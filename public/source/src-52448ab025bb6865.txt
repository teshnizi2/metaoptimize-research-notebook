THIS REFUTES THIS CAMPAIGN'S OWN PRE-REGISTERED ADDENDUM (commit 13b22ed):
"the ceiling is projected to bind anyway in r10/c100, because the top coordinate

THE SECOND RESULT, AND IT IS FREE.  N_eff/m is NON-MONOTONE in the number of

EPS = 1e-3
STEPS_PER_RECORD = 5


# --------------------------------------------------------------- probe reading
def records(d):
    out = []
    with open(os.path.join(d, "probe.jsonl")) as f:
        for line in f:
            out.append(json.loads(line))
    if not out:
        # An empty probe.jsonl has already cost this project a false conclusion
        # once (OPERATIONS: the .out-in-subfolder trap).  Fail loudly, never 0.
        raise ValueError(f"empty probe.jsonl in {d}")
    return out


def occupancy(d, lo, hi, eps=EPS):
    """The FOUR denominators, including the two PATCH_CLIPCOUNT made possible.

    rec_lo / rec_hi   -- fraction of records with any COORDINATE at that guard
                         (from beta_true_min / beta_true_max)
    q4_lo / q4_hi     -- the same, restricted to the last quarter
    coord_lo/coord_hi -- fraction of (record, COORDINATE) cells at that guard
                         (from n_at_lo / n_at_hi / n_beta; None on older probes)
    first_lo/first_hi -- first record index at each guard, or None

    STANDING RULE 5: a registered fraction must name its denominator.
    STANDING RULE 6: it must be measured at the resolution the mechanism
    operates at.  The clamp is applied per coordinate, so `rec_*` and `coord_*`
    are the honest columns and the per-TENSOR mean column is not.
    """
    R = records(d)
    T = len(R)
    q0 = 3 * T // 4
    nlo = nhi = q4lo = q4hi = nq4 = 0
    clo = chi = ctot = 0.0
    first_lo = first_hi = None
    have_counts = "n_at_lo" in R[0]
    for i, r in enumerate(R):
        inq4 = i >= q0
        nq4 += inq4
        if r["beta_true_min"] <= lo + eps:
            nlo += 1
            q4lo += inq4
            if first_lo is None:
                first_lo = i
        if r["beta_true_max"] >= hi - eps:
            nhi += 1
            q4hi += inq4
            if first_hi is None:
                first_hi = i
        if have_counts:
            clo += r["n_at_lo"]
            chi += r["n_at_hi"]
            ctot += r["n_beta"]
    return dict(T=T, rec_lo=nlo / T, rec_hi=nhi / T,
                q4_lo=q4lo / max(nq4, 1), q4_hi=q4hi / max(nq4, 1),
                first_lo=first_lo, first_hi=first_hi,
                coord_lo=(clo / ctot if ctot else None),
                coord_hi=(chi / ctot if ctot else None),
                n_beta=R[0].get("n_beta"),
                min_final=R[-1]["beta_true_min"], max_final=R[-1]["beta_true_max"])


def headroom(d, hi, meta_stepsize=1e-3):
    """How many more records until the TOP coordinate reaches the ceiling.

    Measured on the LAST-QUARTER velocity, never on the startup velocity.  The
    cycle-51 addendum used the startup number and concluded no box-free
    configuration exists; the run it was predicting about is box-free.
    """
    mx = [r["beta_true_max"] for r in records(d)]
    T = len(mx)
    q0 = 3 * T // 4
    v = (mx[-1] - mx[q0]) / max(T - 1 - q0, 1)
    vmax = meta_stepsize * STEPS_PER_RECORD
    head = hi - mx[-1]
    return dict(v_q4=v, frac_vmax=v / vmax, head=head,
                rec_to_hit=(head / v if v > 0 else float("inf")),
                max_final=mx[-1])


def mean(xs):
    return sum(xs) / len(xs) if xs else float("nan")


def agg(pat, lo, hi):
    ds = sorted(glob.glob(pat))
    if not ds:
        return None, 0
    os_ = [occupancy(d, lo, hi) for d in ds]
    keys = ("rec_lo", "rec_hi", "q4_lo", "q4_hi", "coord_lo", "coord_hi",
            "min_final", "max_final")
    out = {}
    for k in keys:
        vs = [o[k] for o in os_ if o[k] is not None]
        out[k] = mean(vs) if vs else None
    out["T"] = os_[0]["T"]
    out["n_beta"] = os_[0]["n_beta"]
    fl = [o["first_lo"] for o in os_ if o["first_lo"] is not None]
    fh = [o["first_hi"] for o in os_ if o["first_hi"] is not None]
    out["first_lo"] = int(mean(fl)) if fl else None
    out["first_hi"] = int(mean(fh)) if fh else None
    return out, len(ds)


# ------------------------------------------------------------------- accuracy
def plateau(prefix, csv_path=None):
    csv_path = csv_path or os.path.join(ROOT, "results", "all_runs.csv")
    v = []
    with open(csv_path) as f:
        for row in csv.DictReader(f):
            if row["run"].startswith(prefix) and row["plateau"]:
                v.append(float(row["plateau"]))
    return mean(v), len(v)


# ------------------------------------------------- N_eff, via the instrument
_ROW = re.compile(r"^\s+(lay|node|w|blk6)\s+([\d,]+)\s+(\d+)\s+\S+\s+\S+\s+"
                  r"([\d,\.]+)\s+([\d,\.]+)\s+([\d,\.]+)\s")


def neff_table(root):
    """-> {family: {rung: (m, N_eff_var, N_eff_var/m)}}, by SHELLING OUT to the
    already-selftested instrument rather than reimplementing it.  A second
    implementation of a number is a second chance to get it wrong."""
    out = subprocess.run([sys.executable, os.path.join(HERE, "neff_instrument.py"), root],
                         capture_output=True, text=True).stdout
    res, fam = {}, None
    for ln in out.splitlines():
        m = re.match(r"=== family (\S+) ===", ln)
        if m:
            fam = m.group(1)
            res[fam] = {}
            continue
        m = _ROW.match(ln)
        if m and fam:
            mm = int(m.group(2).replace(",", ""))
            nv = float(m.group(6).replace(",", ""))
            res[fam][m.group(1)] = (mm, nv, nv / mm)
    return res


_BROW = re.compile(r"^\s+steady \.5-1\s+[\d\.]+x\s+([\-\d\.]+)\s")


def profile_b(root):
    """-> {family: b(steady half)} from probe5_window.py, same reasoning."""
    out = subprocess.run([sys.executable, os.path.join(HERE, "probe5_window.py"), root],
                         capture_output=True, text=True).stdout
    res, fam = {}, None
    for ln in out.splitlines():
        m = re.match(r"=== family (\S+) ===", ln)
        if m:
            fam = m.group(1)
            continue
        m = _BROW.match(ln)
        if m and fam:
            res[fam] = float(m.group(1))
    return res


# ------------------------------------------------------------------- selftest
def selftest():
    import shutil
    import tempfile
    checks = []

    def ok(name, cond):
        checks.append((name, bool(cond)))

    tmp = tempfile.mkdtemp()
    try:
        # A run pinned at the LOW guard for its whole second half, never at HIGH.
        d = os.path.join(tmp, "probe_pin")
        os.makedirs(d)
        with open(os.path.join(d, "probe.jsonl"), "w") as f:
            for i in range(100):
                lo = -30.0 if i >= 50 else -10.0 - 0.1 * i
                f.write(json.dumps(dict(step=5 * i, beta_true_min=lo,
                                        beta_true_max=-5.0,
                                        n_at_lo=(7 if i >= 50 else 0),
                                        n_at_hi=0, n_beta=1000)) + "\n")
        o = occupancy(d, -30.0, 0.0)
        ok("rec_lo counts the pinned half", abs(o["rec_lo"] - 0.50) < 1e-9)
        ok("rec_hi is zero when HI never reached", o["rec_hi"] == 0.0)
        ok("q4_lo is 1.0 when the whole tail is pinned", abs(o["q4_lo"] - 1.0) < 1e-9)
        ok("first_lo is the first pinned record", o["first_lo"] == 50)
        ok("first_hi is None, not 0", o["first_hi"] is None)
        ok("coord_lo uses the per-coordinate denominator",
           abs(o["coord_lo"] - (50 * 7) / (100 * 1000)) < 1e-12)
        ok("n_beta round-trips", o["n_beta"] == 1000)
        ok("T is the record count", o["T"] == 100)

        # A run that never touches either wall -> the box-free case.
        d2 = os.path.join(tmp, "probe_free")
        os.makedirs(d2)
        with open(os.path.join(d2, "probe.jsonl"), "w") as f:
            for i in range(100):
                f.write(json.dumps(dict(step=5 * i, beta_true_min=-16.0,
                                        beta_true_max=-6.0 + 0.02 * i,
                                        n_at_lo=0, n_at_hi=0, n_beta=1000)) + "\n")
        o2 = occupancy(d2, -30.0, 0.0)
        ok("box-free: rec_lo 0", o2["rec_lo"] == 0.0)
        ok("box-free: rec_hi 0", o2["rec_hi"] == 0.0)
        ok("box-free: coord counts 0", o2["coord_lo"] == 0.0 and o2["coord_hi"] == 0.0)
        h = headroom(d2, 0.0)
        # climbs 0.02/record; last value -6.0+0.02*99 = -4.02; head 4.02 -> 201 records
        ok("headroom velocity is the Q4 velocity", abs(h["v_q4"] - 0.02) < 1e-9)
        ok("headroom fraction of vmax", abs(h["frac_vmax"] - 4.0) < 1e-9)
        ok("headroom records-to-hit", abs(h["rec_to_hit"] - 4.02 / 0.02) < 1e-6)

        # A DESCENDING top coordinate must give infinite headroom, not negative.
        d3 = os.path.join(tmp, "probe_down")
        os.makedirs(d3)
        with open(os.path.join(d3, "probe.jsonl"), "w") as f:
            for i in range(100):
                f.write(json.dumps(dict(step=5 * i, beta_true_min=-16.0,
                                        beta_true_max=-4.0 - 0.01 * i,
                                        n_at_lo=0, n_at_hi=0, n_beta=1)) + "\n")
        ok("descending top coordinate -> infinite headroom",
           headroom(d3, 0.0)["rec_to_hit"] == float("inf"))

        # eps must be applied on the correct side of each guard.
        d4 = os.path.join(tmp, "probe_eps")
        os.makedirs(d4)
        with open(os.path.join(d4, "probe.jsonl"), "w") as f:
            f.write(json.dumps(dict(step=0, beta_true_min=-30.0 + 5e-4,
                                    beta_true_max=-5e-4,
                                    n_at_lo=1, n_at_hi=1, n_beta=1)) + "\n")
        o4 = occupancy(d4, -30.0, 0.0)
        ok("within eps of LO counts as bound", o4["rec_lo"] == 1.0)
        ok("within eps of HI counts as bound", o4["rec_hi"] == 1.0)

        # An older probe with no PATCH_CLIPCOUNT fields must give None, not 0.
        d5 = os.path.join(tmp, "probe_old")
        os.makedirs(d5)
        with open(os.path.join(d5, "probe.jsonl"), "w") as f:
            f.write(json.dumps(dict(step=0, beta_true_min=-9.0,
                                    beta_true_max=-5.0)) + "\n")
        o5 = occupancy(d5, -15.0, -2.3026)
        ok("no CLIPCOUNT -> coord columns are None, never 0",
           o5["coord_lo"] is None and o5["coord_hi"] is None)

        # An empty probe must raise, never silently return zeros.
        d6 = os.path.join(tmp, "probe_empty")
        os.makedirs(d6)
        open(os.path.join(d6, "probe.jsonl"), "w").close()
        try:
            records(d6)
            ok("empty probe raises", False)
        except ValueError:
            ok("empty probe raises", True)

        # plateau() must read the CSV, and must skip blank plateau cells.
        cp = os.path.join(tmp, "runs.csv")
        with open(cp, "w") as f:
            f.write("run,plateau\nzz-a-s0,10.0\nzz-a-s1,12.0\nzz-a-s2,\nyy-a-s0,99.0\n")
        m, n = plateau("zz-a-s", cp)
        ok("plateau means only matching, non-blank rows", abs(m - 11.0) < 1e-9 and n == 2)
        ok("plateau does not match a different prefix", plateau("yy-a-s", cp)[1] == 1)

        # agg() must return (None, 0) for a pattern that matches nothing, so a
        # missing arm reads as MISSING and never as a passing zero.
        a, k = agg(os.path.join(tmp, "no_such_*"), -30.0, 0.0)
        ok("agg on no dirs -> (None, 0)", a is None and k == 0)

        # The row regex must pick the N_eff VARIANCE column (the 3rd of three),
        # not N_eff raw or N_eff debias.  This is the one quotable column.
        ln = ("  w       11,173,962  3   1.63e-05       0.004  "
              "5,598,482.0   5,643,111.2   5,636,970.2  yes")
        m6 = _ROW.match(ln)
        ok("row regex matches an instrument row", m6 is not None)
        ok("row regex takes the VARIANCE column",
           m6 and abs(float(m6.group(6).replace(",", "")) - 5636970.2) < 1e-6)
        ok("row regex takes m from column 2",
           m6 and int(m6.group(2).replace(",", "")) == 11173962)
        b6 = _BROW.match("  steady .5-1         3.8x      0.071  k=1: 1.370e-07")
        ok("b regex reads the steady-half exponent",
           b6 and abs(float(b6.group(1)) - 0.071) < 1e-9)
        b7 = _BROW.match("  steady .5-1         2.1x     -0.012  k=1: 6.0e-08")
        ok("b regex accepts a negative exponent",
           b7 and abs(float(b7.group(1)) + 0.012) < 1e-9)
    finally:
        shutil.rmtree(tmp)

    bad = [n for n, c in checks if not c]
    for n in bad:
        print(f"  FAIL {n}")
    print(f"selftest: {len(checks) - len(bad)}/{len(checks)} "
          f"{'PASS' if not bad else 'FAIL'}")
    return 1 if bad else 0

    return "PASS" if lo <= v <= hi else "FAIL"
