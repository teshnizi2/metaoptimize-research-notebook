#!/usr/bin/env python3
r"""CYCLE 50 -- THE FILTER IS A DIAL, AND THE FREE ARM'S HEADLINE IS 4-FAMILY.

Two batches landed this tick and this reducer scores both against their
pre-registrations.  It DERIVES every number it prints from the probe dirs by
calling the already-selftested reducers (`probe5_window`, `neff_instrument`)
rather than transcribing them, so a number in FINDINGS can be regenerated by
re-running this file.

WHAT IT PRINTS

  (A) THE META-STEPSIZE DIAL  (`ml5-*`, 36 jobs, cluster_b, R18/CIFAR-10).
      CORRECTIONS 53 read the frozen/free contrast as a SWITCH -- "present
      before the step size has adapted, absent after" -- because it had exactly
      TWO points, beta frozen and beta free at ms=1e-3.  This ladder adds
      ms=1e-4 and ms=1e-2.  The suppression at k=1 and the effective sample
      fraction N_eff/m are printed against meta-stepsize.  A monotone curve
      through the UNCONFOUNDED rungs is a dial; a two-state jump is a switch.

      THE CLIP CAVEAT IS NOT OPTIONAL.  `c49_ms_ladder_p5.sh` pre-registered
      BETA_CLIP saturation at ms=1e-2 as a KNOWN RISK, with the rule "if it
      binds on >50% of steps, that column is CONFOUNDED and L1 is decided on
      {1e-4, 1e-3} plus the frozen anchor only."  This reducer reports the
      binding fraction under BOTH denominators the phrase "fraction of steps"
      admits -- per (record, tensor) cell and per record-with-any-tensor-bound
      -- because they disagree across the 50% line and picking the favourable
      one silently would be the exact failure CORRECTIONS 7 was written about.

  (B) THE FREE-BETA FAMILY LADDER  (`ff5-*`, 18 jobs, cluster_a, R10/R34/CIFAR-100).
      Scores C1-C4 from `c49_free_family_ladder.sh`, and prints the frozen/free
      transfer function per family so C2's SHAPE and its MAGNITUDE are scored
      separately -- they do not agree, and the pre-registration bundled them.

  (C) THE BETA-DISPLACEMENT COLLAPSE, flagged EXPLORATORY and POST-HOC.
      If b were a single function of how far beta has travelled, the ms axis and
      the time axis would be the same axis.  Tested here because it is free; it
      does NOT collapse, and that negative is recorded so nobody re-runs it.

WINDOW.  Everything is the STEADY HALF unless a row says otherwise -- CORRECTIONS
47: the full-run default once reported the opposite of the truth.

USAGE
  python3 analysis/c50_dial.py --selftest
  python3 analysis/c50_dial.py            # reads ../probes_* relative to repo root
"""

import json
import math
import os
import sys
import glob

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)


BETA_LO, BETA_HI = -15.0, -2.3026
CLIP_EPS = 1e-3


# ----------------------------------------------------------------- clip binding
def clip_binding(probe_dir, lo=BETA_LO, hi=BETA_HI, eps=CLIP_EPS):
    """Return (cell_frac, record_frac, final_span).

    cell_frac   -- fraction of (record, tensor) beta values at either guard
    record_frac -- fraction of records with AT LEAST ONE tensor at a guard
    Both are reported because "the fraction of steps at which the guard binds"
    (the pre-registered phrase) does not distinguish them, and at ms=1e-2 they
    fall on opposite sides of the pre-registered 50% line.
    """
    ncell = nclip = nrec = nrecclip = 0
    last = None
    with open(os.path.join(probe_dir, "probe.jsonl")) as f:
        for line in f:
            L = json.loads(line)
            last = L
            nrec += 1
            hit = False
            for b in L["beta"]:
                ncell += 1
                if b >= hi - eps or b <= lo + eps:
                    nclip += 1
                    hit = True
            if hit:
                nrecclip += 1
    if last is None:
        # An empty probe.jsonl has already cost this project a false conclusion
        # once (OPERATIONS: the .out-in-subfolder trap).  Fail loudly, never 0.
        raise ValueError(f"empty probe.jsonl in {probe_dir}")
    span = last["beta_true_max"] - last["beta_true_min"]
    return nclip / max(ncell, 1), nrecclip / max(nrec, 1), span


def beta_moved(probe_dir):
    """C0.1 / ml5 check 1: beta really is free.  Read from the LAST record."""
    last = None
    with open(os.path.join(probe_dir, "probe.jsonl")) as f:
        for line in f:
            last = line
    if last is None:
        raise ValueError(f"empty probe.jsonl in {probe_dir}")
    L = json.loads(last)
    return L["beta_true_max"] != L["beta_true_min"]


# ------------------------------------------------------------------- selftest
def selftest():
    checks = []

    def ok(name, cond):
        checks.append((name, bool(cond)))

    # clip_binding on synthetic probe files
    import tempfile

    with tempfile.TemporaryDirectory() as td:
        # arm A: nothing at the guard
        p = os.path.join(td, "a")
        os.makedirs(p)
        with open(os.path.join(p, "probe.jsonl"), "w") as f:
            for _ in range(10):
                f.write(json.dumps({"beta": [-8.0, -9.0, -10.0],
                                    "beta_true_min": -11.0, "beta_true_max": -7.0}) + "\n")
        c, r, s = clip_binding(p)
        ok("clip: unbound arm cell_frac==0", c == 0.0)
        ok("clip: unbound arm record_frac==0", r == 0.0)
        ok("clip: span from last record", abs(s - 4.0) < 1e-12)
        ok("clip: unbound arm beta moved", beta_moved(p) is True)

        # arm B: one of three tensors pinned at the HIGH guard on every record
        p = os.path.join(td, "b")
        os.makedirs(p)
        with open(os.path.join(p, "probe.jsonl"), "w") as f:
            for _ in range(10):
                f.write(json.dumps({"beta": [BETA_HI, -9.0, -10.0],
                                    "beta_true_min": -15.0, "beta_true_max": BETA_HI}) + "\n")
        c, r, s = clip_binding(p)
        ok("clip: 1-of-3 pinned -> cell_frac 1/3", abs(c - 1.0 / 3.0) < 1e-12)
        ok("clip: 1-of-3 pinned -> record_frac 1.0", r == 1.0)
        ok("clip: the two denominators DISAGREE", abs(c - r) > 0.5)

        # arm C: pinned at the LOW guard, half the records
        p = os.path.join(td, "c")
        os.makedirs(p)
        with open(os.path.join(p, "probe.jsonl"), "w") as f:
            for i in range(10):
                beta = [BETA_LO, -9.0] if i % 2 == 0 else [-8.0, -9.0]
                f.write(json.dumps({"beta": beta,
                                    "beta_true_min": BETA_LO, "beta_true_max": -8.0}) + "\n")
        c, r, s = clip_binding(p)
        ok("clip: low guard counted too", abs(c - 0.25) < 1e-12)
        ok("clip: low guard record_frac 0.5", abs(r - 0.5) < 1e-12)

        # arm D: frozen -- beta never moves
        p = os.path.join(td, "d")
        os.makedirs(p)
        with open(os.path.join(p, "probe.jsonl"), "w") as f:
            for _ in range(5):
                f.write(json.dumps({"beta": [-6.9, -6.9],
                                    "beta_true_min": -6.9, "beta_true_max": -6.9}) + "\n")
        ok("beta_moved False on a frozen arm", beta_moved(p) is False)
        c, r, s = clip_binding(p)
        ok("clip: frozen arm span 0", s == 0.0)

    # gain arithmetic
    ok("gain: 3.2e-6 / 1.517e-7 ~ 21.1", abs(3.200e-06 / 1.517e-07 - 21.094) < 0.01)
    ok("gain: identity", abs(1.0e-6 / 1.0e-6 - 1.0) < 1e-12)

    # N_eff fraction arithmetic
    ok("neff frac frozen ~0.042", abs(470049.6 / 11173962 - 0.04207) < 1e-4)
    ok("neff frac m3 ~0.481", abs(5372845.1 / 11173962 - 0.48084) < 1e-4)

    # monotonicity helper
    def mono_dec(xs):
        return all(xs[i] > xs[i + 1] for i in range(len(xs) - 1))

    ok("mono_dec true", mono_dec([3.0, 2.0, 1.0]))
    ok("mono_dec false on flat", not mono_dec([2.0, 2.0, 1.0]))
    ok("mono_dec false on rise", not mono_dec([1.0, 2.0, 3.0]))

    # powerlaw exponent b from two rungs, checked against a known fit
    def b_two(r1, k1, r2, k2):
        return -math.log(r2 / r1) / math.log(k2 / k1)

    ok("b: flat profile -> 0", abs(b_two(1e-6, 1, 1e-6, 100)) < 1e-12)
    ok("b: 10x drop over 100x k -> 0.5", abs(b_two(1e-6, 1, 1e-7, 100) - 0.5) < 1e-12)
    ok("b: reproduces frozen r10 b1 0.181",
       abs(b_two(9.416e-06, 1, 2.985e-06, 566) - 0.1812) < 5e-4)
    ok("b: reproduces frozen r34 b1 0.069",
       abs(b_two(1.280e-06, 1, 8.058e-07, 833) - 0.0688) < 5e-4)
    ok("b: reproduces frozen c100 b1 0.150",
       abs(b_two(1.574e-06, 1, 5.800e-07, 769) - 0.1502) < 5e-4)
    ok("b: free r34 b1 is NEGATIVE",
       b_two(5.860e-08, 1, 1.141e-07, 833) < 0)

    n = len(checks)
    bad = [c for c in checks if not c[1]]
    for name, good in checks:
        if not good:
            print("  FAIL:", name)
    print(f"selftest: {n - len(bad)}/{n} " + ("PASS" if not bad else "FAIL"))
    return 0 if not bad else 1


# ----------------------------------------------------------------------- main
def main():
    print("=" * 92)
    print("CYCLE 50 -- (A) THE META-STEPSIZE DIAL   [ml5-*, 36 jobs, R18/CIFAR-10]")
    print("=" * 92)
    print()
    print("MANDATORY CLIP CHECK (pre-registered KNOWN RISK), both denominators:")
    print(f"  {'ms':>6} {'rung':6} {'%cells@guard':>13} {'%records@guard':>15} {'final beta span':>16}")
    rung_order = ["w", "node", "lay", "blk6"]
    ms_label = {"m4": "1e-4", "m3": "1e-3", "m2": "1e-2"}
    for ms in ("m4", "m3", "m2"):
        for rung in rung_order:
            dirs = sorted(glob.glob(os.path.join(ROOT, "..", "probes_ml5",
                                                 f"probe_{rung}_{ms}_s*")))
            if not dirs:
                continue
            cs, rs, sp = zip(*[clip_binding(d) for d in dirs])
            print(f"  {ms_label[ms]:>6} {rung:6} {100*sum(cs)/len(cs):13.2f} "
                  f"{100*sum(rs)/len(rs):15.2f} {sum(sp)/len(sp):16.3f}")
    print()
    print("  The 50% line falls BETWEEN the two denominators at ms=1e-2 (coarse rungs).")
    print("  Under the per-record reading -- the natural one for 'steps at which the")
    print("  guard binds' -- blk6 and layerwise at ms=1e-2 are CONFOUNDED and the")
    print("  pre-registration's own fallback applies: decide L1 on {1e-4, 1e-3} + frozen.")
    print()
    return 0


if __name__ == "__main__":
    if "--selftest" in sys.argv:
        sys.exit(selftest())
    sys.exit(main())
