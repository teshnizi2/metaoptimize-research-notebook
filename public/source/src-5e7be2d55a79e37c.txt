set -e
cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
# CYCLE 9 addendum to e3a_budget.sh. At 100 epochs, alpha0=1e-6, layerwise:
#   additive r=0.05  93.41 +-0.06 (n=2)  train 99.55
#   shrink   lam=0.1 92.58 +-0.10 (n=12) train 97.78   <- still climbing (gotcha 19)
# additive's best cell beats shrink's best by +0.83pp AND fits better, so additive looks
# like a Pareto improvement. But shrink's 92.58 is measured while it is STILL FITTING, so
# that comparison is budget-bounded exactly the way gotcha 19 warns about. The e3a block
# as first submitted has no shrink cell, so it cannot settle it. These two jobs add one.
# Identical config to e3a (alpha0=1e-6, 300 ep) so it is the matched contrast, not a new
# experiment. nice=1000: behind the seed-0 additive sweep, ahead of the seed-1 backfill.
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 300 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
for s in 0 1; do
  sbatch --parsable --job-name=e3a-lsh01-s$s --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 \
    --time=03:00:00 --nice=1000 --export="ALL,AUGMENT=1,$C,HIER=shrink,LAM=0.1" \
    jobs/run_cifar.sh $B --stepsize-groups layerwise --seed $s \
    --save-directory $R/e3a --run-name e3a-lsh01_s$s >/dev/null
done
sleep 4
echo "E3A_TOTAL=$(squeue -h -u account_a -o '%j' | grep -c '^e3a-')"
squeue -h -u account_a -o '%i %j %y %T' | grep '^.* e3a-' | sort -k3,3n
