#!/usr/bin/env python3
# =============================================================================
# cVH1_smoke_gen.py -- SYNTHETIC, HARNESS-FORMAT `cvh1` RUNS FOR THE SELFTEST
#   OF analysis/cVH1_vgghorizon_score.py.
#
# EVERY ACCURACY AND EVERY BETA WRITTEN BY THIS FILE IS INVENTED.  NOTHING IT
# WRITES IS EVIDENCE ABOUT THE REAL BATCH.  It exists for ONE reason:
# CORRECTIONS 171 (cdn1 crashed unconditionally at scoring while its selftest
# passed 34/34) -- a selftest must drive the REAL score() on runs that look
# exactly like the harness's own output, to a FINAL line, including a dropped
# run (which must give INCOMPLETE).
#
# WHAT IT WRITES under <outdir> (which MUST NOT be a real runs directory):
#   <outdir>/cvh1-<arm>-s<seed>-<jobid>.out    NODE / ARGS / ENV / PROBE_TENSOR
#                                              lines in the live runner's format
#                                              (copied from cvi1's own .out
#                                              files), 328 `Epoch` lines,
#                                              `NN  minutes`, RUN_DONE
#   <outdir>/cvh1/probe_<run>/probe.jsonl      1,640 records (PROBE=100 x 500
#                                              meta-steps/epoch x 328 epochs)
#   <outdir>/cvh1/probe_<run>/block_sizes.json the harness's n_b
#   <outdir>/cvh1/probe_<run>/probe_tensor.json
#   <outdir>/cvh1/PARTITION-MANIFEST.txt       the launcher's guard-4 format
#
# THE ISO COMPLEMENT (beta[0]) HAS THREE REGISTERED SHAPES, because the gate
# this batch replaces failed on exactly one of them:
#   chatter  descends, first touches -15.0 at epoch 112.0, then CHATTERS: 38 %
#            of the remaining records sit EXACTLY on the clamp and the other
#            62 % sit 0.001-0.050 nats above it.  Exact-clamp dwell 0.38 < 0.50
#            -- the ciso2 PIN_OCC_MIN trigger (CORRECTIONS 196.5) -- while the
#            step size never exceeds exp(0.05) = 1.051x the floor.
#   pinned   the same descent, then exactly on the clamp forever.
#   free     descends to -13.4 by epoch 100 and then settles at -11.6
#            (exp(3.4) = 30x the floor) for the rest of the run.
#
# usage: python3 analysis/cVH1_smoke_gen.py <outdir> <scenario>
#        python3 analysis/cVH1_smoke_gen.py --list
# =============================================================================

PREFIX = "cvh1-"
BATCH = "cvh1"
SEEDS = (63, 64, 65)
ARMS = ("k01", "kL", "ISO", "CTL")
EPOCHS = 328
PROBE = 100
REC_PER_EPOCH = 5
N_RECORDS = EPOCHS * REC_PER_EPOCH           # 1,640
NET, DSET = "VGG11_bn_c100", "CIFAR100"
CLIP = "-15:-2.3026"
BETA_LO = -15.0
B_INIT = -13.815510557964274                 # ln(alpha0 = 1e-6)
SAVE = "/cluster/account_b/metaopt/runs/cvh1"    # only ever printed into the synthetic lines

SPEC = {
    "k01": "scalar",
    "kL": "layerwise",
    "ISO": "sets:1-22,24-26/bn8.weight",
    "CTL": "sets:1-19,21-26/bn7.weight",
}
PT_TYPE = {"k01": "scalar", "kL": "layerwise", "ISO": "blockwise", "CTL": "blockwise"}
TENSOR_NAMES = (
    "conv1.weight", "bn1.weight", "bn1.bias", "conv2.weight", "bn2.weight", "bn2.bias",
    "conv3.weight", "bn3.weight", "bn3.bias", "conv4.weight", "bn4.weight", "bn4.bias",
    "conv5.weight", "bn5.weight", "bn5.bias", "conv6.weight", "bn6.weight", "bn6.bias",
    "conv7.weight", "bn7.weight", "bn7.bias", "conv8.weight", "bn8.weight", "bn8.bias",
    "linear.weight", "linear.bias",
)
TENSOR_NUMELS = (
    1728, 64, 64, 73728, 128, 128, 294912, 256, 256, 589824, 256, 256, 1179648,
    512, 512, 2359296, 512, 512, 2359296, 512, 512, 2359296, 512, 512, 51200, 100,
)
GROUPS = {
    "ISO": ([i for i in range(1, 27) if i != 23], [23]),
    "CTL": ([i for i in range(1, 27) if i != 20], [20]),
}
# cvi1's own layerwise terminal pinned set (CORRECTIONS 203.7), reused as shape only
KL_PINNED = (2, 5, 8, 9, 11, 14, 17, 20, 21, 23, 24, 25)
SEED_OFF = {63: -0.20, 64: 0.00, 65: 0.20}

# arm -> (TEST@100, TEST@E, TRAIN@100, TRAIN@E): the value every epoch of the
# plateau window carries, before the per-seed offset.  INVENTED.
BASE_LEVELS = {
    "k01": (35.57, 35.60, 35.56, 35.60),
    "kL": (66.30, 67.40, 99.34, 99.80),
    "ISO": (66.85, 66.80, 95.97, 96.30),
    "CTL": (35.55, 35.58, 35.49, 35.52),
}

SCENARIOS = {
    # the registered SURVIVE account, with the ciso2-style chatter at the clamp
    "survive_chatter": dict(iso_b0="chatter"),
    "survive_pinned": dict(iso_b0="pinned"),
    # the complement never reaches the floor: the primary still lands; the pin stamp says so
    "survive_free": dict(iso_b0="free", levels={"ISO": (66.85, 67.60, 95.97, 98.10)}),
    "collapse": dict(iso_b0="chatter", levels={"ISO": (66.85, 38.00, 95.97, 40.00)}),
    "attenuate": dict(iso_b0="chatter", levels={"ISO": (66.85, 54.40, 95.97, 70.00)}),
    "no_rescue": dict(iso_b0="chatter", levels={"ISO": (40.00, 40.00, 42.00, 42.00)}),
    # ISO - kL at E = +1.60 pp: just OVER the frozen READ_BAR (1.5114).  A scorer
    # whose bar read a live corpus sigma > 0.98 would stamp CEIL-TRACKS instead.
    "near_bar": dict(iso_b0="chatter", levels={"kL": (66.30, 65.20, 99.34, 99.80)}),
    "anchor": dict(iso_b0="chatter", levels={"kL": (55.00, 56.00, 90.00, 91.00)}),
    "diverged": dict(iso_b0="chatter", seed_levels={("ISO", 65): (74.85, 74.80, 95.97, 96.30)}),
    "dropped": dict(iso_b0="chatter", mutate="drop"),
    "truncated": dict(iso_b0="chatter", mutate="truncate"),
    "badspec": dict(iso_b0="chatter", mutate="badspec"),
    "nomanifest": dict(iso_b0="chatter", mutate="nomanifest"),
}


def _pl(points, ep):
    """piecewise-linear interpolation over (epoch, beta) knots"""
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        if ep <= x1:
            return y1 if x1 == x0 else y0 + (y1 - y0) * (ep - x0) / (x1 - x0)
    return points[-1][1]


K01_B = [(0, B_INIT), (16.8, -5.48), (36.0, -15.0), (1e9, -15.0)]
CTL_B1 = [(0, B_INIT), (13.6, -7.10), (29.6, -15.0), (1e9, -15.0)]
ISO_B1 = [(0, B_INIT), (8.7, -9.47), (20.0, -15.0), (1e9, -15.0)]
ISO_B0_DESC = [(0, B_INIT), (17.3, -5.20), (100.0, -13.40), (112.0, -15.0)]
ISO_B0_FREE = [(0, B_INIT), (17.3, -5.20), (100.0, -13.40), (110.0, -11.60), (1e9, -11.60)]
KL_FREE = [(0, B_INIT), (20.0, -8.0), (1e9, -9.3)]
KL_PIN = [(0, B_INIT), (10.0, -9.0), (30.0, -15.0), (1e9, -15.0)]
TOUCH_REC = int(112.0 * REC_PER_EPOCH)      # 560


def iso_b0(mode, i):
    ep = i / float(REC_PER_EPOCH)
    if mode == "free":
        return _pl(ISO_B0_FREE, ep)
    if i < TOUCH_REC:
        return _pl(ISO_B0_DESC, ep)
    if mode == "pinned":
        return BETA_LO
    k = i - TOUCH_REC
    if (k * 7919) % 100 < 38:
        return BETA_LO
    return BETA_LO + 0.001 + 0.049 * (((k * 104729) % 1000) / 1000.0)


def betas(arm, i, mode):
    ep = i / float(REC_PER_EPOCH)
    if arm == "k01":
        return [_pl(K01_B, ep)]
    if arm == "kL":
        return [_pl(KL_PIN if (j + 1) in KL_PINNED else KL_FREE, ep) for j in range(26)]
    if arm == "ISO":
        return [iso_b0(mode, i), _pl(ISO_B1, ep)]
    if arm == "CTL":
        return [_pl(K01_B, ep), _pl(CTL_B1, ep)]
    raise ValueError(arm)


def acc_series(lv, off):
    """(train, test) per epoch 0..EPOCHS-1.  Flat at the @100 level over epochs
    60..114, then a 20-epoch linear move to the @E level, flat from 135 on."""
    p100, pe, t100, te = lv
    out = []
    for e in range(EPOCHS):
        if e < 60:
            f = (e + 1) / 60.0
            ts, tr = 1.0 + (p100 - 1.0) * f, 1.0 + (t100 - 1.0) * f
        elif e < 115:
            ts, tr = p100, t100
        elif e < 135:
            f = (e - 115 + 1) / 20.0
            ts, tr = p100 + (pe - p100) * f, t100 + (te - t100) * f
        else:
            ts, tr = pe, te
        out.append((min(100.0, tr + off), min(100.0, ts + off)))
    return out


def args_line(arm, seed, rn, epochs=EPOCHS, spec=None):
    return ("ARGS: --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 "
            "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 "
            "--dataset %s --NN-name %s --batch-size 100 --max-time 999:00:00 --gamma 1 "
            "--meta-stepsize 1e-3 --alpha0 1e-6 --num-epochs %d --stepsize-groups %s --seed %d "
            "--save-directory %s --run-name %s" % (DSET, NET, epochs, spec or SPEC[arm], seed, SAVE, rn))


def env_line(rn):
    return ("ENV: AUGMENT=1 BETA_CLIP=%s HIER=none LAM=na ETA_RATIO=na COS_TOTAL=default "
            "COS_WARMUP=default SCHED=none SCHED_TOTAL=none SCHED_WARMUP=none SCHED_MIN=none "
            "PROBE=%d PROBE_DIR=%s/probe_%s EB_RHO=na EB_LOG=0" % (CLIP, PROBE, SAVE, rn))


def write_out(path, arm, seed, rn, jid, series, done=True, spec=None, n_epochs=EPOCHS):
    with open(path, "w") as fh:
        fh.write("NODE=[HOST_af5f3a3791] | JOB=%s %d | AUGMENT=1\n" % (rn, jid))
        fh.write(args_line(arm, seed, rn, spec=spec) + "\n")
        fh.write(env_line(rn) + "\n")
        fh.write("NVIDIA A100 80GB PCIe, 81920 MiB\n\n")
        fh.write("Files already downloaded and verified\nFiles already downloaded and verified\n")
        fh.write("PROBE_TENSOR: on every=%d type=%s tensors=26 meta_alg=Lion momentum_param=0.99 "
                 "Lion_beta2=0.9 dir=%s/probe_%s\n" % (PROBE, PT_TYPE[arm], SAVE, rn))
        for e in range(n_epochs):
            tr, ts = series[e]
            fh.write("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%\n" % (e, tr, ts))
        if done:
            fh.write("91  minutes\nRUN_DONE\n")


def write_probe(pdir, arm, mode):
    os.makedirs(pdir, exist_ok=True)
    if arm == "k01":
        nb = [sum(TENSOR_NUMELS)]
    elif arm == "kL":
        nb = list(TENSOR_NUMELS)
    else:
        g0, g1 = GROUPS[arm]
        nb = [sum(TENSOR_NUMELS[j - 1] for j in g0), sum(TENSOR_NUMELS[j - 1] for j in g1)]
    json.dump({"stepsize_type": PT_TYPE[arm], "n_b": nb}, open(os.path.join(pdir, "block_sizes.json"), "w"))
    json.dump({"PROBE_TENSOR": 1, "every": PROBE, "stepsize_type": PT_TYPE[arm], "n_tensors": 26,
               "param_numels": list(TENSOR_NUMELS), "len_beta_list": 1, "meta_alg": "Lion",
               "momentum_param": 0.99, "Lion_beta2": 0.9},
              open(os.path.join(pdir, "probe_tensor.json"), "w"))
    with open(os.path.join(pdir, "probe.jsonl"), "w") as fh:
        for i in range(N_RECORDS):
            b = betas(arm, i, mode)
            rec = {"step": i * PROBE, "beta": b, "n_beta": len(b),
                   "n_at_lo": sum(1 for v in b if v <= BETA_LO + 1e-9), "pt_mp": 0.99, "pt_b2": 0.9}
            if arm in GROUPS:
                zt = [((i * 31 + j * 17) % 9 - 4) * 0.125 for j in range(26)]
                mt = [((i * 13 + j * 7) % 7 - 3) * 0.25 for j in range(26)]
                g0, g1 = GROUPS[arm]
                rec["z_tensor"], rec["m_tensor"] = zt, mt
                rec["z_agg"] = [[sum(zt[j - 1] for j in g0), sum(zt[j - 1] for j in g1)]]
                rec["mom_pre"] = [[sum(mt[j - 1] for j in g0), sum(mt[j - 1] for j in g1)]]
            fh.write(json.dumps(rec) + "\n")


def write_manifest(path):
    names = list(TENSOR_NAMES)
    L = ["NUM_PARAM_TENSORS 26", "TOTAL_PARAMS %d" % sum(TENSOR_NUMELS), "NETWORK %s" % NET,
         "CLIP_C %s" % CLIP, "EPOCHS %d" % EPOCHS, "META_STEPS %d" % (EPOCHS * 500)]
    for i in range(26):
        L.append("TENSOR %d %s %d" % (i + 1, names[i], TENSOR_NUMELS[i]))
    L.append("ARMSPEC k01 SPEC scalar M 1 SIZES 26 PARAMS %d GROUP0 %s GROUP1 -"
             % (sum(TENSOR_NUMELS), " ".join(names)))
    L.append("ARMSPEC kL SPEC layerwise M 26 SIZES %s PARAMS %s GROUP0 %s GROUP1 -"
             % (",".join("1" for _ in names), ",".join(str(x) for x in TENSOR_NUMELS), " ".join(names)))
    for arm in ("ISO", "CTL"):
        g0, g1 = GROUPS[arm]
        L.append("ARMSPEC %s SPEC %s M 2 SIZES %d,%d PARAMS %d,%d GROUP0 %s GROUP1 %s"
                 % (arm, SPEC[arm], len(g0), len(g1),
                    sum(TENSOR_NUMELS[j - 1] for j in g0), sum(TENSOR_NUMELS[j - 1] for j in g1),
                    " ".join(names[j - 1] for j in g0), " ".join(names[j - 1] for j in g1)))
        L.append("ISOLATED %s %s" % (arm, ",".join(str(j) for j in g1)))
    L.append("SYMDIFF ISO CTL bn7.weight,bn8.weight")
    open(path, "w").write("\n".join(L) + "\n")


def build(outdir, scenario):
    """write one synthetic batch for `scenario` into outdir (created fresh)."""
    if scenario not in SCENARIOS:
        raise ValueError("unknown scenario %r; known: %s" % (scenario, sorted(SCENARIOS)))
    sc = SCENARIOS[scenario]
    if os.path.exists(outdir):
        shutil.rmtree(outdir)
    os.makedirs(os.path.join(outdir, BATCH))
    levels = dict(BASE_LEVELS)
    levels.update(sc.get("levels", {}))
    mode = sc.get("iso_b0", "chatter")
    jid = 7100000
    for s in SEEDS:
        for arm in ARMS:
            jid += 1
            rn = "%s%s-s%d" % (PREFIX, arm, s)
            lv = sc.get("seed_levels", {}).get((arm, s), levels[arm])
            series = acc_series(lv, SEED_OFF[s])
            done, n_ep, spec = True, EPOCHS, None
            if sc.get("mutate") == "truncate" and (arm, s) == ("ISO", 65):
                done, n_ep = False, 200
            if sc.get("mutate") == "badspec" and (arm, s) == ("CTL", 63):
                spec = SPEC["ISO"]
            if sc.get("mutate") == "drop" and (arm, s) == ("CTL", 64):
                pass
            else:
                write_out(os.path.join(outdir, "%s-%d.out" % (rn, jid)), arm, s, rn, jid, series,
                          done=done, spec=spec, n_epochs=n_ep)
            write_probe(os.path.join(outdir, BATCH, "probe_" + rn), arm, mode)
    if sc.get("mutate") != "nomanifest":
        write_manifest(os.path.join(outdir, BATCH, "PARTITION-MANIFEST.txt"))
    return outdir


def main(argv):
    if len(argv) == 2 and argv[1] == "--list":
        for k in sorted(SCENARIOS):
            print(k)
        return 0
    if len(argv) != 3:
        print(__doc__ or "usage: cVH1_smoke_gen.py <outdir> <scenario> | --list")
        return 2
    build(argv[1], argv[2])
    print("wrote synthetic cvh1 scenario %r to %s (EVERY NUMBER INVENTED)" % (argv[2], argv[1]))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
