#!/bin/bash
# =============================================================================
# c79_argmax_robustness.sh -- `ar1`, IS THE MATCHED-COUNT PARTITION GAP AN
# ARGMAX ARTEFACT?  AND DOES THE FIELD PREDICT ACCURACY -- PRE-REGISTERED?
# Runs on cluster_a ONLY (needs PATCH_CHUNKWISE and PATCH_NODEBN).
# 12 jobs: {nodewise, chunk777, nodewise1d, chunk2325} x seeds 0-2, ms=3e-4,
# 100 ep, probe ON.  NO NEW PATCH -- every granularity here already ships and is
# already covered by an equivalence suite.
#
# WHY THIS BATCH EXISTS -- REASON 1, THE ONE THAT COULD BOUND EVERYTHING
#   The ENTIRE matched-count partition programme was run at ms=1e-4:
#     mm1  D = chunk777 - nodewise   +0.485  t 3.01   (FINDINGS 76.x)
#     pp1  D                          +0.581  t 4.11   (FINDINGS 77.2)
#     bn1  G = chunk2325 - nodewise1d +0.295  t 6.15   (FINDINGS 78.1, UNDECIDED)
#
#   Re-read from the CSV this cycle, at the SAME 13-axis signature (clip
#   -15:-2.3026, augment 1, alpha0 1e-3, ResNet18, CIFAR10, 100 ep):
#
#     nodewise @ 1e-4   92.073 +-0.044  (n=13)
#     nodewise @ 3e-4   92.493 +-0.092  (n= 8)   **+0.420 pp, t ~ 4.1**
#
#   **ms=1e-4 IS NOT nodewise's ARGMAX.  It is 0.42 pp below it -- comparable to
#   the whole gap the campaign has been explaining.**  chunk777 has NEVER been run
#   at any stepsize other than 1e-4, so its own ms curve is entirely unmeasured.
#   If chunk's optimum sits at or near 1e-4 while nodewise's sits at 3e-4, then a
#   material part of D is "we compared two partitions at one partition's favourable
#   stepsize", and every D in the campaign is bounded to ms=1e-4.
#   bn1's own registered limits already said "ms is held at 1e-4 and nodewise1d's
#   argmax has never been measured".  This batch stops deferring that.
#
# WHY THIS BATCH EXISTS -- REASON 2, PRE-REGISTERING WHAT 78.6 FOUND POST-HOC
#   `analysis/c79_field_vs_accuracy.py` assembled every matched-count pair the
#   campaign owns and found the sign-agreement field NEVER once predicts accuracy:
#     mm1 chunk777  - nodewise    dacc +0.485 (t  3.01)  dN_eff/m -0.0180 (t -10.62)
#     pp1 permnode  - nodewise    dacc -0.009 (t -0.06)  dN_eff/m -0.0133 (t -13.12)
#     bn1 chunk2325 - nodewise1d  dacc +0.295 (t  6.15)  dN_eff/m -0.0485 (t -21.90)
#   CONCORDANT 0, ANTI-CONCORDANT 2, accuracy-unresolved 1.  **All three POST-HOC.**
#   That is the sharpest evidence the campaign has against the brief's default
#   direction C -- the programme of measuring sign agreement across granularities --
#   and it cannot be quoted, because nothing about it was registered.
#   A3 below registers it: same rule, same sign convention, fixed BEFORE the data
#   exist, at a stepsize where neither the accuracy nor the field has been read.
#
# THE FOUR ARMS, ONE BATCH, ms=3e-4, seeds 0-2.  TWO MATCHED-COUNT PAIRS:
#     node  nodewise    m=14,420  |  ch   chunk777   m=14,421   (1 group apart)
#     n1d   nodewise1d  m= 4,851  |  c23  chunk2325  m= 4,851   (EXACT)
#
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
#
#   A0    VALIDITY.  n_records == 10000; beta moved; epochs_done == requested == 100.
#   A0.2  n_beta EXACT on EVERY record: node 14420, ch 14421, n1d 4851, c23 4851.
#         Guard 4 MEASURES all four from the ALLOCATED beta on the real built
#         network before anything is submitted.
#   A0.3  THE INSTRUMENT FIRED.  neg_counts.json, n_tot == n_beta, npy shape read
#         from its HEADER == (n_tot,).  (Header, never file size -- that inference
#         produced c74's false VOID on 12/12 arms.)
#   A0.4  BOX-FREE at BOTH guards, the published rec_-based 5% gate, PRIMARY and
#         UNCHANGED.  **ms is 3x larger here than in any probed batch, so a bind is
#         a REAL possibility and is registered as such, not treated as a surprise.**
#         A bind VOIDS A3 (a box-bound arm's N_eff/m is not interpretable) and
#         leaves A1 and A2 standing (they are accuracy-only).
#
#   A1    **THE PRIMARY.  FIVE-WAY, SYMMETRIC, WITHIN-BATCH, COUNT MATCHED.**
#         D' = plateau5(chunk777) - plateau5(nodewise) at ms=3e-4.
#           D' >  +0.30      -> **THE GAP SURVIVES THE STEPSIZE MOVE.**  mm1/pp1's D
#                is not an argmax artefact and the partition result is robust across
#                a 3x change in ms.  This is the branch that STRENGTHENS the campaign.
#           (+0.15, +0.30]   -> **UNDECIDED**, registered in advance.
#           [-0.15, +0.15]   -> **THE GAP COLLAPSES AT ms=3e-4.**  Every D the
#                campaign quotes is then bounded to ONE meta stepsize.  CORRECTIONS
#                107.2/107.4 and FINDINGS 76.x/77.x must all carry "at ms=1e-4", and
#                "uniform beats architecture-aligned" becomes a statement about a
#                stepsize, not about partitions.  **This is the branch that would
#                cost the campaign its central partition claim, and it is registered
#                with exactly the same width as the branch that confirms it.**
#           [-0.30, -0.15)   -> **UNDECIDED**, registered in advance.
#           D' <= -0.30      -> **REVERSAL: AT 3e-4 THE ALIGNED PARTITION WINS.**
#                Reported as an inversion of the campaign's central partition claim,
#                not as noise.
#
#   A2    **THE SECONDARY.  bn1's T1 QUESTION AT A SECOND STEPSIZE.**
#         G' = plateau5(chunk2325) - plateau5(nodewise1d), both m = 4,851 EXACTLY.
#         **SAME five-way bands as bn1's T1**, transcribed unchanged:
#           G' > +0.30 SURVIVES | (+0.15,+0.30] UND | [-0.15,+0.15] TAIL WAS THE
#           CARRIER | [-0.30,-0.15) UND | G' <= -0.30 REVERSAL
#         **A2 DOES NOT AMEND bn1's T1.**  T1 returned +0.295 = UNDECIDED and stays
#         UNDECIDED whatever A2 returns; A2 is a reading at a DIFFERENT ms, not a
#         re-run of T1 with more seeds.  Registered so that a batch run after seeing
#         T1 land 0.005 pp below its threshold cannot be used to nudge it across.
#
#   A3    **THE CONCORDANCE TEST.  PRE-REGISTERED, WITH ITS SIGN CONVENTION FIXED
#         HERE, BEFORE ANY ar1 RUN EXISTS.**
#         For each of the two matched-count pairs (ch - node, c23 - n1d):
#           d_acc = dplateau5,  d_fld = dN_eff/m, each with a Welch t at n=3.
#           A channel is RESOLVED at |t| >= 2.0.
#         **CONVENTION, FIXED NOW:** higher N_eff/m -> higher plateau5 = CONCORDANT.
#         This is the direction the noise-averaging literature implies (more
#         effective independence = more information per meta-step).
#         Registered outcomes over the two pairs:
#           both pairs resolved on BOTH channels, both CONCORDANT
#              -> **THE FIELD PREDICTS.**  Direction C's premise is supported and the
#                 sign-agreement programme is worth running.
#           both pairs resolved on BOTH channels, both ANTI-CONCORDANT
#              -> **THE FIELD SYSTEMATICALLY ANTI-PREDICTS**, now PRE-REGISTERED and
#                 at a stepsize where neither channel had been read.  FINDINGS 78.6's
#                 post-hoc pattern becomes a tested prediction and direction C's
#                 premise is refuted on its own terms.
#           both pairs resolved on both channels, MIXED
#              -> **THE FIELD IS NOT A SUFFICIENT STATISTIC**: no monotone function
#                 of N_eff/m orders these arms by accuracy.
#           a pair RESOLVED on field but UNRESOLVED on accuracy
#              -> **DISSOCIATION** at a new stepsize -- the FINDINGS 77.5 pattern,
#                 which until now stands at ONE contrast.
#           a pair UNRESOLVED on FIELD
#              -> that pair is UNINFORMATIVE, is reported as such, and is NOT folded
#                 into a verdict.
#         **A3 IS VOID IF A0.4 FAILS ON ANY ARM.**
#
#   A4    THE ARGMAX METER.  **DESCRIPTIVE, CROSS-BATCH, NO DIRECTION REGISTERED.**
#         Per arm, plateau5 here at 3e-4 vs the CSV's value at 1e-4.  Carries the
#         unmodelled +-0.25 pp cross-batch offset (CORRECTIONS 106.3) on every row.
#         Its ONLY job is to say which stepsize each arm prefers.  It cannot gate
#         A1, A2 or A3, and it is computed and printed AFTER them.
#
#   A5    THE BATCH-OFFSET METER.  nodewise@3e-4 here vs the CSV's 92.493 (n=8,
#         at1/rs/hz9), bar +-0.50.  **CANNOT VOID A1 OR A2** -- both are within-batch
#         differences and any offset common to their arms cancels exactly.
#
# WHAT THIS BATCH EXPLICITLY DOES NOT CLAIM
#   * It does not measure any arm's OWN argmax.  Two stepsizes are two points; the
#     argmax of chunk777, chunk2325 and nodewise1d remains unmeasured, and a
#     three-point curve is the next thing this batch would justify, not a claim it
#     makes.
#   * It does not separate "the group-size distribution" from "the parameter role"
#     on the 1-D tensors -- on ResNet18 those coincide EXACTLY (FINDINGS 78.1).
#   * Every arm is a WITHIN-TENSOR partition.  Layer boundaries stay untested
#     (CORRECTIONS 107.7 item 2 remains the campaign's rank-2 open question).
#   * A3 is two pairs at n=3 on one architecture and one dataset.  It registers a
#     direction; it does not establish a law.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 tests/test_nodebn.py           (CPU, N0 asserts determinism first)
#   2. python3 tests/test_chunkwise.py        (CPU)
#   3. python3 analysis/c55_neff_noise.py --selftest
#   4. `ar1` registered in c55_neff_noise.BOXES BEFORE scoring, from THIS CLIP=.
#   5. plateau5 is PRIMARY.  best_test inflates ~0.37 pp.
#
# USAGE
#   bash c79_argmax_robustness.sh            # dry run
#   bash c79_argmax_robustness.sh --submit   # submits, ONLY if the guards pass
# =============================================================================

import csv, sys, statistics, math
csvp, clip, epochs, mst, mst_ref = sys.argv[1:6]
rd = list(csv.DictReader(open(csvp)))
AX = ['network','dataset','batch_size','base','meta','alpha0','gamma',
      'augment','beta_clip','hier','lam','eta_ratio','epochs_done']
ref = [r for r in rd if r['run'].startswith('rs-lay-1e4')]
if not ref:
    sys.exit("GUARD FAIL: no rs-lay-1e4 reference row")
sig = {a: ref[0][a] for a in AX}
mine = {'network':'ResNet18','dataset':'CIFAR10','batch_size':'100','base':'SGDm',
        'meta':'Lion','alpha0':'1e-3','gamma':'1','augment':'1',
        'beta_clip':clip,'epochs_done':epochs}
bad = [(a, mine[a], sig[a]) for a in mine if mine[a] != sig[a]]
if bad:
    for a,m,s in bad: print("  axis %s: this batch %r vs reference %r" % (a,m,s))
    sys.exit("GUARD FAIL: ar1 does not match the ladder's axis signature")
print("guard 2: all 13 axes match the rs-lay-1e4 signature read FROM THE CSV")

def cell(gran, ms):
    """plateau5 of every 100-ep run at THIS batch's exact 13-axis signature."""
    out = []
    for r in rd:
        if r['granularity'] != gran or r['meta_stepsize'] != ms: continue
        if r['epochs_done'] != epochs or r['beta_clip'] != clip: continue
        if r['augment'] != '1' or r['alpha0'] != '1e-3': continue
        if r['network'] != 'ResNet18' or r['dataset'] != 'CIFAR10': continue
        if not r.get('plateau5','').strip(): continue
        out.append(float(r['plateau5']))
    return out

def sem(v):
    return statistics.stdev(v)/math.sqrt(len(v)) if len(v) > 1 else float('nan')

# guard 2b: THE PREMISE.  ms=1e-4 must actually be off nodewise's argmax, re-derived
# from the CSV RIGHT NOW.  If it is not, this batch has no reason to exist.
lo, hi = cell('nodewise', mst_ref), cell('nodewise', mst)
if len(lo) < 5 or len(hi) < 5:
    sys.exit("GUARD FAIL: nodewise cells too thin (%d @%s, %d @%s)"
             % (len(lo), mst_ref, len(hi), mst))
d = statistics.mean(hi) - statistics.mean(lo)
se = math.sqrt(sem(hi)**2 + sem(lo)**2)
print("guard 2b: THE PREMISE -- nodewise @%s %.3f +-%.3f (n=%d) vs @%s %.3f +-%.3f "
      "(n=%d): %+.3f pp, t %.2f" % (mst, statistics.mean(hi), sem(hi), len(hi),
      mst_ref, statistics.mean(lo), sem(lo), len(lo), d, d/se))
if d < 0.25:
    sys.exit("GUARD FAIL: nodewise gains only %+.3f pp from %s to %s.  The premise "
             "that %s is materially off nodewise's argmax does NOT hold on the "
             "current CSV -- do not submit." % (d, mst_ref, mst, mst_ref))

# guard 2c: chunk777's ms curve must be genuinely UNMEASURED away from 1e-4, or A1

        sys.exit("GUARD FAIL: %d rows of %s already exist at ms=%s -- A1/A2 would "

      "the accuracy and the field are unread there" % mst)

D_ref = statistics.mean(mmc) - statistics.mean(mmn)
g_c, g_n = cell('chunk2325', mst_ref), cell('nodewise1d', mst_ref)
G_ref = statistics.mean(g_c) - statistics.mean(g_n)
print("guard 2d: at ms=%s the CSV reads D = %+.3f (n=%d vs %d) and G = %+.3f "
      "(n=%d vs %d).  A1 and A2 re-read these at ms=%s."
      % (mst_ref, D_ref, len(mmc), len(mmn), G_ref, len(g_c), len(g_n), mst))
if D_ref < 0.30:
    sys.exit("GUARD FAIL: D at ms=%s reads %+.3f, below the +0.30 it was confirmed "
             "at.  The CSV moved; re-reduce before submitting." % (mst_ref, D_ref))

import sys, torch
sys.path.insert(0, ".")
from build_network import build_network
from Optimizers.HF import HF
K, K2, want_node, want_chunk, want_n1d, want_chunk2 = (int(x) for x in sys.argv[1:7])
MST = float(sys.argv[7])

BASE = {"alg": "SGDm", "weight_decay": 0.1, "momentum_param": 0.99}
META = {"alg": "Lion", "meta_stepsize": MST, "momentum_param": 0.99,
        "Lion_beta2": 0.9, "weight_decay": 0}

class NullWriter:
    def add_scalar(self, *a, **k):
        pass

def build(gran):
    torch.manual_seed(0)
    net = build_network("ResNet18", "cpu")
    return HF(net, stepsize_groups=gran, alpha0=1e-3, args_base=dict(BASE),
              args_meta=dict(META), gamma=1, writer=NullWriter())

def m_of(opt):
    return int(sum(int(b.numel()) for b in opt.beta))

net = build_network("ResNet18", "cpu")
shapes = [tuple(p.shape) for p in net.parameters()]

got = {}
for name, gran, want in (("nodewise", "nodewise", want_node),
                         ("chunk%d" % K, "chunk%d" % K, want_chunk),
                         ("nodewise1d", "nodewise1d", want_n1d),
                         ("chunk%d" % K2, "chunk%d" % K2, want_chunk2)):
    m = m_of(build(gran))
    got[name] = m
    print("guard 4: %-12s ALLOCATED beta m=%-8d (registered %d)" % (name, m, want))
    if m != want:
        sys.exit("GUARD FAIL: m(%s) is %d, not the %d registered" % (name, m, want))

# A1's pair must be matched to <= 1 group; A2's must be matched EXACTLY.
d1 = abs(got["chunk%d" % K] - got["nodewise"])
d2 = abs(got["chunk%d" % K2] - got["nodewise1d"])
if d1 > 1:
    sys.exit("GUARD FAIL: A1's arms are %d groups apart, not <= 1" % d1)
if d2 != 0:
    sys.exit("GUARD FAIL: A2's arms are %d groups apart, not 0" % d2)
print("guard 4b: A1's pair is %d group apart (m=%d vs %d); A2's pair is EXACT at m=%d"
      % (d1, got["chunk%d" % K], got["nodewise"], got["nodewise1d"]))

# nodewise1d must still differ from nodewise ONLY on the 1-D tensors, and the 9,610
# size-1 groups FINDINGS 77.6 measured must still be there to be removed.
o_n1d = build("nodewise1d")
want_groups = [1 if len(s) == 1 else s[0] for s in shapes]
if list(o_n1d.n1d_groups) != want_groups:
    sys.exit("GUARD FAIL: nodewise1d group counts are not (1 per 1-D tensor, shape[0] else)")
n_deg = sum(s[0] for s in shapes if len(s) == 1)
if n_deg != 9610:
    sys.exit("GUARD FAIL: nodewise has %d size-1 groups, not the 9,610 FINDINGS 77.6 "
             "measured -- the network changed under the finding" % n_deg)
print("guard 4c: nodewise1d isolates exactly the 1-D tensors; the size-1 tail is "
      "still %d groups, as FINDINGS 77.6 measured" % n_deg)
PYEOF
) || { echo "GUARD 4 FAILED -- nothing submitted"; exit 1; }
