#!/bin/bash
# =============================================================================
# cUC1_unaug_c100_denominator.sh -- `cuc1`: THE DENOMINATOR ON UNAUGMENTED
#   CIFAR-100, WITH THE METHOD GIVEN A GRID.
#   ONE SUBMISSION, 30 JOBS (5 SGD+cosine rungs + 5 MetaOptimize cells, x 3
#   seeds), ResNet18_c100 / CIFAR-100 / batch 100 / 100 epochs / AUGMENT=0,
#   SEEDS {66,67,68} (the block pre-assigned to this track).
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches `cluster_a` (Saber's shared account, READ ONLY); every guard, audit and
# scancel line below is filtered on the `cuc1-` prefix alone.
#
# REGISTERED SCORER: analysis/cUC1_unaug_c100_score.py, git-committed BEFORE
# this batch is submitted (STANDING RULE 21).  Its header freezes the question,
# the PRIMARY (and why it stays an endpoint), every bar as a LITERAL (O2), the
# branch map with REVERSES reachable, the floor gate and the scope stamps.
# This script IMPORTS it and proves, line by line, that every composed
# command is the design the scorer will check (guards 1d and 6b).
#
# THE DOCUMENTED SCORER INVOCATION -- one argument:
#     python3 analysis/cUC1_unaug_c100_score.py <WS>/runs
#
# -----------------------------------------------------------------------------
# THE ARMS
#   cuc1-lr{005,01,02,04,08}-s{66,67,68}   15 jobs.  --optimizer SGD
#       (build_optimizer.py SGD_optimizer, UNPATCHED: SGD_WD 5e-4, SGD_MOM 0.9,
#       cosine with warmup), COS_TOTAL=50000, COS_WARMUP=1000 -- cdn1 arm A's
#       and cau1's recipe.  LR ladder {0.05, 0.1, 0.2, 0.4, 0.8}.
#   cuc1-m<ms>a<a0>-s{66,67,68}            15 jobs.  cdn1-m's spec (SGDm base
#       0.99/0.1 + Lion meta 0.99/0.9/0, gamma 1, chunk771, BETA_CLIP
#       -15:-2.3026, PROBE=5) on a PLUS grid:
#         m1e4a1e3 (REF = cdn1-m/gm2-ch)  m3e4a1e3 (CENTER)  m1e3a1e3
#         m3e4a1e4  m3e4a1e2
#       cdn1-m also exported PROBE5=1,PROBE5_WRITE_EVERY=500 (not echoed by
#       the runner, an instrument); cuc1 exports NEITHER -- every exported
#       variable is one run_cifar.sh echoes.
#
# SGD_WD AND SGD_MOM ARE NOT ECHOED BY run_cifar.sh.  cau1's mechanism
# (CORRECTIONS 206.3 / 209.3), reused: guard 0 REPORTS any such variable into
# PROVENANCE.txt (SHELL_LEAK) and UNSETS it with every echoed variable before
# composing; build_optimizer.py's sha256 pins the defaults (guard 3c, and its
# mtime is stamped for the ingest-time re-check); bin/cUC1_envwitness_poll.sh
# records each running job's /proc/<pid>/environ (SGD_count) and --witness
# audits it at 30/30.
#
# -----------------------------------------------------------------------------
# WALLTIME AND PARTITIONS -- MEASURED (results/all_runs.csv, wallclock_min,
# 100-epoch CIFAR-100 rows of the same harness):
#   SGD+cosine    cdn1-lr (18): median 29 min, worst 67
#   chunk771 meta cdn1-m (3) 51/52, gm2-ch (3) median 51 worst 118, gc1 (4) 56/65
# WALL=03:30:00 is 1.78x the worst chunk771 run and FITS gpu-short's 4:00:00,
# so _lib_guards.sh's composer ADDS gpu-short.  PARTS is never hard-coded
# (CORRECTIONS 178).
# COST: expected 15 x 29 + 15 x 51 min = 20.0 GPU-h; worst observed per family
#       15 x 67 + 15 x 118 = 46.3 GPU-h; HARD BOUND = WALL x 30 = 105 GPU-h.
#
# USAGE   bash bin/cUC1_unaug_c100_denominator.sh              # DRY RUN, submits NOTHING
#         bash bin/cUC1_unaug_c100_denominator.sh --dry-run    # identical, explicit
#         bash bin/cUC1_unaug_c100_denominator.sh --envaudit   # AFTER launch: ENV audit
#         bash bin/cUC1_unaug_c100_denominator.sh --witness    # AFTER launch: SGD_* witness audit
#         bash bin/cUC1_unaug_c100_denominator.sh --submit     # the ONLY submitting form
# =============================================================================
set -u
SUBMIT=0; MODE=dry
case "${1:-}" in
  --submit)          SUBMIT=1; MODE=submit ;;
  ""|--dry-run)      SUBMIT=0; MODE=dry ;;
  --envaudit)        SUBMIT=0; MODE=envaudit ;;
  --witness)         SUBMIT=0; MODE=witness ;;
  *) echo "usage: $0 [--dry-run|--submit|--envaudit|--witness]"; exit 2 ;;
esac
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

# ---- environment hygiene: REPORT, then UNSET, everything the runner echoes --
LEAK=""
for V in SGD_WD SGD_MOM PROBE5 PROBE5_WRITE_EVERY; do
  [ -n "${!V+x}" ] && LEAK="$LEAK $V=${!V}"
done
unset SGD_WD SGD_MOM AUGMENT BETA_CLIP HIER LAM ETA_RATIO COS_TOTAL COS_WARMUP \
      SCHED SCHED_TOTAL SCHED_WARMUP SCHED_MIN PROBE PROBE_DIR PROBE5 \
      PROBE5_WRITE_EVERY PROBE_TENSOR EB_RHO EB_LOG

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cuc1
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}
SC="$REPO/analysis/cUC1_unaug_c100_score.py"
CSVP="$REPO/results/all_runs.csv"
WITDIR=${WITNESS_DIR:-$HOME/cuc1_envwitness}

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

EPOCHS=100; BATCH=100; AUG=0; GAMMA=1
NET=ResNet18_c100; DSET=CIFAR100
SEEDS="66 67 68"
LRS="0.05:005 0.1:01 0.2:02 0.4:04 0.8:08"
COS_T=50000; COS_W=1000
GRID="m1e4a1e3:1e-4:1e-3 m3e4a1e3:3e-4:1e-3 m1e3a1e3:1e-3:1e-3 m3e4a1e4:3e-4:1e-4 m3e4a1e2:3e-4:1e-2"
CLIP="-15:-2.3026"; GRAN=chunk771; PROBE_M=5
NJOBS=30
# pins -- the live files these arms depend on (both read on cluster_b 2026-09-15; = cau1's pins)
BOPT_SHA=25a899b3745e9d66fbf630795a1202c6075e2067daffa077e4a54c28e54c7ec2
LOAD_SHA=b52b58a35dcecd80ce6d8f8033b09c20dad9f4827b808f4ea2a828ce88138c46
WALL=03:30:00
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

# =============================================================================
# post-launch ENV audit (RULE 20's SEPARATE half), checked against the
# SCORER's expected_env() so the audit and the scorer cannot disagree.
# =============================================================================
if [ "$MODE" = envaudit ]; then
  echo "=== cuc1 ENV-LINE AUDIT (separate from RULE 20's ARGS audit) ==="
  python3 - "$SC" "$WS/runs" <<'PYEOF'
import sys, re, importlib.util
spec = importlib.util.spec_from_file_location("cuc1", sys.argv[1])
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
runs, dups = m.read_runs(sys.argv[2])
bad = n = 0
fam = {}
for (a, s), rec in sorted(runs.items()):
    n += 1
    if len(rec["env"]) != 1:
        print("  %s: %d ENV lines" % (rec["file"], len(rec["env"]))); bad += 1; continue
    got = m.parse_env_line(rec["env"][0])
    for k, w in m.expected_env(a, s).items():
        if not m._match(w, got.get(k)):
            print("  %s: %s=%s want %s" % (rec["file"], k, got.get(k), w)); bad += 1
    if not rec["node"] or "AUGMENT=0" not in rec["node"][0]:
        print("  %s: NODE header not AUGMENT=0" % rec["file"]); bad += 1
    fam.setdefault("lr" if a.startswith("lr") else a, set()).add(re.sub(r" PROBE_DIR=\S+", "", rec["env"][0]))
for k, v in sorted(fam.items()):
    print("  family %-9s %d distinct ENV line(s) (PROBE_DIR stripped)" % (k, len(v)))
    for x in sorted(v):
        print("      " + x)
for d in dups:
    print("  NOTE duplicate .out: %s s%d %s / %s" % d)
print("  %d/%d runs read, %d violations" % (n, m.NJOBS, bad))
print("ENV AUDIT VERDICT: %s" % ("PASS" if bad == 0 and n == m.NJOBS else
                                 ("FAIL" if bad else "UNVERIFIED (%d/%d present)" % (n, m.NJOBS))))
sys.exit(0 if bad == 0 and n == m.NJOBS else (1 if bad else 2))
PYEOF
  exit $?
fi

# =============================================================================
# post-launch SGD_* runtime-witness audit (bin/cUC1_envwitness_poll.sh output)
# =============================================================================
if [ "$MODE" = witness ]; then
  echo "=== cuc1 RUNTIME /proc/<pid>/environ WITNESS AUDIT ($WITDIR) ==="
  python3 - "$SC" "$WITDIR" <<'PYEOF'
import sys, os, re, importlib.util
spec = importlib.util.spec_from_file_location("cuc1", sys.argv[1])
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
wd = sys.argv[2]
files = sorted(os.listdir(wd)) if os.path.isdir(wd) else []
seen, bad = {}, 0
for f in files:
    mm = re.match(r"^(cuc1-(lr\d+|m\de\da\de\d)-s(\d+))-(\d+)\.txt$", f)
    if not mm:
        continue
    rn, arm, seed, jid = mm.group(1), mm.group(2), int(mm.group(3)), mm.group(4)
    kv = {}
    for ln in open(os.path.join(wd, f)):
        if "=" in ln and not ln.startswith(("PID", "SGD_VAR")):
            k, v = ln.strip().split("=", 1)
            kv[k] = v
    want = {"AUGMENT": "0", "SLURM_JOB_NAME": rn, "SLURM_JOB_ID": jid, "SGD_count": "0"}
    if arm.startswith("lr"):
        want.update({"COS_TOTAL": m.COS_TOTAL, "COS_WARMUP": m.COS_WARMUP, "PROBE": "0"})
    else:
        want.update({"BETA_CLIP": m.CLIP, "PROBE": m.PROBE_META})
    errs = ["%s=%s want %s" % (k, kv.get(k), w) for k, w in want.items() if kv.get(k) != w]
    errs += ["%s present" % k for k in ("PROBE5", "PROBE5_WRITE_EVERY") if k in kv]
    if errs:
        bad += 1
        print("  %s: %s" % (f, "; ".join(errs)))
    seen[(arm, seed)] = f
want_all = set((a, s) for a in m.ARMS for s in m.SEEDS)
miss = sorted(want_all - set(seen))
print("  coverage %d/%d, SGD_count=0 and positive control on %d" % (len(seen), len(want_all), len(seen) - bad))
for a, s in miss:
    print("  NO WITNESS: %s" % m.run_name(a, s))
print("WITNESS VERDICT: %s" % ("PASS" if not bad and not miss else ("FAIL" if bad else
                               "UNVERIFIED (%d/%d witnessed)" % (len(seen), len(want_all)))))
sys.exit(0 if not bad and not miss else (1 if bad else 2))
PYEOF
  exit $?
fi

echo "=============================================================="
echo " cuc1 -- THE DENOMINATOR ON UNAUGMENTED CIFAR-100, METHOD ON A GRID"
echo " $DSET / $NET / batch $BATCH / $EPOCHS epochs / AUGMENT=$AUG / seeds $SEEDS"
echo "   lr*  SGD+cosine, COS_TOTAL=$COS_T COS_WARMUP=$COS_W, ladder $(echo $LRS | sed 's/:[0-9]*//g')"
echo "   m*   SGDm+Lion $GRAN, box $CLIP, grid (tag:ms:alpha0) $GRID"
echo " 10 arms x 3 seeds = $NJOBS jobs, ONE submission"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY: GAP_END = plateau5(best rung) - plateau5(best meta cell), IN BATCH"
echo " MODE: $MODE"
echo "=============================================================="

# ---- guard 0: environment hygiene -------------------------------------------
if [ -n "$LEAK" ]; then
  echo "guard 0: the submitting shell carried$LEAK -- REPORTED and UNSET"
else
  echo "guard 0: SGD_WD / SGD_MOM / PROBE5 / PROBE5_WRITE_EVERY were not set in the submitting shell"
fi
echo "guard 0b: every variable run_cifar.sh echoes was UNSET before composition"

# ---- guard 1a: EVERY repo file this launcher needs ---------------------------
NEED="analysis/cUC1_unaug_c100_score.py bin/cUC1_unaug_c100_denominator.sh bin/_lib_guards.sh
analysis/argsline_guard.py results/all_runs.csv bin/PROTECTED.txt bin/cUC1_envwitness_poll.sh
analysis/cUC1_smoke_gen.py"
for F in $NEED; do
  [ -f "$REPO/$F" ] && echo "guard 1a: $F  sha256 $(sha "$REPO/$F")" || guard_fail "guard 1a: missing $REPO/$F"
done
echo "guard 1a: CSV $(($(wc -l < "$CSVP") - 1)) rows"

# ---- guard 1b: committed (RULE 21) -------------------------------------------
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  # shellcheck disable=SC2086
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEED 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: every file in guard 1a is committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires them committed BEFORE any run exists: $DIRTY"
  DEL=$(git -C "$REPO" diff HEAD --numstat -- analysis/ bin/ 2>/dev/null | awk '$2+0>0{print}')
  [ -z "$DEL" ] && echo "guard 1b2: no uncommitted DELETIONS under analysis/ or bin/ (RULE 16)" \
               || guard_fail "guard 1b2: uncommitted deletions: $DEL"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 is verified on the committing"
  echo "          host by wall clock (git %ct) against the earliest sacct Submit, and guard 1a's"
  echo "          sha256 list must equal the committed blobs (git hash-object is not sha256: compare"
  echo "          against 'git show HEAD:<path> | sha256sum' on the Mac)."
fi

# ---- guard 1c: the registered scorer selftests ON THIS HOST -------------------
ST=/tmp/cuc1_selftest.$$
if python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$ST" 2>&1; then
  echo "guard 1c: scorer --selftest PASSES: $(tail -1 "$ST")"
else
  guard_fail "guard 1c: scorer --selftest FAILED"; grep -E '^  FAIL|FAILED|Traceback' "$ST" | head -20
fi
NB1=$(grep -c "^  PASS B' " "$ST"); NB2=$(grep -c "^  PASS B'' " "$ST")
[ "$NB1" = 4 ] && [ "$NB2" = 3 ] \
  && echo "guard 1c2: B' (4 raw-.out pool families) and B'' (3 cdn1-m probes) RAN on this host, not skipped" \
  || guard_fail "guard 1c2: B' passed $NB1/4, B'' passed $NB2/3 -- the frozen raw pools / travel premise not re-derived here"
grep -q "^  PASS ZERO rows outside cuc1- carry seed 66-68" "$ST" \
  && echo "guard 1c3: the scorer's own fresh-seed check ran and passed" \
  || guard_fail "guard 1c3: the scorer's fresh-seed check did not pass"
grep -q "^  PASS whole stdout BYTE-IDENTICAL" "$ST" && grep -q "^  PASS NOT VACUOUS" "$ST" \
  && echo "guard 1c4: section F (frozen bars, O2) passed and is not vacuous" \
  || guard_fail "guard 1c4: section F (frozen bars) did not pass"
NSK=$(grep -c "^  SKIP" "$ST")
[ "$NSK" = 0 ] && echo "guard 1c5: 0 SKIP lines in the selftest" \
               || guard_fail "guard 1c5: $NSK selftest sections SKIPPED on this host"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 1d: this script's design IS the scorer's design ------------------
python3 - "$SC" "$LRS" "$SEEDS" "$COS_T" "$COS_W" "$GRID" "$CLIP" "$GRAN" "$EPOCHS" "$NJOBS" "$NET" "$DSET" <<'PYEOF' \
  || guard_fail "guard 1d: the launcher and the scorer disagree on the design"
import sys, importlib.util
sc, lrs, seeds, ct, cw, grid, clip, gran, ep, nj, net, ds = sys.argv[1:13]
spec = importlib.util.spec_from_file_location("cuc1", sc)
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
bad = []
if [tuple(x.split(":")) for x in lrs.split()] != m.LADDER: bad.append("ladder")
if tuple(int(s) for s in seeds.split()) != tuple(m.SEEDS): bad.append("seeds")
if (ct, cw) != (m.COS_TOTAL, m.COS_WARMUP): bad.append("cosine horizon")
g = [tuple(x.split(":")) for x in grid.split()]
if [t for t, _, _ in g] != list(m.META_ARMS) or any((m.META[t]["ms"], m.META[t]["alpha0"]) != (ms, a0) for t, ms, a0 in g):
    bad.append("meta grid %s" % g)
if (clip, gran, net, ds) != (m.CLIP, m.GRAN, m.NET, m.DSET): bad.append("box/granularity/net/dataset")
if int(ep) != m.EPOCHS or int(nj) != m.NJOBS or len(m.arm_names()) != m.NJOBS: bad.append("epochs / job count")
if bad:
    print("!!! guard 1d: " + "; ".join(bad)); sys.exit(1)
print("guard 1d: ladder, seeds, cosine horizon, the 5-cell grid, box, granularity, net, dataset,"
      " epochs and the %d names are IDENTICAL in this script and the scorer" % m.NJOBS)
PYEOF

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ----------
grep -q '^cuc1-' "$CSVP" 2>/dev/null && guard_fail "guard 2: cuc1-* already in the CSV" \
                                     || echo "guard 2: no cuc1-* name in the corpus"
NOUT=$(ls "$WS/runs"/cuc1-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cuc1-*.out on disk" \
                     || guard_fail "guard 2b: $NOUT cuc1-*.out files already exist"
NSEED=$(grep -l -E -- '--seed (66|67|68)( |$)' "$WS"/runs/*.out "$WS"/runs/*/*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NSEED:-0}" = 0 ] && echo "guard 2c: ZERO .out files under \$WS/runs carry --seed 66-68" \
                      || guard_fail "guard 2c: $NSEED .out files already carry a seed in 66-68"
NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cuc1-')
[ "${NQ:-0}" = 0 ] && echo "guard 2d: no cuc1-* job in the queue" \
                   || guard_fail "guard 2d: $NQ cuc1-* jobs already queued"
NS=$(sacct -u "$USER_NAME" -S 2026-01-01 -X -n -o JobName%60 2>/dev/null | grep -c '^ *cuc1-')
[ "${NS:-0}" = 0 ] && echo "guard 2e: no cuc1-* record in sacct since 2026-01-01" \
                   || guard_fail "guard 2e: $NS cuc1-* records already in sacct"

# ---- guard 3: the live tree -------------------------------------------------
[ -f "$RUNNER" ] && echo "guard 3: runner $RUNNER present (sha256 $(sha "$RUNNER"))" || guard_fail "guard 3: no runner"
grep -q 'ARGS: \$@' "$RUNNER" 2>/dev/null && echo "guard 3a: runner echoes the ARGS line" \
  || guard_fail "guard 3a: runner does not echo ARGS -- RULE 20 unauditable"
for K in AUGMENT BETA_CLIP COS_TOTAL COS_WARMUP PROBE PROBE_DIR SCHED HIER; do
  grep -q "ENV: .*$K=" "$RUNNER" 2>/dev/null || guard_fail "guard 3a: runner ENV line lacks $K"
done
echo "guard 3a: runner ENV line carries AUGMENT BETA_CLIP COS_TOTAL COS_WARMUP PROBE PROBE_DIR SCHED HIER"
LD="$CIF/load_data.py"
[ "$(sha "$LD")" = "$LOAD_SHA" ] && echo "guard 3b: load_data.py sha256 matches the pin" \
                                 || guard_fail "guard 3b: load_data.py sha256 $(sha "$LD"), pinned $LOAD_SHA"
python3 - "$LD" <<'PYEOF' || guard_fail "guard 3b2: AUGMENT=0 does not provably disable CIFAR-100 augmentation"
import sys
src = open(sys.argv[1]).read()
blk = src.split('elif dataset_name == "CIFAR100":', 1)[1].split("elif dataset_name", 1)[0]
ok = ('os.environ.get("AUGMENT", "0") == "1"' in blk and "train_transform = test_transform" in blk
      and "RandomCrop" in blk and blk.index('== "1"') < blk.index("RandomCrop")
      < blk.index("train_transform = test_transform") and "datasets.CIFAR100(" in blk)
print("guard 3b2: CIFAR100 branch: RandomCrop/Flip ONLY under AUGMENT == \"1\"; otherwise the TRAIN"
      " transform IS the TEST transform: %s" % ok)
sys.exit(0 if ok else 1)
PYEOF
BO="$CIF/Optimizers/build_optimizer.py"
[ "$(sha "$BO")" = "$BOPT_SHA" ] && echo "guard 3c: build_optimizer.py sha256 matches cdn1's/cau1's pin (SGD path unpatched)" \
                                 || guard_fail "guard 3c: build_optimizer.py sha256 $(sha "$BO"), pinned $BOPT_SHA"
grep -q "if args.optimizer == 'SGD'" "$BO" && grep -q "SGD_WD', 5e-4" "$BO" \
  && grep -q "SGD_MOM', 0.9" "$BO" && grep -q "COS_TOTAL" "$BO" && grep -q "COS_WARMUP" "$BO" \
  && echo "guard 3c2: SGD branch present; SGD_WD default 5e-4, SGD_MOM default 0.9, COS_* read" \
  || guard_fail "guard 3c2: the SGD branch or its documented defaults are not there"
BO_MTIME=$(stat -c %y "$BO" 2>/dev/null || stat -f %Sm "$BO")
echo "guard 3c3: build_optimizer.py mtime $BO_MTIME (re-check sha + mtime at ingest, 206.3)"
HFLIVE="$CIF/Optimizers/HF.py"
grep -q "BETA_CLIP" "$HFLIVE" 2>/dev/null && echo "guard 3d: live HF.py reads BETA_CLIP" \
  || guard_fail "guard 3d: live HF.py does not read BETA_CLIP"
grep -q "self.args_meta\['meta_stepsize'\] \* torch.sign(" "$HFLIVE" \
  && echo "guard 3d2: live HF.py Lion meta update is beta -= ms * sign(.) -- the travel arithmetic's premise" \
  || guard_fail "guard 3d2: the Lion meta update is not beta -= ms * sign(.) -- the grid's travel argument fails"
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3e: cifar-100-python staged under $CIF/data" \
  || guard_fail "guard 3e: CIFAR-100 is not staged at $CIF/data/cifar-100-python"
HFSHA=$(sha "$HFLIVE"); TRSHA=$(sha "$CIF/train.py"); BNSHA=$(sha "$CIF/build_network.py")

# ---- guard 4: provenance ------------------------------------------------------
# A DRY RUN WRITES NOTHING UNDER $WS/runs.  Its provenance goes to /tmp.
if [ "$SUBMIT" = 1 ]; then PROV="$SAVE/PROVENANCE.txt"; MANI="$SAVE/PARTITION-MANIFEST.txt"; mkdir -p "$SAVE"
else PROV=/tmp/cuc1_PROVENANCE.$$; MANI=/tmp/cuc1_MANIFEST.$$; fi
{ echo "BATCH cuc1"; echo "MODE $( [ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN )";
  echo "STAMP_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)";
  [ "$SUBMIT" = 1 ] && echo "SUBMIT_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)";
  echo "SCORER_SHA256 $(sha "$SC")"; echo "LAUNCHER_SHA256 $(sha "$0")";
