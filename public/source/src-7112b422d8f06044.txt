def mean(v):
    return sum(v) / float(len(v))


def sd(v):
    if len(v) < 2:
        return 0.0
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def file_sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()


# =====================================================================================================================
# THE DECISION (pure functions of numbers; the selftest drives them exhaustively)
# =====================================================================================================================
def rung_state(d, se, diverged, healthy, boxfree):
    if diverged:
        return "DIVERGED"
    if not healthy:
        return "UNHEALTHY"
    if not boxfree:
        return "BOXBOUND"
    if d <= -NULLBAR and d < -RES * se:
        return "REVERSES"
    if d <= NULLBAR and d + RES * se < POOL_EXCLUDE:
        return "VANISHES"
    if d >= SURV and d > RES * se:
        return "SURVIVES"
    return "UNDECIDED"


def scalar_of(k_level, k_div, cmp_ok, t_ch, t_nd, se):
    if k_div:
        return "SCALAR-DIVERGED"
    if k_level < HEALTH_MIN:
        return "SCALAR-COLLAPSED"
    if not cmp_ok:
        return "SCALAR-UNREADABLE"
    tmin = min(t_ch, t_nd)
    if t_ch >= SURV and t_nd >= SURV and t_ch > RES * se and t_nd > RES * se:
        return "SCALAR-BEATS-BEST"
    if tmin - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tmin <= -SURV and tmin < -RES * se:
        return "SCALAR-BELOW-BEST"
    return "SCALAR-UNRESOLVED"


def agreement(s_test, s_val):
    if s_test == s_val:
        return "SAME"
    if any({s_test, s_val} == o for o in OPPOSITE):
        return "FLIP"
    return "DIFF"


def primary_of(st, gap_res=None):
    """st[reader][claim] -> the primary token.  gap_res[claim] = the paired gap is RESOLVED."""
    gap_res = gap_res or {}
    if any(st["TEST"][c] in GATE_STATES for c in CLAIMS):
        return "UNREADABLE-TEST"
    if any(st["VAL"][c] in GATE_STATES for c in CLAIMS):
        return "UNREADABLE-VAL"
    ag = [agreement(st["TEST"][c], st["VAL"][c]) for c in CLAIMS]
    if "FLIP" in ag:
        return "VAL-FLIPS"
    if all(a == "SAME" for a in ag):
        return "VAL-AGREES"
    if any(gap_res.get(c) for c in CLAIMS):
        return "VAL-SHIFTS"
    return "VAL-DIFFERS-UNRESOLVED"


def gap_resolved(g, se):
    return abs(g) >= SURV and abs(g) > RES * se


def select_of(levels_test, levels_val, se_test):
    """levels_X[arm] -> (token, pick_test, pick_val, regret)."""
    pt = max(sorted(levels_test), key=lambda a: levels_test[a])
    pv = max(sorted(levels_val), key=lambda a: levels_val[a])
    reg = levels_test[pt] - levels_test[pv]
    if pt == pv:
        return "SELECT-SAME", pt, pv, reg
    if reg >= SURV and reg > RES * se_test:
        return "SELECT-DIFFERS-RESOLVED", pt, pv, reg
    return "SELECT-DIFFERS-UNRESOLVED", pt, pv, reg


def compose_sigma(sigma_in):
    return (SIGMA_PRIOR, "SIGMA_PRIOR_frozen") if SIGMA_PRIOR >= sigma_in else (sigma_in, "SIGMA_INBATCH")


def reader_states(A, sig):
    """A[arm] = dict(level, range, n, boxfree) for ONE reader."""
    st, K = {}, {}
    for r in D.RUNG_IDS:
        ch, nd = A["ch" + r], A["nd" + r]
        d = ch["level"] - nd["level"]
        se = sig * math.sqrt(1.0 / ch["n"] + 1.0 / nd["n"])
        st[r] = rung_state(d, se, ch["range"] > DIVERGED or nd["range"] > DIVERGED,
                           ch["level"] >= HEALTH_MIN and nd["level"] >= HEALTH_MIN, ch["boxfree"] and nd["boxfree"])
        K["D_" + r] = (d, se)
        k = A["k01" + r]
        for g in ("ch", "nd", "kL"):
            x = A[g + r]
            K["T%s_%s" % (g, r)] = (k["level"] - x["level"], sig * math.sqrt(1.0 / k["n"] + 1.0 / x["n"]))
    k4, c4, n4 = A["k01W4"], A["chW4"], A["ndW4"]
    cmp_ok = (c4["level"] >= HEALTH_MIN and n4["level"] >= HEALTH_MIN and c4["range"] <= DIVERGED and n4["range"] <= DIVERGED)
    st["SC"] = scalar_of(k4["level"], k4["range"] > DIVERGED, cmp_ok, K["Tch_W4"][0], K["Tnd_W4"][0], K["Tch_W4"][1])
    return st, K


def decide(AT, AV, sig_t, sig_v, s_delta):
    stT, KT = reader_states(AT, sig_t)
    stV, KV = reader_states(AV, sig_v)
    st = {"TEST": stT, "VAL": stV}
    se_gap = s_delta * math.sqrt(0.5)
    gaps = {"W1": KV["D_W1"][0] - KT["D_W1"][0], "W4": KV["D_W4"][0] - KT["D_W4"][0],
            "SC": min(KV["Tch_W4"][0], KV["Tnd_W4"][0]) - min(KT["Tch_W4"][0], KT["Tnd_W4"][0])}
    gres = dict((c, gap_resolved(gaps[c], se_gap)) for c in CLAIMS)
    KT["GAPS"], KT["GAP_SE"], KT["GAP_RES"] = gaps, se_gap, gres
    prim = primary_of(st, gres)
    sel = select_of(dict((a, AT[a]["level"]) for a in D.ARMS), dict((a, AV[a]["level"]) for a in D.ARMS),
                    sig_t * math.sqrt(0.5))
    return prim, sel, st, KT, KV

def parse_args_line(line):
    toks = shlex.split(line[len("ARGS:"):])
    pairs, seen, rep, i = [], set(), [], 0
    while i < len(toks):
        t = toks[i]
        if not t.startswith("--"):
            return None, ["stray token %r" % t]
        if i + 1 >= len(toks) or toks[i + 1].startswith("--"):
            return None, ["flag %s has no value" % t]
        f = t[2:]
        if f in seen:
            rep.append(f)
        seen.add(f)
        pairs.append((f, toks[i + 1]))
        i += 2
    return pairs, rep


def read_out(path):
    L = open(path, errors="replace").read().splitlines()
    info = dict(path=path, args=None, env=None, gpu=None, done=False, eps={}, val={}, val_bad=[], kinds_on=[],
                vs=[], pt=False, order_ok=True)
    last_ep = None
    for ln in L:
        if ln.startswith("ARGS:") and info["args"] is None:
            info["args"] = ln
        elif ln.startswith("ENV:") and info["env"] is None:
            info["env"] = ln
        elif ln.startswith("NVIDIA") and info["gpu"] is None:
            info["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            info["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            info["pt"] = True
        elif ln.startswith("VAL_SPLIT"):
            info["vs"].append(ln)
        elif ln.startswith("VAL:"):
            m = VAL_RE.match(ln)
            if not m or int(m.group(3)) != D.N_VAL:
                info["val_bad"].append(ln[:80])
                continue
            e = int(m.group(1))
            if e in info["val"] or last_ep != e:
                info["order_ok"] = False
            info["val"][e] = float(m.group(2))
        else:
            m = EPTR_RE.search(ln)
            if m:
                info["eps"][int(m.group(1))] = (float(m.group(2)), float(m.group(3)))
                last_ep = int(m.group(1))
            k = KIND_ON_RE.match(ln)
            if k:
                info["kinds_on"].append(ln.strip())
    info["complete"] = (info["done"] and all(e in info["eps"] for e in range(D.EPOCHS)))
    return info


def find_outs(runsdir):
    out = []
    for base in (runsdir, os.path.join(runsdir, PREFIX)):
        out += glob.glob(os.path.join(base, "%s-*.out" % PREFIX))
    return sorted(set(os.path.realpath(p) for p in out))


def probe_path(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, D.probe_dir_name(arm, seed), "probe.jsonl")
        if os.path.exists(p):
            return p
    return None


def read_probe(path):
    import json
    recs = []
    for ln in open(path, errors="replace"):
        if not ln.strip():
            continue
        try:
            r = json.loads(ln)
        except ValueError:
            recs.append(None)
            continue
        recs.append({k: r.get(k) for k in ("step", "n_beta", "beta_true_min", "beta_true_max", "n_at_lo", "n_at_hi")}
                    if isinstance(r, dict) else None)
    return recs


def probe_audit(recs, m_want, n_records=None, win_step=None):
    """-> (ok, why, occ).  G-STRUCT: n_records records at steps 0,5,..; n_beta == the arm's m on EVERY record.
    At 45,000 training images there are 9,000 records (a 50,000-image run's 10,000 FAIL here: the count is the witness
    that the run trained on 450 batches per epoch)."""
    n_records = D.N_RECORDS if n_records is None else n_records
    win_step = WIN_STEP if win_step is None else win_step
    if recs is None:
        return False, "no probe.jsonl", None
    if len(recs) != n_records:
        return False, "%d records, want %d" % (len(recs), n_records), None
    if any(r is None for r in recs):
        return False, "an unreadable record", None
    if [r["step"] for r in recs] != list(range(0, n_records * D.PROBE, D.PROBE)):
        return False, "record steps are not 0,5,...,%d" % ((n_records - 1) * D.PROBE), None
    nb = set(r["n_beta"] for r in recs)
    if nb != {m_want}:
        return False, "n_beta %s != the arm's registered m %d" % (sorted(nb)[:3], m_want), None
    mins = [r["beta_true_min"] for r in recs]
    maxs = [r["beta_true_max"] for r in recs]
    if not (max(maxs) - min(mins) > 1e-9):
        return False, "beta never moved", None
    T = float(len(recs))
    rec_lo = sum(1 for x in mins if x <= D.LO + BOX_EPS) / T
    rec_hi = sum(1 for x in maxs if x >= D.HI - BOX_EPS) / T
    win = [r for r in recs if r["step"] >= win_step]
    return True, "", dict(rec_lo=rec_lo, rec_hi=rec_hi, boxfree=(rec_lo < BOXFREE_MAX and rec_hi < BOXFREE_MAX),
                          beta_min_win=min(r["beta_true_min"] for r in win), beta_max_win=max(r["beta_true_max"] for r in win))


def read_corpus_disclosure(csvpath):

        return None
    sys.path.insert(0, HERE)
    try:
        import corpus_exclusions as ce
    finally:
        sys.path.pop(0)
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    return len(ce.filter_rows(rows))

def score(runsdir, csvpath=DEFAULT_CSV, out=print):
    P = out
    P("=" * 110)
    P("cvl1 -- DO THE AUDIT'S TEST RANKINGS HOLD ON A HELD-OUT VALIDATION SPLIT?  (CORRECTIONS 303; ICML-PLAN 1.14)")
    P("scorer sha256 %s   design sha256 %s" % (file_sha(os.path.abspath(__file__)), file_sha(os.path.join(HERE, "cvl1_design.py"))))
    P("runsdir %s" % runsdir)
    P("=" * 110)
    harness = []

    def hfail(msg):
        harness.append(msg)
        P("  !!! HARNESS %s" % msg)

    found = {}
    for p in find_outs(runsdir):
        m = OUT_RE.match(os.path.basename(p))
        if not m:
            hfail("unparseable file name %s" % os.path.basename(p))
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        if arm not in D.ARMS or seed not in D.SEEDS[arm]:
            hfail("a %s .out that is not a registered run: %s" % (PREFIX, os.path.basename(p)))
            continue
        found.setdefault((arm, seed), []).append((jid, p))
    runs, missing, ignored = {}, [], []
    for (arm, seed) in D.RUNS:
        cands = [(jid, p, read_out(p)) for jid, p in sorted(found.get((arm, seed), []))]
        comp = [c for c in cands if c[2]["complete"]]
        ignored += [os.path.basename(c[1]) for c in cands if not c[2]["complete"]]
        if not comp:
            missing.append(D.run_name(arm, seed))
        elif len(comp) > 1:
            hfail("%s has %d COMPLETE .out files -- ambiguous" % (D.run_name(arm, seed), len(comp)))
        else:
            runs[(arm, seed)] = comp[0]
    P("COMPLETION  %d / %d registered runs complete (100 epoch lines + RUN_DONE)" % (len(runs), D.NJOBS))
    if ignored:
        P("  ignored (incomplete): %s" % ", ".join(sorted(ignored)))
    if missing:
        P("  MISSING / INCOMPLETE: %s" % ", ".join(missing))

    prov = resolve_prov(runsdir)
    if prov is None:
        hfail("G-PROV: no PROVENANCE.txt under %s/%s" % (runsdir, PREFIX))
    else:
        kv = dict(ln.split(" ", 1) for ln in open(prov).read().splitlines() if " " in ln)
        want = {"SCORER_SHA256": file_sha(os.path.abspath(__file__)), "DESIGN_SHA256": file_sha(os.path.join(HERE, "cvl1_design.py")),
                "HF_SHA256": D.HF_LIVE_SHA, "LOAD_DATA_SHA256": D.LOADDATA_POST_SHA, "TRAIN_SHA256": D.TRAIN_POST_SHA}
        for k, v in sorted(want.items()):
            if kv.get(k, "").strip() != v:
                hfail("G-PROV: %s in %s is %r, want %s" % (k, prov, kv.get(k), v[:16]))
        P("G-PROV  %s  MODE %s  SCORER %s  DESIGN %s  LOAD_DATA %s  TRAIN %s" % (prov, kv.get("MODE", "?"),
          kv.get("SCORER_SHA256", "?")[:16], kv.get("DESIGN_SHA256", "?")[:16], kv.get("LOAD_DATA_SHA256", "?")[:16],
          kv.get("TRAIN_SHA256", "?")[:16]))

    per, gpus, save_dirs = {}, {}, set()
    for (arm, seed), (jid, path, info) in sorted(runs.items()):
        rn = D.run_name(arm, seed)
        if info["args"] is None:
            hfail("G-ARGS %s: no ARGS line" % rn)
            continue
        pairs, rep = parse_args_line(info["args"])
        if pairs is None or rep:
            hfail("G-ARGS %s: %s" % (rn, rep))
            continue
        sd_ = dict(pairs).get("save-directory", "")
        save_dirs.add(sd_)
        want = D.expected_args(arm, seed, sd_)
        if pairs != want:
            diff = [(f, v, dict(want).get(f)) for f, v in pairs if dict(want).get(f) != v]
            hfail("G-ARGS %s: %d flags differ from the registered line %s" % (rn, len(diff), diff[:4]))
            continue
        if info["env"] != D.ENV_TEMPLATE.replace("<P>", os.path.join(sd_, D.probe_dir_name(arm, seed))):
            hfail("G-ENV %s: ENV line is not the registered one (%r)" % (rn, (info["env"] or "")[:160]))
            continue
        if info["vs"] != [D.witness_on()]:
            hfail("G-VALSPLIT %s: the VAL_SPLIT witness is %r, want exactly ONE %r" % (rn, info["vs"][:2], D.witness_on()[:90]))
            continue
        others = [k for k in info["kinds_on"] if not k.startswith("VAL_SPLIT")]
        if others or info["pt"]:
            hfail("G-KIND %s: prints another patch witness %s" % (rn, (others or ["PROBE_TENSOR"])[:2]))
            continue
        if info["val_bad"] or sorted(info["val"]) != list(range(D.EPOCHS)) or not info["order_ok"]:
            hfail("G-VAL %s: VAL lines are not epochs 0..99 once each, each right after its Epoch line (%d read, bad %r)"
                  % (rn, len(info["val"]), info["val_bad"][:1]))
            continue
        pp = probe_path(runsdir, arm, seed)
        ok, why, occ = probe_audit(read_probe(pp) if pp else None, D.M_OF_ARM[arm])
        if not ok:
            hfail("G-STRUCT %s: %s" % (rn, why))
            continue
        eps, val = info["eps"], info["val"]
        per[(arm, seed)] = dict(TEST=mean([eps[e][1] for e in range(PL_LO, PL_HI + 1)]),
                                VAL=mean([val[e] for e in range(PL_LO, PL_HI + 1)]),
                                train=mean([eps[e][0] for e in range(PL_LO, PL_HI + 1)]), occ=occ, jid=jid)

        hfail("G-ARGS: the runs name %d different save directories" % len(save_dirs))

        return 3
    if missing:
        P("\nFINAL INCOMPLETE | %d of %d runs missing or incomplete; nothing is read" % (len(missing), D.NJOBS))
        return 2

    AR = {"TEST": {}, "VAL": {}}

    for X in READERS:

        for a in D.ARMS:

            ss += sum((x - m_) ** 2 for x in xs)
            df += len(xs) - 1
            AR[X][a] = dict(level=m_, range=max(xs) - min(xs), n=len(xs), sd=sd(xs),
                            boxfree=all(per[(a, s)]["occ"]["boxfree"] for s in D.SEEDS[a]))
        sig_in[X] = (math.sqrt(ss / df) if df else 0.0, df)
    # the paired VAL - TEST gap per run
    dss, ddf = 0.0, 0
    for a in D.ARMS:
        ds = [per[(a, s)]["VAL"] - per[(a, s)]["TEST"] for s in D.SEEDS[a]]
        m_ = mean(ds)
        dss += sum((x - m_) ** 2 for x in ds)
        ddf += len(ds) - 1
    s_delta = math.sqrt(dss / ddf) if ddf else 0.0

    P("\nARMS  plateau5 over epochs %d-%d, TEST and VAL from the SAME runs; box from each run's OWN probe.jsonl" % (PL_LO, PL_HI))
    P("  %-6s %-4s %-9s %2s %9s %7s %7s %9s %7s %7s %9s  %-5s" % ("arm", "wd", "grain", "n", "TEST", "sd", "range",
      "VAL", "sd", "range", "VAL-TEST", "box"))
    for a in D.ARMS:
        t, v = AR["TEST"][a], AR["VAL"][a]
        P("  %-6s %-4s %-9s %2d %9.4f %7.3f %7.3f %9.4f %7.3f %7.3f %+9.4f  %-5s" % (a, D.WD[a], D.SPEC[a], t["n"], t["level"],
          t["sd"], t["range"], v["level"], v["sd"], v["range"], v["level"] - t["level"], "free" if t["boxfree"] else "BOUND"))

    P("\nSIGMA  O2: max(frozen floor, in-batch) per reader -- nothing from the corpus")
    for X in READERS:

    P("  paired (VAL - TEST) per-run sd %.6f (df %d) -- DESCRIPTIVE gaps only" % (s_delta, ddf))

    prim, sel, st, KT, KV = decide(AR["TEST"], AR["VAL"], sig["TEST"][0], sig["VAL"][0], s_delta)
    P("\nCONTRASTS (WITHIN a rung, WITHIN a reader; +/-2 SE is a HALF-WIDTH, not a bound on any effect)")

        for X, K in (("TEST", KT), ("VAL", KV)):
            d, se = K["D_" + r]

        g = KT["GAPS"][r]
        P("       GAP D_%s VAL - TEST = %+.4f pp = %+.2f paired SE (SE %.6f) [%+.4f, %+.4f]  %s"

    for X, K in (("TEST", KT), ("VAL", KV)):
        P("  %-4s T_W4 = k01 - ch %+.4f (%+.2f SE), k01 - nd %+.4f (%+.2f SE)  state %s" % (X, K["Tch_W4"][0],
          K["Tch_W4"][0] / K["Tch_W4"][1], K["Tnd_W4"][0], K["Tnd_W4"][0] / K["Tnd_W4"][1], st[X]["SC"]))
        for r in D.RUNG_IDS:
            P("  %-4s G_%s = kL - k01 = %+.4f pp   DESCRIPTIVE" % (X, r, -K["TkL_" + r][0]))
    tv = KT["GAPS"]["SC"]
    P("       GAP min(T_W4) VAL - TEST = %+.4f pp = %+.2f paired SE [%+.4f, %+.4f]  %s" % (tv, tv / sedd if sedd else 0.0,
      tv - RES * sedd, tv + RES * sedd, "RESOLVED" if KT["GAP_RES"]["SC"] else "not resolved"))
    tok, pt, pv, reg = sel
    P("\nSELECTION  over the 8 configurations: TEST picks %s (%.4f), VAL picks %s (%.4f); TEST regret of VAL's pick %+.4f pp"
      " (SE %.6f)  -> %s" % (pt, AR["TEST"][pt]["level"], pv, AR["VAL"][pv]["level"], reg, sig["TEST"][0] * math.sqrt(0.5), tok))
    stamps = list(BOUNDS)
    for r in D.RUNG_IDS:
        gt = max(D.GRAIN_IDS, key=lambda g: AR["TEST"][g + r]["level"])
        gv = max(D.GRAIN_IDS, key=lambda g: AR["VAL"][g + r]["level"])
        P("  rung %s: best grain on TEST %s, on VAL %s" % (r, D.SPEC_OF_GRAIN[gt], D.SPEC_OF_GRAIN[gv]))
        stamps.append("SELECT-%s-%s" % (r, "SAME" if gt == gv else "DIFFERS"))
    for X in READERS:
        stamps.append("%s-%s" % (X, "SIGMA-PRIOR-FROZEN" if sig[X][1] == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH"))
    for c in CLAIMS:
        if KT["GAP_RES"][c]:
            stamps.append("GAP-RESOLVED-%s" % c)
    # non-gating: the in-batch TEST states against cgw1's landed (50,000-image) ones
    for c in CLAIMS:
        if st["TEST"][c] != D.CGW1_STATES["SCALAR" if c == "SC" else c]:
            stamps.append("TEST-STATE-DIFFERS-FROM-CGW1-%s" % c)
    for a in D.ARMS:
        if abs(AR["TEST"][a]["level"] - D.CGW1_TEST[a]) > MATCH:
            stamps.append("TEST-LEVEL-DIFFERS-FROM-CGW1-%s" % a)
    for a in D.ARMS:
        if not AR["TEST"][a]["boxfree"] and D.GRAIN_OF[a] in ("k01", "kL"):
            stamps.append("BOX-BOUND-%s" % a)
    stateword = "+".join("%s-T%s/V%s" % (c, st["TEST"][c], st["VAL"][c]) for c in CLAIMS)
    P("\nPER CLAIM  " + "  ".join("%s: TEST %s | VAL %s -> %s" % (c, st["TEST"][c], st["VAL"][c],
                                  agreement(st["TEST"][c], st["VAL"][c])) for c in CLAIMS))
    P("\nLICENCE (primary)    %s" % LICENCE.get(prim, ""))
    P("LICENCE (selection)  %s" % LICENCE[tok])
    P("\nFINAL %s | %s | %s | %s" % (prim, tok, stateword, " ".join(stamps)))
    return 0

RESULTS = {"PASS": 0, "FAIL": 0, "SKIP": 0}


def chk(cond, label):
    RESULTS["PASS" if cond else "FAIL"] += 1
    print("  %s %s" % ("PASS" if cond else "FAIL", label))
    return cond


def skip(label):
    RESULTS["SKIP"] += 1
    print("  SKIP %s" % label)


def _write_out(d, arm, seed, jid, level, vlevel, save_dir, args_over=None, env_over=None, extra_line=None, done=True,
               n_ep=D.EPOCHS, vs_over=None, val_over=None, gpu="NVIDIA L4, 23034 MiB"):
    rn = D.run_name(arm, seed)
    pairs = D.expected_args(arm, seed, save_dir)
    if args_over:
        pairs = args_over(pairs)
    env = D.ENV_TEMPLATE.replace("<P>", os.path.join(save_dir, D.probe_dir_name(arm, seed)))
    if env_over:
        env = env_over(env)
    L = ["NODE=[HOST_2f2f29d9e1] | JOB=%s %d | AUGMENT=1" % (rn, jid), "ARGS: " + " ".join("--%s %s" % kv for kv in pairs), env, gpu, ""]
    L += [D.witness_on()] if vs_over is None else vs_over
    if extra_line:
        L.append(extra_line)
    for e in range(n_ep):
        te = level if e >= PL_LO - 5 else level * (1 - math.exp(-(e + 1) / 12.0))
        ve = vlevel if e >= PL_LO - 5 else vlevel * (1 - math.exp(-(e + 1) / 12.0))
        L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.99, te + 5.0), te))
        vl = "VAL: epoch %d val_acc %.2f %% n_val 5000" % (e, round(ve * 50) / 50.0)
        if val_over:
            vl = val_over(e, vl)
        if vl is not None:
            L.append(vl)
    L.append("45  minutes")
    if done:
        L.append("RUN_DONE")
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(L) + "\n")


def _write_probe(pdir, m, lo=-8.0, hi=-5.0, n=D.N_RECORDS, nb=None, hi_frac=0.0):
    os.makedirs(pdir, exist_ok=True)
    with open(os.path.join(pdir, "probe.jsonl"), "w") as fh:
        for i in range(n):
            bmax = D.HI if i >= n * (1 - hi_frac) else hi + 1e-4 * (i % 7)
            fh.write('{"step": %d, "beta_true_min": %.6f, "beta_true_max": %.6f, "n_beta": %d, "n_at_lo": 0, '
                     '"n_at_hi": %d}\n' % (i * D.PROBE, lo - 1e-4 * (i % 5), bmax, nb if nb is not None else m,
                                           1 if bmax >= D.HI else 0))


JIT = (0.02, 0.04, -0.04, -0.02)    # zero-mean over 4 seeds


def _build(root, template, levels, vlevels=None, save_dir="/SAVE/runs/cvl1", over=None, drop=(), extra=None,
           probe_over=None, prov_over=None, jit=None, vjit=None):
    os.makedirs(os.path.join(root, PREFIX), exist_ok=True)
    vlevels = vlevels or levels
    jid = 7100000
    for (arm, seed) in D.RUNS:
        jid += 1
        if (arm, seed) in drop:
            continue
        i = D.SEEDS[arm].index(seed)
        lv = levels[arm] + (JIT[i] if jit is None else jit(arm, i))
        vv = vlevels[arm] + (JIT[i] if vjit is None else vjit(arm, i))
        _write_out(os.path.join(root, PREFIX), arm, seed, jid, lv, vv, save_dir, **dict((over or {}).get((arm, seed), {})))
        dst = os.path.join(root, PREFIX, D.probe_dir_name(arm, seed))
        if probe_over and (arm, seed) in probe_over:
            _write_probe(dst, D.M_OF_ARM[arm], **probe_over[(arm, seed)])
        else:
            os.symlink(os.path.join(template, D.probe_dir_name(arm, seed)), dst)
    for f in (extra or []):
        f(os.path.join(root, PREFIX))
    kv = {"MODE": "submit", "SCORER_SHA256": file_sha(os.path.abspath(__file__)),
          "DESIGN_SHA256": file_sha(os.path.join(HERE, "cvl1_design.py")), "HF_SHA256": D.HF_LIVE_SHA,
          "LOAD_DATA_SHA256": D.LOADDATA_POST_SHA, "TRAIN_SHA256": D.TRAIN_POST_SHA}
    kv.update(prov_over or {})
    open(os.path.join(root, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % x for x in kv.items()))
    return root


def _run(root, csvpath=None):
    lines = []
    rc = score(root, csvpath, out=lines.append)
    lines = [x for ln in lines for x in ln.split("\n")]
    fin = [ln for ln in lines if ln.startswith("FINAL ")]
    return rc, (fin[0] if fin else ""), lines


def _levels(w1=(92.20, 91.70, 91.90, 92.50), w4=(87.60, 87.35, 90.20, 89.60)):
    L = {}
    for r, v in (("W1", w1), ("W4", w4)):
        for g, x in zip(D.GRAIN_IDS, v):
            L[g + r] = x
    return L


def selftest(csvpath, runsdir):
    print("cvl1 SELFTEST -- design, frozen floor through corpus_exclusions.filter_rows, decision tables with exhaustive")
    print("reachability, and the REAL score() on synthetic runs (every account, every gate, the harness breaks).")
    src = open(os.path.abspath(__file__)).read()
    print("\nA  DESIGN")
    chk(D.NJOBS == 32 and len(D.ARMS) == 8, "A 8 arms, 32 runs (4 grains x 2 rungs x 4 seeds)")
    chk(all(set(D.SEEDS[a]) == {184, 185, 186, 187} for a in D.ARMS) and set(D.UNUSED_SEEDS) == {188, 189, 190, 191}
        and set(D.SEED_BLOCK) == set(range(184, 192)), "A seeds {184..187} on every arm, from the assigned block {184..191}")

    chk(D.M_OF_GRAIN == {"ch": 14421, "nd": 14420, "k01": 1, "kL": 62}, "A registered group counts 14421/14420/1/62")
    chk(D.VAL_SPLIT == "5000:302" and D.N_TRAIN == 45000 and D.STEPS_PER_EPOCH == 450 and D.N_RECORDS == 9000,
        "A VAL_SPLIT 5000:302 -> 45,000 train, 450 steps / epoch, 9,000 probe records")
    chk(re.match(r"^[0-9a-f]{64}$", D.VAL_SHA or "") is not None and re.match(r"^[0-9a-f]{64}$", D.TRAIN_SPLIT_SHA or "") is not None,
        "A the design pins the REAL split's val_sha / train_sha (from the proof job, 302)")
    chk(all(len(D.expected_args(a, s, "/x")) == 20 for a, s in D.RUNS), "A every expected ARGS line has 20 flags")
    chk(abs(SIGMA_PRIOR * math.sqrt(0.5) - SE4) < 1e-6, "A SE4 = SIGMA_PRIOR * sqrt(2/4)")
    chk((SURV, NULLBAR, POOL_EXCLUDE, RES, HEALTH_MIN, DIVERGED, BOXFREE_MAX, MATCH) == (0.30, 0.15, 0.5556, 2.0, 85.0, 2.0, 0.05, 0.50),
        "A the frozen bars are cgw1's: SURV 0.30 NULL 0.15 POOL 0.5556 RES 2 HEALTH 85 DIVERGED 2 BOX 0.05 MATCH 0.50")
    dec_src = src[src.index("def rung_state("):src.index("# READING THE RUNS")]
    chk("CGW1" not in dec_src and "corpus" not in dec_src and "PRED" not in dec_src,
        "A no landed level, prediction or corpus figure enters the decision functions (O2, read from source)")
    sc_src = src[src.index("def score("):src.index("# SELFTEST")]
    chk(sc_src.count("CGW1_TEST") == 1 and sc_src.count("CGW1_STATES") == 1, "A cgw1's landed values are read ONCE each, for stamps only")
    chk(all(t in LICENCE for t in PRIMARY_TOKENS + SELECT_TOKENS), "A every primary / selection token has a licence")
    chk(WIN_STEP == 42750, "A plateau window starts at meta step 95 x 450 = 42,750")

    print("\nB  THE FROZEN FLOOR, RE-DERIVED THROUGH corpus_exclusions.filter_rows")
    if csvpath and os.path.exists(csvpath):
        sys.path.insert(0, HERE)
        import corpus_exclusions as ce
        sys.path.pop(0)
        rows = [r for r in csv.DictReader(open(csvpath)) if not r["run"].startswith(PREFIX + "-")]
        CK = ["network", "dataset", "granularity", "base", "meta", "meta_stepsize", "alpha0", "gamma", "augment",
              "beta_clip", "batch_size", "epochs_requested", "hier", "lam", "eta_ratio"]

        def pooled(R):
            cells = {}
            for r in R:
                if (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
                        and (r.get("plateau5") or "").strip() and r["network"] == D.NET and r["dataset"] == D.DSET
                        and r["meta_stepsize"] == D.MST and r["alpha0"] in (D.A0, "0.001") and r["beta_clip"] == D.CLIP
                        and r["epochs_requested"] == str(D.EPOCHS) and r["augment"] == D.AUG and r["batch_size"] == str(D.BATCH)):
                    cells.setdefault(tuple(r[k] for k in CK), []).append(float(r["plateau5"]))
            s2, df_, nc = 0.0, 0, 0
            for v in cells.values():
                if len(v) > 1:
                    mm = mean(v)
                    s2 += sum((x - mm) ** 2 for x in v)
                    df_ += len(v) - 1
                    nc += 1
            return math.sqrt(s2 / df_), df_, nc
        filt = ce.filter_rows(rows)
        sF, dF, nF = pooled(filt)
        sN, _dN, _ = pooled(rows)
        chk(abs(sF - SIGMA_PRIOR) < 1e-12 and dF == SIGMA_PRIOR_DF and nF == 34,
            "B SIGMA_PRIOR: the FROZEN literal %r IS the FILTERED value (df %d, %d cells) on this corpus (%d rows)"
            % (SIGMA_PRIOR, dF, nF, len(rows)))
        chk(abs(sN - SIGMA_PRIOR_NAIVE) < 1e-12 and sN > 3 * sF,
            "B the UNFILTERED value %.6f differs (cgw1's wd-deviating rows) -- the filter is load-bearing; disclosure" % sN)
        dm = ce._pooled(filt, "ResNet18_c100")
        print("  NOTE the corpus_exclusions noise-floor DEMONSTRATION (SIGMA_R18ALL, filtered) on this corpus is %.6f (df %d);"
              " quoted at registration as %.6f -- no bar reads it" % (dm[0], dm[1], DEMO_R18ALL_AT_REG))
    else:
        skip("B no corpus at %s" % csvpath)

    print("\nC  THE DECISION, AT ITS BARS AND EXHAUSTIVELY")
    se = SE4
    chk(rung_state(0.30 + 1e-9, se, False, True, True) == "SURVIVES" and rung_state(0.30 - 1e-9, se, False, True, True) == "UNDECIDED",
        "C the SURVIVES edge +0.30 (cgw1's)")
    chk(rung_state(0.15, se, False, True, True) == "VANISHES" and rung_state(0.15 + 1e-9, se, False, True, True) == "UNDECIDED",
        "C the VANISHES edge +0.15")
    chk(rung_state(0.10, 0.2276, False, True, True) == "VANISHES" and rung_state(0.10, 0.2279, False, True, True) == "UNDECIDED",
        "C VANISHES needs D + 2 SE < +0.5556 (a noisy reader cannot print VANISHES)")
