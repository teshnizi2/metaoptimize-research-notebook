set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
# CYCLE 9. NAIL THE ADDITIVE PEAK LOCATION.
# ad5_sweep.sh was designed when the peak looked like a tie between r=0.03 and r=0.1.
# With those two now at n=5 the picture is different -- the new r=0.05 cell is ABOVE
# BOTH of them and the ladder has a single clear maximum:
#   r=0    92.52 +-0.11 (n=2)   r=0.03 92.96 +-0.15 (n=5)
#   r=0.05 93.41 +-0.06 (n=2)   r=0.1  92.86 +-0.19 (n=5)
#   r=0.2  91.65      (n=1)     r=0.3  91.44 +-0.02 (n=2)
# r=0.05 sits +0.45pp over r=0.03 and +0.55pp over r=0.1 against sd ~0.15, so the peak
# has MOVED off both cells the last cycle was choosing between -- but it is n=2.
# (a) take r=0.05 to n=5 (s2 already running -> s3,s4 here)
# (b) add r=0.07 at n=3 to test whether the maximum is smooth or a spike between
#     0.05 and 0.1. A spike would be an artefact; a smooth cap is the operator's optimum.
# Same base config as ad5_sweep.sh so these are drop-in cells of the SAME ladder:
# alpha0=1e-6, 100 epochs, SGDm+Lion, guard on, augment on, layerwise.
# --time from measurement: 0.30-0.31 min/ep on L4 -> 100ep = 31 min. 01:15:00 = 2.4x.
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ sbatch --parsable --job-name=$1-s$3 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 --time=01:15:00 \
  --export="ALL,AUGMENT=1,$C,HIER=additive,ETA_RATIO=$2" \
  jobs/run_cifar.sh $B --stepsize-groups layerwise --seed $3 \
  --save-directory $R/addit --run-name $1_s$3 >/dev/null; }
# (a) peak cell to n=5
sub ad-l-r005 0.05 3
sub ad-l-r005 0.05 4
# (b) bracket the maximum
for s in 0 1 2; do sub ad-l-r007 0.07 $s; done
sleep 4
echo "AD7_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -cE '^ad-l-(r005-s[34]|r007)')"
