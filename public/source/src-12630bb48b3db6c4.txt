#!/bin/bash
# =============================================================================
# cVT6_complementpath.sh -- `cvt6`: DOES 50's LARGE STEP SIZE STALL THE NETWORK DIRECTLY, OR THROUGH THE COMPLEMENT's
#   EARLY COLLAPSE?  ONE SUBMISSION, 21 JOBS (7 arms x 3 seeds), 100 EPOCHS, SEEDS {96,97,98}, PlainNet18_c100 /
#   CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.  CORRECTIONS 242.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here touches the other account;
# every guard and scancel line is filtered on `cvt6-`.
#
# REGISTERED SCORER: analysis/cVT6_complementpath_score.py, committed BEFORE this batch is submitted (RULE 21).
# Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT6_complementpath_score.py <runsdir>
#
# NEW HARNESS CODE, ISOLATED AND OPT-IN.  The batch runs from $WS/harness_cvt6/cifar10 = cvt4's tree (unchanged,
# re-verified by bin/cVT4_stage_harness.sh) + PATCH_COMPHOLD (patches/patch_comphold.py) in HF.py + the replay file
# patches/cvt6_headpath.json, built and verified by bin/cVT6_stage_harness.sh, and the runner
# $WS/jobs/run_cifar_cvt6.sh (run_cifar.sh with the cd line changed).  COMP_HOLD unset == bitwise cvt4's tree; guard
# 4f requires the inertness / bite log (tests/test_comphold.py + tests/test_comphold_realrun.py,
# bin/cVT6_realrun_inertness.sbatch) on THIS HF.py and THIS scorer.
#
# THE ARMS.  PlainNet indices (53 tensors); guard 4 re-derives every one on the LIVE model by NAME, constructs each
# arm's HF with its BETA_HOLD and COMP_HOLD, checks both witnesses, and writes the manifest.
#   k01           scalar                                BETA_HOLD unset                      COMP_HOLD unset
#   HEAD          sets:1-49,51-53/layer4.1.bn2.weight   BETA_HOLD unset                      COMP_HOLD unset
#   HOLDLOW       HEAD's   BETA_HOLD=layer4.1.bn2.weight:floor                               COMP_HOLD unset
#   HOLDHIGH      HEAD's   BETA_HOLD=layer4.1.bn2.weight:tri:9428                            COMP_HOLD unset
#   HIGHHEADPATH  HEAD's   BETA_HOLD=layer4.1.bn2.weight:tri:9428                            COMP_HOLD=rec:cvt6_headpath
#   LOWMUTEPATH   HEAD's   BETA_HOLD=layer4.1.bn2.weight:floor                               COMP_HOLD=tri:9428
#   LOWHEADPATH   HEAD's   BETA_HOLD=layer4.1.bn2.weight:floor                               COMP_HOLD=rec:cvt6_headpath
#
# RULE 20.  Every ARGS value is ONE shell token; every held arm's ARGS line is HEAD's except the run name.
# BETA_HOLD and COMP_HOLD ride sbatch's --export list (value class [A-Za-z0-9_.:], no comma) and CANNOT ride the
# runner's ENV line: the patches print `BETA_HOLD: ...` and `COMP_HOLD: ...` at construction on EVERY run, and those
# lines are what the separate ENV audit and the scorer's G-HOLDW / G-CHOLDW gates read.  analysis/argsline_guard.py
# IS NOT EDITED.
#
# COST.  cvt4 by sacct: 0.66 GPU-h per PlainNet run off [HOST_3c1b003035] (median 2,381 s), 0.83 including [HOST_3c1b003035].
# 21 jobs ~= 13.9 GPU-h expected (17.4 at 0.83); HARD BOUND = WALL x 21 = 63 GPU-h (WALL 03:00:00; cvt4's slowest run
# was 5,334 s on [HOST_3c1b003035]).
#
# USAGE   bash bin/cVT6_complementpath.sh               # dry run: every guard, every composed line printed
#         bash bin/cVT6_complementpath.sh --dry-run     #   submits nothing
#         bash bin/cVT6_complementpath.sh --submit      # the ONLY submitting form
# ENV     CVT6_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# no arm may inherit a VOTE_W, BETA_HOLD or COMP_HOLD through --export=ALL
unset VOTE_W BETA_HOLD COMP_HOLD
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cvt6/cifar10
HARN4=$WS/harness_cvt4/cifar10
HARN1=$WS/harness_cvt1/cifar10
RUNNER=$WS/jobs/run_cifar_cvt6.sh
SAVE=$WS/runs/cvt6

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true
export PYTHONDONTWRITEBYTECODE=1

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=PlainNet18_c100
SEEDS="96 97 98"
MST=1e-3
A0=1e-6
NTENS=53
TOTPAR=11046308
G_K01='scalar'
G_HEAD='sets:1-49,51-53/layer4.1.bn2.weight'
B_LOW='layer4.1.bn2.weight:floor'
B_HIGH='layer4.1.bn2.weight:tri:9428'
C_HEADPATH='rec:cvt6_headpath'
C_MUTEPATH='tri:9428'
ARMS="k01 HEAD HOLDLOW HOLDHIGH HIGHHEADPATH LOWMUTEPATH LOWHEADPATH"
NJOBS=21
WALL=03:00:00
POST_SHA=e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_CVT1_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_PRE_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
HF_POST_SHA=f1b9c8aa87953c2fd3f64145126591608d332812183911a39ce9f525101edd2e
REPLAY_SHA=74be71fa524ad0122d1408e01dd2b633b004b2e27228393fe6e593f494358a5d
RUNNER_SHA=9aa2ed5c15b8ecd8c22d5137ab4be169b97f9556ebe67b8460c4971d4f84783f
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT6_complementpath_score.py"
SELF="$REPO/bin/cVT6_complementpath.sh"
DERIV="$REPO/analysis/cvt6_registration_derivations.py"
DERIV4="$REPO/analysis/cvt4_registration_derivations.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cVT6_realrun_inertness.sbatch"
TESTC="$REPO/tests/test_comphold.py"
TESTCR="$REPO/tests/test_comphold_realrun.py"
STAGE6="$REPO/bin/cVT6_stage_harness.sh"
STAGE4="$REPO/bin/cVT4_stage_harness.sh"
STAGE1="$REPO/bin/cVT1_stage_harness.sh"
PATCHC="$REPO/patches/patch_comphold.py"
REPLAY="$REPO/patches/cvt6_headpath.json"
TESTDOWN="$REPO/patches/cvt6_testdown.json"
PATCHB="$REPO/patches/patch_betahold.py"
PATCHV="$REPO/patches/patch_voteweight.py"
PATCHF="$REPO/patches/patch_plainnet.py"
TESTB="$REPO/tests/test_betahold.py"
TESTR="$REPO/tests/test_voteweight_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_plainnet.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFCH="$HARN/Optimizers/HF.py.pre_comphold"
HFBH="$HARN/Optimizers/HF.py.pre_betahold"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
RPH="$HARN/Optimizers/cvt6_headpath.json"
BNH="$HARN/build_network.py"
ILOG="$SAVE/inertness_comphold.log"

echo "=============================================================="
echo " cvt6 -- does 50's large step size stall directly, or via the complement's collapse?  (PATCH_COMPHOLD)"
echo " $DSET / $NET (PlainNet indices)   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01           $G_K01   BETA_HOLD unset              COMP_HOLD unset"
echo "   HEAD          $G_HEAD   BETA_HOLD unset   COMP_HOLD unset"
echo "   HOLDLOW       HEAD's   BETA_HOLD=$B_LOW      COMP_HOLD unset (free Lion)"
echo "   HOLDHIGH      HEAD's   BETA_HOLD=$B_HIGH   COMP_HOLD unset (free Lion)"
echo "   HIGHHEADPATH  HEAD's   BETA_HOLD=$B_HIGH   COMP_HOLD=$C_HEADPATH"
echo "   LOWMUTEPATH   HEAD's   BETA_HOLD=$B_LOW      COMP_HOLD=$C_MUTEPATH"
echo "   LOWHEADPATH   HEAD's   BETA_HOLD=$B_LOW      COMP_HOLD=$C_HEADPATH"
echo " 7 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cvt4's + PATCH_COMPHOLD)   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " CO-PRIMARY: P_50 = LOWHEADPATH - HIGHHEADPATH; P_C = LOWHEADPATH - LOWMUTEPATH"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT6_complementpath_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${COMP_HOLD+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD and COMP_HOLD are UNSET in the launcher's environment (no arm can inherit them)" \
  || guard_fail "guard 0b: VOTE_W, BETA_HOLD or COMP_HOLD is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$SELF" "$DERIV" "$DERIV4" "$EXCL" "$EXCLT" "$RRSB" "$TESTC" "$TESTCR" "$STAGE6" "$STAGE4" "$STAGE1" "$PATCHC" \
         "$REPLAY" "$TESTDOWN" "$PATCHB" "$PATCHV" "$PATCHF" "$TESTB" "$TESTR" "$TESTT" "$TESTF" "$LIBF" "$AGF" "$CSVP" \
         "$RUNNER" "$HFH" "$HFCH" "$HFBH" "$HFVW" "$HFPRE" "$RPH" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cVT6_complementpath_score.py bin/cVT6_complementpath.sh analysis/cvt6_registration_derivations.py bin/cVT6_realrun_inertness.sbatch tests/test_comphold.py tests/test_comphold_realrun.py patches/patch_comphold.py patches/cvt6_headpath.json patches/cvt6_testdown.json bin/cVT6_stage_harness.sh"
REUSEF="analysis/cvt4_registration_derivations.py analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv tests/test_voteweight_realrun.py tests/test_betahold.py tests/test_probe_tensor.py tests/test_plainnet.py patches/patch_betahold.py patches/patch_voteweight.py patches/patch_plainnet.py bin/cVT4_stage_harness.sh bin/cVT1_stage_harness.sh bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the ten registration files and fourteen reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT6_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT6_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT6_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT6_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING O2, RULE P / RULE H, the real-record nulls and the invocation ----
SELFT=/tmp/cvt6_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
grep -qE "^  PASS O2 DIRECT +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt6's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT6_complementpath_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 96, 97 or 98" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cVT4's" "$SELFT" && grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cVT1's" "$SELFT" \
  && grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cPL2's" "$SELFT" && grep -q "^  PASS k01's spec is BYTE-IDENTICAL to cVT1's" "$SELFT" \
  && echo "guard 1c5: HEAD is byte-identical to cvt4's, cvt1's and cpl2's HEAD; k01 to cvt1's" || guard_fail "guard 1c5: HEAD / k01 not proved byte-identical"
grep -q "^  PASS B RULE P re-run on THIS host's landed records == P_HIGH 9428" "$SELFT" \
  && grep -q "^  PASS B RULE H re-run on THIS host's landed records reproduces the replay file BYTE FOR BYTE" "$SELFT" \
  && echo "guard 1c6: RULE P (9428, MUTEPATH) and RULE H (the replay file, byte for byte) re-derived from the landed records ON THIS HOST" \
  || guard_fail "guard 1c6: RULE P / RULE H were not re-derived on this host"
grep -q "^  PASS B G-BITE's Lion half PASSES on all 24 landed" "$SELFT" && grep -q "^  PASS B the 6 landed cvt4 HOLDLOW / HOLDHIGH runs PASS" "$SELFT" \
  && grep -q "^  PASS B THE BROKEN-COMP_HOLD NULL ON REAL RECORDS" "$SELFT" && grep -q "^  PASS B non-vacuity" "$SELFT" \
  && grep -q "^  PASS B every landed HEAD run read as each of the 5 held arms FAILS" "$SELFT" \
  && echo "guard 1c7: G-BITE passes 24 landed controls and 6 landed free twins, FAILS the 9 real-record BROKEN-COMP_HOLD nulls and 60 HEAD nulls on this host" \
  || guard_fail "guard 1c7: the real-record G-BITE checks did not pass"
grep -q "^  PASS SIGMA_PLAIN: the FROZEN literal 0.44614812708363655 IS the FILTERED value" "$SELFT" \
  && grep -q "^  PASS the NAIVE PlainNet floor is" "$SELFT" \
  && echo "guard 1c8: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value, and NOT the naive one" \
  || guard_fail "guard 1c8: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E BROKEN COMP_HOLD: every forced complement UNHELD" "$SELFT" && grep -q "^  PASS E COMP HOLD ONLY AT PROBE STEPS" "$SELFT" \
  && grep -q "^  PASS E WRONG COMPLEMENT PATH: HIGHHEADPATH" "$SELFT" && grep -q "^  PASS E FREE COMPLEMENT TOUCHED" "$SELFT" \
  && echo "guard 1c9: the broken-COMP_HOLD null, a probe-step-only hold, a wrong complement path and a touched free complement reach PATCH-NOT-VERIFIED" \
  || guard_fail "guard 1c9: G-BITE break cases did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 1d: the derivation reproduces RULE H, RULE P and the floor on THIS host ----------
DOUT=/tmp/cvt6_deriv.$$
python3 "$DERIV" "$WS/runs" "$CSVP" >"$DOUT" 2>&1 \
  && grep -q "^  REPLAY FILE cvt6_headpath.json  10828 bytes  sha256 $REPLAY_SHA$" "$DOUT" \
  && grep -q "^  RULE P on this host: P_HIGH 9428" "$DOUT" \
  && grep -q "SIGMA_PLAIN   FILTERED 0.44614812708363655" "$DOUT" \
  && echo "guard 1d: analysis/cvt6_registration_derivations.py reproduces RULE H, RULE P and the filtered floor on this host (output sha $(sha "$DOUT"))" \
  || { guard_fail "guard 1d: the derivation did not reproduce RULE H / RULE P / the floor"; tail -5 "$DOUT"; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt6-' "$CSVP" && guard_fail "guard 2: cvt6-* already in the CSV" || echo "guard 2: no cvt6-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
gn = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], gn))
raise SystemExit(1 if hit or not set(gn) <= {"cpl1", "cpl2", "cvt1", "cvt2", "cvt3", "cvt4", "cvt5"} else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt6-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt6-*.out on disk" || guard_fail "guard 2b: $NOUT cvt6-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (96|97|98)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 96/97/98" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 96/97/98"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt6-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt6-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt6-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt6-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt6-* in squeue" || guard_fail "guard 2f: $NQ cvt6-* already queued"
fi

# ---- guard 3/4a: cvt1's and cvt4's trees unchanged; cvt6's tree consistent and sees CIFAR-100 -----------
bash "$STAGE1" >/tmp/cvt6_stage1.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT1_stage_harness.sh (verify mode) VERIFIES cvt1's tree (unchanged):"; grep -E 'V-h2|V-b2|VERIFIED' /tmp/cvt6_stage1.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt1's tree is NOT consistent"; tail -20 /tmp/cvt6_stage1.$$; }
bash "$STAGE4" >/tmp/cvt6_stage4.$$ 2>&1 \
  && { echo "guard 4a1: bin/cVT4_stage_harness.sh (verify mode) VERIFIES cvt4's tree (the base, unchanged):"; grep -E 'V-h2|VERIFIED' /tmp/cvt6_stage4.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a1: cvt4's tree is NOT consistent"; tail -20 /tmp/cvt6_stage4.$$; }
bash "$STAGE6" >/tmp/cvt6_stage6.$$ 2>&1 \
  && { echo "guard 4a2: bin/cVT6_stage_harness.sh (verify mode) VERIFIES cvt6's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt6_stage6.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a2: cvt6's tree is NOT consistent"; tail -20 /tmp/cvt6_stage6.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "$POST_SHA" ] && echo "guard 3b: build_network.py == cpl1's POST_SHA e65e6773...a18b" \
  || guard_fail "guard 3b: build_network.py is not cpl1's POST_SHA"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFCH")" = "$HF_PRE_SHA" ] && [ "$(sha "$HARN4/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && [ "$(sha "$HFBH")" = "$HF_CVT1_SHA" ] && [ "$(sha "$HARN1/Optimizers/HF.py")" = "$HF_CVT1_SHA" ] \
  && [ "$(sha "$HFVW")" = "$HF_LIVE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA f1b9c8aa...; .pre_comphold == cvt4's HF.py == 84b345ad...; .pre_betahold == cvt1's == 3f2b98e1...; .pre_voteweight == the LIVE HF.py == 4732b74a...; .pre_probe_tensor == live" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or cvt4's / cvt1's / the live HF.py moved)"
[ "$(sha "$RPH")" = "$REPLAY_SHA" ] && [ "$(sha "$REPLAY")" = "$REPLAY_SHA" ] \
  && echo "guard 3f: the tree's replay file == the repo's == 74be71fa... (RULE H)" || guard_fail "guard 3f: replay file sha is not the registered one"

DESIGN=(alg-base=SGDm alg-meta=Lion num-epochs=$EPOCHS batch-size=$BATCH \
        gamma=$GAMMA dataset=$DSET NN-name=$NET weight-decay-meta=0 \
        meta-stepsize=$MST alpha0=$A0)
N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    case $ARM in
      k01)          G=$G_K01;  WANTB="";       WANTC="" ;;
      HEAD)         G=$G_HEAD; WANTB="";       WANTC="" ;;
      HOLDLOW)      G=$G_HEAD; WANTB="$B_LOW";  WANTC="" ;;
      HOLDHIGH)     G=$G_HEAD; WANTB="$B_HIGH"; WANTC="" ;;
      HIGHHEADPATH) G=$G_HEAD; WANTB="$B_HIGH"; WANTC="$C_HEADPATH" ;;
      LOWMUTEPATH)  G=$G_HEAD; WANTB="$B_LOW";  WANTC="$C_MUTEPATH" ;;
      LOWHEADPATH)  G=$G_HEAD; WANTB="$B_LOW";  WANTC="$C_HEADPATH" ;;
    esac
    BH=""; [ -n "$WANTB" ] && BH=",BETA_HOLD=$WANTB"
    CH=""; [ -n "$WANTC" ] && CH=",COMP_HOLD=$WANTC"
    RN="cvt6-$ARM-s$S"
    PDIR="$SAVE/probe_$RN"
    CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP_C,HIER=none,SCHED=none,PROBE=$PROBE,PROBE_TENSOR=$PT,PROBE_DIR=$PDIR$BH$CH \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
--gamma $GAMMA --meta-stepsize $MST --alpha0 $A0 --num-epochs $EPOCHS \
--stepsize-groups $G --seed $S --save-directory $SAVE --run-name $RN"
    guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1 \
      || { echo "!!! RULE 20 PRE-CHECK FAILED for $RN"; guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1)); }
    guard_presubmit "$CMD" "stepsize-groups=$G" "seed=$S" "run-name=$RN" >/dev/null 2>&1 \
      || { echo "!!! RULE 20: an axis does not round-trip for $RN"; guard_presubmit "$CMD" "stepsize-groups=$G" "seed=$S" "run-name=$RN"; BAD=$((BAD+1)); }
    GOTB=$(printf '%s\n' "$CMD" | grep -o 'BETA_HOLD=[^ ,]*')
    GOTC=$(printf '%s\n' "$CMD" | grep -o 'COMP_HOLD=[^ ,]*')
    NBH=$(printf '%s\n' "$GOTB" | grep -c 'BETA_HOLD=')
    NCH=$(printf '%s\n' "$GOTC" | grep -c 'COMP_HOLD=')
    NVW=$(printf '%s\n' "$CMD" | grep -c 'VOTE_W=')
    NTD=$(printf '%s\n' "$CMD" | grep -c 'testdown\|tri:100')
    [ "$NVW" = 0 ] && [ "$NTD" = 0 ] || { echo "!!! $RN carries a VOTE_W or a test-only hold"; BAD=$((BAD+1)); }
    if [ -z "$WANTB" ]; then
      [ "$NBH" = 0 ] || { echo "!!! $RN carries a BETA_HOLD"; BAD=$((BAD+1)); }
    else
      [ "$NBH" = 1 ] && [ "$GOTB" = "BETA_HOLD=$WANTB" ] || { echo "!!! $RN carries $NBH BETA_HOLD or not its own string"; BAD=$((BAD+1)); }
    fi
    if [ -z "$WANTC" ]; then
      [ "$NCH" = 0 ] || { echo "!!! $RN carries a COMP_HOLD"; BAD=$((BAD+1)); }
    else
      [ "$NCH" = 1 ] && [ "$GOTC" = "COMP_HOLD=$WANTC" ] || { echo "!!! $RN carries $NCH COMP_HOLD or not its own string"; BAD=$((BAD+1)); }
    fi
    if [ "$SUBMIT" = 1 ]; then
      mkdir -p "$PDIR"
      eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
    else
      echo "$CMD"; echo; N=$((N+1))
    fi
  done
done
echo "guard 6: $N composed command lines, $BAD failed the RULE 20 / BETA_HOLD / COMP_HOLD pre-check"
[ "$BAD" = 0 ] || { echo "!!! composed lines are malformed"; exit 2; }
[ "$N" = "$NJOBS" ] || { echo "!!! composed $N lines, the design says $NJOBS"; exit 2; }
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'DRY RUN, NOTHING SUBMITTED' )); $REJ rejected ----"

if [ "$SUBMIT" = 1 ]; then
  grep -qx 'cvt6-' "$REPO/bin/PROTECTED.txt" 2>/dev/null || echo "cvt6-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- cvt6- present in bin/PROTECTED.txt ----"
  guard_postlaunch "$WS/runs" "cvt6-" "${DESIGN[@]}"; RC=$?
  case $RC in
    0) echo "guard 7: RULE 20 post-launch check PASSED against a run's own ARGS line" ;;
    2) echo "guard 7: UNVERIFIED -- no cvt6 job started inside the window.  Re-check with:"
       echo "         python3 $AGF $WS/runs --name cvt6- --batch-consistency --vary stepsize-groups --vary seed --vary run-name" ;;
    *) echo "!!! guard 7: RULE 20 MISMATCH -- scancel THIS batch only:"
       echo "    squeue -h -u $USER_NAME -o '%i %j' | awk '\$2 ~ /^cvt6-/ {print \$1}' | xargs -r scancel" ;;
  esac
  echo
  echo "ENV AUDIT:"
  echo "    grep -h '^ENV:' $WS/runs/cvt6-*.out | sed 's/ PROBE_DIR=[^ ]*//' | sort | uniq -c    # ONE line, x21"
  echo "    grep -h '^PROBE_TENSOR:' $WS/runs/cvt6-*.out | sed 's/ dir=.*//' | sort | uniq -c    # scalar x3, blockwise x18, tensors=53"
  echo "    grep -h '^VOTE_W' $WS/runs/cvt6-*.out | sort | uniq -c                               # VOTE_W: off x21"
  echo "    grep -h '^BETA_HOLD' $WS/runs/cvt6-*.out | sort | uniq -c                            # off x6, floor x9, tri 9428 x6"
  echo "    grep -h '^COMP_HOLD' $WS/runs/cvt6-*.out | sort | uniq -c                            # off x12, rec x6, tri 9428 x3"
  echo
  echo "SCORE IT WITH, AND ONLY WITH (at 21/21 RUN_DONE):"
  echo "    python3 $REPO/analysis/cVT6_complementpath_score.py $WS/runs"
else
  echo
  echo "NOTHING WAS SUBMITTED.  The submitting form is:  bash bin/cVT6_complementpath.sh --submit"
  [ "$FAIL" = 1 ] && { echo "(dry run ended with guard failures -- exit 2)"; exit 2; }
fi
exit 0
