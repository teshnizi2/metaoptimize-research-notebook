# =============================================================================
# the hold schedule and the Lion recomputation -- pure functions
# =============================================================================
def u_of(n, P):
    if n <= 0:
        return 0
    return min(n - 1, P) - max(0, n - 1 - P)


def v_of(n, P, b0=B0):
    return f32(min(HI, max(LO, b0 + MS * u_of(n, P))))


def lion_natural(beta_pre, mom_pre, z):
    """-> (value, tie).  clamp(beta_pre - ms*sign(b2*m + (1-b2)*z)); tie iff |L| is at rounding level."""
    a, b = B2 * mom_pre, (1 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(LO, min(HI, beta_pre - MS * s)), tie


def hold_value(arm, n):
    m, P = MODE[arm]
    if m == "floor":
        return LO
    return v_of(n, P)

# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.

def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

def r18_tensors():
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if stride != 1 or cin != w:
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]

def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names=None):
    names = names or NAMES
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "bh": [ln for ln in lines if ln.startswith("BETA_HOLD")],
           "gh": [ln for ln in lines if ln.startswith("GROUP_HOLD")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest INTEGER job id among COMPLETE runs wins; a truncated
    resubmission cannot shadow a complete run."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):

def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm

def load_probe(runsdir, arm, seed, prefix=PREFIX, max_step=None):
    """-> list of records, or None if the file is absent.  Unparseable lines are None entries so a garbled
    file cannot pass the record-count check.  json accepts NaN/Infinity tokens.  max_step (selftest only, for
    ciso2's 250-epoch records) keeps records with step < max_step."""
    for base in (os.path.join(runsdir, prefix), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (prefix, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            if max_step is not None:
                recs = [r for r in recs if isinstance(r, dict) and isinstance(r.get("step"), int) and r["step"] < max_step]
            return recs
    return None


GH_KEYS = ("gh_n", "gh_active", "gh_nat", "gh_held")
BH_KEYS = ("bh_n", "bh_active", "bh_nat", "bh_held")


def _groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    return [float(x) for x in v[0]]


def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail).  Pure function of the records."""
    ng = len(arm_groups(arm))
    held = arm in HELD
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "bad_sched": 0,
         "bad_sched_pre": 0, "bad_gh": 0, "gh_on_control": 0, "bh_any": 0, "active_records": 0,
         "gh_active_last": None, "worst_lion": 0.0, "worst_sched": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    prev_active = -1
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) \
                or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre = _groupvec(r, "beta_pre", ng)
            mom = _groupvec(r, "mom_pre", ng)
            z = _groupvec(r, "z_agg", ng)
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        s = r["step"]
        if s != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        d["bh_any"] += any(key in r for key in BH_KEYS)
        unheld = range(ng) if not held else (0,)
        for g in unheld:
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst_lion"] = max(d["worst_lion"], e)
            d["bad_lion"] += e > BH_TOL
        has_gh = any(key in r for key in GH_KEYS)
        if not held:
            d["gh_on_control"] += has_gh
            continue
        n = s + 2
        es, ep = abs(beta[1] - hold_value(arm, n)), abs(bpre[1] - hold_value(arm, n - 1))
        d["worst_sched"] = max(d["worst_sched"], es, ep)
        d["bad_sched"] += es > BH_TOL
        d["bad_sched_pre"] += ep > BH_TOL
        try:
            gn, ga, gnat, gheld = int(r["gh_n"]), int(r["gh_active"]), float(r["gh_nat"]), float(r["gh_held"])
        except (KeyError, TypeError, ValueError):
            d["bad_gh"] += 1
            continue
        nat1, tie1 = lion_natural(bpre[1], mom[1], z[1])
        ok = (gn == n and abs(gheld - beta[1]) <= BH_TOL and 0 <= ga <= gn and ga >= prev_active
              and (tie1 or abs(gnat - nat1) <= BH_TOL))
        d["bad_gh"] += not ok
        if abs(gnat - gheld) > BH_TOL:
            d["active_records"] += 1
        prev_active = max(prev_active, ga)
        d["gh_active_last"] = ga
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_step"] == 0 and d["bad_lion"] == 0 and d["bh_any"] == 0)
    if held:
        ok = (ok and d["bad_sched"] == 0 and d["bad_sched_pre"] == 0 and d["bad_gh"] == 0
              and (d["gh_active_last"] is None or d["gh_active_last"] >= d["active_records"]))
    else:
        ok = ok and d["gh_on_control"] == 0
    return ok, d

def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord \
                and isinstance(r.get("step"), int):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    at = [v <= LO + 1e-4 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    win = [math.exp(v + 15.0) for s, v in vals if 15 <= s / float(STEPS_PER_EPOCH) < 30]
    return {"last": vals[-1][1], "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH),
            "pin_epoch": pin, "r1530": mean(win), "n": len(vals)}

def state(L, M):
    if L >= M["ISO"] - MATCH_BAR:
        return "AT-ISO"
    if L <= M["k01"] + NULL_BAR:
        return "AT-K01"
    return "BETWEEN"


def decide(M, diverged=False):
    """M: arm -> mean plateau5."""
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M["ISO"] - M["k01"] < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED"
    st = dict((a, state(M[a], M)) for a in HELD)
    if st["HOLDISO"] == "AT-K01":
        return "HOLD-CONTROL-FAILED"
    if st["HOLDISO"] == "BETWEEN":
        return "HOLD-CONTROL-PARTIAL"
    if st["HOLDLOW"] == "AT-K01":
        return "LOW-STALLS"
    if st["HOLDLOW"] == "BETWEEN":
        return "LOW-PARTIAL"
    return {"AT-K01": "TRANSFERS", "BETWEEN": "TRANSFERS-GRADED", "AT-ISO": "DOES-NOT-TRANSFER"}[st["HOLDHIGH"]]

def _e_probe_records(arm, seed=0, n=N_RECORDS, as_arm=None, unheld=False, touch_complement=False, pre_natural=False,
                     gh_forge=False, gh_off_by_one=False, strip_gh=False, add_gh=False, add_bh=False):
    """synthetic PROBE_TENSOR records, consistent with the harness's Lion step and the arm's hold.
    as_arm: build the HOLD of another arm (wrong P / wrong mode).  unheld: group 1 follows Lion (the null)."""
    import random
    rnd = random.Random(7000 + seed + 7 * ARMS.index(arm))
    ng = len(arm_groups(arm))
    hold_arm = as_arm or arm
    out = []
    active = 0
    for k in range(n):
        s = 100 * k
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        z = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [f32(max(LO, min(HI, -13.8 + 0.3 * math.sin(k / 30.0) - 0.004 * k))) for _ in range(ng)]
        beta = [f32(lion_natural(bpre[g], mom[g], z[g])[0]) for g in range(ng)]
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99}
        if ng == 2 and arm in HELD and not unheld:
            bpre[1] = hold_value(hold_arm, s + 1) if not pre_natural else f32(bpre[1] + 0.25)
            beta[1] = hold_value(hold_arm, s + 2)
            nat1 = f32(lion_natural(bpre[1], mom[1], z[1])[0])
            if abs(nat1 - beta[1]) > BH_TOL:
                active += 1
            rec.update({"gh_n": s + 2 + (1 if gh_off_by_one else 0), "gh_active": active,
                        "gh_nat": (nat1 + 0.01) if gh_forge else nat1, "gh_held": beta[1]})
            if strip_gh:
                for key in GH_KEYS:
                    rec.pop(key)
        if touch_complement:
            beta[0] = f32(beta[0] + 0.002)
        if add_gh:
            rec.update({"gh_n": s + 2, "gh_active": 0, "gh_nat": beta[-1], "gh_held": beta[-1]})
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[-1], "bh_held": beta[-1]})
        rec.update({"beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [z],
                    "z_tensor": [0.0] * NTENS, "m_tensor": [0.0] * NTENS})
        out.append(rec)
    return out

                    for r in pb[(a, s)]:

            elif probe_drop != (a, s):

            _e_write(d, a, s, jid, L, env=env, spec=spec, gh=gh, vw=vw, bh=bh)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def _spec_from_scorer(fname, key):
    p = os.path.join(HERE, fname)
    if not os.path.exists(p):
        return None
    import ast
    txt = open(p).read()
    m = re.search(r"^SPEC = (\{.*?^\})", txt, re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)

def main():
    ap = argparse.ArgumentParser(description="cvt7 registered scorer.  Documented invocation: "
                                             "python3 analysis/cVT7_grouphold_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt7/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
    sys.exit(main())
