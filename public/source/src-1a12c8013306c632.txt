#!/bin/bash
# =============================================================================
# cL1_cliff_tensor_split.sh -- `cts1`, THE SINGLE-TENSOR RESOLUTION OF THE
#                              k = 49 -> k = 52 CLIFF, SUBMITTED AS ONE BATCH.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.
# Nothing here touches /cluster/account_a and nothing is submitted to cluster_a.
# Every guard and every scancel line below is filtered on the `cts1-` prefix
# alone, so no job this script did not submit can be affected.
#
# REGISTERED SCORER: analysis/cL1_cts1_score.py, git-committed BEFORE this
# batch is submitted (STANDING RULE 21).  Every bar quoted here is frozen
# there, and that file's --selftest re-derives every archived number from
# results/all_runs.csv, re-derives cpk1's own cliff from the corpus, re-derives
# every partition AND every moved tensor from its own frozen per-tensor NAME
# and PARAMETER tables, and asserts each constant against THIS FILE'S own text.
#
# NO CODE PATCH.  Optimizers/HF.py::polish_the_stepsize_groups already parses
# an explicit int list given as the STRING "[a,b]" into CONTIGUOUS groups in
# named_parameters() order.  GUARD 4 below re-proves that on the LIVE model at
# submission time and writes PARTITION-MANIFEST.txt (sizes, per-group PARAMS,
# and the full FIRSTGROUP / LASTGROUP tensor NAME lists), which the scorer's
# G1 and G1b both REQUIRE.  The shared tree is not modified by this batch.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# `cpk1` (39 jobs, CORRECTIONS 146) swept the cut position k at fixed m = 2 and
# returned POSITION-SINGLE-PEAKED with argmax k* = 49.  Its ONE grid step
# k = 49 -> k = 52 cost -0.364021 CAPTURE (-17.105 pp, t -25.55, df 4): half
# the entire scalar-to-layerwise benefit.  Those two arms differ by 0.0118 of
# parameter share.  CORRECTIONS 146.4 recorded this as DESCRIPTIVE -- matched
# on nothing, not registered -- and named it the next test.
#
# On the LIVE model, tensors 50, 51 and 52 (1-based, named_parameters() order)
# are exactly the three that leave the FINE group between k = 49 and k = 52:
#
#     50  layer4.0.bn2.weight              512   moves at k=49 -> k=50
#     51  layer4.0.bn2.bias                512   moves at k=50 -> k=51
#     52  layer4.0.shortcut.0.weight    131072   moves at k=51 -> k=52
#     (132,096 total = 2.6931 % of the fine group's 4,905,060 at k = 49)
#
# So the two MISSING grid points k = 50 and k = 51 cut the cliff into three
# steps that each move EXACTLY ONE TENSOR.  Below one tensor there is no
# experiment: this is the finest resolution the partition axis admits.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  6 arms x 3 seeds = 18 jobs, ONE submission.
#
#   ANCHORS, run IN BATCH so CAPTURE is a within-batch quantity:
#     k01   scalar      m = 1
#     k62   layerwise   m = 62
#
#   THE FOUR CUTS, [k, 62-k]:
#     k = 49  50  51  52
#
#   BOTH CLIFF EDGES (k = 49 and k = 52) ARE RE-RUN HERE RATHER THAN SPLICED
#   FROM cpk1.  Batch, not seed, is the unit of replication in this corpus
#   (F(62,85) = 5.47, p 6.9e-13 for batch; F(11,1976) = 0.152 for seed).
#   Splicing would make two of the three steps cross-batch and one
#   within-batch, and the three would not be comparable.  Six extra jobs buy a
#   cliff that sits ENTIRELY INSIDE ONE BATCH.
#
#   CIFAR-100 / ResNet18_c100 / SGDm base / Lion meta / ms 1e-3 / alpha0 1e-6 /
#   100 ep / batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / HIER
#   unset / PROBE=0.  Matched cell-for-cell to cpk1, READ OUT OF
#   bin/cK1_cut_position_sweep.sh rather than reconstructed.
#
#   CAPTURE(arm) = ( M(arm) - M(k01) ) / ( M(k62) - M(k01) ) on plateau5, both
#   anchors measured IN BATCH.  DROP1 = CAP(49)-CAP(50), DROP2 = CAP(50)-
#   CAP(51), DROP3 = CAP(51)-CAP(52), TOTAL = CAP(49)-CAP(52).
#
# THE TWO ACCOUNTS ON RECORD BEFORE THE DATA, frozen in the scorer:
#   A1 PARAMETER MASS (the RIVAL TO BEAT, inherited from the cycle briefing):
#      the shortcut projection carries it, being 99.225 % of the moved
#      parameters.  Under A1, SHARE3 ~ 1 and DROP1 = DROP2.
#   A2 BATCHNORM GAIN LEVERAGE (this agent's POINT PREDICTION): a per-group
#      stepsize is one scalar, so what a group costs is how badly its members
#      disagree about the stepsize they want, not their parameter count.
#      layer4.0.bn2.weight is a multiplicative gain on layer4.0.conv2.weight
#      (2,359,296 parameters, ALREADY coarse at k = 49) and is the one free
#      knob that lets the fine group compensate for a coarse stepsize that is
#      wrong for layer4.0's residual branch.  Under A2 the collapse is DROP1,
#      and DROP1 >> DROP2 even though bn2.weight and bn2.bias hold IDENTICAL
#      parameter counts.  Mass has already been shown not to govern this cliff
#      (0.0118 of share buying 0.3640 of CAPTURE), so a mass-proportional
#      split repeats the account the parent finding already embarrassed.
#   NO THIRD ACCOUNT MAY BE INTRODUCED AFTER THE DATA.
#
# THE MATCHED CONTROL THIS DESIGN GETS FOR FREE: layer4.0.bn2.weight and
# layer4.0.bn2.bias hold EXACTLY 512 parameters each, are adjacent in
# named_parameters(), belong to the same BatchNorm module, and move at
# consecutive steps.  Under ANY account in which the currency is parameter
# mass, DROP1 and DROP2 must be EQUAL.  Unlike CORRECTIONS 146.4's
# k49-vs-k52 comparison, this one is matched.
#
# PRE-REGISTERED PREDICTIONS, frozen in the scorer:
#   Q1 (A2) the largest drop is DROP1 (k=49->50, layer4.0.bn2.weight).
#   Q2 (A1) the largest drop is DROP3 (k=51->52, layer4.0.shortcut.0.weight).
#   Q3 the cliff REPRODUCES: TOTAL >= 0.1820.
#   Q4 the split is NOT even.
#   Q5 SCALE-SHIFT ASYMMETRY: |DROP1 - DROP2| > 0.05.
#   Q6 TRAIN and TEST agree on which step is largest.
#
# PRE-REGISTERED REFUTATION CONDITIONS, frozen, CHECKED IN THIS ORDER:
#   R1 CLIFF-NOT-REPRODUCED   TOTAL < 0.1820, i.e. less than HALF of cpk1's
#                             own 0.364021.  THE refutation that VOIDS the
#                             question, checked FIRST.  cts1 exists only to
#                             decompose a cliff cpk1 measured; if the cliff is
#                             not here there is nothing to decompose, NO TENSOR
#                             MAY BE NAMED, the deliverable is UNRESOLVED, and
#                             cpk1's arms may NOT be spliced in to rescue it.
#                             0.1820 is 8.51 pp and 14.4 SE of a 3-seed CAPTURE
#                             difference at the re-derived noise floor.
#   R2 SPLIT-EVEN             all three drops within 0.05 of TOTAL/3.  Then it
#                             is NOT a single-tensor effect, no tensor may be
#                             named, and A1 and A2 both fail.  0.05 is 4.5 SE
#                             on DROP1/DROP3 and 3.7 SE on DROP2.
#   R3 NO-MAJORITY            the leader does not exceed TOTAL/2, or does not
#                             beat the runner-up by 0.05.  Naming requires the
#                             CONJUNCTION of majority AND separation.
#
# MAY NOT CLAIM: anything about the group COUNT m (every cut arm is m = 2 and
# the anchors are anchors, not rungs), about resnet18_blocks, CIFAR-10,
# Tiny-ImageNet, ImageNet-489, ResNet-50, alpha0 1e-3, any meta-stepsize but
# 1e-3, any optimiser pair but SGDm+Lion, or any hierarchical/shrinkage
# operator.  A tensor named here is named at ONE SITE in ONE architecture in
# ONE named_parameters() order.  Normalised entropy stays DISQUALIFIED by
# cbl1's R5 and is NOT rehabilitated by this batch.
#
# WALLTIME, MEASURED not guessed.  cpk1 ran this exact cell on this account
# and measured 42.0 min mean per run over 39 jobs, 0 TIMEOUT and 0 FAILED.
# 02:30:00 is 3.6x that mean, is cpk1's own request, and stays inside
# gpu-short's 4 h ceiling so no partition in the list is excluded.
# Total cost of this batch: 18 jobs x ~42 min ~= 13 GPU-hours.
#
# USAGE   bash bin/cL1_cliff_tensor_split.sh            # dry run, submits nothing
#         bash bin/cL1_cliff_tensor_split.sh --submit
# =============================================================================
set -u
SUBMIT=0; [ "${1:-}" = "--submit" ] && SUBMIT=1
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

USER_NAME="${USER:-$(whoami)}"
# NOTE: bin/_lib_guards.sh re-assigns WS at source time from METAOPT_WS with a
# /cluster/account_a default, which would clobber this value and point
# guard_postlaunch at the wrong account's directory (it did exactly that to tn1
# and to hb1).  Exporting METAOPT_WS here is the documented work-around; the
# library is NOT patched while jobs are queued.
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cts1
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP="-15:-2.3026"; MST=1e-3; ALPHA0=1e-6; EPOCHS=100; BATCH=100
GAMMA=1; AUG=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="0 1 2"
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
WALL=02:30:00
NTENS=62

# ---- the registered grid and the registered specs, written out LITERALLY ----
# Both forms are here on purpose: K_LIST is the grid the scorer cross-checks,
# and the G_K* strings are the exact argv the jobs receive.  guard 4c below
# proves they agree, so neither can drift from the other or from the scorer.
K_LIST="49 50 51 52"
G_K49='[49,13]'
G_K50='[50,12]'
G_K51='[51,11]'
G_K52='[52,10]'
ARMS="k01 k49 k50 k51 k52 k62"

echo "=============================================================="
echo " cts1 -- CIFAR-100 SINGLE-TENSOR resolution of the k=49 -> k=52 cliff"
echo " $DSET / $NET  box $CLIP  ms $MST  alpha0 $ALPHA0  AUGMENT=$AUG"
echo " cut positions k = $K_LIST   plus scalar and layerwise anchors"
echo " 6 arms x 3 seeds = 18 jobs, one submission"
echo " scorer: analysis/cL1_cts1_score.py (registered before any run exists)"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
CSVP="$REPO/results/all_runs.csv"
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- guard 1: the registered scorer exists, is committed, and selftests ------
SC="$REPO/analysis/cL1_cts1_score.py"
[ -f "$SC" ] || guard_fail "guard 1a: scorer missing"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  DIRTY=$(git -C "$REPO" status --porcelain -- analysis/cL1_cts1_score.py bin/cL1_cliff_tensor_split.sh 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: scorer AND this script are committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires the scorer committed BEFORE any run exists: $DIRTY"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 was verified"
  echo "          on the host that committed the scorer.  Not a pass, not a fail."
fi
python3 "$SC" --selftest >/tmp/cts1_selftest.$$ 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES" \
  || { guard_fail "guard 1c: scorer selftest FAILED"; tail -25 /tmp/cts1_selftest.$$; }

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ---------
grep -q '^cts1-' "$CSVP" && guard_fail "guard 2: cts1-* already in the CSV" \
                         || echo "guard 2: no cts1-* name collision in the corpus"
if [ -d "$WS/runs" ]; then
  NOUT=$(ls "$WS/runs"/cts1-*.out 2>/dev/null | wc -l | tr -d ' ')
  [ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cts1-*.out on disk" \
                       || guard_fail "guard 2b: $NOUT cts1-*.out files already exist"
fi

# ---- guard 3: the CIFAR-100 data is actually staged on THIS account ---------
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3: cifar-100-python present under $CIF/data" \
  || guard_fail "guard 3: CIFAR-100 is not staged at $CIF/data/cifar-100-python"

# ---- guard 4: EVERY PARTITION PROVEN ON THE LIVE MODEL, AND WRITTEN DOWN ----
# No assumption about tensor ordering, tensor count or parameter count:
# named_parameters() is read from the model this batch will actually train,
# polish_the_stepsize_groups is called on it, the GROUP MEMBERSHIP of every arm
# is printed in full, and the manifest records sizes AND per-group parameter
# counts AND the full tensor NAME lists so the scorer's G1 and G1b can check
# all three.  Guard 4d additionally prints, from the live model, WHICH TENSOR
# moves at each of the three registered single-tensor steps -- the whole point
# of the batch -- and fails if any step moves more or fewer than one tensor.
mkdir -p "$SAVE"
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"

# guard 4c: the literal spec strings and the grid must agree, BEFORE the model
# is ever built.  A typo in one of the four G_K* strings dies here.
SPECS=""
for K in $K_LIST; do
  eval "LIT=\${G_K$K:-}"
  WANT="[$K,$((NTENS-K))]"
  if [ "$LIT" != "$WANT" ]; then
    guard_fail "guard 4c: G_K$K is '$LIT', but the grid says '$WANT'"
  fi
  SPECS="$SPECS $LIT"
done
NK=$(echo $K_LIST | wc -w | tr -d ' ')
NA=$(echo $ARMS | wc -w | tr -d ' ')
[ "$NA" = "$((NK+2))" ] || guard_fail "guard 4c: $NA arms for $NK cut positions + 2 anchors"
# the four cuts must be CONSECUTIVE, or the steps are not single-tensor
PREV=""
for K in $K_LIST; do
  if [ -n "$PREV" ] && [ "$K" != "$((PREV+1))" ]; then
    guard_fail "guard 4c: the grid is not consecutive ($PREV -> $K); the steps would not be single-tensor"
  fi
  PREV=$K
done
echo "guard 4c: $NK literal specs agree with the grid, and the grid is consecutive; $NA arms in total"
# shellcheck disable=SC2086
python3 - "$CIF" "$MANIFEST" "$NTENS" $SPECS <<'PYEOF' \
  || guard_fail "guard 4: live partition check FAILED"
import sys, os
cif, out, ntens = sys.argv[1], sys.argv[2], int(sys.argv[3])
specs = sys.argv[4:]
sys.path.insert(0, cif)
os.chdir(cif)
from build_network import build_network
from Optimizers.HF import HF
net = build_network("ResNet18_c100", "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in net.named_parameters()]
numel = [int(p.numel()) for _, p in net.named_parameters()]
tot = sum(numel)
T = len(nps)
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot]
print("guard 4: live ResNet18_c100 has %d parameter tensors, %d parameters"
      % (T, tot))
if T != ntens:
    print("!!! guard 4: the live model has %d tensors, the batch assumes %d"
          % (T, ntens))
    raise SystemExit(1)

groups, cur = [], None
for i, (n, _) in enumerate(nps, 1):
    key = n.split(".")[0]
    if key.startswith("layer"):
        key = ".".join(n.split(".")[:2])
    if key != cur:
        groups.append([key, i, i])
        cur = key
    else:
        groups[-1][2] = i
print("guard 4: named_parameters() structure, measured on the live model:")
for name, lo, hi in groups:
    print("         %-12s tensors %2d..%-2d  params %9d"
          % (name, lo, hi, sum(numel[lo - 1:hi])))
print("guard 4: the layer4.0 block, tensor by tensor -- this is where the cut sits:")
for i in range(45, 54):
    print("         %2d  %-32s %9d" % (i + 1, names[i], numel[i]))

bad = 0
seen = {}
for spec in specs:
    g = HF.polish_the_stepsize_groups(None, spec, nps)
    sizes = [len(x) for x in g]
    if sum(sizes) != T:
        print("!!! guard 4: %s does not cover the model (%d of %d)"
              % (spec, sum(sizes), T))
        bad += 1
        continue
    if len(g) != 2:
        print("!!! guard 4: %s composed to m=%d, but every cut arm must be m=2"
              % (spec, len(g)))
        bad += 1
        continue
    want = [int(x) for x in spec[1:-1].split(",")]
    if sizes != want:
        print("!!! guard 4: %s composed to sizes %s on the live model" % (spec, sizes))
        bad += 1
        continue
    k = sizes[0]
    if list(g[0]) != names[:k] or list(g[-1]) != names[k:]:
        print("!!! guard 4: %s does not split named_parameters() at %d BY NAME"
              % (spec, k))
        bad += 1
        continue
    pc, i = [], 0
    for s in sizes:
        pc.append(sum(numel[i:i + s]))
        i += s
    lines.append("SPEC %s M %d SIZES %s PARAMS %s FIRSTGROUP %s LASTGROUP %s"
                 % (spec, len(g), ",".join(str(s) for s in sizes),
                    ",".join(str(s) for s in pc), " ".join(g[0]), " ".join(g[-1])))
    print("guard 4: %-9s m=2  sizes %-8s  params %9d / %-9d  share %.6f"
          % (spec, str(sizes), pc[0], pc[1], pc[0] / float(tot)))
    print("         coarse group (%d tensors) ends at   %s" % (k, g[0][-1]))
    print("         fine   group (%d tensors) starts at %s" % (sizes[1], g[1][0]))
    print("         FULL coarse = %s" % " ".join(g[0]))
    print("         FULL fine   = %s" % " ".join(g[1]))
    seen[k] = (set(g[-1]), pc[0])

# guard 4d: THE POINT OF THE BATCH.  Which tensor moves at each step, measured.
ks = sorted(seen)
print("guard 4d: the single-tensor steps, MEASURED on the live model:")
moved_total = 0
for a, b in zip(ks, ks[1:]):
    moved = [n for n in names if n in seen[a][0] and n not in seen[b][0]]
    dp = seen[b][1] - seen[a][1]
    moved_total += dp
    flag = "" if len(moved) == 1 else "   <<< NOT A SINGLE TENSOR"
    print("         k=%d -> k=%d moves %-30s %8d params%s"
          % (a, b, " ".join(moved), dp, flag))
    if len(moved) != 1:
        bad += 1
    if sum(numel[names.index(n)] for n in moved) != dp:
        print("!!! guard 4d: parameter bookkeeping does not close at k=%d->%d" % (a, b))
        bad += 1
if ks:
    print("guard 4d: the %d steps move %d parameters in total, %.4f %% of the "
          "fine group at k=%d"
          % (len(ks) - 1, moved_total,
             100.0 * moved_total / float(sum(numel[ks[0]:])), ks[0]))

# The anchors are NAMED, not listed: init_meta routes them by exact string
# match and they never reach polish_the_stepsize_groups.  That is the confound
# the scorer discloses, so the guard proves the routing on the LIVE source
# rather than assuming it.
import inspect, re as _re
src = inspect.getsource(HF.init_meta)
mm = _re.search(r"stepsize_groups\s+in\s+\[([^\]]*)\]", src)
if not mm:
    print("!!! guard 4: could not find init_meta's exact-match routing list")
    bad += 1
else:
    named = [x.strip().strip("'\"") for x in mm.group(1).split(",")]
    print("guard 4: init_meta routes these names away from blockwise: %s" % named)
    for name in ("scalar", "layerwise"):
        if name in named:
            print("guard 4: anchor %-9s takes its OWN code path (not blockwise)" % name)
        else:
            print("!!! guard 4: anchor %s is NOT in init_meta's named list; it would "
                  "fall through to blockwise" % name)
            bad += 1
    for spec in specs:
        if spec in named:
            print("!!! guard 4: cut spec %s collides with a named granularity" % spec)
            bad += 1

if bad:
    raise SystemExit(1)
open(out, "w").write("\n".join(lines) + "\n")
print("guard 4: %d cut specs verified; PARTITION-MANIFEST.txt written to %s"
      % (len(specs), out))
PYEOF
[ -s "$MANIFEST" ] && echo "guard 4b: manifest is non-empty" || guard_fail "guard 4b: no manifest"

# ---- guard 5: capacity ------------------------------------------------------
FREE=$(df -BG "$WS" 2>/dev/null | tail -1 | awk '{print $4}' | tr -d 'G')
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: ${FREE:-?} GB free on \$WS; $PEND jobs pending, adding 18"
[ "${PEND:-0}" -le 40 ] || guard_fail "guard 5: $PEND pending, over cap"

[ "$FAIL" = 1 ] && { echo "!!! SUBMIT ABORTED -- nothing submitted."; exit 2; }

# ---- compose, guard each line under RULE 20, then submit --------------------
. "$REPO/bin/_lib_guards.sh"
DESIGN=(alg-base=SGDm alg-meta=Lion meta-stepsize=$MST alpha0=$ALPHA0 \
        num-epochs=$EPOCHS batch-size=$BATCH gamma=$GAMMA dataset=$DSET NN-name=$NET)

N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    case $ARM in
      k01) G=scalar ;;
      k49) G=$G_K49 ;;
      k50) G=$G_K50 ;;
      k51) G=$G_K51 ;;
      k52) G=$G_K52 ;;
      k62) G=layerwise ;;
      *)   echo "!!! unknown arm $ARM"; exit 2 ;;
    esac
    # NOTE the single quotes around $G below: "[50,12]" is a bash bracket GLOB.
    # Unquoted, inside the eval, it would expand against the CWD if a file named
    # 5 or 0 or , ever existed there.  Quoted, argv reaches train.py literally
    # and the runner's `echo "ARGS: $@"` prints it unchanged for RULE 20.
    CMD="sbatch --job-name=cts1-$ARM-s$S --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,HIER=none,SCHED=none,PROBE=0 \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
--gamma $GAMMA --meta-stepsize $MST --alpha0 $ALPHA0 --num-epochs $EPOCHS \
--stepsize-groups '$G' --seed $S --save-directory $SAVE --run-name cts1-$ARM-s$S"
    if guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1; then :; else
      echo "!!! RULE 20 PRE-CHECK FAILED for cts1-$ARM-s$S"
      guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1))
    fi
    if [ "$SUBMIT" = 1 ]; then eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
    else echo "$CMD"; echo; N=$((N+1)); fi
  done
done
echo "guard 6: $N composed command lines, $BAD failed the RULE 20 pre-check"
[ "$BAD" = 0 ] || { echo "!!! composed lines are malformed"; exit 2; }
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'dry run, nothing submitted' )); $REJ rejected ----"

if [ "$SUBMIT" = 1 ]; then
  grep -qx 'cts1-' "$REPO/bin/PROTECTED.txt" 2>/dev/null || echo "cts1-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- cts1- added to bin/PROTECTED.txt ----"
  # jobs/run_cifar.sh hard-codes #SBATCH --output=$WS/runs/%x-%j.out, so the .out
  # files land in $WS/runs, NOT in --save-directory.
  guard_postlaunch "$WS/runs" "cts1-" "${DESIGN[@]}"; RC=$?
  case $RC in
    0) echo "guard 7: RULE 20 post-launch check PASSED against a run's own ARGS line" ;;
