def r18_tensors():
    out = [("conv1.weight", "Conv2d"), ("bn1.weight", "BatchNorm2d"), ("bn1.bias", "BatchNorm2d")]
    for L, down in ((1, False), (2, True), (3, True), (4, True)):
        for b in (0, 1):
            p = "layer%d.%d." % (L, b)
            out += [(p + "conv1.weight", "Conv2d"), (p + "bn1.weight", "BatchNorm2d"), (p + "bn1.bias", "BatchNorm2d"),
                    (p + "conv2.weight", "Conv2d"), (p + "bn2.weight", "BatchNorm2d"), (p + "bn2.bias", "BatchNorm2d")]
            if down and b == 0:
                out += [(p + "shortcut.0.weight", "Conv2d"), (p + "shortcut.1.weight", "BatchNorm2d"),
                        (p + "shortcut.1.bias", "BatchNorm2d")]
    out += [("linear.weight", "Linear"), ("linear.bias", "Linear")]
    return [(n, _NUMEL[i], t) for i, (n, t) in enumerate(out)]


NAMES = [n for n, _k, _t in r18_tensors()]


def parse_sets(spec, names=NAMES):
    """Re-typed PATCH_NAMESETS grammar: -> list of lists of 1-based indices.  Raises on anything the harness rejects."""
    if not spec.startswith("sets:"):
        raise ValueError("not a sets: spec")
    idx = dict((n, i + 1) for i, n in enumerate(names))
    out, seen = [], set()
    for g in spec[5:].split("/"):
        if not g:
            raise ValueError("empty group")
        cur = []
        for it in g.split(","):
            m = re.match(r"^(\d+)-(\d+)$", it)
            if m:
                a, b = int(m.group(1)), int(m.group(2))
                if not 1 <= a <= b <= len(names):
                    raise ValueError("bad range %s" % it)
                cur += list(range(a, b + 1))
            elif re.match(r"^\d+$", it):
                cur.append(int(it))
            elif it in idx:
                cur.append(idx[it])
            else:
                raise ValueError("unknown item %s" % it)
        for i in cur:
            if i in seen or not 1 <= i <= len(names):
                raise ValueError("index %d repeated or out of range" % i)
            seen.add(i)
        out.append(sorted(cur))
    if seen != set(range(1, len(names) + 1)):
        raise ValueError("not total")
    return out

    return [[i] for i in range(1, 46)] + [merged] + [[i] for i in single] + [[61], [62]]

def mean(v):
    return sum(v) / len(v)


def sd(v):
    if len(v) < 2:
        return float("nan")
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def file_sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        for b in iter(lambda: fh.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()

def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    dup = False
    for a, b, c in EPTR_RE.findall(txt):
        if int(a) in eps:
            dup = True
        try:
            eps[int(a)] = (float(b), float(c))
        except ValueError:
            eps[int(a)] = (float("nan"), float("nan"))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps), "dup_epoch": dup,
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "foreign": [ln for ln in lines if ln.startswith(FOREIGN_PREFIXES)],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"] and not rec["traceback"]
                       and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest job id among COMPLETE runs wins."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s" % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def load_probe(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        d = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed))
        p = os.path.join(d, "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            meta = {}
            for fn in ("probe_tensor.json", "block_sizes.json"):
                try:
                    meta[fn] = json.load(open(os.path.join(d, fn)))
                except Exception:
                    meta[fn] = None
            return recs, meta
    return None, {}

def partition_ok(rec, arm):
    """One record: z_agg / mom_pre / beta_pre have the arm's group count, and every group's z_agg and mom_pre equal the
    sum of its registered members' z_tensor / m_tensor (singletons at TOL_SINGLE, merged at TOL_MERGED)."""
    try:
        g = groups_of(arm)
        za, ma, bp = rec["z_agg"][0], rec["mom_pre"][0], rec["beta_pre"][0]
        z, m = rec["z_tensor"], rec["m_tensor"]
        if not (len(za) == len(ma) == len(bp) == len(g)) or len(z) != NTENS or len(m) != NTENS:
            return False
        for k, grp in enumerate(g):
            tol = TOL_SINGLE if len(grp) == 1 else TOL_MERGED
            for agg, per in ((za, z), (ma, m)):
                s = sum(per[i - 1] for i in grp)
                if not (math.isfinite(s) and math.isfinite(agg[k])):
                    return False
                if abs(s - agg[k]) > tol * max(abs(agg[k]), abs(s), 1e-30):
                    return False
        return True
    except (KeyError, IndexError, TypeError):
        return False


def partition_check(arm, recs, meta):
    d = {"n": 0 if recs is None else len(recs), "bad": 0, "steps_ok": False, "meta_ok": False, "why": ""}
    if recs is None:
        d["why"] = "probe absent"
        return False, d
    good = [r for r in recs if isinstance(r, dict)]
    steps = [r.get("step") for r in good]
    d["steps_ok"] = len(recs) == N_RECORDS and steps == list(range(0, N_RECORDS * PROBE, PROBE))
    d["bad"] = sum(1 for r in recs if not (isinstance(r, dict) and partition_ok(r, arm)))
    pt, bs = meta.get("probe_tensor.json"), meta.get("block_sizes.json")
    want_nb = [sum(_NUMEL[i - 1] for i in x) for x in groups_of(arm)]
    d["meta_ok"] = (isinstance(pt, dict) and pt.get("stepsize_type") == "blockwise" and pt.get("n_tensors") == NTENS
                    and pt.get("param_numels") == _NUMEL and isinstance(bs, dict) and bs.get("n_b") == want_nb)
    ok = d["steps_ok"] and d["bad"] == 0 and d["meta_ok"]
    if not ok:
        d["why"] = "steps %s, bad records %d, meta %s" % (d["steps_ok"], d["bad"], d["meta_ok"])
    return ok, d


def pin_fraction(recs, arm, gpos, lo_step=PIN_LATE_STEP, hi_step=None):
    v = [r["beta_pre"][0][gpos] <= PIN_LO for r in recs if isinstance(r, dict)
         and r.get("step", -1) >= lo_step and (hi_step is None or r["step"] < hi_step)]
    return (sum(v) / len(v)) if v else float("nan")


def pin_word(f):
    if f != f:
        return "NA"
    return "PINS" if f >= PIN_YES else ("FREE" if f <= PIN_NO else "NEITHER")


# =============================================================================
def decide(M, diverged=False):
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["KLS"] < REF_MIN:
        return "UNRESOLVED-REFERENCE-UNHEALTHY"
    d_car, d_ctl = M["KLS"] - M["MCAR"], M["KLS"] - M["MCTL"]
    did = M["MCTL"] - M["MCAR"]
    if d_car >= MARGIN and did >= MARGIN and d_ctl < MARGIN:
        return "CARRIER-MERGE-HARMS"
    if d_car >= MARGIN and did >= MARGIN and d_ctl >= MARGIN:
        return "BOTH-HARM-CARRIER-MORE"
    if d_car >= MARGIN and d_ctl >= MARGIN and abs(did) < MARGIN:
        return "BOTH-HARM-ALIKE"
    if d_ctl >= MARGIN and did <= -MARGIN:
        return "CONTROL-MERGE-HARMS-MORE"
    if d_car < MARGIN and d_ctl < MARGIN and abs(did) < MARGIN:
        return "NO-MERGE-HARMS"
    return "UNRESOLVED-PARTIAL"


def arm_state(D, se):
    if abs(D) <= READ_SE * se:
        return "AT-KLS"
    if D >= MARGIN:
        return "BELOW-KLS"
    if D <= -MARGIN:
        return "ABOVE-KLS"
    return "NEAR-KLS-BELOW" if D > 0 else "NEAR-KLS-ABOVE"


def corpus_disclosure(csvpath):
    """NON-GATING: the corpus row count and the filtered layerwise mean at the cell.  No bar reads it."""
    if not csvpath or not os.path.exists(csvpath):
        return "not readable"
    sys.path.insert(0, HERE)
    try:
        import corpus_exclusions as CE
    finally:
        sys.path.pop(0)
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    filt = CE.filter_rows(rows)
    v = [float(r["plateau5"]) for r in filt if r.get("network") == NET and r.get("dataset") == DSET
         and r.get("granularity") == "layerwise" and r.get("epochs_requested") == "100" and r.get("meta_stepsize") == "1e-3"
         and r.get("alpha0") == "1e-6" and r.get("augment") == "1" and r.get("beta_clip") == "-15:-2.3026"
         and r.get("batch_size") == "100" and r.get("complete") == "1" and (r.get("plateau5") or "").strip()
         and r.get("hier", "") == "" and r.get("superseded") == "0" and r.get("collapsed") == "0"]
    return "%d rows after filter_rows (%s- excluded); layerwise at the cell n %d mean %s" % (
        len(filt), PREFIX, len(v), ("%.4f" % mean(v)) if v else "-")

def _e_probe_dir(cache, arm, as_arm, pin):
    """A cached synthetic probe dir: records for `arm`'s levels but grouped as `as_arm`; G* pinned late iff pin."""
    key = (arm, as_arm, pin)
    if key in cache:
        return cache[key]
    d = os.path.join(cache["_root"], "probe_%s_%s_%s" % key)
    os.makedirs(d)
    rng = random.Random(sum(ord(ch) * (k + 1) for k, ch in enumerate("%s|%s|%s" % key)))
    g = groups_of(as_arm)
    with open(os.path.join(d, "probe.jsonl"), "w") as fh:
        for k in range(N_RECORDS):
            z = [round(rng.uniform(-1, 1), 3) for _ in range(NTENS)]
            m = [round(rng.uniform(-1, 1), 3) for _ in range(NTENS)]
            step = k * PROBE
            bp = []
            for gi, grp in enumerate(g):
                if len(grp) > 1:
                    bp.append(-15.0 if (pin and step >= 20000) else -8.0)
                else:
                    bp.append(-9.0)
            rec = {"step": step, "z_tensor": z, "m_tensor": m, "z_agg": [[sum(z[i - 1] for i in grp) for grp in g]],
                   "mom_pre": [[sum(m[i - 1] for i in grp) for grp in g]], "beta_pre": [bp]}
            fh.write(json.dumps(rec) + "\n")
    json.dump({"PROBE_TENSOR": 1, "every": 100, "stepsize_type": "blockwise", "n_tensors": NTENS, "param_numels": _NUMEL},
              open(os.path.join(d, "probe_tensor.json"), "w"))
    json.dump({"stepsize_type": "blockwise", "n_b": [sum(_NUMEL[i - 1] for i in x) for x in g]},
              open(os.path.join(d, "block_sizes.json"), "w"))
    cache[key] = d
    return d


def _e_build(root, cache, levels, jit=(0.3, -0.3, 0.1, -0.1), drop=None, over=None, probe_as=None, no_manifest=False,
             bad_manifest=False, prov_mode="submit", resubmit=None, pin=None):
    d = tempfile.mkdtemp(dir=root)
    os.makedirs(os.path.join(d, PREFIX))
    jid = 9000000
    for s_i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            if drop == (a, s):
                continue
            kw = (over or {}).get((a, s), {})
            _e_write(d, a, s, jid, levels[a], jitter=jit[s_i], **kw)
            if resubmit == (a, s):
                _e_write(d, a, s, jid - 5000, levels[a], n_ep=40, run_done=False, jitter=jit[s_i])
            src = _e_probe_dir(cache, a, (probe_as or {}).get((a, s), a), bool((pin or {}).get(a, a == "MCAR")))
            os.symlink(src, os.path.join(d, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s)))
    if not no_manifest:
        txt = synthetic_manifest_text()
        open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(txt + ("X\n" if bad_manifest else ""))
    open(os.path.join(d, PREFIX, "PROVENANCE.txt"), "w").write(
        "MODE %s\nHF_SHA256 %s\nBUILD_NETWORK_SHA256 %s\nRUNNER_SHA256 %s\nSCORER_SHA256 %s\n"
        % (prov_mode, LIVE_HF_SHA, LIVE_BN_SHA, RUNNER_SHA, file_sha(os.path.abspath(__file__))))
    return d


def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()


def _e_csv_variants(src, dst_dir):
    out = {"no-corpus": os.path.join(dst_dir, "does_not_exist.csv")}
    base = {"dataset": DSET, "base": "SGDm", "meta": "Lion", "meta_stepsize": "1e-3", "alpha0": "1e-6", "gamma": "1",
            "augment": "1", "beta_clip": "-15:-2.3026", "batch_size": "100", "epochs_requested": "100", "hier": "",
            "lam": "na", "eta_ratio": "na", "superseded": "0", "collapsed": "0", "complete": "1", "network": NET}
    inv = []
    for i, (gran, v) in enumerate([("layerwise", 1.0), ("layerwise", 99.0), (SPEC["KLS"], 3.0), (SPEC["MCAR"], 90.0),
                                   (SPEC["MCTL"], 2.0), ("scalar", 95.0)]):
        inv.append(dict(base, granularity=gran, run="zstress-%d-s%d" % (i, 900 + i), job_id=str(9990000 + i),
                        seed=str(900 + i), plateau5="%.4f" % v))
    if src and os.path.exists(src):
        rows = list(csv.DictReader(open(src)))
        hdr = list(rows[0].keys())
        out["real"] = src
        p = os.path.join(dst_dir, "stress.csv")
        with open(p, "w", newline="") as fh:
            w = csv.DictWriter(fh, fieldnames=hdr, extrasaction="ignore")
            w.writeheader()
            for r in rows:
                w.writerow(r)
            for r in inv:
                w.writerow(dict((k, r.get(k, "")) for k in hdr))
        out["real+invented"] = p
    hdr = ["run", "job_id", "seed", "plateau5", "granularity"] + list(base.keys())
    p = os.path.join(dst_dir, "minimal.csv")
    with open(p, "w", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=hdr, extrasaction="ignore")
        w.writeheader()
        for r in inv:
            w.writerow(r)
    out["minimal-invented"] = p
    return out

def selftest(csvpath, runsdir):

    print("corpus_exclusions.filter_rows, the accounts and the 164.6 floor table, G-PARTITION on synthetic AND real records,")
    print("the REAL score() on synthetic runs, corpus invariance, and the documented one-argument invocation.\n")

    print("A. DESIGN")
    t = r18_tensors()
    chk(len(t) == NTENS and sum(k for _n, k, _t in t) == TOTPAR, "62 tensors, 11,220,132 parameters (re-typed)")
    chk([NAMES[i - 1] for i in C_SET] == ["layer4.0.bn2.weight", "layer4.0.shortcut.1.weight", "layer4.1.bn2.weight"],
        "C = the carriers {50,53,59} (184 / 187)")

        "T = cdep1's DEPTH triple {47,48,56} (188.1)")

        and all(t[i - 1][2] == "BatchNorm2d" and _NUMEL[i - 1] == 512 for i in C_SET + T_SET),
        "C and T: count 3, numel 1,536, each 512-wide BatchNorm2d, all in layer4")
    l4 = [i for i in range(1, 63) if NAMES[i - 1].startswith("layer4.")]
    chk(sorted(N_SET + C_SET + T_SET) == l4 and len(N_SET) == 9, "N u C u T = layer4's 15 tensors exactly; N has 9",
        "N numel %d" % sum(_NUMEL[i - 1] for i in N_SET))
    for a in ARMS:
        try:
            g = parse_sets(SPEC[a])
            chk(g == groups_of(a), "%s spec parses (re-typed PATCH_NAMESETS grammar) to the registered %d groups" % (a, len(g)))
        except ValueError as ex:
            chk(False, "%s spec parses" % a, str(ex))
        chk(re.match(r"^[A-Za-z0-9_.,:/-]+$", SPEC[a]) is not None, "%s spec is ONE shell token of the NAMESETS alphabet" % a)
    gc, gt = groups_of("MCAR"), groups_of("MCTL")
    chk([len(x) for x in gc] == [len(x) for x in gt] and
        [sum(_NUMEL[i - 1] for i in x) for x in gc] == [sum(_NUMEL[i - 1] for i in x) for x in gt],
        "MCAR and MCTL: identical group count, sizes and per-group numel (51 groups; G* 12 tensors, 8,392,192 params)")
    chk(set(gc[MERGED_G]) - set(gt[MERGED_G]) == set(C_SET) and set(gt[MERGED_G]) - set(gc[MERGED_G]) == set(T_SET),
        "G*(MCAR) and G*(MCTL) differ exactly by C vs T")
    chk(SPEC["MCAR"] != SPEC["MCTL"] != SPEC["KLS"], "three distinct CSV granularity strings (corpus-separable)")
    for bad in ("sets:1-62/1", "sets:1-61", "sets:1-62/", "sets:foo/1-62"):
        try:
            parse_sets(bad)
            chk(False, "parse_sets rejects %r" % bad)
        except ValueError:
            chk(True, "parse_sets rejects %r" % bad)
    txt = synthetic_manifest_text()
    chk(txt.count("\nTENSOR ") == 62 and "ARMMERGED MCAR layer4.0.conv1.weight layer4.0.conv2.weight layer4.0.bn2.weight" in txt,
        "synthetic manifest carries 62 TENSOR lines and the merged-group NAMES", "%d bytes, sha %s" % (len(txt), hashlib.sha256(txt.encode()).hexdigest()[:16]))
    chk(all(not p.startswith(q) and not q.startswith(p) for p in FOREIGN_PREFIXES for q in FOREIGN_PREFIXES if p != q),
        "no foreign witness prefix is a prefix of another")

    print("\nB. THE NOISE FLOOR through corpus_exclusions.filter_rows (OPERATIONS 36), registration corpus")
    if csvpath and os.path.exists(csvpath):
        raw = open(csvpath, newline="").read().splitlines(True)
        if len(raw) - 1 >= REG_ROWS:
            blob = (raw[0] + "".join(raw[1:1 + REG_ROWS])).encode()
            chk(hashlib.sha256(blob).hexdigest() == REG_CORPUS_SHA, "the CSV's first %d data rows ARE the registration corpus" % REG_ROWS)
            sys.path.insert(0, HERE)
            try:
                import corpus_exclusions as CE
            finally:
                sys.path.pop(0)
            rows = list(csv.DictReader([raw[0]] + raw[1:1 + REG_ROWS]))
            filt = CE._pooled(CE.filter_rows(rows), NET)
            naive = CE._pooled(rows, NET)
            chk(abs(filt[0] - SIGMA_PRIOR) < 1e-12 and filt[1] == SIGMA_PRIOR_DF,
                "SIGMA_PRIOR: the FROZEN literal %r IS the FILTERED value" % SIGMA_PRIOR, "%r df %d" % (filt[0], filt[1]))
            chk(abs(naive[0] - SIGMA_NAIVE) < 1e-9 and abs(naive[0] - SIGMA_PRIOR) > 1.0, "... and is NOT the naive (unfiltered) value",
                "naive %r" % naive[0])
            lw = [float(r["plateau5"]) for r in CE.filter_rows(rows) if r["network"] == NET and r["granularity"] == "layerwise"
                  and r["epochs_requested"] == "100" and r["meta_stepsize"] == "1e-3" and r["alpha0"] == "1e-6"
                  and r["augment"] == "1" and r["beta_clip"] == "-15:-2.3026" and r["batch_size"] == "100"
                  and r["complete"] == "1" and r["hier"] == "" and r["superseded"] == "0" and r["collapsed"] == "0"
                  and r["dataset"] == DSET and r["base"] == "SGDm" and r["meta"] == "Lion"]
            chk(len(lw) == 32 and abs(mean(lw) - LAYERWISE_MEAN) < 5e-5 and abs(sd(lw) - LAYERWISE_SD) < 5e-5,
                "LAYERWISE_MEAN / SD re-derived (filtered, n 32)", "%.6f / %.6f" % (mean(lw), sd(lw)))
            chk(not any(r["run"].startswith(PREFIX + "-") or r["seed"] in ("124", "125", "126", "127")
                        for r in csv.DictReader(open(csvpath))), "ZERO corpus rows are cmg1- or carry seed 124-127")
        else:
            skip("registration corpus", "CSV shorter than %d rows" % REG_ROWS)
    else:
        skip("noise floor", "no CSV")
    chk(abs(REF_MIN - (LAYERWISE_MEAN - MARGIN)) < 1e-9, "REF_MIN = LAYERWISE_MEAN - MARGIN")

    print("\nC. ACCOUNTS AND THE 164.6 FLOOR TABLE")
    se0 = SIGMA_PRIOR * math.sqrt(0.5)
    for acc, lv in ACCOUNTS.items():
        b = decide(lv)
        chk(b == ACCOUNT_BRANCH[acc], "%s predicts %s" % (acc, ACCOUNT_BRANCH[acc]), "decide() -> %s" % b)
        print("        KLS %.4f MCAR %.4f MCTL %.4f | DELTA_ID %+.4f (%+.1f SE) D_CAR %+.4f D_CTL %+.4f"
              % (lv["KLS"], lv["MCAR"], lv["MCTL"], lv["MCTL"] - lv["MCAR"], (lv["MCTL"] - lv["MCAR"]) / se0,
                 lv["KLS"] - lv["MCAR"], lv["KLS"] - lv["MCTL"]))
        chk(all(not (SAT_BAND[0] <= v <= SAT_BAND[1]) for v in lv.values()), "%s: no arm predicted inside the floor saturation band" % acc)

    sep12 = abs((a1["MCTL"] - a1["MCAR"]) - (a2["MCTL"] - a2["MCAR"])) / se0
    chk(sep12 >= 10, "the PRIMARY separates A1 from A2 by >= 10 SE (not floor-bounded)", "%.1f SE" % sep12)
    sep13 = abs((a1["KLS"] - a1["MCAR"]) - (a3["KLS"] - a3["MCAR"])) / se0
    chk(sep13 >= 10, "D_CAR separates A1/A2 from A3 by >= 10 SE", "%.1f SE" % sep13)

    chk("BOUND" in " ".join(NOT_LICENSED) and "164.6" in " ".join(NOT_LICENSED), "an AT-KLS reading is registered as a BOUND (164.6)")

        M = {"KLS": rng.uniform(55, 75), "MCAR": rng.uniform(40, 75), "MCTL": rng.uniform(40, 75)}

        chk(own == 3 * nrec, "every real-record synthetic record passes as its own arm", "%d / %d" % (own, 3 * nrec))
        chk(cross == 0, "NO real-record synthetic record passes as either other arm (the broken-partition null is excluded)",
            "%d of %d cross reads pass" % (cross, 6 * nrec))
        recs0, meta0 = [], {}
        for ln in open(os.path.join(real[0], "probe.jsonl")):
            recs0.append(json.loads(ln))
        for fn in ("probe_tensor.json", "block_sizes.json"):
            p0 = os.path.join(real[0], fn)

            "records bad %d, meta_ok %s" % (d0["bad"], d0["meta_ok"]))
    else:

    print("\nF. THE REAL score() ON SYNTHETIC RUNS (every number invented)")
    root = tempfile.mkdtemp(prefix="cmg1_selftest_")
    cache = {"_root": os.path.join(root, "_cache")}
    os.makedirs(cache["_root"])
    csvs = _e_csv_variants(csvpath, root)
    cases = [
        ("CARRIER-MERGE-HARMS", _levels(69.5, 60.0, 68.5), {}, 0),
        ("BOTH-HARM-CARRIER-MORE", _levels(69.5, 52.0, 63.0), {}, 0),
        ("BOTH-HARM-ALIKE", _levels(69.5, 60.0, 61.0), {}, 0),
        ("CONTROL-MERGE-HARMS-MORE", _levels(69.5, 68.0, 60.0), {}, 0),
        ("NO-MERGE-HARMS", _levels(69.5, 68.5, 68.8), {}, 0),

        ("NO-MERGE-HARMS", _levels(69.5, 64.6, 69.45), {}, 0),                    # D_CAR 4.9, DELTA 4.85: just under

        ("NO-MERGE-HARMS", _levels(69.5, 64.6, 64.6), {}, 0),                     # both 4.9 below

    chk("MCAR-GSTAR-PINS" in _e_score(_e_build(root, cache, _levels(69.5, 60.0, 68.5)), csvs["no-corpus"])[1]
        and "MCTL-GSTAR-FREE" in _e_score(_e_build(root, cache, _levels(69.5, 60.0, 68.5)), csvs["no-corpus"])[1],

    # harness breaks
    breaks = [
        ("wrong spec on the ARGS line (MCAR ran MCTL's)", {"over": {("MCAR", 124): {"args": expected_args("MCTL", 124, _E_SAVE).replace("cmg1-MCTL", "cmg1-MCAR")}}}),
        ("repeated flag", {"over": {("KLS", 125): {"args": expected_args("KLS", 125, _E_SAVE) + " --alpha0 1e-3"}}}),

        ("ENV BETA_CLIP none", {"over": {("KLS", 126): {"env": ENV_EXPECTED.replace("BETA_CLIP=-15:-2.3026", "BETA_CLIP=none").replace(" EB_RHO", " PROBE_DIR=%s/probe_cmg1-KLS-s126 EB_RHO" % _E_SAVE)}}}),
        ("PROBE_DIR of another run", {"over": {("KLS", 126): {"env": ENV_EXPECTED.replace(" EB_RHO", " PROBE_DIR=%s/probe_cmg1-MCAR-s126 EB_RHO" % _E_SAVE)}}}),
        ("PROBE_TENSOR missing", {"over": {("MCAR", 126): {"pt": "Files already downloaded and verified"}}}),
        ("PROBE_TENSOR type=layerwise", {"over": {("KLS", 124): {"pt": "PROBE_TENSOR: on every=100 type=layerwise tensors=62 meta_alg=Lion momentum_param=0.99 Lion_beta2=0.9 dir=/x"}}}),
        ("a VOTE_W witness (wrong tree)", {"over": {("MCTL", 125): {"foreign": "VOTE_W: off"}}}),
        ("a SHADOW_VOTE witness", {"over": {("MCTL", 125): {"foreign": "SHADOW_VOTE: on x"}}}),

    ]

        d = _e_build(root, cache, _levels(69.5, 60.0, 68.5), **kw)
        rc, fin, out = _e_score(d, csvs["no-corpus"])
        chk(rc == 1 and fin.startswith("FINAL: HARNESS-UNSOUND"), "HARNESS break: %s -> HARNESS-UNSOUND, exit 1" % lab, (fin or "")[:90])
    pbreaks = [
        ("MCAR's records grouped as MCTL (swapped)", {("MCAR", 124): "MCTL"}),
        ("MCTL's records grouped as MCAR", {("MCTL", 127): "MCAR"}),

    d = _e_build(root, cache, _levels(69.5, 60.0, 68.5))
    pd = os.path.join(d, PREFIX, "probe_cmg1-MCTL-s125")
    os.unlink(pd)
    shutil.copytree(_e_probe_dir(cache, "MCTL", "MCTL", False), pd)

    print("\nG. O2 -- the FINAL line and every bar line are invariant across four corpora (injected rows at cmg1's cells)")

        rc, fin, out = _e_score(d, p)
        finals[nm] = fin

    chk(len(set(finals.values())) == 1, "O2 CARRIER-MERGE-HARMS FINAL identical across %s" % ", ".join(sorted(finals)))

    d = _e_build(root, cache, _levels(69.5, 60.0, 68.5))
    p = subprocess.run([sys.executable, os.path.abspath(__file__), d], capture_output=True, text=True)
    chk(p.returncode == 0 and "FINAL: CARRIER-MERGE-HARMS" in p.stdout, "`python3 cMG1_mergecarrier_score.py <runsdir>` exits 0 on synthetic runs")
    d = _e_build(root, cache, _levels(69.5, 60.0, 68.5), drop=("KLS", 127))
    p = subprocess.run([sys.executable, os.path.abspath(__file__), d], capture_output=True, text=True)
    chk(p.returncode == 2 and "FINAL: INCOMPLETE" in p.stdout, "... and on a dropped run exits 2")
    shutil.rmtree(root, ignore_errors=True)

    print("\nI. RULE 21 PREMISE: no cmg1 run may exist yet")
    runs, _n = read_runs(runsdir) if runsdir else ({}, [])
    chk(not runs, "no cmg1-*.out under %s" % runsdir, "%d found" % len(runs))

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))
    for f in GATE_FAIL:
        print("   FAIL", f)
    return 1 if GATE_FAIL else 0


def main():
    ap = argparse.ArgumentParser(description="cmg1 registered scorer.  Documented invocation: "
                                             "python3 analysis/cMG1_mergecarrier_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cmg1/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc

    sys.exit(main())
