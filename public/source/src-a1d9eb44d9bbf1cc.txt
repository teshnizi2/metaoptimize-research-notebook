#!/bin/bash
# =============================================================================
# cGN1_stage_harness.sh -- BUILD OR VERIFY THE ISOLATED `cgn1` HARNESS TREE.
#   (CORRECTIONS 212.)  Touches NO existing file.  Submits nothing.
#
# WHY AN ISOLATED TREE AND NOT THE LIVE ONE.  `cgn1` needs one new network name,
# `ResNet18_gn_c100` (patches/patch_resnet_gn_c100.py -- one inserted dispatch
# region).  Measured on cluster_b at registration, on a scratch copy of the live
# tree: with that region in the LIVE build_network.py,
#     tests/test_vggbn2.py V0f FAILS   (67 PASS / 1 FAIL; the unpatched live
#                                        tree passes 68 / 0)
# because V0f proves PATCH_VGGBN additive by deleting ITS two regions and
# comparing to build_network.py.pre_vggbn byte for byte -- any other region
# breaks that.  After a cgn1 ingest V1 fails as well (`ResNet18_gn_c100` raises
# on --pre).  test_vggbn2.py is guard 1d3 of the registered VGG launchers
# (bin/cVK2_vgg_cut_ladder.sh), so a live apply would disable every VGG batch.
# The isolated tree leaves the live tree BYTE-UNTOUCHED.
#
# WHAT IT BUILDS (under $METAOPT_WS, every path new and cgn1-namespaced):
#   harness_cgn1/cifar10/                a copy of the live harness's *.py files
#                                        (train.py, load_data.py, tin_data.py,
#                                        build_network.py, Optimizers/*.py),
#                                        byte-identical to the live tree
#   harness_cgn1/cifar10/data  ->        SYMLINK to the live tree's data/ (no copy)
#   harness_cgn1/cifar10/build_network.py.pre_gn_c100
#                                        the pinned pre-patch bytes (sha below)
#   harness_cgn1/cifar10/build_network.py  = pre + the ONE patch region
#   jobs/run_cifar_cgn1.sh               jobs/run_cifar.sh with EXACTLY ONE line
#                                        changed: the `cd` into the harness
#   harness_cgn1/STAGE-MANIFEST.txt      sha256 of every staged file
#
# THE PINNED PRE-PATCH FILE.  The patch was proved against the live
# build_network.py of 2026-09-15, sha256 PRE_SHA below.  If the live file has
# moved since (another track's patch), point CGN1_PRE_SRC at a file carrying
# exactly those bytes (e.g. that track's .pre_<patch> backup); the sha is
# checked either way.
#
# USAGE
#   bash bin/cGN1_stage_harness.sh            # VERIFY ONLY (exit 0 iff consistent)
#   bash bin/cGN1_stage_harness.sh --stage    # build it (refuses if it exists)
# =============================================================================
set -u
MODE=verify; [ "${1:-}" = "--stage" ] && MODE=stage
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HROOT=$WS/harness_cgn1
DEST=$HROOT/cifar10
RUNNER_SRC=$WS/jobs/run_cifar.sh
RUNNER=$WS/jobs/run_cifar_cgn1.sh
PATCH=$REPO/patches/patch_resnet_gn_c100.py
PRE_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
POST_SHA=9f6e4ec97157311bb3b664509258144235319db88962b9d8c3a4484b1837c892
PRE_SRC=${CGN1_PRE_SRC:-$LIVE/build_network.py}
FAIL=0; bad() { echo "!!! STAGE FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

# the harness files that are COPIED (everything else under the live tree -- data,
# backups, __pycache__, .DS_Store -- is NOT a harness source file)
live_py() { (cd "$LIVE" && find . -name '*.py' -not -path '*/__pycache__/*' -not -path './data/*' | sort); }

echo "cGN1_stage_harness: mode=$MODE  WS=$WS"
echo "  live tree   $LIVE"
echo "  cgn1 tree   $DEST"
echo "  runner      $RUNNER  (from $RUNNER_SRC)"

if [ "$MODE" = stage ]; then
  [ -e "$HROOT" ] && { echo "!!! $HROOT already exists -- refusing to overwrite. Verify it instead."; exit 2; }
  [ -e "$RUNNER" ] && { echo "!!! $RUNNER already exists -- refusing to overwrite."; exit 2; }
  [ -f "$PATCH" ] || { echo "!!! no patch at $PATCH"; exit 2; }
  [ "$(sha "$PRE_SRC")" = "$PRE_SHA" ] || { echo "!!! $PRE_SRC sha $(sha "$PRE_SRC") != pinned $PRE_SHA"; echo "!!! set CGN1_PRE_SRC to a file with the pinned bytes"; exit 2; }
  mkdir -p "$DEST" || exit 2
  for f in $(live_py); do
    mkdir -p "$DEST/$(dirname "$f")"; cp -p "$LIVE/$f" "$DEST/$f" || exit 2
  done
  cp -p "$PRE_SRC" "$DEST/build_network.py" || exit 2
  cp -p "$PRE_SRC" "$DEST/build_network.py.pre_gn_c100" || exit 2
  ln -s "$LIVE/data" "$DEST/data" || exit 2
  BUILD_NET_PATH="$DEST/build_network.py" python3 "$PATCH" || { echo "!!! patch failed"; exit 2; }
  sed "s#^cd $LIVE\$#cd $DEST#" "$RUNNER_SRC" > "$RUNNER" || exit 2
  chmod --reference="$RUNNER_SRC" "$RUNNER" 2>/dev/null || chmod 755 "$RUNNER"
  { echo "STAGED_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "LIVE $LIVE"; echo "DEST $DEST"
    echo "PRE_SRC $PRE_SRC"; echo "PATCH_SHA256 $(sha "$PATCH")"
    for f in $(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' | sort); do
      echo "FILE $(sha "$DEST/$f") $f"; done
    echo "RUNNER $(sha "$RUNNER") $RUNNER"
    echo "RUNNER_SRC $(sha "$RUNNER_SRC") $RUNNER_SRC"; } > "$HROOT/STAGE-MANIFEST.txt"
  echo "staged; now verifying"
fi

# ---------------------------------------------------------------- VERIFY ------
[ -d "$DEST" ] || { echo "!!! no cgn1 tree at $DEST -- run with --stage"; exit 2; }
# V-a: every live harness .py has a byte-identical copy, EXCEPT build_network.py
N=0
for f in $(live_py); do
  [ "$f" = "./build_network.py" ] && continue
  if [ ! -f "$DEST/$f" ]; then bad "V-a $f missing from the cgn1 tree"; continue; fi
  [ "$(sha "$LIVE/$f")" = "$(sha "$DEST/$f")" ] && N=$((N+1)) || bad "V-a $f differs from the LIVE tree (re-stage)"
done
echo "V-a: $N live harness .py files byte-identical in the cgn1 tree (build_network.py excepted)"
# V-a2: the cgn1 tree carries no .py the live tree lacks, other than none
EXTRA=$(cd "$DEST" && find . -name '*.py' -not -path '*/__pycache__/*' -not -path './data/*' | sort | while read -r f; do [ -f "$LIVE/$f" ] || echo "$f"; done)
[ -z "$EXTRA" ] && echo "V-a2: no .py in the cgn1 tree that the live tree lacks" || bad "V-a2 extra .py in the cgn1 tree: $EXTRA"
# V-b: the pre snapshot is the pinned bytes; build_network.py is the pinned post
[ "$(sha "$DEST/build_network.py.pre_gn_c100")" = "$PRE_SHA" ] && echo "V-b: pre snapshot sha == pinned $PRE_SHA" || bad "V-b pre snapshot sha != pinned"
[ "$(sha "$DEST/build_network.py")" = "$POST_SHA" ] && echo "V-b2: cgn1 build_network.py sha == pinned post $POST_SHA" || bad "V-b2 cgn1 build_network.py sha $(sha "$DEST/build_network.py") != pinned post $POST_SHA"
LSHA=$(sha "$LIVE/build_network.py")
[ "$LSHA" = "$PRE_SHA" ] && echo "V-b3: the LIVE build_network.py is still the pinned pre (untouched)" \
  || echo "V-b3: NOTE -- the LIVE build_network.py has moved since registration ($LSHA); the cgn1 tree keeps the pinned pre. Not a failure: V-a still binds every OTHER harness file."
grep -q PATCH_RESNET_GN_C100 "$LIVE/build_network.py" && bad "V-b4 the LIVE build_network.py carries PATCH_RESNET_GN_C100 -- it must not" \
  || echo "V-b4: the LIVE build_network.py carries NO PATCH_RESNET_GN_C100 marker"
# V-c: data is a symlink to the live data
[ -L "$DEST/data" ] && [ "$(readlink "$DEST/data")" = "$LIVE/data" ] && echo "V-c: data -> $LIVE/data (symlink)" || bad "V-c data is not a symlink to $LIVE/data"
[ -d "$DEST/data/cifar-100-python" ] && echo "V-c2: CIFAR-100 visible through the symlink" || bad "V-c2 cifar-100-python not visible"
# V-d: the runner differs from run_cifar.sh in EXACTLY the cd line
if [ -f "$RUNNER" ]; then
  D=$(diff "$RUNNER_SRC" "$RUNNER" | grep -c '^[<>]')
  DL=$(diff "$RUNNER_SRC" "$RUNNER" | grep '^[<>]')
  if [ "$D" = 2 ] && echo "$DL" | grep -qx "< cd $LIVE" && echo "$DL" | grep -qx "> cd $DEST"; then
    echo "V-d: $RUNNER differs from run_cifar.sh in EXACTLY one line: cd $LIVE -> cd $DEST"
  else
    bad "V-d the runner diff is not exactly the cd line:"; diff "$RUNNER_SRC" "$RUNNER"
  fi
else
  bad "V-d no runner at $RUNNER"
fi
[ "$FAIL" = 0 ] && { echo "cGN1_stage_harness: VERIFIED"; exit 0; } || { echo "cGN1_stage_harness: NOT CONSISTENT"; exit 2; }
