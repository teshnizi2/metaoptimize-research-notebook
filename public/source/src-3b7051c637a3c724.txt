# cDP1_cdep1_depth_score.py -- THE REGISTERED SCORER FOR `cdep1`:
# IS THE `ciso1` RESCUE DRIVEN BY TENSOR *IDENTITY* OR BY NETWORK *DEPTH*?
#
# RULE 21: this file is git-committed BEFORE any `cdep1` run exists; the
# CORRECTIONS entry states the commit time, the earliest sacct Submit and the
# margin in seconds.  RULE 16: it is run UNEDITED.  If it is broken it is
# FROZEN and a successor is registered (precedent cN1/cN2 at 149, cdn1/cdn2
# at 175).
#
# THE DOCUMENTED INVOCATION (no second argument to forget; --manifest is
# OPTIONAL and DEFAULTED -- CORRECTIONS 164.2 / 165.7):
#     python3 analysis/cDP1_cdep1_depth_score.py <runsdir>
#     python3 analysis/cDP1_cdep1_depth_score.py --selftest [--runsdir <runsdir>]
#
# =============================================================================
# THE QUESTION, STATED BEFORE ANY NUMBER IT PRODUCES IS READ
# =============================================================================
#   `ciso1` (CORRECTIONS 187) measured, at the standard CIFAR-100 cell, that
#   moving three named 512-parameter layer4 BatchNorm SCALES (1-based tensors
#   50 `layer4.0.bn2.weight`, 53 `layer4.0.shortcut.1.weight`, 59
#   `layer4.1.bn2.weight`) out of the shared step-size group into their own
#   lifted plateau5 from 23.2807 to 70.2113 -- +46.9307 pp, the WHOLE in-batch
#   scalar->layerwise gap -- while isolating their layer-2 structural
#   homologues (20, 23, 29) gave -0.0633 pp.
#
#   `187.4` records the binding limitation: `ISO` and `CTRL` differ
#   SIMULTANEOUSLY in tensor identity, in network depth (stage 4 vs stage 2)
#   AND in isolated parameter count (1,536 vs 384).  Nothing in `ciso1` breaks
#   that three-way confound, so `ciso1` licenses "isolating THESE three rescues
#   and isolating THOSE three does not" and does NOT license "it is their
#   identity, not their depth".
#
#   `cdep1` is the depth control.  `DEPTH` isolates three 512-parameter
#   layer-4 BatchNorm parameters that contain NO carrier, at group sizes
#   [59,3] and isolated numel 1,536 -- IDENTICAL to `ISO` on group sizes, on
#   isolated parameter count and on network depth, differing ONLY in WHICH
#   layer-4 512-parameter BatchNorm parameters are isolated.
#
# =============================================================================
# THE ARMS (x seeds 24, 25, 26; ONE submission; PROBE=100, PROBE_TENSOR=1)
# =============================================================================
#   k01     scalar                                m=1      FLOOR anchor, in batch
#   k62     layerwise                             m=62     CEILING anchor, in batch
#   ISO     sets:<59 others>/50,53,59   [59,3]    THE POSITIVE CONTROL, re-run
#                                                 in batch at FRESH seeds.
#   DEPTH   sets:<59 others>/47,48,56   [59,3]    THE ARM.  layer4.0.bn1.weight,
#                                                 layer4.0.bn1.bias,
#                                                 layer4.1.bn1.weight.
#   DEPTH2  sets:<60 others>/47,56      [60,2]    THE CLASS-PURE VARIANT:
#                                                 the only two non-carrier BN
#                                                 SCALES in layer4.  Matched on
#                                                 class and depth, unmatched on
#                                                 count and numel (1,024).
#   ONE     sets:<61 others>/50         [61,1]    THE CONTENT-FREE RIVAL'S
#                                                 KILLER.  See "H-BROKEN" below.
#
#   WHY `ISO` IS RE-RUN RATHER THAN SPLICED.  BATCH is the unit of replication
#   in this campaign (F(62,85)=5.47) and SEED is null (F(11,1976)=0.152).  A
#   `DEPTH` measured here against a `ciso1` `ISO` would be a cross-batch
#   contrast on the campaign's own largest variance component.  Every contrast
#   below is WITHIN batch.  Cost of that decision: 3 jobs, about 2.2 GPU-h.
#
#   WHY `ONE` IS IN THE BATCH -- THIS IS THE `164.6` LESSON APPLIED.  The modal
#   predicted outcome is that `DEPTH` lands AT THE FLOOR.  A floored arm is
#   consistent with IDENTITY-OPERATIVE and equally consistent with the
#   content-free rival
#       H-BROKEN: "any two-group partition other than the exact `ISO` set
#                  floors, for reasons having nothing to do with which tensors
#                  are isolated."
#   `cpr1`'s headline died at `164.6` because exactly this rival was never
#   registered and its floored treatment arm could not be told from "it died".
#   `ONE` isolates a DIFFERENT set (tensor 50 alone) at a DIFFERENT size
#   ([61,1], 512 params) and rescued by +41.4693 pp in `ciso1`.  If `ONE`
#   rescues here and `DEPTH` floors, H-BROKEN is dead IN BATCH and the floored
#   `DEPTH` is a real null rather than an artefact of "not being `ISO`".
#   3 jobs, about 2.2 GPU-h, and without it the modal outcome is unreadable.
#
# =============================================================================
# THE PARTITIONS, RE-DERIVED FROM THE LIVE MODEL, NOT INHERITED (`187.4` notes
# that the spec string as first written there was wrong and corrected in place;
# every string below was re-derived here from the live 62-tensor manifest and
# is re-proved by --selftest and by the launcher's guard 4 on the live model)
# =============================================================================
#   Every 512-parameter tensor on ResNet18_c100, verified by name on the live
#   model (build_network("ResNet18_c100")) on 2026-09-09:
#       47 layer4.0.bn1.weight        512  BN SCALE      non-carrier
#       48 layer4.0.bn1.bias          512  BN BIAS       non-carrier
#       50 layer4.0.bn2.weight        512  BN SCALE  <== CARRIER
#       51 layer4.0.bn2.bias          512  BN BIAS       non-carrier
#       53 layer4.0.shortcut.1.weight 512  BN SCALE  <== CARRIER
#       54 layer4.0.shortcut.1.bias   512  BN BIAS       non-carrier
#       56 layer4.1.bn1.weight        512  BN SCALE      non-carrier
#       57 layer4.1.bn1.bias          512  BN BIAS       non-carrier
#       59 layer4.1.bn2.weight        512  BN SCALE  <== CARRIER
#       60 layer4.1.bn2.bias          512  BN BIAS       non-carrier
#   ResNet-18 has exactly FIVE BatchNorm scales in layer4 (47, 50, 53, 56, 59)
#   and every 512-wide BN scale in the whole model is one of them (layer1 64,
#   layer2 128, layer3 256).  THREE are carriers.  A three-element arm that is
#   class-matched (BN scale), depth-matched (layer4), numel-matched (1,536) and
#   carrier-free therefore DOES NOT EXIST on this model.  This is re-proved in
#   --selftest section A, not asserted.
#
#   THE THIRD MEMBER, CHOSEN BY A RULE STATED BEFORE THE ARM WAS FIXED.  Among
#   the five 512-parameter layer-4 non-carriers that are not 47 or 56
#   ({48, 51, 54, 57, 60}), 51 / 54 / 60 are the affine BIAS PARTNERS of the
#   three carriers 50 / 53 / 59 and are excluded so that no carrier's own
#   partner sits in the isolated group; that leaves {48, 57}, the bias partners
#   of the two non-carrier scales, which are indistinguishable on every
#   measured quantity.  48 is taken -- it completes the affine pair of 47 and
#   it is the index `187.4` put on record, so the arm is neither inflated nor
#   deflated by a late choice.  57 was an equally admissible alternative.
#
#   THE SPEC STRINGS, RE-DERIVED:
#     DEPTH  isolates {47,48,56}; complement {1..46} u {49..55} u {57..62}
#            = 46 + 7 + 6 = 59 tensors ->
#            sets:1-46,49-55,57-62/layer4.0.bn1.weight,layer4.0.bn1.bias,layer4.1.bn1.weight
#     DEPTH2 isolates {47,56};    complement {1..46} u {48..55} u {57..62}
#            = 46 + 8 + 6 = 60 tensors ->
#            sets:1-46,48-55,57-62/layer4.0.bn1.weight,layer4.1.bn1.weight
#   The isolated group is written by NAME and the coarse group by INDEX RANGES,
#   so an off-by-one in either half is a duplicate or a gap and the harness's
#   PATCH_NAMESETS parser raises at parse time.
#
# =============================================================================
# WHAT THIS DESIGN CONTROLS FOR, AND WHAT IT DOES **NOT** -- STATED BEFORE ANY
# RESULT EXISTS, BECAUSE IT BINDS EVERY SENTENCE WRITTEN ABOUT THE OUTCOME
# =============================================================================
#   CONTROLLED.  Group count (m=2), coarse-group size (59), isolated-group size
#   (3), isolated parameter count (1,536), network depth (every isolated tensor
#   is in layer4, the last stage), parameter WIDTH (512 each), normalisation
#   layer type (all BatchNorm affine parameters), batch, cell, horizon, seeds.
#
#   NOT CONTROLLED, AND ARCHITECTURALLY UNCONTROLLABLE ON THIS MODEL:
#   (1) PARAMETER CLASS.  `DEPTH`'s third member is a BN BIAS, because only TWO
#       non-carrier BN scales exist in layer4.  `DEPTH2` removes this at the
#       cost of the count/numel match; running both is how the batch answers
#       the "the bias ruined it" objection IN BATCH rather than in prose.
#   (2) DYNAMICAL MAGNITUDE.  On `ciso1`'s own scalar records (1,500
#       determinate / 945 pinned, pooled over 3 seeds; re-derived by --selftest)
#       the mean |L_i| = |0.9 m_i + 0.1 z_i| of the three carriers is
#       2.81e-01 / 1.83e-01 / 3.19e-01 (ranks 2 / 3 / 1 of 62), while EVERY
#       512-parameter layer-4 non-carrier sits between 1.73e-06 and 1.29e-03
#       (ranks 33 to 62) against a median tensor of 1.94e-03.  The best
#       possible three-member non-carrier set sums to 3.48e-03 against the
#       carriers' 7.83e-01 -- a factor of 225.  NO choice of members closes
#       that gap.  Consequence, binding: a null `DEPTH` is consistent with
#       "identity matters" AND with "magnitude matters", and this batch CANNOT
#       separate those two.  What it CAN kill is "DEPTH alone" -- the claim
#       that isolating ANY three matched layer-4 512-parameter BN parameters
#       rescues.  That, and only that, is the registered question.
#
# =============================================================================
# THE NOISE FLOOR, RE-DERIVED AT REGISTRATION FROM THE 2,761-ROW CORPUS, WITH
# `cdep1-` EXCLUDED FROM EVERY READER (the 165.4 pattern: every frozen premise
# is invariant under this batch's own ingest).  NOTHING IS COPIED FROM `ciso1`.
# =============================================================================
#   cR1's estimator: pooled within-(batch x granularity) SD of plateau5 over
#   m=2 arms in this exact cell at 100 epochs, superseded and collapsed rows
#   dropped, re-derived on 2026-09-09 at 2,761 rows:
#     SIGMA_NARROW  (^\[\d+,\d+\]$ matcher)          0.925518  df 60  30 cells   90 members
#     SIGMA_WIDE    (matcher extended to sets: m=2)  0.863472  df 84  42 cells  126 members
#     SIGMA_W = max                                  0.925518
#   SE_ARM_DIFF = SIGMA_W * sqrt(2/3) = 0.755682   (3-seed arm vs 3-seed arm)
#   NOTE, DISCLOSED: SIGMA_NARROW is unchanged from `ciso1`'s registration
#   (`ciso1` added no [a,b] granularity rows), while SIGMA_WIDE has DRIFTED
#   0.889041 -> 0.863472 with `ciso1`'s five sets: cells ingested.  The max is
#   the narrow one either way, so `cdep1`'s bars are directly commensurable
#   with `ciso1`'s -- which is why the pp bars below are deliberately the same
#   numbers.  This was verified, not assumed (sigma_w drifted after
#   registration twice, `156` and `159`).
#
# =============================================================================
# THE BARS, FROZEN, IN pp AND IN SE
# =============================================================================
#   LEVELS (3-seed means of plateau5 = mean test accuracy over epochs 95..99,
#   read from each run's OWN .out; the CSV `plateau` column is BANNED):
#     F = k01     C = k62     I = ISO     D = DEPTH     D2 = DEPTH2   O = ONE
#   CONTRASTS:
#     D_ISO    = I  - F        D_DEPTH  = D  - F        D_DEPTH2 = D2 - F
#     D_ONE    = O  - F        D_CEIL   = C  - F
#     DELTA_ID = I  - D        <== ***THE PRIMARY STATISTIC***
#     DELTA_ID2= I  - D2       DELTA_BIAS = D - D2
#
#   WHY `DELTA_ID` IS PRIMARY AND NOT `D_DEPTH`.  `164.6`: an arm whose
#   predicted level sits AT a saturation boundary cannot disagree with the
#   account that predicts it.  Under IDENTITY-OPERATIVE `D_DEPTH` is predicted
#   at 0.00 -- ON the floor.  `DELTA_ID` is a difference of two arms and is
#   bounded in NEITHER direction: IDENTITY-OPERATIVE predicts +47.30 pp
#   (+62.6 SE), DEPTH-OPERATIVE predicts 0.00 pp, and the measurement is free
#   to land anywhere including negative.  The primary read therefore has room
#   to disagree with both live accounts.  `D_DEPTH` is retained as the
#   branch-mapping statistic with the BOUND discipline registered below.
#
#   SUPPORT_BAR = +25.0 pp = 33.08 SE.  Carried over from `ciso1` DELIBERATELY
#       so the two batches are commensurable, and re-justified here: the
#       largest rescue any m=2 partition of this cell has produced is
#       k49 - k01 = 55.4688 - 22.9141 = +32.55 pp; +25 is 77 % of it.
#   REFUTE_BAR  = +5.0 pp = 6.62 SE.  The worst non-floor m=2 configuration on
#       the corpus ladder is k50 at 30.3370 = +7.42 pp over the floor; +5 sits
#       below it.
#   READ_BAR    = 2 SE = 1.511364 pp.  |x| <= READ_BAR is "indistinguishable".
#   IDENTITY_BAR = SUPPORT_BAR = 25.0 pp, applied to DELTA_ID.
#   DEAD_BAR    = 5.00 pp: a within-arm seed RANGE above it in k01, ISO or
#       DEPTH is UNRESOLVED-DIVERGED (a bimodal arm must be reported, not
#       averaged).
#   Anchor bands (provenance, re-derived at score time, cdep1 excluded):
#       k01 inside 22.914057 +/- 4 x 0.570874 = [20.6306, 25.1976]  (n 35, 12 cells)
#       k62 inside 69.443586 +/- 4 x 0.512042 = [67.3954, 71.4918]  (n 29, 10 cells)
#
# =============================================================================
# THE FLOOR / SATURATION REGISTRATION -- `164.6`'s DEFECT, DESIGNED OUT IN THE
# ONLY WAY IT CAN BE FOR A NULL CONTROL: BY REGISTERING THE BOUND IN ADVANCE
# =============================================================================
#   PREDICTED LEVELS AND MARGINS OVER THE PREDICTED IN-BATCH FLOOR (22.914057,
#   the corpus's 35 same-cell scalar rows over 12 batches, cdep1 excluded):
#
#     arm    | IDENTITY-OPERATIVE | DEPTH-OPERATIVE | BOTH (midpoint) | worst margin
#     -------|--------------------|-----------------|-----------------|-------------
#     k01    | 22.914             | 22.914          | 22.914          | it IS the floor
#     k62    | 69.444             | 69.444          | 69.444          | +46.53 = +61.6 SE
#     ISO    | 70.211             | 70.211          | 70.211          | +47.30 = +62.6 SE
#     ONE    | 64.750             | 64.750          | 64.750          | +41.84 = +55.4 SE
#     DEPTH  | 22.914  AT FLOOR   | 70.211          | ~46.56          | 0.00 (IDENTITY)
#     DEPTH2 | 22.914  AT FLOOR   | 70.211          | ~46.56          | 0.00 (IDENTITY)
#
#   TWO ARMS ARE PREDICTED AT THE FLOOR UNDER A LIVE ACCOUNT.  That is not
#   avoidable: the predicted level of a null control IS the null level.  What
#   `164.6` forbids is treating a floored arm's AGREEMENT with a point
#   prediction as evidence.  Registered here, in advance:
#     (a) THE PRIMARY READ IS NOT FLOOR-BOUNDED.  DELTA_ID's worst-case
#         distance from any boundary is +47.30 pp under IDENTITY-OPERATIVE and
#         0.00 pp under DEPTH-OPERATIVE, and it may take any real value.
#     (b) A FLOORED `DEPTH` ENTITLES A BOUND, NEVER A MAGNITUDE.  If
#         D_DEPTH <= REFUTE_BAR the scorer stamps DEPTH-FLOOR-SATURATED and
#         what may be written is "D_DEPTH <= 5.0 pp" and "DELTA_ID >= 25.0 pp".
#         What may NOT be written: the point value of D_DEPTH as an effect
#         size, any residual against 22.914 as agreement, or the phrase
#         "DEPTH floored exactly as IDENTITY-OPERATIVE predicted".
#     (c) THE SATURATION BAND IS REGISTERED FROM THE CORPUS NOW, so no band is
#         chosen after seeing the result: the 35 same-cell scalar rows plus
#         every m=2 cell whose mean is at or below floor + REFUTE_BAR =
#         27.914057 (42 further rows over 14 cells) run [21.0940, 27.9520],
#         width 6.8580 pp = 9.08 SE.  ANY level inside that band is
#         unrejectable against the floor at the 2-SE bar.
#     (d) THE CONTENT-FREE RIVAL IS NAMED AND MADE TESTABLE IN BATCH.  See
#         H-BROKEN above and gate [H] below: `ONE` is the arm that kills it.
#
# =============================================================================
# THE MECHANISM PREMISE, MEASURED BEFORE THIS FILE WAS COMMITTED
# =============================================================================
#   On `ciso1`'s own k01 (scalar) PROBE_TENSOR records, pooled over its three
#   seeds -- 1,500 determinate records, 945 pinned -- with
#   L_i = 0.9 m_i + 0.1 z_i and s_agg = sign(0.9 m_agg + 0.1 z_agg), the share
#   of records on which REMOVING a set S flips the remainder's sign
#   (`187.6`'s statistic, which that entry established is the operative one --
#   NOT domination):
#       S = {50,53,59}  flip 0.8260 determinate / 1.0000 pinned  (dominates 0.9714)
#       S = {50}        flip 0.8000 / 1.0000                     (dominates 0.0000)
#       S = {47,48,56}  flip 0.0000 / 0.0000                     (dominates 0.0000)
#       S = {47,56}     flip 0.0000 / 0.0000                     (dominates 0.0000)
#       S = {20,23,29}  flip 0.0000 / 0.0000                     (ciso1's CTRL)
#   So the sign account predicts `DEPTH` and `DEPTH2` NULL, ex ante, on the
#   statistic `187.6` established.  --selftest re-derives all five shares from
#   `ciso1`'s records when they are reachable; the score run re-derives the
#   in-batch versions on `cdep1`'s OWN k01 and stamps PREMISE-REPLICATES.
#   THIS IS A PREDICTION, NOT A DEMONSTRATION: it is measured on the COLLAPSED
#   scalar trajectory and says what the shared group's vote would have been,
#   not what a run with a different partition does.
#
#   A SECOND REGISTERED PREDICTION, FROM THE SAME RECORDS:  `DEPTH` and
#   `DEPTH2` should be INDISTINGUISHABLE (|DELTA_BIAS| <= READ_BAR), because
#   the bias member 48 contributes 1.73e-06 of the isolated set's 2.2504e-03
#   mean-|L| mass -- 0.08 % -- and the flip share is 0.0000 for both sets.  If
#   they differ by more than READ_BAR, something outside the sign account is
#   operating and the batch says so.
#
# =============================================================================
# THE BRANCH MAP, FIRST MATCH WINS
# =============================================================================
#   UNRESOLVED-MANIFEST-MISMATCH   no manifest; live model != the frozen 62
#                                  (index, name, numel); any arm's group index
#                                  sets != registered; sizes != [59,3] (ISO),
#                                  [59,3] (DEPTH), [60,2] (DEPTH2), [61,1]
#                                  (ONE); ISO ^ DEPTH != exactly the six names.
#   UNRESOLVED-PROVENANCE          ARGS / ENV / PROBE_TENSOR line not the
#                                  registered one; run count != 18; missing
#                                  probe records or probe_tensor.json; the
#                                  decomposition (group sums of z_tensor /
#                                  m_tensor vs the harness's own z / momentum)
#                                  fails on > 1 % of records.
#   UNRESOLVED-DIVERGED            no RUN_DONE / < 100 epochs / traceback on
#                                  any run; k01 or k62 outside its corpus band;
#                                  seed range > DEAD_BAR in k01, ISO or DEPTH.
#   UNRESOLVED-NO-POSITIVE-CONTROL D_ISO < SUPPORT_BAR.  The `ciso1` rescue did
#                                  not reproduce in batch at fresh seeds; no
#                                  DEPTH contrast is readable and the batch
#                                  says so instead of reading one.  Checked
#                                  BEFORE any DEPTH read.
#   DEPTH-OPERATIVE                D_DEPTH >= SUPPORT_BAR and |DELTA_ID| <=
#                                  READ_BAR.  Isolating three matched
#                                  carrier-free layer-4 512-parameter BN
#                                  parameters rescues just as well: the
#                                  `ciso1` result is about DEPTH (or about
#                                  the partition's shape), NOT about which
#                                  tensors.
#   IDENTITY-OPERATIVE             D_DEPTH <= REFUTE_BAR and DELTA_ID >=
#                                  IDENTITY_BAR.  A size-, numel- and
#                                  depth-matched carrier-free set does NOT
#                                  rescue: the rescue tracks WHICH tensors are
#                                  isolated.  (Subject to the magnitude caveat
#                                  above, which the FINAL line carries.)
#   BOTH-OPERATIVE                 D_DEPTH >= SUPPORT_BAR and DELTA_ID >
#                                  READ_BAR.  DEPTH rescues, but by
#                                  measurably less than ISO.  BOTH the depth /
#                                  shape of the partition AND the identity of
#                                  the isolated tensors contribute.  THIS IS A
#                                  RESULT, NOT A FAILURE.
#   UNRESOLVED-PARTIAL-RESCUE      REFUTE_BAR < D_DEPTH < SUPPORT_BAR.  DEPTH
#                                  lands between the floor and a rescue.
#                                  REGISTERED MEANING: both matter -- depth
#                                  buys a partial escape from the collapse and
#                                  identity buys the rest.  ALSO NOT A FAILURE;
#                                  the middle is a substantive outcome and the
#                                  scorer prints the fraction of the in-batch
#                                  gap recovered, D_DEPTH / D_ISO.
#   UNRESOLVED-PARTIAL             anything else (e.g. D_DEPTH <= REFUTE_BAR
#                                  with DELTA_ID < IDENTITY_BAR, which can
#                                  only happen if ISO itself is weak).
#
#   STAMPS (never the branch):
#     DEPTH-FLOOR-SATURATED | DEPTH-ABOVE-BAND   (the (b)/(c) bound discipline)
#     DEPTH2-RESCUES | DEPTH2-PARTIAL | DEPTH2-FLOOR
#     BIAS-NULL | BIAS-MATTERS                   (|DELTA_BIAS| vs READ_BAR)
#     H-BROKEN-DEAD | H-BROKEN-ALIVE             (gate [H]; ONE rescues or not)
#     ISO-REPLICATES | ISO-SHIFTED               (|I - 70.2113| vs 4*SIGMA_W)
#     ISO-59-FREE|PINS|NEITHER, ISO-3-...,       (the `187.7` beta reading,
#     DEPTH-59-..., DEPTH-3-..., DEPTH2-60-...,   re-measured at fresh seeds)
#     DEPTH2-2-..., ONE-61-...
#     PREMISE-REPLICATES | PREMISE-NOT-REPLICATED
#     FLOOR-NOT-INGESTED   when the CSV has no cdep1 row (the levels are read
#                          from the RAW .out files either way; the CSV is
#                          never the primary source here).
#
# plateau5 is PRIMARY; TRAIN is printed beside TEST at every arm and contrast;
# the CSV `plateau` column is not read anywhere; best_test is not read anywhere.
#
# DISCLOSURE.  `ciso1`'s 15 runs, its probe records, its PARTITION-MANIFEST,
# CORRECTIONS 187 and the corpus at 2,761 rows were all visible when this file
# was written.  The premise statistics quoted above were computed on `ciso1`'s

# =============================================================================

# the `sets:` grammar, re-implemented HERE so the registered index sets are a
# property of THIS file and not of the harness (G1 compares both)
# =============================================================================
def spec_to_groups(spec, names=TENSOR_NAMES):
    """`sets:<g>/<g>` -> list of 1-based index lists in model order, exactly the
    harness's rule (range, then int, then name; total, loud)."""
    if spec in ("scalar",):
        return [list(range(1, len(names) + 1))]
    if spec in ("layerwise",):
        return [[i] for i in range(1, len(names) + 1)]
    assert spec.startswith("sets:"), spec
    T = len(names)
    owner = {}
    out = []
    for gi, g in enumerate(spec[5:].split("/")):
        idx = []
        for item in g.split(","):
            m = re.match(r"^(\d+)-(\d+)$", item)
            if m:
                lo, hi = int(m.group(1)), int(m.group(2))
                assert 1 <= lo <= hi <= T, item
                got = list(range(lo, hi + 1))
            elif re.match(r"^\d+$", item):
                got = [int(item)]
            else:
                assert item in names, item
                got = [names.index(item) + 1]
            for i in got:
                assert i not in owner, (i, owner.get(i), gi)
                owner[i] = gi
            idx.extend(got)
        out.append(sorted(idx))
    assert len(owner) == T, (len(owner), T)
    return out


def tail5(d, budget=EPOCHS):
    v = [d[e] for e in range(budget - 5, budget) if e in d]
    return (sum(v) / 5.0) if len(v) == 5 else None


def carrying_set(X, s_agg):
    order = sorted(range(len(X)), key=lambda i: -abs(X[i]))
    tot_abs = sum(abs(x) for x in X)
    part, rem = 0.0, tot_abs
    for k, i in enumerate(order, 1):
        part += X[i]
        rem -= abs(X[i])
        if sgn(part) == s_agg and abs(part) > rem:
            return order[:k]
    return order

M2_NARROW = re.compile(r"^\[\d+,\d+\]$")


def _is_m2(g, wide):
    g = str(g or "")
    if M2_NARROW.match(g):
        return True
    if wide and g.startswith("sets:"):
        return g.count("/") == 1
    return False


def noise_floor_from_csv(path, wide):
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not _in_cell(r) or not _is_m2(r["granularity"], wide):
            continue
        try:
            cells[(_batch_of(r["run"]), r["granularity"])].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        nm += len(v)
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc, nm


def cell_level(path, grans):
    v, b = [], set()
    for r in _csv_rows(path):
        if _in_cell(r) and str(r["granularity"]) in grans:
            try:
                v.append(float(r["plateau5"]))
                b.add((_batch_of(r["run"]), r["granularity"]))
            except (TypeError, ValueError):
                pass
    if not v:
        return None
    return {"n": len(v), "mean": statistics.mean(v),
            "sd": statistics.stdev(v) if len(v) > 1 else float("nan"), "cells": len(b)}


def band(path, gran):
    L = cell_level(path, (gran,))
    if L is None or L["n"] < 2:
        return None
    return {"n": L["n"], "mean": L["mean"], "sd": L["sd"],
            "lo": L["mean"] - BAND_K * L["sd"], "hi": L["mean"] + BAND_K * L["sd"]}


def saturation_band(path):
    """The registered band: same-cell scalar rows plus every m=2 cell whose mean
    is at or below floor + REFUTE_BAR.  Re-derived, never copied."""
    sc = [float(r["plateau5"]) for r in _csv_rows(path)
          if _in_cell(r) and r["granularity"] == "scalar" and r["plateau5"] not in (None, "")]
    if not sc:
        return None
    cap = statistics.mean(sc) + REFUTE_BAR
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        if _in_cell(r) and _is_m2(r["granularity"], True):
            try:
                cells[(_batch_of(r["run"]), r["granularity"])].append(float(r["plateau5"]))
            except (TypeError, ValueError):
                pass
    extra = [x for v in cells.values() if statistics.mean(v) <= cap for x in v]
    allv = sc + extra
    return {"lo": min(allv), "hi": max(allv), "n_scalar": len(sc), "n_extra": len(extra),
            "n_cells": sum(1 for v in cells.values() if statistics.mean(v) <= cap), "cap": cap}

def read_manifest(path):
    man = {"tensors": [], "arms": {}, "meta": {}}
    for ln in open(path):
        p = ln.split()
        if not p:
            continue
        if p[0] == "TENSOR" and len(p) >= 4:
            man["tensors"].append((int(p[1]), p[2], int(p[3])))
        elif p[0] == "ARMSPEC" and len(p) >= 4:
            arm = p[1]
            d = {"spec": p[3], "m": int(p[5]), "sizes": tuple(int(x) for x in p[7].split(",")),
                 "params": tuple(int(x) for x in p[9].split(","))}
            i0 = p.index("GROUP0")
            i1 = p.index("GROUP1")
            d["group0"] = p[i0 + 1:i1]
            d["group1"] = p[i1 + 1:]
            man["arms"][arm] = d
        elif p[0] == "ISOLATED" and len(p) >= 3:
            man["meta"].setdefault("isolated", {})[p[1]] = tuple(int(x) for x in p[2].split(","))
        else:
            man["meta"][p[0]] = " ".join(p[1:])
    return man


def load_outs(runsdir):

    pats = [os.path.join(runsdir, PREFIX + "*.out"), os.path.join(runsdir, BATCH, PREFIX + "*.out")]

def occupancy(recs, k):
    n = sum(1 for r in recs if "beta" in r and len(r["beta"]) > k)
    at = sum(1 for r in recs if "beta" in r and len(r["beta"]) > k and r["beta"][k] <= BETA_LO + PIN_EPS)
    term = recs[-1]["beta"][k] if recs and "beta" in recs[-1] and len(recs[-1]["beta"]) > k else None
    return (at / n if n else None), term, n


def free_state(occ, term):
    """-> 'FREE' | 'PINS' | 'NEITHER' | 'UNKNOWN' by the registered rule."""
    if occ is None or term is None:
        return "UNKNOWN"
    if occ <= FREE_OCC_MAX and term > BETA_LO + FREE_TERMINAL_MARGIN:
        return "FREE"
    if occ >= PIN_OCC_MIN:
        return "PINS"
    return "NEITHER"


def level_token(d):
    if d is None:
        return "UNKNOWN"
    if d >= SUPPORT_BAR:
        return "RESCUES"
    if d <= REFUTE_BAR:
        return "FLOOR"
    return "PARTIAL"


def decide(d_iso, d_depth, delta_id, stamps):
    """The registered branch.  Gates (manifest, provenance, divergence) are
    applied by the caller before this is reached."""
    if d_iso is None or d_depth is None or delta_id is None:
        return "UNRESOLVED-PARTIAL"
    if d_iso < SUPPORT_BAR:
        return "UNRESOLVED-NO-POSITIVE-CONTROL"
    if d_depth >= SUPPORT_BAR:
        if abs(delta_id) <= READ_BAR:
            return "DEPTH-OPERATIVE"
        return "BOTH-OPERATIVE"
    if d_depth <= REFUTE_BAR:
        if delta_id >= IDENTITY_BAR:
            return "IDENTITY-OPERATIVE"
        return "UNRESOLVED-PARTIAL"
    return "UNRESOLVED-PARTIAL-RESCUE"


# =============================================================================
# the premise, re-derived from a scalar arm's PROBE_TENSOR records.
# `187.6` established that the OPERATIVE statistic is the sign of the
# remainder sum, not domination.  Both are reported; the flip share governs.
# =============================================================================
FLIP_SETS = {"TRIO": (50, 53, 59), "ONE": (50,), "DEPTH": (47, 48, 56), "DEPTH2": (47, 56)}


def premise_from_scalar_records(recs_by_seed):
    car = [i - 1 for i in FLIP_SETS["TRIO"]]
    n_det = n_dis = n_pin = 0
    dis_car = collections.Counter()
    flip_det = collections.Counter()
    flip_pin = collections.Counter()
    dom_pin = collections.Counter()
    absL = collections.defaultdict(float)
    absN = 0
    rest_up = 0
    for recs in recs_by_seed:
        for r in recs:
            try:
                z, m = r["z_tensor"], r["m_tensor"]
                zh, mh = float(r["z_agg"][0][0]), float(r["mom_pre"][0][0])
            except (KeyError, IndexError, TypeError):
                continue
            b2 = float(r.get("pt_b2", B2))
            L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
            s = sgn(b2 * mh + (1 - b2) * zh)
            if s == 0 or sgn(sum(L)) != s:
                continue
            pos = sum(1 for x in L if x > 0)
            neg = sum(1 for x in L if x < 0)
            smaj = sgn(pos - neg)
            if smaj == 0:
                continue
            n_det += 1
            if s != smaj:
                n_dis += 1
                for i in carrying_set(L, s):
                    dis_car[i] += 1
            pinned = (r.get("n_at_lo") == 1)
            if pinned:
                n_pin += 1
                absN += 1
                for i, x in enumerate(L):
                    absL[i] += abs(x)
                rest_up += sum(L[i] for i in range(len(L)) if i not in car) < 0
            for lab, S in FLIP_SETS.items():
                idx = [i - 1 for i in S]
                rem = sum(L[i] for i in range(len(L)) if i not in idx)
                ssum = sum(L[i] for i in idx)
                arest = sum(abs(L[i]) for i in range(len(L)) if i not in idx)
                if sgn(rem) != s:
                    flip_det[lab] += 1
                    if pinned:
                        flip_pin[lab] += 1
                if pinned and sgn(ssum) == s and abs(ssum) > arest:
                    dom_pin[lab] += 1
    shares = {TENSOR_NAMES[i]: (dis_car[i] / n_dis if n_dis else None) for i in car}
    top = [(TENSOR_NAMES[i], c / n_dis) for i, c in dis_car.most_common(6)] if n_dis else []
    mean_abs = {i + 1: (absL[i] / absN) for i in range(N_TENSORS)} if absN else {}
    return {"n_det": n_det, "n_dis": n_dis, "R_T": (n_dis / n_det if n_det else None),
            "shares": shares, "top": top, "pin_n": n_pin,
            "flip_det": {k: (flip_det[k] / n_det if n_det else None) for k in FLIP_SETS},
            "flip_pin": {k: (flip_pin[k] / n_pin if n_pin else None) for k in FLIP_SETS},
            "dom_pin": {k: (dom_pin[k] / n_pin if n_pin else None) for k in FLIP_SETS},
            "mean_abs": mean_abs,
            "rest_up": (rest_up / n_pin if n_pin else None)}


def premise_ok(P):
    """The in-batch premise rule: the carriers flip on >= 0.90 of PINNED
    records and DEPTH's own set flips on <= 0.10 of them."""
    fp = P["flip_pin"]
    if P["pin_n"] == 0 or fp["TRIO"] is None or fp["DEPTH"] is None:
        return False
    return fp["TRIO"] >= PREMISE_SHARE and fp["DEPTH"] <= PREMISE_NULL_SHARE

    names = list(TENSOR_NAMES)

    # ---- G1: the manifest ------------------------------------------------------------
    print("\n[G1] MANIFEST -- the live model and the partitions, as written at submit time")
    mp, how = resolve_manifest(runsdir, manifest)
    if mp is None:
        chk(False, "manifest found (%s)" % how)
        unresolved.append("UNRESOLVED-MANIFEST-MISMATCH")
    else:
        print("  manifest: %s (%s)" % (mp, how))
        man = read_manifest(mp)
        chk([(i, n, k) for i, n, k in man["tensors"]] == [(i + 1, names[i], TENSOR_NUMEL[i]) for i in range(N_TENSORS)],
            "live model == the frozen 62 (index, name, numel)")
        for arm in M2_ARMS:
            d = man["arms"].get(arm)
            reg = spec_to_groups(SPEC[arm])
            ok = d is not None and d["spec"] == SPEC[arm] and d["m"] == 2 and d["sizes"] == SIZES[arm] \
                and d["params"] == PARAMS[arm] \
                and [names.index(n) + 1 for n in d["group0"]] == reg[0] \
                and [names.index(n) + 1 for n in d["group1"]] == reg[1] \
                and tuple(reg[1]) == ISOLATED[arm]
            chk(ok, "%s: spec, m=2, sizes %s, params %s, both groups EXACTLY as registered; isolated %s"
                % (arm, SIZES[arm], PARAMS[arm], ISOLATED[arm]))
            if d:
                print("       GROUP1 = %s" % " ".join(d["group1"]))
        iso, dep, dep2, one = (man["arms"].get(a) for a in ("ISO", "DEPTH", "DEPTH2", "ONE"))
        if iso and dep:
            sd = set(iso["group1"]) ^ set(dep["group1"])
            chk(sd == set(CARRIERS) | set(DEPTH_SET) and len(iso["group1"]) == len(dep["group1"]) == 3
                and set(iso["group0"]) ^ set(dep["group0"]) == sd,
                "ISO ^ DEPTH == exactly the six names, three swapped at identical group sizes [59,3]",
                str(sorted(sd)))
            n_iso = [TENSOR_NUMEL[names.index(n)] for n in iso["group1"]]
            n_dep = [TENSOR_NUMEL[names.index(n)] for n in dep["group1"]]
            chk(sum(n_iso) == sum(n_dep) == 1536 and n_iso == n_dep == [512, 512, 512],
                "isolated numels MATCHED: ISO %s == DEPTH %s (1,536 each)" % (n_iso, n_dep))
            chk(all(n.startswith("layer4.") for n in iso["group1"] + dep["group1"]),
                "every isolated tensor of ISO and DEPTH lives in layer4 -- DEPTH IS MATCHED")
            chk(not (set(dep["group1"]) & set(CARRIERS)), "DEPTH's isolated group contains NO carrier")
        if dep2:
            chk(dep2["group1"] == list(DEPTH2_SET) and not (set(dep2["group1"]) & set(CARRIERS)),
                "DEPTH2 isolates the only two non-carrier layer4 BN SCALES, and no carrier")
        if iso and one:
            chk(set(iso["group1"]) - set(one["group1"]) == set(CARRIERS[1:]) and one["group1"] == [CARRIERS[0]],
                "ONE isolates exactly layer4.0.bn2.weight; ISO adds exactly 53 and 59")
        if GATE_FAIL:
            unresolved.append("UNRESOLVED-MANIFEST-MISMATCH")
    g1_fail = list(GATE_FAIL)

    # ---- G0: provenance ---------------------------------------------------------------
    print("\n[G0] PROVENANCE -- every run's OWN lines")
    chk(len(outs) == N_RUNS, "%d (arm, seed) runs with a .out" % N_RUNS, "%d" % len(outs))
    for arm in ARMS:
        for seed in SEEDS:
            tag = "%s-s%d" % (arm, seed)
            oo = outs.get((arm, seed))
            if not oo:
                chk(False, "%s: .out present" % tag)
                continue
            o = oo[-1]
            if len(oo) > 1:
                print("  NOTE %s: %d .out files, the latest job (%d) is audited" % (tag, len(oo), o["job"]))
            am = ARGS_RE[arm].match(o["args"][-1]) if o["args"] else None
            chk(am is not None and int(am.group(1)) == seed and int(am.group(2)) == seed,
                "%s: ARGS line is the registered line for %s" % (tag, arm))
            env = re.sub(r" PROBE_DIR=\S+", "", o["env"][-1]).replace("ENV: ", "") if o["env"] else ""
            chk(env == ENV_FIXED, "%s: ENV line (modulo PROBE_DIR) is the registered line" % tag)
            pm = PT_RE.match(o["pt"][-1]) if o["pt"] else None
            chk(pm is not None and pm.group(1) == PT_TYPE[arm], "%s: PROBE_TENSOR line present, type=%s" % (tag, PT_TYPE[arm]))
            P = probes.get((arm, seed))
            h = P["hdr"] if P else None
            chk(h is not None and h.get("PROBE_TENSOR") == 1 and h.get("every") == PROBE_EVERY
                and h.get("stepsize_type") == PT_TYPE[arm] and h.get("n_tensors") == N_TENSORS
                and tuple(h.get("param_numels", ())) == TENSOR_NUMEL and h.get("meta_alg") == "Lion",
                "%s: probe_tensor.json is the registered header" % tag)
            n = len(P["recs"]) if P else 0
            chk(P is not None and n == N_RECORDS and P["bad"] == 0
                and [r.get("step") for r in P["recs"]] == list(range(0, META_STEPS, PROBE_EVERY)),
                "%s: %d probe records at steps 0..49900" % (tag, n))
    if len(GATE_FAIL) > len(g1_fail):
        unresolved.append("UNRESOLVED-PROVENANCE")
    g0_fail = list(GATE_FAIL)

    # ---- A: decomposition on every record of every arm --------------------------------
    print("\n[A] DECOMPOSITION -- group sums of z_tensor / m_tensor vs the harness's own z / momentum; applied sign")
    n_rec = n_bad_z = n_bad_m = n_sign = n_sign_bad = n_clamped = 0
    worst_z = worst_m = 0.0
    for (arm, seed), P in sorted(probes.items()):
        groups = [[i - 1 for i in g] for g in spec_to_groups(SPEC[arm])]
        for r in P["recs"]:
            try:
                z, m = r["z_tensor"], r["m_tensor"]
                za, ma = r["z_agg"][0], r["mom_pre"][0]
                b2 = float(r.get("pt_b2", B2))
                bpre, bpost = r["beta_pre"][0], r["beta"]
            except (KeyError, IndexError, TypeError):
                continue
            if len(za) != len(groups) or len(ma) != len(groups) or len(bpre) != len(groups) or len(bpost) != len(groups):
                n_bad_z += 1
                n_rec += 1
                continue
            n_rec += 1
            gz = [sum(z[i] for i in g) for g in groups]
            gm = [sum(m[i] for i in g) for g in groups]
            sz = sum(abs(v) for v in z) + 1e-30
            sm = sum(abs(v) for v in m) + 1e-30
            dz = max(abs(p - q) for p, q in zip(gz, za)) / sz
            dm = max(abs(p - q) for p, q in zip(gm, ma)) / sm
            worst_z, worst_m = max(worst_z, dz), max(worst_m, dm)
            n_bad_z += dz > DECOMP_TOL_Z
            n_bad_m += dm > DECOMP_TOL_M
            for k in range(len(groups)):
                Lh = b2 * ma[k] + (1 - b2) * za[k]
                if not (bpost[k] > BETA_LO + 1e-9 and bpost[k] < BETA_HI - 1e-9):
                    n_clamped += 1
                    continue
                n_sign += 1
                n_sign_bad += abs(-(bpost[k] - bpre[k]) / MS - sgn(Lh)) > 1e-3
    ok_a = chk(n_rec > 0 and n_bad_z / max(n_rec, 1) <= DECOMP_FAIL_MAX,
               "group sums of z_tensor == harness z within %.0e on >= 99%% of records" % DECOMP_TOL_Z,
               "%d/%d fail, worst rel %.2e" % (n_bad_z, n_rec, worst_z))
    ok_a &= chk(n_rec > 0 and n_bad_m / max(n_rec, 1) <= DECOMP_FAIL_MAX,
                "group sums of m_tensor == harness momentum_meta within %.0e on >= 99%% of records" % DECOMP_TOL_M,
                "%d/%d fail, worst rel %.2e" % (n_bad_m, n_rec, worst_m))
    ok_a &= chk(n_sign > 0 and n_sign_bad / max(n_sign, 1) <= SIGN_FAIL_MAX,
                "-(beta - beta_pre)/ms == sign(b2*mom_pre + (1-b2)*z_agg) on >= 99%% of unclamped coordinates",
                "%d/%d fail, %d clamped" % (n_sign_bad, n_sign, n_clamped))
    if not ok_a and "UNRESOLVED-PROVENANCE" not in unresolved:
        unresolved.append("UNRESOLVED-PROVENANCE")

    # ---- R: completion, bands, divergence ------------------------------------------------
    print("\n[R] COMPLETION, ANCHOR BANDS, WITHIN-ARM DIVERGENCE")
    lev, levt = {}, {}
    for arm in ARMS:
        vs, ts = [], []
        for seed in SEEDS:
            oo = outs.get((arm, seed))
            if not oo:
                continue
            o = oo[-1]
            ok = chk(o["done"] and not o["traceback"] and o["n_ep"] >= EPOCHS and o["p5"] is not None,
                     "%s-s%d: RUN_DONE, no traceback, %d epoch lines, plateau5 %s (train %s)"
                     % (arm, seed, o["n_ep"], fmt(o["p5"], 3), fmt(o["t5"], 3)))
            if not ok:
                unresolved.append("UNRESOLVED-DIVERGED")
            if o["p5"] is not None:
                vs.append(o["p5"])
            if o["t5"] is not None:
                ts.append(o["t5"])
        if len(vs) == len(SEEDS):
            lev[arm] = statistics.mean(vs)
            levt[arm] = statistics.mean(ts) if len(ts) == len(SEEDS) else None
            rng = max(vs) - min(vs)
            print("  %-6s plateau5 %.4f (seeds %s)  range %.3f   TRAIN %s"
                  % (arm, lev[arm], ", ".join("%.3f" % v for v in vs), rng, fmt(levt[arm], 3)))
            if arm in ("k01", "ISO", "DEPTH") and rng > DEAD_BAR:
                chk(False, "%s: seed range %.3f <= DEAD_BAR %.2f" % (arm, rng, DEAD_BAR))
                unresolved.append("UNRESOLVED-DIVERGED")
            elif rng > DEAD_BAR:
                stamps.append("%s-SEEDS-DIVERGE" % arm)
        else:
            print("  %-6s INCOMPLETE (%d of %d plateau5)" % (arm, len(vs), len(SEEDS)))
    for arm, gran in (("k01", "scalar"), ("k62", "layerwise")):
        b = band(csv_path, gran)
        if b is None:
            skip("%s: corpus band unavailable" % arm)
            continue
        print("  corpus %-9s n %2d  %.4f +/- %.4f  band [%.3f, %.3f]" % (gran, b["n"], b["mean"], b["sd"], b["lo"], b["hi"]))
        if arm in lev:
            if not chk(b["lo"] <= lev[arm] <= b["hi"], "%s in-batch level %.4f inside the corpus band" % (arm, lev[arm])):
                unresolved.append("UNRESOLVED-DIVERGED")
    if own_rows(csv_path) == 0:
        stamps.append("FLOOR-NOT-INGESTED")

    # ---- B: the beta states ------------------------------------------------------------------
    print("\n[B] BETA STATES on the m=2 arms (coordinate 0 = the coarse group, 1 = the isolated group)")
    state = {}
    for arm in M2_ARMS:
        for k, lab in ((0, "coarse"), (1, "isolated")):
            occs, terms = [], []
            for seed in SEEDS:
                P = probes.get((arm, seed))
                if not P or not P["recs"]:
                    continue
                occ, term, n = occupancy(P["recs"], k)
                if occ is not None:
                    occs.append(occ)
                if term is not None:
                    terms.append(term)
                print("  %-6s s%d  beta[%d] (%-8s) floor occupancy %s  terminal %s  -> %s"
                      % (arm, seed, k, lab, fmt(occ), fmt(term, 3), free_state(occ, term)))
            if occs and terms:
                st = free_state(statistics.mean(occs), statistics.mean(terms))
                state[(arm, k)] = st
                print("  %-6s     beta[%d] pooled occupancy %.4f  mean terminal %.3f  -> %s"
                      % (arm, k, statistics.mean(occs), statistics.mean(terms), st))
    for arm, k, tag in (("ISO", 0, "ISO-59"), ("ISO", 1, "ISO-3"),
                        ("DEPTH", 0, "DEPTH-59"), ("DEPTH", 1, "DEPTH-3"),
                        ("DEPTH2", 0, "DEPTH2-60"), ("DEPTH2", 1, "DEPTH2-2"),
                        ("ONE", 0, "ONE-61")):
        stamps.append(tag + "-" + state.get((arm, k), "UNKNOWN"))

    # ---- P: the premise, replicated in batch on k01 -------------------------------------------
    print("\n[P] PREMISE -- the in-batch scalar anchor's own records (`187.6`'s flip statistic is PRIMARY)")
    have = [probes[("k01", s)]["recs"] for s in SEEDS if ("k01", s) in probes]
    P = premise_from_scalar_records(have) if have else None
    if P is None or P["n_det"] == 0:
        skip("no in-batch k01 probe records")
        stamps.append("PREMISE-NOT-REPLICATED")
    else:
        print("  determinate %d  DISAGREE %d  R_T %s  pinned %d  59-rest votes UP %s"
              % (P["n_det"], P["n_dis"], fmt(P["R_T"]), P["pin_n"], fmt(P["rest_up"])))
        for nm, v in P["shares"].items():
            print("    %-30s carrier share on DISAGREE %s" % (nm, fmt(v)))
        print("    REMOVING S FLIPS THE REMAINDER'S SIGN (`187.6`) -- determinate / pinned / dominates(pinned):")
        for lab in ("TRIO", "ONE", "DEPTH", "DEPTH2"):
            ref = CISO1_FLIP.get(lab)
            print("      %-7s %s = %s / %s / %s      (ciso1 ex ante: %.4f / %.4f)"
                  % (lab, FLIP_SETS[lab], fmt(P["flip_det"][lab]), fmt(P["flip_pin"][lab]),
                     fmt(P["dom_pin"][lab]), ref[0], ref[1]))
        if P["mean_abs"]:
            mo = sorted(P["mean_abs"].items(), key=lambda kv: -kv[1])
            rk = {i: r for r, (i, _) in enumerate(mo, 1)}
            print("    mean |L| on pinned records, the two isolated sets:")
            for i in (50, 53, 59, 47, 48, 56):
                print("      idx %2d %-30s %.4e   rank %2d" % (i, TENSOR_NAMES[i - 1], P["mean_abs"][i], rk[i]))
            print("      SUM ISO %.4e   SUM DEPTH %.4e   ratio %.1f  (the magnitude match this design CANNOT make)"
                  % (sum(P["mean_abs"][i] for i in (50, 53, 59)), sum(P["mean_abs"][i] for i in (47, 48, 56)),
                     sum(P["mean_abs"][i] for i in (50, 53, 59)) / max(sum(P["mean_abs"][i] for i in (47, 48, 56)), 1e-30)))
        stamps.append("PREMISE-REPLICATES" if premise_ok(P) else "PREMISE-NOT-REPLICATED")

    # ---- H: the content-free rival ----------------------------------------------------------
    print("\n[H] H-BROKEN -- 'any partition other than ISO floors'.  ONE is the arm that kills it.")
    d_one = lev["ONE"] - lev["k01"] if "ONE" in lev and "k01" in lev else None
    print("  D_ONE = ONE - k01 : %s   (ciso1 reference %+.4f)" % (se_str(d_one), ONE_REF - L_FLOOR))
    if d_one is not None and d_one >= SUPPORT_BAR:
        print("  ONE rescues in batch at a DIFFERENT isolated set and a DIFFERENT size [61,1]:")
        print("  H-BROKEN is DEAD in batch, so a floored DEPTH is a null about WHICH tensors, not about")
        print("  'not being ISO'.")
        stamps.append("H-BROKEN-DEAD")
    else:
        print("  ONE did NOT rescue in batch.  H-BROKEN is NOT excluded: a floored DEPTH cannot be told")
        print("  from 'any partition other than ISO floors'.  Say so; do not read DEPTH as an identity result.")
        stamps.append("H-BROKEN-ALIVE")
    stamps.append("ONE-" + level_token(d_one))

    # ---- D: the contrasts ------------------------------------------------------------------------
    print("\n[D] CONTRASTS, ALL WITHIN BATCH (SE_ARM_DIFF %.6f pp)" % SE_ARM_DIFF)
    def L(a):
        return lev.get(a)
    d_iso = (L("ISO") - L("k01")) if L("ISO") is not None and L("k01") is not None else None
    d_depth = (L("DEPTH") - L("k01")) if L("DEPTH") is not None and L("k01") is not None else None
    d_depth2 = (L("DEPTH2") - L("k01")) if L("DEPTH2") is not None and L("k01") is not None else None
    d_ceil = (L("k62") - L("k01")) if L("k62") is not None and L("k01") is not None else None
    delta_id = (L("ISO") - L("DEPTH")) if L("ISO") is not None and L("DEPTH") is not None else None
    delta_id2 = (L("ISO") - L("DEPTH2")) if L("ISO") is not None and L("DEPTH2") is not None else None
    delta_bias = (L("DEPTH") - L("DEPTH2")) if L("DEPTH") is not None and L("DEPTH2") is not None else None
    print("  *** PRIMARY ***  DELTA_ID  = ISO - DEPTH  : %s" % se_str(delta_id))
    for lab, d in (("D_ISO     = ISO    - k01", d_iso), ("D_DEPTH   = DEPTH  - k01", d_depth),
                   ("D_DEPTH2  = DEPTH2 - k01", d_depth2), ("D_ONE     = ONE    - k01", d_one),
                   ("D_CEIL    = k62    - k01", d_ceil), ("DELTA_ID2 = ISO - DEPTH2 ", delta_id2),
                   ("DELTA_BIAS= DEPTH - DEPTH2", delta_bias)):
        print("  %s : %s" % (lab, se_str(d)))
    if all(a in levt and levt[a] is not None for a in ARMS):
        print("  TRAIN (plateau5-window mean train accuracy), all vs k01:")
        print("    " + "  ".join("%s %+.3f" % (a, levt[a] - levt["k01"]) for a in ARMS if a != "k01"))
    print("  bars: SUPPORT %.1f pp (%.2f SE)  REFUTE %.1f pp (%.2f SE)  READ %.6f pp (2 SE)  IDENTITY %.1f pp"
          % (SUPPORT_BAR, SUPPORT_BAR / SE_ARM_DIFF, REFUTE_BAR, REFUTE_BAR / SE_ARM_DIFF, READ_BAR, IDENTITY_BAR))
    if d_iso and d_depth is not None and d_iso != 0:
        print("  FRACTION OF THE IN-BATCH SCALAR->ISO GAP RECOVERED BY DEPTH: D_DEPTH / D_ISO = %.4f" % (d_depth / d_iso))
    if d_ceil and d_iso is not None and d_ceil != 0:
        print("  D_ISO / D_CEIL = %.4f (ISO vs the in-batch layerwise ceiling)" % (d_iso / d_ceil))

    # the BOUND discipline, applied automatically
    sb = saturation_band(csv_path)
    if sb:
        print("  registered saturation band re-derived: [%.4f, %.4f] (%d scalar + %d m=2 rows over %d cells, cap %.4f)"
              % (sb["lo"], sb["hi"], sb["n_scalar"], sb["n_extra"], sb["n_cells"], sb["cap"]))
    for arm, d in (("DEPTH", d_depth), ("DEPTH2", d_depth2)):
        if d is None or L(arm) is None:
            continue
        if sb and sb["lo"] <= L(arm) <= sb["hi"]:
            stamps.append("%s-FLOOR-SATURATED" % arm)
            print("  %s level %.4f is INSIDE the registered saturation band: its level is a BOUND, NOT a"
                  % (arm, L(arm)))
            print("     magnitude.  Entitled: 'D_%s <= %.1f pp'.  NOT entitled: the point value as an effect"
                  % (arm, REFUTE_BAR))
            print("     size, or any residual against the floor read as agreement (`164.6`).")
        else:
            stamps.append("%s-ABOVE-BAND" % arm)
    if "ISO" in lev:
        shift = abs(lev["ISO"] - ISO_REF)
        stamps.append("ISO-REPLICATES" if shift <= BAND_K * SIGMA_W else "ISO-SHIFTED")
        print("  ISO in batch %.4f vs ciso1 %.4f: |shift| %.4f against 4*SIGMA_W = %.4f -> %s"
              % (lev["ISO"], ISO_REF, shift, BAND_K * SIGMA_W, stamps[-1]))
    if delta_bias is not None:
        stamps.append("BIAS-NULL" if abs(delta_bias) <= READ_BAR else "BIAS-MATTERS")
    stamps.append("DEPTH2-" + level_token(d_depth2))

    # ---- FINAL ------------------------------------------------------------------------------------
    for u in ("UNRESOLVED-MANIFEST-MISMATCH", "UNRESOLVED-PROVENANCE", "UNRESOLVED-DIVERGED"):
        if u in unresolved:
            return finish(u, stamps)
    return finish(decide(d_iso, d_depth, delta_id, stamps), stamps)


def finish(branch, stamps):
    print("\n" + "=" * 78)
    print("FINAL: %s | %s" % (branch, " | ".join(stamps) if stamps else "no stamps"))
    print("=" * 78)
    print("PRIMARY: DELTA_ID = plateau5(ISO) - plateau5(DEPTH) @ 100 epochs, from the runs' own .out files;")
    print("every contrast WITHIN batch; bars frozen in this file (SE_ARM_DIFF %.6f from a corpus reader" % SE_ARM_DIFF)
    print("that excludes cdep1-* rows).  SCOPE, BINDING ON EVERY SENTENCE: this batch matches DEPTH to ISO on")
    print("group sizes, isolated numel, width, depth and normalisation-layer type.  It does NOT and CANNOT")
    print("match them on DYNAMICAL MAGNITUDE -- on the scalar trajectory the carriers' mean |L| exceeds every")
    print("512-parameter layer-4 non-carrier's by two to five orders of magnitude, and no set of members")
    print("closes that gap.  An IDENTITY-OPERATIVE reading therefore means 'not any three matched layer-4 BN")
    print("parameters', and does NOT separate tensor identity from term magnitude.")
    return branch

def selftest(csv_path, runsdir):
    print("=" * 78)
    print("cDP1 --selftest: the grammar, the partitions, the impossibility proof, the noise floor,")
    print("the anchors, the saturation band, the branch map, and the ex-ante premise on ciso1")
    print("=" * 78)
    names = TENSOR_NAMES

    print("\n[A] THE MODEL AND THE IMPOSSIBILITY PROOF")
    chk(len(names) == N_TENSORS and len(TENSOR_NUMEL) == N_TENSORS and sum(TENSOR_NUMEL) == N_PARAMS,
        "62 names, 62 numels, 11,220,132 parameters")
    chk(all(n in names for n in CARRIERS + DEPTH_SET), "carriers and the DEPTH set are live parameter names")
    chk(tuple(names.index(n) + 1 for n in CARRIERS) == (50, 53, 59), "carrier 1-based indices are (50, 53, 59)")
    chk(tuple(names.index(n) + 1 for n in DEPTH_SET) == (47, 48, 56), "DEPTH 1-based indices are (47, 48, 56)")
    chk(tuple(names.index(n) + 1 for n in DEPTH2_SET) == (47, 56), "DEPTH2 1-based indices are (47, 56)")
    chk(all(TENSOR_NUMEL[names.index(n)] == 512 for n in CARRIERS + DEPTH_SET),
        "every carrier AND every DEPTH member has exactly 512 parameters -- the numel match is EXACT")
    chk(all(n.startswith("layer4.") for n in CARRIERS + DEPTH_SET),
        "every carrier AND every DEPTH member lives in layer4 -- the depth match is EXACT")
    chk(not (set(DEPTH_SET) & set(CARRIERS)) and not (set(DEPTH2_SET) & set(CARRIERS)),
        "neither DEPTH nor DEPTH2 contains a carrier")
    bn_scale = [n for n in names if n.endswith(".weight") and (".bn" in n or "shortcut.1" in n)]
    l4_scale = [n for n in bn_scale if n.startswith("layer4.")]
    wide512_scale = [n for n in bn_scale if TENSOR_NUMEL[names.index(n)] == 512]
    chk(len(l4_scale) == 5 and set(wide512_scale) == set(l4_scale),
        "ResNet-18 has exactly FIVE BN scales in layer4 and every 512-wide BN scale in the model is one of them",
        str([names.index(n) + 1 for n in l4_scale]))
    non_car = [n for n in l4_scale if n not in CARRIERS]
    chk(len(non_car) == 2 and set(non_car) == set(DEPTH2_SET),
        "exactly TWO non-carrier layer4 BN scales exist -> a class+depth+numel-matched carrier-free TRIPLE is"
        " IMPOSSIBLE, and DEPTH2 is the maximal class-pure control", str(non_car))
    chk(names[47] == "layer4.0.bn1.bias" and TENSOR_NUMEL[47] == 512,
        "index 48 (1-based) is layer4.0.bn1.bias -- a BatchNorm BIAS, 512 params, NOT a scale; this is the"
        " ONE dimension on which DEPTH is unmatched, and DEPTH2 is why it is testable")
    for i in (51, 54, 60):
        chk(names[i - 1].endswith(".bias") and names[i - 1].startswith("layer4."),
            "index %d is the affine bias partner of a carrier (%s) -- excluded from DEPTH by the registered rule"

        "the PRIMARY statistic DELTA_ID is >= 30 SE from zero under IDENTITY-OPERATIVE and 0 under"
        " DEPTH-OPERATIVE -- it is bounded in NEITHER direction, so the primary read can disagree with both")

    print("       DEPTH and DEPTH2 do NOT clear the floor under IDENTITY-OPERATIVE.  That is unavoidable for a")

    chk(ARGS_RE["DEPTH"].match(a % (SPEC["ISO"], "DEPTH")) is None, "ARGS regex rejects a swapped spec")
    chk(ARGS_RE["DEPTH"].match(a % (SPEC["DEPTH2"], "DEPTH")) is None, "ARGS regex rejects DEPTH2's spec on DEPTH")
    chk(ARGS_RE["k01"].match((a % ("scalar", "k01")).replace("1e-3", "1e-4")) is None, "ARGS regex rejects a wrong ms")

    e = ("ENV: AUGMENT=1 BETA_CLIP=-15:-2.3026 HIER=none LAM=na ETA_RATIO=na COS_TOTAL=default COS_WARMUP=default "
         "SCHED=none SCHED_TOTAL=none SCHED_WARMUP=none SCHED_MIN=none PROBE=100 "
         "PROBE_DIR=/cluster/account_b/metaopt/runs/cdep1/probe_cdep1-DEPTH-s24 EB_RHO=na EB_LOG=0")
    chk(re.sub(r" PROBE_DIR=\S+", "", e).replace("ENV: ", "") == ENV_FIXED, "ENV line modulo PROBE_DIR is the registered one")
    chk(re.sub(r" PROBE_DIR=\S+", "", e.replace("BETA_CLIP=-15:-2.3026", "BETA_CLIP=-20:-2.3026")).replace("ENV: ", "")
        != ENV_FIXED, "ENV audit catches a wrong BETA_CLIP (it cannot ride the ARGS line)")
    chk(PT_RE.match("PROBE_TENSOR: on every=100 type=blockwise tensors=62 meta_alg=Lion momentum_param=0.99 "
                    "Lion_beta2=0.9 dir=/x/probe_cdep1-DEPTH-s24") is not None, "PROBE_TENSOR regex matches a blockwise ON line")
    chk(EPTR_RE.search("Epoch 99, Train Accuracy: 23.65 %, Test Accuracy: 23.68 %") is not None, "epoch-line regex")
    for f, want in (("cdep1-DEPTH-s24-4930000.out", "DEPTH"), ("cdep1-DEPTH2-s25-4930001.out", "DEPTH2"),
                    ("cdep1-ISO-s26-4930002.out", "ISO")):
        m = OUT_RE.match(f)
        chk(m is not None and m.group(1) == want, "out-file regex resolves %s to arm %s" % (f, want))
    chk(OUT_RE.match("ciso1-ISO-s21-4925520.out") is None, "out-file regex rejects another batch's file")
    chk(PROBE_DIR_RE.match("probe_cdep1-DEPTH2-s24").group(1) == "DEPTH2", "probe-dir regex resolves DEPTH2 before DEPTH")

    print("\n[F] THE BRANCH MAP")
    st = []
    chk(decide(47.0, 0.3, 46.7, st) == "IDENTITY-OPERATIVE", "decide: ISO rescues, DEPTH floors, big gap -> IDENTITY")
    st = []
    chk(decide(47.0, 47.0, 0.0, st) == "DEPTH-OPERATIVE", "decide: both rescue, indistinguishable -> DEPTH-OPERATIVE")
    st = []
    chk(decide(47.0, 30.0, 17.0, st) == "BOTH-OPERATIVE", "decide: DEPTH rescues but by less -> BOTH-OPERATIVE")
    st = []
    chk(decide(47.0, 15.0, 32.0, st) == "UNRESOLVED-PARTIAL-RESCUE", "decide: DEPTH in the middle -> PARTIAL-RESCUE (both matter)")
    st = []

    st = []

    chk(free_state(0.0, -8.0) == "FREE" and free_state(0.63, -15.0) == "PINS" and free_state(0.3, -15.0) == "NEITHER"

    recs = [{"beta": [-15.0, -15.0]}] * 6 + [{"beta": [-10.0, -15.0]}] * 4

    chk(carrying_set([5.0, -1.0, -1.0, -1.0, 0.5], 1) == [0], "carrying set: one dominant term")
    P0 = {"pin_n": 100, "flip_pin": {"TRIO": 1.0, "DEPTH": 0.0, "ONE": 1.0, "DEPTH2": 0.0}}

    import tempfile

        for lab, grans, want, wn in (("floor (scalar)", ("scalar",), L_FLOOR, L_FLOOR_N),

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default=None)
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default=None)
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default=None, help="selftest: where ciso1/ lives")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if a.runsdir is None:
        ap.error("runsdir is required (or --selftest)")
    score(a.runsdir, a.csv, a.manifest)


if __name__ == "__main__":
    main()
