def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

# =============================================================================
# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.
# =============================================================================
def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


# =============================================================================
# the frozen manifest, composed from ResNet-18's layout (no torch).  The
# launcher's guard 4d proves the LIVE manifest BYTE-IDENTICAL to this text.

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists (each sorted), re-implemented
    HERE so the registered index sets are checked without importing HF.py."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def synthetic_manifest_text():
    t = r18_tensors(True)
    names = [n for n, _q, _o in t]
    numel = [q for _n, q, _o in t]
    lines = ["NETWORK %s" % NET, "NUM_PARAM_TENSORS %d" % len(t), "TOTAL_PARAMS %d" % sum(numel),
             "NORM BatchNorm2d %d GroupNorm 0" % N_BATCHNORM, "PARENT %s %d" % (PARENT, PARENT_NTENS),
             "PARENT_REMOVED %s" % " ".join(SHORTCUT_9),
             "CLIP_C %s" % CLIP, "EPOCHS %d" % EPOCHS, "META_STEPS %d" % (EPOCHS * STEPS_PER_EPOCH)]
    for i, (n, q, o) in enumerate(t, 1):
        lines.append("TENSOR %d %s %d %s" % (i, n, q, o))
    lines.append("W512 %s" % ",".join("%d:%s" % x for x in zip(BN512_IDX, BN512_NAMES)))
    lines.append("ARMSPEC k01 SPEC scalar TYPE scalar M 1")
    lines.append("ARMSPEC kL SPEC layerwise TYPE layerwise M %d" % len(t))
    for a in BLOCKWISE_ARMS:
        g = parse_sets(SPEC[a], names)
        lines.append("ARMSPEC %s SPEC %s TYPE blockwise M %d SIZES %s PARAMS %s GROUP1 %s"
                     % (a, SPEC[a], len(g), ",".join(str(len(x)) for x in g),
                        ",".join(str(sum(numel[i - 1] for i in x)) for x in g),
                        " ".join(names[i - 1] for i in g[1])))
        lines.append("ISOLATED %s %s" % (a, ",".join(str(i) for i in g[1])))
    return "\n".join(lines) + "\n"

        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  Among several files for one (arm, seed) the highest
    INTEGER job id among COMPLETE runs wins; a truncated resubmission cannot shadow
    a complete run.  -> (runs, notes)"""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, any_gran=False):
    return ((any_gran or r.get("granularity") in ("scalar", "layerwise"))
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra, any_gran=False):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, any_gran) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm


def net_is(net):
    return lambda r: r.get("network") == net and r.get("dataset") == DSET


def prefix_arm_mean(rows, prefix, gran, net):
    v = [float(r["plateau5"]) for r in rows if r.get("run", "").startswith(prefix + "-")
         and usable(r) and std_cell(r, True) and r.get("network") == net and r.get("granularity") == gran]
    return (mean(v), len(v)) if v else (None, 0)

def load_probe(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    continue
                if isinstance(r.get("beta"), list) and isinstance(r.get("step"), int):
                    recs.append(r)
            return recs
    return None


def beta_summary(recs, coord):
    vals = [(r["step"], r["beta"][coord]) for r in recs if len(r["beta"]) > coord]
    if not vals:
        return None
    at = [v <= CLAMP_LO + 1e-4 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    return {"last": vals[-1][1], "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH),
            "frac_clamp": sum(at) / float(len(at)), "pin_epoch": pin, "n": len(vals)}


NOM_SETS = {"50": [50], "44": [44], "47": [47], "44_50": [44, 50]}


def pinned_nomination(recs):
    """the 184/194.5 statistic on k01's PINNED records (1-based sets); descriptive"""
    ok = [r for r in recs if isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
          and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS and len(r["beta"]) == 1
          and r["beta"][0] <= CLAMP_LO + 1e-4]
    if not ok:
        return None
    mass = [0.0] * NTENS
    flips = dict((k, 0) for k in NOM_SETS)
    for r in ok:
        b2 = r.get("pt_b2", 0.9)
        L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
        S = sum(L)
        s0 = (S > 0) - (S < 0)
        for i, v in enumerate(L):
            mass[i] += abs(v)
        for k, idx in NOM_SETS.items():
            Sr = S - sum(L[i - 1] for i in idx)
            if ((Sr > 0) - (Sr < 0)) != s0:
                flips[k] += 1
    tm = sum(mass) or 1.0
    return {"n": len(ok), "share": [x / tm for x in mass],
            "flip": dict((k, v / float(len(ok))) for k, v in flips.items())}


def descriptive_probe_readout(runsdir):
    print("\nDESCRIPTIVE, NON-GATING -- beta trajectories from the runs' own probe.jsonl")
    print("  (cpl1: k01 pinned at epoch ~36; ISO's complement at ~60-62.  Nothing below gates anything.)")
    any_rec = False
    for a in ARMS:
        for s in SEEDS:
            recs = load_probe(runsdir, a, s)
            if not recs:
                continue
            any_rec = True
            if a == "kL":
                last = recs[-1]["beta"]
                print("  %-4s s%d  %d records; groups at the clamp at the last record: %d of %d; max beta %.3f"
                      % (a, s, len(recs), sum(v <= CLAMP_LO + 1e-4 for v in last), len(last), max(last)))
                continue
            for coord, lab in ((0, "beta[0]" if a == "k01" else "complement beta[0]"),
                               (1, "isolated beta[1]")):
                if a == "k01" and coord == 1:
                    continue
                bs = beta_summary(recs, coord)
                if bs:
                    print("  %-4s s%d  %-19s last %8.3f  peak %8.3f @ep %5.1f  at-clamp frac %.3f  "
                          "pinned-from-epoch %s" % (a, s, lab, bs["last"], bs["peak"], bs["peak_epoch"],
                                                    bs["frac_clamp"],
                                                    "never" if bs["pin_epoch"] is None else "%.1f" % bs["pin_epoch"]))
    if not any_rec:
        print("  (no probe records reachable under %s)" % runsdir)
    print("\nDESCRIPTIVE, NON-GATING -- the 184/194.5 statistic on THIS batch's k01 PINNED records")
    got = False
    for s in SEEDS:
        recs = load_probe(runsdir, "k01", s)
        nm = pinned_nomination(recs) if recs else None
        if nm is None:
            continue
        got = True
        order = sorted(range(NTENS), key=lambda i: -nm["share"][i])[:4]
        print("  k01 s%d  %d pinned records; top share %s"
              % (s, nm["n"], ", ".join("idx %d %.4f" % (i + 1, nm["share"][i]) for i in order)))
        print("          removing {50} flips %.4f | {44} %.4f | {47} %.4f | {44,50} %.4f"
              % (nm["flip"]["50"], nm["flip"]["44"], nm["flip"]["47"], nm["flip"]["44_50"]))
    if not got:
        print("  (no pinned k01 records with per-tensor terms reachable)")

# THE BRANCH MAP -- a pure function of the in-batch means and two flags
# =============================================================================
def decide(M, trains_ok=True, diverged=False):
    D_GAP = M["kL"] - M["k01"]
    D_ISO = M["ISO"] - M["k01"]
    D_HEAD = M["HEAD"] - M["k01"]
    D_TWIN = M["TWIN"] - M["k01"]
    DELTA_HEAD = M["HEAD"] - M["TWIN"]
    D_PAIR = M["ISO"] - M["HEAD"]
    if not trains_ok:
        return "PLAIN-UNTRAINABLE"
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if D_GAP < GAP_MIN:
        return "GAP-NOT-REPRODUCED-PLAIN"
    if D_ISO < RESCUE_BAR:
        return "POSITIVE-CONTROL-FAILED-PLAIN"
    rH, rT = D_HEAD >= RESCUE_BAR, D_TWIN >= RESCUE_BAR
    if not rH and not rT:
        return "PAIR-REQUIRED-PLAIN" if D_HEAD <= NULL_BAR else "HEAD-PARTIAL-PLAIN"
    if not rH:
        return "UNEXPECTED-PATTERN"
    if DELTA_HEAD <= -IDENTITY_BAR:
        return "CONTROL-DOMINATES-PLAIN"
    if DELTA_HEAD >= IDENTITY_BAR:
        return "HEAD-CARRIES-ALONE-PLAIN" if D_PAIR <= PAIR_BAR else "HEAD-CARRIES-MOST-PLAIN"
    if rT and abs(DELTA_HEAD) <= NULL_BAR:
        return "CLASS-OPERATIVE-PLAIN"
    return "IDENTITY-ATTENUATED-PLAIN"

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound, 2 incomplete."""
    print("=" * 78)
    print(" cpl2 -- TURN cpl1's SUBTRACTION INTO A MEASUREMENT  (PlainNet18_c100)")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-4s %s" % (a, SPEC[a]))
    print(" PRIMARY: DELTA_HEAD = plateau5(HEAD) - plateau5(TWIN); KEY SECONDARY D_HEAD = HEAD - k01.")
    print("          plateau5 = mean TEST over epochs %d..%d of each run's own .out." % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2); the corpus cannot move them.")
    print("=" * 78)
    rows = read_corpus(csvpath)
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)"
          % ("not readable" if rows is None else "%d rows, %s- excluded" % (len(rows), PREFIX)))

    runs, rnotes = read_runs(runsdir)

    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing),
                                                            len(ARMS) * len(SEEDS), ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    every run's OWN ARGS line says what its file name registers")
    for s in SEEDS:

                continue
            f = parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s)}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad])
                 if (dup or bad) else "")

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),

             if len(r["pt"]) != 1 or not PT_RE.match(r["pt"][0])

    hchk(not badpt, "G-ENV every run printed ONE `PROBE_TENSOR: on every=%d type=<its arm's> tensors=%d`"

             "G-STRUCT byte-identical to synthetic_manifest_text() (53 tensors, no shortcut, 3 m=2 partitions)",

             str(prov.get("SCORER_SHA256"))[:16])

    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH; tail slope = TEST OLS over epochs 80-99 (descriptive)")
    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)

        v, t = arm_v[a], arm_t[a]

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def _spec_from_scorer(fname, key):
    p = os.path.join(os.path.dirname(os.path.abspath(__file__)), fname)
    if not os.path.exists(p):
        return None
    import ast
    m = re.search(r"^SPEC = (\{.*?^\})", open(p).read(), re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)

        cpl1_ctl = _spec_from_scorer("cPL1_plainnet_residual_score.py", "CTL") or ""
        cpl1_one = _spec_from_scorer("cPL1_plainnet_residual_score.py", "ONE") or ""
        for lab, gran, frozen in (("k01", "scalar", R_CPL1["k01"]), ("kL", "layerwise", R_CPL1["kL"]),
                                  ("ISO", SPEC["ISO"], R_CPL1["ISO"]), ("CTL", cpl1_ctl, R_CPL1["CTL"]),
                                  ("ONE", cpl1_one, R_CPL1["ONE"])):
            v, n = prefix_arm_mean(rows, "cpl1", gran, NET)
            if v is not None and abs(v - frozen) < 5e-5:
                chk(n == 3, "frozen reference cpl1 %-4s re-derives == %.4f (n=3)" % (lab, frozen))
            else:
                note("frozen reference cpl1 %s: %.4f, live %s (non-gating)" % (lab, frozen, fmt(v)))
        n_pl = [r["run"] for r in rows if r.get("network") == NET]
        chk(all(x.startswith("cpl1-") for x in n_pl), "%s appears ONLY in cpl1's rows (%d)" % (NET, len(n_pl)))
        chk(not any(r.get("granularity") in (SPEC["HEAD"], SPEC["TWIN"]) for r in rows),
            "neither singleton spec (HEAD, TWIN) has EVER run: 50 was never isolated alone")
        chk(not any(r.get("seed") in ("75", "76", "77") for r in rows), "ZERO corpus rows carry seed 75, 76 or 77")

    print("\nD. THE FLOOR TABLE (164.6).  Chance %.2f pp, ceiling 100.  Levels UNSURE." % CHANCE)
    print("   'at-k01' = predicted at the IN-BATCH k01 level (on PlainNet the scalar beta pins at epoch ~36, cpl1)")
    lo_all = 100.0
    for acc, pred in ACCOUNTS.items():
        print("  %-11s" % acc)
        for a in ARMS:
            lo, hi = pred[a]
            role = ("the in-batch floor ANCHOR" if a == "k01" else
                    "at-k01" if pred[a] == pred["k01"] else
                    "OFF k01 by >= %.0f pp" % (lo - pred["k01"][1]))
            print("      %-4s %4.0f-%-4.0f  above chance %+5.1f..%+5.1f  below ceiling %5.1f..%5.1f  %s"
                  % (a, lo, hi, lo - CHANCE, hi - CHANCE, 100 - hi, 100 - lo, role))
            lo_all = min(lo_all, lo)
        dlo = pred["HEAD"][0] - pred["TWIN"][1]
        dhi = pred["HEAD"][1] - pred["TWIN"][0]
        print("      PRIMARY DELTA_HEAD predicted %+.0f..%+.0f  (hard range +-%.0f)" % (dlo, dhi, 100 - CHANCE))
        chk(dhi <= (100 - CHANCE) - 25 and dlo >= -(100 - CHANCE) + 25,
            "%-11s DELTA_HEAD's predicted interval is >= 25 pp inside BOTH hard bounds" % acc)
        chk(pred["ISO"][0] - pred["k01"][1] >= 4 * RESCUE_BAR,
            "%-11s the positive control ISO is predicted >= %.0f pp off k01 (kills pair-level H-BROKEN)"
            % (acc, 4 * RESCUE_BAR))
        off = [a for a in ("HEAD", "TWIN") if pred[a] != pred["k01"]]
        if acc in ("PAIR", "H-SINGLETON"):
            chk(not off, "%-11s puts BOTH singletons at k01 -- declared: PAIR-REQUIRED cannot exclude H-SINGLETON" % acc)
        else:
            chk(bool(off) and all(pred[a][0] - pred["k01"][1] >= RESCUE_BAR for a in off),
                "%-11s has an off-k01 singleton >= %.0f pp clear of k01 (kills H-SINGLETON in batch)"
                % (acc, RESCUE_BAR), ",".join(off))
        mid = dict((a, (pred[a][0] + pred[a][1]) / 2.0) for a in ARMS)
        chk(decide(mid) == EXPECTED_BRANCH[acc], "%-11s midpoint prediction lands in %s" % (acc, EXPECTED_BRANCH[acc]),
            decide(mid))
    chk(lo_all - CHANCE >= 7.0, "the LOWEST level ANY account predicts for ANY arm is >= 7 pp above chance",
        "%.1f pp (cpl1's k01 was 11.79)" % (lo_all - CHANCE))
    chk(ACCOUNTS["PAIR"] == ACCOUNTS["H-SINGLETON"], "PAIR and H-SINGLETON predict IDENTICALLY on every arm (declared)")
    chk(all(pred["kL"][0] - pred["k01"][1] >= GAP_MIN + 20 for pred in ACCOUNTS.values()),
        "every account predicts D_GAP >= GAP_MIN + 20")

    print("\nE. THE REAL score() ON SYNTHETIC RUNS (invented numbers; evidence of nothing)")
    tmp = tempfile.mkdtemp(prefix="cpl2_selftest_")
    gnman = synthetic_manifest_text().replace("PlainNet18_c100", "ResNet18_gn_c100")
    cases = [
        ("SUBTRACT", {}, {}, "HEAD-CARRIES-ALONE-PLAIN"),
        ("JOINT", {"HEAD": 41.0}, {}, "HEAD-CARRIES-MOST-PLAIN"),
        ("PAIR/H-SINGLETON", {"HEAD": 12.2}, {}, "PAIR-REQUIRED-PLAIN"),
        ("head partial", {"HEAD": 17.0}, {}, "HEAD-PARTIAL-PLAIN"),
        ("CLASS1", {"TWIN": 65.5}, {}, "CLASS-OPERATIVE-PLAIN"),
        ("positive control fails", {"ISO": 13.0, "HEAD": 12.1}, {}, "POSITIVE-CONTROL-FAILED-PLAIN"),
        ("gap not reproduced", {"kL": 31.0, "ISO": 30.0, "HEAD": 29.0}, {}, "GAP-NOT-REPRODUCED-PLAIN"),
        ("untrainable", {"kL": 22.0, "ISO": 21.0, "HEAD": 20.0}, {}, "PLAIN-UNTRAINABLE"),
        ("unexpected", {"HEAD": 12.5, "TWIN": 50.0}, {}, "UNEXPECTED-PATTERN"),
        ("control dominates", {"HEAD": 30.0, "TWIN": 60.0}, {}, "CONTROL-DOMINATES-PLAIN"),
        ("identity attenuated", {"HEAD": 60.0, "TWIN": 50.0}, {}, "IDENTITY-ATTENUATED-PLAIN"),
        ("near-bar PAIR just inside", {"HEAD": 67.5 - PAIR_BAR + 0.05}, {}, "HEAD-CARRIES-ALONE-PLAIN"),
        ("near-bar PAIR just outside", {"HEAD": 67.5 - PAIR_BAR - 0.05}, {}, "HEAD-CARRIES-MOST-PLAIN"),
        ("near-bar HEAD null edge", {"HEAD": 12.0 + NULL_BAR + 0.05}, {}, "HEAD-PARTIAL-PLAIN"),
        ("near-bar HEAD rescue edge", {"HEAD": 12.0 + RESCUE_BAR - 0.05}, {}, "HEAD-PARTIAL-PLAIN"),
        ("diverged", {}, {"over": {("HEAD", 77): 55.0}}, "UNRESOLVED-DIVERGED"),
        ("harness: every arm tiny", dict((a, 3.0) for a in ARMS), {}, "HARNESS-UNSOUND"),
        ("harness: AUGMENT=0 on one run", {}, {"env_break": ("TWIN", 77)}, "HARNESS-UNSOUND"),
        ("harness: ARGS spec != arm", {}, {"spec_break": ("HEAD", 76)}, "HARNESS-UNSOUND"),
        ("harness: ARGS name the ResNet parent", {}, {"net_break": ("ISO", 75)}, "HARNESS-UNSOUND"),
        ("harness: manifest names another net", {}, {"manifest": gnman}, "HARNESS-UNSOUND"),
        ("harness: provenance is the live tree", {}, {"prov": {"bn": PRE_SHA}}, "HARNESS-UNSOUND"),
        ("harness: provenance dry-run", {}, {"prov": {"mode": "dry-run"}}, "HARNESS-UNSOUND"),
        ("harness: provenance names another scorer", {}, {"prov": {"scorer": "0" * 64}}, "HARNESS-UNSOUND"),
        ("a DROPPED run", {}, {"drop": ("HEAD", 76)}, "INCOMPLETE"),
        ("a TRUNCATED higher-jid resubmission", {}, {"shadow": ("TWIN", 75)}, "HEAD-CARRIES-ALONE-PLAIN"),
    ]
    results = {}
    for lab, lv, kw, want in cases:
        d = os.path.join(tmp, re.sub(r"\W+", "_", lab))
        os.makedirs(d)
        _e_build(d, levels=lv, **kw)
        rc, fin, out = _e_score(d, csvpath)
        results[lab] = (rc, fin, out, d)
        chk(fin.startswith("FINAL: %s |" % want), "E %-40s -> %s" % (lab, want), fin[:80])
    rc, fin, out, dpath = results["SUBTRACT"]
    chk(rc == 0 and "HARNESS-CLEAN" in fin and "SINGLETON-BROKEN-EXCLUDED" in fin
        and "POSITIVE-CONTROL-REPRODUCES" in fin and "HORIZON-100-ONLY" in fin
        and "SINGLE-TENSOR-DECOMPOSITION-OF-cpl1" in fin,
        "E SUBTRACT exits 0 with HARNESS-CLEAN, SINGLETON-BROKEN-EXCLUDED and the unconditional stamps")
    chk(results["a DROPPED run"][0] == 2, "E a DROPPED run exits 2 (INCOMPLETE)")
    chk(results["harness: AUGMENT=0 on one run"][0] == 1, "E a harness failure exits 1")
    chk(results["a TRUNCATED higher-jid resubmission"][1] == results["SUBTRACT"][1],
        "E a truncated higher-jid resubmission does not shadow a complete run")
    chk("SINGLETON-BROKEN-NOT-EXCLUDED" in results["PAIR/H-SINGLETON"][1]
        and "FLOOR-READINGS-ARE-BOUNDS" in results["PAIR/H-SINGLETON"][1]
        and "POSITIVE-CONTROL-REPRODUCES" in results["PAIR/H-SINGLETON"][1],
        "E PAIR-REQUIRED stamps SINGLETON-BROKEN-NOT-EXCLUDED, FLOOR-READINGS-ARE-BOUNDS, POSITIVE-CONTROL-REPRODUCES")
    d = os.path.join(tmp, "probe")
    os.makedirs(d)
    _e_build(d)
    _e_probe(d)
    rc, fin, out = _e_score(d, csvpath)
    chk(fin == results["SUBTRACT"][1] and "removing {50} flips 1.0000" in out and "{47} 0.0000" in out
        and "pinned-from-epoch 6.0" in out,
        "E synthetic pinned records: the readout runs (pin epoch, flips) and FINAL does not move")
    d2 = os.path.join(tmp, "probe_garbled")
    os.makedirs(d2)
    _e_build(d2)
    _e_probe(d2)
    open(os.path.join(d2, PREFIX, "probe_cpl2-k01-s75", "probe.jsonl"), "w").write("{not json\n\x00\x01\n")
    rc, fin, out = _e_score(d2, csvpath)
    chk(rc == 0 and fin == results["SUBTRACT"][1], "E a GARBLED probe file changes nothing in FINAL or rc")

    print("\n   O2 INVARIANCE: the SAME synthetic runs against four corpora")
    variants = _e_csv_variants(csvpath if csvpath and os.path.exists(csvpath) else None, tmp)
    for case in ("SUBTRACT", "PAIR/H-SINGLETON"):
        ref_fin, ref_bars = None, None
        dd = results[case][3]
        for lab in ("real", "real+invented", "minimal-invented", "no-corpus"):
            if lab not in variants:
                skip("O2 %s corpus not available on this host" % lab)
                continue
            rc, fin, out = _e_score(dd, variants[lab])
            bars = _bar_lines(out)
            if ref_fin is None:
                ref_fin, ref_bars = fin, bars
            chk(fin == ref_fin and bars == ref_bars and len(bars) == 4,
                "O2 %-16s %-17s FINAL and SIGMA/SE/bars/PRIMARY lines are IDENTICAL" % (case, lab), fin[:44])
    if "real+invented" in variants:
        a_ = pooled_sigma(read_corpus(variants["real"]), net_is(NET), True)[0]

            "O2 non-vacuity: the invented rows MOVE the live PlainNet corpus sigma (%.4f -> %.4f) and FINAL did not move"

    print("\n   THE DOCUMENTED ONE-ARGUMENT INVOCATION, as a subprocess")
    me = os.path.abspath(__file__)
    try:
        p = subprocess.run([sys.executable, me, dpath], stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                           universal_newlines=True, timeout=300)
        fl = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 0 and fl and fl[-1] == results["SUBTRACT"][1],
            "`python3 %s <runsdir>` exits 0 with the in-process FINAL line" % os.path.basename(me),
            "rc %d" % p.returncode)

                           stderr=subprocess.STDOUT, universal_newlines=True, timeout=300)

            "... and on a dropped run exits 2 with FINAL: INCOMPLETE", "rc %d" % p.returncode)
    except Exception as e:
        chk(False, "subprocess invocation", "%s: %s" % (type(e).__name__, e))

    print("\nF. THE BRANCH MAP IS TOTAL -- 4,000 random in-batch means, every branch registered")
    import random
    rnd = random.Random(20260916)
    seen = set()
    for _ in range(4000):
        m = dict((a, rnd.uniform(1.0, 80.0)) for a in ARMS)
        seen.add(decide(m, rnd.random() > 0.05, rnd.random() < 0.05))
    chk(seen <= set(LICENSE), "every branch decide() returns has a registered licence text", "%d distinct" % len(seen))
    reached = set(r[1].split(" |")[0][len("FINAL: "):] for r in results.values())

    if csvpath and os.path.exists(csvpath):
        chk(not any(r.get("run", "").startswith(PREFIX + "-") for r in csv.DictReader(open(csvpath))),

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))

def main():
    ap = argparse.ArgumentParser(description="cpl2 registered scorer.  Documented invocation: "
                                             "python3 analysis/cPL2_plainnet_head_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cpl2/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
    sys.exit(main())
