#!/bin/bash
# =============================================================================
# Cycle 49 -- the META-STEPSIZE LADDER.  Runs on cluster_b.  36 jobs.
# This is CORRECTIONS 52.4's named next experiment, pre-registered here before it runs.
#
# OPEN QUESTION.  Cycle 48 measured the scale profile at exactly TWO distances from the
# meta-optimum: beta frozen (b = 0.343) and beta free at ms=1e-3 (b = 0.065).  CORRECTIONS
# 47's reading is that a step-size adapter is a HIGH-PASS FILTER on meta-gradient
# co-fluctuation -- suppressing 22.92x at k=1, 5.49x at k=775, and 0.77x (not at all) at
# k=180,225.  A filter has a STRENGTH.  If that reading is right, b is a CONTINUOUS
# function of how hard the adapter is driven, and meta-stepsize is the drive.  Two points
# cannot distinguish a dial from a switch.  This ladder turns it into a curve.
#
# PRE-REGISTERED PREDICTIONS -- write the verdict against these, not around them.
#   (L0) VALIDITY GATE, and it gates everything else.  The ms=1e-3 rung is a straight
#        re-run of the c48 `fr5-*-a3` config and must reproduce it: free rho_s(weightwise)
#        within 3x of **8.888e-08**, and b within the seed spread of **0.065**.
#        It is included at full n=3 rather than reused from c48 SO THAT all three rungs are
#        measured by ONE instrument in ONE batch -- cross-batch instrument drift is exactly
#        the confound CORRECTIONS 51 found in `s`.  If L0 fails, nothing else is quoted.
#   (L1) THE MONOTONE PREDICTION.  b falls monotonically as meta-stepsize rises:
#            b(1e-4) > b(1e-3) > b(1e-2),  with b(1e-4) in [0.10, 0.30]
#        i.e. ms=1e-4 sits between the frozen 0.343 and the adapted 0.065, and ms=1e-2 is
#        at or below 0.065.  Anchors: frozen b = 0.343 (48.5), free-at-1e-3 b = 0.065 (48.14).
#        REFUTATION: b non-monotone in ms, OR the three b values flat to within their seed
#        spread.  Either outcome means "frozen vs free" is a SWITCH, not a dial, and
#        CORRECTIONS 47's filter language must be withdrawn and rewritten as a two-state
#        contrast.  Write it as that, not as a null.
#   (L2) SCALE-SELECTIVITY IS PRESERVED AT EVERY RUNG.  At each ms, the frozen/free
#        suppression ratio is monotone DECREASING in k: weightwise > nodewise > layerwise,
#        with layerwise ~1 (no suppression).  And the k=1 ratio grows with ms.
#        REFUTATION: the ordering inverts at any ms, or the k=1 ratio is flat in ms.
#   (L3) Reported as a BOUND, never as an absence: the layerwise rung (see RESOLUTION).
#
#   RESOLUTION -- restated before submission per CORRECTIONS 33's standing rule.  rho_min is
#   set by T and by m; it does NOT depend on meta-stepsize, so c48's measured floors carry
#   over unchanged (FINDINGS table at 48, "rung | m | frozen rho_s | rho_min | ..."):
#       rung        rho_min (T=1000 steady, 3 seeds pooled)   verdict
#       weightwise  8.5e-09                                   RESOLVES (free rho_s ~ 3.1x it)
#       nodewise    7.0e-06                                   RESOLVES (~1.4x it; thin)
#       layerwise   1.3e-03  (~7.4e-04 pooled)                WILL NOT -- report as a bound
#       blk6        >8.6e-02                                  DID NOT RESOLVE even frozen
#   blk6 is retained ONLY because b is fit ACROSS rungs and dropping it would leave a
#   3-point fit, which CORRECTIONS 51 already flags as a summary rather than a description.
#   Its own rho is not quotable.  **A layerwise null here means "no correlation above
#   rho_s = 1.3e-03", NOT "no correlation".**
#
#   KNOWN RISK, PRE-REGISTERED SO IT CANNOT BE READ AS A RESULT.  BETA_CLIP=-15:-2.3026 is
#   on every arm (it must be -- removing it would change more than one field).  At ms=1e-2
#   beta moves ~10x faster than the calibrated 1e-3 recipe and MAY saturate the guard.  A
#   clip-saturated arm measures a CLIPPED process, not a faster-adapting one -- this is
#   FINDINGS 36.3's confound arriving on a new axis.  MANDATORY CHECK before quoting the
#   ms=1e-2 column: report the fraction of steps at which the guard binds, per arm.  If it
#   binds on >50% of steps, that column is reported as CONFOUNDED and L1 is decided on
#   {1e-4, 1e-3} plus the frozen anchor only.
#
# DESIGN.  Byte-for-byte the c48 `fr5-*-a3` free recipe with ONE field varied,
# `--meta-stepsize`, so the ladder is not confounded by anything else.  20 epochs, PROBE=5,
# T=2000 records -- changing the horizon would break L0, which is the point of L0.
# 4 rungs x ms {1e-4, 1e-3, 1e-2} x 3 seeds = 36.  n=3 (not CORRECTIONS 52.4's "~12 jobs"
# at n=1) because the resolution above is quoted for 3 pooled seeds; n=1 would not resolve
# the layerwise bound and would leave every b a single-draw fit.
#
# WHY cluster_b.  Its comparator is the c48 `fr5-*` free batch, which ran on cluster_b.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. beta_true_max != beta_true_min on >=1 record -- beta really is FREE.  Run, not assumed.
#   2. neg_counts.json exists, n_records == 2000, n_tot matches the arm
#   3. sum(neg_counts)/(n_records*n_tot) == time-mean of frac_neg to <1e-6
#   4. python3 analysis/probe5_floor.py --selftest && analysis/probe5_window.py --selftest
#   5. NEW: guard-binding fraction per arm (the KNOWN RISK above)
#   6. Window scan on every arm.  CORRECTIONS 47: the full-run default would have reported
#      the opposite of the truth.  Never quote a profile number without its window.
#
# USAGE
#   bash c49_ms_ladder_p5.sh            # dry run
#   bash c49_ms_ladder_p5.sh --submit   # submits, only if the guards pass
# =============================================================================
set -u

WS=/cluster/account_b/metaopt
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/ml5
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_b
PEND_CAP=40
NJOBS=36
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c49 META-STEPSIZE LADDER + PROBE5 (cluster_b) ==="

if ! grep -q PATCH_PROBE5 "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5 not present in $HF"; exit 1
fi
if ! grep -q PATCH_PROBE5_FIX "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5_FIX not present -- unfixed patch crashes at record 500"; exit 1
fi
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX present  OK"

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 2: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
for G in resnet18_blocks layerwise nodewise weightwise; do
  case $G in
    resnet18_blocks) TAG=blk6 ;;
    layerwise)       TAG=lay  ;;
    nodewise)        TAG=node ;;
    weightwise)      TAG=w    ;;
  esac
  for MS in 1e-4:m4 1e-3:m3 1e-2:m2; do
    MSV=${MS%:*}; MST=${MS#*:}
    for S in 0 1 2; do
      RN="ml5-${TAG}-${MST}-s${S}"
      PD="$SAVE/probe_${TAG}_${MST}_s${S}"
      CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
        --export="ALL,AUGMENT=1,BETA_CLIP=-15:-2.3026,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
        "$RUNNER"
        --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
        --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
        --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
        --max-time 999:00:00 --gamma 1 --meta-stepsize "$MSV" --alpha0 1e-3
        --num-epochs 20 --stepsize-groups "$G" --seed "$S"
        --save-directory "$SAVE" --run-name "$RN")
      N=$((N+1))
      if [ "$SUBMIT" = "1" ]; then "${CMD[@]}" >/dev/null; else printf '%s\n' "${CMD[*]}"; fi
    done
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo SUBMITTED || echo 'dry run, nothing submitted' )) ----"
echo "reduce with:  python3 analysis/probe5_window.py <probe root>   (steady half)"
