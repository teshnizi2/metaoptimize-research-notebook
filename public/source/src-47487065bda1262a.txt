# cVT2_injectladder_score.py -- THE REGISTERED SCORER FOR `cvt2`:
#   AN INJECT DOSE LADDER -- IS THE COLLAPSE A THRESHOLD OR GRADED IN K, AND WHAT
#   WAS cvt1's 0.34 RESIDUE?  (CORRECTIONS 233)
#   ONE SUBMISSION, 21 JOBS (7 arms x 3 seeds), 100 EPOCHS, SEEDS {87,88,89},
#   PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM,
#   run from cvt1's ISOLATED tree $WS/harness_cvt1 (PATCH_PLAINNET + PATCH_VOTEWEIGHT),
#   UNCHANGED (no new harness code).
#
# Committed BEFORE any cvt2 run exists (STANDING RULE 21).  RUN IT UNEDITED
# (RULE 16).  The documented invocation takes ONE argument:
#
#     python3 analysis/cVT2_injectladder_score.py <runsdir>
#
# --manifest and --csv are OPTIONAL and DEFAULTED (the 164.2 pattern).
#
# =============================================================================
# FROZEN BARS (O2) -- READ THIS FIRST
# =============================================================================
# Every bar, sigma, tolerance and threshold score() uses is a LITERAL here.
# score() NEVER reads a bar term from results/all_runs.csv (it prints a row count
# as a disclosure and nothing else).  The ONLY data-dependent term that can enter
# a bar is SIGMA_INBATCH, through compose_sigma(), a max() with the frozen floor
# taking ONE argument.  --selftest proves it: the SAME synthetic runs give a
# byte-identical FINAL line (and identical SIGMA/SE/bar/PRIMARY lines) against the
# real corpus, the real corpus + INVENTED rows at THIS batch's own cells
# (PlainNet18_c100 `scalar` for k01, and the HEAD `sets:` string every ladder arm
# carries, since the CSV has no VOTE_W column) and at the ResNet18_c100 cells the
# frozen sigma came from, invented rows only, and no corpus.
#
# THE FROZEN NOISE FLOOR WAS DERIVED THROUGH analysis/corpus_exclusions.filter_rows
# (docs/OPERATIONS.md section 36): cvt1's 9 listed MUTE / DOSE / INJECT rows are
# dropped BEFORE any cell is pooled.  Section C of --selftest re-derives both
# sigmas through filter_rows on the registration corpus and shows the naive
# (unfiltered) PlainNet sigma is NOT the frozen literal.
#
# =============================================================================
# THE QUESTION (cvt1, CORRECTIONS 227 + 230)
# =============================================================================
# On PlainNet18_c100, isolating layer4.1.bn2.weight (1-based idx 50) rescues the
# scalar collapse (cvt1 HEAD 64.79 vs k01 11.95); adding its bn1 twin's term
# (idx 47) x691 INSIDE HEAD's rescued complement re-collapses the complement's step
# size (pin 37.6) but the arm keeps 0.3442 of the HEAD-k01 gap (INJECT 30.14).
# OPEN: is the collapse a THRESHOLD in the twin's weight K or GRADED, and is the
# 0.34 residue dose/onset (a larger K would remove it) or does it belong to HEAD's
# isolated tensor (every K above the knee floors near 30)?
#
# THE LADDER (derived from cvt1's own probe records, not from the brief's estimates;
# analysis/cvt2_registration_derivations.py, CORRECTIONS 233.3).  Lion moves beta by
# ms*sign(b2*mom + (1-b2)*z): only the SIGN of the complement's sum matters, not the
# twin's SHARE of its |L| mass.  Two record-level quantities bracket the knee:
#   K_INIT    = 5     the smallest K that turns >= 0.95 of HEAD's epoch 18-36
#                     complement records DOWN on every seed (HEAD's trajectory,
#                     counterfactual): below it the collapse cannot START.
#   K_SUSTAIN = 84.407 (min over seeds of the median; 85.28 / 99.78 / 84.41), max
#                     117.78 over every record from epoch 20: on INJECT's OWN collapsed
#                     records the other 51 terms vote UP on every record, and the twin
#                     at K keeps the record DOWN only if K > -rest/L_47.  Below it the
#                     collapse cannot HOLD.
#   rungs 13, 33   = K_INIT * (K_SUSTAIN/K_INIT)^(1/3), ^(2/3) = 12.83, 32.90 -- the
#                     initiate-but-not-sustain interval split in three equal log parts;
#   rung 152       (the brief's) = 1.29 x the largest K_SUSTAIN: sustains on every record;
#   rung 691       cvt1's INJECT, replicated;
#   rung 2000      the brief's top dose.
#   The brief's 258 is DROPPED: like 152 and 691 it sits above every collapsed record's
#   K_SUSTAIN (x2.19), so it carries no shape information 152 lacks.  The brief's
#   share-based estimates re-derive on INJECT's records (epochs 17-36, pooled) as
#   K(share 0.50) = 116.8, K(0.63) = 212.5 -- both ABOVE the sustain knee.
#
# THE ARMS (every name re-derived on the LIVE 53-tensor model by the launcher)
#   k01     scalar                                   VOTE_W unset      the floor anchor
#   HEAD    sets:1-49,51-53/layer4.1.bn2.weight      VOTE_W unset      positive control (K = 1)
#   K13     HEAD's string   VOTE_W=layer4.1.bn1.weight:13
#   K33     HEAD's string   VOTE_W=layer4.1.bn1.weight:33
#   K152    HEAD's string   VOTE_W=layer4.1.bn1.weight:152
#   K691    HEAD's string   VOTE_W=layer4.1.bn1.weight:691    (== cvt1's INJECT, byte for byte)
#   K2000   HEAD's string   VOTE_W=layer4.1.bn1.weight:2000
#
# DOSE AND ONSET ARE CONFOUNDED -- DISCLOSED.  PATCH_VOTEWEIGHT has no time gate: K
# acts from step 0.  On HEAD's records the twin's OWN term votes DOWN only from about
# epoch 16 (no record in epochs 0-14 has an UP-voting complement and a DOWN-voting
# twin), so on HEAD's trajectory the epoch from which the twin dominates the
# complement moves only from 19.4 (K 10) to 16.4 (K 2000) to 15.4 (K 30000): onset
# is nearly fixed by the twin's own sign.  The ladder varies dose; it cannot move
# onset by more than ~3 epochs, and it cannot separate "late" from "not 50's".
# The per-arm DOM epoch below is REPORTED BESIDE DOSE, descriptively.
#
# =============================================================================
# THE REGISTERED ACCOUNTS (every arm's predicted LEVEL; all UNSURE)
# =============================================================================
# Anchors (cvt1, between batch, non-gating): k01 11.9493, HEAD 64.7940, INJECT 30.1400.
# K01_PL 8-16 (ANCHOR) | FREE_PL 56-72 | PLAT_PL 22-38 | BETW_HI 48-56 | BETW_LO 38-48
# | BETW_152 34-42 | TOPDROP_PL 16-22.
#   SIGN-SAT-THRESHOLD   below K_SUSTAIN the collapse starts but cannot hold and the
#                        complement recovers to HEAD; above it Lion's sign saturates
#                        and every K is the same run.  K13, K33 FREE; K152, K691,
#                        K2000 PLAT.           -> THRESHOLD | TOP-FLAT (knee (33,152])
#   SIGN-SAT-GRADED      below K_SUSTAIN the complement settles at a K-dependent
#                        equilibrium; above it, sign saturation.  K13 BETW_HI, K33
#                        BETW_LO; K152, K691, K2000 PLAT.   -> GRADED | TOP-FLAT
#   KNEE-AT-INITIATION   any K that starts the collapse holds it.  Every rung PLAT.
#                                              -> KNEE-BELOW-LADDER | TOP-FLAT
#   DOSE-THROUGH         graded all the way up (the brief's dose/onset reading of the
#                        0.34): K13 BETW_HI, K33 BETW_LO, K152 BETW_152, K691 PLAT,
#                        K2000 TOPDROP.        -> GRADED | TOP-DROPS
#   THRESHOLD-TOP-DROP   a sustain knee, and the top dose removes the residue.
#                        K13, K33 FREE; K152, K691 PLAT; K2000 TOPDROP.
#                                              -> THRESHOLD | TOP-DROPS
#   BROKEN-PATCH (null)  the weights never reach the sum: every ladder arm == HEAD.
#                                              -> REPLICATE-FAILED; G-BITE separates it.
# FLOOR GATE (164.6).  k01 is the ANCHOR and is the floor by construction (its and any
# floor-band reading is a LOCATION).  NO OTHER ARM is predicted at the floor under ANY
# account: every non-anchor lower bound is >= K01_REF + NULL_BAR + 1.0 = 14.95 pp
# (lowest: TOPDROP 16).  The brief's "K 2000 drops INJECT TO k01's LEVEL" is therefore
# NOT registered as a point prediction: TOP-DROPS is read as a DROP from K691 by
# >= COLLAPSE_BAR, and a K2000 reading inside the floor band is stamped TOP-AT-K01 /
# FLOOR-READINGS-ARE-BOUNDS (a bound, not an agreement).
#
# =============================================================================
# THE NOISE FLOOR -- FROZEN (derived through corpus_exclusions.filter_rows)
# =============================================================================
# results/all_runs.csv at 2,983 rows (sha 1aa245d6...), 9 listed rows dropped
# (CORPUS-EXCLUSIONS.tsv sha 31105e32...), pooled within-cell SD of plateau5 at the
# standard cell (100 ep, AUGMENT=1, BETA_CLIP -15:-2.3026, ms 1e-3, alpha0 1e-6,
# batch 100; superseded / collapsed / incomplete dropped; 15 CELLKEYS):
#   SIGMA_R18ALL  0.694442846939599     ResNet18_c100, every granularity (df 209)
#   SIGMA_PLAIN   0.46281639069317043   PlainNet18_c100 = cpl1 + cpl2 + cvt1 k01/HEAD (df 29, 7 cells)
#                 (UNFILTERED it would read 7.895507, df 38 -- the trap)
#   SIGMA_PRIOR := max = 0.694442846939599;  SE_ARM_DIFF at the prior 0.567010 pp.
#
# =============================================================================
# STATISTICS (all IN BATCH; plateau5 from each run's own .out; TRAIN beside TEST)
# =============================================================================
#   D_HEAD  = HEAD - k01                  positive control (>= GAP_MIN)
#   P691    = HEAD - K691                 the replicate's collapse (>= SHAPE_GAP)
#   PRIMARY SHAPE: each rung K in (K13, K33, K152) is, against HEAD and PLATEAU := K691,
#     AT-HEAD     HEAD - L_K <= MATCH_BAR
#     AT-PLATEAU  |L_K - L_691| <= MATCH_BAR
#     BETWEEN     otherwise (more than MATCH_BAR from both)
#   PRIMARY TOP:  TOP = L_691 - L_2000
#   DESCRIPTIVE:  SHARE_KEPT_K = (L_K - k01)/(HEAD - k01) for every rung.
#
# =============================================================================
# THE BARS, FROZEN (pp; SE at the frozen prior 0.567010 in brackets)
# =============================================================================
#   GAP_MIN       20.0 [35.27]  D_HEAD >= 20
#   K01_MAX       20.0          k01 mean <= 20
#   SHAPE_GAP     15.0 [26.45]  P691 >= 15, so AT-HEAD, BETWEEN and AT-PLATEAU are disjoint
#                               and BETWEEN is >= 5 pp wide (cvt1: 34.65)
#   MATCH_BAR      5.0 [ 8.82]  "at" a level; also the monotonicity tolerance
#   COLLAPSE_BAR  10.0 [17.64]  TOP >= 10 -> TOP-DROPS
#   NULL_BAR       2.0 [ 3.53]  within 2 pp of k01 -> a floor-band reading (a LOCATION)
#   DIVERGED_BAR   5.0          within-arm seed range
#   FLOOR_MIN     15.0 / CEIL_MAX 90.0   max arm mean -> HARNESS-UNSOUND
#   G-BITE: BITE_TOL 1e-4 (relative to sum_i |w_i z_tensor_i| of the group); INFORM 1e-2;
#           >= 50 informative records on every ladder run; 500 records on every run.
#
# =============================================================================
# HARNESS GATES (-> HARNESS-UNSOUND, rc 1)
# =============================================================================
#   G-ARGS   every run's OWN ARGS line (exactly one, no repeated flag) has its arm's
#            registered stepsize-groups, PlainNet18_c100, CIFAR100, 100 ep, batch 100,
#            ms 1e-3, alpha0 1e-6, SGDm/Lion/HF, its seed and run-name.
#   G-ENV    one distinct ENV line (PROBE_DIR stripped) == cvt1's == cpl2's; one
#            `PROBE_TENSOR: on every=100 type=<arm's> tensors=53` per run.
#   G-VOTEW  exactly one `VOTE_W:` line per run, == WITNESS[arm], byte for byte.
#   G-STRUCT PARTITION-MANIFEST.txt BYTE-IDENTICAL to synthetic_manifest_text().
#   G-PROV   PROVENANCE.txt: MODE submit; BUILD_NETWORK_SHA256 == POST_SHA; HF_SHA256 ==
#            HF_POST_SHA (cvt1's PATCH_VOTEWEIGHT tree); SCORER_SHA256 == this file.
#   G-BITE   (-> PATCH-NOT-VERIFIED, rc 1) per run, from <runsdir>/cvt2/probe_cvt2-<arm>-s<seed>/:
#            500 parseable records; z_agg == sum_i w_i z_tensor_i and mom_pre == sum_i w_i
#            m_tensor_i per group at every record; every ladder run has >= 50 records where
#            weighted and unweighted sums differ by > INFORM and on NONE of them the
#            unweighted sum matches.  A wrong K (e.g. K2000 carrying 691) fails here.
#
# =============================================================================
# THE BRANCH MAP, FROZEN (decide(); first match wins).  FINAL: <SHAPE> | <TOP> | stamps
# =============================================================================
#   INCOMPLETE / HARNESS-UNSOUND / PATCH-NOT-VERIFIED   (above)
#   UNRESOLVED-DIVERGED        an arm's seed range > 5
#   SCALAR-NOT-COLLAPSED       k01 > 20
#   POSITIVE-CONTROL-FAILED    D_HEAD < 20
#   REPLICATE-FAILED           P691 < 15 (cvt1's INJECT did not re-collapse enough to host a shape)
#   LADDER-ABOVE-HEAD          some rung - HEAD > 5 (unregistered; stop)
#   NON-MONOTONE               along HEAD(K=1), K13, K33, K152, K691, K2000 some level RISES
#                              by > 5 over its predecessor (unregistered; stop)
#   -- SHAPE (K13, K33, K152 against HEAD and K691):
#   GRADED                     any rung BETWEEN
#   KNEE-BELOW-LADDER          every rung AT-PLATEAU (knee at K <= 13; shape unresolved)
#   THRESHOLD                  AT-HEAD rungs then AT-PLATEAU rungs, none BETWEEN
#                              (stamp KNEE-IN-(a,b] with the bracketing rungs; all three
#                              AT-HEAD -> KNEE-IN-(152,691])
#   -- TOP:
#   TOP-FLAT                   |TOP| <= 5
#   TOP-ATTENUATED             5 < TOP < 10
#   TOP-DROPS                  TOP >= 10
#
# STAMPS (never the branch): HARNESS-CLEAN; PATCH-BITES; ONE-NETWORK-PLAINNET;
#   HORIZON-100-ONLY; DOSE-ONSET-CONFOUNDED; SIGMA-PRIOR-FROZEN / SIGMA-INBATCH;
#   POSITIVE-CONTROL-REPRODUCES; K13-/K33-/K152-AT-HEAD / -BETWEEN / -AT-PLATEAU;
#   KNEE-IN-(a,b]; TOP-AT-K01; FLOOR-READINGS-ARE-BOUNDS; TRAIN-AGREES / TRAIN-DISAGREES.
#
# plateau5 is PRIMARY (mean TEST over epochs 95..99 of each run's own .out); the CSV
# `plateau` column is read NOWHERE; best_test NOWHERE.  The probe readout after FINAL
# (pins, the twin's DOM epoch per arm, the TEST trajectory) is DESCRIPTIVE and cannot
# move FINAL.
# =============================================================================

def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def file_sha(path):

def q(v, p):
    v = sorted(v)
    if not v:
        return None
    k = (len(v) - 1) * p
    lo = int(k)
    hi = min(lo + 1, len(v) - 1)
    return v[lo] + (v[hi] - v[lo]) * (k - lo)

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists (each sorted); no HF.py import."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names):
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest INTEGER job id among COMPLETE runs wins."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes

def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm

def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail dict).  Pure function of the records."""
    names = [n for n, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    W = WEIGHTS[arm]
    d = {"n": 0, "bad_rec": 0, "bad_z": 0, "bad_m": 0, "worst_z": 0.0, "worst_m": 0.0,
         "informative": 0, "unweighted_match_on_informative": 0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for r in recs:
        ok = (r is not None and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS
              and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
              and isinstance(r.get("z_agg"), list) and len(r["z_agg"]) == 1 and isinstance(r["z_agg"][0], list)
              and len(r["z_agg"][0]) == len(groups)
              and isinstance(r.get("mom_pre"), list) and len(r["mom_pre"]) == 1 and isinstance(r["mom_pre"][0], list)
              and len(r["mom_pre"][0]) == len(groups))
        if not ok:
            d["bad_rec"] += 1
            continue
        try:
            zt = [float(x) for x in r["z_tensor"]]
            mt = [float(x) for x in r["m_tensor"]]
            za = [float(x) for x in r["z_agg"][0]]
            ma = [float(x) for x in r["mom_pre"][0]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in zt + mt + za + ma):
            d["bad_rec"] += 1
            continue
        d["n"] += 1
        ez = em = 0.0
        informative = False
        um = True
        for g, zg, mg in zip(groups, za, ma):
            sz = sum(abs(W.get(i, 1.0) * zt[i - 1]) for i in g) + 1e-30
            sm = sum(abs(W.get(i, 1.0) * mt[i - 1]) for i in g) + 1e-30
            wz = sum(W.get(i, 1.0) * zt[i - 1] for i in g)
            wm = sum(W.get(i, 1.0) * mt[i - 1] for i in g)
            uz = sum(zt[i - 1] for i in g)
            ez = max(ez, abs(wz - zg) / sz)
            em = max(em, abs(wm - mg) / sm)
            if W and abs(wz - uz) / sz > INFORM:
                informative = True
                um = um and abs(uz - zg) / sz <= BITE_TOL
        d["worst_z"], d["worst_m"] = max(d["worst_z"], ez), max(d["worst_m"], em)
        d["bad_z"] += ez > BITE_TOL
        d["bad_m"] += em > BITE_TOL
        if informative:
            d["informative"] += 1
            d["unweighted_match_on_informative"] += um
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] == N_RECORDS
          and d["bad_z"] == 0 and d["bad_m"] == 0)
    if W:
        ok = ok and d["informative"] >= MIN_INFORMATIVE and d["unweighted_match_on_informative"] == 0
    return ok, d

def beta_pin(recs, coord):
    vals = [(r["step"], r["beta"][coord]) for r in recs
            if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord
            and isinstance(r.get("step"), int)]
    if not vals:
        return None
    at = [v <= CLAMP_LO + 1e-3 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    return {"pin": pin, "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH), "last": vals[-1][1],
            "frac_clamp": sum(at) / float(len(at))}


def dom_epoch(recs, arm):
    """the registered DOM definition: in the group that holds the twin, the twin's weighted Lion term
    w*L_47 (L = b2*m + (1-b2)*z, raw per-tensor terms) outweighs the NET of the group's other terms and
    shares the total's sign; DOM epoch = first record from which >= DOM_FRAC of the next DOM_WIN
    records are dominated."""
    names = [n for n, _q, _o in r18_tensors(True)]
    g0 = arm_groups(arm, names)[0]
    w = WEIGHTS[arm].get(IDX_TWIN, 1.0)
    flags, eps = [], []
    for r in recs:
        if not (isinstance(r, dict) and isinstance(r.get("m_tensor"), list) and isinstance(r.get("z_tensor"), list)
                and len(r["m_tensor"]) == NTENS and len(r["z_tensor"]) == NTENS and isinstance(r.get("step"), int)):
            continue
        b2 = r.get("pt_b2", 0.9)
        L = [b2 * m + (1 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
        t = w * L[IDX_TWIN - 1]
        rest = math.fsum(L[i - 1] for i in g0 if i != IDX_TWIN)
        flags.append(abs(t) > abs(rest) and ((t > 0) == (t + rest > 0)))
        eps.append(r["step"] / float(STEPS_PER_EPOCH))
    for j in range(len(flags) - DOM_WIN + 1):
        if sum(flags[j:j + DOM_WIN]) >= DOM_FRAC * DOM_WIN:
            return eps[j]
    return None

# THE BRANCH MAP -- a pure function of the in-batch means and one flag
# =============================================================================
def rung_state(M, a):
    if M["HEAD"] - M[a] <= MATCH_BAR:
        return "AT-HEAD"
    if abs(M[a] - M["K691"]) <= MATCH_BAR:
        return "AT-PLATEAU"
    return "BETWEEN"


def decide(M, diverged=False):
    """-> (shape_branch, top_token or None, knee_stamp or None)."""
    if diverged:
        return "UNRESOLVED-DIVERGED", None, None
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED", None, None
    if M["HEAD"] - M["k01"] < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED", None, None
    if M["HEAD"] - M["K691"] < SHAPE_GAP:
        return "REPLICATE-FAILED", None, None
    if any(M[a] - M["HEAD"] > MATCH_BAR for a in LADDER):
        return "LADDER-ABOVE-HEAD", None, None
    if any(M[CHAIN[i + 1]] - M[CHAIN[i]] > MATCH_BAR for i in range(len(CHAIN) - 1)):
        return "NON-MONOTONE", None, None
    st = [rung_state(M, a) for a in SHAPE_RUNGS]
    knee = None
    if "BETWEEN" in st:
        shape = "GRADED"
    elif all(x == "AT-PLATEAU" for x in st):
        shape = "KNEE-BELOW-LADDER"
    else:
        shape = "THRESHOLD"
        n_head = sum(1 for x in st if x == "AT-HEAD")
        bounds = (1, 13, 33, 152, 691)
        knee = "KNEE-IN-(%d,%d]" % (bounds[n_head], bounds[n_head + 1])
    top = M["K691"] - M["K2000"]
    if abs(top) <= MATCH_BAR:
        tok = "TOP-FLAT"
    elif top >= COLLAPSE_BAR:
        tok = "TOP-DROPS"
    else:
        tok = "TOP-ATTENUATED"
    return shape, tok, knee

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    nall = len(ARMS) * len(SEEDS)
    print("=" * 78)
    print(" cvt2 -- AN INJECT DOSE LADDER ON PlainNet18_c100: THRESHOLD OR GRADED, AND WHAT WAS THE 0.34?")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-6s %-38s VOTE_W=%s" % (a, SPEC[a], VOTEW[a] or "(unset)"))
    print(" PRIMARY: SHAPE of K13/K33/K152 against HEAD and K691's plateau;  TOP = K691 - K2000.")
    print("          plateau5 = mean TEST over epochs %d..%d of each run's own .out." % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2); the corpus cannot move them.  DOSE AND ONSET ARE CONFOUNDED (no time gate).")
    print("=" * 78)
    rows = read_corpus(csvpath)
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)"
          % ("not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)))

    runs, rnotes = read_runs(runsdir)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback" % (nall, EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (nall - len(missing), nall, ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    every run's OWN ARGS line says what its file name registers")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s)}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad])
                 if (dup or bad) else "")

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),
         "G-ENV one distinct ENV line, one per run", "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
    hchk(envs == [ENV_EXPECTED], "G-ENV the ENV line is cvt1's (== cpl2's), byte for byte (PROBE_DIR stripped)")
    badpt = [(a, s) for (a, s), r in sorted(runs.items())
             if len(r["pt"]) != 1 or not PT_RE.match(r["pt"][0])
             or PT_RE.match(r["pt"][0]).group(1) != PT_TYPE[a]]

         % (PROBE, NTENS), "bad: %s" % badpt[:5] if badpt else "%d runs" % len(runs))

    print("\nG-VOTEW   every run printed exactly ONE VOTE_W line, == its arm's registered witness")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == the frozen manifest, byte for byte")
    mp, how = resolve_manifest(runsdir, manifest_path)
    if not mp:
        hchk(False, "G-STRUCT manifest present", how)

        print("  manifest: %s (%s)" % (mp, how))
        hchk(txt == synthetic_manifest_text(),
             "G-STRUCT byte-identical to synthetic_manifest_text() (53 tensors, [52,1] on six arms, the VOTE_W table)",
             "%d vs %d bytes" % (len(txt), len(synthetic_manifest_text())))

    print("\nG-PROV    PROVENANCE.txt")
    pp, phow = resolve(runsdir, "", "PROVENANCE.txt")
    prov = read_kv(pp)
    if prov is None:
        hchk(False, "G-PROV PROVENANCE.txt present", phow)
    else:
        print("  provenance: %s (%s)" % (pp, phow))
        hchk(prov.get("MODE") == "submit", "G-PROV MODE submit", str(prov.get("MODE")))

             "G-PROV build_network.py sha256 == POST_SHA %s..." % POST_SHA[:12], str(prov.get("BUILD_NETWORK_SHA256"))[:16])
        hchk(prov.get("HF_SHA256") == HF_POST_SHA,
             "G-PROV HF.py sha256 == HF_POST_SHA %s... (cvt1's PATCH_VOTEWEIGHT tree)" % HF_POST_SHA[:12],
             str(prov.get("HF_SHA256"))[:16])
        me = file_sha(os.path.abspath(__file__))
        hchk(prov.get("SCORER_SHA256") == me, "G-PROV SCORER_SHA256 == this scorer's sha256 %s..." % me[:12],
             str(prov.get("SCORER_SHA256"))[:16])

    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH; tail slope = TEST OLS over epochs 80-99 (descriptive)")
    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)
    arm_t = dict((a, [runs[(a, s)]["train5"] for s in SEEDS]) for a in ARMS)
    for a in ARMS:
        v, t = arm_v[a], arm_t[a]
        sl = [tail_slope(runs[(a, s)]["eps"]) for s in SEEDS]

    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    hi = max(M.values())
    print("\nG-FLOOR / G-CEIL")
    hchk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE), "max %.4f" % hi)

    if harness:

            len(harness), "; ".join(h.split()[0] for h in harness))

    print("\nG-BITE    the registered weight reached the complement's sum -- read from the runs' OWN probe records")

def _e_probe_records(arm, weights=None, n=N_RECORDS, seed=0):
    """synthetic PROBE_TENSOR records consistent with `weights` (default: the arm's own)."""
    import random
    rnd = random.Random(1000 + seed)
    W = WEIGHTS[arm] if weights is None else weights
    names = [nm for nm, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    out = []
    for k in range(n):
        zt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        mt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        zt[IDX_HEAD - 1] = 5e-2 * (1 + 0.1 * rnd.random())
        mt[IDX_HEAD - 1] = 4e-2
        zt[IDX_TWIN - 1] = 1e-4 * (1 + 0.1 * rnd.random())
        mt[IDX_TWIN - 1] = 8e-5
        za = [sum(W.get(i, 1.0) * zt[i - 1] for i in g) for g in groups]
        ma = [sum(W.get(i, 1.0) * mt[i - 1] for i in g) for g in groups]
        b = max(-15.0, -6.0 - 0.02 * k)
        out.append({"step": 100 * k, "beta": [b] * len(groups), "pt_b2": 0.9, "pt_mp": 0.99,
                    "z_tensor": zt, "m_tensor": mt, "z_agg": [za], "mom_pre": [ma],
                    "beta_pre": [[b] * len(groups)]})
    return out

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def _dict_from_scorer(fname, key, var="SPEC"):
    p = os.path.join(_HERE, fname)
    if not os.path.exists(p):
        return None
    import ast
    m = re.search(r"^%s = (\{.*?^\})" % var, open(p).read(), re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)


def _rederive_ladder(cvt1dir):
    """K_INIT, the K_SUSTAIN medians / max, and the rungs, from cvt1's records (E3/E5/E6 re-typed)."""
    def L(r, i):
        b2 = r.get("pt_b2", 0.9)
        return b2 * r["m_tensor"][i] + (1 - b2) * r["z_tensor"][i]
    k_init = None
    for K in (1.0, 2.0, 3.0, 5.0, 10.0):
        fr = []
        for s in DERIV_SEEDS:
            sel = [r for r in load_probe(cvt1dir, "HEAD", s, "cvt1") if 18 <= r["step"] / 500.0 < 36]
            n = 0
            for r in sel:
                b2 = r.get("pt_b2", 0.9)
                Ac = b2 * r["mom_pre"][0][0] + (1 - b2) * r["z_agg"][0][0]
                t = L(r, IDX_TWIN - 1)
                n += (Ac - t + K * t) > 0
            fr.append(n / float(len(sel)))
        if min(fr) >= 0.95:
            k_init = K
            break
    meds, mx = [], 0.0
    for s in DERIV_SEEDS:
        post = []
        for r in load_probe(cvt1dir, "INJECT", s, "cvt1"):
            e = r["step"] / 500.0
            t = L(r, IDX_TWIN - 1)
            rn = math.fsum(L(r, i) for i in range(NTENS) if i not in (IDX_HEAD - 1, IDX_TWIN - 1))
            if t > 0 and rn < 0:
                if e >= 20:
                    mx = max(mx, -rn / t)
                if e >= 38:
                    post.append(-rn / t)
        meds.append(q(post, 0.5))
    return k_init, tuple(meds), mx

def main():
    ap = argparse.ArgumentParser(description="cvt2 registered scorer.  Documented invocation: "
                                             "python3 analysis/cVT2_injectladder_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt2/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc
