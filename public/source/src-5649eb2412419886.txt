#!/bin/bash
# =============================================================================
# cAW2_stdrecipe.sh -- `caw2`: THE STANDARD-RECIPE GATE, WITH EACH CONFOUND IN ITS OWN CELL.  CORRECTIONS 290.
#   ONE SUBMISSION, 27 JOBS (9 arms x 3 seeds), 100 EPOCHS, SEEDS {160,161,162}, ResNet18_c100 / CIFAR-100,
#   PROBE=100, PROBE_TENSOR=1 ON EVERY ARM, --constraint=L4 ON EVERY JOB.
#   caw2 is the SUCCESSOR of caw1 (cad9fba / 16faa5f), WITHDRAWN BEFORE LAUNCH (the verifier's V2); no caw1 job exists.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Every guard and scancel line is filtered on `caw2-`.
#
# REGISTERED SCORER: analysis/cAW2_confound_score.py (+ analysis/caw2_design.py, and cwd1's analysis/cwd_design.py and
# analysis/cwd_common.py REUSED UNEDITED), committed and pushed BEFORE this batch is submitted (RULE 21).
#   DOCUMENTED INVOCATION:  python3 analysis/cAW2_confound_score.py <runsdir>
#
# THE ARMS (analysis/caw2_design.py; no hold patch, no mask, on any arm)
#   K01      SGDm(0.99) + Lion(0.99,0.9)    wd 0.1  scalar        the mechanism cell: IN-BATCH POSITIVE CONTROL
#   MS | ML  SGDm(0.99) + Adam(0.9,0.999)   wd 0.1  both grains   the META swap alone
#   LS | LL  AdamW(0.9,0.999) + Lion        wd 0.1  both grains   the BASE swap alone
#   AS | AL  AdamW + Adam                   wd 0.1  both grains   the STANDARD RECIPE
#   XS | XL  AdamW + Adam                   wd 1.0  both grains   the DOSE arm
#
# NO NEW HARNESS CODE: every flag is an existing CLI flag of train.py; cwd1's tree $WS/harness_cwd1/cifar10 (HF.py
# 94aedc33...) and runner $WS/jobs/run_cifar_cwd1.sh (34a8c90e...) run UNCHANGED.  The harness-null proof is
# bin/cAW2_realrun_proof.sbatch (tests/test_caw2_realrun.py), whose .out guard 4f reads at $WS/runs/caw2/proof_caw2.log.
#
# RULE 20.  Every ARGS value is ONE shell token.  The base half, the meta half, the decay and the grain VARY BY ARM,
# so each composed line is checked against its OWN arm's full registered flag list (analysis/caw2_design.args_pairs),
# not a batch-wide array.  analysis/argsline_guard.py IS NOT EDITED.  Exclusion rows owed at landing: LS/LL/AS/AL
# ARGS_MOMENTUM_BASE; XS/XL two ARGS kinds (0.9 and 1.0; corpus_exclusions.py cannot list that today -- reported,
# not fixed, and corpus_exclusions.py is NOT edited); K01/MS/ML none.
#
# HARDWARE: every job carries --constraint=L4; the scorer only DISCLOSES the device (G-HW).
# COST: cwd5's sacct mean 0.6855 GPU-h per run of this cell on L4 -> 27 jobs ~= 18.5 GPU-h expected (the AdamW base is
#   not timed on this cell; UNSURE by +/- 20 %).  HARD BOUND = WALL x 27 = 81 GPU-h.
#
# USAGE   bash bin/cAW2_stdrecipe.sh [--dry-run]   |   bash bin/cAW2_stdrecipe.sh --submit   (the ONLY submitting form)
# ENV     CAW2_REGISTERED_COMMIT=<40-hex sha>   on a host with no git repo, --submit requires it (271.6 F1: full sha)
# =============================================================================
set -u
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
unset VOTE_W BETA_HOLD COMP_HOLD WINDOW_HOLD GROUP_HOLD REST_HOLD DECAY_MASK
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
B=caw2
HARN=$WS/harness_cwd1/cifar10
RUNNER=$WS/jobs/run_cifar_cwd1.sh
SAVE=$WS/runs/caw2

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true
export PYTHONDONTWRITEBYTECODE=1

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="160 161 162"
MST=1e-3
A0=1e-6
NTENS=62
TOTPAR=11220132
ARMS="K01 MS ML LS LL AS AL XS XL"
NJOBS=27
WALL=03:00:00
GPU_CONSTRAINT=L4
BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_SHA=94aedc33046afb12782ffb7f2eb729bafc7fcba5f42227fec373fa67b6f58ebf
RUNNER_SHA=34a8c90ee3a0bee00aa2d0b4a3ad5d16d461fbf5a1f0cda5797e44e44c158371
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL" "gpu-l4-24g")

SC="$REPO/analysis/cAW2_confound_score.py"
DES="$REPO/analysis/caw2_design.py"
WDES="$REPO/analysis/cwd_design.py"
COM="$REPO/analysis/cwd_common.py"
SELF="$REPO/bin/cAW2_stdrecipe.sh"
R20="$REPO/bin/cAW2_rule20.sh"
R20E="$REPO/analysis/caw2_rule20_envaudit.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cAW2_realrun_proof.sbatch"
TEST2="$REPO/tests/test_caw2_realrun.py"
TEST1="$REPO/tests/test_caw1_realrun.py"
TESTR="$REPO/tests/test_grouphold_realrun.py"
STAGEW="$REPO/bin/cWD_stage_harness.sh"
ARCHR="$REPO/jobs/run_cifar_cwd1.sh"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
BNH="$HARN/build_network.py"
PLOG="$SAVE/proof_caw2.log"

echo "=============================================================="
echo " caw2 -- THE STANDARD-RECIPE GATE, EACH CONFOUND IN ITS OWN CELL (successor of the WITHDRAWN caw1)"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C   ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"
echo "   K01      SGDm(0.99)+Lion   wd 0.1  scalar       (positive control)"
echo "   MS | ML  SGDm(0.99)+Adam   wd 0.1  both grains  (META swap alone)"
echo "   LS | LL  AdamW+Lion        wd 0.1  both grains  (BASE swap alone)"
echo "   AS | AL  AdamW+Adam        wd 0.1  both grains  (the STANDARD RECIPE)"
echo "   XS | XL  AdamW+Adam        wd 1.0  both grains  (the DOSE arm)"
echo " 9 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cwd1's, UNCHANGED)   runner $RUNNER (cwd1's, UNCHANGED)   partitions $PARTS   constraint $GPU_CONSTRAINT   walltime $WALL"
echo " DOCUMENTED INVOCATION: python3 analysis/cAW2_confound_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${GROUP_HOLD+x}" ] && [ -z "${REST_HOLD+x}" ] && [ -z "${DECAY_MASK+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD, GROUP_HOLD, REST_HOLD and DECAY_MASK are UNSET in the launcher's environment" \
  || guard_fail "guard 0b: a hold / mask variable is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" || guard_fail "guard 0: no CSV at $CSVP"

for f in "$SC" "$DES" "$WDES" "$COM" "$SELF" "$R20" "$R20E" "$EXCL" "$EXCLT" "$RRSB" "$TEST2" "$TEST1" "$TESTR" "$STAGEW" \
         "$ARCHR" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

NEWF="analysis/cAW2_confound_score.py analysis/caw2_design.py bin/cAW2_stdrecipe.sh bin/cAW2_rule20.sh analysis/caw2_rule20_envaudit.py bin/cAW2_realrun_proof.sbatch tests/test_caw2_realrun.py"
REUSEF="analysis/cwd_design.py analysis/cwd_common.py analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv tests/test_caw1_realrun.py tests/test_grouphold_realrun.py bin/cWD_stage_harness.sh bin/_lib_guards.sh analysis/argsline_guard.py jobs/run_cifar_cwd1.sh analysis/caw1_design.py analysis/cAW1_stdrecipe_score.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the registration files and reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CAW2_REGISTERED_COMMIT:-}" ]; then
    if printf '%s' "$CAW2_REGISTERED_COMMIT" | grep -qE '^[0-9a-f]{40}$'; then
      echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CAW2_REGISTERED_COMMIT (40 hex, 271.6 F1)"
    else
      guard_fail "guard 1b: CAW2_REGISTERED_COMMIT is not a full 40-hex commit name (271.6 F1): $CAW2_REGISTERED_COMMIT"
    fi
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CAW2_REGISTERED_COMMIT is unset"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CAW2_REGISTERED_COMMIT=<40-hex sha>"
  fi
fi
[ "$(sha "$ARCHR")" = "$RUNNER_SHA" ] && [ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] \
  && echo "guard 1b2: the archived jobs/run_cifar_cwd1.sh == \$WS/jobs/run_cifar_cwd1.sh == $RUNNER_SHA (cwd1's runner, UNCHANGED)" \
  || guard_fail "guard 1b2: the archived runner or the live runner is not $RUNNER_SHA"
[ "$(sha "$WDES")" = 03d2d7614915757798699ff75ca971a11fca2056d383649fecbbf5adf06c0195 ] && [ "$(sha "$COM")" = d7ddb49e71013e3a21eaf5129861ab43dbc16e181bbd92af7c70cdc95e82bf52 ] \
  && [ "$(sha "$TEST1")" = 72bcb3e17ab7a8dfd41e2057671d237c46e1c3f5c4a5b09e0cbdac9959483f5a ] \
  && [ "$(sha "$REPO/analysis/caw1_design.py")" = 0795aedd68be22a6741241c667e20e9b9ca92a23399cc4a16fc59fb441a9bfac ] \
  && [ "$(sha "$REPO/analysis/cAW1_stdrecipe_score.py")" = bd259052a1a738cad71ff565dd8cc94c7888087f6d860c5ed7925a6f26e94793 ] \
  && echo "guard 1b3: the REUSED cwd_design.py / cwd_common.py / test_caw1_realrun.py and the WITHDRAWN caw1's design + scorer are their registered bytes (RULE 16)" \
  || guard_fail "guard 1b3: a reused or withdrawn registration file moved"

SELFT=/tmp/caw2_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION\|Traceback' "$SELFT" | head -20; tail -5 "$SELFT"; }
[ "$(grep -c '^  SKIP' "$SELFT")" = 0 ] && echo "guard 1c1: no SKIP (real records and the corpus were found on this host)" || guard_fail "guard 1c1: the selftest SKIPPED a section"
grep -q "^  PASS F FINAL identical over 3 corpus variants" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS E .python3 cAW2_confound_score.py <runsdir>. exits 0" "$SELFT" && grep -q "^  PASS E a dropped run -> INCOMPLETE rc 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS C ZERO corpus rows OUTSIDE caw2 carry a seed of the block 160..165" "$SELFT" && grep -q "^  PASS C ZERO corpus rows carry the caw2- prefix" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed / fresh-prefix checks ran and passed" \
  || guard_fail "guard 1c4: fresh-seed / prefix check did not pass"
grep -q "^  PASS B CALIBRATION: on the landed COLLAPSED arm (cwd5 k01W1)" "$SELFT" \
  && grep -q "^  PASS B DOSE CALIBRATION" "$SELFT" && grep -q "^  PASS B the dose word separates" "$SELFT" \
  && echo "guard 1c5: DOM_C and the DOSE were calibrated on landed cwd5 records (the verifier's peak shrinks reproduced)" \
  || guard_fail "guard 1c5: the landed-record calibration did not pass"
grep -q "^  PASS C SIGMA_R18ALL: the FROZEN literal 0.6415731883291765 IS the FILTERED value" "$SELFT" \
  && echo "guard 1c6: the frozen floor IS the corpus_exclusions.filter_rows value on the CURRENT corpus, not the naive one" \
  || guard_fail "guard 1c6: the frozen floor was not shown to be the filtered value on the current corpus"
grep -q "^  PASS A K01's flags == the landed cwd5 k01W1's ARGS line" "$SELFT" && [ "$(grep -c "^  PASS A .*'s optimiser flags == " "$SELFT")" = 3 ] \
  && grep -q "^  PASS A caw1 (WITHDRAWN BEFORE LAUNCH) analysis/cAW1_stdrecipe_score.py is its registered bytes" "$SELFT" \
  && echo "guard 1c7: K01 == landed cwd5 k01W1; AS / MS / LS == the landed CIFAR-10 cells' optimiser flags; caw1 unedited" \
  || guard_fail "guard 1c7: the landed-ARGS / withdrawn-predecessor checks did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

grep -q '^caw2-' "$CSVP" && guard_fail "guard 2: caw2-* already in the CSV" || echo "guard 2: no caw2-* row in the corpus"
NOUT=$(ls "$WS/runs"/caw2-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no caw2-*.out on disk" || guard_fail "guard 2b: $NOUT caw2-*.out already exist"
NOUT1=$(ls "$WS/runs"/caw1-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT1:-0}" = 0 ] && echo "guard 2b1: no caw1-*.out on disk (caw1 was WITHDRAWN BEFORE LAUNCH)" || guard_fail "guard 2b1: $NOUT1 caw1-*.out exist -- caw1 was NOT withdrawn before launch"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (16[0-5])( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry a seed of the block 160..165" \
                      || guard_fail "guard 2d: $NSEED .out files already ran a seed of 160..165"
NWD=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- "--weight-decay-base (1.0|1)( |$)")
echo "guard 2c: $NWD existing .out ARGS lines already carry --weight-decay-base 1.0 (DISCLOSURE: the X cell is new)"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^caw[12]-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 caw1-* / caw2-* jobs" || guard_fail "guard 2e: sacct shows $NS caw1-/caw2-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^caw2-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 caw2-* in squeue" || guard_fail "guard 2f: $NQ caw2-* already queued"
fi

# ---- guard 3/4a: cwd1's tree consistent (caw2 runs on it, as cwd3 / cwd4 / cwd5 did) ------------------------------
bash "$STAGEW" cwd1 >/tmp/caw2_stagew.$$ 2>&1 \
  && { echo "guard 4a1: cWD_stage_harness.sh cwd1 (verify mode) VERIFIES cwd1's tree, which caw2 uses UNCHANGED:"; grep -E 'V-a|V-h|V-d|V-f|VERIFIED' /tmp/caw2_stagew.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a1: cwd1's tree is NOT consistent"; tail -20 /tmp/caw2_stagew.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" || guard_fail "guard 3: CIFAR-100 not visible"
[ "$(sha "$BNH")" = "$BN_SHA" ] && echo "guard 3b: build_network.py == cvt8's ($BN_SHA)" || guard_fail "guard 3b: build_network.py"
[ "$(sha "$HFH")" = "$HF_SHA" ] && echo "guard 3c: HF.py == $HF_SHA (cwd1's)" || guard_fail "guard 3c: the tree's HF.py is not cwd1's"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == $RUNNER_SHA (cwd1's run_cifar_cwd1.sh)" || guard_fail "guard 3e: runner sha"
M3F=0; for fl in --alg-base --normalizer-param-base --momentum-param-base --weight-decay-base --alg-meta --normalizer-param-meta --momentum-param-meta --Lion-beta2-meta --weight-decay-meta; do
  grep -q -- "\"$fl\"" "$HARN/train.py" || { M3F=1; echo "    missing $fl"; }
done
[ "$M3F" = 0 ] && echo "guard 3f: train.py declares every base / meta flag the arms use (EXISTING registered CLI flags; no new code)" \
  || guard_fail "guard 3f: train.py lacks a flag"

# ---- guard 4f: the harness-null proof log ----------------------------------------------------------------------------
if [ -f "$PLOG" ] && grep -q "^REALRUN_RC 0$" "$PLOG" && grep -q "^ALL PASS$" "$PLOG" && [ "$(grep -c '^  FAIL' "$PLOG")" = 0 ] \
   && grep -q "^BATCH caw2$" "$PLOG" && grep -q "^NVIDIA L4, " "$PLOG" \
   && grep -q "^DESIGN_SHA256 $(sha "$DES")$" "$PLOG" && grep -q "^DRIVER_SHA256 $(sha "$TEST2")$" "$PLOG" \
   && grep -q "^REUSED_CHILD_SHA256 $(sha "$TEST1")$" "$PLOG" && grep -q "^CWD_DESIGN_SHA256 $(sha "$WDES")$" "$PLOG" \
   && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTR")$" "$PLOG" \
   && grep -q "^TREE $HARN HF $(sha "$HFH") build_network $(sha "$BNH") " "$PLOG" \
   && [ "$(grep -c "PASS RC1 .* BITWISE the formula with THE PASSED wd" "$PLOG")" = 9 ] \
   && [ "$(grep -c "PASS RC2 .* meta: beta (and its moments) BITWISE" "$PLOG")" = 9 ] \
   && [ "$(grep -c "PASS RC1 .* THE HARNESS-NULL IS EXCLUDED" "$PLOG")" = 6 ] \
   && [ "$(grep -c "PASS RC4 .* printed ONE PROBE_TENSOR line == the design's witness" "$PLOG")" = 9 ] \
   && [ "$(grep -c "PASS RC5 " "$PLOG")" = 5 ] \
   && grep -q "PASS RC6 AS run twice: identical final weights" "$PLOG" \
   && grep -q "PASS RC7 the nine arms' final weights are PAIRWISE DISTINCT" "$PLOG" \
   && grep -q "PASS RC8 at alpha0 1e-2 the wd-0.1 (AS) and wd-1.0 (XS) runs' final weights DIFFER" "$PLOG"; then
  echo "guard 4f: caw2's harness-null proof ALL PASS on THIS tree, THIS design, THIS driver: $(grep -c '^  PASS' "$PLOG") PASS / 0 FAIL (sha $(sha "$PLOG" | cut -c1-8))"
  echo "          RC1/RC2 every arm's base and meta BITWISE at its passed wd (9+9), the SGDm fallback excluded on the 6 AdamW"
  echo "          arms, RC4 the nine witnesses MEASURED, RC5 the vote decomposition on all five scalar arms, RC6-RC8"
else
  guard_fail "guard 4f: no ALL PASS caw2 proof log for HF $(sha "$HFH") / design $(sha "$DES") at $PLOG -- sbatch $RRSB, then copy its .out there"
fi

# ---- guard 4c': the launcher's literals == the design module's ------
python3 - "$DES" "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$ARMS" "$BN_SHA" "$HF_SHA" "$RUNNER_SHA" "$STAGEW" "$GPU_CONSTRAINT" \
          "$NJOBS" "$WALL" "$MST" "$A0" "$DSET" "$EPOCHS" "$BATCH" "$CLIP_C" "$PROBE" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"
import sys, re, importlib.util
spm = importlib.util.spec_from_file_location("caw2_design", sys.argv[1]); Dm = importlib.util.module_from_spec(spm); spm.loader.exec_module(Dm)
(net, ntens, totpar, seeds, arms, bn, hf, runner, stage, gpu, njobs, wall, mst, a0, dset, ep, bat, clip, probe) = sys.argv[2:21]
bad = []
if tuple(arms.split()) != Dm.ARMS: bad.append("ARMS")
if net != Dm.NET or int(ntens) != Dm.NTENS or int(totpar) != Dm.TOTPAR: bad.append("NET/NTENS/TOTPAR")
if tuple(int(x) for x in seeds.split()) != Dm.SEEDS: bad.append("SEEDS")
if (bn, hf, runner) != (Dm.BN_SHA, Dm.HF_SHA, Dm.RUNNER_SHA): bad.append("SHAS")
if gpu != Dm.GPU_CONSTRAINT or int(njobs) != Dm.NJOBS or wall != Dm.WALL: bad.append("GPU/NJOBS/WALL")
if (mst, a0, dset, int(ep), int(bat), clip, int(probe)) != (Dm.MST, Dm.A0, Dm.DSET, Dm.EPOCHS, Dm.BATCH, Dm.CLIP, Dm.PROBE): bad.append("CELL")
st = open(stage).read()
blk = st[st.index("  cwd1) "):st.index(";;", st.index("  cwd1) "))]
for name, v in (("BN_SHA", bn), ("HF_POST_SHA", hf), ("RUNNER_SHA", runner)):
    m = re.search(r"%s=([0-9a-f]{64})" % name, blk)
    if not m or m.group(1) != v: bad.append("stage " + name)
if bad:
    print("!!! guard 4c': disagreement on %s" % sorted(bad)); raise SystemExit(1)
print("guard 4c': ARMS, SEEDS, NET, NTENS, TOTPAR, the cell, NJOBS, WALL, the GPU constraint and the tree / runner shas")
print("           agree in launcher / design / cwd1's stage block")
for a in Dm.ARMS:
