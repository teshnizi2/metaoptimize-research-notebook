#!/usr/bin/env python3
r"""CYCLE 54 -- score the three cycle-53 pre-registrations.

  `br6-*`  (cluster_b, 12) -- THE BUDGET REPLICATION, box-free BY CONSTRUCTION at
      BETA_CLIP=-30:2.0, 40 epochs, R18/ms=1e-3, rungs w/node/lay, seeds 0-3.
      Registered C0, C0.3, C0.4, C1, C2, C2.5 in bin/c53_budget_replication.sh.
      SCORING ORDER: C0, C0.3 (PER SEED), C0.4, then **C2 BEFORE C1**.
      C2 is the INDEPENDENT prediction (the lay-stationary / node-drifting
      asymmetry, informed by nothing post-hoc).  C1 is LABELLED NOT INDEPENDENT
      -- it was registered after seeing bl5's post-hoc box-free window, so a
      confirmation is a REPLICATION and must be written as one.  Reading the
      independent prediction first is the cycle-51 W2-before-W1 lesson applied.

  `ns6-*`  (cluster_a, 3)  -- N1's third seed at ms=2e-4.  Named `ns5-*-m2p4-s2` so
      it POOLS with ns5; N1 is re-read by `c53_score.py`, not here.  This file
      only scores M0.3, the per-seed box-free gate, which c53_score.py pools.

  `bo6-*`  (cluster_a, 6)  -- AdamW BASE.  Registered V0, V0.3, V0.4, V1, V2.
      SCORING ORDER: V0, V0.3 (PER SEED), V0.4, then **V2 BEFORE V1**.
      V2 decides whether CORRECTIONS 70 may cite Adam-mini / Adalayer / SGG at
      all; V1 has NO registered direction and so cannot be over-read.

STANDING RULE (8), new this cycle and the reason every gate below is per seed:
a gate on an event that either happens or does not must be scored PER SEED, never
on the pooled fraction.  `bl5`'s weightwise arm read "8.03% of records at HI",
which was one seed at 24.07% and two at 0.00% -- the pooled row concealed both
the phenomenon and its size (CORRECTIONS 72).

Run `--selftest` before trusting any number this prints.
"""
import glob
import math
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from c52_boxfree import mean, neff_table, occupancy, plateau  # noqa: E402
from c53_score import argmin_rung  # noqa: E402
from c53_budget_window import neff_over  # noqa: E402

P = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "..")
RUNGS = ("lay", "node", "w")
NBETA = {"w": 11_173_962, "node": 14_420, "lay": 62}
BAR_NEFF = 0.10
BAR_BOXFREE = 0.05

# controls, all re-derived in-run where possible (rule 1); quoted for drift checks
CTRL20 = {"lay": 0.7262, "node": 0.4056, "w": 0.5064}   # cl5 cU, 20 ep, steady half
BL5_40 = {"lay": 0.6912, "node": 0.2774, "w": 0.1483}   # bl5, 40 ep, HI=0.0
BL5_PLATEAU = {"w": 86.367, "node": 89.971, "lay": 88.828}
# The BOX-FREE 4-family values, CORRECTIONS 68.  The range is taken from the RAW
# numbers, not from the "15.7%-54.7%" the paper text rounds them to: R34's raw
# value is 0.5474, which is OUTSIDE the rounded 0.547 endpoint.  A bar built from
# a rounded quotation would pass 0.5472 and fail 0.5474 for no reason a referee
# could defend.  (Caught by this file's own selftest.)
BOXFREE_4 = {"R10": 0.1571, "C100": 0.3124, "R18": 0.5045, "R34": 0.5474}
RANGE_LO, RANGE_HI = min(BOXFREE_4.values()), max(BOXFREE_4.values())


def fmt(v, nd=4):
    return "--" if v is None or (isinstance(v, float) and not math.isfinite(v)) else f"{v:.{nd}f}"


def argmin_blockers(rows, stable):
    """Which rungs make an argmin over `rows` UNINTERPRETABLE?  (cycle 54)

    An argmin is a statistic OVER the rungs, so it inherits EVERY rung's
    interpretability -- not just that of the rung it happens to select.  This
    file's first version gated V1 on the weightwise rung alone and left V2
    ungated, then printed "V2 REFUTED ... the Adam-mini sentence must be
    DELETED" from an argmin whose two smallest rungs had both failed the
    box-free gate on 2 of 2 seeds, while the only box-free rung was the
    maximum.  Returns ([blocking rung names], reason) -- empty list == usable.
    """
    bad = sorted(g for g, rs in rows.items() if not all(o["free"] for o in rs))
    if bad:
        return bad, f"the box-free gate failed on {', '.join(bad)}"
    if not stable:
        return [], "the stability gate failed"
    return [], ""


def per_seed_boxfree(pat, lo, hi, rec_per_epoch):
    """STANDING RULE (8): the box-free gate, PER SEED.  Returns a list of dicts."""
    out = []
    for d in sorted(glob.glob(pat)):
        o = occupancy(d, lo, hi)
        o["dir"] = os.path.basename(d)
        o["free"] = max(o["rec_lo"], o["rec_hi"]) < BAR_BOXFREE
        o["ep_hi"] = (o["first_hi"] / rec_per_epoch) if o["first_hi"] is not None else None
        out.append(o)
    return out


def boxfree_table(title, pats, lo, hi, rec_per_epoch, expect_T):
    print(f"\n{title}")
    print(f"  bar: every seed < {100 * BAR_BOXFREE:.0f}% of records at either guard "
          f"(STANDING RULE 8 -- PER SEED, never pooled)")
    print(f"\n  {'arm':26}{'T':>7}{'n_beta':>13}{'%rec LO':>10}{'%rec HI':>10}"
          f"{'%Q4 HI':>9}{'%coordHI':>11}{'1stHI':>8}{'epoch':>8}{'final max':>11}{'free?':>7}")
    allfree, allvalid = True, True
    rows = {}
    for rung, pat in pats.items():
        rs = per_seed_boxfree(pat, lo, hi, rec_per_epoch)
        rows[rung] = rs
        for o in rs:
            valid = (o["T"] == expect_T) and (o["n_beta"] == NBETA[rung])
            allvalid &= valid
            allfree &= o["free"]
            s_coord = ("--" if o["coord_hi"] is None else f"{100 * o['coord_hi']:.5f}")
            s_first = ("-" if o["first_hi"] is None else str(o["first_hi"]))
            s_ep = ("-" if o["ep_hi"] is None else f"{o['ep_hi']:.1f}")
            print(f"  {o['dir']:26}{o['T']:>7}{o['n_beta']:>13,}{100 * o['rec_lo']:>10.4f}"
                  f"{100 * o['rec_hi']:>10.4f}{100 * o['q4_hi']:>9.2f}{s_coord:>11}"
                  f"{s_first:>8}{s_ep:>8}{o['max_final']:>11.3f}"
                  f"{('yes' if o['free'] else 'NO'):>7}")
    nb = {r: sum(1 for o in rows[r] if o["free"]) for r in rows}
    print(f"\n  box-free seeds: " + "   ".join(f"{r} {nb[r]}/{len(rows[r])}" for r in rows))
    print(f"  validity (T=={expect_T} and n_beta byte-match): {'PASS' if allvalid else 'FAIL'}")
    print(f"  --> box-free gate {'PASS on every seed' if allfree else 'FAIL -- see the NO rows'}")
    if not allfree:
        print("      Per the registrations: a bound seed's primary test is UNINTERPRETABLE, and")
        print("      its first_hi/epoch IS the result -- a direct measurement of the budget at")
        print("      which the box becomes unavoidable at this ceiling.")
    return rows, allfree, allvalid


def profile(root, fam, window=(0.5, 1.0)):
    """N_eff/m per rung.  Uses the shared, selftested instrument."""
    if window == (0.5, 1.0):
        t = neff_table(root).get(fam, {})
        return {g: (t[g][2] if g in t else None) for g in RUNGS}
    return {g: mean([neff_over(d, window)[0]
                     for d in sorted(glob.glob(os.path.join(root, f"probe_{g}_*")))])
            for g in RUNGS}


def selftest():
    import json
    import tempfile
    checks = []

    def ok(n, c):
        checks.append((n, bool(c)))

    with tempfile.TemporaryDirectory() as td:
        # one seed binds at 3500, two never do -- the bl5 shape, which the POOLED
        # row hid and which STANDING RULE 8 exists to surface.
        for s, binds in enumerate([3500, None, None]):
            d = os.path.join(td, f"probe_w_c2_s{s}")
            os.makedirs(d)
            with open(os.path.join(d, "probe.jsonl"), "w") as f:
                for t in range(4000):
                    hit = binds is not None and t >= binds
                    f.write(json.dumps(dict(
                        step=t * 5, beta_true_min=-25.0,
                        beta_true_max=(2.0 if hit else -1.0),
                        n_at_lo=0, n_at_hi=(3 if hit else 0),
                        n_beta=NBETA["w"])) + "\n")
        rs = per_seed_boxfree(os.path.join(td, "probe_w_c2_s*"), -30.0, 2.0, 100)
        ok("three seeds seen", len(rs) == 3)
        ok("only the binding seed is flagged", [o["free"] for o in rs] == [False, True, True])
        ok("first_hi is exact", rs[0]["first_hi"] == 3500)
        ok("epoch = first_hi / 100", abs(rs[0]["ep_hi"] - 35.0) < 1e-9)
        ok("free seeds have no first_hi", rs[1]["first_hi"] is None)
        # the POOLED fraction would be 12.5%/3 = 4.2% -- UNDER the 5% bar.
        pooled = mean([o["rec_hi"] for o in rs])
        ok("pooled fraction would PASS the bar and hide the bind", pooled < BAR_BOXFREE)
        ok("per-seed scoring FAILS it, which is the point",
           not all(o["free"] for o in rs))
        ok("n_beta byte-match is read per seed", all(o["n_beta"] == NBETA["w"] for o in rs))

    # --- the V2 gate (cycle 54).  These encode the exact bo6 shape that the
    #     first version of this file scored as "V2 REFUTED": lay box-free, node
    #     and w both bound, and the argmin sitting on a bound rung.
    _free = [{"free": True}, {"free": True}]
    _bound = [{"free": False}, {"free": False}]
    ok("V2 gate: the literal bo6 shape is UNINTERPRETABLE",
       argmin_blockers({"lay": _free, "node": _bound, "w": _bound}, True)[0] == ["node", "w"])
    ok("V2 gate: a bound rung that is NOT the argmin still blocks",
       argmin_blockers({"lay": _bound, "node": _free, "w": _free}, True)[0] == ["lay"])
    ok("V2 gate: all rungs box-free and stable is usable",
       argmin_blockers({"lay": _free, "node": _free, "w": _free}, True) == ([], ""))
    ok("V2 gate: instability blocks even when every rung is box-free",
       argmin_blockers({"lay": _free, "node": _free, "w": _free}, False)[1] != "")
    ok("V2 gate: one bound seed of two is enough to block (STANDING RULE 8)",
       argmin_blockers({"lay": _free, "node": _free,
                        "w": [{"free": True}, {"free": False}]}, True)[0] == ["w"])

    ok("argmin imported live", argmin_rung({"lay": .7, "node": .4, "w": .5})[0] == "node")
    ok("argmin undecided inside 0.02",
       argmin_rung({"lay": .7, "node": .40, "w": .41})[1] is False)
    ok("range brackets all 4 box-free families",
       all(RANGE_LO <= v <= RANGE_HI for v in BOXFREE_4.values()))
    ok("range is built from RAW values, not the rounded quotation",
       RANGE_HI > 0.547 and RANGE_LO <= 0.1571)

    n = sum(1 for _, c in checks if c)
    for name, c in checks:
        if not c:
            print(f"  FAIL: {name}")
    print(f"selftest: {n}/{len(checks)} PASS")
    return 0 if n == len(checks) else 1


# ------------------------------------------------------------------------ br6
def score_br6():
    print("=" * 112)
    print("BR6 -- THE BUDGET REPLICATION, BOX-FREE BY CONSTRUCTION.  40 ep, BETA_CLIP=-30:2.0, n=4")
    print("  ORDER: C0, C0.3 (PER SEED), C0.4, then **C2 BEFORE C1** -- the independent")
    print("  prediction is read before the labelled-dependent one.")
    print("=" * 112)
    pats = {g: f"{P}/probes_br6/probe_{g}_c2_s*" for g in RUNGS}
    if not glob.glob(pats["w"]):
        print("\n  no br6 probes on disk yet -- SKIPPED.")
        return
    rows, allfree, allvalid = boxfree_table(
        "(C0 + C0.3) VALIDITY AND THE BOX-FREE GATE", pats, -30.0, 2.0, 100, 4000)

    print("\n(C0.4) THE STABILITY GATE.  alpha may now reach e^2 = 7.39.  X3's 'the operating")
    print("  point is INTERIOR' was measured at 20 epochs and is now known to be budget-scoped.")
    print(f"\n  {'rung':6}{'br6 plateau':>14}{'n':>3}{'bl5 (HI=0.0)':>15}{'delta':>9}"
          f"{'bar':>8}{'verdict':>10}")
    c04 = True
    for g in RUNGS:
        p, n = plateau(f"br6-{g}-c2")
        d = p - BL5_PLATEAU[g]
        good = abs(d) <= 2.0
        c04 &= good
        print(f"  {g:6}{p:>14.3f}{n:>3}{BL5_PLATEAU[g]:>15.3f}{d:>+9.3f}"
              f"{'2.0 pp':>8}{('PASS' if good else 'FAIL'):>10}")
    print(f"  --> C0.4 {'PASS -- the ceiling is not load-bearing at 40 epochs either' if c04 else 'FAIL -- the ceiling IS load-bearing at 40 epochs; C1/C2 UNINTERPRETABLE, and THAT IS THE RESULT (it narrows X3 to 20 epochs)'}")

    pr = profile(f"{P}/probes_br6", "c2")

    print("\n(C2) **THE INDEPENDENT PREDICTION, READ FIRST.**  Registered: the bl5 asymmetry")
    print("  replicates at n=4 under a different ceiling -- lay within +-0.10 of 0.7262,")
    print("  node BELOW 0.31.  Nothing post-hoc informed this.")
    print(f"\n  {'rung':6}{'20 ep':>10}{'bl5 40 ep':>12}{'br6 40 ep':>12}{'delta vs 20':>13}"
          f"{'registered':>22}{'verdict':>10}")
    lay_ok = pr["lay"] is not None and abs(pr["lay"] - CTRL20["lay"]) <= BAR_NEFF
    node_ok = pr["node"] is not None and pr["node"] < 0.31
    for g, reg, good in (("lay", "|d| <= 0.10 of 0.7262", lay_ok),
                         ("node", "< 0.31", node_ok)):
        print(f"  {g:6}{CTRL20[g]:>10.4f}{BL5_40[g]:>12.4f}{fmt(pr[g]):>12}"
              f"{(pr[g] - CTRL20[g] if pr[g] is not None else float('nan')):>+13.4f}"
              f"{reg:>22}{('PASS' if good else 'FAIL'):>10}")
    print(f"  --> C2 {'CONFIRMED -- budget-robustness IS granularity-dependent, replicated at n=4' if (lay_ok and node_ok) else 'REFUTED -- FINDINGS 53.4s asymmetry was a 3-seed accident'}")
    print("      Read the layerwise row as a BOUND: m=62 is marginal against rho_min ~9.5e-04,")
    print("      so 'not shown to move', never 'shown not to move'.")

    print("\n(C1) THE PRIMARY TEST -- **LABELLED NOT INDEPENDENT.**  Registered AFTER seeing")
    print("  bl5's post-hoc box-free window, so a confirmation is a REPLICATION, not a")
    print("  discovery, and must be written as one.")
    print("  Registered: DRIFTING DOWN, N_eff/m < 0.40, expected ~0.26.")
    print("  Refutation: inside [0.41, 0.61] -> the fall was the box and the headline is")
    print("  budget-robust after all.")
    v = pr["w"]
    if v is not None:
        drift = v < 0.40
        refut = abs(v - CTRL20["w"]) <= BAR_NEFF
        print(f"\n  20-ep control  {CTRL20['w']:.4f}")
        print(f"  br6 40 ep      {v:.4f}   delta {v - CTRL20['w']:+.4f}")
        wfree = all(o["free"] for o in rows["w"])
        if not wfree:
            print("  --> C1 UNINTERPRETABLE: the weightwise rung bound even at HI=+2.0.")
        elif not c04:
            print("  --> C1 UNINTERPRETABLE: C0.4 failed, so the run is not a measurement.")
        elif drift:
            print("  --> C1 CONFIRMED (REPLICATION, not a discovery). N_eff/m is NOT budget-stable")
            print("      at weightwise.  CORRECTIONS 73's quoting rule is now on pre-registered")
            print("      evidence at n=4 rather than a post-hoc window at n=2.")
        elif refut:
            print("  --> C1 REFUTED. The bl5 fall WAS the box, the post-hoc window misled, and")
            print("      the headline is budget-robust.  Write that, and withdraw 73's weightwise")
            print("      row (its NODEWISE row is independent and would still stand).")
        else:
            print("  --> C1 lands between the branches (0.40-0.41 or outside both). Report the")
            print("      number and claim neither branch.")

    print("\n(C2.5) THE ORDERING.  Registered: **w < node < lay** replicates.")
    print("  Refutation: node < w < lay returns -> the bl5 flip WAS the box and CORRECTIONS 70")
    print("  is not budget-scoped after all.")
    if all(pr[g] is not None for g in RUNGS):
        order = " < ".join(k for k, _ in sorted(pr.items(), key=lambda kv: kv[1]))
        am, dec = argmin_rung(pr)
        print(f"\n  br6: lay {pr['lay']:.4f}  node {pr['node']:.4f}  w {pr['w']:.4f}   -> {order}")
        print(f"  argmin = {am} ({'decided' if dec else 'UNDECIDED, gap <= 0.02'})")
        if not dec:
            print("  --> C2.5 UNDECIDED -- the two smallest are inside the tolerance. Do not read")
            print("      the nominal argmin (CORRECTIONS 75).")
        elif order == "w < node < lay":
            print("  --> C2.5 CONFIRMED: the nodewise minimum is BUDGET-SPECIFIC, replicated.")
        elif order == "node < w < lay":
            print("  --> C2.5 REFUTED: the bl5 flip was the box; CORRECTIONS 74 must be revisited.")
        else:
            print(f"  --> C2.5: a THIRD ordering ({order}). Neither branch fires; report it as is.")


# ------------------------------------------------------------------------ bo6
def score_bo6():
    print("\n\n" + "=" * 112)
    print("BO6 -- BASE-OPTIMIZER GENERALITY: AdamW base, Lion meta.  20 ep, -30:0.0, n=2")
    print("  ORDER: V0, V0.3 (PER SEED), V0.4, then **V2 BEFORE V1**.  V2 decides whether")
    print("  CORRECTIONS 70 may cite Adam-mini / Adalayer / SGG at all.")
    print("=" * 112)
    pats = {g: f"{P}/probes_bo6/probe_{g}_adw_s*" for g in RUNGS}
    if not glob.glob(pats["w"]):
        print("\n  no bo6 probes on disk yet -- SKIPPED.")
        return
    rows, allfree, allvalid = boxfree_table(
        "(V0 + V0.3) VALIDITY AND THE BOX-FREE GATE", pats, -30.0, 0.0, 100, 2000)
    if not allvalid:
        print("\n  *** V0 FAILS. n_beta is a property of the NETWORK; a mismatch means the")
        print("      network changed, and the batch is VOID rather than interesting.")

    print("\n(V0.4) THE TRAINS-AT-ALL GATE.  ms and alpha0 are carried over from the SGDm")
    print("  ladder and are UNTUNED for a normalised base update; a correlation statistic")
    print("  computed on a diverged run measures nothing.  BAR: plateau > 40.0 per seed.")
    print(f"\n  {'rung':6}{'plateau':>10}{'n':>3}{'bar':>9}{'verdict':>10}")
    v04 = True
    for g in RUNGS:
        p, n = plateau(f"bo6-{g}-adw")
        good = p > 40.0
        v04 &= good
        print(f"  {g:6}{p:>10.3f}{n:>3}{'> 40.0':>9}{('PASS' if good else 'FAIL'):>10}")
    print(f"  --> V0.4 {'PASS' if v04 else 'FAIL -- the arm is VOID, not refuted; the ladder needs its own ms, which is the NEXT batch and not a result'}")
    print("  (NOT an optimizer comparison: no plateau here may be set against an SGDm plateau.)")

    pr = profile(f"{P}/probes_bo6", "adw")

    print("\n(V2) **THE SHAPE -- READ FIRST.**  Registered: argmin stays `node` under AdamW.")
    print("  SGDm control (cl5 cU, n=3): lay 0.7258 / node 0.4016 / w 0.5045, argmin node,")
    print("  gap 0.103 -- decided.")
    if all(pr[g] is not None for g in RUNGS):
        am, dec = argmin_rung(pr)
        order = " < ".join(k for k, _ in sorted(pr.items(), key=lambda kv: kv[1]))
        print(f"\n  AdamW: lay {pr['lay']:.4f}  node {pr['node']:.4f}  w {pr['w']:.4f}   -> {order}")
        print(f"  argmin = {am} ({'decided' if dec else 'UNDECIDED, gap <= 0.02'})")
        # ---------------------------------------------------------------------
        # THE GATE V2 WAS MISSING (cycle 54).  An argmin is a statistic OVER the
        # rungs, so it inherits EVERY rung's interpretability, not just its own.
        # The first version of this file gated V1 on rows["w"] (below) and left
        # V2 ungated -- and then printed "V2 REFUTED ... the Adam-mini sentence
        # must be DELETED" from an argmin whose two smallest rungs had BOTH
        # failed V0.3 on 2 of 2 seeds, while `lay`, the only box-free rung, was
        # the maximum.  That is the FOURTH time this campaign a summary statistic
        # has been read past its own uncertainty flag (CORRECTIONS 75: `s` in 51,
        # the pooled clip fraction in 62, the argmin in 74, this).
        # bo6's own registration fixes the consequence: "that arm's V1 is
        # UNINTERPRETABLE, and the batch is re-run at a raised ceiling rather
        # than reported", and its RESOLUTION note states the general principle --
        # an untrustworthy rung makes the argmin UNDECIDABLE "rather than
        # defaulting it to one of the other two.  Say which."
        bad, why = argmin_blockers(rows, v04)
        if why:
            print(f"  --> V2 UNINTERPRETABLE -- {why}.")
            print("      An argmin inherits the interpretability of EVERY rung it ranks, and")
            print(f"      {'/'.join(bad) if bad else 'this batch'} feeds it.  The nominal ordering above is NOT a verdict.")
            print("      Registered consequence (bo6 V0.3): RE-RUN AT A RAISED CEILING.")
            print("      The Adam-mini sentence is SUSPENDED: it may not be written, and it")
            print("      may not be deleted either, until the re-run scores.")
        elif not dec:
            print("  --> V2 UNDECIDED. Buy a third seed, as ns5/ns6 did. Do NOT read the")
            print("      nominal argmin (CORRECTIONS 75).")
        elif am == "node":
            print("  --> V2 CONFIRMED. The nodewise minimum transfers to a second base optimizer.")
            print("      CORRECTIONS 70's Adam-mini / Adalayer / SGG sentence becomes defensible")
            print("      in the family those methods actually live in.")
        else:
            print(f"  --> V2 REFUTED (argmin = {am}). The nodewise minimum is an SGDm phenomenon.")
            print("      CORRECTIONS 70 gains a BASE-OPTIMIZER scope on top of its ms and budget")
            print("      scopes, and **the Adam-mini sentence must be DELETED, not hedged.**")

    print("\n(V1) DOES THE HEADLINE RANGE SURVIVE?  N_eff/m weightwise vs the published")
    print(f"  BOX-FREE 4-family range [{RANGE_LO:.4f}, {RANGE_HI:.4f}] (CORRECTIONS 68, RAW")
    print("  endpoints -- the text rounds these to 15.7%-54.7% and 0.5474 falls outside 0.547).")
    print("  NO DIRECTION WAS REGISTERED, deliberately -- there is no mechanism to predict")
    print("  one, and inventing one to collect a confirmation is the uc5 failure mode.")
    v = pr["w"]
    if v is not None:
        inside = RANGE_LO <= v <= RANGE_HI
        print(f"\n  AdamW-base R18 weightwise: {v:.4f}   range [{RANGE_LO:.4f}, {RANGE_HI:.4f}]")
        if not all(o["free"] for o in rows["w"]) or not v04:
            print("  --> V1 UNINTERPRETABLE (V0.3 or V0.4 failed on this rung).")
        elif inside:
            print("  --> V1 PASSES. The retained fraction is a property of the network and task,")
            print("      not of SGDm.  CORRECTIONS 59 needs no base-optimizer caveat --")
            print("      **but it still needs its BUDGET one (73).**")
        else:
            print("  --> V1 REFUTED. The 16%-55% range is SGDm-specific and every quotation of")
            print("      it needs a base-optimizer column.  That is a real narrowing.")


def main():
    score_br6()
    score_bo6()
    print("\n\n  N1 (the ns6 third seed) is scored by `python3 analysis/c53_score.py`, which")
    print("  re-reads the ms=2e-4 rung at n=3 automatically because ns6 writes into probes_ns5.")
    return 0


if __name__ == "__main__":
    sys.exit(selftest() if "--selftest" in sys.argv else main())
