set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
# 20-epoch DIAGNOSTIC runs: measure how fast the single shared beta moves as a
# function of N (the number of pooled meta-gradient estimates). Every arm below
# ends each step with exactly ONE shared step size; only N differs.
# Short + right-sized --time so they backfill (gotcha 12).
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 20 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ sbatch --parsable --job-name=$1 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 --time=00:45:00 \
  --export="ALL,AUGMENT=1,$C,$3,PROBE=100,PROBE_DIR=$R/bdrift/$1" \
  jobs/run_cifar.sh $B --stepsize-groups $2 --seed 0 \
  --save-directory $R/bdrift --run-name $1 >/dev/null; }
sub p2-scal      scalar          "HIER="
sub p2-blk6-L1p0 resnet18_blocks "HIER=shrink,LAM=1.0"
sub p2-lay-L1p0  layerwise       "HIER=shrink,LAM=1.0"
sub p2-node-L1p0 nodewise        "HIER=shrink,LAM=1.0"
sub p2-w-L1p0    weightwise      "HIER=shrink,LAM=1.0"
sub p2-lay-plain layerwise       "HIER="
sleep 3
echo "BDRIFT_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -c '^p2-')"
echo "TOTAL_QUEUE=$(squeue -h -u account_b | wc -l)"
