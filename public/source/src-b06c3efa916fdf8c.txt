def mean(v):
    return sum(v) / float(len(v))


def sd(v):
    if len(v) < 2:
        return 0.0
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def median(v):
    s = sorted(v)
    n = len(s)
    return s[n // 2] if n % 2 else 0.5 * (s[n // 2 - 1] + s[n // 2])


def file_sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()

def rung_state(d, se, diverged, healthy, boxfree):
    """cgw1's rung_state, VERBATIM."""
    if diverged:
        return "DIVERGED"
    if not healthy:
        return "UNHEALTHY"
    if not boxfree:
        return "BOXBOUND"
    if d <= -NULLBAR and d < -RES * se:
        return "REVERSES"
    if d <= NULLBAR and d + RES * se < POOL_EXCLUDE:
        return "VANISHES"
    if d >= SURV and d > RES * se:
        return "SURVIVES"
    return "UNDECIDED"


def _grp(s):
    return "B" if s in BAD else ("S" if s == "SURVIVES" else ("X" if s in ABSENT else "U"))


def primary_of(s5, s4):
    g5, g4 = _grp(s5), _grp(s4)
    if g5 == "B" and g4 == "B":
        return "AI-UNREADABLE"
    if g5 == "B" or g4 == "B":
        o = g4 if g5 == "B" else g5
        return {"S": "AI-ONE-RUNG-SURVIVES", "X": "AI-ONE-RUNG-ABSENT", "U": "AI-ONE-RUNG-UNDECIDED"}[o]
    pair = set((g5, g4))
    if pair == {"S"}:
        return "AI-SURVIVES"
    if pair == {"X"}:
        return "AI-ARTEFACT-REVERSES" if "REVERSES" in (s5, s4) else "AI-ARTEFACT-VANISHES"
    if pair == {"S", "X"}:
        return "AI-LAMBDA-DEPENDENT"
    if pair == {"S", "U"}:
        return "AI-SURVIVES-ONE-LAMBDA"
    if pair == {"X", "U"}:
        return "AI-ABSENT-ONE-LAMBDA"
    return "AI-UNDECIDED"


def scalar_of(k_level, k_div, cmp_ok, t_ch, t_nd, se):
    """cgw1's scalar_of, VERBATIM."""
    if k_div:
        return "SCALAR-DIVERGED"
    if k_level < HEALTH_MIN:
        return "SCALAR-COLLAPSED"
    if not cmp_ok:
        return "SCALAR-UNREADABLE"
    tmin = min(t_ch, t_nd)
    if t_ch >= SURV and t_nd >= SURV and t_ch > RES * se and t_nd > RES * se:
        return "SCALAR-BEATS-BEST"
    if tmin - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tmin <= -SURV and tmin < -RES * se:
        return "SCALAR-BELOW-BEST"
    return "SCALAR-UNRESOLVED"


def sc_of(t5, t4):
    if t5 in SC_GATED and t4 in SC_GATED:
        return "SC-UNREADABLE"
    if t5 == t4 == "SCALAR-BEATS-BEST":
        return "SC-BEATS-BOTH"
    if t5 == t4 == "SCALAR-BELOW-BEST":
        return "SC-BELOW-BOTH"
    ok = ("SCALAR-BEATS-BEST", "SCALAR-TIES-BEST")
    if t5 in ok and t4 in ok:
        return "SC-TIES-OR-BEATS-BOTH"
    if (t5 in ok and t4 == "SCALAR-BELOW-BEST") or (t4 in ok and t5 == "SCALAR-BELOW-BEST"):
        return "SC-LAMBDA-DEPENDENT"
    return "SC-PARTIAL"


def compose_sigma(sigma_in):
    return (SIGMA_PRIOR, "SIGMA_PRIOR_frozen") if SIGMA_PRIOR >= sigma_in else (sigma_in, "SIGMA_INBATCH")


def decide(A, sigma_used):
    """A[arm] = dict(level, train, range, n, boxfree).  Returns (primary, sc, states, scal, keys)."""
    states, scal, keys = {}, {}, {}
    for r in D.RUNG_IDS:
        ch, nd = A["ch" + r], A["nd" + r]
        d = ch["level"] - nd["level"]
        se = sigma_used * math.sqrt(1.0 / ch["n"] + 1.0 / nd["n"])
        states[r] = rung_state(d, se, ch["range"] > DIVERGED or nd["range"] > DIVERGED,
                               ch["level"] >= HEALTH_MIN and nd["level"] >= HEALTH_MIN,
                               ch["boxfree"] and nd["boxfree"])
        keys["D_" + r] = (d, se)
        keys["DTRAIN_" + r] = ch["train"] - nd["train"]
        k, L = A["k01" + r], A["kL" + r]
        for g in ("ch", "nd"):
            x = A[g + r]
            keys["T%s_%s" % (g, r)] = (k["level"] - x["level"], sigma_used * math.sqrt(1.0 / k["n"] + 1.0 / x["n"]))
        keys["G_" + r] = (L["level"] - k["level"], sigma_used * math.sqrt(1.0 / k["n"] + 1.0 / L["n"]))
        cmp_ok = (ch["level"] >= HEALTH_MIN and nd["level"] >= HEALTH_MIN and ch["range"] <= DIVERGED
                  and nd["range"] <= DIVERGED)
        scal[r] = scalar_of(k["level"], k["range"] > DIVERGED, cmp_ok, keys["Tch_" + r][0], keys["Tnd_" + r][0],
                            keys["Tch_" + r][1])
    primary = primary_of(states["I5"], states["I4"])
    sc = sc_of(scal["I5"], scal["I4"])
    d5, se5 = keys["D_I5"]
    d4, se4 = keys["D_I4"]
    keys["DD"] = (d4 - d5, math.sqrt(se5 ** 2 + se4 ** 2))
    return primary, sc, states, scal, keys

def parse_args_line(line):
    """-> (pairs in order, repeated flags).  argparse semantics for this harness: every value is ONE token."""
    toks = shlex.split(line[len("ARGS:"):])
    pairs, seen, rep, i = [], set(), [], 0
    while i < len(toks):
        t = toks[i]
        if not t.startswith("--"):
            return None, ["stray token %r" % t]
        if i + 1 >= len(toks) or toks[i + 1].startswith("--"):
            return None, ["flag %s has no value" % t]
        f = t[2:]
        if f in seen:
            rep.append(f)
        seen.add(f)
        pairs.append((f, toks[i + 1]))
        i += 2
    return pairs, rep


def read_out(path):
    L = open(path, errors="replace").read().splitlines()
    info = dict(path=path, args=None, env=None, gpu=None, done=False, eps={}, eps_dup=[], kinds_on=[], node=None,
                pt=False, dr=[])
    for ln in L:
        if ln.startswith("ARGS:") and info["args"] is None:
            info["args"] = ln
        elif ln.startswith("ENV:") and info["env"] is None:
            info["env"] = ln
        elif ln.startswith("NODE=") and info["node"] is None:
            info["node"] = ln.split("|")[0].strip()
        elif ln.startswith("NVIDIA") and info["gpu"] is None:
            info["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            info["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            info["pt"] = True
        elif ln.startswith("DECAY_ROUTE"):
            info["dr"].append(ln.rstrip())
        else:
            m = EPTR_RE.search(ln)
            if m:
                e = int(m.group(1))
                if e in info["eps"]:
                    info["eps_dup"].append(e)      # cgw1's latent RULE 16 defect F1 (296.6), closed here: a gate
                info["eps"][e] = (float(m.group(2)), float(m.group(3)))
            k = KIND_ON_RE.match(ln)
            if k:
                info["kinds_on"].append(ln.strip())
    info["complete"] = info["done"] and all(e in info["eps"] for e in range(D.EPOCHS))
    return info


def find_outs(runsdir):
    out = []
    for base in (runsdir, os.path.join(runsdir, PREFIX)):
        out += glob.glob(os.path.join(base, "%s-*.out" % PREFIX))
    return sorted(set(os.path.realpath(p) for p in out))


def probe_path(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, D.probe_dir_name(arm, seed), "probe.jsonl")
        if os.path.exists(p):
            return p
    return None


PROBE_KEYS = ("step", "n_beta", "beta_true_min", "beta_true_max", "n_at_lo", "n_at_hi", "beta", "dr_mode", "dr_lam",
              "dr_n", "dr_shrink_meas", "dr_shrink_applied", "dr_trace_decay")


def read_probe(path):
    recs = []
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            r = json.loads(ln)
        except ValueError:
            recs.append(None)
            continue
        recs.append({k: r.get(k) for k in PROBE_KEYS} if isinstance(r, dict) else None)
    return recs


def probe_struct(recs, m_want):
    """cgw1's G-STRUCT (10,000 records at steps 0,5,..; n_beta == the arm's m) + box occupancy.  -> (ok, why, occ)."""
    if recs is None:
        return False, "no probe.jsonl", None
    if len(recs) != D.N_RECORDS:
        return False, "%d records, want %d" % (len(recs), D.N_RECORDS), None
    if any(r is None for r in recs):
        return False, "an unreadable record", None
    steps = [r["step"] for r in recs]
    if steps != list(range(0, D.N_RECORDS * D.PROBE, D.PROBE)):
        return False, "record steps are not 0,5,...,%d" % ((D.N_RECORDS - 1) * D.PROBE), None
    nb = set(r["n_beta"] for r in recs)
    if nb != {m_want}:
        return False, "n_beta %s != the arm's registered m %d (the grain is not what ARGS says)" % (sorted(nb)[:3], m_want), None
    mins = [r["beta_true_min"] for r in recs]
    maxs = [r["beta_true_max"] for r in recs]
    if not (max(maxs) - min(mins) > 1e-9):
        return False, "beta never moved", None
    T = len(recs)
    rec_lo = sum(1 for x in mins if x <= D.LO + BOX_EPS) / float(T)
    rec_hi = sum(1 for x in maxs if x >= D.HI - BOX_EPS) / float(T)
    win = [r for r in recs if r["step"] >= WIN_STEP]
    tot = float(sum(r["n_beta"] for r in win))
    have = all(r.get("n_at_lo") is not None and r.get("n_at_hi") is not None for r in win)
    a_win = []
    for r in win:
        b = r.get("beta")
        if isinstance(b, list) and b:
            a_win.append(median([math.exp(x) for x in b]))
    occ = dict(rec_lo=rec_lo, rec_hi=rec_hi,
               coord_lo_win=(sum(r["n_at_lo"] for r in win) / tot) if have else None,
               coord_hi_win=(sum(r["n_at_hi"] for r in win) / tot) if have else None,
               boxfree=(rec_lo < BOXFREE_MAX and rec_hi < BOXFREE_MAX),
               a_plateau=median(a_win) if a_win else None)
    return True, "", occ


def probe_decay(recs, lam):
    """G-SHRINK: every record is an alpha_indep record at THIS rung's LAMBDA, and the decay MEASURABLY bit.
    -> (ok, why, stats)."""
    devs = []
    for r in recs:
        if r.get("dr_mode") != "alpha_indep":
            return False, "step %s: dr_mode %r, want 'alpha_indep' (no DECAY_ROUTE record: the unpatched tree?)" % (
                r.get("step"), r.get("dr_mode")), None
        if r.get("dr_lam") != lam:
            return False, "step %s: dr_lam %r != this rung's LAMBDA %r" % (r.get("step"), r.get("dr_lam"), lam), None
        if r.get("dr_n") != r["step"] + 2:
            return False, "step %s: dr_n %r != step + 2" % (r["step"], r.get("dr_n")), None
        sm, sa, tdc = r.get("dr_shrink_meas"), r.get("dr_shrink_applied"), r.get("dr_trace_decay")
        if sm is None or sa is None or tdc is None:
            return False, "step %s: a null dr_shrink_* / dr_trace_decay" % r["step"], None
        if abs(sa / lam - 1.0) > APPLIED_TOL or abs(tdc - (1.0 - lam)) > TRACE_TOL:
            return False, "step %s: dr_shrink_applied %r / dr_trace_decay %r are not LAMBDA / 1 - LAMBDA" % (r["step"], sa, tdc), None
        dv = sm / lam - 1.0
        if abs(dv) > SHRINK_TOL:
            return False, "step %s: MEASURED shrink %r is %.2e relative from LAMBDA %r (> %g): the decay did not bite as registered" % (
                r["step"], sm, dv, lam, SHRINK_TOL), None
        devs.append(dv)
    return True, "", dict(max_rel_dev=max(abs(x) for x in devs), median_meas=lam * (1.0 + median(devs)))


def read_corpus_disclosure(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    sys.path.insert(0, HERE)
    try:
        import corpus_exclusions as ce
    finally:
        sys.path.pop(0)
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    return len(ce.filter_rows(rows))


def resolve_prov(runsdir):
    for p in (os.path.join(runsdir, PREFIX, "PROVENANCE.txt"), os.path.join(runsdir, "PROVENANCE.txt")):
        if os.path.exists(p):
            return p
    return None

def score(runsdir, csvpath=DEFAULT_CSV, out=print):
    P = out
    P("=" * 110)
    P("cai1 -- THE COUNT-MATCHED AUDIT UNDER ALPHA-INDEPENDENT DECAY  (CORRECTIONS 306; ICML-PLAN 1.6, DECOUPLED-NOT-TESTED)")
    P("scorer sha256 %s   design sha256 %s" % (file_sha(os.path.abspath(__file__)), file_sha(os.path.join(HERE, "cai1_design.py"))))
    P("runsdir %s" % runsdir)
    P("=" * 110)
    harness = []

    def hfail(msg):
        harness.append(msg)
        P("  !!! HARNESS %s" % msg)

    found = {}
    for p in find_outs(runsdir):
        m = OUT_RE.match(os.path.basename(p))
        if not m:
            hfail("unparseable file name %s" % os.path.basename(p))
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        if arm not in D.ARMS or seed not in D.SEEDS[arm]:
            hfail("a %s .out that is not a registered run: %s" % (PREFIX, os.path.basename(p)))
            continue
        found.setdefault((arm, seed), []).append((jid, p))
    runs, missing, ignored = {}, [], []
    for (arm, seed) in D.RUNS:
        cands = [(jid, p, read_out(p)) for jid, p in sorted(found.get((arm, seed), []))]
        comp = [c for c in cands if c[2]["complete"]]
        ignored += [os.path.basename(c[1]) for c in cands if not c[2]["complete"]]
        if not comp:
            missing.append(D.run_name(arm, seed))
        elif len(comp) > 1:
            hfail("%s has %d COMPLETE .out files (%s) -- ambiguous" % (D.run_name(arm, seed), len(comp),
                  ", ".join(os.path.basename(c[1]) for c in comp)))
        else:
            runs[(arm, seed)] = comp[0]
    P("COMPLETION  %d / %d registered runs complete (100 epoch lines + RUN_DONE)" % (len(runs), D.NJOBS))
    if ignored:
        P("  ignored (incomplete, a complete twin or none): %s" % ", ".join(sorted(ignored)))
    if missing:
        P("  MISSING / INCOMPLETE: %s" % ", ".join(missing))

    prov = resolve_prov(runsdir)
    if prov is None:
        hfail("G-PROV: no PROVENANCE.txt under %s/%s" % (runsdir, PREFIX))
    else:
        kv = dict(ln.split(" ", 1) for ln in open(prov).read().splitlines() if " " in ln)
        want = {"SCORER_SHA256": file_sha(os.path.abspath(__file__)),
                "DESIGN_SHA256": file_sha(os.path.join(HERE, "cai1_design.py")),
                "HF_SHA256": D.HF_POST_SHA}
        for k, v in sorted(want.items()):
            if kv.get(k, "").strip() != v:
                hfail("G-PROV: %s in %s is %r, the registered / running file is %s" % (k, prov, kv.get(k), v[:16]))
        P("G-PROV  %s  MODE %s  SCORER %s  DESIGN %s  HF %s" % (prov, kv.get("MODE", "?"), kv.get("SCORER_SHA256", "?")[:16],
          kv.get("DESIGN_SHA256", "?")[:16], kv.get("HF_SHA256", "?")[:16]))

    per = {}
    gpus = {}
    save_dirs = set()
    for (arm, seed), (jid, path, info) in sorted(runs.items()):
        rn = D.run_name(arm, seed)
        r = D.RUNG_OF[arm]
        if info["eps_dup"]:
            hfail("G-EPOCH %s: epoch line(s) %s printed more than once" % (rn, sorted(set(info["eps_dup"]))[:5]))
            continue
        if info["args"] is None:
            hfail("G-ARGS %s: no ARGS line" % rn)
            continue
        pairs, rep = parse_args_line(info["args"])
        if pairs is None or rep:
            hfail("G-ARGS %s: %s" % (rn, rep))
            continue
        sd_ = dict(pairs).get("save-directory", "")
        save_dirs.add(sd_)
        want = D.expected_args(arm, seed, sd_)
        if pairs != want:
            diff = [(f, v, dict(want).get(f)) for f, v in pairs if dict(want).get(f) != v]
            miss = [f for f, _v in want if f not in dict(pairs)]
            hfail("G-ARGS %s: %d flags differ from the registered line %s missing %s" % (rn, len(diff), diff[:4], miss[:4]))
            continue
        pdir = os.path.join(sd_, D.probe_dir_name(arm, seed))
        if info["env"] != D.ENV_TEMPLATE.replace("<P>", pdir):
            hfail("G-ENV %s: ENV line is not the registered one (%r)" % (rn, (info["env"] or "")[:160]))
            continue
        if info["dr"] != [D.witness_on(r)]:
            hfail("G-WITNESS %s: DECAY_ROUTE lines %r, want exactly ONE: %r" % (rn, [x[:110] for x in info["dr"]][:2], D.witness_on(r)))
            continue
        other = [k for k in info["kinds_on"] if not k.startswith("DECAY_ROUTE")]
        if other or info["pt"]:
            hfail("G-KIND %s: prints another patch witness %s" % (rn, (other or ["PROBE_TENSOR"])[:2]))
            continue
        pp = probe_path(runsdir, arm, seed)
        recs = read_probe(pp) if pp else None
        ok, why, occ = probe_struct(recs, D.M_OF_ARM[arm])
        if not ok:
            hfail("G-STRUCT %s: %s" % (rn, why))
            continue
        ok, why, dst = probe_decay(recs, D.LAM_OF_ARM[arm])
        if not ok:
            hfail("G-SHRINK %s: %s" % (rn, why))
            continue
        eps = info["eps"]
        per[(arm, seed)] = dict(level=mean([eps[e][1] for e in range(PL_LO, PL_HI + 1)]),
                                train=mean([eps[e][0] for e in range(PL_LO, PL_HI + 1)]), occ=occ, dst=dst, jid=jid)
        gpus[info["gpu"]] = gpus.get(info["gpu"], 0) + 1
    if len(save_dirs) > 1:
        hfail("G-ARGS: the runs name %d different save directories %s" % (len(save_dirs), sorted(save_dirs)))
    P("G-ARGS / G-ENV / G-WITNESS / G-KIND / G-STRUCT / G-SHRINK  %d of %d complete runs pass (20 flags each, wd 0, the "
      "registered ENV line, ONE own-rung DECAY_ROUTE witness, no other patch witness, 10,000 probe records with n_beta == "
      "m, every record's measured shrink within %g of its LAMBDA)" % (len(per), len(runs), SHRINK_TOL))
    P("G-HW (DISCLOSURE ONLY, never a gate): %s" % ", ".join("%s x%d" % (k, v) for k, v in sorted(gpus.items(), key=str)))
    n_corp = read_corpus_disclosure(csvpath)
    P("corpus: %s  (DISCLOSURE ONLY -- enters no bar, sigma, level, state, contrast or stamp)"
      % ("not readable" if n_corp is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (n_corp, PREFIX)))
    if harness:
        P("\nFINAL HARNESS-FAIL | %d harness failure(s); nothing is read" % len(harness))
        return 3
    if missing:
        P("\nFINAL INCOMPLETE | %d of %d runs missing or incomplete; nothing is read" % (len(missing), D.NJOBS))
        return 2

    A = {}
    ss, df = 0.0, 0
    P("\nARMS  plateau5 = mean TEST over epochs %d-%d from each run's OWN .out; box occupancy, measured shrink and learned"
      " step size from its OWN probe.jsonl" % (PL_LO, PL_HI))
    P("  %-6s %-4s %-7s %-9s %2s %9s %7s %7s %9s  %-5s %7s %7s %12s %10s %12s %9s" % (
        "arm", "rung", "LAMBDA", "grain", "n", "plateau5", "sd", "range", "train5", "box", "rec_lo", "rec_hi",
        "shrink_meas", "maxreldev", "a_plateau", "LAM/a"))
    for a in D.ARMS:
        xs = [per[(a, s)]["level"] for s in D.SEEDS[a]]
        tr = [per[(a, s)]["train"] for s in D.SEEDS[a]]
        occs = [per[(a, s)]["occ"] for s in D.SEEDS[a]]
        dsts = [per[(a, s)]["dst"] for s in D.SEEDS[a]]
        bf = all(o["boxfree"] for o in occs)
        m_ = mean(xs)
        ss += sum((x - m_) ** 2 for x in xs)
        df += len(xs) - 1
        A[a] = dict(level=m_, train=mean(tr), range=max(xs) - min(xs), n=len(xs), boxfree=bf, sd=sd(xs), seeds=xs)
        ap = [o["a_plateau"] for o in occs if o["a_plateau"] is not None]
        apm = median(ap) if ap else None
        P("  %-6s %-4s %-7s %-9s %2d %9.4f %7.3f %7.3f %9.4f  %-5s %7.4f %7.4f %12.6e %10.2e %12s %9s" % (
            a, D.RUNG_OF[a], D.LAM_TEXT[D.RUNG_OF[a]], D.SPEC[a], len(xs), m_, A[a]["sd"], A[a]["range"], A[a]["train"],
            "free" if bf else "BOUND", max(o["rec_lo"] for o in occs), max(o["rec_hi"] for o in occs),
            median([x["median_meas"] for x in dsts]), max(x["max_rel_dev"] for x in dsts),
            "na" if apm is None else "%.6e" % apm, "na" if apm is None else "%.3g" % (D.LAM_OF_ARM[a] / apm)))
    P("  (shrink_meas / a_plateau / LAM/a are DESCRIPTIVE: a_plateau = median over the epoch-95+ records of the median over"
      " the record's stored step sizes of exp(beta); for chunk777 / nodewise the probe stores 62 per-tensor MEANS of log"
      " step size, not the per-group values -- APPROXIMATE, as 296.4)")
    sig_in = math.sqrt(ss / df) if df else 0.0
    sig, which = compose_sigma(sig_in)
    P("\nSIGMA  O2: max(frozen floor, in-batch) -- nothing from the corpus")
    P("  SIGMA_PRIOR_frozen   %.6f  (df %d, filtered, literal)" % (SIGMA_PRIOR, SIGMA_PRIOR_DF))
    P("  SIGMA_INBATCH        %.6f  (df %d, pooled within the 8 arms)" % (sig_in, df))
    P("  SIGMA_USED           %.6f  (= %s)" % (sig, which))

    primary, sc, states, scal, K = decide(A, sig)
    P("\nCONTRASTS (every one WITHIN a LAMBDA rung, IN BATCH; +/-2 SE is a HALF-WIDTH, not a bound on any effect)")
    for r in D.RUNG_IDS:
        d, se = K["D_" + r]
        P("  D_%s  = ch%s - nd%s = %+.4f pp = %+.2f SE (SE %.6f)  [%+.4f, %+.4f]  TRAIN %+.4f   state %s"
          % (r, r, r, d, d / se, se, d - RES * se, d + RES * se, K["DTRAIN_" + r], states[r]))
    v, se = K["DD"]
    P("  DD   = D_I4 - D_I5 = %+.4f pp = %+.2f SE (SE %.6f)   DESCRIPTIVE" % (v, v / se, se))
    for r in D.RUNG_IDS:
        for g in ("ch", "nd"):
            v, se = K["T%s_%s" % (g, r)]
            P("  T%s_%s = k01%s - %s%s = %+.4f pp = %+.2f SE" % (g, r, r, g, r, v, v / se))
        P("    scalar reading at %s: %s" % (r, scal[r]))
    for r in D.RUNG_IDS:
        v, se = K["G_" + r]
        P("  G_%s  = kL%s - k01%s = %+.4f pp = %+.2f SE   DESCRIPTIVE (layerwise vs scalar in this cell)" % (r, r, r, v, v / se))
    P("\nLADDERS (LEVELS, not effects; this cell only)")
    for g in D.GRAIN_IDS:
        P("  %-9s %s" % (D.SPEC_OF_GRAIN[g], " | ".join("LAMBDA %s %.4f" % (D.LAM_TEXT[r], A[g + r]["level"]) for r in D.RUNG_IDS)))
    P("\nREFERENCE (between batch, never pooled; cgw1 296.3, alpha-scaled): D_W1 %+.4f %s, D_W2 %+.4f %s, D_W4 %+.4f %s; "
      "scalar at W4 %s" % (D.CGW1_D["W1"], D.CGW1_STATES["W1"], D.CGW1_D["W2"], D.CGW1_STATES["W2"], D.CGW1_D["W4"],
                           D.CGW1_STATES["W4"], D.CGW1_STATES["SCALAR_W4"]))

    stamps = list(BOUNDS)
    stamps.append("SIGMA-PRIOR-FROZEN" if which == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH")
    for r in D.RUNG_IDS:
        stamps.append("%s-VS-CGW1-W1-%s" % (r, "SAME" if states[r] == D.CGW1_STATES["W1"] else "DIFFERS"))
        if states[r] in ("SURVIVES", "REVERSES"):
            stamps.append("TRAIN-%s-%s" % ("AGREES" if (K["DTRAIN_" + r] > 0) == (K["D_" + r][0] > 0) else "DISAGREES", r))
        g_, gse = K["G_" + r]
        if g_ <= -SURV and g_ < -RES * gse:
            stamps.append("LAYERWISE-BELOW-SCALAR-%s" % r)
        elif g_ >= SURV and g_ > RES * gse:
            stamps.append("LAYERWISE-ABOVE-SCALAR-%s" % r)
        for g in ("k01", "kL"):
            if not A[g + r]["boxfree"]:
                stamps.append("BOX-BOUND-%s%s" % (g, r))
    for a in D.ARMS:
        if A[a]["range"] > DIVERGED:
            stamps.append("DIVERGED-%s" % a)
        if A[a]["level"] < HEALTH_MIN:
            stamps.append("UNHEALTHY-%s" % a)
    stateword = "+".join("%s-%s" % (r, states[r]) for r in D.RUNG_IDS)
    scalword = "+".join("%s-%s" % (r, scal[r]) for r in D.RUNG_IDS)
    P("\nLICENCE (primary)  %s" % LICENCE[primary])
    P("LICENCE (scalar)   %s" % SC_LICENCE[sc])
    P("\nFINAL %s | %s | %s | %s | %s" % (primary, sc, stateword, scalword, " ".join(stamps)))
    return 0

RESULTS = {"PASS": 0, "FAIL": 0, "SKIP": 0}


def chk(cond, label):
    RESULTS["PASS" if cond else "FAIL"] += 1
    print("  %s %s" % ("PASS" if cond else "FAIL", label))
    return cond


def skip(label):
    RESULTS["SKIP"] += 1
    print("  SKIP %s" % label)


def _curve(level):
    return [min(99.0, level * (1 - math.exp(-(e + 1) / 12.0))) for e in range(D.EPOCHS)]


def _write_out(d, arm, seed, jid, level, save_dir, args_over=None, env_over=None, extra_line=None, done=True,
               n_ep=D.EPOCHS, gpu="NVIDIA L4, 23034 MiB", witness=None, dup_epoch=False):
    rn = D.run_name(arm, seed)
    pairs = D.expected_args(arm, seed, save_dir)
    if args_over:
        pairs = args_over(pairs)
    args = "ARGS: " + " ".join("--%s %s" % kv for kv in pairs)
    env = D.ENV_TEMPLATE.replace("<P>", os.path.join(save_dir, D.probe_dir_name(arm, seed)))
    if env_over:
        env = env_over(env)
    c = _curve(level)
    L = ["NODE=[HOST_2f2f29d9e1] | JOB=%s %d | AUGMENT=1" % (rn, jid), args, env, gpu, "",
         "args_base includes unnecessary attributes {'normalizer_param'} "]
    wl = [D.witness_on(D.RUNG_OF[arm])] if witness is None else witness
    L += wl
    L.append("")
    if extra_line:
        L.append(extra_line)
    for e in range(n_ep):
        te = level if e >= PL_LO - 5 else c[e]
        L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.99, te + 5.0), te))
        if dup_epoch and e == 97:
            L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, 50.0, 50.0))
    L.append("%d  minutes" % 50)
    if done:
        L.append("RUN_DONE")
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(L) + "\n")


def _write_probe(pdir, m, lam, lo=-8.0, hi=-5.0, n=D.N_RECORDS, nb=None, hi_frac=0.0, lam_rec=None, meas_mult=1.0,
                 no_dr=False, applied=None, dr_n_off=0):
    os.makedirs(pdir, exist_ok=True)
    with open(os.path.join(pdir, "probe.jsonl"), "w") as fh:
        for i in range(n):
            bmax = D.HI if i >= n * (1 - hi_frac) else hi + 1e-4 * (i % 7)
            s = '{"step": %d, "beta": [%.6f], "beta_true_min": %.6f, "beta_true_max": %.6f, "n_beta": %d, "n_at_lo": 0, ' \
                '"n_at_hi": %d' % (i * D.PROBE, 0.5 * (lo + hi), lo - 1e-4 * (i % 5), bmax, nb if nb is not None else m,
                                   1 if bmax >= D.HI else 0)
            if not no_dr:
                L_ = lam if lam_rec is None else lam_rec
                s += (', "dr_mode": "alpha_indep", "dr_lam": %r, "dr_n": %d, "dr_shrink_meas": %r, "dr_shrink_applied": %r, '
                      '"dr_trace_decay": %r' % (L_, i * D.PROBE + 2 + dr_n_off,
                                                L_ * meas_mult * (1.0 + 2e-4 * ((i % 3) - 1)),
                                                L_ if applied is None else applied, 1.0 - L_))
            fh.write(s + "}\n")


JIT = (0.02, 0.03, -0.03, -0.02)   # zero-mean over 4 seeds


def _build(root, template, levels, save_dir="/SAVE/runs/cai1", over=None, drop=(), extra=None, probe_over=None,
           prov_over=None, jit=None):
    os.makedirs(os.path.join(root, PREFIX), exist_ok=True)
    jid = 7000000
    for (arm, seed) in D.RUNS:
        jid += 1
        if (arm, seed) in drop:
            continue
        i = D.SEEDS[arm].index(seed)
        lv = levels[arm] + (JIT[i] if jit is None else jit(arm, i))
        kw = dict((over or {}).get((arm, seed), {}))
        _write_out(os.path.join(root, PREFIX), arm, seed, jid, lv, save_dir, **kw)
        dst = os.path.join(root, PREFIX, D.probe_dir_name(arm, seed))
        if probe_over and (arm, seed) in probe_over:
            _write_probe(dst, D.M_OF_ARM[arm], D.LAM_OF_ARM[arm], **probe_over[(arm, seed)])
        else:
            os.symlink(os.path.join(template, D.probe_dir_name(arm, seed)), dst)
    for f in (extra or []):
        f(os.path.join(root, PREFIX))
    kv = {"MODE": "submit", "SCORER_SHA256": file_sha(os.path.abspath(__file__)),
          "DESIGN_SHA256": file_sha(os.path.join(HERE, "cai1_design.py")), "HF_SHA256": D.HF_POST_SHA}
    kv.update(prov_over or {})
    open(os.path.join(root, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % x for x in kv.items()))
    return root


def _run(root, csvpath=None):
    lines = []
    rc = score(root, csvpath, out=lines.append)
    lines = [x for ln in lines for x in ln.split("\n")]
    fin = [ln for ln in lines if ln.startswith("FINAL ")]
    return rc, (fin[0] if fin else ""), lines
