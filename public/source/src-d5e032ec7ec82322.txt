# layer4.0.bn2.weight, 53 layer4.0.shortcut.1.weight, 59 layer4.1.bn2.weight) are
# EXACTLY the 512-wide BatchNorm scales whose output is a SUMMAND OF A RESIDUAL
# ADDITION, and its two non-carriers (47, 56, the bn1 scales) are exactly the two
# that are not.  On a ResNet that reading is inseparable from "the deepest BN
# scales", because every deep BN scale feeds an add.
#
# VGG11_bn has no addition but differs from ResNet18 in depth, channel schedule
# and downsampling as well; every VGG verdict carries NOT-A-ONE-VARIABLE-
# ABLATION.  PlainNet18_c100 (patches/patch_plainnet.py) is ResNet(PlainBlock,
# [2,2,2,2], 100) where PlainBlock is BasicBlock minus `out += self.shortcut(x)`
# and the three projection-shortcut modules are NOT CONSTRUCTED (198: they would
# otherwise leave 9 tensors with no gradient path in the list HF.init_meta
# consumes positionally).  Same class, stem, stages, strides, widths, BN layers,
# ReLUs, pool and head.  tests/test_plainnet.py P3e proves it FUNCTIONALLY: with
# ResNet18_c100's own weights loaded, PlainNet18_c100's output is bitwise equal
# to ResNet18_c100's modules walked with every addition removed.
#
#   THIS BATCH IS THE ONE-VARIABLE RESIDUAL ABLATION.  The stamp
#   ONE-VARIABLE-RESIDUAL-ABLATION is printed unconditionally, with the two
#   declared consequences of removing the addition that are NOT extra
#   variables: (i) THIRD-CARRIER-ABSENT -- layer4.0.shortcut.1.weight no longer
#   exists, so only the two surviving carriers can be isolated; (ii) the 53-tensor
#   base changes what `scalar` sums and `layerwise` counts, exactly as removing
#   the shortcut modules must.
#
# THE SHARP QUESTION: once the ex-carriers no longer feed an addition, do they
# still carry the scalar->layerwise gap?
#

# of block 0's addition (and ONE rescued +41.47 pp there); on PlainNet it feeds

def fmt(x, nd=4):
    return "n/a" if x is None else ("%.*f" % (nd, x))


def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


# =============================================================================
# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.
# =============================================================================
def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


# =============================================================================
# the frozen PlainNet18_c100 manifest, composed from ResNet-18's layout.  The
# launcher's guard 4 proves the LIVE model equals it name by name.
# =============================================================================
def resnet18_c100_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9), ("bn1.weight", 64), ("bn1.bias", 64)]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9), (pre + "bn1.weight", w),
                  (pre + "bn1.bias", w), (pre + "conv2.weight", w * w * 9),
                  (pre + "bn2.weight", w), (pre + "bn2.bias", w)]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin), (pre + "shortcut.1.weight", w),
                      (pre + "shortcut.1.bias", w)]
        inp = w
    return t + [("linear.weight", 512 * 100), ("linear.bias", 100)]


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists, re-implemented HERE so the
    registered index sets are checked without importing HF.py."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(idx)
    return out

# run readers -- the RAW .out files are PRIMARY
# =============================================================================
def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    args = next((ln[len("ARGS:"):].strip() for ln in lines if ln.startswith("ARGS:")), "")
    rec = {"n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "args": args, "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (rec["n_epochs"] == EPOCHS and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  Among several files for one (arm, seed) the highest
    job id among COMPLETE runs wins; a truncated resubmission cannot shadow a
    complete run.  -> (runs, notes)"""
    cand = {}
    notes = []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def read_manifest(path):
    if not path or not os.path.exists(path):
        return None
    d = {"tensors": [], "armspec": {}, "isolated": {}, "raw": {}}
    for ln in open(path):
        p = ln.split()
        if not p:
            continue
        if p[0] == "TENSOR" and len(p) >= 4:
            d["tensors"].append((int(p[1]), p[2], int(p[3])))
        elif p[0] == "ARMSPEC" and len(p) >= 3:
            d["armspec"][p[1]] = " ".join(p[2:])
        elif p[0] == "ISOLATED" and len(p) >= 3:
            d["isolated"][p[1]] = p[2]
        elif len(p) >= 2:
            d["raw"][p[0]] = " ".join(p[1:])
    return d


def resolve_manifest(runsdir, explicit):
    """DEFAULT resolution, so the documented ONE-argument invocation cannot be got
    wrong (the 164.2 pattern).  -> (path, how)"""
    if explicit:
        return explicit, "explicit --manifest"
    for cand, how in (
        (os.path.join(runsdir, PREFIX, "PARTITION-MANIFEST.txt"), "<runsdir>/cpl1/"),
        (os.path.join(runsdir, PREFIX + "-PARTITION-MANIFEST.txt"), "<runsdir>/cpl1-"),
    ):
        if os.path.exists(cand):
            return cand, how
    return None, "not found"

def read_corpus(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    return [r for r in csv.DictReader(open(csvpath))
            if not r.get("run", "").startswith(PREFIX + "-")]


def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, any_gran=False):
    return ((any_gran or r.get("granularity") in ("scalar", "layerwise"))
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra, any_gran=False):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, any_gran) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm, top = 0.0, 0, 0, 0, 0.0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        s = sum((x - m) ** 2 for x in v)
        ss += s
        df += len(v) - 1
        nc += 1
        nm += len(v)
        top = max(top, s)
    return (math.sqrt(ss / df) if df else None), df, nc, nm, (top / ss if ss else None)


def is_r18c100(r):
    return r.get("network") == PARENT and r.get("dataset") == DSET


def prefix_mean(rows, prefix, gran):
    v = [float(r["plateau5"]) for r in rows if r.get("run", "").startswith(prefix + "-")
         and usable(r) and std_cell(r, True) and is_r18c100(r) and r.get("granularity") == gran]
    return (mean(v), len(v)) if v else (None, 0)

# the nomination readout -- DESCRIPTIVE, NON-GATING.  Reads this batch's own k01
# PROBE_TENSOR records (pinned scalar records) the way cVI1 read cvg1's.
# =============================================================================
NOM_SETS = {"44": [43], "50": [49], "44_50": [43, 49], "41_47": [40, 46]}


def nomination(runsdir):
    base = None
    for cand in (os.path.join(runsdir, PREFIX), runsdir):
        if os.path.isdir(cand) and any(d.startswith("probe_cpl1-k01-s")
                                       for d in os.listdir(cand)):
            base = cand
            break
    if base is None:
        return None
    share_acc, flip_acc, npin, nseeds = {}, {}, 0, 0
    for d in sorted(x for x in os.listdir(base) if x.startswith("probe_cpl1-k01-s")):
        p = os.path.join(base, d, "probe.jsonl")
        if not os.path.exists(p):
            continue
        pin = []
        for ln in open(p):
            ln = ln.strip()
            if not ln:
                continue
            try:
                r = json.loads(ln)
            except ValueError:
                continue
            if (r.get("n_beta", 0) > 0 and r.get("n_at_lo", -1) == r.get("n_beta")
                    and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
                    and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS):
                pin.append(r)
        if not pin:
            continue
        nseeds += 1
        npin += len(pin)
        mass = [0.0] * NTENS
        fl = dict((k, 0) for k in NOM_SETS)
        for r in pin:
            b2 = r.get("pt_b2", 0.9)
            L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
            S = sum(L)
            s0 = (S > 0) - (S < 0)
            for i, v in enumerate(L):
                mass[i] += abs(v)
            for k, idx in NOM_SETS.items():
                Sr = S - sum(L[i] for i in idx)
                if ((Sr > 0) - (Sr < 0)) != s0:
                    fl[k] += 1
        tm = sum(mass) or 1.0
        for i in range(NTENS):
            share_acc[i] = share_acc.get(i, 0.0) + mass[i] / tm
        for k in NOM_SETS:
            flip_acc[k] = flip_acc.get(k, 0.0) + fl[k] / len(pin)
    if not nseeds:
        return None
    return (dict((i, v / nseeds) for i, v in share_acc.items()),
            dict((k, v / nseeds) for k, v in flip_acc.items()), npin, nseeds)

# THE BRANCH MAP -- a pure function of the in-batch means and the gates
# =============================================================================
def decide(M, floor_ok=True, trains_ok=True, diverged=False):
    D_GAP = M["kL"] - M["k01"]
    D_ISO = M["ISO"] - M["k01"]
    D_CTL = M["CTL"] - M["k01"]
    D_ONE = M["ONE"] - M["k01"]
    DELTA_ID = M["ISO"] - M["CTL"]
    if not floor_ok:
        return "HARNESS-UNSOUND"
    if not trains_ok:
        return "PLAIN-UNTRAINABLE"
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if D_GAP < GAP_ABSENT_BAR:
        return "GAP-ABSENT-PLAIN"
    if D_GAP < GAP_MIN:
        return "GAP-ATTENUATED-PLAIN"
    rI, rC, rO = D_ISO >= RESCUE_BAR, D_CTL >= RESCUE_BAR, D_ONE >= RESCUE_BAR
    if not (rI or rC or rO):
        return "NO-RESCUE-PLAIN"
    if not rI:
        return "UNEXPECTED-PATTERN"
    if DELTA_ID <= -IDENTITY_BAR:
        return "CONTROL-DOMINATES-PLAIN"
    if DELTA_ID >= IDENTITY_BAR:
        if rO:
            return "CARRIER-SURVIVES-PLAIN"
        if D_ONE <= NULL_BAR:
            return "HEAD-CARRIES-PLAIN"
        return "ONE-PARTIAL-PLAIN"
    if rC and abs(DELTA_ID) <= NULL_BAR:
        return "CLASS-OPERATIVE-PLAIN"
    return "IDENTITY-ATTENUATED-PLAIN"

def score(runsdir, csvpath, manifest_path):
    print("=" * 78)
    print(" cpl1 -- ResNet-18 WITHOUT ITS RESIDUAL ADDITIONS: THE ONE-VARIABLE ABLATION")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d"

        print("   %-4s %s" % (a, SPEC[a]))

    print("          plateau5 = mean TEST accuracy over epochs %d..%d of each run's own .out."
          % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2); the corpus cannot move them.")
    print("=" * 78)

    runs, rnotes = read_runs(runsdir)

    # ---- G-STRUCT ------------------------------------------------------------------
    print("\nG-STRUCT  the live model and the three partitions are what the batch claims")
    mp, how = resolve_manifest(runsdir, manifest_path)
    man = read_manifest(mp)
    if man is None:
        chk(False, "G-STRUCT PARTITION-MANIFEST.txt present", how)
    else:
        print("  manifest: %s (%s)" % (mp, how))
        names = [t[1] for t in man["tensors"]]
        numel = dict((t[0], t[2]) for t in man["tensors"])
        want = resnet18_c100_tensors(True)
        chk([(n, q) for _i, n, q in man["tensors"]] == want,
            "G-STRUCT the %d tensors (name, numel, order) are PlainNet18_c100's" % NTENS,
            "%d tensors" % len(man["tensors"]))
        chk(sum(numel.values()) == TOTPAR, "G-STRUCT %d parameters" % TOTPAR,
            str(sum(numel.values())))
        w512 = tuple((i, n) for i, n, q in man["tensors"] if q == 512 and n.endswith(".weight"))
        chk(w512 == tuple(zip(BN512_IDX, BN512_NAMES)),
            "G-STRUCT the 512-wide BN scales are at 41/44/47/50", str(w512))
        chk(not any("shortcut" in n for n in names),
            "G-STRUCT NO parameter name contains `shortcut`")

            "G-STRUCT the 9 tensors removed from ResNet18_c100 are exactly its shortcut tensors",

        for a in BLOCKWISE_ARMS:
            spec = man["armspec"].get(a, "")
            m = re.search(r"SIZES (\S+) PARAMS (\S+) GROUP0 .* GROUP1 (.*)$", spec)
            if not m:
                chk(False, "G-STRUCT %s ARMSPEC line present" % a, spec[:60])

            sizes = [int(x) for x in m.group(1).split(",")]
            pars = [int(x) for x in m.group(2).split(",")]
            iso = m.group(3).split()
            chk(sizes == WANT_SIZES[a] and pars == WANT_PARAMS[a]
                and iso == [n for n, _i in ISOLATED[a]] and spec.startswith("SPEC " + SPEC[a] + " "),
                "G-STRUCT %s composes to sizes %s params %s isolating %s"
                % (a, WANT_SIZES[a], WANT_PARAMS[a], "/".join(n for n, _i in ISOLATED[a])),
                "sizes %s params %s group1 %s" % (sizes, pars, iso))
        chk(len(set(man["armspec"].get(a, a) for a in BLOCKWISE_ARMS)) == 3,
            "G-STRUCT ISO, CTL and ONE are three DIFFERENT partitions")

                                                         kv.get("--stepsize-groups"),
                                                         kv.get("--seed")))
    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]))

        "G-ENV every run printed ONE `PROBE_TENSOR: on ... tensors=%d` line of its arm's type" % NTENS,

    # ---- the levels ------------------------------------------------------------------
    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH")
    arm_v, arm_t = {}, {}
    for a in ARMS:
        rs = [runs[(a, s)] for s in SEEDS if (a, s) in runs and runs[(a, s)]["complete"]]
        arm_v[a] = [r["plateau5"] for r in rs]
        arm_t[a] = [r["train5"] for r in rs]
        if arm_v[a]:
            print("  %-4s TEST %.4f  sd %s  range %.3f  n=%d   TRAIN %s   %s"
                  % (a, mean(arm_v[a]), fmt(sd(arm_v[a]), 3), max(arm_v[a]) - min(arm_v[a]),
                     len(arm_v[a]), fmt(mean(arm_t[a])), SPEC[a]))
            print("       seeds: %s" % ", ".join("s%d %.4f" % (r["seed"], r["plateau5"]) for r in rs))

    if not all(len(arm_v[a]) == len(SEEDS) for a in ARMS):
        print("\nFINAL: INCOMPLETE -- %s"
              % ", ".join("%s %d/%d" % (a, len(arm_v[a]), len(SEEDS)) for a in ARMS))
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")

    # ---- sigma (O2) --------------------------------------------------------------------
    print("\nSIGMA     O2: max(frozen floor, in-batch) -- nothing from the corpus")
    ss = sum(sum((x - mean(arm_v[a])) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    which, sigma_used, cands = compose_sigma(sigma_in)
    for nm, v in cands:
        print("  %-20s %.6f" % (nm, v))
    se = sigma_used * math.sqrt(2.0 / 3.0)
    print("  SIGMA_USED           %.6f  (= %s)" % (sigma_used, which))
    print("  SE_ARM_DIFF          %.6f  = SIGMA_USED * sqrt(2/3)  (df_inbatch %d)" % (se, df))
    try:                                     # NON-GATING disclosure; assigns nothing a bar reads

            _a = pooled_sigma(_rows, is_r18c100, True)[0]
            note("live corpus (NON-GATING, NOT USED): NARROW %s, R18ALL %s; frozen %.6f / %.6f"
                 % (fmt(_n, 6), fmt(_a, 6), SIGMA_NARROW, SIGMA_R18ALL))
    except Exception as _e:                  # a disclosure must never crash the scorer
        note("live corpus disclosure skipped: %s" % type(_e).__name__)

    # ---- G-FLOOR / G-TRAINS / G-CEIL / G-DIVERGE -----------------------------------------
    print("\nG-FLOOR / G-TRAINS / G-CEIL / G-DIVERGE")
    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    hi = max(M.values())
    floor_ok = chk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE),
                   "max %.4f" % hi)
    trains_ok = chk(M["kL"] >= KL_MIN, "G-TRAINS kL mean >= %.2f pp (the plain net trains at this cell)"
                    % KL_MIN, "kL %.4f" % M["kL"])
    chk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)

def _e_probe(d, dominant_idx0, nrec=40, nseeds=3):
    """synthetic pinned k01 records where ONE tensor (0-based) dominates the sign"""
    for s in SEEDS[:nseeds]:
        p = os.path.join(d, PREFIX, "probe_cpl1-k01-s%d" % s)
        os.makedirs(p, exist_ok=True)
        with open(os.path.join(p, "probe.jsonl"), "w") as f:
            for k in range(nrec):
                # every other tensor votes the OTHER way, so removing the dominant one
                # flips the remainder's sign and removing any other set does not
                m = [-1e-4 - 1e-6 * ((i + k) % 3) for i in range(NTENS)]
                z = [0.0] * NTENS
                m[dominant_idx0] = 0.5
                f.write(json.dumps({"step": 100 * k, "n_beta": 1, "n_at_lo": 1, "pt_b2": 0.9,
                                    "m_tensor": m, "z_tensor": z}) + "\n")

def _e_score(d, csvpath):
    """the REAL score(), stdout captured, GATE_FAIL saved and restored"""
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc = score(d, csvpath, "")
    except Exception as ex:               # a crash is a FAIL line, never a traceback
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), ""
    finally:
        GATE_FAIL[:] = saved
    out = buf.getvalue()
    fin = [ln for ln in out.splitlines() if ln.startswith("FINAL:")]
    return rc, (fin[-1] if fin else "(no FINAL line)"), out

                used.add(int(r.get("seed", "")))
            except ValueError:
                pass
        chk(not (set(SEEDS) & used), "ZERO corpus rows ANYWHERE carry seed 69, 70 or 71")

    print("\nD. THE FLOOR TABLE (164.6).  Chance %.2f pp, ceiling 100.  Levels UNSURE." % CHANCE)
    lo_all, hi_all = 100.0, 0.0
    for acc, pred in ACCOUNTS.items():
        at = [a for a in BLOCKWISE_ARMS if pred[a] == pred["k01"]]
        off = [a for a in BLOCKWISE_ARMS if pred[a][0] > pred["k01"][1] or pred["k01"] == FREE_PRED]
        lo_all = min(lo_all, min(v[0] for v in pred.values()))
        hi_all = max(hi_all, max(v[1] for v in pred.values()))
        print("  %-9s %s   at-k01: %-12s off-floor m=2: %-12s"
              % (acc, "  ".join("%s %2.0f-%2.0f" % (a, pred[a][0], pred[a][1]) for a in ARMS),
                 ",".join(at) or "-", ",".join(off) or "NONE"))
        mid = dict((a, (pred[a][0] + pred[a][1]) / 2.0) for a in ARMS)
        if acc == "IDENT":
            mid["ONE"] = 58.0
        chk(decide(mid) == EXPECTED_BRANCH[acc],
            "%-9s midpoint prediction lands in %s" % (acc, EXPECTED_BRANCH[acc]), decide(mid))
    chk(lo_all - CHANCE >= 15.0,
        "the LOWEST level ANY account predicts for ANY arm is >= 15 pp clear of chance",
        "%.1f pp margin" % (lo_all - CHANCE))
    chk(100.0 - hi_all >= 25.0, "the HIGHEST predicted level is >= 25 pp clear of the ceiling",
        "%.1f pp margin" % (100.0 - hi_all))
    chk(ACCOUNTS["R-ONLY"] == ACCOUNTS["H-BROKEN"],
        "R-ONLY and H-BROKEN predict IDENTICALLY on every arm -- declared: NO-RESCUE cannot "
        "confirm R-ONLY (stamped H-BROKEN-NOT-EXCLUDED)")
    print("  Every OTHER account puts at least one m=2 arm OFF the floor (or has no collapse),")
    print("  so a floored ONE/CTL under PATH/IDENT is readable as a real null, not 'it died'.")
    print("  DELTA_ID under R-ONLY/H-BROKEN sits at the floor-difference (a soft bound); that")
    print("  reading is therefore never quoted as a point agreement.")

    print("\nE. THE REAL score() ON SYNTHETIC RUNS (invented numbers; evidence of nothing)")
    tmp = tempfile.mkdtemp(prefix="cpl1_selftest_")
    cases = [
        ("PATH", {}, "HEAD-CARRIES-PLAIN"),
        ("IDENT", {"ONE": 58.0}, "CARRIER-SURVIVES-PLAIN"),
        ("CLASS", {"CTL": 62.5, "ONE": 60.0}, "CLASS-OPERATIVE-PLAIN"),
        ("R-ONLY/H-BROKEN", {"ISO": 25.6}, "NO-RESCUE-PLAIN"),
        ("R-NOGAP", {"k01": 60.0, "ISO": 63.0, "CTL": 61.0, "ONE": 62.0}, "GAP-ABSENT-PLAIN"),
        ("attenuated gap", {"k01": 48.0}, "GAP-ATTENUATED-PLAIN"),
        ("untrainable", {"k01": 18.0, "kL": 22.0, "ISO": 21.0, "CTL": 18.2, "ONE": 18.1},
         "PLAIN-UNTRAINABLE"),
        ("harness", dict((a, 1.0) for a in ARMS), "HARNESS-UNSOUND"),
        ("one-partial", {"ONE": 31.0}, "ONE-PARTIAL-PLAIN"),
        ("control-dominates", {"ISO": 36.0, "CTL": 62.0}, "CONTROL-DOMINATES-PLAIN"),
        ("unexpected", {"ISO": 25.5, "ONE": 50.0}, "UNEXPECTED-PATTERN"),
        ("identity-attenuated", {"ISO": 55.0, "CTL": 45.0}, "IDENTITY-ATTENUATED-PLAIN"),
        ("near-bar ONE just rescues", {"ONE": 25.0 + RESCUE_BAR + 0.05}, "CARRIER-SURVIVES-PLAIN"),
        ("near-bar ONE just misses", {"ONE": 25.0 + RESCUE_BAR - 0.05}, "ONE-PARTIAL-PLAIN"),
        ("near-bar gap", {"kL": 25.0 + GAP_MIN - 0.05, "ISO": 44.0}, "GAP-ATTENUATED-PLAIN"),
    ]
    results = {}
    for lab, lv, want in cases:
        d = os.path.join(tmp, re.sub(r"\W+", "_", lab))
        os.makedirs(d)
        _e_build(d, levels=lv)
        rc, fin, out = _e_score(d, csvpath)
        results[lab] = (rc, fin, out, d)
        chk(fin.startswith("FINAL: %s |" % want), "E %-26s -> %s" % (lab, want), fin[:70])
    rc, fin, out, dpath = results["PATH"]
    chk(rc == 0 and "GATES-CLEAN" in fin and "H-BROKEN-EXCLUDED" in fin
        and "DECOMPOSITION-BY-SUBTRACTION" in fin and "ONE-VARIABLE-RESIDUAL-ABLATION" in fin,
        "E PATH scores exit 0, GATES-CLEAN, H-BROKEN-EXCLUDED, the unconditional stamps", fin[:90])
    chk("H-BROKEN-NOT-EXCLUDED" in results["R-ONLY/H-BROKEN"][1]
        and "FLOOR-READINGS-ARE-BOUNDS" in results["R-ONLY/H-BROKEN"][1],
        "E NO-RESCUE stamps H-BROKEN-NOT-EXCLUDED and FLOOR-READINGS-ARE-BOUNDS")
    d = os.path.join(tmp, "dropped")
    os.makedirs(d)
    _e_build(d, drop=("ONE", 70))
    rc, fin, out = _e_score(d, csvpath)
    chk(rc == 2 and fin.startswith("FINAL: INCOMPLETE"), "E a DROPPED run -> FINAL: INCOMPLETE, exit 2",
        fin[:70])
    d = os.path.join(tmp, "shadow")
    os.makedirs(d)
    _e_build(d, shadow=("ISO", 69))
    rc, fin, out = _e_score(d, csvpath)
    chk(fin == results["PATH"][1], "E a TRUNCATED higher-jid resubmission does not shadow a complete run")
    d = os.path.join(tmp, "envbreak")
    os.makedirs(d)
    _e_build(d, env_break=("CTL", 71))
    rc, fin, out = _e_score(d, csvpath)
    chk(rc == 1 and "GATES-" in fin and "GATES-CLEAN" not in fin, "E an AUGMENT=0 run fails G-ENV", fin[:60])
    d = os.path.join(tmp, "specbreak")
    os.makedirs(d)
    _e_build(d, spec_break=("ISO", 70))
    rc, fin, out = _e_score(d, csvpath)
    chk(rc == 1 and "GATES-CLEAN" not in fin, "E a run whose ARGS spec != its arm label fails G-ARGS",
        fin[:60])
    d = os.path.join(tmp, "nomanifest")
    os.makedirs(d)
    _e_build(d)
    os.remove(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"))
    rc, fin, out = _e_score(d, csvpath)
    chk(rc == 1 and "GATES-CLEAN" not in fin, "E a MISSING manifest fails G-STRUCT (not silently skipped)")
    d = os.path.join(tmp, "probe")
    os.makedirs(d)
    _e_build(d)
    _e_probe(d, 49)
    rc, fin, out = _e_score(d, csvpath)
    chk(fin == results["PATH"][1] and "removing {50} flips the remainder's sign on 1.0000" in out
        and "removing {41_47} flips the remainder's sign on 0.0000" in out,
        "E the NON-GATING nomination readout runs on synthetic records and does not touch FINAL")

    print("\n   O2 INVARIANCE: the SAME synthetic runs against five corpora")
    variants = _e_csv_variants(csvpath if csvpath and os.path.exists(csvpath) else None, tmp)
    ref_fin, ref_bars = None, None
    for lab in ("real", "real+invented", "minimal-invented", "no-corpus"):
        if lab not in variants:
            skip("O2 %s corpus not available on this host" % lab)
            continue
        rc, fin, out = _e_score(dpath, variants[lab])
        bars = _bar_lines(out)
        if ref_fin is None:
            ref_fin, ref_bars = fin, bars
        chk(fin == ref_fin and bars == ref_bars and len(bars) == 4,
            "O2 %-17s FINAL and the SIGMA/SE/bars/PRIMARY lines are IDENTICAL" % lab, fin[:50])
    if "real+invented" in variants:
        live_real = pooled_sigma(read_corpus(variants["real"]), is_r18c100, True)[0]
        live_inv = pooled_sigma(read_corpus(variants["real+invented"]), is_r18c100, True)[0]
        chk(live_inv is not None and live_real is not None and live_inv > 2 * live_real,
            "O2 non-vacuity: the invented rows MOVE the live corpus sigma (%.4f -> %.4f), and the "
            "FINAL line above did not move" % (live_real or 0, live_inv or 0))
    mv = dict(_E_BASE)
    chk(all(decide(dict(mv, **{a: mv[a] + 1e-9})) == decide(mv) for a in ARMS),
        "O2 decide() is a function of the in-batch means only (no hidden state)")

    print("\n   THE DOCUMENTED ONE-ARGUMENT INVOCATION, as a subprocess")
    me = os.path.abspath(__file__)
    try:
        p = subprocess.run([sys.executable, me, dpath], capture_output=True, text=True, timeout=300)
        fin = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 0 and fin and fin[-1] == results["PATH"][1],
            "`python3 %s <runsdir>` exits 0 with the in-process FINAL line" % os.path.basename(me),
            "rc %d" % p.returncode)
        p = subprocess.run([sys.executable, me, os.path.join(tmp, "dropped")],
                           capture_output=True, text=True, timeout=300)
        fin = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 2 and fin and fin[-1].startswith("FINAL: INCOMPLETE"),
            "... and on a dropped run exits 2 with FINAL: INCOMPLETE", "rc %d" % p.returncode)
    except Exception as e:
        chk(False, "subprocess invocation", "%s: %s" % (type(e).__name__, e))

    print("\nF. THE BRANCH MAP IS TOTAL -- 4,000 random in-batch means, no exception, "
          "every branch registered")
    import random
    rnd = random.Random(20260915)
    seen = set()
    for _ in range(4000):
        m = dict((a, rnd.uniform(1.0, 80.0)) for a in ARMS)
        seen.add(decide(m, rnd.random() > 0.05, rnd.random() > 0.05, rnd.random() < 0.05))
    chk(seen <= set(LICENSE), "every branch decide() returns has a registered licence text",
        "%d distinct" % len(seen))
    note("branches reached by the random sweep: %s" % ", ".join(sorted(seen)))
    reached = set(r[1].split(" |")[0][len("FINAL: "):] for r in results.values())
    chk(reached | seen == set(LICENSE),
        "every registered branch is reached by section E or the sweep (none is dead text)",
        "missing %s" % sorted(set(LICENSE) - reached - seen))

    print("\nG. RULE 21 PREMISE: no cpl1 run may exist yet")
    runs, _n = read_runs(runsdir) if runsdir else ({}, [])
    chk(not runs, "no cpl1-*.out under %s" % runsdir, "%d found" % len(runs))
    if csvpath and os.path.exists(csvpath):
        chk(not any(r.get("run", "").startswith(PREFIX + "-") for r in csv.DictReader(open(csvpath))),
            "no cpl1- row in the CSV")

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))
    for f in GATE_FAIL:
        print("   FAIL", f)
    return 1 if GATE_FAIL else 0

def main():
    ap = argparse.ArgumentParser(
        description="cpl1 registered scorer.  Documented invocation: "
                    "python3 analysis/cPL1_plainnet_residual_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cpl1/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    return score(runsdir, a.csv, a.manifest)


if __name__ == "__main__":
    sys.exit(main())
