def file_sha(path):

def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"

def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d

def gpu_of(path):
    """the GPU name from the run's own nvidia-smi line (`<name>, <n> MiB`), or None.  DISCLOSURE ONLY."""
    for ln in open(path, errors="replace"):
        ln = ln.strip()

    return None

def lion_natural(beta_pre, mom_pre, z):
    return S9.lion_natural(beta_pre, mom_pre, z)


def lion_check(recs, ng):
    d = {"n": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "nonfinite": 0, "worst": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre, mom, z = S9._groupvec(r, "beta_pre", ng), S9._groupvec(r, "mom_pre", ng), S9._groupvec(r, "z_agg", ng)
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        if r["step"] != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        for g in range(ng):
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst"] = max(d["worst"], e)
            d["bad_lion"] += e > BH_TOL
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS and d["bad_step"] == 0
          and d["bad_lion"] == 0)
    return ok, d


def bite_run(arm, recs):
    okl, dl = lion_check(recs, NG[arm])
    hk = C.hold_keys_present(recs)
    okd, dd = C.dm_audit(recs, K_MASKED[arm], N_RECORDS)
    return okl and okd and hk == 0, dl, dd, hk

def rung_state(M, r):
    k, L = M[SCALAR_OF[r]], M[LAYER_OF[r]]
    if L < REF_MIN:
        return "UNREADABLE"
    if L > 0 and k <= R50 * L:
        return "COLLAPSE"
    if L - k < GAP_BAR:
        return "NOGAP"
    return "PARTIAL"


def carrier_state(M, states):
    """the carrier companion at rung W2.  UNREADABLE unless that rung is COLLAPSE (cmo1's ISO-UNREADABLE precedent)."""
    if states[RUNG_IDS.index(CARRIER_RUNG)] != "COLLAPSE":
        return "CAR-UNREADABLE"
    if M[CAR] >= M[LAYER_OF[CARRIER_RUNG]] - MATCH_BAR:
        return "CAR-REC"
    if M[CAR] <= M[SCALAR_OF[CARRIER_RUNG]] + NULL_BAR:
        return "CAR-NULL"
    return "CAR-PART"


def account_of(states):
    """states: the four rung states in RUNG_IDS order, none UNREADABLE and the first COLLAPSE."""
    c = 0
    while c < len(states) and states[c] == "COLLAPSE":
        c += 1
    if "COLLAPSE" in states[c:]:
        return "NON-MONOTONE"
    if c == len(states):
        return "COLLAPSE-AT-ALL-VALUES"
    if "PARTIAL" in states[c:]:
        return "GRADED-ABOVE-THRESHOLD-%s" % RUNG_IDS[c - 1]
    return "THRESHOLD-%s-%s" % (RUNG_IDS[c - 1], RUNG_IDS[c])


def decide(M, diverged=False, carrier_ok=True):
    """-> (token, state word or None)."""
    if diverged:
        return "UNRESOLVED-DIVERGED", None
    bad_ref = [r for r in RUNG_IDS if M[LAYER_OF[r]] < REF_MIN]
    if bad_ref:
        return "UNRESOLVED-REFERENCE", "REF-BELOW-%.0f-AT-%s" % (REF_MIN, "-".join(bad_ref))
    states = [rung_state(M, r) for r in RUNG_IDS]
    if M[SCALAR_OF["W1"]] > K01_MAX or states[0] != "COLLAPSE":
        return "ANCHOR-NOT-COLLAPSED", "W1-%s" % states[0]
    tok = account_of(states)
    cw = carrier_state(M, states) if carrier_ok else "CAR-PATCH-NOT-VERIFIED"
    return tok, "+".join("%s-%s" % (r, s) for r, s in zip(RUNG_IDS, states)) + "+" + cw

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    print("=" * 78)
    print(" cwd5 -- ResNet18_c100: THE COUPLED WEIGHT-DECAY LADDER.  Does the shared-step-size collapse exist at")
    print("         NORMAL weight-decay values, or only at the campaign's 0.1?")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for r in RUNG_IDS:
        print("   rung %s  --weight-decay-base %-5s   %-7s (scalar) | %-7s (layerwise)%s"
              % (r, D.WD_TOKEN[r], SCALAR_OF[r], LAYER_OF[r],
                 "   + %s (scalar, DECAY_MASK = the three ctd1 carriers)" % CAR if r == CARRIER_RUNG else ""))
    print(" CO-PRIMARY: G_W4 = %s - %s, the granularity gap at the STANDARD CIFAR weight decay."
          % (LAYER_OF["W4"], SCALAR_OF["W4"]))
    print(" Every state is WITHIN its rung.  plateau5 = mean TEST over epochs %d..%d of each run's own .out."
          % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2).  DECOUPLED WEIGHT DECAY IS NOT TESTED (281.2) and is stamped on every FINAL.")
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = C.read_runs(runsdir, PREFIX, ARMS, EPOCHS)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS / G-WDPATH   every run's OWN ARGS line says what its file name registers, INCLUDING its rung's")
    print("          --weight-decay-base token and its grain.  That the flag reaches the update is proved on the real")
    print("          GPU path by the bite job (bitwise w'/h'/m' at all four values) and in batch by %s's wd=0.01 witness." % CAR)
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = C.parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s),
                    "momentum-param-base": "0.99", "weight-decay-base": B.WD[a], "momentum-param-meta": "0.99",
                    "Lion-beta2-meta": "0.9", "weight-decay-meta": "0", "gamma": "1"}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d (wd %s, %s)" % (a, s, B.WD[a], SPEC[a]),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad]) if (dup or bad) else "")
    wds = sorted(set(C.parse_args_line(runs[(a, s)]["args"][0]).get("weight-decay-base", [None])[-1]
                     for a in ARMS for s in SEEDS if len(runs[(a, s)]["args"]) == 1))
    hchk(wds == sorted(set(D.WD_TOKEN[r] for r in RUNG_IDS)),
         "G-WDPATH the batch carries EXACTLY the four registered weight-decay tokens and no other", str(wds))

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(" ".join(t for t in ln.split() if not t.startswith("PROBE_DIR=")) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),
         "G-ENV one distinct ENV line (PROBE_DIR stripped), one per run", "%d distinct" % len(envs))

        print("        %s" % ln)
    hchk(envs == [ENV_EXPECTED], "G-ENV the ENV line is cvt1's ... cvt9's / cwd1's / cwd3's / cwd4's, byte for byte")

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == the frozen manifest, byte for byte")
    mp, how = resolve_manifest(runsdir, manifest_path)
    if not mp:
        hchk(False, "G-STRUCT manifest present", how)
    else:
        txt = open(mp).read()
        print("  manifest: %s (%s)" % (mp, how))
        hchk(txt == D.manifest_text(B), "G-STRUCT byte-identical to cwd5_design.manifest_text(CWD5)",
             "%d vs %d bytes" % (len(txt), len(D.manifest_text(B))))

    print("\nG-PROV    PROVENANCE.txt")
    pp, phow = resolve(runsdir, "", "PROVENANCE.txt")
    prov = read_kv(pp)
    if prov is None:
        hchk(False, "G-PROV PROVENANCE.txt present", phow)
    else:
        print("  provenance: %s (%s)" % (pp, phow))
        me = file_sha(os.path.abspath(__file__))
        for key, want, lab in (("MODE", "submit", "MODE submit"),
                               ("BUILD_NETWORK_SHA256", B.BN_SHA, "build_network.py == cvt8's"),
                               ("HF_SHA256", B.HF_POST_SHA, "HF.py == cwd1's PATCH_DECAYMASK tree, UNCHANGED"),
                               ("HF_PRE_DECAYMASK_SHA256", B.HF_PARENT_SHA, "HF.py.pre_decaymask == cvt8's HF.py"),
                               ("RUNNER_SHA256", B.RUNNER_SHA, "runner == run_cifar_cwd1.sh, UNCHANGED"),
                               ("SCORER_SHA256", me, "SCORER == this file"),
                               ("DESIGN_SHA256", file_sha(D.__file__), "DESIGN == the cwd5_design.py this scorer loaded"),
                               ("CWD_DESIGN_SHA256", file_sha(W.__file__), "the reused cwd_design.py this scorer loaded"),
                               ("COMMON_SHA256", file_sha(C.__file__), "COMMON == the cwd_common.py this scorer loaded")):
            hchk(prov.get(key) == want, "G-PROV %s" % lab, "%s=%s" % (key, str(prov.get(key))[:16]))

    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH; tail slope = TEST OLS over epochs 80-99 (descriptive)")
    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)
    arm_t = dict((a, [runs[(a, s)]["train5"] for s in SEEDS]) for a in ARMS)
    for a in ARMS:
        v, t = arm_v[a], arm_t[a]
        sl = [C.tail_slope(runs[(a, s)]["eps"]) for s in SEEDS]
        print("  %-7s wd %-5s %-9s TEST %.4f  sd %s  range %.4f   TRAIN %.4f   tail slope %s pp/ep"

    T = dict((a, mean(arm_t[a])) for a in ARMS)
    hi = max(M.values())
    print("\nG-FLOOR / G-CEIL")
    hchk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE), "max %.4f" % hi)
    hchk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)

    if harness:
        final = "FINAL: HARNESS-UNSOUND | %d harness gate(s) failed: %s" % (len(harness), "; ".join(h.split()[0] for h in harness))
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)
        for h in harness:
            print("   FAILED  %s" % h)
        for ln in LICENSE["HARNESS-UNSOUND"]:

    print("\nG-BITE    SPLIT, as registered.  LADDER gate (the 8 unmasked arms) is HARD; CARRIER gate (%s) suppresses" % CAR)
    print("          only the carrier word.  Lion is recomputed on each arm's OWN group count (scalar 1, layerwise %d)." % NTENS)
    ladder_fail, carrier_fail = [], []
    nonfin = {}
    for s in SEEDS:
        for a in ARMS:
            recs = C.load_probe(runsdir, PREFIX, a, s)

                     ("  (%s)" % (dl.get("why") or dd.get("why"))) if (dl.get("why") or dd.get("why")) else ""))

    carrier_ok = not carrier_fail

    print("\nSIGMA     O2: max(frozen floor, in-batch) -- nothing from the corpus")
    ss = sum(sum((x - M[a]) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    which, sigma_used, cands = compose_sigma(sigma_in)
    for nm, v in cands:
        print("  %-20s %.6f" % (nm, v))

    print("  SIGMA_USED           %.6f  (= %s)" % (sigma_used, which))
    print("  SE_ARM_DIFF          %.6f  = SIGMA_USED * sqrt(2/%d)  (df_inbatch %d)" % (se, len(SEEDS), df))

    print("\nTHE LADDER (every contrast WITHIN its rung)")
    states = [rung_state(M, r) for r in RUNG_IDS]
    CP = {}
    for r, st in zip(RUNG_IDS, states):

                 " / ".join("%+.3f" % (runs[(L, s)]["plateau5"] - runs[(k, s)]["plateau5"]) for s in SEEDS)))

              " reference healthy iff layerwise >= %.1f" % (R50 * M[L], R50, GAP_BAR, REF_MIN))

          % (CP["G_W4"][0], CP["G_W4"][0] / se, D.WD_TOKEN["W4"]))
    print("  DESCRIPTIVE ladders (levels, not effects): scalar %s"
          % "  ".join("%s %.4f" % (D.WD_TOKEN[r], M[SCALAR_OF[r]]) for r in RUNG_IDS))

          % "  ".join("%s %.4f" % (D.WD_TOKEN[r], M[LAYER_OF[r]]) for r in RUNG_IDS))

    cw = carrier_state(M, states) if carrier_ok else "CAR-PATCH-NOT-VERIFIED"
    print("\nTHE CARRIER COMPANION at rung %s (wd %s), read ONLY if that rung is COLLAPSE" % (CARRIER_RUNG, D.WD_TOKEN[CARRIER_RUNG]))
    if not carrier_ok:
        print("  %s -- G-BITE's carrier gate failed; NO carrier number is read." % cw)
    elif states[RUNG_IDS.index(CARRIER_RUNG)] != "COLLAPSE":
        print("  %s -- rung %s is %s, so there is no collapse for the mask to remove.  %s %.4f is PRINTED as a level"
              % (cw, CARRIER_RUNG, states[RUNG_IDS.index(CARRIER_RUNG)], CAR, M[CAR]))
        print("  and NOTHING is read from it (cmo1's ISO-UNREADABLE at wd 0, CORRECTIONS 264).")
    else:
        k2, l2 = SCALAR_OF[CARRIER_RUNG], LAYER_OF[CARRIER_RUNG]
        CP["P_CARW2"] = (M[CAR] - M[k2], T[CAR] - T[k2])

def _e_records(arm, seed, ng=None, touch=False, add_bh=False, nodm=False, adddm=False, dmforge=None, n=N_RECORDS, k=None):
    rnd = random.Random(7700 + seed + 17 * ARMS.index(arm))
    ng = ng or NG[arm]
    out = []
    for kk in range(n):
        s = 100 * kk
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        z = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [S9.f32(max(S9.LO, min(S9.HI, -13.8 + 0.3 * math.sin(kk / 30.0 + g) - 0.004 * kk))) for g in range(ng)]
        beta = [S9.f32(lion_natural(bpre[g], mom[g], z[g])[0]) for g in range(ng)]
        if touch:
            beta[0] = S9.f32(beta[0] + 0.002)
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99, "beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [z]}
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[0], "bh_held": beta[0]})
        out.append(rec)
    kk_ = K_MASKED[arm] if k is None else k
    if (kk_ and not nodm) or adddm:
        out = C.add_dm_keys(out, kk_ or 3, forge=dmforge)
    return out

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None, env_break=None,
             spec_break=None, wd_break=None, dm_break=None, extra_break=None, manifest=None, prov=None,
             probe_break=None, probe_drop=None, gpu=None):
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(manifest if manifest is not None else D.manifest_text(B))
    _e_prov(d, prov)
    jid = 9770000
    pb = probe_break or {}
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = ("layerwise" if SPEC[a] == "scalar" else "scalar") if spec_break == (a, s) else None
            wd = wd_break[1] if wd_break and wd_break[0] == (a, s) else None
            dm = dm_break[1] if dm_break and dm_break[0] == (a, s) else None
            extra = extra_break[1] if extra_break and extra_break[0] == (a, s) else None
            g = (gpu or {}).get((a, s), "NVIDIA L4, 23034 MiB")
            _e_write(d, a, s, jid, L, env=env, spec=spec, wd=wd, dm=dm, extra=extra, gpu=g)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")

        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

    ap.add_argument("--csv", default=S9.DEFAULT_CSV)
    ap.add_argument("--manifest", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir is required")
    rc, _fin = score(a.runsdir, a.csv, a.manifest)
    raise SystemExit(rc)
