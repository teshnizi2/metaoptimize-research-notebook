# ---------------------------------------------------------------- arithmetic that is stable on every host --
def lsum(v):
    """Left-to-right float accumulation.  Bit-identical on every IEEE-754 host and on every CPython."""
    acc = 0.0
    for x in v:
        acc = acc + x
    return acc


def lmean(v):
    return lsum(v) / len(v) if v else None


def xmean(v):
    """Exactly-summed mean (math.fsum): the alternative a CPython >= 3.12 `sum()` can produce."""
    return math.fsum(v) / len(v) if v else None


def both(fmtstr, v_lr, v_ex, tag):
    """-> (printed_string, {acceptable renderings}).  Prints the left-to-right one; accepts either."""
    a = fmtstr % v_lr
    b = fmtstr % v_ex
    if a != b:
        TIES.append("%s: left-to-right %s vs exact %s" % (tag, a, b))
    return a, set([a, b])


def f32(x):
    return struct.unpack("f", struct.pack("f", x))[0]

def u_of(n, P):
    if n <= 0:
        return 0
    return min(n - 1, P) - max(0, n - 1 - P)


def v_of(n, P=P_BIG):
    return f32(min(HI, max(LO, B0 + MS * u_of(n, P))))


def lion_natural(beta_pre, mom_pre, z):
    a = B2 * mom_pre
    b = (1.0 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(LO, min(HI, beta_pre - MS * s)), tie


def sha256_of(path):
    h = hashlib.sha256()
    with open(path, "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


# ---------------------------------------------------------------- the architecture, re-typed --------------
def r18_tensors():
    """ResNet18 for CIFAR-100, re-typed from the architecture (3x3 stem, 4 stages of 2 basic blocks, 1x1 projection
    shortcut wherever the stage changes width or stride)."""
    t = [("conv1.weight", 64 * 3 * 3 * 3, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    cin = 64
    for li, w in enumerate([64, 128, 256, 512], 1):
        for b in [0, 1]:
            stride = 2 if (li > 1 and b == 0) else 1
            inp = cin if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * inp * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if stride != 1 or inp != w:
                t += [(pre + "shortcut.0.weight", w * inp, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        cin = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]

def all_digits(s):
    return len(s) > 0 and all(c in "0123456789" for c in s)


def parse_sets(spec, names):
    """No regex: a token is `a-b`, `a`, or a tensor name."""
    if not spec.startswith("sets:"):
        raise ValueError(spec)
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            lo_hi = tok.split("-")
            if len(lo_hi) == 2 and all_digits(lo_hi[0]) and all_digits(lo_hi[1]):
                idx += list(range(int(lo_hi[0]), int(lo_hi[1]) + 1))
            elif all_digits(tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm):
    if SPEC[arm] == "scalar":
        return [list(range(1, len(NAMES) + 1))]
    return parse_sets(SPEC[arm], NAMES)

def parse_epoch_line(ln):
    """`Epoch 18, Train Accuracy: 16.48 %, Test Accuracy: 16.83 %` -> (epoch, train, test) or None.  No regex."""
    if not ln.startswith("Epoch "):
        return None
    head, _sep, rest = ln.partition(",")
    ep = head[len("Epoch "):].strip()
    if not all_digits(ep):
        return None
    if "Train Accuracy:" not in rest or "Test Accuracy:" not in rest:
        return None
    tr_part, _s2, te_part = rest.partition("Test Accuracy:")
    tr = tr_part.split("Train Accuracy:")[1].split("%")[0].strip()
    te = te_part.split("%")[0].strip()
    try:
        return int(ep), float(tr), float(te)
    except ValueError:
        return None


def parse_out(path):
    txt = open(path, errors="replace").read()
    lines = txt.splitlines()
    eps = {}
    for ln in lines:
        p = parse_epoch_line(ln)
        if p:
            eps[p[0]] = (p[1], p[2])
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "bh": [ln for ln in lines if ln.startswith("BETA_HOLD")],
           "gh": [ln for ln in lines if ln.startswith("GROUP_HOLD")],
           "rh": [ln for ln in lines if ln.startswith("REST_HOLD")],
           "ch": [ln for ln in lines if ln.startswith("COMP_HOLD")],
           "wh": [ln for ln in lines if ln.startswith("WINDOW_HOLD")]}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    ok_tail = all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail)
    rec["plateau5"] = lmean([x[1] for x in tail]) if ok_tail else None
    rec["plateau5_x"] = xmean([x[1] for x in tail]) if ok_tail else None
    rec["train5"] = lmean([x[0] for x in tail]) if ok_tail else None
    rec["train5_x"] = xmean([x[0] for x in tail]) if ok_tail else None
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"] and not rec["traceback"]
                       and rec["plateau5"] is not None)
    return rec


def parse_args_line(ln):
    toks = ln[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def strip_probe_dir(env_line):
    """Drop the ` PROBE_DIR=<token>` field.  No regex."""
    out = [t for t in env_line.split(" ") if not t.startswith("PROBE_DIR=")]
    return " ".join(out)


def name_of_out(fn):
    """cvt10-<arm>-s<seed>-<jobid>.out -> (arm, seed, jobid) or None.  No regex."""
    if not fn.startswith(PREFIX + "-") or not fn.endswith(".out"):
        return None
    core = fn[len(PREFIX) + 1:-4]
    parts = core.split("-")
    if len(parts) != 3:
        return None
    arm, sseed, sjid = parts
    if arm not in ARMS or not sseed.startswith("s") or not all_digits(sseed[1:]) or not all_digits(sjid):
        return None
    return arm, int(sseed[1:]), int(sjid)


def load_probe(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    recs.append(json.loads(ln))
                except ValueError:
                    recs.append(None)
            return recs
    return None


def read_kv(path):
    d = {}
    if not path or not os.path.exists(path):
        return None
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d

def state_of(L, M):
    if L >= M["ISO"] - MATCH_BAR:
        return "AT-ISO"
    if L <= M["k01"] + NULL_BAR:
        return "AT-K01"
    return "BETWEEN"


def single_word(M, s):
    free, big, pfx = "ONE%d" % s, "ONE%dBIG" % s, "ONE%d" % s
    if M[free] < M["k01"] + READ_MARGIN:
        return pfx + "-UNREADABLE"
    if M[big] <= M["k01"] + NULL_BAR:
        return pfx + "-STALLS"
    if M[big] >= M[free] - MATCH_BAR:
        return pfx + "-NO-EFFECT"
    return pfx + "-PARTIAL"


def split_word(M):
    if M["ISOSPLIT"] <= M["k01"] + NULL_BAR:
        return "SPLIT-STALLS"
    if M["ISOSPLIT"] >= M["ISO"] - MATCH_BAR:
        return "SPLIT-NO-EFFECT"
    return "SPLIT-PARTIAL"


def summary_word(w50, w59, w53):
    if any(w.endswith("-STALLS") for w in (w50, w59, w53)):
        return "ONE-SUFFICES"
    if not w50.endswith("-UNREADABLE") and not w59.endswith("-UNREADABLE"):
        return "NO-ONE-SUFFICES"
    return "ONE-UNRESOLVED"


def decide(M, diverged):
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:

def groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    try:
        return [float(x) for x in v[0]]
    except (TypeError, ValueError):
        return None


def bite(arm, recs):
    ng = len(arm_groups(arm))
    held = arm in HELD
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "bad_sched": 0,
         "bad_sched_pre": 0, "bad_gh": 0, "keys_wrong": 0, "active_records": 0, "gh_active_last": None,
         "worst_lion": 0.0, "worst_sched": 0.0, "ng": ng}
    if recs is None:
        return False, d
    prev = -1
    for k, r in enumerate(recs):
        if (not isinstance(r, dict) or not isinstance(r.get("step"), int)
                or not isinstance(r.get("beta"), list) or len(r["beta"]) != ng):
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        bpre, mom, z = groupvec(r, "beta_pre", ng), groupvec(r, "mom_pre", ng), groupvec(r, "z_agg", ng)
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        s = r["step"]
        if s != PROBE * k:
            d["bad_step"] += 1
        d["n"] += 1
        n = s + 2
        for g in range(ng):
            if held and g == 1:
                continue
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst_lion"] = max(d["worst_lion"], e)
            d["bad_lion"] += e > BH_TOL
        d["keys_wrong"] += any(key in r for key in FOREIGN_KEYS)
        has_gh = any(key in r for key in GH_KEYS)
        if not held:
            d["keys_wrong"] += has_gh
            continue
        es, ep = abs(beta[1] - v_of(n)), abs(bpre[1] - v_of(n - 1))
        d["worst_sched"] = max(d["worst_sched"], es, ep)
        d["bad_sched"] += es > BH_TOL
        d["bad_sched_pre"] += ep > BH_TOL
        try:
            cn, ca, cnat, cheld = int(r["gh_n"]), int(r["gh_active"]), float(r["gh_nat"]), float(r["gh_held"])
        except (KeyError, TypeError, ValueError):
            d["bad_gh"] += 1
            continue
        nat1, tie1 = lion_natural(bpre[1], mom[1], z[1])
        ok1 = (cn == n and abs(cheld - beta[1]) <= BH_TOL and 0 <= ca <= cn and ca >= prev
               and (tie1 or abs(cnat - nat1) <= BH_TOL))
        d["bad_gh"] += not ok1
        d["active_records"] += abs(cnat - cheld) > BH_TOL
        prev = max(prev, ca)
        d["gh_active_last"] = ca
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_step"] == 0 and d["bad_lion"] == 0 and d["keys_wrong"] == 0)
    if held:
        ok = (ok and d["bad_sched"] == 0 and d["bad_sched_pre"] == 0 and d["bad_gh"] == 0
              and (d["gh_active_last"] is None or d["gh_active_last"] >= d["active_records"]))
    return ok, d


def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if (isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord
                and isinstance(r.get("step"), int)):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    kpk = max(range(len(vals)), key=lambda i: vals[i][1])
    pin = None
    for k in range(kpk, len(vals)):
        if all(math.exp(v + 15.0) <= 2.0 for _s, v in vals[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    win = [math.exp(v + 15.0) for s, v in vals if 15 <= s / float(STEPS_PER_EPOCH) < 30]
    win2 = [math.exp(v + 15.0) for s, v in vals if 30 <= s / float(STEPS_PER_EPOCH) < 60]
    return {"peak": vals[kpk][1], "peak_epoch": vals[kpk][0] / float(STEPS_PER_EPOCH), "pin_epoch": pin,
            "r1530": lmean(win), "r3060": lmean(win2),
            "r1530x": xmean(win), "r3060x": xmean(win2)}


def first_stays(gap, pred):
    for e in range(EPOCHS):
        if all(pred(gap[f]) for f in range(e, EPOCHS)):
            return e
    return None
