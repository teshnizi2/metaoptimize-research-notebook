set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
CLIP="BETA_CLIP=-15:-2.3026"
COM="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"
LION="--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9"
ADAM="--alg-meta Adam --normalizer-param-meta 0.999 --momentum-param-meta 0.9"

sub () { # 1=name 2=granularity 3=seed 4=metacfg
  sbatch --parsable --job-name=$1-s$3 --partition=gpu-l4-24g --gres=gpu:l4:1 --time=03:00:00 \
    --export="ALL,AUGMENT=1,$CLIP" \
    jobs/run_cifar.sh $COM $4 --stepsize-groups $2 --seed $3 --save-directory $R/guarded --run-name $1_s$3 >/dev/null
}

for s in 0 1 2; do
  # A. SGDm+Lion guarded -- completes the guarded headline grid (layerwise/weightwise exist as d4-clip*)
  sub g4-sgdmLion-scal  scalar          $s "$LION"
  sub g4-sgdmLion-blk6  resnet18_blocks $s "$LION"
  # B. SGDm+Adam guarded -- CONTROL for the "granularity rescues diverging seeds" claim.
  #    Unguarded, scalar collapsed on 2 of 3 seeds (88.45 / 18.47 / 20.71). If the GUARD
  #    alone rescues those seeds, the stability claim is confounded and must be rewritten.
  sub g4-sgdmAdam-scal  scalar          $s "$ADAM"
  sub g4-sgdmAdam-blk6  resnet18_blocks $s "$ADAM"
  sub g4-sgdmAdam-layer layerwise       $s "$ADAM"
done
sleep 3
echo "G4_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -c '^g4')"
echo "TOTAL=$(squeue -h -u account_b | wc -l)"
