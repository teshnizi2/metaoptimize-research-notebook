# =============================================================================

  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then

  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^csh1-')

          "$NJOBS" "$WALL" "$MST" "$A0" "$DSET" "$EPOCHS" "$BATCH" "$CLIP_C" "$PROBE" "$WDB" "$TRAIN_SHA" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"

# ---- guard 4: THE LIVE MODEL AND THE LIVE OPTIMISER, per arm, from cwd1's tree (CPU construction only) --------------

python3 - "$HARN" "$MTMP" "$DES" <<'PYEOF' || guard_fail "guard 4: live model / optimiser / witness / manifest check FAILED"
import sys, os, io, ast, types, argparse, contextlib, importlib.util, tempfile, shutil
