#!/usr/bin/env bash
# =============================================================================
# c87_an0_analysis.sh -- `an0`, THE THREE ANALYSIS-ONLY FIXES.
# ZERO SLURM JOBS.  ZERO GPU-HOURS.  ZERO SEEDS.  ZERO EPOCHS.
#
# This file is shaped exactly like a submit script -- the same guard ladder, the
# same emitter region, the same refusal semantics -- because the guards are what
# make a result quotable, and an0's product (two CHANGED PUBLISHED NUMBERS and a
# figure) needs them at least as much as a GPU batch does.  The one difference is
# the one that matters: **THE EMITTER CONTAINS NO sbatch AND GUARD 9 PROVES IT.**
#
# -----------------------------------------------------------------------------
# WHY THIS RUNS BEFORE ANY FURTHER SUBMISSION
# -----------------------------------------------------------------------------
# an0 costs nothing and two of its three fixes change numbers that are already
# written down.  Submitting a new batch on top of a headline that inverts when
# you put it on a commensurable scale, or on top of a count correction imported
# from a slope the record cannot agree with itself about, spends GPU-hours
# defending a sentence that is going to be rewritten anyway.
#
#   F1  COMMENSURABILITY.  Every D in the corpus is quoted in percentage POINTS
#       and compared across cells whose base error budgets differ by 3.7x.  The
#       campaign registered the fix for this -- gn1's T0.6 commensurability gate
#       and the D/(100-level) print, CORRECTIONS 114.2(e), with the explicit
#       reasoning that "a 1.6x change in the scale the effect lives on ... if the
#       gap is even partly multiplicative distorts D_GN by ~60%" -- and then did
#       not apply it to its own cross-dataset headline.  an0 applies it to the
#       headline.  If the argmax cell moves between scales, the sentence naming a
#       largest D without naming a scale is DELETED.
#
#   F2  THE PRACTITIONER'S NUMBER.  What a reader applies is T = nodewise1d -
#       nodewise, "merge the normalisation scalars".  T is count-confounded by
#       construction, and the record corrects it with a slope IMPORTED from the
#       ck1 chunk ladder -- a different batch, a different count range, and rungs
#       that SPLIT the very 1-D tensors whose merging is the prescription.  The
#       correction does not need an import at all:
#             T = (D - G) + U,   U = chunk_coarse - chunk_fine
#       is an EXACT ALGEBRAIC IDENTITY in which every term is a within-batch
#       contrast the corpus already owns.  U is the in-batch estimate of exactly
#       the quantity the slope was imported for.  GUARD 4 measures every m in
#       that identity off the ALLOCATED beta on the LIVE network.
#
#   F3  THE HORIZON TRAJECTORY.  D(t) and (D-G)(t) are computable TODAY from the
#       .out epoch series on disk.  A referee holding the artefact computes them
#       in an afternoon.  The paper publishes the trajectory itself, per cell,
#       whatever a longer-horizon batch may later say.
#
# -----------------------------------------------------------------------------
# **RULE 19 IS NOT SATISFIED BY an0 AND an0 DOES NOT CLAIM IT.**
# -----------------------------------------------------------------------------
# STANDING RULE 19 requires the scorer committed BEFORE the runs exist.  Every
# run an0 reads was submitted, completed and scored long before this file was
# written.  Pretending otherwise would be precisely the record-hygiene failure
# the batch exists to fix.  What IS frozen before the scorer's first run: the
# metrics, the decision rules, the algebraic identity, and the trajectory grid.
# What is NOT frozen: the data.  Every an0 verdict is therefore labelled POST-HOC
# and is a CORRECTION, never a test.  The scorer prints that disclosure at the
# top of its own output and its selftest asserts the disclosure is present.
#
# -----------------------------------------------------------------------------
# WHAT an0 MAY NOT CLAIM.  Registered here.
# -----------------------------------------------------------------------------
#   * NOT a pre-registered test of anything.  See the disclosure above.
#   * NOT a RULE-11 result on any cell.  ms is untuned on ResNet34 and on
#     CIFAR-100 and every verdict line says so.
#   * NOT a new effect.  an0 re-expresses numbers the corpus already owns.
#   * NO POOLING across architectures or datasets.  F1 makes cells COMPARABLE;
#     comparable is not poolable.
#   * NO FIELD STATISTIC.  Direction C was dropped on its own pre-registered
#     adoption gate; the scorer's selftest asserts the absence over its own AST.
#   * ar1 IS GUARD-SUSPECT and the scorer says so per seed rather than quietly
#     dropping it: the box gate runs FIRST and prints occupancy at BOTH
#     denominators beside every number.
#
# STANDING RULES HONOURED: 10 (the arm is held fixed inside every contrast), 13
# (the two load-bearing guards are tested in BOTH directions -- run --rule13), 14
# (every PRIMARY contrast is WITHIN one batch; the one cross-batch fit in the
# file, the ck1/cx2 ladder, is labelled DESCRIPTIVE and carries the floor in its
# own line).  RULE 19 is DISCLOSED AS UNMET, above.
# =============================================================================

  GUARD_FAILS=$((GUARD_FAILS + 1))

    exit 1

  echo "     an0's OUTPUT MUST NOT BE QUOTED UNTIL THIS GUARD PASSES.)"
}

# --- GUARD 1b -- **THE ENV/ARGS PIN.  READ OFF EVERY .out HEADER ON DISK.** --
# For a GPU batch this guard pins the sbatch --export line.  an0 submits nothing,
# so the analogue -- and the stronger check -- is the SELECTION FILTER: every run
# an0 consumes must actually have run the configuration an0 attributes to it.
# The .out ARGS:/ENV: header is the only instrument that can prove that; the CSV
# is a derived artefact, and an unidentified schedule has already made one batch
# of this corpus unreadable (CLOSEOUT 5.1, the I1 family).
python3 - "$SCORER" "$RAW" "$RAW2" <<'PYEOF' || guard_fail "guard 1b: ENV/ARGS pin"
import os, sys, importlib.util
scorer, raw, raw2 = sys.argv[1:4]
spec = importlib.util.spec_from_file_location("an0", scorer)
S = importlib.util.module_from_spec(spec); spec.loader.exec_module(S)
roots = [d for d in (raw, raw2) if os.path.isdir(d)]
PINS_ARGS = ["optimizer", "alg-base", "alg-meta", "dataset", "NN-name",
             "batch-size", "meta-stepsize", "alpha0", "num-epochs",
             "stepsize-groups", "gamma", "seed"]
bad, n = [], 0
for fam, sp in S.CELLS.items():
    for short, gran in sp["arms"].items():
        for s in sp["seeds"]:
            f = S.find_out(roots, fam, short, s)
            if f is None: continue
            a, e = S.header_pins(f)
            n += 1
            miss = [k for k in PINS_ARGS if k not in a]
            if miss:
                bad.append("%s-%s-s%d: ARGS header missing %s" % (fam, short, s, miss)); continue
            for key, want in (("dataset", sp["dset"]), ("NN-name", sp["net"]),
                              ("meta-stepsize", sp["ms"]), ("alpha0", sp["alpha0"]),
                              ("num-epochs", str(sp["epochs"])),
                              ("stepsize-groups", gran), ("seed", str(s)),
                              ("optimizer", "HF"), ("alg-base", "SGDm"),
                              ("alg-meta", "Lion"), ("gamma", "1"),
                              ("batch-size", "100")):
                if a.get(key) != want:
                    bad.append("%s-%s-s%d: --%s is %r, registered %r"
                               % (fam, short, s, key, a.get(key), want))
            for key, want in (("BETA_CLIP", sp["box"]), ("AUGMENT", "1"),
                              ("HIER", "none"), ("SCHED", "none")):
                if e.get(key) != want:
                    bad.append("%s-%s-s%d: ENV %s is %r, registered %r"
                               % (fam, short, s, key, e.get(key), want))
            # the forbidden ones: a stray schedule or pooling dial silently turns
            # a headline cell into a different experiment (the dc-aw-* failure).
            for k in ("SCHED_TOTAL", "SCHED_WARMUP", "SCHED_MIN"):
                if e.get(k) not in (None, "none"):
                    bad.append("%s-%s-s%d: ENV %s=%r -- AN UNIDENTIFIED SCHEDULE"
                               % (fam, short, s, k, e.get(k)))
            for k in ("LAM", "ETA_RATIO"):
                if e.get(k) not in (None, "na", "none", "0"):
                    bad.append("%s-%s-s%d: ENV %s=%r -- a pooling dial is set"
                               % (fam, short, s, k, e.get(k)))
print("guard 1b: %d .out headers parsed; %d ARGS pins and 4 ENV pins checked on each"
      % (n, len(PINS_ARGS)))
if bad:
    sys.exit("GUARD FAIL: %d header pin violation(s):\n  %s"
             % (len(bad), "\n  ".join(bad[:12])))
print("guard 1b: EVERY run an0 consumes actually ran the configuration an0 "
      "attributes to it -- net, dataset, ms, alpha0, box, epochs, granularity, "
      "seed, base/meta optimiser, AUGMENT, and NO schedule and NO pooling dial.")
PYEOF

# --- GUARD 2 -- CSV PRECONDITIONS, INCLUDING A SECOND INSTRUMENT ON plateau5 -
python3 - "$CSVP" "$SCORER" "$RAW" "$RAW2" <<'PYEOF' || guard_fail "guard 2: CSV preconditions"
import csv, os, sys, math, importlib.util
csvp, scorer, raw, raw2 = sys.argv[1:5]
spec = importlib.util.spec_from_file_location("an0", scorer)
S = importlib.util.module_from_spec(spec); spec.loader.exec_module(S)
roots = [d for d in (raw, raw2) if os.path.isdir(d)]

# 2a -- THE SCHEMA NOTE.  `plateau` is mean-of-last-20 and is NOT comparable
# across budgets; `plateau5` is the PRIMARY.  Confusing the two has already
# produced one 22.8 pp error in this project (CLOSEOUT item 3a).
cols = R[0].keys()
for c in ("plateau", "plateau5", "epochs_done", "epochs_requested", "collapsed"):
    if c not in cols:
        sys.exit("GUARD FAIL: results/all_runs.csv has no %r column" % c)
print("guard 2a: CSV carries BOTH `plateau` (mean-of-last-20, NOT comparable "
      "across budgets) and `plateau5` (PRIMARY).  an0 reads NEITHER as its "
      "primary: plateau5 is RE-DERIVED from the .out series and the CSV column "
      "is used only as the SECOND INSTRUMENT below.")

# 2b -- SECOND INSTRUMENT.  Every run an0 reads: .out-derived plateau5 must
# agree with the CSV column.  A disagreement means one of them is stale.
worst, n, miss = 0.0, 0, []
for fam, sp in S.CELLS.items():
    for short in sp["arms"]:
        for s in sp["seeds"]:
            f = S.find_out(roots, fam, short, s)
            if f is None: continue
            ser = S.series_from_out(f)
            p5 = S.window_mean(ser, len(ser))
            rn = "%s-%s-s%d" % (fam, short, s)
            row = by_run.get(rn)
            if row is None or not row.get("plateau5"):
                miss.append(rn); continue
            d = abs(p5 - float(row["plateau5"]))
            worst = max(worst, d); n += 1
print("guard 2b: %d runs cross-checked .out-derived plateau5 vs the CSV column; "
      "worst |difference| = %.6f pp" % (n, worst))
if miss:
    print("guard 2b: %d run(s) have no plateau5 in the CSV: %s" % (len(miss), miss[:6]))
if n < 100:
    sys.exit("GUARD FAIL: fewer than 100 runs cross-checked (%d) -- the second "
             "instrument is not exercised" % n)
if worst > 1e-3:
    sys.exit("GUARD FAIL: the CSV and the .out disagree by up to %.6f pp on "
             "plateau5.  One of the two is stale and an0 must not run until it is "
             "known which." % worst)

# 2c -- THE COMPLETENESS PREMISE.  an0 reads only complete, non-collapsed,
# non-superseded runs, at the budget its cells register.
bad = []
for fam, sp in S.CELLS.items():
    for short in sp["arms"]:
        for s in sp["seeds"]:
            rn = "%s-%s-s%d" % (fam, short, s)
            r = by_run.get(rn)
            if r is None: continue
            if r["collapsed"] != "0" or r["superseded"] not in ("0", ""):
                bad.append("%s collapsed=%s superseded=%s" % (rn, r["collapsed"], r["superseded"]))
            if r["epochs_done"] != str(sp["epochs"]):
                bad.append("%s epochs_done=%s, registered %d" % (rn, r["epochs_done"], sp["epochs"]))
if bad:
    sys.exit("GUARD FAIL: %d run(s) fail the completeness premise:\n  %s"
             % (len(bad), "\n  ".join(bad[:10])))
print("guard 2c: every run an0 reads is complete, non-collapsed, non-superseded, "
      "and at its registered budget")

# 2d -- THE SLOPE THE RECORD CANNOT AGREE WITH ITSELF ABOUT.  This is a
# RECORD-HYGIENE guard: it asserts the defect F2 exists to fix is still there,
# so that a later reader can tell whether an0 has already been applied.
vals = sorted(S.SLOPE_RECORD.values())
sp_ = vals[-1] - vals[0]
print("guard 2d: the record carries %d values for ONE count slope, spread %.4f "
      "pp/decade: %s" % (len(vals), sp_,
                         ", ".join("%+0.4f" % v for v in vals)))
if sp_ <= 0.10:
    print("guard 2d: the spread is already inside the bar -- F2's slope leg may "
          "have been applied already.  CHECK THE RECORD BEFORE RE-RUNNING.")

# 2e -- THE LADDER RUNGS SPLIT THE TAIL.  A structural fact, asserted here so it
# lands in the log beside the slope it discredits.
mx = S.MAX_1D["ResNet18"]
split = [g for g in list(S.LADDER["ck1"].values()) + list(S.LADDER["cx2"].values())
         if int(g.replace("chunk", "")) <= mx]
keep = [g for g in list(S.LADDER["ck1"].values()) + list(S.LADDER["cx2"].values())
        if int(g.replace("chunk", "")) > mx]
print("guard 2e: of the %d ladder rungs, %d use K <= %d and therefore SPLIT the "
      "1-D tensors whose merging is the prescription (%s); %d do not (%s).  The "
      "ladder is a MIXED instrument and its slope is the wrong tool for a count "
      "correction between two TAIL-PRESERVING partitions."
      % (len(split) + len(keep), len(split), mx, ", ".join(split), len(keep),
         ", ".join(keep)))

# 2g -- NAME COLLISION.  an0 must never create a run row, now or ever.
clash = [r["run"] for r in R if r["run"].startswith("an0-")]
if clash:
    sys.exit("GUARD FAIL: %d an0-* run rows already exist in the CSV: %s.  an0 "
             "emits ZERO jobs and must therefore own ZERO run rows -- their "
             "existence means something submitted under this tag."
             % (len(clash), clash[:5]))
print("guard 2g: no an0-* run row exists in the CSV (%d rows scanned), and none "
      "ever may: an0 emits no jobs (GUARD 9)." % len(R))
PYEOF

# --- GUARD 3 -- EVERY BOX an0 READS IS REGISTERED, AND MATCHES THE DISK ------
python3 - "$REPO" "$SCORER" "$RAW" "$RAW2" <<'PYEOF' || guard_fail "guard 3: box registration"
import os, re, sys, importlib.util
repo, scorer, raw, raw2 = sys.argv[1:5]
spec = importlib.util.spec_from_file_location("an0", scorer)
S = importlib.util.module_from_spec(spec); spec.loader.exec_module(S)
src = open(os.path.join(repo, "analysis", "c55_neff_noise.py")).read()
roots = [d for d in (raw, raw2) if os.path.isdir(d)]
fams = list(S.CELLS) + list(S.LADDER)
for fam in fams:
    m = re.search(r'"%s":\s*\(([-0-9.eE]+),\s*([-0-9.eE]+),' % re.escape(fam), src)
    if not m:
        sys.exit("GUARD FAIL: `%s` is not registered in c55_neff_noise.BOXES.  A "
                 "root with no registered box is REFUSED, never guessed." % fam)
    rlo, rhi = float(m.group(1)), float(m.group(2))
    if fam in S.CELLS:
        lo, hi = (float(x) for x in S.CELLS[fam]["box"].split(":"))
        if abs(rlo - lo) > 1e-9 or abs(rhi - hi) > 1e-9:
            sys.exit("GUARD FAIL: c55 registers %s as (%s, %s); the scorer uses (%s, %s)"
                     % (fam, rlo, rhi, lo, hi))
        # and the DISK must agree with both
        sp = S.CELLS[fam]
        for short in sp["arms"]:

# filesystem a single micro-benchmark FLAPS: the same corpus projected 980 s and

# the guard's own verdict flipped without the corpus changing.  A guard that flaps
# is not a guard.
def rate(files, fn, n=3):

    for f in sorted(files, key=lambda x: -x[-1])[:n]:
        t0 = time.time()
        fn(f)
        dt = time.time() - t0
        rs.append(dt / (f[-1] / 1e6))          # seconds per MB
    return sorted(rs)[len(rs) // 2], max(rs), rs
