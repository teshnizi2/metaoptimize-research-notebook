def sha_file(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()

ARG_RE = re.compile(r"--([A-Za-z0-9-]+) (\S+)")


def find_out(run):
    """The .out for a run name: <run>-<job>.out in RUNS_A2 or RUNS_1 root; the highest-job COMPLETE one."""
    cands = []
    for root in (RUNS_A2, RUNS_1):
        for d in (root, os.path.join(root, run.split("-")[0])):
            if not os.path.isdir(d):
                continue
            for fn in os.listdir(d):
                if fn.startswith(run + "-") and fn.endswith(".out") and fn[len(run) + 1:-4].isdigit():
                    cands.append(os.path.join(d, fn))
    if not cands:
        return None
    parsed = [(core.parse_out(p), p) for p in cands]
    comp = [x for x in parsed if x[0]["complete"]]
    pick = max(comp or parsed, key=lambda x: int(os.path.basename(x[1])[len(run) + 1:-4]))
    return pick


def run_info(run, pdir):
    got = find_out(run)
    info = {"run": run, "batch": run.split("-")[0], "pdir": pdir, "out": None}
    m = re.match(r"^(.*)-s(\d+)$", run[len(info["batch"]) + 1:])
    info["arm"], info["seed"] = (m.group(1), int(m.group(2))) if m else (run, -1)
    if got is None:
        return info
    rec, path = got
    info["out"] = path
    args = dict(ARG_RE.findall(rec["args"][0])) if rec["args"] else {}
    info.update({"wd": float(args.get("weight-decay-base", "nan")), "wd_tok": args.get("weight-decay-base"),
                 "mom": args.get("momentum-param-base"), "base": args.get("alg-base"), "meta": args.get("alg-meta"),
                 "net": args.get("NN-name"), "dset": args.get("dataset"), "ms": args.get("meta-stepsize"),
                 "a0": args.get("alpha0"), "grain": args.get("stepsize-groups"),
                 "bs": int(args.get("batch-size", "0") or 0), "p5": rec["p5"], "t5": rec["t5"],
                 "complete": rec["complete"], "n_ep": rec["n_ep"]})
    info["spe"] = TRAIN_SET.get(info["dset"], 0) // info["bs"] if info["bs"] else None
    lines = open(path, errors="replace").read().splitlines()
    fl = [l for l in lines if l.startswith(FOREIGN)]
    info["foreign_on"] = sorted({l.split(":")[0] for l in fl if re.search(r":\s*on\b", l)})
    info["mask_idx"] = []
    for l in fl:
        if l.startswith("DECAY_MASK") and re.search(r":\s*on\b", l):
            mi = re.search(r"\bidx=(\S+)", l)
            if mi:
                info["mask_idx"] = [int(x) for x in mi.group(1).split(",") if x.isdigit()]
    return info

def selftest(cache):
    print("=" * 110)
    print("SELFTEST -- the registered carrier-vote statistic, re-derived through cCV0 on the committed records")
    print("=" * 110)
    ok = True
    calib = [("ctd1", "sc", (18, 19, 20)), ("ciso1", "k01", (21, 22, 23)), ("cdep1", "k01", (24, 25, 26))]
    want = [0.820, 0.784, 0.810, 0.806, 0.786, 0.806, 0.814, 0.816, 0.820]
    got, pooled = [], []
    for b, a, seeds in calib:
        for s in seeds:
            recs = cache["%s-%s-s%d" % (b, a, s)]
            got.append(core.vote_stats(recs)["DOM_C"])
            pooled += recs
    pv = core.vote_stats(pooled)["DOM_C"]
    c1 = all(abs(g - w) < 5e-4 for g, w in zip(got, want)) and abs(pv - 0.8069) < 5e-5
    print("  %s 256.3 calibration DOM_C per run %s pooled %.4f (registered 0.820/.../0.820, pooled 0.8069)"
          % ("PASS" if c1 else "FAIL", " ".join("%.3f" % g for g in got), pv))
    ct = []
    for s in (116, 117, 118):
        ct += cache["cct1-k01-s%d" % s]
    st = core.vote_stats(ct)
    c2 = st["DOM_C"] == 0.0 and abs(st["TOP3_C"] - 0.6940) < 5e-5
    print("  %s 265 cct1 k01 pooled DOM_C %.4f TOP3_C %.4f (registered 0.0000 / 0.6940)"
          % ("PASS" if c2 else "FAIL", st["DOM_C"], st["TOP3_C"]))
    # N_eff arithmetic on synthetic vectors
    c3 = abs(neff([1.0] * 62) - 62.0) < 1e-9 and abs(neff([1.0] + [0.0] * 61) - 1.0) < 1e-12 \
        and abs(neff([1.0, 1.0, 0.0]) - 2.0) < 1e-12
    print("  %s N_eff(uniform 62)=62, N_eff(one-hot)=1, N_eff(two equal)=2" % ("PASS" if c3 else "FAIL"))
    ok = c1 and c2 and c3
    print("  SELFTEST %s" % ("PASS" if ok else "FAIL"))
    return ok


def neff(v):
    a = [abs(x) for x in v]
    t = sum(a)
    if t <= 0:
        return None
    return 1.0 / sum((x / t) ** 2 for x in a)

def shrink_rows(i, recs):
    """per record: (epoch, s_max, s_med, s_car) with s = wd*exp(beta_g) over UNMASKED coordinates.
    Masked tensors get 0 (reported separately); for scalar runs every unmasked tensor gets the one value."""
    spe = i["spe"]
    wd = i["wd"]
    out = []
    for r in recs:
        b = r.get("beta")
        if not b:
            continue
        a = [math.exp(x) for x in b]
        s = [wd * x for x in a]
        car = None
        if len(b) == 62:
            car = statistics.mean(s[k] for k in core.C_IDX0)
        out.append((r["step"] // spe, max(s), statistics.median(s), car, a))
    return out

def dom_series(recs):
    out = []
    for r in recs:
        try:
            L, b2 = core.record_L(r)
            s = core.shared_sign(r, L, b2)
        except (KeyError, IndexError, TypeError, ValueError):
            out.append(None)
            continue
        if len(L) != 62 or s == 0:
            out.append(None)
            continue
        out.append(1 if core.fixes_sign(L, core.C_IDX0, s) else 0)
    return out


def first_window(dom, w, mode, bar=0.5):
    n = len(dom)
    h = w // 2
    for k in range(n):
        if mode == "centred":
            lo, hi = max(0, k - h), min(n, k + h + 1)
        elif mode == "forward":
            lo, hi = k, min(n, k + w)
        else:
            lo, hi = max(0, k - w + 1), k + 1
        v = [x for x in dom[lo:hi] if x is not None]
        if v and sum(v) / len(v) >= bar:
            return k
    return None


def beta_events(i, recs, ms):
    b = [r["beta"][0] for r in recs]
    steps = [r["step"] for r in recs]
    b0 = recs[0]["beta_pre"][0][0] if recs[0].get("beta_pre") else b[0] - ms
    dev = [b0 + ms * (steps[k] + 1) - b[k] for k in range(len(b))]
    ev = {}
    for d in (0.02, 0.1):
        ev["exit%.2f" % d] = next((k for k in range(len(b)) if dev[k] > d), None)
    ev["peak"] = max(range(len(b)), key=lambda k: b[k])
    pk = ev["peak"]
    ev["desc1"] = next((k for k in range(pk, len(b)) if b[k] <= b[pk] - 1.0), None)
    ev["floor"] = next((k for k in range(len(b)) if b[k] <= BETA_LO + 1e-3), None)
    ev["noise_pre"] = max(abs(dev[k]) for k in range(min(40, len(dev))))
    return ev, b, steps

def neff_rows(i, recs):
    spe = i["spe"]
    out = []
    for r in recs:
        if "z_tensor" not in r or "m_tensor" not in r:
            continue
        b2 = float(r.get("pt_b2", core.B2_REG))
        L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
        out.append((r["step"] / spe, neff(L), neff(r["z_tensor"])))
    return out

def best_threshold(pts):
    xs = sorted({p[0] for p in pts})
    if not xs:
        return "n/a"
    best = None
    cands = [xs[0] - 1e-9] + [(xs[k] + xs[k + 1]) / 2 for k in range(len(xs) - 1)] + [xs[-1] + 1e-9]
    for t in cands:
        for dirn in (1, -1):  # 1: collapsed iff N_eff < t
            err = sum(1 for p in pts if ((p[0] < t) if dirn == 1 else (p[0] > t)) != p[1])
            if best is None or err < best[0]:
                best = (err, t, dirn)
    return "%d/%d misclassified at N_eff %s %.2f => COLLAPSED" % (best[0], len(pts), "<" if best[2] == 1 else ">",
                                                                  best[1])
