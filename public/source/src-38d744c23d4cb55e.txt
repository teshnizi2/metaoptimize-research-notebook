# Cycle 30: ResNet34 FINE r-grid below the current peak.  CIFAR-10, layerwise, a0=1e-6, 100ep.
#
# OPEN QUESTION THIS ANSWERS (one question, 16 jobs):
#   Where is r* on ResNet34, and does the pooling MASS r*.m stay constant across model scale?
#
# WHAT THE COMPLETE 100-EPOCH DATA NOW SAYS (all_runs.csv, M1 additive, layerwise, CIFAR-10,
# a0=1e-6, epochs_done==100, not collapsed -- plateau = mean of last 5 epochs):
#   ResNet10  m=38   r*=0.10   82.114 -> 91.590   (+9.476, but r=0 is a near-collapse: DO NOT
#                                                  quote this gain beside the other two)
#   ResNet18  m=62   r*=0.06   92.587 -> 93.222   (+0.636, n=8/10)
#   ResNet34  m=110  r*=0.02   93.098 -> 93.884   (+0.786, n=3/3, t=6.6)
#   => r* FALLS with model size: 0.10 -> 0.06 -> 0.02.  The GAIN does not vanish at scale
#      (R34 >= R18), which is the point the parent paper's premise turns on.
#
# THE GAP.  R34's grid has NO point below 0.02, so r* is unresolved downward.  And 0.02 vs
# 0.03 is NOT separable (93.884 +-0.194 vs 93.813 +-0.100, n=3 each, t=0.56) -- the peak is
# a plateau over [0.02,0.03], not a located point.  r=0.04 IS down (t=3.66).
#
# PRE-REGISTERED.
#   r*.m is 3.8 (R10), 3.7 (R18), 2.75 (R34 at r=0.025).  A constant-mass law predicts
#   r*(R34) ~ 0.034 -- already strained, since 0.03 does not beat 0.02.
#   PREDICTED: the peak stays in [0.015,0.03] and r*.m keeps DECLINING with scale
#              (i.e. total pooling mass is NOT scale-invariant; large models want
#              proportionally less pooling than a constant-mass law allows).
#   FALSIFIED IF: r=0.005 or 0.01 matches or beats 0.02 -- then r* is still falling fast,
#              r*.m collapses, and the "shrink r as the model grows" rule needs a steeper
#              exponent than one power of m.
#   ALSO FALSIFIED IF: the whole 0.005-0.03 range is flat within noise -- then the R34
#              "interior optimum" is a plateau starting at 0.005 and the r*-falls-with-scale
#              reading (which the three-model table above rests on) is much weaker than it looks.
#
# Also takes the two cells the claim rests on, r=0 and r=0.02, from n=3 to n=5.
set -e
cd ~/metaopt
OUT=~/metaopt/runs/r34f
mkdir -p $OUT
COMMON="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 \
--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 \
--dataset CIFAR10 --batch-size 100 --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 \
--alpha0 1e-6 --num-epochs 100 --NN-name ResNet34 --stepsize-groups layerwise"
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
sub(){ sbatch --job-name=$1 --partition=$PARTS --cpus-per-task=6 --mem=14G --time=03:50:00 \
  --export=ALL,AUGMENT=1,BETA_CLIP=-15:-2.3026,HIER=additive,ETA_RATIO=$2 \
  jobs/run_cifar.sh $COMMON --seed $3 --save-directory $OUT --run-name $1 >/dev/null; }
n=0
# fine grid below the current peak
for spec in 0.005:r0005 0.01:r001 0.015:r0015 0.025:r0025; do
  R=${spec%%:*}; TAG=${spec##*:}
  for S in 0 1 2; do sub r34f-$TAG-s$S $R $S; n=$((n+1)); done
done
# n=3 -> n=5 on the two cells the +0.786 rests on
for spec in 0:r0 0.02:r002; do
  R=${spec%%:*}; TAG=${spec##*:}
  for S in 3 4; do sub r34f-$TAG-s$S $R $S; n=$((n+1)); done
done
echo "SUBMITTED r34f: $n"
echo "TOTAL_cluster_b=$(squeue -u $USER -h | wc -l)"
