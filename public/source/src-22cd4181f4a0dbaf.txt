def mean(v):

def sdev(v):

def sha256_of(path):
    h = hashlib.sha256()
    fh = open(path, "rb")
    while True:
        b = fh.read(1 << 20)
        if not b:
            break
        h.update(b)
    fh.close()
    return h.hexdigest()

def r18():
    """ResNet-18 for CIFAR (3x3 stem, no maxpool, 100-way head): (name, numel, kind) in named_parameters order."""
    t = [("conv1.weight", 64 * 3 * 3 * 3, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    cin = 64
    for li, w in ((1, 64), (2, 128), (3, 256), (4, 512)):
        for b in (0, 1):
            pre = "layer%d.%d." % (li, b)
            fin = cin if b == 0 else w
            stride = 2 if (li > 1 and b == 0) else 1
            t.append((pre + "conv1.weight", w * fin * 3 * 3, "Conv2d"))
            t.append((pre + "bn1.weight", w, "BatchNorm2d"))
            t.append((pre + "bn1.bias", w, "BatchNorm2d"))
            t.append((pre + "conv2.weight", w * w * 3 * 3, "Conv2d"))
            t.append((pre + "bn2.weight", w, "BatchNorm2d"))
            t.append((pre + "bn2.bias", w, "BatchNorm2d"))
            if stride != 1 or fin != w:
                t.append((pre + "shortcut.0.weight", w * fin, "Conv2d"))
                t.append((pre + "shortcut.1.weight", w, "BatchNorm2d"))
                t.append((pre + "shortcut.1.bias", w, "BatchNorm2d"))
        cin = w
    t.append(("linear.weight", 512 * 100, "Linear"))
    t.append(("linear.bias", 100, "Linear"))
    return t

def read_out(path):
    txt = open(path, errors="replace").read()
    eps = {}
    rec = {"args": [], "env": [], "pt": [], "sw": [], "run_done": False,
           "traceback": ("Traceback (most recent call last)" in txt)}
    for ln in txt.splitlines():
        if ln.startswith("ARGS:"):
            rec["args"].append(ln)
        elif ln.startswith("ENV:"):
            rec["env"].append(ln)
        elif ln.startswith("PROBE_TENSOR:"):
            rec["pt"].append(ln)
        elif ln.strip() == "RUN_DONE":
            rec["run_done"] = True
        else:
            for sw in SWITCHES:
                if ln.startswith(sw):
                    rec["sw"].append(ln)
                    break
        if ln.startswith("Epoch "):
            f = ln.split(",")
            if len(f) >= 3 and "Train Accuracy:" in f[1] and "Test Accuracy:" in f[2]:
                try:
                    e = int(f[0].split()[1])
                    tr = float(f[1].split(":")[1].strip().split()[0])
                    te = float(f[2].split(":")[1].strip().split()[0])
                except (ValueError, IndexError):
                    continue
                eps[e] = (tr, te)
    rec["eps"] = eps
    rec["n_epochs"] = len(eps)
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    else:
        rec["plateau5"] = rec["train5"] = None
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"] and not rec["traceback"]
                       and rec["plateau5"] is not None)
    return rec


def split_args(line):
    """`ARGS: --a 1 --b x y` -> {a: '1', b: 'x y'} plus the count of repeats. No regex."""
    toks = line[len("ARGS:"):].split()
    flags, order, cur = {}, [], None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            order.append(cur)
            flags[cur] = None
        elif cur is not None:
            flags[cur] = t if flags[cur] is None else flags[cur] + " " + t
    dup = sorted(set(k for k in order if order.count(k) > 1))
    return flags, dup


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None


# ---------------------------------------------------------------- TFRecord + masked CRC-32C (independent)
_CRC_TABLE = []


def _crc_table():
    if not _CRC_TABLE:
        poly = 0x82F63B78
        for i in range(256):
            c = i
            for _ in range(8):
                c = (c >> 1) ^ (poly if (c & 1) else 0)
            _CRC_TABLE.append(c)
    return _CRC_TABLE


def crc32c(data):
    tab = _crc_table()
    c = 0xFFFFFFFF
    for b in data:
        c = tab[(c ^ b) & 0xFF] ^ (c >> 8)
    return c ^ 0xFFFFFFFF


def unmask(m):
    m = (m - 0xA282EAD8) & 0xFFFFFFFF
    return ((m >> 17) | (m << 15)) & 0xFFFFFFFF


def tfrecords(path):
    """Yield (payload, crc_len_ok, crc_data_ok) for every frame of a TFRecord file."""
    data = open(path, "rb").read()
    off = 0
    while off + 12 <= len(data):
        (ln,) = struct.unpack("<Q", data[off:off + 8])
        (mlen,) = struct.unpack("<I", data[off + 8:off + 12])
        end = off + 12 + ln
        if end + 4 > len(data):
            return
        payload = data[off + 12:end]
        (mdat,) = struct.unpack("<I", data[end:end + 4])
        yield payload, (unmask(mlen) == crc32c(data[off:off + 8])), (unmask(mdat) == crc32c(payload))
        off = end + 4


def tb_table(runsdir, a, s):
    """key -> set of values, read out of the run's own events file(s); plus (n_files, n_frames, n_crc_bad)."""
    d = os.path.join(runsdir, PREFIX, "Tensorboard_outputs", run_name(a, s))
    files = sorted(f for f in os.listdir(d) if f.startswith("events.out.tfevents.")) if os.path.isdir(d) else []
    out, frames, crcbad = {}, 0, 0
    for f in files:
        for payload, c1, c2 in tfrecords(os.path.join(d, f)):
            frames += 1
            if not (c1 and c2):
                crcbad += 1
                continue
            if b"|" not in payload:
                continue
            for raw in payload.split(b"\n"):
                if raw.count(b"|") != 3 or not raw.startswith(b"|") or not raw.endswith(b"|"):
                    continue
                parts = raw.split(b"|")
                k = parts[1].decode("utf-8", "replace")
                v = parts[2].decode("utf-8", "replace")
                out.setdefault(k, set()).add(v)
    return out, len(files), frames, crcbad


# ---------------------------------------------------------------- the branch map, re-typed
def collapse_word(k, l):
    if l < REF_MIN:
        return "REF-UNHEALTHY"
    if l - k <= MATCH_BAR:
        return "COLLAPSE-IS-CONFIG"
    if k <= R50 * l and l - k >= GAP_MIN:
        return "COLLAPSE-PERSISTS"
    return "COLLAPSE-ATTENUATED"


def iso_word(k, l, i, cw):
    if cw in ("REF-UNHEALTHY", "COLLAPSE-IS-CONFIG") or l - k <= RESCUE_BAR:
        return "ISO-UNREADABLE"
    if i >= l - MATCH_BAR:
        return "ISO-RESCUES"
    if i <= k + MATCH_BAR:
        return "ISO-FAILS"
    return "ISO-PARTIAL"


def decide(M, div):
    if any(a in div for a in BASEARMS):
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX or M["kL"] - M["k01"] < GAP_MIN or M["ISO"] - M["k01"] < GAP_MIN:
        return "ANCHOR-FAILED"
    halves = []
    for X in FACTORS:
        if any((X + b) in div for b in BASEARMS):
            halves.append("%s:DIVERGED" % X)
            continue
        k, l, i = M[X + "k01"], M[X + "kL"], M[X + "ISO"]
        cw = collapse_word(k, l)
        halves.append("%s:%s/%s" % (X, cw, iso_word(k, l, i, cw)))
    return "+".join(halves)

        "  Licensed: 'at {X} the collapse does not run through THESE carriers' -- the carrier account is configuration-bound",
        "  even though the collapse is not.  Not licensed: that no other small set carries it at {X} (not re-nominated)."],
    "ISO-PARTIAL": [
        "  {X} ISO-PARTIAL: isolating the carriers recovers more than 5 pp but stays more than 5 pp below layerwise.",
        "  Licensed: 'the carriers carry part of the collapse at {X}'.  No full-rescue sentence."],
    "ISO-UNREADABLE": ["  {X} ISO-UNREADABLE: no gap (or no healthy reference) to rescue; ISO's level is stamped, not read."],
}


def licence_lines(branch):
    out = []
    for X, h in zip(FACTORS, branch.split("+")):
        body = h[len(X) + 1:]
        sub = {"X": X, "F": FACTOR_TEXT[X], "V": VALUE_TEXT[X], "R": ROUTE_TEXT[X]}
        if body == "DIVERGED":
            out += [ln.format(**sub) for ln in LIC_C["DIVERGED"]]
            continue
        c, i = body.split("/", 1)
        out += [ln.format(**sub) for ln in LIC_C[c]] + [ln.format(**sub) for ln in LIC_I[i]]
    out += ["EFFECT ON " + READING + ": as the two halves say, ONE value per factor, one factor at a time, at ciso1's cell only",
            "(ResNet18_c100, CIFAR-100, Lion meta, meta step 1e-3, alpha0 1e-6, clip, 100 epochs).  Readings at k01 are LOCATIONS."]
    return out

        print("  %-6s carriers 50/53/59 = %.3f / %.3f / %.3f (sum %.3f); top tensor %s %.3f"
              % (a, tot[49], tot[52], tot[58], tot[49] + tot[52] + tot[58], NAMES[top[0]], tot[top[0]]))

    emit("")
    emit("WHAT THIS DECIDES -- as registered, before any run existed:")
    for ln in licence_lines(branch):
        emit("  " + ln)
    emit("")
    emit("WHAT IT DOES NOT LICENSE, UNCONDITIONALLY:")

    emit("  * The parent's exact configuration: momentum 0.9 here keeps alpha0 1e-6 (parent Table 4 row: 1e-5, alignment UNSURE).")

    emit("  * For W0, which route: WD on the gammas vs WD inside the meta trace (both removed together).")
    emit("  * Point agreement for an arm at the k01 level: such readings are LOCATIONS (164.6).")

    # ---------------- [7] line-by-line agreement
    print("\n[7] LINE-BY-LINE AGREEMENT WITH THE SCORER LOG")
    got = open(logpath).read().splitlines()
    nmis = 0
    if len(got) != len(L):
        bad("scorer log has %d lines, this parser rebuilt %d" % (len(got), len(L)))
    for i in range(max(len(got), len(L))):
        gl = got[i] if i < len(got) else "<missing>"
        ml = L[i] if i < len(L) else "<not rebuilt>"
        if ml.startswith("@SUFFIX@"):
            if gl.endswith(ml[len("@SUFFIX@"):]) and gl.startswith("  manifest: " if "MANIFEST" in ml else "  provenance: "):
                continue
            nmis += 1
            bad("line %d host-path line does not end with the expected suffix: %r" % (i + 1, gl))
            continue
        if gl != ml:
            nmis += 1
            if nmis <= 20:
                bad("line %d\n      scorer: %r\n      mine:   %r" % (i + 1, gl, ml))
    (ok if nmis == 0 else bad)("every one of the %d scorer lines rebuilt independently and matched (2 host-path lines by suffix)"
                               % len(got), "%d mismatched" % nmis)
    gfin = [ln for ln in got if ln.startswith("FINAL:")]
    (ok if gfin == [final] else bad)("FINAL identical, token by token")
    for tok_mine, tok_log in zip(final.split(" | "), (gfin[0] if gfin else "").split(" | ")):
        if tok_mine != tok_log:
            bad("FINAL token %r vs %r" % (tok_mine, tok_log))

    # ---------------- descriptive, not registered
    print("\n[D1] DESCRIPTIVE / UNSURE -- the ARGS witness per arm (for the corpus-exclusion stage, 255.7)")
    print("     definition: the tokens of the run's own argparse line; identical on all 3 seeds of the arm.")
    for a in ARMS:
        m, w, gr = argswit[a]

              % (a, m, w, gr if gr != ISOSPEC else "ISO(sets:1-49,51-52,54-58,60-62/3 carriers)"))
    nondef = [a for a in ARMS if (MOM[a], WDC[a]) != ("0.99", "0.1")]
    print("     rows whose ARGS carries a NON-DEFAULT base momentum or base WD (the exclusion stage owes these):")
    for a in nondef:
        for s in SEEDS:
            r = runs[(a, s)]
            wit = ("ARGS --momentum-param-base 0.9" if factor_of(a) == "M9" else "ARGS --weight-decay-base 0")
            print("       %-14s job %d   witness: %s" % (run_name(a, s), r["job"], wit))
    print("     %d rows (9 M9 + 9 W0); the 9 anchor rows are plain 0.99 / 0.1 and pool normally." % (len(nondef) * len(SEEDS)))

    print("\n[D2] DESCRIPTIVE / UNSURE -- does k01 still collapse, does ISO still rescue (same numbers, in words)")
    for X in FACTORS:
        k, l, i = M[X + "k01"], M[X + "kL"], M[X + "ISO"]
        print("     %s: k01 %.4f vs its own kL %.4f -> gap %.4f pp, ratio %.4f (collapse bars: gap >= 20 AND ratio <= 0.50)"
              % (X, k, l, l - k, k / l))
        print("        ISO %.4f; ISO - kL = %+.4f (rescue bar: within 5 pp of kL); ISO - k01 = %+.4f" % (i, i - l, i - k))
    print("     anchor in the same batch: k01 %.4f, kL %.4f, ISO %.4f (gap %.4f, ratio %.4f)"
          % (M["k01"], M["kL"], M["ISO"], M["kL"] - M["k01"], M["k01"] / M["kL"]))

    print("\n[D3] DESCRIPTIVE / UNSURE -- carrier nomination from the probe records (500 records per run, all steps)")
    for a in ("k01", "M9k01", "W0k01"):
        tot = shares[a]
        top = sorted(range(NTENS), key=lambda j: -tot[j])[:5]
        print("     %-6s ciso1's trio share %.3f; top5 %s"
              % (a, tot[49] + tot[52] + tot[58], ", ".join("%s %.3f" % (NAMES[j], tot[j]) for j in top)))
    print("     UNSURE: a share readout is not a nomination test; no control arm at M9 / W0 (S4 is SHOULD-tier).")

    print("\n[D4] DESCRIPTIVE / UNSURE -- TEST (TRAIN) at epochs 9 / 19 / 49 / 99, arm mean over the 3 seeds")
    print("     record boundary: the run's own `Epoch <n>` lines, 0-indexed; the final epoch line is printed twice by")
    print("     the harness and is read once (dict by epoch); seeds are the batch's own 108/109/110, no split.")
    for a in ARMS:
        cells = []
        for e in (9, 19, 49, 99):
            te = mean([runs[(a, s)]["eps"][e][1] for s in SEEDS])
            tr = mean([runs[(a, s)]["eps"][e][0] for s in SEEDS])
            cells.append("e%-2d %6.2f (%6.2f)" % (e, te, tr))
        print("     %-6s %s" % (a, "  ".join(cells)))

    print("\n" + "=" * 100)
    print("VIOLATIONS %d" % len(VIOL))
    print("=" * 100)
    return 0 if not VIOL else 1
