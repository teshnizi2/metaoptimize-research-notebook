# cL1_cts1_score.py -- THE REGISTERED SCORER FOR `cts1`, THE SINGLE-TENSOR
#                      RESOLUTION OF THE k = 49 -> k = 52 CLIFF ON CIFAR-100.
#
# STANDING RULE 21: this file is git-committed BEFORE any cts1 run exists.
# Every arm, bar, gate, prediction, verdict tree and refusal below is frozen
# here.  --selftest re-derives every archived constant from
# results/all_runs.csv, re-derives cpk1's own cliff from the corpus, re-derives
# every partition and every moved tensor from the frozen per-tensor NAME and
# PARAMETER tables, asserts that the corpus contains ZERO cts1 rows at
# registration time, and cross-checks each constant against
# bin/cL1_cliff_tensor_split.sh's OWN TEXT so registration and batch cannot
# drift.
#
# STANDING RULE 16: run this file UNEDITED and quote its verdict.  Passing its
# documented arguments (--runs, --manifest, --selftest) is NOT editing it.
# Once a cts1 run exists this file is frozen for good.
#
# =============================================================================
# WHERE THIS COMES FROM
# =============================================================================
# `cpk1` (39 jobs, CIFAR-100, one submission, scored by its own registered
# scorer UNEDITED, CORRECTIONS 146) swept the cut position k at fixed m = 2
# over eleven grid points and returned
#     SHAPE POSITION-SINGLE-PEAKED | BOUNDARY BOUNDARY-REFUTED
#     | PARAM-BAL PARAM-BALANCE-FAVOURED | REPLICATION CBL1-ORDER-REPLICATES
# with argmax k* = 49.  Re-derived here from results/all_runs.csv in
# --selftest, never quoted from prose: cpk1's in-batch anchors were
# scalar 22.720667 and layerwise 69.710667 (gap 46.990000 pp), and
#
#     k    spec       plateau5   CAPTURE      parameter share of the COARSE group
#     49   [49,13]     55.4473   +0.696460    0.562834
#     52   [52,10]     38.3420   +0.332440    0.574607
#
# ONE STEP OF THE GRID, k = 49 -> k = 52, COSTS -0.364021 CAPTURE
# (-17.1053 pp, se 0.6695, t -25.55, df 4).  Half of the entire
# scalar-to-layerwise benefit disappears.  Those two arms differ by 0.011773
# of parameter share.  CORRECTIONS 146.4 recorded this as DESCRIPTIVE -- it
# was not registered, the comparison was matched on nothing -- and named it
# the next test.  Parameter balance predicted the argmax location exactly
# (its pre-registered rival test P5 was refuted in its favour) and yet CANNOT
# produce a 36-point collapse from a 1.2-point change in the split.
#
# =============================================================================
# THE NETWORK, MEASURED NOT ASSUMED
# =============================================================================
# ResNet18_c100 as built by build_network("ResNet18_c100", torch.device("cpu"))
# has exactly 62 parameter tensors and 11,220,132 parameters.  Read off the
# LIVE model at registration time; the launcher re-proves it at submission time
# and writes PARTITION-MANIFEST.txt, which G1 below REQUIRES and checks in
# SIZES, in PARAMETER COUNTS, and in the TENSOR NAMES on both sides of the cut.
#
# In named_parameters() order (1-based; a cut at k puts tensors 1..k in the
# COARSE group and k+1..62 in the FINE group), layer4.0 occupies 46..54:
#
#     46  layer4.0.conv1.weight        1179648
#     47  layer4.0.bn1.weight              512
#     48  layer4.0.bn1.bias                512
#     49  layer4.0.conv2.weight        2359296
#     50  layer4.0.bn2.weight              512   <- moves at k=49 -> k=50
#     51  layer4.0.bn2.bias                512   <- moves at k=50 -> k=51
#     52  layer4.0.shortcut.0.weight    131072   <- moves at k=51 -> k=52
#     53  layer4.0.shortcut.1.weight       512
#     54  layer4.0.shortcut.1.bias         512
#
# So the two missing grid points k = 50 and k = 51 cut the cpk1 cliff into
# THREE STEPS THAT EACH MOVE EXACTLY ONE TENSOR, total 132,096 parameters =
# 2.6931 % of the fine group's 4,905,060 at k = 49.  Nothing else about the
# run changes.  This is the finest resolution the partition axis admits: below
# one tensor there is no experiment.
#
# =============================================================================
# THE DESIGN.  6 arms x 3 seeds = 18 jobs, ONE submission
# =============================================================================
#   ANCHORS, run IN BATCH so CAPTURE is a within-batch quantity:
#     k01   scalar      m =  1
#     k62   layerwise   m = 62
#
#   THE FOUR CUTS, [k, 62-k]:
#     k = 49   the cpk1 ARGMAX, re-run IN BATCH
#     k = 50   NEW
#     k = 51   NEW
#     k = 52   the far side of the cpk1 cliff, re-run IN BATCH
#
#   WHY BOTH CLIFF EDGES ARE RE-RUN RATHER THAN SPLICED FROM cpk1.  BATCH, not
#   seed, is the unit of replication in this corpus (F(62,85) = 5.47,
#   p 6.9e-13 for batch; F(11,1976) = 0.152 for seed).  Splicing cpk1's k49 and
#   k52 onto two fresh interior points would make two of the three steps
#   cross-batch and one within-batch, and the three would not be comparable.
#   Six arms in one submission cost 6 extra jobs and buy a cliff that sits
#   ENTIRELY INSIDE ONE BATCH.  cpk1's own values enter only as DESCRIPTION.
#
#   CIFAR-100 / ResNet18_c100 / SGDm base / Lion meta / meta-stepsize 1e-3 /
#   alpha0 1e-6 / 100 epochs / batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 /
#   AUGMENT=1 / HIER unset / PROBE=0.  Matched cell-for-cell to cpk1, read out
#   of bin/cK1_cut_position_sweep.sh rather than reconstructed.
#
# =============================================================================
# THE PRIMARY STATISTIC, FROZEN
# =============================================================================
#   plateau5 = mean of the LAST 5 test epochs.  The CSV `plateau` column
#   (mean-of-last-20) is BANNED as primary and is never read by this file;
#   `best_test` is not a plateau and is never read either.  train5 = mean of
#   the last 5 TRAIN epochs, reported for EVERY arm.
#
#   CAPTURE(arm) = ( M(arm) - M(k01) ) / ( M(k62) - M(k01) )   on plateau5,
#   both anchors measured IN BATCH.
#
#   THE THREE SINGLE-TENSOR STEPS, as DROPS (positive = capture falls):
#     DROP1 = CAPTURE(k49) - CAPTURE(k50)   moves layer4.0.bn2.weight      512
#     DROP2 = CAPTURE(k50) - CAPTURE(k51)   moves layer4.0.bn2.bias        512
#     DROP3 = CAPTURE(k51) - CAPTURE(k52)   moves layer4.0.shortcut.0.weight
#                                                                      131072
#     TOTAL = CAPTURE(k49) - CAPTURE(k52) = DROP1 + DROP2 + DROP3
#     SHARE_i = DROP_i / TOTAL
#
#   DISCLOSED CONFOUND, inherited from cpk1, cbl1 and hb1 and not discovered
#   later: the two anchors use the `scalar` and `layerwise` code paths in
#   HF.init_meta while all four cuts use the `blockwise` path.  It rescales
#   CAPTURE by a constant and cannot reorder the four cuts, which share one
#   code path -- and every verdict below depends only on DIFFERENCES among
#   those four.
#
# =============================================================================
# THE TWO ACCOUNTS ON RECORD BEFORE THE DATA
# =============================================================================
# A1  PARAMETER MASS.  The step that moves the most parameters carries the
#     collapse.  layer4.0.shortcut.0.weight is 131,072 of the 132,096 moved
#     parameters = 99.225 %; the two BN tensors are 0.388 % each.  Under A1,
#     SHARE3 is near 1 and SHARE1 = SHARE2 (both tensors hold 512 parameters).
#     This is the reading carried into this batch from the cycle briefing, and
#     it is registered as the RIVAL TO BEAT.
#
# A2  BATCHNORM GAIN LEVERAGE.  layer4.0.bn2.weight is a multiplicative gain
#     on the output of layer4.0.conv2.weight -- 2,359,296 parameters that are
#     ALREADY in the coarse group at k = 49.  A per-group stepsize is a single
#     scalar, so what a group costs is not its parameter count but how badly
#     its members disagree about the stepsize they want; a BN scale is the one
#     free knob that lets the fine group compensate for a coarse-group
#     stepsize that is wrong for layer4.0's residual branch.  Under A2 the
#     collapse is carried by DROP1, and DROP1 >> DROP2 even though bn2.weight
#     and bn2.bias hold IDENTICAL parameter counts -- a scale is multiplicative
#     and has leverage, a shift is additive and has much less.
#
# A2 IS THIS AGENT'S POINT PREDICTION AND A1 IS THE BRIEFING'S.  They are
# registered together because they name DIFFERENT steps of the same three-step
# grid, so one submission separates them.  Recording that A1 is the inherited
# reading is what makes its defeat -- or its survival -- informative rather
# than a confirmation.  NO THIRD ACCOUNT MAY BE INTRODUCED AFTER THE DATA.
#
# =============================================================================
# THE MATCHED CONTROL THIS DESIGN GETS FOR FREE
# =============================================================================
# layer4.0.bn2.weight and layer4.0.bn2.bias hold EXACTLY 512 parameters each,
# sit adjacent in named_parameters(), belong to the same BatchNorm module, and
# move at consecutive steps of this grid.  Under ANY account in which the
# currency is parameter mass, DROP1 and DROP2 must be equal.  A separation
# between them is therefore direct, MATCHED evidence that mass is not the
# currency -- unlike CORRECTIONS 146.4's k49-vs-k52 comparison, which was
# matched on nothing.  This is registered as Q5 and as the SCALE-SHIFT verdict.
#
# =============================================================================
# PREDICTIONS, ON RECORD BEFORE SUBMISSION
# =============================================================================
#   Q1  (A2, THIS AGENT'S POINT PREDICTION)  The largest of the three drops is
#       DROP1, the step k = 49 -> k = 50 that moves layer4.0.bn2.weight.
#   Q2  (A1, THE RIVAL TO BEAT)  The largest is DROP3, the step
#       k = 51 -> k = 52 that moves layer4.0.shortcut.0.weight.
#   Q3  The cliff REPRODUCES in this batch: TOTAL >= 0.1820.
#   Q4  The split is NOT even: some drop is further than 0.05 from TOTAL/3.
#   Q5  SCALE-SHIFT ASYMMETRY: |DROP1 - DROP2| > 0.05, between two tensors of
#       IDENTICAL parameter count.
#   Q6  The TRAIN curve and the TEST curve agree on which step is largest.
#
#   Exactly one of Q1 and Q2 can hold, and both can fail (DROP2 largest, or an
#   even split).  I do NOT predict the magnitudes.
#
# =============================================================================
# REFUTATION CONDITIONS, FROZEN, CHECKED IN THIS ORDER
# =============================================================================
#   R1  CLIFF-NOT-REPRODUCED.  TOTAL < 0.1820, i.e. less than HALF of cpk1's
#       own 0.364021.  THIS IS THE REFUTATION THAT VOIDS THE QUESTION, AND IT
#       IS CHECKED FIRST.  cts1 exists only to decompose a cliff cpk1 measured;
#       if the cliff is not here, there is nothing to decompose, no tensor may
#       be named, and the deliverable is the word UNRESOLVED.  It is NOT a
#       licence to report the three steps anyway, and it is NOT a licence to
#       rescue the story by splicing cpk1's arms in.  0.1820 in CAPTURE units
#       is 8.51 pp and 14.4 SE of a 3-seed difference at the re-derived noise
#       floor.  A negative TOTAL (the cliff reversed) also lands here.
#   R2  SPLIT-EVEN.  All three drops lie within 0.05 of an even split
#       (TOTAL/3 each).  Then this is NOT a single-tensor effect, the collapse
#       is distributed across the whole three-tensor move, and NO TENSOR MAY BE
#       NAMED.  Verdict SPLIT-EVEN-NO-TENSOR-NAMED; both A1 and A2 fail.
#   R3  NO-MAJORITY.  The largest drop does not exceed HALF of TOTAL, or it
#       does not exceed the second-largest drop by more than 0.05.  Then the
#       largest step is REPORTED as the largest and NO TENSOR IS NAMED as
#       carrying the cliff.  Verdict SPLIT-UNEVEN-NO-MAJORITY.
#       Naming a tensor requires the CONJUNCTION: majority AND separation.
#
# WHAT DOES NOT COUNT AS REFUTATION, AND EQUALLY DOES NOT COUNT AS RESCUE
#   * A drop of the wrong sign (a step that RAISES capture).  Reported with
#     its t; it can push a SHARE above 1, which is allowed and changes no bar.
#   * An in-batch k49 or k52 differing from cpk1's 55.447 / 38.342.  cpk1 is
#     ANOTHER BATCH.  Only this batch's own TOTAL is tested (R1); cpk1's
#     magnitudes are printed as description and gate nothing.
#   * TOTAL larger than cpk1's.  Allowed, reported, changes no bar.
#   * A pretty fit of some quantity nobody named.  A1 and A2 are the only two
#     accounts on record and neither may be swapped for a third.
#   * No post-hoc change of statistic, window, scale, grid or bar.
#
# =============================================================================
# THE GATES, IN ORDER.  THE ORDER IS PART OF THE REGISTRATION
# =============================================================================
#   G0  PROVENANCE (RULE 20), ABSOLUTELY GATING.  Every arm is MEASURED from
#       the run's OWN `ARGS:`/`ENV:` lines, never from its filename.
#   G1  THE PARTITION GATE.  PARTITION-MANIFEST.txt, written by the launcher
#       from the LIVE model, must show 62 tensors, 11,220,132 parameters, and
#       for all four specs the registered size vector, the registered per-group
#       parameter counts, AND the registered tensor NAMES on both sides of the
#       cut.  G1b then derives WHICH TENSOR MOVES at each of the three steps
#       from the manifest alone and requires the registered three.  Without
#       this, "k = 50" is a claim about a string, not about a network.
#   G2  COMPLETENESS.  >= 95 of 100 epochs and all 5 tail epochs present.  Any
#       registered arm with fewer than 2 usable seeds VOIDS the batch.
#   G3  THE ARCHIVE GATE.  In-batch k01 and k62 within +/-2.00 pp of their
#       archived cell means (22.7670 and 69.5171).  A miss is VOID-TREE.
#   G4  THE MANIFOLD GATE.  In-batch (k62 - k01) >= 23.3750 pp, half the
#       archived gap.  Below it CAPTURE is a ratio of noise: VOID-NOMANIFOLD.
#   G5a THE CLIFF GATE (R1).  Does the cpk1 cliff exist inside THIS batch?
#   G5b THE PRIMARY.  R2 and R3 -> the SPLIT verdict, the named tensor if any,
#       then the MASS (A1), BN-LEVERAGE (A2) and SCALE-SHIFT verdicts.
#   G6  DESCRIPTIVE, NEVER GATING.  The train ladder, the parameter shares,
#       the per-step t statistics, and the cross-batch cpk1 comparison.
#
# =============================================================================
# WHAT THIS SCORER WILL NOT DO
# =============================================================================
#   * It will not read the CSV `plateau` (last-20) column, or `best_test`.
#   * It will not pool cts1 with cpk1 or with any archived batch to raise n,
#     and it will not substitute a cpk1 arm for a missing cts1 arm.  Archived
#     rows enter only as G3's and G4's comparators and as the noise floor.
#   * It will not claim anything about CIFAR-10, Tiny-ImageNet, ImageNet-489,
#     ResNet-50, resnet18_blocks, alpha0 1e-3, any meta-stepsize but 1e-3, any
#     optimiser pair but SGDm+Lion, any m other than 2 (the anchors are
#     anchors, not rungs), or any hierarchical / shrinkage operator.
#   * It will not generalise a named tensor beyond THIS network in THIS
#     named_parameters() order.  A different ordering is a different
#     experiment, and "BN scales matter" is a hypothesis this batch can
#     support at exactly one site, not a law.
#   * It will not soften a bar after the fact.  If the answer lands in a gap,
#     the deliverable is the word UNRESOLVED.
#
# =============================================================================
# USAGE
#   python3 analysis/cL1_cts1_score.py --selftest
#   python3 analysis/cL1_cts1_score.py --runs /cluster/account_b/metaopt/runs

def ktag(k):
    return "k%02d" % k


def kspec(k):
    return "[%d,%d]" % (k, N_TENSORS - k)


ARMS = collections.OrderedDict()          # tag -> --stepsize-groups value
ARMS[TAG_LO] = "scalar"
for _k in KGRID:
    ARMS[ktag(_k)] = kspec(_k)
ARMS[TAG_HI] = "layerwise"


# =============================================================================
# the partition algebra.  Sizes, parameter counts and the moved tensor come
# from the SPEC STRING and the frozen tables, so the manifest and this file can
# be compared without either quoting the other.
# =============================================================================
def spec_sizes(spec):
    if not (spec.startswith("[") and spec.endswith("]")):
        return None
    return [int(x) for x in spec[1:-1].split(",")]


def group_params(sizes):
    out, i = [], 0
    for s in sizes:
        out.append(sum(TENSOR_NUMEL[i:i + s]))
        i += s
    return out


def param_share(k):
    """Fraction of the network's parameters in the COARSE group of [k, 62-k]."""
    return float(sum(TENSOR_NUMEL[:k])) / float(N_PARAMS)


def fine_params(k):
    return sum(TENSOR_NUMEL[k:])


def moved_tensor(k_from, k_to):
    """(names, numel) leaving the FINE group when the cut moves k_from->k_to."""
    names = TENSOR_NAMES[k_from:k_to]
    return names, sum(TENSOR_NUMEL[k_from:k_to])


def struct_at(k):
    for lo, hi, name in STRUCT:
        if lo <= k <= hi:
            return ("%s ends at %d" % (name, hi) if k == hi
                    else "inside %s (%d..%d)" % (name, lo, hi))
    return "?"


def arm_sizes(tag):
    spec = ARMS[tag]
    if spec == "scalar":
        return [N_TENSORS]
    if spec == "layerwise":
        return [1] * N_TENSORS
    return spec_sizes(spec)


def arm_m(tag):
    return len(arm_sizes(tag))

def _csv_rows(path=CSV):
    with open(path) as fh:
        return [r for r in csv.DictReader(fh) if (r.get("superseded") or "0") != "1"]


def _batch_of(run):
    return re.split(r"[-_]", str(run))[0]


def _in_cell(r):
    return (r["network"] == NET and r["dataset"] == DSET
            and r["base"] == BASE_ALG and r["meta"] == META_ALG
            and r["meta_stepsize"] == MS and r["alpha0"] == ALPHA0
            and r["gamma"] == GAMMA and r["augment"] == AUG
            and r["beta_clip"] == CLIP and not (r.get("hier") or "").strip()
            and r["batch_size"] == str(BATCH) and r["epochs_done"] == str(EPOCHS)
            and r["collapsed"] == "0")


def _cell_p5(r):
    if not _in_cell(r):
        return None
    try:
        return float(r["plateau5"])
    except (TypeError, ValueError):
        return None


def archived_arms(path=CSV):
    """{granularity: (mean plateau5, n, distinct batches, mean final_train)}."""
    acc = collections.defaultdict(list)
    for r in _csv_rows(path):
        p5 = _cell_p5(r)
        if p5 is None:
            continue
        try:
            tr = float(r["final_train"])
        except (TypeError, ValueError):
            tr = None
        acc[r["granularity"]].append((p5, _batch_of(r["run"]), tr))
    out = {}
    for k, v in acc.items():
        trs = [t for _, _, t in v if t is not None]
        out[k] = (statistics.mean(x for x, _, _ in v), len(v),
                  len(set(b for _, b, _ in v)),
                  statistics.mean(trs) if trs else float("nan"))
    return out


def noise_floor_from_csv(path=CSV):
    """Pooled WITHIN-CELL SD of plateau5 over (batch x granularity) cells."""
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        p5 = _cell_p5(r)
        if p5 is not None:
            cells[(_batch_of(r["run"]), r["granularity"])].append(p5)
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc


def cpk1_from_csv(path=CSV):
    """cpk1's own CAPTURE at the two cliff edges, from ITS OWN anchors."""
    g = collections.defaultdict(list)
    n = 0
    for r in _csv_rows(path):
        if not str(r.get("run") or "").startswith(CPK1_PREFIX):
            continue
        n += 1
        try:
            g[r["granularity"]].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    if "scalar" not in g or "layerwise" not in g:
        return None
    lo, hi = statistics.mean(g["scalar"]), statistics.mean(g["layerwise"])
    caps = {}
    for k in (KCLIFF_HI, KCLIFF_LO):
        s = kspec(k)
        if s in g:
            caps[k] = (statistics.mean(g[s]) - lo) / (hi - lo)
    return {"n_rows": n, "lo": lo, "hi": hi, "caps": caps,
            "p5": dict((s, list(v)) for s, v in g.items())}


def premise_no_cts1_rows(path=CSV):
    """RULE 21: no run of this batch may exist in the corpus yet."""
    return sum(1 for r in _csv_rows(path)
               if str(r.get("run") or "").startswith(PREFIX))


# =============================================================================
# the SE arithmetic.  Each contrast gets the SE of ITS OWN linear combination,
# so a bar is never quoted against the wrong denominator.
# =============================================================================
def se_of_combo(coefs, sigma=SIGMA_W, n=3, gap=ARCH_GAP):
    """SE, in CAPTURE units, of sum_i coefs[i] * mean(arm_i) at n seeds each."""
    return sigma * math.sqrt(sum(c * c for c in coefs) / float(n)) / gap


# the linear combinations the bars are actually applied to
SE_ARM_DIFF = se_of_combo((1, -1))            # any DROP_i, and TOTAL
SE_ADJ_STEPS = se_of_combo((1, -2, 1))        # DROP1-DROP2 or DROP2-DROP3
SE_END_STEPS = se_of_combo((1, -1, -1, 1))    # DROP1-DROP3
SE_EVEN_13 = se_of_combo((2 / 3.0, -1, 1 / 3.0))          # DROP1 - TOTAL/3
SE_EVEN_2 = se_of_combo((-1 / 3.0, 1, -1, 1 / 3.0))       # DROP2 - TOTAL/3

def _tokens(line):
    import shlex
    try:
        return shlex.split(line)
    except ValueError:
        return line.split()


def parse_args_line(line):
    """(effective_flags, repeated_flags).  argparse keeps the LAST occurrence."""
    toks = _tokens(line)
    occ, i, n = [], 0, len(toks)
    while i < n:
        t = toks[i]
        if t.startswith("--") and len(t) > 2:
            if "=" in t:
                f, v = t.split("=", 1)
                occ.append((f.lstrip("-"), v))
                i += 1
                continue
            vals, j = [], i + 1
            while j < n and not (toks[j].startswith("--") and len(toks[j]) > 2):
                vals.append(toks[j])
                j += 1
            occ.append((t.lstrip("-"), " ".join(vals)))
            i = j
            continue
        i += 1
    eff, seen = collections.OrderedDict(), collections.OrderedDict()
    for f, v in occ:
        eff[f] = v
        seen.setdefault(f, []).append(v)
    return eff, dict((f, v) for f, v in seen.items() if len(v) > 1)


def parse_env_line(line):
    out = {}
    for tok in line.split():
        if "=" in tok:
            k, v = tok.split("=", 1)
            out[k] = v
    return out


def read_out(path):
    args = env = None
    test, train = {}, {}
    try:
        with open(path, "r", errors="replace") as fh:
            for line in fh:
                if args is None:
                    m = ARGS_RE.match(line)
                    if m:
                        args = m.group(1).strip()
                        continue
                if env is None:
                    m = ENV_RE.match(line)
                    if m:
                        env = m.group(1).strip()
                        continue
                m = EP_RE.search(line)
                if m:
                    test[int(m.group(1))] = float(m.group(2))
                m = EPTR_RE.search(line)
                if m:
                    train[int(m.group(1))] = float(m.group(2))
    except IOError:
        return None
    if args is None:
        return None
    eff, dup = parse_args_line(args)
    return {"args": eff, "dup": dup, "argsline": args,
            "env": parse_env_line(env or ""), "test": test, "train": train,
            "path": path}


def tail5(series, budget=EPOCHS, w=5):
    """PRIMARY window.  Mean of the LAST 5 epochs; train.py prints 0-indexed."""
    v = [series[e] for e in range(budget - w, budget) if e in series]
    return sum(v) / len(v) if len(v) == w else None


def norm_num(s):
    try:
        return "%g" % float(s)
    except (TypeError, ValueError):
        return None


def arm_of(a):
    """The arm a run ACTUALLY ran, from its own flags.  None if it is not ours."""
    if a.get("dataset") != DSET or a.get("NN-name") != NET:
        return None
    g = a.get("stepsize-groups")
    for tag, spec in ARMS.items():
        if g == spec:
            return tag
    return None

def gate0(recs):
    kept, voided = [], []
    for r in recs:
        a, e = r["args"], r["env"]
        why = []
        if r["dup"]:
            why.append("REPEATED FLAG %s" % sorted(r["dup"]))
        tag = arm_of(a)
        if tag is None:
            why.append("arm not in the registered design (dataset=%s NN=%s groups=%s)"
                       % (a.get("dataset"), a.get("NN-name"),
                          a.get("stepsize-groups")))
        for flag, want in (("meta-stepsize", MS), ("alpha0", ALPHA0),
                           ("num-epochs", str(EPOCHS)), ("batch-size", str(BATCH)),
                           ("gamma", GAMMA), ("alg-base", BASE_ALG),
                           ("alg-meta", META_ALG)):
            got = a.get(flag)
            same = (norm_num(got) == norm_num(want)) if norm_num(want) is not None \
                else (got == want)
            if not same:
                why.append("%s=%s (want %s)" % (flag, got, want))
        if e.get("AUGMENT") != AUG:
            why.append("AUGMENT=%s" % e.get("AUGMENT"))
        if e.get("BETA_CLIP") != CLIP:
            why.append("BETA_CLIP=%s" % e.get("BETA_CLIP"))
        if (e.get("HIER") or "none") != "none":
            why.append("HIER=%s (this is a PARTITION sweep; HIER must be unset)"
                       % e.get("HIER"))
        r["arm"] = tag
        r["seed"] = a.get("seed")
        (voided if why else kept).append((r, why))
    return kept, voided

MAN_RE = re.compile(
    r"^SPEC\s+(\S+)\s+M\s+(\d+)\s+SIZES\s+(\S+)\s+PARAMS\s+(\S+)\s+"
    r"FIRSTGROUP\s+(.*?)\s+LASTGROUP\s+(.*?)\s*$")


def read_manifest(path):
    if not path or not os.path.exists(path):
        return None
    out, ntens, npar = {}, None, None
    with open(path) as fh:
        for line in fh:
            s = line.strip()
            m = re.match(r"^NUM_PARAM_TENSORS\s+(\d+)$", s)
            if m:
                ntens = int(m.group(1))
                continue
            m = re.match(r"^TOTAL_PARAMS\s+(\d+)$", s)
            if m:
                npar = int(m.group(1))
                continue
            m = MAN_RE.match(s)
            if m:
                out[m.group(1)] = {
                    "m": int(m.group(2)),
                    "sizes": [int(x) for x in m.group(3).split(",") if x != ""],
                    "params": [int(x) for x in m.group(4).split(",") if x != ""],
                    "first": m.group(5).split(),
                    "last": m.group(6).split(),
                }
    return {"n_tensors": ntens, "n_params": npar, "specs": out}


def gate1(man):
    """[] if the manifest proves every registered partition; else the reasons."""
    if man is None:
        return ["PARTITION-MANIFEST.txt is missing"]
    why = []
    if man["n_tensors"] != N_TENSORS:
        why.append("NUM_PARAM_TENSORS=%s (want %d)" % (man["n_tensors"], N_TENSORS))
    if man["n_params"] != N_PARAMS:
        why.append("TOTAL_PARAMS=%s (want %d)" % (man["n_params"], N_PARAMS))
    for tag, spec in ARMS.items():
        want = spec_sizes(spec)
        if want is None:                 # scalar / layerwise: named, not listed
            continue
        got = man["specs"].get(spec)
        if got is None:
            why.append("no manifest line for %s (arm %s)" % (spec, tag))
            continue
        if got["sizes"] != want:
            why.append("%s (arm %s) composed to sizes %s on the live model, "
                       "registered %s" % (spec, tag, got["sizes"], want))
        if got["m"] != len(want):
            why.append("%s (arm %s) is m=%d, registered m=%d"
                       % (spec, tag, got["m"], len(want)))
        if sum(got["sizes"]) != N_TENSORS:
            why.append("%s (arm %s) sizes sum to %d, not %d"
                       % (spec, tag, sum(got["sizes"]), N_TENSORS))
        wantp = group_params(want)
        if got["params"] != wantp:
            why.append("%s (arm %s) holds %s parameters per group on the live "
                       "model, registered %s" % (spec, tag, got["params"], wantp))
        k = want[0]
        if got["first"] != list(TENSOR_NAMES[:k]):
            why.append("%s (arm %s) COARSE group is not tensors 1..%d by NAME "
                       "on the live model" % (spec, tag, k))
        if got["last"] != list(TENSOR_NAMES[k:]):
            why.append("%s (arm %s) FINE group is not tensors %d..%d by NAME "
                       "on the live model" % (spec, tag, k + 1, N_TENSORS))
    return why


def gate1b(man):
    """[] if the manifest proves WHICH TENSOR moves at each registered step.

    Derived from the manifest ALONE: the tensors that leave the FINE group
    between two specs are exactly set(last[k_from]) - set(last[k_to])."""
    if man is None:
        return ["PARTITION-MANIFEST.txt is missing"]
    why = []
    for a, b in STEPS:
        sa, sb = man["specs"].get(kspec(a)), man["specs"].get(kspec(b))
        if sa is None or sb is None:
            why.append("step k=%d -> k=%d: a manifest line is missing" % (a, b))
            continue
        moved = [n for n in sa["last"] if n not in set(sb["last"])]
        want_name, want_numel = STEP_TENSOR[(a, b)]
        if moved != [want_name]:
            why.append("step k=%d -> k=%d moves %s on the live model, "
                       "registered exactly ['%s']" % (a, b, moved, want_name))
            continue
        got_numel = sb["params"][0] - sa["params"][0]
        if got_numel != want_numel:
            why.append("step k=%d -> k=%d moves %d parameters on the live "
                       "model, registered %d" % (a, b, got_numel, want_numel))
    return why

def mean_se(v):
    m = statistics.mean(v)
    return (m, float("nan")) if len(v) < 2 else \
        (m, statistics.stdev(v) / math.sqrt(len(v)))


def contrast(a, b):
    """(diff, se, t, df) for mean(a) - mean(b), independent 3-seed arms."""
    ma, mb = statistics.mean(a), statistics.mean(b)
    if len(a) < 2 or len(b) < 2:
        return ma - mb, float("nan"), float("nan"), 0
    se = math.sqrt(statistics.variance(a) / len(a) + statistics.variance(b) / len(b))
    t = (ma - mb) / se if se > 0 else float("inf")
    return ma - mb, se, t, len(a) + len(b) - 2


def capture(m_arm, m_lo, m_hi):
    return (m_arm - m_lo) / (m_hi - m_lo)


def drops_from_caps(caps):
    """[DROP1, DROP2, DROP3], positive = CAPTURE falls across that step."""
    return [caps[a] - caps[b] for a, b in STEPS]


# =============================================================================
# THE FROZEN VERDICT TREES.  Exhaustive, deterministic, checked in this order.
# =============================================================================
def cliff_verdict(total):
    """R1.  Checked FIRST.  Below the bar the whole question is void."""
    if total < CLIFF_BAR:
        return "CLIFF-NOT-REPRODUCED"
    return "CLIFF-REPRODUCES"


def split_verdict(drops, total):
    """R2 then R3.  Only meaningful once cliff_verdict is CLIFF-REPRODUCES."""
    if all(abs(d - total / 3.0) <= EVEN_BAND for d in drops):
        return "SPLIT-EVEN-NO-TENSOR-NAMED"                  # R2
    order = sorted(range(len(drops)), key=lambda i: -drops[i])
    top, second = drops[order[0]], drops[order[1]]
    if top / total > MAJORITY_SHARE and (top - second) > SEPARATION_BAR:
        return "SINGLE-TENSOR-MAJORITY"
    return "SPLIT-UNEVEN-NO-MAJORITY"                        # R3


def named_tensor(drops, total):
    """The tensor this batch is entitled to name, or None.  None is a result."""
    if cliff_verdict(total) != "CLIFF-REPRODUCES":
        return None
    if split_verdict(drops, total) != "SINGLE-TENSOR-MAJORITY":
        return None
    i = max(range(len(drops)), key=lambda j: drops[j])
    return STEP_TENSOR[STEPS[i]][0]


def mass_verdict(share3):
    """A1, the briefing's account: the projection carries it because it is
    99.225 % of the moved parameters."""
    if share3 >= MASS_HIGH:
        return "MASS-PROPORTIONAL-CONSISTENT"
    if share3 < MASS_LOW:
        return "MASS-REFUTED"
    return "MASS-UNRESOLVED"


def bn_verdict(share_bn):
    """A2, this agent's account: the BN pair carries it on leverage, not mass."""
    if share_bn > BN_HIGH:
        return "BN-LEVERAGE-FAVOURED"
    if share_bn < BN_LOW:
        return "BN-LEVERAGE-REFUTED"
    return "BN-LEVERAGE-UNRESOLVED"


def scaleshift_verdict(d_scale, d_shift):
    """Q5, the MATCHED control: two tensors of IDENTICAL parameter count."""
    if abs(d_scale - d_shift) > ASYM_BAR:
        return "SCALE-SHIFT-ASYMMETRIC"
    return "SCALE-SHIFT-SYMMETRIC"

def selftest():
    fails = []

    def ok(label, cond, extra=""):
        print("  %-70s %s %s" % (label, "PASS" if cond else "**FAIL**", extra))
        if not cond:
            fails.append(label)

    print("cL1_cts1_score selftest -- every constant re-derived, none quoted")

    print("-- the archived CIFAR-100 cell -------------------------------------")
    g = archived_arms()
    for k in ("scalar", "layerwise"):
        ok("archived %s arm present" % k, k in g,
           ("mean %.4f n=%d batches=%d train %.3f" % g[k]) if k in g else "")
    if all(k in g for k in ("scalar", "layerwise")):
        gap = g["layerwise"][0] - g["scalar"][0]
        ok("ARCH_SCAL matches the CSV", abs(g["scalar"][0] - ARCH_SCAL) < 0.005,
           "%.4f" % g["scalar"][0])
        ok("ARCH_LAY matches the CSV", abs(g["layerwise"][0] - ARCH_LAY) < 0.005,
           "%.4f" % g["layerwise"][0])
        ok("ARCH_GAP matches the CSV", abs(gap - ARCH_GAP) < 0.005, "%.4f pp" % gap)
        ok("MANIFOLD_BAR is exactly half the archived gap",
           abs(MANIFOLD_BAR - gap / 2.0) < 0.005, "%.4f" % (gap / 2.0))
        ok("archived scalar UNDERFITS (the failure is train-side)",
           abs(g["scalar"][3] - ARCH_TRAIN_SCAL) < 0.05
           and abs(g["scalar"][3] - g["scalar"][0]) < 2.0,
           "train %.3f vs test %.4f" % (g["scalar"][3], g["scalar"][0]))
        ok("archived layerwise FITS", abs(g["layerwise"][3] - ARCH_TRAIN_LAY) < 0.05,
           "train %.3f" % g["layerwise"][3])
    sd, df, nc = noise_floor_from_csv()
    ok("SIGMA_W matches the CSV", abs(sd - SIGMA_W) < 0.005,
       "re-derived %.4f df=%d cells=%d" % (sd, df, nc))
    ok("SIGMA_DF matches the CSV", df == SIGMA_DF, "%d" % df)

    print("-- RULE 21 premise: nothing of this batch may exist yet ------------")
    n = premise_no_cts1_rows()
    ok("corpus holds ZERO %s rows" % PREFIX, n == 0, "found %d" % n)

    print("-- cpk1's cliff, re-derived from the corpus ------------------------")
    c = cpk1_from_csv()
    ok("cpk1 is in the corpus", c is not None)
    if c:
        ok("cpk1 contributed %d rows" % CPK1_N_ROWS, c["n_rows"] == CPK1_N_ROWS,
           "found %d" % c["n_rows"])
        ok("cpk1's in-batch anchors re-derive",
           abs(c["lo"] - CPK1_LO) < 5e-4 and abs(c["hi"] - CPK1_HI) < 5e-4,
           "scalar %.4f layerwise %.4f gap %.4f"
           % (c["lo"], c["hi"], c["hi"] - c["lo"]))
        ok("both cliff edges are in the corpus",
           set(c["caps"]) == {KCLIFF_HI, KCLIFF_LO}, str(sorted(c["caps"])))
        for k in (KCLIFF_HI, KCLIFF_LO):
            if k in c["caps"]:
                ok("cpk1 CAPTURE at k=%d re-derives to %.6f" % (k, CPK1_CAP[k]),
                   abs(c["caps"][k] - CPK1_CAP[k]) < 5e-5, "%.6f" % c["caps"][k])
        if set(c["caps"]) == {KCLIFF_HI, KCLIFF_LO}:
            tot = c["caps"][KCLIFF_HI] - c["caps"][KCLIFF_LO]
            ok("cpk1's cliff TOTAL re-derives to %.6f" % CPK1_TOTAL,
               abs(tot - CPK1_TOTAL) < 5e-5, "%.6f" % tot)
            ok("CLIFF_BAR is exactly HALF of cpk1's own TOTAL",
               abs(CLIFF_BAR - tot / 2.0) < 5e-4, "%.6f" % (tot / 2.0))
            a = c["p5"][kspec(KCLIFF_HI)]
            b = c["p5"][kspec(KCLIFF_LO)]
            dd, se_d, t_d, df_d = contrast(a, b)
            ok("the cpk1 cliff is far outside noise in ITS OWN batch",
               abs(t_d) > 10, "%+.3f pp se %.3f t %+.2f df %d"
               % (dd, se_d, t_d, df_d))
            ok("the two cliff edges differ by barely any parameter SHARE",
               abs(param_share(KCLIFF_LO) - param_share(KCLIFF_HI)) < 0.02,
               "%.6f vs %.6f (delta %.6f)"
               % (param_share(KCLIFF_HI), param_share(KCLIFF_LO),
                  param_share(KCLIFF_LO) - param_share(KCLIFF_HI)))

    print("-- the network, and the three single-tensor steps over it ----------")
    ok("the frozen NAME table has %d entries" % N_TENSORS,
       len(TENSOR_NAMES) == N_TENSORS, "%d" % len(TENSOR_NAMES))
    ok("the frozen NUMEL table has %d entries" % N_TENSORS,
       len(TENSOR_NUMEL) == N_TENSORS, "%d" % len(TENSOR_NUMEL))
    ok("the frozen NUMEL table sums to %d parameters" % N_PARAMS,
       sum(TENSOR_NUMEL) == N_PARAMS, "%d" % sum(TENSOR_NUMEL))
    ok("every tensor name is distinct", len(set(TENSOR_NAMES)) == N_TENSORS)
    ok("STRUCT tiles 1..%d exactly once" % N_TENSORS,
       [i for lo, hi, _ in STRUCT for i in range(lo, hi + 1)]
       == list(range(1, N_TENSORS + 1)))
    ok("layer4.0 occupies tensors 46..54",
       [(lo, hi) for lo, hi, n in STRUCT if n == "layer4.0"] == [(46, 54)])
    ok("6 registered arms", len(ARMS) == 6, str(list(ARMS)))
    ok("the grid is the four CONSECUTIVE cuts 49..52",
       list(KGRID) == [49, 50, 51, 52])
    ok("every cut arm has m = 2", all(arm_m(ktag(k)) == 2 for k in KGRID))
    ok("every cut spec covers the model",
       all(sum(arm_sizes(ktag(k))) == N_TENSORS for k in KGRID))
    ok("the anchors bracket every cut arm in m",
       arm_m(TAG_LO) == 1 and arm_m(TAG_HI) == N_TENSORS)
    ok("the three steps tile the cliff exactly once",
       [a for a, _ in STEPS] == [49, 50, 51]
       and [b for _, b in STEPS] == [50, 51, 52]
       and STEPS[0][0] == KCLIFF_HI and STEPS[-1][1] == KCLIFF_LO)
    print("      %-12s %-30s %-9s %-11s %s"
          % ("step", "tensor that moves", "params", "share of it", "structure"))
    tot_moved = 0
    for a, b in STEPS:
        names, numel = moved_tensor(a, b)
        want_name, want_numel = STEP_TENSOR[(a, b)]
        tot_moved += numel
        ok("step k=%d -> k=%d moves EXACTLY ONE tensor" % (a, b), len(names) == 1,
           str(list(names)))
        ok("step k=%d -> k=%d moves %s" % (a, b, want_name),
           list(names) == [want_name], str(list(names)))
        ok("step k=%d -> k=%d moves %d parameters" % (a, b, want_numel),
           numel == want_numel, "%d" % numel)
        print("      %-12s %-30s %-9d %-11.5f %s"
              % ("k=%d->%d" % (a, b), want_name, want_numel,
                 want_numel / 132096.0, struct_at(b)))
    ok("the three steps move 132,096 parameters in total", tot_moved == 132096,
       "%d" % tot_moved)
    ok("that is the whole k=49 -> k=52 move and nothing else",
       tot_moved == sum(TENSOR_NUMEL[KCLIFF_HI:KCLIFF_LO]))
    ok("the moved mass is 2.69 %% of the fine group at k=49",
       abs(tot_moved / float(fine_params(KCLIFF_HI)) - 0.026931) < 1e-5,
       "%.6f of %d" % (tot_moved / float(fine_params(KCLIFF_HI)),
                       fine_params(KCLIFF_HI)))
    ok("the projection is 99.2 %% of the moved parameters (A1's premise)",
       abs(STEP_TENSOR[MASS_STEP][1] / float(tot_moved) - 0.992249) < 1e-5,
       "%.6f" % (STEP_TENSOR[MASS_STEP][1] / float(tot_moved)))
    ok("the MATCHED CONTROL: the two BN tensors hold IDENTICAL counts",
       STEP_TENSOR[SCALE_STEP][1] == STEP_TENSOR[SHIFT_STEP][1] == 512,
       "%d vs %d" % (STEP_TENSOR[SCALE_STEP][1], STEP_TENSOR[SHIFT_STEP][1]))
    ok("the scale precedes the shift in named_parameters()",
       TENSOR_NAMES.index("layer4.0.bn2.weight")
       + 1 == TENSOR_NAMES.index("layer4.0.bn2.bias"))
    ok("the conv the BN scales is ALREADY coarse at k=49",
       TENSOR_NAMES.index("layer4.0.conv2.weight") + 1 <= KCLIFF_HI,
       "layer4.0.conv2.weight is tensor %d with %d parameters"
       % (TENSOR_NAMES.index("layer4.0.conv2.weight") + 1,
          TENSOR_NUMEL[TENSOR_NAMES.index("layer4.0.conv2.weight")]))
    ok("param_share is strictly increasing across the four cuts",
       all(param_share(KGRID[i]) < param_share(KGRID[i + 1])
           for i in range(len(KGRID) - 1)))
    for k in KGRID:
        print("      k=%-3d %-9s coarse %9d  fine %9d  share %.6f  %s"
              % (k, kspec(k), sum(TENSOR_NUMEL[:k]), fine_params(k),
                 param_share(k), struct_at(k)))
    ok("group_params agrees with param_share at every cut",
       all(abs(group_params(arm_sizes(ktag(k)))[0] / float(N_PARAMS)
               - param_share(k)) < 1e-12 for k in KGRID))
    ok("group_params of every cut sums to the whole network",
       all(sum(group_params(arm_sizes(ktag(k)))) == N_PARAMS for k in KGRID))
    ok("no cut in this batch lands on a layer4.0 module edge",
       all(k not in (45, 54) for k in KGRID))

    print("-- the power arithmetic: EVERY bar in the SE of ITS OWN contrast ---")
    se_mean = SIGMA_W / math.sqrt(3.0)
    ok("SE of a 3-seed mean", abs(se_mean - 0.4183) < 0.002, "%.4f pp" % se_mean)
    ok("SE of a difference of two 3-seed means",
       abs(SIGMA_W * math.sqrt(2 / 3.0) - 0.5916) < 0.002,
       "%.4f pp" % (SIGMA_W * math.sqrt(2 / 3.0)))
    ok("SE(CAPTURE difference) at n=3, the headline denominator",
       abs(SE_ARM_DIFF - 0.012654) < 5e-5, "%.6f" % SE_ARM_DIFF)
    ok("SE(DROP_i - DROP_j) for two ADJACENT steps sharing an arm",
       abs(SE_ADJ_STEPS - 0.021917) < 5e-5, "%.6f" % SE_ADJ_STEPS)
    ok("SE(DROP1 - DROP3), the two disjoint end steps",
       abs(SE_END_STEPS - 0.017897) < 5e-5, "%.6f" % SE_END_STEPS)
    ok("SE(DROP1 - TOTAL/3) = SE(DROP3 - TOTAL/3)",
       abs(SE_EVEN_13 - 0.011160) < 5e-5, "%.6f" % SE_EVEN_13)
    ok("SE(DROP2 - TOTAL/3)", abs(SE_EVEN_2 - 0.013338) < 5e-5,
       "%.6f" % SE_EVEN_2)
    print("      BAR                 value      in SE of its own contrast")
    print("      CLIFF_BAR (R1)      %.4f     %5.1f SE   (= %.3f pp)"
          % (CLIFF_BAR, CLIFF_BAR / SE_ARM_DIFF, CLIFF_BAR * ARCH_GAP))
    print("      EVEN_BAND (R2)      %.4f     %5.1f SE on DROP1/DROP3, "
          "%.1f SE on DROP2" % (EVEN_BAND, EVEN_BAND / SE_EVEN_13,
                                EVEN_BAND / SE_EVEN_2))
    print("      SEPARATION_BAR (R3) %.4f     %5.1f SE adjacent, %.1f SE "
          "end-to-end" % (SEPARATION_BAR, SEPARATION_BAR / SE_ADJ_STEPS,
                          SEPARATION_BAR / SE_END_STEPS))
    print("      ASYM_BAR (Q5)       %.4f     %5.1f SE   (DROP1 vs DROP2, "
          "adjacent)" % (ASYM_BAR, ASYM_BAR / SE_ADJ_STEPS))
    print("      cpk1's own TOTAL    %.4f     %5.1f SE   (= %.3f pp)"
          % (CPK1_TOTAL, CPK1_TOTAL / SE_ARM_DIFF, CPK1_TOTAL * ARCH_GAP))
    ok("CLIFF_BAR is >= 8 SE of its own contrast",
       CLIFF_BAR / SE_ARM_DIFF >= 8, "%.1f SE" % (CLIFF_BAR / SE_ARM_DIFF))
    ok("EVEN_BAND is >= 3 SE of its own contrast",
       min(EVEN_BAND / SE_EVEN_13, EVEN_BAND / SE_EVEN_2) >= 3,
       "%.1f SE" % min(EVEN_BAND / SE_EVEN_13, EVEN_BAND / SE_EVEN_2))
    ok("SEPARATION_BAR is >= 2 SE of its own contrast, and is a CONJUNCT with "
       "the majority clause, never used alone",
       SEPARATION_BAR / SE_ADJ_STEPS >= 2,
       "%.1f SE" % (SEPARATION_BAR / SE_ADJ_STEPS))
    ok("cpk1's TOTAL is more than 20 SE, so R1 has real power",
       CPK1_TOTAL / SE_ARM_DIFF > 20, "%.1f SE" % (CPK1_TOTAL / SE_ARM_DIFF))

    print("-- the verdict trees, exercised in EVERY direction -----------------")
    # R1
    ok("CLIFF-REPRODUCES on a cliff the size cpk1 measured",
       cliff_verdict(CPK1_TOTAL) == "CLIFF-REPRODUCES")
    ok("CLIFF-REPRODUCES exactly AT the bar", cliff_verdict(CLIFF_BAR)
       == "CLIFF-REPRODUCES")
    ok("CLIFF-NOT-REPRODUCED just below the bar",
       cliff_verdict(CLIFF_BAR - 1e-9) == "CLIFF-NOT-REPRODUCED")
    ok("CLIFF-NOT-REPRODUCED on a vanished cliff",
       cliff_verdict(0.02) == "CLIFF-NOT-REPRODUCED")
    ok("CLIFF-NOT-REPRODUCED on a REVERSED cliff",
       cliff_verdict(-0.30) == "CLIFF-NOT-REPRODUCED")
    ok("a voided cliff names NO tensor whatever the drops look like",
       named_tensor([0.10, 0.0, 0.0], 0.10) is None)
    # R2
    even = [0.364021 / 3.0] * 3
    ok("SPLIT-EVEN on an exactly even split",
       split_verdict(even, sum(even)) == "SPLIT-EVEN-NO-TENSOR-NAMED")
    ok("an even split names NO tensor",
       named_tensor(even, sum(even)) is None)
    nearly = [0.121 + 0.049, 0.121, 0.364021 - 0.121 - 0.121 - 0.049]
    ok("SPLIT-EVEN still, at 0.049 from the even split (just inside the band)",
       split_verdict(nearly, sum(nearly)) == "SPLIT-EVEN-NO-TENSOR-NAMED",
       "drops %s, TOTAL/3 %.4f" % (["%.4f" % d for d in nearly],
                                   sum(nearly) / 3.0))
    # R3 / naming
    a2win = [0.28, 0.03, 0.054021]
    ok("SINGLE-TENSOR-MAJORITY when one step carries it (A2's shape)",
       split_verdict(a2win, sum(a2win)) == "SINGLE-TENSOR-MAJORITY")
    ok("...and the named tensor is layer4.0.bn2.weight",
       named_tensor(a2win, sum(a2win)) == "layer4.0.bn2.weight",
       str(named_tensor(a2win, sum(a2win))))
    a1win = [0.02, 0.014021, 0.33]
    ok("SINGLE-TENSOR-MAJORITY on A1's shape too",
       split_verdict(a1win, sum(a1win)) == "SINGLE-TENSOR-MAJORITY")
    ok("...and the named tensor is layer4.0.shortcut.0.weight",
       named_tensor(a1win, sum(a1win)) == "layer4.0.shortcut.0.weight",
       str(named_tensor(a1win, sum(a1win))))
    mid = [0.19, 0.17, 0.004021]
    ok("SPLIT-UNEVEN-NO-MAJORITY when the leader has a majority but no "
       "SEPARATION", split_verdict(mid, sum(mid)) == "SPLIT-UNEVEN-NO-MAJORITY",
       "share %.4f, lead %.4f vs bar %.2f"
       % (mid[0] / sum(mid), mid[0] - mid[1], SEPARATION_BAR))
    ok("...and it names NO tensor", named_tensor(mid, sum(mid)) is None)
    nomaj = [0.17, 0.014021, 0.18]
    ok("SPLIT-UNEVEN-NO-MAJORITY when no step reaches half",
       split_verdict(nomaj, sum(nomaj)) == "SPLIT-UNEVEN-NO-MAJORITY",
       "best share %.4f" % (max(nomaj) / sum(nomaj)))
    ok("...and it names NO tensor", named_tensor(nomaj, sum(nomaj)) is None)
    ok("R2 outranks R3: an even split is never a majority",
       split_verdict(even, sum(even)) == "SPLIT-EVEN-NO-TENSOR-NAMED")
    neg = [0.42, -0.06, 0.004021]
    ok("a step of the WRONG SIGN is allowed and can push a share above 1",
       split_verdict(neg, sum(neg)) == "SINGLE-TENSOR-MAJORITY"
       and neg[0] / sum(neg) > 1.0, "share %.4f" % (neg[0] / sum(neg)))
    # A1 / A2 / Q5 trees
    ok("MASS-PROPORTIONAL-CONSISTENT at the mass share A1 implies",
       mass_verdict(0.992249) == "MASS-PROPORTIONAL-CONSISTENT")
    ok("MASS-PROPORTIONAL-CONSISTENT exactly at MASS_HIGH",
       mass_verdict(MASS_HIGH) == "MASS-PROPORTIONAL-CONSISTENT")
    ok("MASS-UNRESOLVED just below it",
       mass_verdict(MASS_HIGH - 1e-9) == "MASS-UNRESOLVED")
    ok("MASS-UNRESOLVED exactly at MASS_LOW", mass_verdict(MASS_LOW)
       == "MASS-UNRESOLVED")
    ok("MASS-REFUTED just below MASS_LOW",
       mass_verdict(MASS_LOW - 1e-9) == "MASS-REFUTED")
    ok("BN-LEVERAGE-FAVOURED when the BN pair carries most of it",
       bn_verdict(0.85) == "BN-LEVERAGE-FAVOURED")
    ok("BN-LEVERAGE-UNRESOLVED exactly at BN_HIGH",
       bn_verdict(BN_HIGH) == "BN-LEVERAGE-UNRESOLVED")
    ok("BN-LEVERAGE-REFUTED when the BN pair is negligible",
       bn_verdict(0.008) == "BN-LEVERAGE-REFUTED")
    ok("BN-LEVERAGE-UNRESOLVED exactly at BN_LOW",
       bn_verdict(BN_LOW) == "BN-LEVERAGE-UNRESOLVED")
    ok("A1 and A2 cannot BOTH be favoured (the shares sum to 1)",
       not (mass_verdict(0.95) == "MASS-PROPORTIONAL-CONSISTENT"
            and bn_verdict(1 - 0.95) == "BN-LEVERAGE-FAVOURED"))
    ok("SCALE-SHIFT-ASYMMETRIC when two 512-param tensors disagree",
       scaleshift_verdict(0.28, 0.03) == "SCALE-SHIFT-ASYMMETRIC")
    ok("SCALE-SHIFT-SYMMETRIC exactly at ASYM_BAR",
       scaleshift_verdict(0.10, 0.10 - ASYM_BAR) == "SCALE-SHIFT-SYMMETRIC")
    ok("SCALE-SHIFT-ASYMMETRIC just above ASYM_BAR",
       scaleshift_verdict(0.10, 0.10 - ASYM_BAR - 1e-9)
       == "SCALE-SHIFT-ASYMMETRIC")
    ok("drops_from_caps reads the grid in k order",
       drops_from_caps({49: 0.70, 50: 0.60, 51: 0.55, 52: 0.33})
       == [0.70 - 0.60, 0.60 - 0.55, 0.55 - 0.33])
    ok("the three drops always sum to TOTAL",
       abs(sum(drops_from_caps({49: 0.7, 50: 0.42, 51: 0.41, 52: 0.33}))
           - (0.7 - 0.33)) < 1e-12)

    print("-- the parsers, on adversarial input -------------------------------")
    eff, dup = parse_args_line("--alg-meta Lion --dataset CIFAR100 --alg-meta RMSProp")
    ok("a repeated flag is DETECTED", "alg-meta" in dup, str(dup))
    ok("argparse's last-wins is reproduced", eff["alg-meta"] == "RMSProp")
    line = ("--dataset CIFAR100 --NN-name ResNet18_c100 --stepsize-groups %s "
            "--seed 0" % kspec(50))
    eff2, dup2 = parse_args_line(line)
    ok("no false positive on a clean line", not dup2)
    ok("the bracket spec survives tokenisation",
       eff2.get("stepsize-groups") == kspec(50), eff2.get("stepsize-groups"))
    ok("arm_of reads k=50 from the flags", arm_of(eff2) == "k50")
    for tag, spec in ARMS.items():
        ok("arm_of round-trips %s" % tag,
           arm_of({"dataset": DSET, "NN-name": NET,
                   "stepsize-groups": spec}) == tag)
    ok("arm_of rejects the CIFAR-10 network",
       arm_of({"dataset": "CIFAR10", "NN-name": "ResNet18",
               "stepsize-groups": "layerwise"}) is None)
    for off in ("[45,17]", "[47,15]", "[48,14]", "[53,9]", "[55,7]"):
        ok("arm_of rejects the off-grid cut %s" % off,
           arm_of({"dataset": DSET, "NN-name": NET,
                   "stepsize-groups": off}) is None)
    ok("tail5 needs all 5 tail epochs",
       tail5({e: 50.0 for e in range(95, 100)}) == 50.0
       and tail5({e: 50.0 for e in range(96, 100)}) is None)
    ok("tail5 ignores everything but the tail",
       abs(tail5(dict([(e, 1.0) for e in range(95)]
                      + [(e, 7.0) for e in range(95, 100)])) - 7.0) < 1e-9)

    print("-- the partition gate, on a good manifest and five bad ones --------")
    lines = ["NUM_PARAM_TENSORS %d" % N_TENSORS, "TOTAL_PARAMS %d" % N_PARAMS]
    for spec in ARMS.values():
        s = spec_sizes(spec)
        if s is None:
            continue
        lines.append("SPEC %s M %d SIZES %s PARAMS %s FIRSTGROUP %s LASTGROUP %s"
                     % (spec, len(s), ",".join(str(x) for x in s),
                        ",".join(str(x) for x in group_params(s)),
                        " ".join(TENSOR_NAMES[:s[0]]),
                        " ".join(TENSOR_NAMES[s[0]:])))
    tmp = os.path.join(os.environ.get("TMPDIR", "/tmp"), "cts1_selftest_manifest.txt")

    def _write(txt):
        with open(tmp, "w") as fh:
            fh.write(txt + "\n")
        return read_manifest(tmp)

    m_ok = _write("\n".join(lines))
    ok("a correct manifest passes G1", gate1(m_ok) == [], str(gate1(m_ok)))

        "layer4.0.bn2.weight layer4.0.bn2.bias layer4.0.shortcut.0.weight",

       any("by NAME" in w for w in gate1(m)), str(gate1(m)[:1]))
    m = _write("\n".join(l for l in lines if "[51,11]" not in l))
    ok("a manifest missing an arm FAILS G1",
       any("no manifest line" in w for w in gate1(m)))
    ok("...and FAILS G1b too", any("missing" in w for w in gate1b(m)))
    m = _write("\n".join(lines).replace("NUM_PARAM_TENSORS 62",
                                        "NUM_PARAM_TENSORS 60"))
    ok("a manifest with the wrong tensor count FAILS G1",
       any("NUM_PARAM_TENSORS" in w for w in gate1(m)))
    ok("a missing manifest FAILS G1 and G1b",
       gate1(read_manifest(tmp + ".nope")) != []
       and gate1b(read_manifest(tmp + ".nope")) != [])
    # G1b must catch a manifest that is internally consistent but moves the

    swapped = []
    for l in lines:
        if l.startswith("SPEC [50,12]"):
            names = list(TENSOR_NAMES)
            names[49], names[50] = names[50], names[49]
            l = ("SPEC [50,12] M 2 SIZES 50,12 PARAMS %s FIRSTGROUP %s "

                 % (",".join(str(x) for x in group_params([50, 12])),

    ok("a manifest that moves the WRONG tensor at k=49->50 FAILS G1b",
       any("moves" in w for w in gate1b(m)), str(gate1b(m)[:1]))
    try:
        os.remove(tmp)
    except OSError:
        pass

    print("-- the batch script's own text must match this registration --------")
    if not os.path.exists(BATCH_SH):
        ok("bin/cL1_cliff_tensor_split.sh exists", False, BATCH_SH)
    else:
        txt = open(BATCH_SH).read()
        for label, needle in (
                ("run-name prefix cts1-", "cts1-"),
                ("dataset CIFAR100", "DSET=CIFAR100"),
                ("network ResNet18_c100", "NET=ResNet18_c100"),
                ("meta-stepsize 1e-3", "MST=1e-3"),
                ("alpha0 1e-6", "ALPHA0=1e-6"),
                ("100 epochs", "EPOCHS=100"),

                ("box -15:-2.3026", 'CLIP="-15:-2.3026"'),

                ("SGDm base", "--alg-base SGDm"),

           "6 arms x 3 seeds = 18 jobs" in txt)

    print("=" * 78)
    if fails:
        print("SELFTEST FAIL -- %d checks failed" % len(fails))
        for f in fails:
            print("   %s" % f)
        return 1
    print("SELFTEST PASS")
    return 0

def score(runs_dir, manifest):
    paths = sorted(glob.glob(os.path.join(runs_dir, "**", PREFIX + "*.out"),
                             recursive=True))
    print("cL1_cts1_score -- `cts1`, the SINGLE-TENSOR resolution of the "
          "k=49 -> k=52 cliff")
    print("runs dir: %s" % runs_dir)
    print("%d .out files match %s*" % (len(paths), PREFIX))
    if not paths:
        print("VERDICT: NO DATA -- the batch has not produced any output yet.")
        return 3
    recs = [r for r in (read_out(p) for p in paths) if r]
    print("%d carry an ARGS line" % len(recs))

    print("\n-- THE TWO ACCOUNTS, restated before the numbers --------------------")
    print("   A1 PARAMETER MASS (the RIVAL TO BEAT, inherited): the collapse "
          "follows the")
    print("      parameters, so %s (%d params, %.3f%% of the "
          "move) carries it."
          % (STEP_TENSOR[MASS_STEP][0], STEP_TENSOR[MASS_STEP][1],
             100.0 * STEP_TENSOR[MASS_STEP][1] / 132096.0))
    print("   A2 BATCHNORM GAIN LEVERAGE (this agent's point prediction): a "
          "group's cost is")
    print("      how badly its members disagree about a stepsize, not its "
          "parameter count,")
    print("      so %s (%d params) carries it -- and DROP1 >> "
          "DROP2 even though" % (STEP_TENSOR[SCALE_STEP][0],
                                 STEP_TENSOR[SCALE_STEP][1]))
    print("      layer4.0.bn2.weight and layer4.0.bn2.bias hold IDENTICAL "
          "parameter counts.")
    print("\n-- REGISTERED PREDICTIONS ------------------------------------------")
    print("   Q1 (A2) the largest drop is DROP1, k=49->50, %s"
          % STEP_TENSOR[SCALE_STEP][0])
    print("   Q2 (A1) the largest drop is DROP3, k=51->52, %s"
          % STEP_TENSOR[MASS_STEP][0])
    print("   Q3 the cliff REPRODUCES: TOTAL >= %.4f (half of cpk1's %.6f)"
          % (CLIFF_BAR, CPK1_TOTAL))
    print("   Q4 the split is NOT even (some drop is > %.2f from TOTAL/3)"
          % EVEN_BAND)
    print("   Q5 SCALE-SHIFT ASYMMETRY: |DROP1 - DROP2| > %.2f" % ASYM_BAR)
    print("   Q6 TRAIN and TEST agree on which step is largest")
    print("\n-- REFUTATION BRANCHES, CHECKED IN THIS ORDER ----------------------")
    print("   R1 CLIFF-NOT-REPRODUCED  TOTAL < %.4f -> the question is VOID, "
          "no tensor" % CLIFF_BAR)
    print("      may be named, the deliverable is UNRESOLVED, and cpk1's arms "
          "may NOT be")
    print("      spliced in to rescue it.")
    print("   R2 SPLIT-EVEN            all three drops within %.2f of TOTAL/3 "
          "-> NOT a" % EVEN_BAND)
    print("      single-tensor effect; no tensor may be named; A1 and A2 both "
          "fail.")
    print("   R3 NO-MAJORITY           the leader does not exceed TOTAL/2, or "
          "does not beat")
    print("      the runner-up by %.2f -> reported as the largest, but NO "
          "tensor is named." % SEPARATION_BAR)

    print("\n-- G0 PROVENANCE (RULE 20): the arm is read from each run's own flags --")
    kept, voided = gate0(recs)
    for r, why in voided:
        print("   VOID %-32s %s" % (os.path.basename(r["path"]), "; ".join(why)))
    print("   %d runs pass provenance, %d VOID" % (len(kept), len(voided)))
    if not kept:
        print("VERDICT: VOID -- no run survived provenance.")
        return 2

    print("\n-- G1 PARTITION GATE: every split must be a fact about the network --")
    if manifest is None:
        for c in (os.path.join(runs_dir, "cts1", "PARTITION-MANIFEST.txt"),
                  os.path.join(runs_dir, "PARTITION-MANIFEST.txt")):
            if os.path.exists(c):
                manifest = c
                break
    man = read_manifest(manifest)
    why1 = gate1(man)
    print("   manifest: %s" % (manifest or "<not found>"))
    if man:
        for k in KGRID:
            d = man["specs"].get(kspec(k))
            if d:
                print("      k=%-3d %-9s sizes %-8s params %s  share %.6f"
                      % (k, kspec(k), ",".join(str(x) for x in d["sizes"]),
                         ",".join(str(x) for x in d["params"]),
                         d["params"][0] / float(man["n_params"] or N_PARAMS)))
                print("            coarse ends at %-28s fine starts at %s"
                      % (d["first"][-1], d["last"][0]))
    for w in why1:
        print("   FAIL %s" % w)
    if why1:
        print("VERDICT: VOID-PARTITION -- without the manifest, 'k = 50' is a "
              "claim about a string, not about a network.")
        return 2
    print("   PASS -- %d parameter tensors, %d parameters; all %d cut specs "
          "compose exactly as registered, in sizes, in parameter counts AND in "
          "tensor names" % (man["n_tensors"], man["n_params"], len(KGRID)))

    print("\n-- G1b THE MOVED TENSOR: which tensor leaves the FINE group, "
          "measured ----")
    why1b = gate1b(man)
    for a, b in STEPS:
        sa, sb = man["specs"].get(kspec(a)), man["specs"].get(kspec(b))
        if sa and sb:
            moved = [n for n in sa["last"] if n not in set(sb["last"])]
            print("   k=%d -> k=%d moves %-30s %8d params  (%.4f of the move)"
                  % (a, b, " ".join(moved), sb["params"][0] - sa["params"][0],
                     (sb["params"][0] - sa["params"][0]) / 132096.0))
    for w in why1b:
        print("   FAIL %s" % w)
    if why1b:
        print("VERDICT: VOID-PARTITION -- the live model does not move the "
              "tensors this batch is registered to move.")
        return 2
    print("   PASS -- each of the three steps moves EXACTLY ONE tensor, and it "
          "is the registered one")

    print("\n-- G2 COMPLETENESS: >= 95/100 epochs and all 5 tail epochs present --")
    cells = collections.defaultdict(list)
    incomplete = []
    for r, _ in kept:
        done = (max(r["test"]) + 1) if r["test"] else 0
        p5 = tail5(r["test"])
        t5 = tail5(r["train"])
        if p5 is None or done < 0.95 * EPOCHS:
            incomplete.append((os.path.basename(r["path"]), done))
            continue
        cells[r["arm"]].append((p5, t5, r["seed"], os.path.basename(r["path"])))
    for n, d in incomplete:
        print("   DROPPED %-34s %d epochs" % (n, d))
    print("   %d complete runs in %d arms" % (sum(len(v) for v in cells.values()),
                                              len(cells)))
    thin = [a for a in ARMS if len(cells.get(a, [])) < 2]
    if thin:
        print("   arms with fewer than 2 usable seeds: %s" % ", ".join(thin))
        print("VERDICT: VOID -- a registered arm has fewer than 2 usable seeds.")
        return 2

    print("\n-- the arms (TRAIN reported alongside TEST at every arm) -------------")
    print("   %-4s %-9s %-3s %-9s %-8s %-9s %-9s  %s"
          % ("arm", "spec", "n", "plateau5", "se", "train5", "share", "seeds"))
    p5s, t5s = {}, {}
    for arm in ARMS:
        v = cells.get(arm, [])
        if not v:
            continue
        a = [x for x, _, _, _ in v]
        b = [y for _, y, _, _ in v if y is not None]
        p5s[arm] = a
        if b:
            t5s[arm] = b
        m, se = mean_se(a)
        k = int(arm[1:]) if arm not in (TAG_LO, TAG_HI) else None
        print("   %-4s %-9s %-3d %-9.3f %-8.3f %-9s %-9s  %s"
              % (arm, ARMS[arm][:9], len(v), m, se,
                 ("%.3f" % statistics.mean(b)) if b else "--",
                 ("%.6f" % param_share(k)) if k else "--",
                 " ".join("s%s=%.2f" % (s, x) for x, _, s, _ in
                          sorted(v, key=lambda z: str(z[2])))))

    print("\n-- G3 ARCHIVE GATE: does this batch reproduce the archived cell? ----")
    g = archived_arms()
    bad = []
    for arm, key, want in ((TAG_LO, "scalar", ARCH_SCAL),
                           (TAG_HI, "layerwise", ARCH_LAY)):
        got = statistics.mean(p5s[arm])
        ref = g[key][0] if key in g else float("nan")
        d = got - ref
        print("   %-4s in-batch %.3f   archived %.3f (n=%d, %d batches)   %+.3f pp"
              % (arm, got, ref, g[key][1], g[key][2], d))
        if not (abs(d) <= TREE_BAND):
            bad.append("%s off by %+.3f pp (band +/-%.2f)" % (arm, d, TREE_BAND))
    if bad:
        for b in bad:
            print("   FAIL %s" % b)
        print("VERDICT: VOID-TREE -- the batch does not reproduce the archived "
              "CIFAR-100 cell; nothing here is comparable.")
        return 2
    print("   PASS -- both anchors inside +/-%.2f pp" % TREE_BAND)

    print("\n-- G4 MANIFOLD GATE: there must be a gap to capture ------------------")
    m_lo, m_hi = statistics.mean(p5s[TAG_LO]), statistics.mean(p5s[TAG_HI])
    gap = m_hi - m_lo
    print("   in-batch GAP = M(%s) - M(%s) = %+.3f pp   (archived %+.3f, bar %.3f)"
          % (TAG_HI, TAG_LO, gap, ARCH_GAP, MANIFOLD_BAR))
    if gap < MANIFOLD_BAR:
        print("VERDICT: VOID-NOMANIFOLD -- the granularity gap did not appear in "
              "this batch, so CAPTURE is a ratio of noise and no verdict is issued.")
        return 2
    print("   PASS")

    caps = {}
    for k in KGRID:
        t = ktag(k)
        if t in p5s:
            caps[k] = capture(statistics.mean(p5s[t]), m_lo, m_hi)
    if len(caps) != len(KGRID):
        print("VERDICT: VOID -- the grid is incomplete (%d of %d cuts)."
              % (len(caps), len(KGRID)))
        return 2

    t_lo = statistics.mean(t5s[TAG_LO]) if TAG_LO in t5s else None
    t_hi = statistics.mean(t5s[TAG_HI]) if TAG_HI in t5s else None
    tcaps = {}
    if t_lo is not None and t_hi is not None and t_hi != t_lo:
        for k in KGRID:
            t = ktag(k)
            if t in t5s:
                tcaps[k] = capture(statistics.mean(t5s[t]), t_lo, t_hi)

    print("\n-- THE FOUR CUTS ------------------------------------------------------")
    print("   %-4s %-9s %-9s %-9s %-9s %-10s  %s"
          % ("k", "spec", "plateau5", "CAPTURE", "TRAINCAP", "share", "structure"))
    for k in KGRID:
        print("   %-4d %-9s %-9.3f %+9.4f %9s %-10.6f  %s"
              % (k, kspec(k), statistics.mean(p5s[ktag(k)]), caps[k],
                 ("%+.4f" % tcaps[k]) if k in tcaps else "--",
                 param_share(k), struct_at(k)))

    print("\n-- G5a THE CLIFF GATE (R1): is cpk1's cliff inside THIS batch? -------")
    total = caps[KCLIFF_HI] - caps[KCLIFF_LO]
    dd, se_d, t_d, df_d = contrast(p5s[ktag(KCLIFF_HI)], p5s[ktag(KCLIFF_LO)])
    cv = cliff_verdict(total)
    print("   TOTAL = CAPTURE(k=%d) - CAPTURE(k=%d) = %+.4f   "
          "(%+.3f pp, se %.3f, t %+.2f, df %d)"
          % (KCLIFF_HI, KCLIFF_LO, total, dd, se_d, t_d, df_d))
    print("   bar %.4f = half of cpk1's own %.6f = %.1f SE at the re-derived "
          "noise floor" % (CLIFF_BAR, CPK1_TOTAL, CLIFF_BAR / SE_ARM_DIFF))
    print("   VERDICT (the CLIFF): %s" % cv)
    if cv == "CLIFF-NOT-REPRODUCED":
        print("\n   R1 FIRES.  cts1 exists ONLY to decompose a cliff cpk1 "
              "measured.  The cliff")
        print("   is not in this batch, so there is nothing to decompose.  NO "
              "TENSOR IS NAMED,")
        print("   the three steps below are NOT reported as an answer, and "
              "cpk1's own k=49 and")
        print("   k=52 arms may NOT be spliced in to rescue the question -- "
              "batch, not seed, is")
        print("   the unit of replication (F(62,85) = 5.47).  What this batch "
              "HAS established is")
        print("   that the cliff is not stable across batches, which is itself "
              "a finding about")
        print("   cpk1 and is the only thing that may be reported.")
        print("\n   FINAL: CLIFF %s | SPLIT UNRESOLVED | TENSOR NONE "
              "| MASS UNRESOLVED | BN UNRESOLVED | SCALE-SHIFT UNRESOLVED" % cv)
        print("   DELIVERABLE: UNRESOLVED.")
        return 0
    print("   PASS -- the cliff is here, in this batch, and can be decomposed")

    print("\n-- G5b THE PRIMARY: WHICH SINGLE TENSOR CARRIES THE COLLAPSE? --------")
    drops = drops_from_caps(caps)
    shares = [d / total for d in drops]
    print("   %-11s %-30s %-9s %-11s %-11s %s"
          % ("step", "tensor that moves", "params", "dCAPTURE", "SHARE", "pp / se / t"))
    for i, (a, b) in enumerate(STEPS):
        name, numel = STEP_TENSOR[(a, b)]
        dd, se_d, t_d, df_d = contrast(p5s[ktag(a)], p5s[ktag(b)])
        print("   k=%-2d->%-4d %-30s %-9d %+11.4f %+11.4f  %+.3f / %.3f / %+.2f (df %d)"
              % (a, b, name, numel, drops[i], shares[i], dd, se_d, t_d, df_d))
    print("   TOTAL %+.4f   (the three drops sum to it by construction: %+.4f)"
          % (total, sum(drops)))
    print("   an even split would be %+.4f each" % (total / 3.0))
    print("   distance from the even split: %s"
          % "  ".join("DROP%d %+.4f (%.1f SE)"
                      % (i + 1, drops[i] - total / 3.0,
                         abs(drops[i] - total / 3.0)
                         / (SE_EVEN_2 if i == 1 else SE_EVEN_13))
                      for i in range(3)))

    sv = split_verdict(drops, total)
    tensor = named_tensor(drops, total)
    order = sorted(range(3), key=lambda i: -drops[i])
    print("\n   frozen bars: R1 CLIFF TOTAL<%.4f | R2 EVEN all within %.2f of "
          "TOTAL/3 | R3 the leader must exceed SHARE %.2f AND beat the "
          "runner-up by %.2f" % (CLIFF_BAR, EVEN_BAND, MAJORITY_SHARE,
                                 SEPARATION_BAR))
    print("   leader DROP%d SHARE %+.4f; runner-up DROP%d; lead %+.4f "
          "(%.1f SE)"
          % (order[0] + 1, shares[order[0]], order[1] + 1,
             drops[order[0]] - drops[order[1]],
             abs(drops[order[0]] - drops[order[1]])
             / (SE_END_STEPS if {order[0], order[1]} == {0, 2} else SE_ADJ_STEPS)))
    print("   VERDICT (the SPLIT):   %s" % sv)
    print("   VERDICT (the TENSOR):  %s"
          % (tensor if tensor else "NONE -- this batch is not entitled to name "
             "a tensor"))

    share3 = shares[2]
    share_bn = shares[0] + shares[1]
    mv = mass_verdict(share3)
    bv = bn_verdict(share_bn)
    ssv = scaleshift_verdict(drops[0], drops[1])
    print("\n   A1 PARAMETER MASS: SHARE(%s) = %+.4f "
          "(its parameter share of the move is %.4f)"
          % (STEP_TENSOR[MASS_STEP][0], share3,
             STEP_TENSOR[MASS_STEP][1] / 132096.0))
    print("   VERDICT (A1 MASS):     %s   [consistent at >= %.2f, refuted "
          "below %.2f]" % (mv, MASS_HIGH, MASS_LOW))
    print("   A2 BN GAIN LEVERAGE: SHARE(bn2.weight) + SHARE(bn2.bias) = %+.4f"
          % share_bn)
    print("   VERDICT (A2 BN):       %s   [favoured above %.2f, refuted below "
          "%.2f]" % (bv, BN_HIGH, BN_LOW))
    print("\n   THE MATCHED CONTROL -- two tensors, IDENTICAL parameter count "
          "(512 vs 512),")
    print("   adjacent in named_parameters(), same BatchNorm module.  Under ANY "
          "account in")
    print("   which the currency is parameter mass these two drops must be "
          "EQUAL.")
    dd12 = drops[0] - drops[1]
    print("   DROP1 (%s) %+.4f  vs  DROP2 (%s) %+.4f   diff %+.4f (%.1f SE, "
          "bar %.2f)"
          % (STEP_TENSOR[SCALE_STEP][0], drops[0], STEP_TENSOR[SHIFT_STEP][0],
             drops[1], dd12, abs(dd12) / SE_ADJ_STEPS, ASYM_BAR))
    print("   VERDICT (SCALE-SHIFT): %s" % ssv)

    print("\n   the registered predictions, scored:")
    print("      Q1 (A2) largest drop is DROP1 ...... %s  (largest is DROP%d)"
          % ("HELD" if order[0] == 0 else "REFUTED", order[0] + 1))
    print("      Q2 (A1) largest drop is DROP3 ...... %s"
          % ("HELD" if order[0] == 2 else "REFUTED"))
    print("      Q3 the cliff reproduces ............ %s  (TOTAL %+.4f, bar %.4f)"
          % ("HELD" if total >= CLIFF_BAR else "REFUTED", total, CLIFF_BAR))
    print("      Q4 the split is NOT even ........... %s"
          % ("HELD" if sv != "SPLIT-EVEN-NO-TENSOR-NAMED" else "REFUTED (R2)"))
    print("      Q5 SCALE-SHIFT asymmetry ........... %s  (|diff| %.4f, bar %.2f)"
          % ("HELD" if ssv == "SCALE-SHIFT-ASYMMETRIC" else "REFUTED",
             abs(dd12), ASYM_BAR))

    print("\n-- G6 DESCRIPTIVE, NEVER GATING --------------------------------------")
    if tcaps and len(tcaps) == len(KGRID):
        tdrops = drops_from_caps(tcaps)
        ttotal = tcaps[KCLIFF_HI] - tcaps[KCLIFF_LO]
        torder = sorted(range(3), key=lambda i: -tdrops[i])
        print("   TRAIN: train5 anchors %s %.3f -> %s %.3f (span %.3f pp)"
              % (TAG_LO, t_lo, TAG_HI, t_hi, t_hi - t_lo))
        print("   TRAIN drops: %s   TOTAL %+.4f"
              % ("  ".join("DROP%d %+.4f" % (i + 1, tdrops[i]) for i in range(3)),
                 ttotal))
        print("      Q6 TRAIN and TEST agree on the largest step .... %s  "
              "(train DROP%d, test DROP%d)"
              % ("HELD" if torder[0] == order[0] else "REFUTED",
                 torder[0] + 1, order[0] + 1))
        if torder[0] != order[0]:
            print("      FLAG T1 TRAIN-TEST-DIVERGENCE: the fitting curve and "
                  "the generalisation")
            print("      curve blame different tensors.  The coarse-partition "
                  "failure has been")
            print("      train-side in every dataset measured so far, so this "
                  "is a real caveat on")
            print("      any mechanism read off the TEST side.  Non-overriding.")
    else:
        print("   train5 unavailable in at least one arm; Q6 not evaluated")

    if any(caps[k] > 1.0 for k in KGRID):
        print("   NOTE: some m=2 arm has CAPTURE > 1, i.e. it beat layerwise. "
              "Reported; it changes no bar.")
    nonmono = [i for i in range(3) if drops[i] < 0]
    if nonmono:
        print("   NOTE: %d of the three steps RAISED capture (DROP%s negative). "
              "Reported; it changes no bar and can put a SHARE above 1."
              % (len(nonmono), ",".join(str(i + 1) for i in nonmono)))

    print("\n   CROSS-BATCH, DESCRIPTIVE ONLY (cpk1 is ANOTHER BATCH and batch "
          "is the unit")
    print("   of replication; only THIS batch's own TOTAL was tested, at G5a. "
          "These")
    print("   magnitudes gate nothing and no cpk1 arm was used in any number "
          "above):")
    c = cpk1_from_csv()
    if c:
        print("      cpk1 anchors scalar %.3f layerwise %.3f (gap %.3f); "
              "cts1 %.3f / %.3f (gap %.3f)"
              % (c["lo"], c["hi"], c["hi"] - c["lo"], m_lo, m_hi, gap))
        for k in (KCLIFF_HI, KCLIFF_LO):
            if k in c["caps"]:
                print("      k=%-3d cts1 CAPTURE %+.4f   cpk1 CAPTURE %+.4f   "
                      "%+.4f" % (k, caps[k], c["caps"][k], caps[k] - c["caps"][k]))
        print("      cliff TOTAL: cts1 %+.4f   cpk1 %+.4f   %+.4f"
              % (total, CPK1_TOTAL, total - CPK1_TOTAL))
    else:
        print("      cpk1 rows not found in the corpus; nothing printed.")

    print("\n-- SCOPE -------------------------------------------------------------")
    print("   CIFAR-100 / ResNet18_c100 / SGDm+Lion / ms 1e-3 / alpha0 1e-6 / 100 ep.")
    print("   EVERY cut arm has m = 2; the anchors are anchors, not rungs, so this")
    print("   batch says NOTHING about the group COUNT, about resnet18_blocks, about")
    print("   CIFAR-10, Tiny-ImageNet, ImageNet-489 or ResNet-50, about any other")
    print("   meta-stepsize or optimiser pair, or about any hierarchical/shrinkage")
    print("   operator (HIER is unset in every arm).  A tensor named here is named at")
    print("   ONE SITE in ONE architecture in ONE named_parameters() order; a")
    print("   different ordering of the same network is a different experiment, and")
    print("   'BN scales matter' would be a hypothesis this batch supports at exactly")
    print("   one site, never a law.  Normalised entropy stays DISQUALIFIED (cbl1 R5)")
    print("   and is NOT rehabilitated here whichever way this falls.")
    print("")
    print("   FINAL: CLIFF %s | SPLIT %s | TENSOR %s | MASS %s | BN %s | "
          "SCALE-SHIFT %s"
          % (cv, sv, tensor if tensor else "NONE", mv, bv, ssv))
    return 0


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs", default="/cluster/account_b/metaopt/runs")
    ap.add_argument("--manifest", default=None)
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    sys.exit(selftest() if a.selftest else score(a.runs, a.manifest))


if __name__ == "__main__":
    main()
