#
# =============================================================================
# THE QUESTION, STATED BEFORE ANY NUMBER IT PRODUCES IS READ
# =============================================================================
# `cvi1` (CORRECTIONS 203) isolated `bn8.weight` (tensor 23, 512 parameters)
# into its own step-size group on VGG11_bn_c100 and recovered the whole in-batch
# scalar->layerwise gap at 100 epochs (D_ISO +31.2833 pp; its exact twin
# `bn7.weight` recovered -0.0200).  It was measured at 100 epochs and at NO
# other.  ciso2 (CORRECTIONS 196) measured the ResNet analogue at 250 epochs
# (RHO 0.9977) but its token is UNRESOLVED-NOT-PINNED because PIN_OCC_MIN
# scored exact-clamp DWELL TIME and bucketed a group chattering within 1.2x of
# the floor with a free arm running at 86-131x.  cvh1 runs the horizon leg on
# VGG with that defect removed BY REGISTRATION (cIS2 is frozen and untouched).
#
# RE-DERIVED HERE FROM cvi1's OWN 1,500 ISO PROBE RECORDS (--selftest repeats
# it on any host that can see them; guard 1c2 requires that it RAN):
#   ISO beta[0] (the 25-tensor complement), seed 41 / 42 / 43:
#     terminal @ record 499   -13.480375 / -13.260286 / -13.526394
#     OLS slope, last 50 rec  -0.0304401 / -0.0235310 / -0.0292064 nats/record
#     OLS slope, last 100 rec -0.0267193 / -0.0254405 / -0.0307080
#     linear extrapolation to the -15.0 clamp (epoch):
#       last-50  109.9844 / 114.7865 / 110.0910
#       last-100 111.3747 / 113.6767 / 109.5975
#     last-quarter median alpha/alpha_floor 26.11 / 30.64 / 25.85 -- NOT pinned
#   ISO beta[1] (bn8.weight) reaches the clamp at epoch 20.2 / 19.8 / 20.0.
#   k01 reaches it at 36.0 x3; CTL's complement at 35.8 / 36.0 / 36.0.
#   ==> once the complement pins, EVERY tensor in ISO sits on exp(-15): the
#       network freezes.  The question is whether the rescue survives that.
#
# =============================================================================
# THE HORIZON, DERIVED -- NOT CHOSEN ROUND (the 190 method)
# =============================================================================
#   1. PIN_BOUND = 114.7865: the LATEST of the six cvi1 extrapolations above
#      (seed 42, last-50).  CORRECTING THE PROSE, NOT QUOTING IT: 203.7 says
#      the late descent "accelerates"; the re-derived slopes say that holds on
#      seed 41 only (last-50 14 % steeper than last-100) and NOT on seeds 42 /
#      43 (last-50 7.5 % / 4.9 % SHALLOWER).  So a linear extrapolation is NOT
#      guaranteed to be an upper bound here, which is why the bound takes the
#      LATEST of BOTH windows on ALL seeds, and why its support is empirical:
#      CALIBRATED ON THE ONE PRECEDENT THAT EXISTS: the same method, applied
#      to ciso1's records, bounded the ARM MAXIMUM of ciso2's measured first
#      touch on both arms it was tried on -- ISO predicted <= 104.7, measured
#      max 103.0; ONE predicted <= 182.1, measured max 151.0.  (196.5 found a
#      +2.7-epoch per-seed "violation" by pairing seed 22 with seed 32; that
#      pairing is arbitrary across two batches, and the arm-level bound held.)
#   2. W_MAX = 213.0: the LONGEST post-pin window this campaign has EVER
#      measured.  A post-pin window is the time an arm is observed after EVERY
#      one of its step-size groups is on the floor (190's usage: 63.2 for
#      ciso1's floored arms -- re-derived, that is CTRL s22's touch at 36.8;
#      ciso1's k01 touches at 37.0 x3, i.e. 63.0 -- when ciso1 ISO's isolated
#      trio had already been on the floor 77.6 epochs and was not counted).
#      Candidates, re-derived from the probe records:
#        ciso2 k01  first touch 37.2 / 37.0 / 37.0 -> 250 - 37.0 = 213.0  <- MAX
#        ciso2 ISO  first touch 102.2 / 103.0 / 99.4 -> 150.6
#        ciso2 ONE  148.4 / 151.0 / 135.8            -> 114.2
#        ciso1 CTRL / cvi1 k01 / cvg1 k01 (100 ep)   -> 63.2 / 64.0 / 64.2
#      NO batch longer than 100 epochs other than ciso2 has a probe record on
#      disk (982 probe directories searched): the 772-epoch cts3/cpk2/cpk3 and
#      the 300/600-epoch CIFAR-10 rows have NO beta trajectory, so their pin
#      epoch is UNMEASURED and they cannot supply a post-pin window.  DISCLOSED:
#      a GROUP-level window (a single group on the floor while the rest of the
#      network trains) is longer -- ciso2 ISO's trio, 228.0 -- but it does not
#      observe a frozen network and 190 did not count its analogue.
#   3. E = ceil(PIN_BOUND + W_MAX) = ceil(327.7865) = 328.  The r<=2 pin
#      definition used by the gate below gives ceil(108.8952 + 214.4) = 324,
#      so 328 is the maximum over both definitions.  ISO's complement then gets
#      >= 328 - 114.7865 = 213.2 post-pin epochs -- as long as the longest
#      window ever measured -- IF the extrapolation holds.  If it does not,
#      the POST-PIN-WINDOW stamp says so; the primary is unaffected.
#
# THE 100-EPOCH CONTROL IS IN THE SAME RUNS.  `num_epochs` occurs in exactly
# TWO places in the live train.py (sha 3fea309e..., the file ciso2 ran on): the
# argparse declaration (line 42) and `for epoch in range(args.num_epochs)`
# (line 135).  No scheduler, total-step count or warmup is keyed to it.  The
# ONE schedule hook anywhere is HF.py's PATCH_SCHED: SCHED=none is truthy, so
# `_sched_factor()` IS called -- and returns exactly 1.0, because
# SCHED_TOTAL is unset (T = 0) and `sched_total_effective` is referenced at
# HF.py:483 and defined NOWHERE in the harness.  The launcher's guard 4h2
# PROVES that on the live class for counters up to 1e7, and guard 4h3 refuses
# if SCHED_TOTAL / SCHED_WARMUP / SCHED_MIN / COS_* are set in the submitting
# shell (--export=ALL would carry them).  --max-time 999:00:00 keeps train.py's
# own time break from firing.  So epochs 95..99 of a 328-epoch run ARE a
# 100-epoch run of an identical configuration: a WITHIN-RUN paired control.
#
# =============================================================================
# THE PIN GATE, REDESIGNED -- AND MOVED AFTER THE PRIMARY, AS A STAMP
# =============================================================================
# PIN_OCC_MIN's defect (196.5): it measured the fraction of records at EXACTLY
# beta = -15.000.  A group re-clamped every few records scores 0.34-0.43 while
# its step size never exceeds 1.19x the floor.  The replacement measures STEP-
# SIZE MAGNITUDE:
#     r = alpha / alpha_floor = exp(beta - BETA_LO)
#     per seed: MED_R = median of r over the LAST QUARTER of the records
#     PINNED  MED_R <= PIN_RATIO_MAX  = 2.0   (step within 2x of the floor)
#     FREE    MED_R >= FREE_RATIO_MIN = 10.0  (an order of magnitude above it)
#     NEITHER otherwise                       (mid-descent)
#     arm class = the class held by >= 2 of 3 seeds, else NEITHER.
# CALIBRATED ON EVERY RECORD SET THAT EXISTS (re-derived by --selftest where
# reachable; frozen here):
#   ciso2 ISO complement MED_R 1.0010 / 1.0025 / 1.0010  -> PINNED  (the arm
#     PIN_OCC_MIN called NOT PINNED: exact dwell 0.3974 / 0.3365 / 0.4295)
#   ciso2 ONE            1.0274 / 1.0555 / 1.0481        -> PINNED
#   ciso2 k62 beta[0]    87.43 / 69.32 / 105.10          -> FREE
#   ciso1 ISO @100       10.80 / 7.86 / 7.89             -> NEITHER (1 FREE)
#   ciso1 ONE @100       57.55 / 46.64 / 48.07           -> FREE
#   cvi1  ISO @100       26.11 / 30.64 / 25.85           -> FREE
#   every k01            1.0000                          -> PINNED
# The largest PINNED-arm median is 1.0555 and the smallest FREE median is
# 25.85: both thresholds sit an order of magnitude inside that gap.
#
# WHY A STAMP AND NOT A GATE.  The question is "does the rescue survive the
# horizon".  RHO answers it whether or not the complement pinned; the pin
# state says WHAT the horizon exercised (a frozen network, or a still-training
# one).  ciso2 put the pin gate BEFORE the primary and lost a measured
# RHO 0.9977 to it.  Here the branch is decided by the primary alone and the
# pin state rides beside it: COMPLEMENT-PINNED + FROZEN-AT-PIN is "the rescue
# survives freezing"; COMPLEMENT-FREE is "the rescue survives 328 epochs of a
# network that never froze", which is a different and weaker sentence.
#
# =============================================================================
# THE NOISE FLOOR -- RE-DERIVED AT REGISTRATION, cvh1- EXCLUDED, THEN FROZEN
# =============================================================================
# Corpus at registration: 2,863 rows, sha256 870c4003...4a3a.  Six defensible
# pooled within-cell SDs of plateau5, superseded / collapsed / incomplete
# dropped, every reader excluding `cvh1-`:
#   VGG_SL_100      0.317817  df 16   VGG, scalar+layerwise, 100 ep (cvi1's def)
#   VGG_ALL_100     0.290870  df 30   VGG, batch x granularity, 100 ep
#   NARROW_SL_100   0.586232  df 77   ResNet18_c100, scalar+layerwise (cvi1's)
#   R250            0.409934  df  8   ResNet18_c100, 250 ep (ciso2, in-batch)
#   R772            0.813024  df 30   ResNet18_c100, 772 ep
#   R100_M2_NARROW  0.925518  df 60   ResNet18_c100, m=2 [a,b], 100 ep (cIS2's)
#   SIGMA_FROZEN = max = 0.925518.  VGG's OWN variants are 2.9-3.2x smaller,
#   so the frozen bar is CONSERVATIVE on this architecture, and no corpus
#   owns a VGG row beyond 100 epochs -- that extrapolation is disclosed.
#   SE_FROZEN = SIGMA_FROZEN * sqrt(2/3) = 0.755682;  READ_BAR_FROZEN = 1.511365.
#
# FROZEN BARS (CORRECTIONS 207/208, O2).  EVERY bar, band and threshold in this
# file is a LITERAL.  score() reads NOTHING from results/all_runs.csv that can
# enter any bar, branch or token -- it opens the CSV only to print a
# descriptive row count.  The ONLY data-dependent term is this batch's OWN
# within-arm SD, and it enters ONLY as  SIGMA_USED = max(SIGMA_FROZEN,
# SIGMA_INBATCH),  which can widen the stamp bar and never narrow it.
# --selftest ASSERTS the invariance: the same synthetic runs give an identical
# FINAL line against the real corpus, against two corpora with invented rows
# injected at cvh1's own cells and at the VGG standard cell, and against no
# corpus at all -- and shows a live-reading scorer WOULD have flipped.
#
# =============================================================================
# THE STATISTICS AND THE BARS, FROZEN
# =============================================================================
#   P100(arm) = mean TEST over epochs 95..99 of that run; PE(arm) the same over
#   epochs 323..327.  T100 / TE on TRAIN.  3-seed arm means, ALL IN BATCH.
#   D_ISO_h = P_h(ISO) - P_h(k01).
#   PRIMARY:  RHO = D_ISO_E / D_ISO_100.  Bounded in NEITHER direction: the
#     numerator exceeds the denominator if ISO keeps improving after 100 epochs
#     and goes negative if ISO falls below the floor.
#   D100_MIN     = 10.0 pp  cvi1's RESCUE_BAR.  Below it there is no in-batch
#                           rescue to retain and RHO is not interpretable.
#   REFUTE_BAR   =  5.0 pp  = 6.62 SE.  ciso2's, carried: D_ISO_E at or below
#                           it means ISO has returned to the floor.
#   SUPPORT_BAR  = 15.0 pp  = 19.85 SE.  cvi1's IDENTITY_BAR: half this cell's
#                           gap (15.63).  Required jointly with SURVIVE_FRAC.
#   SURVIVE_FRAC =  0.80    ciso2's, carried, so the two horizons read on one
#                           scale.  Tolerated loss 0.20 x ~31.3 = ~6.3 pp =
#                           ~8.3 SE: a SCIENTIFIC threshold, stated as one.
#   ATTEN_MILD   =  0.50    RHO >= 0.50 sub-stamps ATTENUATE-MILD, else SEVERE.
#   DEAD_BAR     =  5.0 pp  a within-arm seed range above it, in any arm at
#                           either horizon, is UNRESOLVED-DIVERGED.
#   ANCHORS      k01 P100 in 35.3442 +- 3.702072, kL P100 in 66.2884 +- 3.702072
#                (corpus VGG means n 9 each, cvh1 excluded; half-width
#                BAND_K 4 x SIGMA_FROZEN).  FROZEN.  Outside -> UNRESOLVED-ANCHOR.
#   READ_BAR     2 x SE_USED, for every stamp-level comparison.
#
# =============================================================================
# THE BRANCH MAP, FIRST MATCH WINS
# =============================================================================
#   UNRESOLVED-MANIFEST-MISMATCH  no manifest, or the model / any partition is
#                                 not the frozen one (name AND numel, 1-based).

#   RESCUE-SURVIVES               RHO >= SURVIVE_FRAC AND D_ISO_E >= SUPPORT_BAR.

#   RHO:<x>
#   COMPLEMENT-PINNED | COMPLEMENT-NEITHER | COMPLEMENT-FREE     (the new gate)

#   ISO-TRAIN-FROZEN | ISO-TRAIN-MOVES        |TE(ISO) - T100(ISO)| vs READ_BAR

#   REPLICATES-CVI1 | DIFFERS-FROM-CVI1       |D_ISO_100 - 31.2833| vs READ_BAR;
#                                             BETWEEN-BATCH, NON-GATING.
#   MAGNITUDE-NOT-SEPARATED, NOT-A-ONE-VARIABLE-ABLATION   unconditional.

def chk(cond, label, extra=""):
    global _N_PASS, _N_FAIL
    if cond:
        _N_PASS += 1
    else:
        _N_FAIL += 1
    print("  %s %s%s" % ("PASS" if cond else "FAIL", label, (" -- " + extra) if extra else ""))
    return bool(cond)

def fmt(x, nd=4):

def spec_to_groups(spec, names=TENSOR_NAMES):
    """compose a registered spec into 1-based index groups, without torch."""
    if spec == "scalar":
        return [list(range(1, len(names) + 1))]
    if spec == "layerwise":
        return [[i] for i in range(1, len(names) + 1)]
    if not spec.startswith("sets:") or spec[5:].count("/") != 1:
        raise ValueError("unparsable spec %r" % spec)
    coarse, fine = spec[5:].split("/")
    g0 = []
    for part in coarse.split(","):
        part = part.strip()
        if "-" in part:
            a, b = part.split("-")
            g0.extend(range(int(a), int(b) + 1))
        elif part:
            g0.append(int(part))
    g1 = []
    for nm in fine.split(","):
        nm = nm.strip()
        if nm not in names:
            raise ValueError("spec names %r, which is not on the model" % nm)
        g1.append(names.index(nm) + 1)
    return [g0, g1]


def classify(med_r):
    if med_r is None or med_r != med_r:
        return "UNMEASURED"
    if med_r <= PIN_RATIO_MAX:
        return "PINNED"
    if med_r >= FREE_RATIO_MIN:
        return "FREE"
    return "NEITHER"


def arm_class(classes):
    c = collections.Counter(classes)
    for k in ("PINNED", "FREE"):
        if c.get(k, 0) >= 2:
            return k
    return "NEITHER"


def beta_series(recs, k=0):
    out = []
    for r in recs:
        b = r.get("beta")
        if isinstance(b, list) and len(b) > k:
            out.append(float(b[k]))
    return out


def group_stats(b):
    """step-size-magnitude statistics of ONE beta trajectory."""
    if not b:
        return None
    n = len(b)
    t = b[-max(1, int(n * TAIL_FRAC)):]
    r = [math.exp(v - BETA_LO) for v in t]
    lr2 = BETA_LO + math.log(PIN_RATIO_MAX)
    ft = next((i for i, v in enumerate(b) if v <= BETA_LO + PIN_EPS), None)
    f2 = next((i for i, v in enumerate(b) if v <= lr2), None)
    return {"n": n, "terminal": b[-1], "med_r": statistics.median(r), "max_r": max(r),
            "occ_exact": sum(1 for v in t if v <= BETA_LO + PIN_EPS) / float(len(t)),
            "first_touch": ft, "first_r2": f2}


def ols_slope(y):
    n = len(y)
    if n < 2:
        return float("nan")
    mx, my = (n - 1) / 2.0, sum(y) / n
    den = sum((i - mx) ** 2 for i in range(n))
    return sum((i - mx) * (v - my) for i, v in enumerate(y)) / den


def extrapolate(b, level, window):
    """epoch at which a linear fit of the last `window` records reaches `level`."""
    sl = ols_slope(b[-window:])
    if not (sl < 0) or b[-1] <= level:
        return None, sl
    return (len(b) + (b[-1] - level) / (-sl)) / float(REC_PER_EPOCH), sl


def three_way(delta, lo_tag, mid_tag, hi_tag, bar):
    if delta is None:
        return "UNMEASURED"
    if delta <= -bar:
        return lo_tag
    if delta >= bar:
        return hi_tag
    return mid_tag

EPOCH_RE = re.compile(r"^Epoch (\d+), Train Accuracy: ([-\d.]+) %, Test Accuracy: ([-\d.]+) %")
RUN_RE = re.compile(r"^(%s([A-Za-z0-9]+)-s(\d+))-(\d+)\.out$" % re.escape(PREFIX))
PT_RE = re.compile(r"^PROBE_TENSOR: on every=(\d+) type=(\w+) tensors=(\d+) ")


def parse_args_line(a):
    toks = a.split()[1:] if a.startswith("ARGS:") else a.split()
    d, cnt, cur = {}, collections.Counter(), None
    for t in toks:
        if t.startswith("--"):
            cur = t
            cnt[t] += 1
            d[t] = ""
        elif cur is not None:
            d[cur] = (d[cur] + " " + t).strip()
    return d, sorted(f for f, c in cnt.items() if c > 1)


def parse_env_line(e):
    d = {}
    for t in e.split()[1:]:
        if "=" in t:
            k, v = t.split("=", 1)
            d[k] = v
    return d


def load_outs(runsdir):
    runs, dups = {}, collections.Counter()
    for path in sorted(glob.glob(os.path.join(runsdir, PREFIX + "*.out"))):
        m = RUN_RE.match(os.path.basename(path))
        if not m:
            continue
        rn, arm, seed, jid = m.group(1), m.group(2), int(m.group(3)), int(m.group(4))
        dups[rn] += 1
        d = {"run": rn, "arm": arm, "seed": seed, "job": jid, "path": path,
             "args": None, "env": None, "pt": None, "node": None,
             "train": {}, "test": {}, "done": False, "traceback": False}
        with open(path, errors="replace") as fh:
            for line in fh:
                if line.startswith("ARGS:"):
                    d["args"] = line.rstrip("\n")
                elif line.startswith("ENV:"):
                    d["env"] = line.rstrip("\n")
                elif line.startswith("PROBE_TENSOR:"):
                    d["pt"] = line.rstrip("\n")
                elif line.startswith("NODE="):
                    d["node"] = line.split("|")[0].strip()
                elif line.startswith("RUN_DONE"):
                    d["done"] = True
                elif "Traceback (most recent call last)" in line:
                    d["traceback"] = True
                else:
                    em = EPOCH_RE.match(line)
                    if em:
                        e = int(em.group(1))
                        d["train"][e] = float(em.group(2))
                        d["test"][e] = float(em.group(3))
        if rn not in runs or jid > runs[rn]["job"]:
            runs[rn] = d
    return runs, dups


def load_probes(runsdir):
    out = {}
    for d in sorted(glob.glob(os.path.join(runsdir, BATCH, "probe_" + PREFIX + "*"))):
        rn = os.path.basename(d)[len("probe_"):]
        pj = os.path.join(d, "probe.jsonl")
        if not os.path.isfile(pj):
            continue
        recs = []
        with open(pj) as fh:
            for line in fh:
                line = line.strip()
                if line:
                    try:
                        recs.append(json.loads(line))
                    except ValueError:
                        pass
        e = {"dir": d, "recs": recs, "block": None}
        bs = os.path.join(d, "block_sizes.json")
        if os.path.isfile(bs):
            try:
                e["block"] = json.load(open(bs))
            except ValueError:
                pass
        out[rn] = e
    return out


def resolve_manifest(runsdir, explicit=None):
    if explicit:
        return (explicit, "explicit --manifest") if os.path.isfile(explicit) else (None, "explicit --manifest MISSING")
    for cand, how in ((os.path.join(runsdir, BATCH, "PARTITION-MANIFEST.txt"), "<runsdir>/%s/PARTITION-MANIFEST.txt" % BATCH),
                      (os.path.join(runsdir, PREFIX + "PARTITION-MANIFEST.txt"), "<runsdir>/%sPARTITION-MANIFEST.txt" % PREFIX)):
        if os.path.isfile(cand):
            return cand, "default: " + how
    return None, "default resolution found nothing"


def read_manifest(path):
    tens, arms, iso, head = [], {}, {}, {}
    with open(path) as fh:
        for line in fh:
            p = line.split()
            if not p:
                continue
            if p[0] == "TENSOR" and len(p) >= 4:
                tens.append((int(p[1]), p[2], int(p[3])))
            elif p[0] == "ARMSPEC":
                d, i = {"arm": p[1]}, 2
                while i < len(p):
                    k = p[i]
                    if k in ("SPEC", "M", "SIZES", "PARAMS"):
                        d[k] = p[i + 1]
                        i += 2
                    elif k in ("GROUP0", "GROUP1"):
                        j, vals = i + 1, []
                        while j < len(p) and p[j] not in ("GROUP0", "GROUP1"):
                            vals.append(p[j])
                            j += 1
                        d[k] = vals
                        i = j
                    else:
                        i += 1
                arms[d["arm"]] = d
            elif p[0] == "ISOLATED" and len(p) >= 3:
                iso[p[1]] = tuple(int(x) for x in p[2].split(","))
            elif len(p) >= 2:
                head[p[0]] = p[1]
    return {"tensors": tens, "arms": arms, "isolated": iso, "head": head}


def window_mean(acc, end_epoch):
    v = [acc[e] for e in range(end_epoch - 5, end_epoch) if e in acc]
    return (sum(v) / 5.0) if len(v) == 5 else None


def decompose_check(rec, groups):
    zt, mt, za, mp = rec.get("z_tensor"), rec.get("m_tensor"), rec.get("z_agg"), rec.get("mom_pre")
    if not (isinstance(zt, list) and isinstance(mt, list) and za and mp):
        return 0, 0, 0.0
    za = za[0] if isinstance(za[0], list) else za
    mp = mp[0] if isinstance(mp[0], list) else mp
    nch = nfa = 0
    worst = 0.0
    for k, g in enumerate(groups):
        if k >= len(za) or k >= len(mp):
            continue
        for arr, ref in ((zt, za[k]), (mt, mp[k])):
            terms = [arr[i - 1] for i in g]
            d = abs(sum(terms) - ref)
            # SCALE-AWARE: the error is taken relative to the group's own L1 mass,
            # not to |ref|.  On cvi1's REAL records a 25-tensor complement sum that
            # cancels to ~1e-3 of its largest term carries ~1e-7 ABSOLUTE float32
            # accumulation error; divided by |ref| that reads as up to 5.4e-2 and
            # failed 27 of 12,000 CORRECT groupings (0.5 % on the worst run, half
            # the 1 % ceiling).  A WRONG grouping moves the sum by a whole term,
            # which the negative control in --selftest shows is caught.
            scale = max(abs(ref), sum(abs(x) for x in terms))
            err = d / scale if scale > 1e-30 else d
            worst = max(worst, err)
            nch += 1
            if err > DECOMP_TOL:
                nfa += 1
    return nch, nfa, worst


def corpus_note(csv_path):
    """DESCRIPTIVE ONLY.  Nothing returned here enters any bar, branch or token."""
    if not csv_path or not os.path.isfile(csv_path):
        return "no corpus at %s" % csv_path
    n = own = 0
    with open(csv_path) as fh:
        for r in csv.DictReader(fh):
            n += 1
            own += 1 if str(r.get("run") or "").startswith(PREFIX) else 0
    sha = hashlib.sha256(open(csv_path, "rb").read()).hexdigest()
    return "%d rows (sha %s...), %d named %s*" % (n, sha[:12], own, PREFIX)

def finish(branch, stamps, code=0):
    print("\n" + "=" * 78)
    print("GATES: %d PASS  %d FAIL  %d SKIP" % (_N_PASS, _N_FAIL, _N_SKIP))
    seen, out = set(), []
    for s in stamps:
        if s not in seen:
            seen.add(s)
            out.append(s)
    print("FINAL: " + " | ".join([branch] + out))
    print("=" * 78)
    return code


def score(runsdir, csv_path=None, manifest=None):
    global _N_PASS, _N_FAIL, _N_SKIP
    _N_PASS = _N_FAIL = _N_SKIP = 0
    print("=" * 78)

    print(" runsdir %s" % runsdir)
    print(" design  %d arms x %d seeds = %d jobs, %d epochs (in-run control at %d), PROBE=%d -> %d records"
          % (len(ARMS), len(SEEDS), N_JOBS, EPOCHS, CTRL_EPOCHS, PROBE_EVERY, N_RECORDS))

          % (EPOCHS, CTRL_EPOCHS))
    print(" corpus  %s" % corpus_note(csv_path))
    print("         READ FOR THAT NOTE ONLY: every bar, band and sigma below is a FROZEN LITERAL (O2).")
    print("=" * 78)

    # ---- [G1] manifest --------------------------------------------------------
    print("\n[G1] MANIFEST -- the live model and every registered partition")
    mpath, how = resolve_manifest(runsdir, manifest)
    print("  manifest resolution: %s -> %s" % (how, mpath))
    if not mpath:
        return finish("UNRESOLVED-MANIFEST-MISMATCH", ["NO-MANIFEST"])
    M = read_manifest(mpath)
    ok = chk(len(M["tensors"]) == N_TENSORS, "manifest has %d TENSOR lines" % N_TENSORS,
             "got %d" % len(M["tensors"]))
    if ok:
        ok &= chk(all(M["tensors"][i] == (i + 1, TENSOR_NAMES[i], TENSOR_NUMELS[i]) for i in range(N_TENSORS)),
                  "every tensor matches the frozen NAME AND NUMEL at its 1-based index")
        ok &= chk(sum(t[2] for t in M["tensors"]) == N_PARAMS, "total parameters == %d" % N_PARAMS)
        w512 = tuple((i, n) for i, n, q in M["tensors"] if q == 512 and n.endswith(".weight"))
        ok &= chk(w512 == BN512, "the four 512-wide BN scales are bn5/bn6/bn7/bn8 at 14/17/20/23", str(w512))
        ok &= chk(not any("shortcut" in n for _, n, _ in M["tensors"]), "no parameter name contains `shortcut`")
    for arm in ARMS:
        a = M["arms"].get(arm)
        if not a:
            ok &= chk(False, "manifest carries ARMSPEC %s" % arm)
            continue
        ok &= chk(a.get("SPEC") == SPEC[arm], "%s SPEC is the registered string" % arm, "manifest %r" % a.get("SPEC"))
        if arm in M2_ARMS:
            g0 = [TENSOR_NAMES.index(n) + 1 for n in a.get("GROUP0", []) if n in TENSOR_NAMES]
            g1 = [TENSOR_NAMES.index(n) + 1 for n in a.get("GROUP1", []) if n in TENSOR_NAMES]
            ok &= chk(tuple(sorted(g1)) == ISOLATED[arm], "%s isolates exactly %s (1-based)" % (arm, ISOLATED[arm]),
                      "got %s" % sorted(g1))
            ok &= chk((len(g0), len(g1)) == SIZES[arm], "%s group sizes %s" % (arm, SIZES[arm]))
            pc = (sum(TENSOR_NUMELS[i - 1] for i in g0), sum(TENSOR_NUMELS[i - 1] for i in g1))
            ok &= chk(pc == PARAMS[arm], "%s parameter counts %s" % (arm, PARAMS[arm]), "got %s" % (pc,))
            ok &= chk(sorted(g0 + g1) == list(range(1, N_TENSORS + 1)), "%s covers the 26 tensors, no gap, no duplicate" % arm)
            comp = spec_to_groups(SPEC[arm])
            ok &= chk(sorted(comp[0]) == sorted(g0) and sorted(comp[1]) == sorted(g1),
                      "%s: the registered SPEC re-composes to the manifest's own groups" % arm)
    for k, v in (("EPOCHS", str(EPOCHS)), ("CLIP_C", CLIP_C), ("NETWORK", NET), ("META_STEPS", str(META_STEPS))):
        ok &= chk(M["head"].get(k) == v, "manifest %s == %s" % (k, v), "got %s" % M["head"].get(k))
    if not ok:
        return finish("UNRESOLVED-MANIFEST-MISMATCH", ["MANIFEST-FAIL"])

    # ---- [C] completeness -- a dropped or unfinished run is INCOMPLETE ----------
    print("\n[C] COMPLETENESS -- all %d runs present, RUN_DONE, no traceback, %d epoch lines" % (N_JOBS, EPOCHS))
    runs, dups = load_outs(runsdir)
    want = ["%s%s-s%d" % (PREFIX, a, s) for s in SEEDS for a in ARMS]
    missing = [rn for rn in want if rn not in runs]
    notdone = []
    for rn in want:
        if rn not in runs:
            chk(False, "%s present" % rn, "MISSING")
            continue
        d = runs[rn]
        good = d["done"] and not d["traceback"] and all(e in d["test"] for e in range(EPOCHS))

    print("\n[G0] PROVENANCE -- every run's OWN ARGS / ENV / PROBE_TENSOR line and probe records")
    pok = chk(not extra, "no run outside the registered 12", "extra %s" % extra)

        pok &= chk(not bad and not dupf, "%s ARGS carries every registered flag, none repeated" % rn,

        ev = parse_env_line(d["env"] or "")
        badenv = ["%s=%r" % (k, ev.get(k)) for k, v in REQ_ENV.items() if ev.get(k) != v]
        pdir_ok = str(ev.get("PROBE_DIR", "")).endswith("/%s/probe_%s" % (BATCH, rn))
        pok &= chk(not badenv and pdir_ok, "%s ENV line is the registered one" % rn,
                   "bad %s PROBE_DIR %s" % (badenv, ev.get("PROBE_DIR")))
        envs.add(re.sub(r" PROBE_DIR=\S*", "", d["env"] or ""))
        pm = PT_RE.match(d["pt"] or "")
        pok &= chk(bool(pm) and pm.group(1) == str(PROBE_EVERY) and pm.group(2) == PT_TYPE[d["arm"]]
                   and pm.group(3) == str(N_TENSORS),
                   "%s PROBE_TENSOR: on every=%d type=%s tensors=%d" % (rn, PROBE_EVERY, PT_TYPE[d["arm"]], N_TENSORS),
                   (d["pt"] or "")[:100])
        pr = probes.get(rn)
        if pr is None:
            pok &= chk(False, "%s has probe records" % rn)
            continue
        pok &= chk(len(pr["recs"]) == N_RECORDS, "%s has exactly %d probe records" % (rn, N_RECORDS),
                   "got %d" % len(pr["recs"]))
        want_nb = [N_PARAMS] if d["arm"] == "k01" else (list(TENSOR_NUMELS) if d["arm"] == "kL" else list(PARAMS[d["arm"]]))
        pok &= chk(bool(pr["block"]) and list(pr["block"].get("n_b") or []) == want_nb,
                   "%s block_sizes n_b is the registered arity" % rn)
    pok &= chk(len(envs) == 1, "ONE distinct ENV line across the 12 runs (PROBE_DIR masked)", "%d distinct" % len(envs))
    tot = fail = 0
    worst = 0.0
    for rn in want:
        if runs[rn]["arm"] in M2_ARMS and rn in probes:
            g = spec_to_groups(SPEC[runs[rn]["arm"]])
            for rec in probes[rn]["recs"]:
                c, f, w = decompose_check(rec, g)
                tot, fail, worst = tot + c, fail + f, max(worst, w)
    if tot:
        pok &= chk(fail <= DECOMP_FAIL_MAX * tot, "group sums of z_tensor / m_tensor reproduce z_agg / mom_pre",
                   "%d/%d fail, worst %.3e" % (fail, tot, worst))
    else:
        pok &= chk(False, "decomposition -- no blockwise probe record carries z_tensor")
    if not pok:
        return finish("UNRESOLVED-PROVENANCE", ["PROVENANCE-FAIL"])

    # ---- [L] levels, from the raw .out ----------------------------------------
    print("\n[L] LEVELS -- plateau5 TEST and TRAIN at %d and %d epochs, per seed and per arm" % (CTRL_EPOCHS, EPOCHS))

    for arm in ARMS:
        for s in SEEDS:

        A[arm] = {k: statistics.mean(V[(arm, s)][k] for s in SEEDS) for k in ("P100", "PE", "T100", "TE")}

        A[arm]["RE"] = max(V[(arm, s)]["PE"] for s in SEEDS) - min(V[(arm, s)]["PE"] for s in SEEDS)
        print("  %-4s TEST  @%d %.4f (%s; range %.3f)   @%d %.4f (%s; range %.3f)"
              % (arm, CTRL_EPOCHS, A[arm]["P100"], " ".join("%.3f" % V[(arm, s)]["P100"] for s in SEEDS), A[arm]["R100"],
                 EPOCHS, A[arm]["PE"], " ".join("%.3f" % V[(arm, s)]["PE"] for s in SEEDS), A[arm]["RE"]))
        print("       TRAIN @%d %.4f                              @%d %.4f" % (CTRL_EPOCHS, A[arm]["T100"], EPOCHS, A[arm]["TE"]))
    ss = df = 0
    for arm in ARMS:
        for k in ("P100", "PE"):
            v = [V[(arm, s)][k] for s in SEEDS]
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
    sigma_in = math.sqrt(ss / df)
    sigma_used = max(SIGMA_FROZEN, sigma_in)
    se = sigma_used * math.sqrt(2.0 / 3.0)
    read_bar = 2.0 * se
    print("  SIGMA_FROZEN %.6f (literal)   SIGMA_INBATCH %.6f (df %d, this batch's own seeds)" % (SIGMA_FROZEN, sigma_in, df))
    print("  SIGMA_USED %.6f   SE_USED %.6f   READ_BAR %.6f" % (sigma_used, se, read_bar))

    # ---- [A] anchors / [D] divergence -----------------------------------------
    print("\n[A] ANCHORS -- the in-run 100-epoch floor and ceiling inside their FROZEN bands")
    aok = True
    for arm in ("k01", "kL"):
        lo, hi = ANCHOR[arm] - BAND_HALF, ANCHOR[arm] + BAND_HALF
        aok &= chk(lo <= A[arm]["P100"] <= hi, "%s P100 inside [%.4f, %.4f]" % (arm, lo, hi), "%.4f" % A[arm]["P100"])
    if not aok:
        return finish("UNRESOLVED-ANCHOR", ["ANCHOR-FAIL"])
    print("\n[D] DIVERGENCE -- every arm's seed range <= DEAD_BAR %.1f at both horizons" % DEAD_BAR)
    dok = True
    for arm in ARMS:
        dok &= chk(A[arm]["R100"] <= DEAD_BAR and A[arm]["RE"] <= DEAD_BAR, "%s seed range" % arm,
                   "@%d %.4f  @%d %.4f" % (CTRL_EPOCHS, A[arm]["R100"], EPOCHS, A[arm]["RE"]))
    if not dok:
        return finish("UNRESOLVED-DIVERGED", ["BIMODAL-ARM"])

    # ---- [B] the primary and the branch ----------------------------------------
    print("\n[B] PRIMARY -- all contrasts IN BATCH")
    D100 = A["ISO"]["P100"] - A["k01"]["P100"]
    DE = A["ISO"]["PE"] - A["k01"]["PE"]
    RHO = DE / D100 if abs(D100) > 1e-12 else None
    print("  D_ISO @%d = %+.4f pp (%+.2f SE)   D_ISO @%d = %+.4f pp (%+.2f SE)" % (CTRL_EPOCHS, D100, D100 / se, EPOCHS, DE, DE / se))
    print("  RHO = %s   (SURVIVE needs RHO >= %.2f AND D_ISO@%d >= %.1f; COLLAPSE D_ISO@%d <= %.1f)"
          % (fmt(RHO), SURVIVE_FRAC, EPOCHS, SUPPORT_BAR, EPOCHS, REFUTE_BAR))
    for s in SEEDS:
        d1 = V[("ISO", s)]["P100"] - V[("k01", s)]["P100"]
        d2 = V[("ISO", s)]["PE"] - V[("k01", s)]["PE"]
        print("    seed %d paired: D@%d %+.4f  D@%d %+.4f  rho %s (descriptive)"
              % (s, CTRL_EPOCHS, d1, EPOCHS, d2, fmt(d2 / d1 if abs(d1) > 1e-12 else None)))
    stamps = []
    if D100 < D100_MIN:
        branch = "NO-RESCUE-AT-100"
    elif DE <= REFUTE_BAR:
        branch = "RESCUE-COLLAPSES"
    elif RHO is not None and RHO >= SURVIVE_FRAC and DE >= SUPPORT_BAR:
        branch = "RESCUE-SURVIVES"
    else:
        branch = "RESCUE-ATTENUATES"
    stamps.append("RHO:%s" % fmt(RHO))
    if branch == "RESCUE-ATTENUATES":
        stamps.append("ATTENUATE-MILD" if (RHO is not None and RHO >= ATTEN_MILD_FRAC) else "ATTENUATE-SEVERE")
    print("  BRANCH (decided by the primary alone): %s" % branch)

    # ---- [P] the pin state -- AFTER the primary, as stamps ----------------------
    print("\n[P] PIN STATE -- step-size MAGNITUDE over the last quarter, r = alpha/alpha_floor")
    iso = {}
    for s in SEEDS:
        g = group_stats(beta_series(probes["%sISO-s%d" % (PREFIX, s)]["recs"], 0))
        iso[s] = g
        print("  ISO-s%d complement: median r %.4f  max r %.3f  terminal %.4f  first r<=%.0f epoch %s  first touch %s  "
              "exact-clamp dwell %.4f -> %s"
              % (s, g["med_r"], g["max_r"], g["terminal"], PIN_RATIO_MAX,
                 "%.1f" % (g["first_r2"] / float(REC_PER_EPOCH)) if g["first_r2"] is not None else "never",
                 "%.1f" % (g["first_touch"] / float(REC_PER_EPOCH)) if g["first_touch"] is not None else "never",
                 g["occ_exact"], classify(g["med_r"])))
    cls = arm_class([classify(iso[s]["med_r"]) for s in SEEDS])
    old = sum(1 for s in SEEDS if iso[s]["occ_exact"] >= OLD_PIN_OCC_MIN)
    note("cIS2's PIN_OCC_MIN (exact-clamp dwell >= %.2f) would call %d/3 seeds pinned; it is NOT used" % (OLD_PIN_OCC_MIN, old))
    stamps.append("COMPLEMENT-" + cls)
    if cls == "PINNED":
        pins = [iso[s]["first_r2"] / float(REC_PER_EPOCH) for s in SEEDS
                if classify(iso[s]["med_r"]) == "PINNED" and iso[s]["first_r2"] is not None]
        latest = max(pins)
        window = EPOCHS - latest
        stamps.append("ISO-PIN-EPOCH:%.1f" % statistics.mean(pins))
        stamps.append("POST-PIN-WINDOW:%.1f" % window)
        stamps.append("WINDOW-MEETS-WMAX" if window >= W_MAX else "WINDOW-SHORT")
        print("  pin epochs %s -> latest %.1f; post-pin window %.1f epochs (W_MAX %.1f)" % (pins, latest, window, W_MAX))
        if window >= POST_MIN_WINDOW:
            e0 = int(math.ceil(latest)) + 5
            post = statistics.mean(statistics.mean(runs["%sISO-s%d" % (PREFIX, s)]["test"][e] for e in range(e0, e0 + 5))
                                   for s in SEEDS)
            d_pp = A["ISO"]["PE"] - post
            print("  ISO TEST post-pin (epochs %d..%d) %.4f  vs  end %.4f  delta %+.4f (%+.2f SE)"
                  % (e0, e0 + 4, post, A["ISO"]["PE"], d_pp, d_pp / se))
            stamps.append(three_way(d_pp, "DECAYS-AFTER-PIN", "FROZEN-AT-PIN", "IMPROVES-AFTER-PIN", read_bar))
        else:
            stamps.append("POST-PIN-UNMEASURED")
    else:
        stamps.append("POST-PIN-UNMEASURED")
    for arm, k in (("k01", 0), ("CTL", 0), ("CTL", 1), ("ISO", 1)):
        meds = [group_stats(beta_series(probes["%s%s-s%d" % (PREFIX, arm, s)]["recs"], k))["med_r"] for s in SEEDS]
        print("  %s beta[%d] median r %s (descriptive)" % (arm, k, " / ".join("%.4f" % m for m in meds)))
    kl_free = []
    for s in SEEDS:
        recs = probes["%skL-s%d" % (PREFIX, s)]["recs"]
        nb = len(recs[0]["beta"]) if recs else 0
        kl_free.append(sum(1 for k in range(nb) if classify(group_stats(beta_series(recs, k))["med_r"]) == "FREE"))
    print("  kL groups FREE (median r >= %.0f) per seed: %s of %d" % (FREE_RATIO_MIN, kl_free, N_TENSORS))

    # ---- [S] stamps ----------------------------------------------------------------
    print("\n[S] STAMPS -- none of these moves the branch")
    d = A["ISO"]["PE"] - A["kL"]["PE"]
    print("  ISO - kL @%d = %+.4f pp (%+.2f SE)" % (EPOCHS, d, d / se))
    stamps.append(three_way(d, "CEIL-BELOW", "CEIL-TRACKS", "CEIL-ABOVE", read_bar))
    stamps.append("KL-STILL-TRAINING" if sum(1 for n in kl_free if n >= 1) >= 2 else "KL-FROZEN")
    d = A["k01"]["PE"] - A["k01"]["P100"]
    print("  NULL MODEL  k01 @%d - @%d = %+.4f pp (%+.2f SE)" % (EPOCHS, CTRL_EPOCHS, d, d / se))
    stamps.append("FLOOR-HOLDS" if abs(d) < read_bar else "FLOOR-DRIFTS")
    d = A["CTL"]["PE"] - A["k01"]["PE"]
    print("  CTL - k01 @%d = %+.4f pp" % (EPOCHS, d))
    stamps.append("CTL-AT-FLOOR-AT-E" if abs(d) < read_bar else "CTL-LEAVES-FLOOR")
    d = A["ISO"]["PE"] - A["CTL"]["PE"]
    print("  DELTA_ID @%d = ISO - CTL = %+.4f pp (%+.2f SE)  [@%d %+.4f]"
          % (EPOCHS, d, d / se, CTRL_EPOCHS, A["ISO"]["P100"] - A["CTL"]["P100"]))
    stamps.append("IDENTITY-HOLDS-AT-E" if d >= IDENTITY_BAR else "IDENTITY-FADES-AT-E")
    d = A["ISO"]["TE"] - A["ISO"]["T100"]
    print("  ISO TRAIN @%d - @%d = %+.4f pp" % (EPOCHS, CTRL_EPOCHS, d))
    stamps.append("ISO-TRAIN-FROZEN" if abs(d) < read_bar else "ISO-TRAIN-MOVES")
    stamps.append("SIGMA-FROZEN-DOMINATES" if sigma_used == SIGMA_FROZEN else "SIGMA-INBATCH-DOMINATES")
    d = D100 - CVI1_D_ISO_100
    print("  BETWEEN-BATCH, NON-GATING: D_ISO@%d here %.4f vs cvi1's %.4f, delta %+.4f (%+.2f SE)"
          % (CTRL_EPOCHS, D100, CVI1_D_ISO_100, d, d / se))
    stamps.append("REPLICATES-CVI1" if abs(d) < read_bar else "DIFFERS-FROM-CVI1")
    stamps.append("MAGNITUDE-NOT-SEPARATED")
    stamps.append("NOT-A-ONE-VARIABLE-ABLATION")

    print("\n  REGISTERED POINT PREDICTIONS (descriptive, non-gating):")
    print("    RHO predicted [%.2f, %.2f], measured %s -> %s"
          % (PRED_RHO[0], PRED_RHO[1], fmt(RHO), "IN" if RHO is not None and PRED_RHO[0] <= RHO <= PRED_RHO[1] else "OUT"))
    for arm in ("k01", "CTL", "kL", "ISO"):
        lo, hi = PRED_LEVEL_E[arm]
        v = A[arm]["PE"]
        print("    %-4s @%d predicted [%.1f, %.1f], measured %.4f -> %s" % (arm, EPOCHS, lo, hi, v, "IN" if lo <= v <= hi else "OUT"))
    print("    ISO complement pin (r<=2) predicted <= %.1f, (exact) <= %.1f" % (PIN_BOUND_R2, PIN_BOUND))
    return finish(branch, stamps)

CELLKEYS = ["network", "dataset", "granularity", "base", "meta", "meta_stepsize", "alpha0", "gamma",
            "augment", "beta_clip", "batch_size", "epochs_requested", "hier", "lam", "eta_ratio"]
M2N = re.compile(r"^\[\d+,\d+\]$")


def _rows(path):
    if not path or not os.path.isfile(path):
        return None
    with open(path) as fh:
        return [r for r in csv.DictReader(fh) if not str(r.get("run") or "").startswith(PREFIX)]


def _usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
            and (r.get("plateau5") or "").strip() != "")


def _std_cvi1(r):
    return (r.get("granularity") in ("scalar", "layerwise") and r.get("epochs_requested") == "100"
            and r.get("augment") == "1" and r.get("beta_clip") == CLIP_C and r.get("meta_stepsize") == MS
            and r.get("alpha0") == A0 and r.get("batch_size") == "100")


def _std(r, ep):
    return (r.get("epochs_requested") == str(ep) and r.get("augment") == "1" and r.get("beta_clip") == CLIP_C
            and r.get("meta_stepsize") == MS and r.get("alpha0") == A0 and r.get("batch_size") == "100"
            and r.get("base") == "SGDm" and r.get("meta") == "Lion" and r.get("gamma") == "1"
            and not (r.get("hier") or "").strip())


def _pooled(rows, filt, key):
    cells = collections.defaultdict(list)
    for r in rows:
        if _usable(r) and filt(r):
            cells[key(r)].append(float(r["plateau5"]))
    ss, df = 0.0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = sum(v) / len(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
    return (math.sqrt(ss / df) if df else float("nan")), df


def noise_variants(rows):
    """the six registered definitions, re-derived.  SELFTEST ONLY -- score() never calls this."""
    ck = lambda r: tuple(r.get(k) for k in CELLKEYS)
    bg = lambda r: (re.split(r"[-_]", r["run"])[0], r.get("granularity"), r.get("network"), r.get("epochs_requested"))
    V = lambda r: r.get("network") == NET and r.get("dataset") == DSET
    N = lambda r: r.get("network") == "ResNet18_c100" and r.get("dataset") == "CIFAR100"
    return {
        "VGG_SL_100": _pooled(rows, lambda r: V(r) and _std_cvi1(r), ck),
        "VGG_ALL_100": _pooled(rows, lambda r: V(r) and _std(r, 100), bg),
        "NARROW_SL_100": _pooled(rows, lambda r: r.get("network") == "ResNet18_c100" and r.get("dataset") == "CIFAR100"
                                 and _std_cvi1(r), ck),
        "R250": _pooled(rows, lambda r: N(r) and _std(r, 250), bg),
        "R772": _pooled(rows, lambda r: N(r) and _std(r, 772), bg),
        "R100_M2_NARROW": _pooled(rows, lambda r: N(r) and _std(r, 100) and r.get("epochs_done") == "100"
                                  and bool(M2N.match(r.get("granularity") or "")), bg),
    }


def anchor_levels(rows):
    out = {}
    for arm, gran in (("k01", "scalar"), ("kL", "layerwise")):
        v = [float(r["plateau5"]) for r in rows if _usable(r) and _std_cvi1(r)
             and r.get("network") == NET and r.get("granularity") == gran]
        out[arm] = (statistics.mean(v), len(v)) if v else (None, 0)
    return out


def _final_of(text):
    for line in text.splitlines():
        if line.startswith("FINAL: "):
            return line[len("FINAL: "):]
    return None


def _line_of(text, key):
    for line in text.splitlines():
        if key in line:
            return line.strip()
    return None


def _run_scorer(runsdir, csv_path=None):
    cmd = [sys.executable, THIS, runsdir]
    if csv_path is not None:
        cmd += ["--csv", csv_path]
    p = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, universal_newlines=True, timeout=900)
    return p.returncode, p.stdout

                     plateau5="%.4f" % own_vals[i % len(own_vals)], plateau="", best_test="", final_test="",

            i += 1
    with open(dst, "w", newline="") as fh:

        w.writeheader()

def selftest(csv_path, runsdir, keep=False):
    global _N_PASS, _N_FAIL, _N_SKIP
    _N_PASS = _N_FAIL = _N_SKIP = 0
    print("=" * 78)
    print(" cVH1 SELFTEST -- frozen premises, the gate calibration, and the REAL score()")
    print(" csv %s   runsdir %s" % (csv_path, runsdir))
    print("=" * 78)

    print("\n[S0] the frozen arithmetic")
    chk(abs(SE_FROZEN - SIGMA_FROZEN * math.sqrt(2.0 / 3.0)) < 1e-6, "SE_FROZEN = SIGMA_FROZEN * sqrt(2/3)")
    chk(abs(READ_BAR_FROZEN - 2 * SE_FROZEN) < 2e-6, "READ_BAR_FROZEN = 2 SE_FROZEN")
    chk(SIGMA_FROZEN == max(v for v, _ in SIGMA_VARIANTS.values()), "SIGMA_FROZEN is the MAX of the six registered variants")
    chk(abs(BAND_HALF - BAND_K * SIGMA_FROZEN) < 1e-6, "BAND_HALF = BAND_K x SIGMA_FROZEN")
    chk(PIN_BOUND == max(CVI1_PIN_EXT50 + CVI1_PIN_EXT100), "PIN_BOUND is the latest of the six cvi1 extrapolations")
    chk(EPOCHS == int(math.ceil(max(PIN_BOUND + W_MAX, PIN_BOUND_R2 + W_MAX_R2))),
        "E = ceil(max(PIN_BOUND + W_MAX, PIN_BOUND_R2 + W_MAX_R2)) = %d" % EPOCHS,
        "%.4f / %.4f" % (PIN_BOUND + W_MAX, PIN_BOUND_R2 + W_MAX_R2))
    chk(EPOCHS - PIN_BOUND >= W_MAX, "the complement still gets >= W_MAX post-pin epochs", "%.4f" % (EPOCHS - PIN_BOUND))
    chk(abs(W_MAX - (250 - min(CISO2_K01_TOUCH))) < 1e-9 and abs(W_MAX_R2 - (250 - min(CISO2_K01_R2))) < 1e-9,
        "W_MAX and W_MAX_R2 are 250 minus ciso2 k01's earliest pin")
    chk(META_STEPS == 164000 and N_RECORDS == 1640 and REC_PER_EPOCH == 5 and N_JOBS == 12,
        "164,000 meta-steps, 1,640 records, 5 per epoch, 12 jobs")
    chk(CISO1_ISO_EXT_MAX >= CISO2_ISO_TOUCH_MAX and CISO1_ONE_EXT_MAX >= CISO2_ONE_TOUCH_MAX,
        "the extrapolation method bounded the arm maximum on BOTH ciso2 arms it was tried on")
    for k, v in CAL_MED_R.items():
        chk(arm_class([classify(x) for x in v]) == CAL_WANT[k], "gate classifies %s as %s" % (k, CAL_WANT[k]), str(v))
    chk(max(max(CAL_MED_R["ciso2-ISO"]), max(CAL_MED_R["ciso2-ONE"])) < PIN_RATIO_MAX
        and min(min(CAL_MED_R["ciso2-k62"]), min(CAL_MED_R["cvi1-ISO"]), min(CAL_MED_R["ciso1-ONE"])) > FREE_RATIO_MIN,
        "both gate thresholds sit strictly inside the calibration gap")
    chk(all(x < OLD_PIN_OCC_MIN for x in CISO2_ISO_OCC_EXACT),
        "the frozen record of the defect: ciso2 ISO's exact dwell < 0.50 on 3/3 seeds while PINNED by magnitude")
    chk(D100_MIN > REFUTE_BAR and SUPPORT_BAR > REFUTE_BAR and 0 < SURVIVE_FRAC < 1, "bar ordering")

    print("\n[S1] the corpus, cvh1- excluded from every reader")
    rows = _rows(csv_path)
    if rows is None:
        skip("corpus premises", "no CSV at %s" % csv_path)
    else:
        sha = hashlib.sha256(open(csv_path, "rb").read()).hexdigest()
        at_reg = (sha == REG_CSV_SHA)
        print("  %d usable-or-not rows (cvh1 excluded); sha %s; registration corpus: %s" % (len(rows), sha[:16], at_reg))
        got = noise_variants(rows)
        for k, (reg, rdf) in SIGMA_VARIANTS.items():
            s, df = got[k]
            lab = "sigma %-15s re-derives to %.6f (df %d)" % (k, reg, rdf)
            if at_reg:
                chk(abs(s - reg) < 5e-6 and df == rdf, lab, "got %.6f df %d" % (s, df))
            else:
                note(lab + " -- corpus has MOVED since registration; the bar is FROZEN and cannot move",
                     "live %.6f df %d" % (s, df))
        al = anchor_levels(rows)
        for arm in ("k01", "kL"):
            lab = "anchor %s re-derives to %.4f (n %d)" % (arm, ANCHOR[arm], ANCHOR_N[arm])
            if at_reg:
                chk(al[arm][0] is not None and abs(al[arm][0] - ANCHOR[arm]) < 5e-5 and al[arm][1] == ANCHOR_N[arm],
                    lab, "got %s n %d" % (fmt(al[arm][0]), al[arm][1]))
            else:
                note(lab + " -- corpus moved; the band is FROZEN", "live %s n %d" % (fmt(al[arm][0]), al[arm][1]))
        nseed = sum(1 for r in rows if str(r.get("seed")) in ("63", "64", "65"))
        chk(nseed == 0, "ZERO non-cvh1 rows carry seed 63, 64 or 65", "%d" % nseed)
        chk(not any(r.get("network") == NET and (r.get("epochs_requested") or "0").isdigit()
                    and int(r.get("epochs_requested")) > 100 for r in rows),
            "the corpus owns NO VGG row beyond 100 epochs (the horizon extrapolation of sigma is disclosed)")

    print("\n[S2] the registered specs compose to the registered partitions")
    for arm in M2_ARMS:
        g = spec_to_groups(SPEC[arm])
        chk((len(g[0]), len(g[1])) == SIZES[arm] and tuple(g[1]) == ISOLATED[arm], "%s -> sizes %s isolating %s" % (arm, SIZES[arm], ISOLATED[arm]))
        chk(sorted(g[0] + g[1]) == list(range(1, N_TENSORS + 1)), "%s covers the 26 tensors" % arm)
        chk((sum(TENSOR_NUMELS[i - 1] for i in g[0]), sum(TENSOR_NUMELS[i - 1] for i in g[1])) == PARAMS[arm],
            "%s parameter counts %s" % (arm, PARAMS[arm]))
    chk(TENSOR_NAMES[22] == "bn8.weight" and TENSOR_NAMES[19] == "bn7.weight" and sum(TENSOR_NUMELS) == N_PARAMS,
        "bn8.weight is 23 and bn7.weight is 20 (1-based); numels sum to %d" % N_PARAMS)

    print("\n[S3] the cvi1 PREMISE, from cvi1's own records (the horizon derivation)")
    rd = runsdir or "."
    c1 = os.path.join(rd, "cvi1")
    if not os.path.isfile(os.path.join(c1, "probe_cvi1-ISO-s41", "probe.jsonl")):
        skip("cvi1 premise re-derivation", "no cvi1 probe records under %s" % c1)
    else:
        exts = []
        for i, s in enumerate((41, 42, 43)):
            recs = [json.loads(x) for x in open(os.path.join(c1, "probe_cvi1-ISO-s%d" % s, "probe.jsonl")) if x.strip()]
            b0, b1 = beta_series(recs, 0), beta_series(recs, 1)
            e50, s50 = extrapolate(b0, BETA_LO, 50)
            e100, s100 = extrapolate(b0, BETA_LO, 100)
            g = group_stats(b0)
            print("  ISO-s%d beta[0] terminal %.6f slope50 %.7f slope100 %.7f -> %.4f / %.4f ; median r %.4f ; beta[1] touch %.1f"
                  % (s, b0[-1], s50, s100, e50, e100, g["med_r"], group_stats(b1)["first_touch"] / 5.0))
            chk(abs(b0[-1] - CVI1_ISO_B0_TERMINAL[i]) < 5e-6 and abs(s50 - CVI1_ISO_B0_SLOPE50[i]) < 5e-7
                and abs(s100 - CVI1_ISO_B0_SLOPE100[i]) < 5e-7, "cvi1 ISO-s%d terminal and slopes re-derive" % s)
            chk(abs(g["med_r"] - CAL_MED_R["cvi1-ISO"][i]) < 5e-4 and classify(g["med_r"]) == "FREE",
                "cvi1 ISO-s%d complement is NOT pinned at 100 epochs (median r %.2f)" % (s, g["med_r"]))
            exts += [e50, e100]
        chk(abs(max(exts) - PIN_BOUND) < 5e-4, "cvi1 PIN_BOUND re-derives to %.4f" % PIN_BOUND, "got %.4f" % max(exts))
        ext2 = []
        for s in (41, 42, 43):
            recs = [json.loads(x) for x in open(os.path.join(c1, "probe_cvi1-ISO-s%d" % s, "probe.jsonl")) if x.strip()]
            b0 = beta_series(recs, 0)
            ext2 += [extrapolate(b0, BETA_LO + math.log(PIN_RATIO_MAX), w)[0] for w in (50, 100)]
        chk(abs(max(ext2) - PIN_BOUND_R2) < 5e-4, "cvi1 PIN_BOUND_R2 re-derives to %.4f" % PIN_BOUND_R2, "got %.4f" % max(ext2))
        # the provenance decomposition, on REAL harness records -- with a negative control
        tot = fail = tot_w = fail_w = 0
        for arm in M2_ARMS:
            other = "CTL" if arm == "ISO" else "ISO"
            for s in (41, 42, 43):
                p = os.path.join(c1, "probe_cvi1-%s-s%d" % (arm, s), "probe.jsonl")
                for rec in (json.loads(x) for x in open(p) if x.strip()):
                    c, f, _ = decompose_check(rec, spec_to_groups(SPEC[arm]))
                    tot, fail = tot + c, fail + f
                    c, f, _ = decompose_check(rec, spec_to_groups(SPEC[other]))
                    tot_w, fail_w = tot_w + c, fail_w + f
        chk(tot > 0 and fail <= DECOMP_FAIL_MAX * tot,
            "DECOMPOSITION passes on cvi1's REAL blockwise records under the registered partition", "%d/%d fail" % (fail, tot))
        chk(tot_w > 0 and fail_w >= 0.5 * tot_w,
            "NEGATIVE CONTROL: the SWAPPED partition (ISO<->CTL groups) fails the decomposition on the same real records",
            "%d/%d fail" % (fail_w, tot_w))
        outs = {}
        for arm in ARMS:
            v = []
            for s in (41, 42, 43):
                p = glob.glob(os.path.join(rd, "cvi1-%s-s%d-*.out" % (arm, s)))
                if p:
                    acc = {}
                    for line in open(p[0], errors="replace"):
                        m = EPOCH_RE.match(line)
                        if m:
                            acc[int(m.group(1))] = float(m.group(3))
                    w = window_mean(acc, 100)
                    if w is not None:
                        v.append(w)
            outs[arm] = statistics.mean(v) if len(v) == 3 else None
        if all(outs[a] is not None for a in ARMS):
            for a in ARMS:
                chk(abs(outs[a] - CVI1_P100[a]) < 5e-4, "cvi1 %s plateau5 re-derives to %.4f" % (a, CVI1_P100[a]), "%.4f" % outs[a])
            chk(abs((outs["ISO"] - outs["k01"]) - CVI1_D_ISO_100) < 5e-4, "cvi1 D_ISO re-derives to %.4f" % CVI1_D_ISO_100)
        else:
            skip("cvi1 plateau5", "cvi1 .out files not under %s" % rd)

    print("\n[S4] the ciso1 / ciso2 CALIBRATION, from their own records")
    c2 = os.path.join(rd, "ciso2")
    if not os.path.isfile(os.path.join(c2, "probe_ciso2-ISO-s31", "probe.jsonl")):
        skip("ciso1/ciso2 calibration", "no ciso2 probe records under %s" % c2)
    else:
        def rec_of(b, rn):
            return [json.loads(x) for x in open(os.path.join(rd, b, "probe_" + rn, "probe.jsonl")) if x.strip()]
        touches = [group_stats(beta_series(rec_of("ciso2", "ciso2-k01-s%d" % s), 0)) for s in (31, 32, 33)]
        ft = [t["first_touch"] / 5.0 for t in touches]
        f2 = [t["first_r2"] / 5.0 for t in touches]
        chk(tuple(ft) == CISO2_K01_TOUCH and tuple(f2) == CISO2_K01_R2, "ciso2 k01 pin epochs re-derive", "%s / %s" % (ft, f2))
        chk(abs((250 - min(ft)) - W_MAX) < 1e-9, "ciso2 W_MAX re-derives to %.1f" % W_MAX)
        for key, b, arm, seeds, k in (("ciso2-ISO", "ciso2", "ISO", (31, 32, 33), 0),
                                      ("ciso2-ONE", "ciso2", "ONE", (31, 32, 33), 0),
                                      ("ciso2-k62", "ciso2", "k62", (31, 32, 33), 0),
                                      ("ciso1-ISO", "ciso1", "ISO", (21, 22, 23), 0),
                                      ("ciso1-ONE", "ciso1", "ONE", (21, 22, 23), 0)):
            try:
                gs = [group_stats(beta_series(rec_of(b, "%s-%s-s%d" % (b, arm, s)), k)) for s in seeds]
            except (IOError, OSError):
                skip("%s calibration" % key)
                continue
            meds = tuple(round(g["med_r"], 4) for g in gs)
            chk(all(abs(m - w) < 5e-4 for m, w in zip(meds, CAL_MED_R[key]))
                and arm_class([classify(g["med_r"]) for g in gs]) == CAL_WANT[key],
                "%s median r re-derives and the gate calls it %s" % (key, CAL_WANT[key]), str(meds))
            if key == "ciso2-ISO":
                occ = [g["occ_exact"] for g in gs]
                chk(all(abs(o - w) < 5e-5 for o, w in zip(occ, CISO2_ISO_OCC_EXACT)) and all(o < OLD_PIN_OCC_MIN for o in occ),
                    "ciso2 ISO: the OLD dwell rule FAILS (0.34-0.43 < 0.50) where the magnitude rule PINS -- the defect, fixed",
                    str([round(o, 4) for o in occ]))
                tmax = max(g["first_touch"] for g in gs) / 5.0
                chk(abs(tmax - CISO2_ISO_TOUCH_MAX) < 1e-9, "ciso2 ISO latest first touch re-derives to %.1f" % CISO2_ISO_TOUCH_MAX)
        try:
            e1 = max(extrapolate(beta_series(rec_of("ciso1", "ciso1-ISO-s%d" % s), 0), BETA_LO, 50)[0] for s in (21, 22, 23))
            chk(abs(e1 - CISO1_ISO_EXT_MAX) < 0.05 and e1 >= CISO2_ISO_TOUCH_MAX,
                "ciso1's ISO extrapolation (%.1f) bounded ciso2's latest ISO touch (%.1f)" % (e1, CISO2_ISO_TOUCH_MAX))
        except (IOError, OSError, TypeError):
            skip("ciso1 ISO extrapolation")

    print("\n[S5] SYNTHETIC RUNS THROUGH THE REAL score(), BY THE DOCUMENTED ONE-ARGUMENT INVOCATION")
    if not os.path.isfile(SMOKE_GEN):
        chk(False, "the synthetic generator exists at %s" % SMOKE_GEN)
        return _st_end()
    spec = importlib.util.spec_from_file_location("cvh1_smoke", SMOKE_GEN)
    gen = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(gen)
    chk(gen.EPOCHS == EPOCHS and gen.SEEDS == SEEDS and gen.SPEC == SPEC and tuple(gen.TENSOR_NAMES) == TENSOR_NAMES
        and tuple(gen.TENSOR_NUMELS) == TENSOR_NUMELS and gen.N_RECORDS == N_RECORDS,
        "the generator's design literals equal the scorer's")
    tmp = tempfile.mkdtemp(prefix="cvh1_selftest_")
    real = os.path.realpath(rd)
    chk(not os.path.realpath(tmp).startswith(real + os.sep) and os.path.realpath(tmp) != real,
        "synthetic runs live OUTSIDE the real runs dir", tmp)
    expect = [
        ("survive_chatter", "RESCUE-SURVIVES", ["COMPLEMENT-PINNED", "FROZEN-AT-PIN", "WINDOW-MEETS-WMAX",
                                                "KL-STILL-TRAINING", "FLOOR-HOLDS", "REPLICATES-CVI1", "CEIL-TRACKS"]),
        ("survive_pinned", "RESCUE-SURVIVES", ["COMPLEMENT-PINNED", "FROZEN-AT-PIN"]),
        ("survive_free", "RESCUE-SURVIVES", ["COMPLEMENT-FREE", "POST-PIN-UNMEASURED"]),
        ("collapse", "RESCUE-COLLAPSES", ["COMPLEMENT-PINNED", "DECAYS-AFTER-PIN", "IDENTITY-FADES-AT-E"]),
        ("attenuate", "RESCUE-ATTENUATES", ["ATTENUATE-MILD"]),
        ("no_rescue", "NO-RESCUE-AT-100", ["DIFFERS-FROM-CVI1"]),
        ("near_bar", "RESCUE-SURVIVES", ["CEIL-ABOVE"]),
        ("anchor", "UNRESOLVED-ANCHOR", []),
        ("diverged", "UNRESOLVED-DIVERGED", []),
        ("dropped", "INCOMPLETE", ["MISSING:1", "MISSING-CTL-s64"]),
        ("truncated", "INCOMPLETE", ["NOT-DONE-ISO-s65"]),
        ("badspec", "UNRESOLVED-PROVENANCE", []),
        ("nomanifest", "UNRESOLVED-MANIFEST-MISMATCH", []),
    ]
    finals = {}
    for name, branch, toks in expect:
        d = os.path.join(tmp, name)
        gen.build(d, name)
        rc, out = _run_scorer(d)
        fin = _final_of(out)
        finals[name] = (fin, out)
        ok = fin is not None and fin.split(" | ")[0] == branch and all(t in fin.split(" | ") for t in toks)
        chk(ok, "SYNTHETIC %s -> %s%s" % (name, branch, (" with " + ",".join(toks)) if toks else ""),
            "rc %d FINAL: %s" % (rc, fin))
        if not ok:
            print("\n".join("      | " + x for x in out.splitlines()[-25:]))
    if finals.get("survive_chatter", (None,))[0]:
        g = gen
        recs = [json.loads(x) for x in open(os.path.join(tmp, "survive_chatter", BATCH, "probe_%sISO-s63" % PREFIX, "probe.jsonl"))]
        st = group_stats(beta_series(recs, 0))
        chk(st["occ_exact"] < OLD_PIN_OCC_MIN and classify(st["med_r"]) == "PINNED",
            "the chatter scenario REPRODUCES the ciso2 trigger (dwell %.4f < 0.50) and the magnitude gate PINS it (median r %.4f)"
            % (st["occ_exact"], st["med_r"]))
        del g

    print("\n[S6] FROZEN BARS (O2): score() is INVARIANT to the corpus")
    src = inspect.getsource(score)
    chk(not any(t in src for t in ("noise_variants(", "anchor_levels(", "_pooled(", "_rows(")),
        "score()'s source calls no corpus-derived sigma, band or level")
    base_rows = _rows(csv_path)
    if base_rows is None or not any(r.get("network") == NET and r.get("granularity") == "scalar" for r in base_rows):
        skip("O2 stress corpora", "no VGG template row in %s" % csv_path)
    else:
        header = list(csv.DictReader(open(csv_path)).fieldnames)
        sA, sB = os.path.join(tmp, "stress_A.csv"), os.path.join(tmp, "stress_B.csv")
        _stress_csv(base_rows, header, sA, (20.0, 50.0, 35.0), (11.0, 88.0, 3.0, 64.0))
        _stress_csv(base_rows, header, sB, (35.1, 35.2, 60.0), (35.0, 35.0, 99.0))
        liveA = noise_variants(_rows(sA))
        mx = max(v for v, _ in liveA.values() if v == v)
        rb_live = 2.0 * mx * math.sqrt(2.0 / 3.0)
        chk(mx > SIGMA_FROZEN and rb_live > 1.60,
            "the stress corpus is NOT VACUOUS: a live-max sigma would be %.4f -> READ_BAR %.4f > the near_bar margin 1.60,"
            " so a live-reading scorer would stamp CEIL-TRACKS where this one stamps CEIL-ABOVE" % (mx, rb_live))
        for name in ("near_bar", "survive_chatter", "collapse"):
            d = os.path.join(tmp, name)
            base_fin, base_out = finals[name]
            seen = [("one-arg default corpus", base_fin, _line_of(base_out, "SE_USED"))]
            for lab, cp in (("given corpus", csv_path), ("stress A", sA), ("stress B", sB),
                            ("no corpus", os.path.join(tmp, "absent.csv"))):
                rc, out = _run_scorer(d, cp)
                seen.append((lab, _final_of(out), _line_of(out, "SE_USED")))
            chk(len(set(x[1] for x in seen)) == 1 and len(set(x[2] for x in seen)) == 1 and seen[0][1] is not None,
                "O2 FINAL identical across %d corpora on %s" % (len(seen), name),
                "; ".join("%s: %s" % (x[0], x[1]) for x in seen) if len(set(x[1] for x in seen)) != 1 else seen[0][1])
    if not keep:
        shutil.rmtree(tmp, ignore_errors=True)
    else:
        print("  synthetic runs KEPT at %s" % tmp)
    return _st_end()


def _st_end():
    print("\n" + "=" * 78)
    print("SELFTEST: %d PASS  %d FAIL  %d SKIP" % (_N_PASS, _N_FAIL, _N_SKIP))
    print("=" * 78)
    return 1 if _N_FAIL else 0


def main():
    ap = argparse.ArgumentParser(description="registered scorer for cvh1 (the VGG isolation rescue at a long horizon)")
    ap.add_argument("runsdir", nargs="?", default=None,
                    help="the directory holding cvh1-*.out and cvh1/ (the documented single argument)")
    ap.add_argument("--runsdir", dest="runsdir_kw", default=None, help="same, for --selftest")
    ap.add_argument("--csv", default=DEFAULT_CSV,
                    help="defaults to the repo's results/all_runs.csv; score() reads it for a row-count NOTE only")
    ap.add_argument("--manifest", default=None, help="OPTIONAL: defaults to <runsdir>/cvh1/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--keep", action="store_true", help="--selftest: keep the synthetic runs")
    a = ap.parse_args()
    rd = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, rd or ".", keep=a.keep)
    if not rd:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    return score(rd, a.csv, a.manifest)


if __name__ == "__main__":
    sys.exit(main())
