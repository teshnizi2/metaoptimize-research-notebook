#!/usr/bin/env python3
"""cai1_live_check.py -- `cai1` (CORRECTIONS 306): the spec strings, group counts AND THE DECAY ROUTE RE-DERIVED ON THE
LIVE MODEL.  A copy of analysis/cvl1_live_check.py's construction loop with cai1's design module; guard 4e (the SGDm
source-identity inheritance from cwd5's bite log) is REPLACED by the route check below, because this batch's premise is
the PATCH_DECAYROUTE proof job (305.5) on this very HF.py (sha-pinned by the launcher's guards 3b / 3g), not cwd5's.

Run on cluster_b (CPU ONLY -- model and optimizer CONSTRUCTION, no forward pass, no GPU) from the launcher's guard 4:
    python3 analysis/cai1_live_check.py <tree cifar10 dir> <out manifest path>
For every one of the 8 arms, with DECAY_ROUTE set to the arm's registered value (and SCHED=none, as each job gets it),
it parses the arm's OWN registered ARGS tokens with the tree's OWN `train.parse_args` (lifted by AST), builds the network
with the tree's OWN `build_network` and the optimizer with the tree's OWN `build_optimizer`, and reads back: the stepsize
type, m = sum(numel(beta)), the weight decay the optimizer holds (must be 0.0), the BOUND base update (must be the
patch's `_dr_SGDm_base_update`), the route the patch parsed (`_dr_mode` alpha_indep, `_dr_lam` == the rung's LAMBDA),
and the witness line the construction PRINTED (must equal cai1_design.witness_on(rung) exactly).  It then writes the
LIVE manifest and exits 1 unless it is BYTE-IDENTICAL to cai1_design.manifest_text().  Non-vacuity: with DECAY_ROUTE
unset the same construction must bind the UNPATCHED SGDm_base_update and print `DECAY_ROUTE: off`; and alpha_indep at
--weight-decay-base 0.1 must RAISE the patch's ValueError.
"""
import contextlib
import importlib.util
import io
import os
import sys

tree, out = sys.argv[1], sys.argv[2]
HERE = os.path.dirname(os.path.abspath(__file__))
spec = importlib.util.spec_from_file_location("cai1_design", os.path.join(HERE, "cai1_design.py"))
D = importlib.util.module_from_spec(spec)
spec.loader.exec_module(D)

os.chdir(tree)
sys.path.insert(0, tree)
import torch  # noqa: E402
import argparse  # noqa: E402
import ast  # noqa: E402
_src = open(os.path.join(tree, "train.py")).read()
_fn = [n for n in ast.parse(_src).body if isinstance(n, ast.FunctionDef) and n.name == "parse_args"][0]
_ns = {"argparse": argparse}
exec(compile(ast.Module(body=[_fn], type_ignores=[]), os.path.join(tree, "train.py"), "exec"), _ns)


class train:  # noqa: N801
    parse_args = staticmethod(_ns["parse_args"])


from build_network import build_network  # noqa: E402
from Optimizers.build_optimizer import build_optimizer  # noqa: E402

for k in ("DECAY_ROUTE", "SCHED", "SCHED_WARMUP", "PROBE", "PROBE_TENSOR", "DECAY_MASK", "VAL_SPLIT"):
    os.environ.pop(k, None)
os.environ["SCHED"] = "none"
bad = 0


def build(arm, route, wd_over=None):
    toks = []
    for k, v in D.expected_args(arm, D.SEEDS[arm][0], "/tmp/cai1_livecheck"):
        if k == "weight-decay-base" and wd_over is not None:
            v = wd_over
        toks += ["--" + k, v]
    sys.argv = ["train.py"] + toks
    args = train.parse_args()
    args.device = torch.device("cpu")
    torch.manual_seed(0)
    if route is None:
        os.environ.pop("DECAY_ROUTE", None)
    else:
        os.environ["DECAY_ROUTE"] = route
    net = build_network(args.NN_name, args.device)
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        opt = build_optimizer(net, args, None)
    os.environ.pop("DECAY_ROUTE", None)
    return args, net, opt, [ln for ln in buf.getvalue().splitlines() if ln.startswith("DECAY_ROUTE")]


live_m, ntens, totpar = {}, set(), set()
for a in D.ARMS:
    r = D.RUNG_OF[a]
    args, net, opt, wit = build(a, D.route_value(r))
    m = sum(int(b.numel()) for b in opt.beta)
    nt = len(list(net.parameters()))
    tp = sum(p.numel() for p in net.parameters())
    wd = opt.args_base["weight_decay"]
    bu = getattr(opt.base_update, "__name__", "?")
    ok = (m == D.M_OF_ARM[a] and wd == 0.0 and args.weight_decay_base == D.WDF and bu == "_dr_SGDm_base_update"
          and getattr(opt, "_dr_on", False) and opt._dr_mode == "alpha_indep" and opt._dr_lam == D.LAM[r]
          and wit == [D.witness_on(r)] and args.stepsize_groups == D.SPEC[a]
          and opt.args_meta["meta_stepsize"] == float(D.MST) and float(args.alpha0) == float(D.A0))
    print("guard 4: %-6s spec %-9s type %-10s m %6d (registered %6d)  wd %r  base_update %s  route %s lam %r  witness %s  %s"
          % (a, args.stepsize_groups, opt.stepsize_type, m, D.M_OF_ARM[a], wd, bu, getattr(opt, "_dr_mode", None),
             getattr(opt, "_dr_lam", None), "== registered" if wit == [D.witness_on(r)] else "!!! %r" % wit,
             "OK" if ok else "!!! MISMATCH"))
    bad += not ok
    live_m.setdefault(D.GRAIN_OF[a], set()).add(m)
    ntens.add(nt)
    totpar.add(tp)
# non-vacuity: OFF binds the unpatched update; alpha_indep at wd 0.1 is refused
_a, _n, opt0, wit0 = build("k01I5", None)
nv1 = getattr(opt0.base_update, "__name__", "?") == "SGDm_base_update" and wit0 == ["DECAY_ROUTE: off"]
print("guard 4n1: DECAY_ROUTE unset -> base_update %s, witness %r  %s" % (getattr(opt0.base_update, "__name__", "?"), wit0,
      "OK (non-vacuity: the switch is what rebinds)" if nv1 else "!!! MISMATCH"))
bad += not nv1
try:
    build("k01I5", D.route_value("I5"), wd_over="0.1")
    print("!!! guard 4n2: alpha_indep at --weight-decay-base 0.1 was NOT refused")
    bad += 1
except ValueError as e:
    print("guard 4n2: alpha_indep at --weight-decay-base 0.1 is refused loudly: %s" % str(e)[:90])
if len(ntens) != 1 or len(totpar) != 1:
    print("!!! guard 4: tensor / parameter counts differ across arms: %s %s" % (ntens, totpar))
    bad += 1
nt, tp = ntens.pop(), totpar.pop()
L = ["BATCH %s" % D.PREFIX, "NETWORK %s" % D.NET, "DATASET %s" % D.DSET, "NUM_PARAM_TENSORS %d" % nt,
     "TOTAL_PARAMS %d" % tp, "CLIP_C %s" % D.CLIP, "MS %s ALPHA0 %s" % (D.MST, D.A0), "EPOCHS %d" % D.EPOCHS,
     "WEIGHT_DECAY_BASE %s" % D.WD_TOKEN,
     "META_STEPS %d" % (D.EPOCHS * D.STEPS_PER_EPOCH), "PROBE %d N_RECORDS %d" % (D.PROBE, D.N_RECORDS),
     "DECAY_ROUTE %s" % ",".join("%s=%s" % (r, D.route_value(r)) for r in D.RUNG_IDS)]
for g, spec_, _m in D.GRAINS:
    L.append("GRAIN %s SPEC %s M %s" % (g, spec_, ",".join(str(x) for x in sorted(live_m[g]))))
for a in D.ARMS:
    L.append("ARM %s RUNG %s ROUTE %s SPEC %s M %s SEEDS %s" % (a, D.RUNG_OF[a], D.route_value(D.RUNG_OF[a]), D.SPEC[a],
             ",".join(str(x) for x in sorted(live_m[D.GRAIN_OF[a]])), ",".join(str(s) for s in D.SEEDS[a])))
L.append("NJOBS %d" % D.NJOBS)
txt = "\n".join(L) + "\n"
if txt != D.manifest_text():
    a_, b_ = txt.splitlines(), D.manifest_text().splitlines()
    for k in range(max(len(a_), len(b_))):
        if k >= len(a_) or k >= len(b_) or a_[k] != b_[k]:
            print("!!! guard 4d: live manifest differs at line %d:\n    live      %s\n    design    %s"
                  % (k + 1, a_[k] if k < len(a_) else "", b_[k] if k < len(b_) else ""))
            break
    bad += 1
else:
    print("guard 4d: live %s: %d tensors, %d parameters; the LIVE manifest is BYTE-IDENTICAL to "
          "cai1_design.manifest_text() (%d bytes)" % (D.NET, nt, tp, len(txt)))
if bad:
    raise SystemExit(1)
open(out, "w").write(txt)
