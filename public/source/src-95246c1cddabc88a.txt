def args_line(arm, seed, ws="/cluster/account_b/metaopt"):
    return ARGS_FMT % (DSET, NET, MST, A0, SPEC[arm], seed, ws + "/runs/" + BATCH, "%s-%s-s%d" % (BATCH, arm, seed))


def args_ok(line, arm, seed):
    m = re.match(r"^(.*--save-directory )(\S+)( --run-name \S+)$", line or "")
    if not m or not m.group(2).endswith("/runs/" + BATCH):
        return False
    return line == args_line(arm, seed, m.group(2)[:-len("/runs/" + BATCH)])


def manifest_text():
    names, numel = core.TENSOR_NAMES, core.NUMEL[NET]
    lines = ["BATCH %s" % BATCH, "NETWORK %s" % NET, "NUM_PARAM_TENSORS %d" % len(names),
             "TOTAL_PARAMS %d" % sum(numel)]
    for i, (n, q) in enumerate(zip(names, numel), 1):
        lines.append("TENSOR %d %s %d" % (i, n, q))
    lines.append("CARRIERS %s" % ",".join("%d:%s" % x for x in zip(core.C_IDX1, core.CARRIERS)))
    for a in ARMS:
        g = core.parse_sets(SPEC[a])
        lines.append("ARMSPEC %s SPEC %s GROUPS %s" % (a, SPEC[a], "/".join(",".join(str(i) for i in x) for x in g)))
    return "\n".join(lines) + "\n"

# the decision (pure; selftested)

def nom_word(dom, top3):
    if dom is None or top3 is None:
        return None
    if dom >= DOM_HI and top3 >= TOP3_BAR:
        return "NOMINATED"
    if dom <= DOM_LO:
        return "NOT-NOMINATED"
    return "NOMINATION-PARTIAL"


def arm_word(d, kind):
    if kind == "ISO":
        return "ISO-RESCUES" if d >= RESCUE_BAR else ("ISO-NO-RESCUE" if d <= NULL_BAR else "ISO-PARTIAL")
    return "CTL-RESCUES" if d >= RESCUE_BAR else ("CTL-NULL" if d <= NULL_BAR else "CTL-PARTIAL")


def decide(M):
    """M: {'lev': {arm: mean}, 'rng': {arm: range}, 'runs': [plateau5...], 'dom': DOM_C, 'top3': TOP3_C}
    -> (branch, stamps).  Reads no record."""
    stamps = []
    if any(M["rng"][a] > DIVERGED_BAR for a in ARMS) or any(not (LEVEL_MIN <= x <= LEVEL_MAX) for x in M["runs"]):
        return "UNRESOLVED-DIVERGED", stamps
    nom = nom_word(M["dom"], M["top3"])
    if M["lev"]["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED", ["UNBRANCHED-%s" % nom]
    d_iso = M["lev"]["ISO"] - M["lev"]["k01"]
    d_ctl = M["lev"]["CTL"] - M["lev"]["k01"]
    iso, ctl = arm_word(d_iso, "ISO"), arm_word(d_ctl, "CTL")
    if iso == "ISO-NO-RESCUE" or ctl == "CTL-NULL":
        stamps.append("FLOOR-READINGS-ARE-BOUNDS")
    if M["lev"]["ISO"] < M["lev"]["CTL"]:
        stamps.append("CTL-ABOVE-ISO")
    return "%s+%s+%s" % (nom, iso, ctl), stamps


def all_branches():
    return ["UNRESOLVED-DIVERGED", "SCALAR-NOT-COLLAPSED"] + ["%s+%s+%s" % t for t in
                                                             itertools.product(NOM_WORDS, ISO_WORDS, CTL_WORDS)]


def licence(branch):
    if branch in ("UNRESOLVED-DIVERGED",):
        return "NONE: a seed range > 5 pp or a run outside [5, 90]; no sentence"
    if branch == "SCALAR-NOT-COLLAPSED":
        return "NONE about carriers: k01 > 34 pp, the cell is not an R50 collapse in this batch"
    nom, iso, ctl = branch.split("+")
    head = HEADLINE.get(branch)
    body = "; ".join((NOM_SENT[nom], ISO_SENT[iso], CTL_SENT[ctl]))
    return ("%s. %s." % (head, body)) if head else (body + ".")

def score(runsdir, csv_path=core.DEFAULT_CSV, quiet=False):
    """-> (final_line, rc).  rc 0 scored, 1 gate failed, 2 incomplete."""
    led = core.Ledger(quiet=quiet)
    P = (lambda *a: None) if quiet else print
    P("=" * 100)
    P("cST1 -- `cst1`: the carrier contrast at meta step 3e-4.  runsdir %s" % runsdir)
    P("=" * 100)
    runs, notes = core.read_outs(runsdir, BATCH, ARMS)
    for n in notes:
        led.note(n)

    # ---- INCOMPLETE ------------------------------------------------------------------------------------------------
    P("\n[G-COMPLETE] every (arm, seed) has a COMPLETE .out and 500 probe records")
    probes = {}
    complete = True
    for a in ARMS:
        for s in SEEDS:
            r = runs.get((a, s))
            ok = led.chk(r is not None and r["complete"], "%s-s%d: complete .out (RUN_DONE, epochs 0..99, finite "
                                                          "plateau5)" % (a, s), gate="COMPLETE")
            pr = core.read_probe(runsdir, BATCH, a, s)
            probes[(a, s)] = pr
            ok2 = led.chk(pr is not None and len(pr["recs"]) >= core.N_RECORDS,
                          "%s-s%d: >= %d probe records" % (a, s, core.N_RECORDS),
                          "%d" % (len(pr["recs"]) if pr else 0), gate="COMPLETE")
            complete &= ok and ok2
    if not complete:
        final = "FINAL: INCOMPLETE | %d/%d complete" % (sum(1 for k, v in runs.items() if v["complete"]), N_RUNS)
        P("\n" + final)
        return final, 2

    # ---- HARNESS gates ---------------------------------------------------------------------------------------------
    P("\n[G-CORE / G-PROV / G-STRUCT] the code and the harness that ran")
    led.chk(core.core_sha() == CORE_SHA, "G-CORE: cCV0_carrier_vote_core.py sha256 == registered", core.core_sha(),
            gate="HARNESS")
    prov = core.read_kv(os.path.join(runsdir, BATCH, "PROVENANCE.txt"))
    led.chk(prov is not None, "G-PROV: <runsdir>/%s/PROVENANCE.txt present" % BATCH, gate="HARNESS")
    prov = prov or {}
    for key, want in (("MODE", "submit"), ("HF_SHA256", core.HF_LIVE_SHA), ("BUILD_NETWORK_SHA256",
                                                                            core.BUILD_NETWORK_SHA),
                      ("TRAIN_PY_SHA256", core.TRAIN_PY_SHA), ("RUNNER_SHA256", core.RUNNER_LIVE_SHA),
                      ("CORE_SHA256", CORE_SHA), ("SCORER_SHA256", core.file_sha(os.path.abspath(__file__)))):
        led.chk(prov.get(key) == want, "G-PROV: %s == %s" % (key, want), "got %s" % prov.get(key), gate="HARNESS")
    man = None
    mp = os.path.join(runsdir, BATCH, "PARTITION-MANIFEST.txt")
    if os.path.isfile(mp):
        man = open(mp).read()
    led.chk(man == manifest_text(), "G-STRUCT: PARTITION-MANIFEST.txt byte-identical to manifest_text() "
                                    "(%d bytes)" % len(manifest_text()), gate="HARNESS")

    P("\n[G-ARGS / G-ENV / G-PT / G-FOREIGN / G-HEADER / G-RECORDS] every run's own lines")
    for a in ARMS:
        for s in SEEDS:
            r, pr, tag = runs[(a, s)], probes[(a, s)], "%s-s%d" % (a, s)
            led.chk(len(r["args"]) == 1 and args_ok(r["args"][0], a, s), "G-ARGS %s: one ARGS line == registered"
                    % tag, gate="HARNESS")
            env_ok = len(r["env"]) == 1
            pdir = None
            if env_ok:
                m = re.search(r" PROBE_DIR=(\S+)", r["env"][0])
                pdir = m.group(1) if m else None
                env_ok = (re.sub(r" PROBE_DIR=\S+", "", r["env"][0]) == "ENV: " + core.ENV_FIXED and pdir is not None
                          and pdir.endswith("/runs/%s/probe_%s-%s" % (BATCH, BATCH, tag)))
            led.chk(env_ok, "G-ENV %s: one ENV line == registered (modulo PROBE_DIR, which is the run's own)" % tag,
                    gate="HARNESS")
            m = core.PT_RE.match(r["pt"][0]) if len(r["pt"]) == 1 else None
            led.chk(m is not None and m.group(1) == PT_TYPE[a] and m.group(2) == pdir,
                    "G-PT %s: one PROBE_TENSOR line, type=%s, dir == PROBE_DIR" % (tag, PT_TYPE[a]), gate="HARNESS")
            led.chk(not r["foreign"], "G-FOREIGN %s: no intervention-kind line (the live tree ran)" % tag,
                    "; ".join(r["foreign"])[:120], gate="HARNESS")
            h = pr["hdr"] or {}
            led.chk(h.get("PROBE_TENSOR") == 1 and h.get("every") == core.PROBE_EVERY and h.get("stepsize_type") ==
                    PT_TYPE[a] and h.get("n_tensors") == core.N_TENSORS and tuple(h.get("param_numels", ())) ==
                    core.NUMEL[NET] and h.get("meta_alg") == "Lion" and h.get("Lion_beta2") == 0.9,
                    "G-HEADER %s: probe_tensor.json registered (%s numels)" % (tag, NET), gate="HARNESS")
            led.chk(pr["bad"] == 0 and [x.get("step") for x in pr["recs"]] == list(range(0, core.META_STEPS,
                                                                                          core.PROBE_EVERY)),
                    "G-RECORDS %s: 500 parseable records at steps 0..49,900" % tag, gate="HARNESS")
    if led.fails("HARNESS"):
        final = "FINAL: HARNESS-UNSOUND | %d harness check(s) failed: %s" % (len(led.fails("HARNESS")),
                                                                            led.fails("HARNESS")[0])
        P("\n" + final)
        return final, 1

    P("\n[G-SPEC] the partition that RAN, read off the records (the BROKEN-SPEC null's only separator)")
    for a in ARMS:
        for s in SEEDS:
            recs = probes[(a, s)]["recs"]
            nc = set(len(x.get("beta", [])) for x in recs)
            nz = set(len((x.get("z_agg") or [[]])[0]) for x in recs)
            led.chk(nc == {N_COORDS[a]} and nz == {N_COORDS[a]}, "G-SPEC %s-s%d: every record carries %d beta / z_agg "
                                                                 "coordinate(s)" % (a, s, N_COORDS[a]),
                    "beta %s z_agg %s" % (sorted(nc), sorted(nz)), gate="SPEC")
    if led.fails("SPEC"):
        final = "FINAL: SPEC-NOT-APPLIED | %s" % led.fails("SPEC")[0]
        P("\n" + final)
        return final, 1

    P("\n[G-DECOMP] per-group sums of the capture vs the harness; applied step vs sign(L)")
    for a in ARMS:
        for s in SEEDS:
            d = core.decomposition(probes[(a, s)]["recs"], core.parse_sets(SPEC[a]), MS)
            led.chk(core.decomposition_ok(d), "G-DECOMP %s-s%d" % (a, s),
                    "shape %d z %d m %d (worst %.1e / %.1e) sign %d/%d clamped %d shared %d" % (
                        d["bad_shape"], d["bad_z"], d["bad_m"], d["worst_z"], d["worst_m"], d["bad_sign"],
                        d["n_sign"], d["clamped"], d["bad_shared"]), gate="DECOMP")
    if led.fails("DECOMP"):
        final = "FINAL: UNRESOLVED-DECOMPOSITION | %s" % led.fails("DECOMP")[0]
        P("\n" + final)
        return final, 1

    # ---- NOMINATION (k01 records only) -----------------------------------------------------------------------------
    P("\n[NOMINATION] k01's records only, pooled over seeds (the registered check)")
    for s in SEEDS:
        core.describe_stats("k01-s%d" % s, core.vote_stats(probes[("k01", s)]["recs"]), out=P)
    nst = core.vote_stats([x for s in SEEDS for x in probes[("k01", s)]["recs"]])
    core.describe_stats("k01 POOLED", nst, out=P)
    P("  calibration at ms 1e-3 (registration): scalar DOM_C %.3f-%.3f pooled %.4f; layerwise 0.0000"
      % (CALIB_SCALAR_DOM + (CALIB_SCALAR_POOLED,)))

    # ---- LEVELS ----------------------------------------------------------------------------------------------------
    P("\n[LEVELS] plateau5 (train5) per run; arm means; contrasts in pp and SE (SE_PRIOR %.6f)" % SE_PRIOR)
    lev, rng, trn, allruns = {}, {}, {}, []
    for a in ARMS:
        v = [runs[(a, s)]["p5"] for s in SEEDS]
        t = [runs[(a, s)]["t5"] for s in SEEDS]
        lev[a], rng[a], trn[a] = core.mean(v), max(v) - min(v), core.mean(t)
        allruns += v
        P("  %-4s %s  mean %.4f  range %.3f  | train %s mean %.4f" % (a, " ".join("%.3f" % x for x in v), lev[a],
                                                                      rng[a], " ".join("%.3f" % x for x in t), trn[a]))
    d_iso, d_ctl, p = lev["ISO"] - lev["k01"], lev["CTL"] - lev["k01"], lev["ISO"] - lev["CTL"]
    for nm, d in (("D_ISO = ISO - k01", d_iso), ("D_CTL = CTL - k01", d_ctl), ("P = ISO - CTL", p)):
        P("  %-20s %+8.4f pp = %+8.2f SE" % (nm, d, d / SE_PRIOR))
    ins = [runs[(a, s)]["p5"] - lev[a] for a in ARMS for s in SEEDS]
    sig_in = math.sqrt(sum(x * x for x in ins) / (len(ins) - len(ARMS)))
    P("  in-batch sigma %.4f (df %d); SIGMA_PRIOR %.4f -> %s" % (sig_in, len(ins) - len(ARMS), SIGMA_PRIOR,
                                                                 "SIGMA-PRIOR-FROZEN"))

    M = {"lev": lev, "rng": rng, "runs": allruns, "dom": nst["DOM_C"], "top3": nst["TOP3_C"]}
    branch, stamps = decide(M)
    tw = None
    if branch not in ("UNRESOLVED-DIVERGED", "SCALAR-NOT-COLLAPSED"):
        tw = "%s+%s" % (arm_word(trn["ISO"] - trn["k01"], "ISO"), arm_word(trn["CTL"] - trn["k01"], "CTL"))
        stamps.append("TRAIN-AGREES" if tw == "+".join(branch.split("+")[1:]) else "TRAIN-DISAGREES(%s)" % tw)
    stamps += ["DOM_C=%.4f" % nst["DOM_C"], "TOP3_C=%.4f" % nst["TOP3_C"],
               "PINNED-FRAC=%s" % core.fmt(nst["PINNED_FRAC"]), "HARNESS-CLEAN", "SPEC-APPLIED", "DECOMPOSITION-OK",
               "ONE-CELL-MS-3E-4", "ALPHA0-1E-6-ONLY", "HORIZON-100-ONLY", "ONE-NETWORK-RESNET18_C100",
               "CTL-HAS-A-BIAS-MEMBER", "NO-LAYERWISE-IN-BATCH", "NOMINATION-FROM-K01-RECORDS-ONLY",
               "SIGMA-PRIOR-FROZEN"]

    # ---- DESCRIPTIVE ----------------------------------------------------------------------------------------------
    P("\n[DESCRIPTIVE, gating nothing] the counterfactual shared vote inside ISO and CTL; beta per group")
    for a in ("ISO", "CTL"):
        core.describe_stats("%s POOLED" % a, core.vote_stats([x for s in SEEDS for x in probes[(a, s)]["recs"]]), out=P)
    for a in ARMS:
        for s in SEEDS:
            recs = probes[(a, s)]["recs"]
            for c in range(N_COORDS[a]):
                b = [x["beta"][c] for x in recs]
                pk = max(range(len(b)), key=lambda i: b[i])
                P("  beta %s-s%d g%d: peak %.3f at step %d; terminal %.3f; share at floor %.3f"
                  % (a, s, c, b[pk], recs[pk]["step"], b[-1], sum(1 for x in b if x <= core.BETA_LO + 1e-6) / len(b)))
    rows = core.read_corpus(csv_path, exclude_prefix=BATCH)
    if rows is not None:
        P("  corpus anchors (non-gating, filtered): %s" % "; ".join("%s %.4f" % kv for kv in ANCHOR.items()))

    P("\nBRANCH   %s" % branch)
    P("LICENCE  %s" % licence(branch))
    P("NOT LICENSED  %s" % NOT_LICENSED)
    final = "FINAL: %s | %s" % (branch, " | ".join(stamps))
    P("\n" + final)
    return final, 0

def _synth_batch(root, levels, dom, rng, spec_break=None, decomp_break=None, drop=None, no_done=None, args_mut=None,
                 env_mut=None, pt_mut=None, foreign=None, hdr_mut=None, short=None, prov_mut=None, man_mut=None,
                 seed_jitter=None, kind_else="nodom", cache=None, extra_trunc=None, ws="/cluster/account_b/metaopt"):
    os.makedirs(os.path.join(root, BATCH), exist_ok=True)
    job = 7000000
    for a in ARMS:
        for s in SEEDS:
            job += 1
            if drop == (a, s):
                continue
            groups = core.parse_sets(SPEC[a])
            share = dom if a == "k01" else 0.0
            key = (a, share, kind_else, spec_break == (a, s), decomp_break == (a, s), short == (a, s))
            recs = cache.get(key) if cache is not None else None
            if recs is None:
                recs = core.synth_records(groups, share, rng, MS, kind_else=kind_else,
                                          break_decomp=decomp_break == (a, s),
                                          n_coords=1 if spec_break == (a, s) else None)
                if short == (a, s):
                    recs = recs[:-1]
                if cache is not None:
                    cache[key] = recs
            lvl = levels[a] + (seed_jitter.get((a, s), 0.0) if seed_jitter else 0.0)
            pdir = os.path.join(ws, "runs", BATCH, "probe_%s-%s-s%d" % (BATCH, a, s))
            env = core.ENV_FIXED.replace("PROBE=100", "PROBE=100 PROBE_DIR=" + pdir)
            if env_mut == (a, s):
                env = env.replace("AUGMENT=1", "AUGMENT=0")
            al = args_line(a, s, ws)
            if args_mut == (a, s):
                al = al.replace("--meta-stepsize 3e-4", "--meta-stepsize 1e-3")
            run = core.write_synth_run(root, BATCH, a, s, job, al, PT_TYPE[a] if pt_mut != (a, s) else "scalar",
                                       lvl, recs=recs, drop_run_done=no_done == (a, s), env=env,
                                       hdr_override={"param_numels": list(core.NUMEL["ResNet18"])}
                                       if hdr_mut == (a, s) else None)
            # the PT line must name the run's PROBE_DIR (the synthetic writer names its local dir): rewrite it
            outp = os.path.join(root, "%s-%d.out" % (run, job))
            txt = open(outp).read()
            txt = re.sub(r"(PROBE_TENSOR: on .* dir=)\S+", lambda m: m.group(1) + pdir, txt)
            if foreign and foreign[0] == (a, s):
                txt = txt.replace("\nEpoch 0,", "\n%s\nEpoch 0," % foreign[1], 1)
            open(outp, "w").write(txt)
            if extra_trunc == (a, s):
                job2 = job + 500
                # a later, truncated resubmission (higher job id, 40 epochs, no RUN_DONE); recs=None leaves the
                # complete run's probe records in place
                core.write_synth_run(root, BATCH, a, s, job2, al, PT_TYPE[a], 1.0, recs=None, n_epochs=40,
                                     drop_run_done=True, env=env)
    prov = {"MODE": "submit", "HF_SHA256": core.HF_LIVE_SHA, "BUILD_NETWORK_SHA256": core.BUILD_NETWORK_SHA,
            "TRAIN_PY_SHA256": core.TRAIN_PY_SHA, "RUNNER_SHA256": core.RUNNER_LIVE_SHA, "CORE_SHA256": CORE_SHA,
            "SCORER_SHA256": core.file_sha(os.path.abspath(__file__))}
    if prov_mut:
        prov.update(prov_mut)
    with open(os.path.join(root, BATCH, "PROVENANCE.txt"), "w") as f:
        for k, v in prov.items():
            f.write("%s %s\n" % (k, v))
    mt = manifest_text()
    if man_mut:
        mt = mt.replace("layer4.1.bn2.weight 512", "layer4.1.bn2.weight 513")
    open(os.path.join(root, BATCH, "PARTITION-MANIFEST.txt"), "w").write(mt)


def selftest(csv_path, runsdir=None):
    led = core.Ledger()
    print("=" * 100)
    print("cST1 SELFTEST")
    print("=" * 100)

    print("\n[A] design, literals, bars, branches")
    led.chk(core.core_sha() == CORE_SHA, "the imported core is the registered core", core.core_sha())
    led.chk(len(ARMS) * len(SEEDS) == 9 and set(SEEDS) | set(UNUSED_SEEDS) == {112, 113, 114, 115},
            "9 runs; seeds 112-114 used, 115 unused")
    try:
        import cDP1_cdep1_depth_score as dp1
        led.chk(SPEC["ISO"] == dp1.SPEC["ISO"] and SPEC["CTL"] == dp1.SPEC["DEPTH"],
                "ISO and CTL strings byte-identical to cdep1's registered ISO and DEPTH (cDP1 imported)")
    except Exception as e:  # noqa: BLE001
        led.chk(False, "cDP1 importable", str(e))
    try:
        import cIS1_ciso1_isolate_score as is1
        spec_iso = getattr(is1, "SPEC", {}).get("ISO")
        led.chk(spec_iso == SPEC["ISO"], "ISO string byte-identical to ciso1's registered ISO (cIS1 imported)",
                str(spec_iso)[:40])
    except Exception as e:  # noqa: BLE001
        led.chk(False, "cIS1 importable", str(e))
    g_iso, g_ctl = core.parse_sets(SPEC["ISO"]), core.parse_sets(SPEC["CTL"])
    led.chk(tuple(g_iso[1]) == ISOLATED["ISO"] == core.C_IDX1 and len(g_iso[0]) == 59
            and sorted(g_iso[0] + g_iso[1]) == list(range(1, 63)), "ISO parses to [59, 3], isolated 50, 53, 59")
    led.chk(tuple(g_ctl[1]) == ISOLATED["CTL"] and len(g_ctl[0]) == 59 and sorted(g_ctl[0] + g_ctl[1]) ==
            list(range(1, 63)) and not set(g_ctl[1]) & set(core.C_IDX1),
            "CTL parses to [59, 3], isolated 47, 48, 56, no carrier")
    led.chk([core.NUMEL[NET][i - 1] for i in g_iso[1]] == [core.NUMEL[NET][i - 1] for i in g_ctl[1]] == [512] * 3,
            "isolated numels matched 3 x 512")
    led.chk(0 < DOM_LO < DOM_HI < 1 and NULL_BAR < RESCUE_BAR and K01_MAX < ANCHOR["layerwise@3e-4 (cru1 n3)"] / 2
            and K01_MAX > ANCHOR["k01@3e-4 (cru1 n3)"] + 4 * SIGMA_PRIOR, "bar ordering; K01_MAX below R50 of the "
                                                                          "corpus layerwise level, above k01 + 4 sigma")
    led.chk(CALIB_SCALAR_DOM[0] - DOM_HI >= 0.25 and DOM_LO >= 0.10, "nomination bars sit >= 0.25 below the collapse "
                                                                     "calibration and >= 0.10 above layerwise (0.000)")
    br = all_branches()
    led.chk(len(br) == 29 and all(licence(b) for b in br), "29 branches, every one has a licence")
    import inspect
    src = inspect.getsource(decide)
    led.chk("probe" not in src and "recs" not in src and "read_" not in src, "decide() reads no record")
    led.chk(abs(SE_PRIOR - 0.5561961163) < 1e-9, "SE_PRIOR = SIGMA_PRIOR * sqrt(2/3) = 0.556196")
    led.chk(args_line("ISO", 112).count("--") == 20, "the registered ARGS line has 20 flags, none repeated")

    print("\n[B] the nomination premise on LANDED records (ms 1e-3), with the core this scorer runs")
    calib = [("ctd1", "sc", (18, 19, 20)), ("ciso1", "k01", (21, 22, 23)), ("cdep1", "k01", (24, 25, 26))]
    lay = [("ctd1", "lay", (18, 19, 20)), ("ciso1", "k62", (21, 22, 23)), ("cdep1", "k62", (24, 25, 26))]
    if runsdir and all(core.read_probe(runsdir, b, a, s[0]) for b, a, s in calib + lay):
        for grp, want in ((calib, "NOMINATED"), (lay, "NOT-NOMINATED")):
            recs = []
            for b, a, ss in grp:
                for s in ss:
                    recs += core.read_probe(runsdir, b, a, s)["recs"]
            st = core.vote_stats(recs)
            led.chk(nom_word(st["DOM_C"], st["TOP3_C"]) == want and len(recs) == 4500,
                    "PREMISE %s: 9 landed runs read %s" % (grp[0][1], want),
                    "DOM_C %.4f TOP3_C %.4f" % (st["DOM_C"], st["TOP3_C"]))
        st = core.vote_stats(sum((core.read_probe(runsdir, b, a, s)["recs"] for b, a, ss in calib for s in ss), []))
        led.chk(abs(st["DOM_C"] - CALIB_SCALAR_POOLED) < 5e-5, "PREMISE the pooled scalar DOM_C == the registered "
                                                               "0.8069", "%.5f" % st["DOM_C"])
    else:
        print("  SKIP the landed ctd1 / ciso1 / cdep1 probe records are not under --runsdir")

    print("\n[C] the noise floor through filter_rows; freshness")
    rows = core.read_corpus(csv_path)
    raw = core.read_corpus(csv_path, filtered=False)
    if rows is None:
        print("  SKIP no corpus")
    else:
        sig, df, nc = core.pooled_sigma(rows, NET, DSET)
        sha = core.file_sha(csv_path)
        if sha == REG_CORPUS_SHA:
            led.chk(abs(sig - SIGMA_PRIOR) < 1e-12 and df == 221 and nc == 46, "SIGMA_R18ALL FILTERED == SIGMA_PRIOR "
                                                                               "(registration corpus)", repr(sig))
        else:
            print("  NOTE the corpus moved since registration (sha %s..., %d rows): filtered sigma now %r (df %d); "
                  "SIGMA_PRIOR stays frozen (O2)" % (sha[:12], len(raw), sig, df))
        naive = core.pooled_sigma(raw, NET, DSET)[0]
        led.chk(naive is not None and naive > sig, "the NAIVE sigma exceeds the filtered one (the trap is live)",
                "%.4f vs %.4f" % (naive, sig))
        led.chk(not any(r["run"].startswith(BATCH + "-") for r in raw), "no cst1- row in the corpus")
        led.chk(not any(r.get("seed") in ("112", "113", "114", "115") for r in raw), "no row with seed 112-115")

    print("\n[D] accounts and the floor table")
    lows = []
    for name, acc in ACCOUNTS.items():
        mid = lambda k: (acc[k][0] + acc[k][1]) / 2.0  # noqa: E731
        M = {"lev": {a: mid(a) for a in ARMS}, "rng": {a: 0.5 for a in ARMS}, "runs": [mid(a) for a in ARMS],
             "dom": mid("dom"), "top3": 0.8 if mid("dom") >= DOM_HI else 0.0}
        b = decide(M)[0]
        led.chk(b == EXPECTED_BRANCH[name], "account %s's midpoint lands in %s" % (name, EXPECTED_BRANCH[name]), b)
        lows += [acc[a][0] for a in ARMS]
    led.chk(len(set(EXPECTED_BRANCH.values())) == len(EXPECTED_BRANCH), "the 6 accounts land in 6 different branches")
    acc = ACCOUNTS["SAME-MECHANISM"]
    Mn = {"lev": {"k01": 29.0, "ISO": 29.0, "CTL": 29.0}, "rng": {a: 0.5 for a in ARMS}, "runs": [29.0] * 3,
          "dom": 0.7, "top3": 0.8}
    led.chk(decide(Mn)[0] == NULL_ACCOUNT[1] and NULL_ACCOUNT[1] not in EXPECTED_BRANCH.values(),
            "the BROKEN-SPEC null predicts %s, a branch no non-null account owns; only G-SPEC separates it"
            % NULL_ACCOUNT[1])
    led.chk(min(lows) - CHANCE >= 20, "lowest predicted level %d pp is >= 20 pp above chance" % min(lows))

    print("\n[E] the REAL score() on synthetic batches")
    tmp = tempfile.mkdtemp(prefix="cst1_selftest_")
    rng = random.Random(20260917)
    cache = {}
    base = {"k01": 29.0, "ISO": 67.0, "CTL": 29.5}

    def run_case(label, want_prefix, want_rc, levels=None, dom=0.8, **kw):
        root = os.path.join(tmp, re.sub(r"[^A-Za-z0-9]+", "_", label))
        _synth_batch(root, levels or base, dom, rng, cache=cache, **kw)
        final, rc = score(root, csv_path, quiet=True)
        ok = final.startswith("FINAL: " + want_prefix) and rc == want_rc
        led.chk(ok, "E %-58s -> %s (rc %d)" % (label, want_prefix, want_rc), "got %s rc %d" % (final[:90], rc))
        shutil.rmtree(root, ignore_errors=True)
        return final

    run_case("SAME-MECHANISM", "NOMINATED+ISO-RESCUES+CTL-NULL |", 0)
    run_case("ISOLATION-GENERIC", "NOMINATED+ISO-RESCUES+CTL-RESCUES", 0, {"k01": 29, "ISO": 67, "CTL": 66})
    run_case("NOMINATED-INSUFFICIENT", "NOMINATED+ISO-PARTIAL+CTL-NULL", 0, {"k01": 29, "ISO": 46, "CTL": 29})
    run_case("OTHER-CARRIERS (not nominated)", "NOT-NOMINATED+ISO-NO-RESCUE+CTL-NULL", 0,
             {"k01": 29, "ISO": 29.5, "CTL": 29}, dom=0.0)
    run_case("ROLE-NOT-VOTE", "NOT-NOMINATED+ISO-RESCUES+CTL-NULL", 0, dom=0.05)
    run_case("other-trio dominates, C not", "NOT-NOMINATED+ISO-RESCUES", 0, dom=0.0, kind_else="other3")
    run_case("NOMINATION-PARTIAL", "NOMINATION-PARTIAL+ISO-RESCUES+CTL-NULL", 0, dom=0.30)
    run_case("CTL-PARTIAL", "NOMINATED+ISO-RESCUES+CTL-PARTIAL", 0, {"k01": 29, "ISO": 67, "CTL": 40})
    run_case("NO-COLLAPSE-HERE", "SCALAR-NOT-COLLAPSED", 0, {"k01": 41, "ISO": 67, "CTL": 41})
    run_case("near bar: D_ISO 20.02 rescues", "NOMINATED+ISO-RESCUES", 0, {"k01": 29, "ISO": 49.02, "CTL": 29})
    run_case("near bar: D_ISO 19.98 partial", "NOMINATED+ISO-PARTIAL", 0, {"k01": 29, "ISO": 48.98, "CTL": 29})
    run_case("near bar: D_CTL 4.98 null", "NOMINATED+ISO-RESCUES+CTL-NULL", 0, {"k01": 29, "ISO": 67, "CTL": 33.98})
    run_case("near bar: D_CTL 5.02 partial", "NOMINATED+ISO-RESCUES+CTL-PARTIAL", 0, {"k01": 29, "ISO": 67,
                                                                                     "CTL": 34.02})
    run_case("near bar: DOM_C 0.502 nominated", "NOMINATED+", 0, dom=0.502)
    run_case("near bar: DOM_C 0.498 partial", "NOMINATION-PARTIAL+", 0, dom=0.498)
    run_case("near bar: DOM_C 0.100 not nominated", "NOT-NOMINATED+", 0, dom=0.100)
    run_case("near bar: DOM_C 0.104 partial", "NOMINATION-PARTIAL+", 0, dom=0.104)
    run_case("near bar: k01 33.9 collapsed", "NOMINATED+", 0, {"k01": 33.9, "ISO": 67, "CTL": 33.9})
    run_case("near bar: k01 34.1 not collapsed", "SCALAR-NOT-COLLAPSED", 0, {"k01": 34.1, "ISO": 67, "CTL": 34.1})
    run_case("diverged seed (ISO s113 -6)", "UNRESOLVED-DIVERGED", 0, seed_jitter={("ISO", 113): -6.0})
    run_case("dropped run -> INCOMPLETE", "INCOMPLETE", 2, drop=("CTL", 114))
    run_case("no RUN_DONE -> INCOMPLETE", "INCOMPLETE", 2, no_done=("k01", 113))
    run_case("499 records -> INCOMPLETE", "INCOMPLETE", 2, short=("ISO", 112))
    run_case("truncated resubmission does not shadow", "NOMINATED+ISO-RESCUES+CTL-NULL", 0, extra_trunc=("ISO", 113))
    run_case("ARGS: ms 1e-3 on k01-s112", "HARNESS-UNSOUND", 1, args_mut=("k01", 112))
    run_case("ENV: AUGMENT=0", "HARNESS-UNSOUND", 1, env_mut=("CTL", 113))
    run_case("PT type scalar on ISO", "HARNESS-UNSOUND", 1, pt_mut=("ISO", 114))
    run_case("foreign GROUP_HOLD: off line", "HARNESS-UNSOUND", 1, foreign=(("ISO", 112), "GROUP_HOLD: off"))
    run_case("foreign SHADOW_VOTE line", "HARNESS-UNSOUND", 1, foreign=(("k01", 114), "SHADOW_VOTE: off"))
    run_case("header numels of ResNet18 (CIFAR-10)", "HARNESS-UNSOUND", 1, hdr_mut=("k01", 112))
    run_case("provenance HF of cvt8's tree", "HARNESS-UNSOUND", 1, prov_mut={"HF_SHA256": "5197dc2e" + "0" * 56})
    run_case("provenance runner of cvt8", "HARNESS-UNSOUND", 1, prov_mut={"RUNNER_SHA256": "1dbe4ff5" + "0" * 56})
    run_case("provenance MODE dryrun", "HARNESS-UNSOUND", 1, prov_mut={"MODE": "dryrun"})
    run_case("provenance scorer sha wrong", "HARNESS-UNSOUND", 1, prov_mut={"SCORER_SHA256": "0" * 64})
    run_case("manifest numel changed", "HARNESS-UNSOUND", 1, man_mut=True)
