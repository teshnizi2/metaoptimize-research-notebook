#!/bin/bash
# =============================================================================
# c76_matched_m_partition.sh -- `mm1`, THE MATCHED-COUNT PARTITION CONTRAST.
# Runs on cluster_a ONLY (needs PATCH_CHUNKWISE).
# 6 jobs: {chunk777, nodewise} x seeds 0-2, ms=1e-4, 100 ep, probe ON.
#
# WHY THIS BATCH EXISTS
#   `ck1`'s K2 REFUTED "accuracy is a function of the group count m alone":
#   chunk1024 (m=10,944) beat nodewise (m=14,420) by +0.565 pp at the same ms.
#   analysis/c76_partition_vs_count.py decomposed that number and found the count
#   explains only +0.049 pp of it and the PARTITION +0.517 pp.  But that residual
#   rests on TWO things this batch removes outright rather than models:
#
#     1. AN INTERPOLATION.  The count was removed by interpolating ck1's own chunk
#        curve to m=14,420.  Here m is MATCHED BY CONSTRUCTION instead:
#        m(777) = 14,421 against nodewise's 14,420 -- ONE group apart, 7e-5 in
#        relative terms.  Mean group size is then identical to 0.1 weights (774.9).
#     2. A CROSS-BATCH OFFSET.  chunk1024 was a `ck1` run and nodewise a `tw0` run.
#        K1 measured that offset at -0.255 pp, NOT resolved (t=1.74), and it was
#        correctly not applied.  Here BOTH arms run in the SAME batch, so M1 is a
#        WITHIN-BATCH difference and no offset can enter it at all.
#
#   What survives is the cleanest question the campaign can currently ask:
#   at the SAME number of groups, over the SAME weights, does it matter WHICH
#   weights are grouped together?
#
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
#
#   M0    VALIDITY.  n_records == 10000; beta moved; epochs_done == requested == 100.
#   M0.2  n_beta EXACT on EVERY record: chunk777 14421, nodewise 14420.  Guard 4
#         MEASURES both from the ALLOCATED beta on the real built network before
#         anything is submitted -- neither is taken from a comment.
#   M0.3  THE INSTRUMENT FIRED.  neg_counts.json, n_tot == n_beta, npy shape from
#         its HEADER == (n_tot,).
#   M0.4  BOX-FREE at BOTH guards, rec_-based 5% gate, PRIMARY and UNCHANGED.
#         tw0's nodewise@1e-4 was box-free 3/3 and all 15 ck1 arms were box-free,
#         so a bind here would be NEW and would be reported as such.
#
#   M1    **THE PRIMARY, AND IT IS WITHIN-BATCH.**
#         D = plateau5(chunk777) - plateau5(nodewise), both measured HERE.
#         **PREDICTS D > +0.30 pp** -> partition structure matters at matched
#              count; c76's +0.517 pp residual is real and is not an artifact of
#              the interpolation or of the cross-batch offset.  The architecture-
#              AGNOSTIC partition beats the architecture-ALIGNED one.
#         **REFUTES if D <= +0.15 pp (INCLUDING NEGATIVE)** -> K2's +0.565 was
#              carried by the 24% count difference and/or the batch offset, and
#              "granularity == the group count" survives after all.
#         **(+0.15, +0.30] IS REGISTERED IN ADVANCE AS UNDECIDED** so that a
#              result landing in the gap cannot be squeezed into either verdict
#              after the fact.  This band is written down BEFORE the data exists.
#
#   M2    THE BATCH-OFFSET METER.  nodewise@1e-4 here vs tw0's 91.961, bar +-0.50.
#         **THIS CANNOT VOID M1.**  M1 is a within-batch difference and is immune
#         to any common offset.  M2 measures the offset for the RECORD, and a
#         failure would void only the comparability of this batch's ABSOLUTE
#         levels with tw0's -- which M1 does not use.  Stated this way in advance
#         so a VOID cannot later be used to discard an inconvenient M1.
#
#   M3    THE FIELD AT MATCHED COUNT.  **DESCRIPTIVE, NO DIRECTION REGISTERED.**
#         a_raw(chunk777) vs a_raw(nodewise).  ck1's K4b saw the field move with
#         the partition (-0.01556 at nearly fixed m); this is that comparison with
#         the count difference removed entirely.  No prior measurement at EXACTLY
#         matched m exists, so no direction is registered and none may be inferred
#         afterwards.  **The "53.1%" sentence is NOT reproduced and must NOT be
#         written.**
#
# WHAT THIS BATCH DELIBERATELY DOES NOT DO
#   * It does NOT control the group-SIZE DISTRIBUTION.  At matched m the MEAN
#     group size is identical by construction, but nodewise's groups are
#     heterogeneous (per-layer channel counts) and chunk777's are uniform at 777
#     except for ragged tails.  So a confirmed M1 says "the partition matters",
#     NOT yet "architecture alignment matters".  Separating those needs a permuted
#     partition carrying nodewise's exact size multiset -- a CODE change, and it
#     does not belong in the same batch as a measurement.
#   * It does NOT sweep ms.  ms=1e-4 is held because guard 2b re-derives that it is
#     the INTERIOR argmax of the chunk family's endpoints; nodewise's own argmax is
#     3e-4 (72.1b), so nodewise is read OFF its argmax here -- deliberately, because
#     K2's refuted comparison was also at 1e-4 and M1 must be the SAME contrast with
#     the confounds removed, not a different one.  **THE RISK IS STATED: a confirmed
#     M1 is a statement at ms=1e-4, not at each partition's own optimum.**
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 analysis/c75_ck1_score.py --selftest            (74/74)
#   2. python3 analysis/c76_partition_vs_count.py --selftest   (27/27)
#   3. python3 analysis/c55_neff_noise.py --selftest           (51/51)
#   4. `mm1` registered in c55_neff_noise.BOXES BEFORE scoring, from THIS CLIP=.
#   5. plateau5 is PRIMARY.  best_test inflates ~0.37 pp.
#
# USAGE
#   bash c76_matched_m_partition.sh            # dry run
#   bash c76_matched_m_partition.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

WS=/cluster/account_a/metaopt
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/mm1
CIFAR=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=$(whoami)

NJOBS=6
CLIP=-15:-2.3026               # tw0's and ck1's box EXACTLY -- not a new box
MST=1e-4                       # the ms K2's refuted comparison was made at
EPOCHS=100
WALL=03:50:00
WRITE_EVERY=500
CHUNK_K=777                    # m(777) = 14,421, ONE group from nodewise's 14,420
M_CHUNK=14421
M_NODE=14420

REPO="$(cd "$(dirname "$0")/.." && pwd)"
CSVP="$REPO/results/all_runs.csv"

echo "=============================================================="
echo "  mm1 -- THE MATCHED-COUNT PARTITION CONTRAST"
echo "    chunk${CHUNK_K} (m=$M_CHUNK) vs nodewise (m=$M_NODE), seeds 0,1,2 = $NJOBS jobs"
echo "    ms=$MST, BETA_CLIP=$CLIP, ${EPOCHS}ep, PROBE=5 AND PROBE5=1"
echo "    M1 (PRIMARY, WITHIN-BATCH): D = chunk${CHUNK_K} - nodewise"
echo "       PREDICTS D > +0.30 | REFUTES D <= +0.15 | (+0.15,+0.30] UNDECIDED"
echo "=============================================================="

# --- GUARD 0 ---------------------------------------------------------------
[ -f "$CSVP" ] || { echo "GUARD FAIL: no CSV at $CSVP"; exit 1; }
[ -f "$RUNNER" ] || { echo "GUARD FAIL: no runner at $RUNNER"; exit 1; }
echo "guard 0: runner and CSV present ($(wc -l < "$CSVP") CSV lines)"

# --- GUARD 1 -- the probe patch is present AND ENABLED (STANDING RULE 19) ---
python3 - "$WS" "$REPO" "$0" <<'PYEOF' || exit 1
import os, re, sys
ws, repo, me = sys.argv[1], sys.argv[2], sys.argv[3]
hf = os.path.join(ws, "MetaOptimize", "codes", "Supervised_tasks", "MetaOptimize",
                  "cifar10", "Optimizers", "HF.py")
if not os.path.exists(hf):
    sys.exit("GUARD FAIL: %s absent" % hf)
src = open(hf).read()
if not re.search(r"os\.environ\.get\(\s*['\"]PROBE5['\"].*?\)\s*==\s*['\"]1['\"]", src):
    sys.exit("GUARD FAIL: HF.py's PROBE5 gate is not the =='1' form this script exports for")
print("guard 1a: HF.py gates the sign-count writer on PROBE5 == '1'")
mine = [l for l in open(me).read().splitlines()
        if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
if len(mine) != 1:
    sys.exit("GUARD FAIL: expected exactly 1 --export line with PROBE_DIR, found %d" % len(mine))
for need in ("PROBE=5", "PROBE5=1", "PROBE5_WRITE_EVERY="):
    if need not in mine[0]:
        sys.exit("GUARD FAIL: export line missing %r -- that is bf8's bug" % need)
print("guard 1b: this script exports PROBE=5 AND PROBE5=1 AND PROBE5_WRITE_EVERY")
tw0 = os.path.join(repo, "bin", "c74_tuned_weightwise_probe.sh")
ref = [l for l in open(tw0).read().splitlines()
       if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
def keys(l):
    return sorted(k.split("=")[0] for k in re.search(r'"ALL,([^"]+)"', l).group(1).split(","))
if keys(mine[0]) != keys(ref[0]):
    sys.exit("GUARD FAIL: export KEYS differ from tw0's CONFIRMED-FIRING line")
print("guard 1c: export keys identical to tw0's CONFIRMED-FIRING line")
PYEOF

# --- GUARD 1d -- PATCH_CHUNKWISE IS PRESENT ON THIS ACCOUNT ----------------
python3 - "$WS" <<'PYEOF' || exit 1
import os, sys
hf = os.path.join(sys.argv[1], "MetaOptimize", "codes", "Supervised_tasks",
                  "MetaOptimize", "cifar10", "Optimizers", "HF.py")
src = open(hf).read()
if "PATCH_CHUNKWISE" not in src:
    sys.exit("GUARD FAIL: PATCH_CHUNKWISE is NOT applied to %s" % hf)
for need in ("chunkwise", "chunk_size", "repeat_interleave"):
    if need not in src:
        sys.exit("GUARD FAIL: PATCH_CHUNKWISE marker present but %r missing -- partial" % need)
print("guard 1d: PATCH_CHUNKWISE present and complete on this account")
PYEOF

# --- GUARD 2 -- the 13-axis signature, re-derived FROM THE CSV --------------
python3 - "$CSVP" "$CLIP" "$EPOCHS" "$MST" <<'PYEOF' || exit 1
import csv, sys, statistics
csvp, clip, epochs, mst = sys.argv[1:5]
rd = list(csv.DictReader(open(csvp)))
AX = ['network','dataset','batch_size','base','meta','alpha0','gamma',
      'augment','beta_clip','hier','lam','eta_ratio','epochs_done']
ref = [r for r in rd if r['run'].startswith('rs-lay-1e4')]
if not ref:
    sys.exit("GUARD FAIL: no rs-lay-1e4 reference row")
sig = {a: ref[0][a] for a in AX}
mine = {'network':'ResNet18','dataset':'CIFAR10','batch_size':'100','base':'SGDm',
        'meta':'Lion','alpha0':'1e-3','gamma':'1','augment':'1',
        'beta_clip':clip,'epochs_done':epochs}
bad = [(a, mine[a], sig[a]) for a in mine if mine[a] != sig[a]]
if bad:
    for a,m,s in bad: print("  axis %s: this batch %r vs reference %r" % (a,m,s))
    sys.exit("GUARD FAIL: mm1 does not match the ladder's axis signature")
print("guard 2: all 13 axes match the rs-lay-1e4 signature read FROM THE CSV")

# guard 2b: M2's reference is what this script registered, read from the CSV.
tn = [float(r['plateau5']) for r in rd
      if r['run'].startswith('tw0-node-') and r.get('plateau5','').strip()]
if len(tn) != 3 or abs(statistics.mean(tn) - 91.961) > 0.001:
    sys.exit("GUARD FAIL: M2 reference is %.3f (n=%d), not the 91.961 (n=3) registered"
             % (statistics.mean(tn) if tn else float('nan'), len(tn)))
print("guard 2b: M2 reference tw0 nodewise@1e-4 = %.3f (n=3) as registered" % statistics.mean(tn))

# guard 2c: the chunk1024 vs nodewise contrast this batch is de-confounding must
# STILL read the way c76 reduced it -- if the CSV has moved, the batch is stale.
c1024 = [float(r['plateau5']) for r in rd
         if r['run'].startswith('ck1-k1024-') and r.get('plateau5','').strip()]
if len(c1024) != 3:
    sys.exit("GUARD FAIL: expected ck1's 3 chunk1024 rows, found %d" % len(c1024))
raw = statistics.mean(c1024) - statistics.mean(tn)
if abs(raw - 0.565) > 0.002:
    sys.exit("GUARD FAIL: K2's raw contrast reads %+.3f now, not the +0.565 c76 "
             "decomposed.  The CSV moved; re-run the reducer before submitting." % raw)
print("guard 2c: K2's raw contrast still reads %+.3f pp as c76 decomposed it" % raw)
PYEOF

# --- GUARD 3 -- the box is registered BEFORE any job runs -------------------
python3 - "$REPO" "$CLIP" <<'PYEOF' || exit 1
import sys, os
repo, clip = sys.argv[1], sys.argv[2]
sys.path.insert(0, os.path.join(repo, "analysis"))
import c55_neff_noise as c55
if "mm1" not in c55.BOXES:
    sys.exit("GUARD FAIL: `mm1` is not registered in c55_neff_noise.BOXES.  Register "
             "it from THIS script's CLIP= line BEFORE submitting or the root is skipped.")
lo, hi, prov = c55.BOXES["mm1"]
want = tuple(float(x) for x in clip.split(":"))
if (lo, hi) != want:
    sys.exit("GUARD FAIL: c55 registers mm1 as %s but CLIP= is %s" % ((lo,hi), want))
print("guard 3: mm1 registered as (%.1f, %.4f) from %s" % (lo, hi, prov))
PYEOF

# --- GUARD 4 -- m IS MATCHED, MEASURED FROM THE ALLOCATED BETA -------------
# The whole batch is an argument about two partitions having the SAME group count.
# That count is therefore MEASURED here, on the real built network, from the beta
# the optimizer actually allocates -- not computed from a formula and not read off
# a comment.  This is the guard the batch lives or dies by.
(
  module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1
  # shellcheck disable=SC1091
  source "$WS/envs/mo/bin/activate"
  cd "$CIFAR" || exit 1
  python - "$CHUNK_K" "$M_CHUNK" "$M_NODE" <<'PYEOF' || exit 1
import sys, math, torch
sys.path.insert(0, ".")
from build_network import build_network
from Optimizers.HF import HF
K, want_chunk, want_node = int(sys.argv[1]), int(sys.argv[2]), int(sys.argv[3])

# The SAME construction tests/test_chunkwise.py uses, so the count measured here is
# the count that suite verified against the allocated beta.
BASE = {"alg": "SGDm", "weight_decay": 0.1, "momentum_param": 0.99}
META = {"alg": "Lion", "meta_stepsize": 1e-4, "momentum_param": 0.99,
        "Lion_beta2": 0.9, "weight_decay": 0}

class NullWriter:
    def add_scalar(self, *a, **k):
        pass

def alloc(gran):
    torch.manual_seed(0)
    net = build_network("ResNet18", "cpu")
    opt = HF(net, stepsize_groups=gran, alpha0=1e-3, args_base=dict(BASE),
             args_meta=dict(META), gamma=1, writer=NullWriter())
    return int(sum(int(b.numel()) for b in opt.beta))

net = build_network("ResNet18", "cpu")
ns = [int(p.numel()) for p in net.parameters()]
formula = sum(math.ceil(n / K) for n in ns)
m_chunk = alloc("chunk%d" % K)
m_node  = alloc("nodewise")
print("guard 4: chunk%d  formula m=%d   ALLOCATED beta m=%d" % (K, formula, m_chunk))
print("guard 4: nodewise                ALLOCATED beta m=%d" % m_node)
if formula != m_chunk:
    sys.exit("GUARD FAIL: chunk%d formula %d != allocated %d -- PATCH_CHUNKWISE does "
             "not partition the way the ladder claims" % (K, formula, m_chunk))
if m_chunk != want_chunk:
    sys.exit("GUARD FAIL: m(chunk%d) is %d, not the %d registered" % (K, m_chunk, want_chunk))
if m_node != want_node:
    sys.exit("GUARD FAIL: m(nodewise) is %d, not the %d registered" % (m_node, want_node))
d = abs(m_chunk - m_node)
if d > 1:
    sys.exit("GUARD FAIL: the counts differ by %d groups; M1 is not a matched-m "
             "contrast and must not be submitted as one" % d)
print("guard 4b: counts differ by %d group (%.1e relative); mean group size %.1f vs %.1f"
      % (d, d / m_node, sum(ns) / m_chunk, sum(ns) / m_node))
print("guard 4c: MATCHED-m contrast verified from the ALLOCATED beta, not a formula")
PYEOF
) || { echo "GUARD 4 FAILED -- nothing submitted"; exit 1; }

# --- GUARD 5 -- disk -------------------------------------------------------
python3 - "$WS" <<'PYEOF' || exit 1
import shutil, sys
free_gb = shutil.disk_usage(sys.argv[1]).free / 1e9
need_gb = 6 * 0.060
if free_gb < 20 * need_gb:
    sys.exit("GUARD FAIL: %.1f GB free vs ~%.2f GB needed (20x wanted)" % (free_gb, need_gb))
print("guard 5: %.0f GB free vs ~%.2f GB needed  OK" % (free_gb, need_gb))
PYEOF

# --- GUARD 6 -- queue depth ------------------------------------------------
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
RUN=$(squeue -h -u "$USER_NAME" -t RUNNING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 6: FairShare=${FS:-unknown} (informational), pending=$PEND running=$RUN, adding $NJOBS"
if [ "$PEND" -gt 40 ]; then
  echo "GUARD FAIL: $PEND already pending, over the 40 cap"; exit 1
fi

# ---------------------------------------------------------------------------
mkdir -p "$SAVE"
N=0
FAIL=0
SGDM=(--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0)
for ARM in "chunk${CHUNK_K}:ch" "nodewise:node"; do
  GRAN="${ARM%%:*}"; SHORT="${ARM##*:}"
  for S in 0 1 2; do
    RN="mm1-${SHORT}-s${S}"
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --gres=gpu:1 --cpus-per-task=6
      --mem=14G --time="$WALL" --nice=0
      --export="ALL,AUGMENT=1,BETA_CLIP=${CLIP},PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=${WRITE_EVERY},PROBE_DIR=$SAVE/probe_${SHORT}_mm1_s${S}"
      "$RUNNER" "${SGDM[@]}"
      --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
      --max-time 999:00:00 --gamma 1 --meta-stepsize "$MST" --alpha0 1e-3
      --num-epochs "$EPOCHS" --stepsize-groups "$GRAN" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    # COUNT WHAT SLURM ACCEPTED, NOT WHAT WE TRIED (CORRECTIONS 71).
    if [ "$SUBMIT" = "1" ]; then
      if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
    else
      N=$((N+1)); printf '%s\n' "${CMD[*]}"
    fi
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
[ "$N" = "$NJOBS" ] || echo "!!! expected $NJOBS jobs, built $N -- DO NOT TREAT AS SUBMITTED"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
exit 0
