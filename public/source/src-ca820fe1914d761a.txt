NODE_RE = re.compile(r"^NODE=(\S+)\s*\|\s*JOB=(\S+)\s+(\d+)")
GPU_RE = re.compile(r"^\s*(NVIDIA[^,]*),\s*([0-9]+)\s*MiB")
CLIP_RE = re.compile(r"BETA_CLIP=(\S+)")
PROBEENV_RE = re.compile(r"PROBE=(\S+)")


def parse_args_line(line):
    """The run's OWN ARGS line -> {flag: value}, LAST occurrence wins (argparse's
    own rule), plus a list of any flag that appeared more than once."""
    toks = line.split()
    if toks and toks[0] == "ARGS:":
        toks = toks[1:]
    out, seen, dup = {}, {}, []
    i = 0
    while i < len(toks):
        t = toks[i]
        if t.startswith("--"):
            k = t[2:]
            v = "1"
            if i + 1 < len(toks) and not toks[i + 1].startswith("--"):
                v = toks[i + 1]
                i += 1
            seen[k] = seen.get(k, 0) + 1
            if seen[k] == 2:
                dup.append(k)
            out[k] = v
        i += 1
    return out, dup


def provenance(path):
    """Everything hz3q gates on, MEASURED off one .out file.  No declarations."""
    p = dict(path=path, file=os.path.basename(path), node=None, job=None,
             job_id=None, gpu=None, gpu_mib=None, clip=None, probe=None,
             args=None, dup_flags=None, n_epochs=0, last_epoch=None)
    with open(path, "r", errors="replace") as fh:
        for i, line in enumerate(fh):
            if i < 40:
                m = NODE_RE.match(line)
                if m:
                    p["node"], p["job"], p["job_id"] = m.group(1), m.group(2), int(m.group(3))
                    continue
                if line.startswith("ARGS:"):
                    p["args"], p["dup_flags"] = parse_args_line(line)
                    continue
                if line.startswith("ENV:"):
                    m = CLIP_RE.search(line)
                    p["clip"] = m.group(1) if m else None
                    m = PROBEENV_RE.search(line)
                    p["probe"] = m.group(1) if m else None
                    continue
                m = GPU_RE.match(line)
                if m and p["gpu"] is None:
                    p["gpu"], p["gpu_mib"] = m.group(1).strip(), int(m.group(2))
                    continue
            m = C87.EP_RE.search(line)
            if m:
                p["n_epochs"] += 1
                p["last_epoch"] = int(m.group(1))
    return p


def newest_out(runs_dir, run_name):
    """The file the reader would read: same rule as c87_hz3_score.load_arm."""
    if not os.path.isdir(runs_dir):
        return None
    cands = [os.path.join(runs_dir, fn) for fn in os.listdir(runs_dir)
             if fn.startswith(run_name + "-") and fn.endswith(".out")]
    return sorted(cands)[-1] if cands else None


def quartet(runs_dir, tag, seed):
    """{short: provenance} for one tag/seed, or {} if nothing is on disk."""
    out = {}
    for short in ARMS:
        f = newest_out(runs_dir, "%s-%s-s%d" % (tag, short, seed))
        if f:
            out[short] = provenance(f)
    return out


def homogeneous(q, field):
    """(value, ok) -- ok iff every arm reports the same non-None value."""
    vals = [q[s].get(field) for s in sorted(q)]
    if not vals or any(v is None for v in vals):
        return None, False
    return vals[0], len(set(vals)) == 1


def gate_provenance(q, want_clip=CLIP, want_seed=SEED, want_epochs=EPOCHS):
    """The H0 predicate, as a list of failures.  [] means PASS.

    This is the ONE function whose behaviour separates hz3q from the parent
    registration: it reads the GPU model, the beta box and the design off each
    run's own header and compares them to EACH OTHER.  It declares nothing."""
    fails = []
    if sorted(q) != sorted(ARMS):
        fails.append("quartet is incomplete: have %s, need %s"
                     % (sorted(q), sorted(ARMS)))
        return fails
    gpu, ok = homogeneous(q, "gpu")
    if not ok:
        fails.append("GPU MODEL IS NOT SHARED across the four arms: %s"
                     % {s: q[s]["gpu"] for s in sorted(q)})
    clip, ok = homogeneous(q, "clip")
    if not ok:
        fails.append("BETA_CLIP IS NOT SHARED across the four arms: %s"
                     % {s: q[s]["clip"] for s in sorted(q)})
    elif clip != want_clip:
        fails.append("BETA_CLIP is %r on all four arms, not the registered %r"
                     % (clip, want_clip))
    for short in sorted(q):
        a = q[short]["args"] or {}
        gran = ARMS[short][0]
        if a.get("stepsize-groups") != gran:
            fails.append("%s: --stepsize-groups is %r, not the registered %r"
                         % (short, a.get("stepsize-groups"), gran))
        if a.get("seed") != str(want_seed):
            fails.append("%s: --seed is %r, not %d" % (short, a.get("seed"), want_seed))
        if a.get("num-epochs") != str(want_epochs):
            fails.append("%s: --num-epochs is %r, not %d"
                         % (short, a.get("num-epochs"), want_epochs))
        if a.get("meta-stepsize") != MST:
            fails.append("%s: --meta-stepsize is %r, not %r"
                         % (short, a.get("meta-stepsize"), MST))
        if a.get("alpha0") != ALPHA0:
            fails.append("%s: --alpha0 is %r, not %r" % (short, a.get("alpha0"), ALPHA0))
        if q[short]["dup_flags"]:
            fails.append("%s: STANDING RULE 20 -- flag(s) %s appear TWICE in the "
                         "run's own ARGS line" % (short, q[short]["dup_flags"]))
        if q[short]["n_epochs"] < want_epochs:
            fails.append("%s: only %d epochs on disk, needs %d"
                         % (short, q[short]["n_epochs"], want_epochs))
        if q[short]["probe"] not in (None, str(PROBE_EVERY)):
            fails.append("%s: PROBE=%s, not %d" % (short, q[short]["probe"], PROBE_EVERY))
    return fails

# NON-OVERWRITE.  COMBINATION 1/2, asserted rather than asserted-in-prose.
# =============================================================================
def assert_no_overwrite(runs_dir=None):
    """[] means PASS.  Runs the PARENT scorer's own admission predicate over
    hz3q's names and requires it to reject every one of them."""
    fails = []
    for short in ARMS:
        parent_rn = "%s-%s-s%d" % (PARENT_TAG, short, SEED)
        ours = "%s-%s-s%d-4999999.out" % (TAG, short, SEED)
        if ours.startswith(parent_rn + "-"):
            fails.append("%s WOULD BE ADMITTED by the parent scorer's glob for %s "
                         "-- hz3q would OVERWRITE hz3's seed 5" % (ours, parent_rn))
        theirs = "%s-4782027.out" % parent_rn
        if theirs.startswith("%s-%s-s%d-" % (TAG, short, SEED)):
            fails.append("%s would be admitted by hz3q's own glob -- hz3q would "
                         "READ hz3's runs as its own" % theirs)
    pdir_parent = "probe_%%s_%s_s%d" % (PARENT_TAG, SEED)
    pdir_ours = "probe_%%s_%s_s%d" % (TAG, SEED)
    if pdir_parent == pdir_ours:
        fails.append("hz3q's probe directory name collides with hz3's -- HF.py "
                     "appends to probe.jsonl, so two boxes would interleave")
    if runs_dir and os.path.isdir(runs_dir):
        for short in ARMS:
            for fn in os.listdir(runs_dir):
                if fn.startswith("%s-%s-s%d-" % (TAG, short, SEED)) and \
                        fn.startswith("%s-%s-s%d-" % (PARENT_TAG, short, SEED)):
                    fails.append("on-disk collision on %s" % fn)
    return fails

def band_flat(delta, t):
    """H2's verdict on the 100 -> 300 change in D.  See FLAT_T above."""
    if any(x is None or (isinstance(x, float) and math.isnan(x)) for x in (delta, t)):
        return "NOT EVALUABLE"
    if abs(t) < FLAT_T:
        return "FLAT"
    return "NOT FLAT -- D DECLINES WITH BUDGET" if delta < 0 else \
           "NOT FLAT -- D GROWS WITH BUDGET"


def band_class(delta_pp):
    """HC's DISCLOSURE, never a verdict."""
    if delta_pp is None or math.isnan(delta_pp):
        return "NOT MEASURABLE"
    return ("GPU CLASS NOT FIRST-ORDER ON THE LEVEL" if abs(delta_pp) <= CTRL_CLASS_PP
            else "GPU CLASS IS FIRST-ORDER ON THE LEVEL -- DISCLOSE")

def arm_series(runs_dir, tag, short, seed):
    f = newest_out(runs_dir, "%s-%s-s%d" % (tag, short, seed))
    return (C87.series_from_out(f), f) if f else (None, None)


def d_at(ser_hi, ser_lo, budget, w=WINDOW):
    a = C87.window_at(ser_hi, budget, w) if ser_hi else None
    b = C87.window_at(ser_lo, budget, w) if ser_lo else None
    return None if (a is None or b is None) else a - b


def collect(runs_dir, qruns_dir):
    """{seed: {'D': {budget: v}, 'G': {...}, 'acc': {short: {budget: v}},
               'src': tag}} -- seeds 0-4 from the ARCHIVE, seed 5 from BOTH."""
    out = {}
    for seed in PARENT_SEEDS + [SEED]:
        for tag, root in (((PARENT_TAG, runs_dir),) if seed != SEED
                          else ((PARENT_TAG, runs_dir), (TAG, qruns_dir))):
            ser, files = {}, {}
            for short in ARMS:
                ser[short], files[short] = arm_series(root, tag, short, seed)
            if any(ser[s] is None for s in ARMS):
                continue
            rec = dict(tag=tag, files=files,
                       acc={s: {b: C87.window_at(ser[s], b) for b in READINGS}
                            for s in ARMS},
                       D={b: d_at(ser["ch"], ser["node"], b) for b in READINGS},
                       G={b: d_at(ser["c23"], ser["n1d"], b) for b in READINGS})
            out.setdefault(seed, {})[tag] = rec
    return out


def flat_test(rows):
    """rows = [(D100, D300)] -> (delta, se, t, df, n).  Paired within seed."""
    rows = [r for r in rows if r[0] is not None and r[1] is not None]
    if len(rows) < 2:
        return (float("nan"),) * 4 + (len(rows),)
    d = [b - a for a, b in rows]
    m = st.mean(d)
    se = st.stdev(d) / math.sqrt(len(d))
    t = m / se if se else float("nan")
    return m, se, t, len(d) - 1, len(d)

def selftest(out=sys.stdout):
    n = [0, 0]

    def chk(cond, msg):
        n[1] += 1
        if cond:
            n[0] += 1
        else:
            print("  FAIL %2d: %s" % (n[1], msg), file=out)

    repo = os.path.dirname(_HERE)
    bs_path = os.path.join(repo, BATCH_SCRIPT)
    bs = open(bs_path).read() if os.path.exists(bs_path) else ""

    # c1  the batch script exists and is the one registered here
    chk(bool(bs), "the registered batch script %s must exist" % BATCH_SCRIPT)

    # c2-c8  every constant, against the script's own text
    for name, val in (("TAG", TAG), ("SEED", str(SEED)), ("CLIP", CLIP),
                      ("EPOCHS", str(EPOCHS)), ("PART", PART), ("WALL", WALL),
                      ("PROBE_EVERY", str(PROBE_EVERY))):
        chk(re.search(r"^%s=%s\s*(#|$)" % (name, re.escape(val)), bs, re.M) is not None,
            "%s must read %s=%s -- registration and batch would drift"
            % (BATCH_SCRIPT, name, val))

    # c9  all FOUR arms, with their registered m, are in the batch
    for short, (gran, m, _leg) in ARMS.items():

            "which is the whole point of the design" % (gran, short, m))

                "parent scorer sha256 %s != pinned %s -- it has been EDITED"

                                  "num-epochs": str(EPOCHS), "meta-stepsize": MST,

                q[short]["args"][field[4:]] = v
            else:
                q[short][field] = v

    chk(gate_provenance(synth()) == [],

        "RULE 13: the gate must REFUSE a quartet whose GPU model differs on one arm"

    chk(gate_provenance(synth(ch__clip="-15:-2.3026")) != [],
        "RULE 13: the gate must REFUSE a quartet whose BETA_CLIP differs on one arm")

    for name in ("EPOCHS", "STEPS_PER_EPOCH", "PROBE_EVERY", "MST", "ALPHA0",

        "the four arms must be the parent's own arms")

    # c18  the box arithmetic that makes H1's expectation exactly zero
    b0 = math.log(float(ALPHA0))

    chk(C87.window_at({e: 1.0 for e in range(300)}, 300, 5) == 1.0,

    # c20  the parse of a real ARGS line, including the duplicate detector
    real = ("ARGS: --optimizer HF --alg-base SGDm --alg-meta Lion --dataset CIFAR10 "
            "--NN-name ResNet18 --num-epochs 300 --stepsize-groups chunk777 --seed 5")
    a, d = parse_args_line(real)

    a2, d2 = parse_args_line(real + " --alg-meta RMSProp")

    # c21  provenance() on the real archived run, if it is reachable locally
    for cand in (os.path.join(repo, "..", "runs"), os.path.join(repo, "runs")):
        f = newest_out(os.path.abspath(cand), "hz3-node-s5")
        if f:

def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runs", default="runs",
                    help="directory holding the ARCHIVED hz3-*.out (the ROOT of runs/)")
    ap.add_argument("--probes", default=None, help="hz3's probe parent (runs/hz3)")
    ap.add_argument("--qruns", default=None,
                    help="directory holding hz3q-*.out (defaults to --runs)")
    ap.add_argument("--qprobes", default=None,
                    help="hz3q's probe parent (defaults to --probes)")
    a = ap.parse_args()
    if a.selftest:
        sys.exit(0 if selftest() else 1)
    score(a.runs, a.probes or a.runs, a.qruns or a.runs,
          a.qprobes or a.probes or a.runs)


if __name__ == "__main__":
    main()
