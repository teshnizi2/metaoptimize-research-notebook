#!/bin/bash
# =============================================================================
# cVT5_plainhorizon.sh -- `cvt5`: PLAINNET AT 300 EPOCHS -- IS cvt2's GRADED A
#   DELAY, AND DOES HEAD's RESCUE SURVIVE ITS OWN COMPLEMENT PIN?  CORRECTIONS 238.
#   ONE SUBMISSION, 12 JOBS (4 arms x 3 seeds), 300 EPOCHS (the 100-epoch control
#   READ IN-RUN), SEEDS {93,94,95}, PlainNet18_c100 / CIFAR-100, PROBE=100,
#   PROBE_TENSOR=1 ON EVERY ARM.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account; every guard and scancel line is filtered on `cvt5-`.
#
# REGISTERED SCORER: analysis/cVT5_plainhorizon_score.py, committed BEFORE this batch
# is submitted (RULE 21).  Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT5_plainhorizon_score.py <runsdir>
#
# NO NEW HARNESS CODE.  The network runs from cvt1's OWN isolated tree, UNCHANGED:
#   $WS/harness_cvt1/cifar10 (PATCH_PLAINNET + PATCH_VOTEWEIGHT, verified by cvt1's
#   bin/cVT1_stage_harness.sh in VERIFY mode) and $WS/jobs/run_cifar_cvt1.sh.  The
#   four VOTE_W strings are cvt2's byte for byte, and cvt2's real-run ladder log
#   (46/0, $WS/runs/cvt2/inertness_ladder_realrun.log) proved inertness (k01, HEAD) and
#   bite (K13, K33) for exactly these strings on this tree; guard 4g0/4g1 re-require it.
#
# THE ARMS (PlainNet indices; guard 4 re-derives every one on the LIVE 53-tensor model)
#   k01    scalar                                VOTE_W unset
#   HEAD   sets:1-49,51-53/layer4.1.bn2.weight   VOTE_W unset                    (K = 1)
#   K13    HEAD's string                         VOTE_W=layer4.1.bn1.weight:13
#   K33    HEAD's string                         VOTE_W=layer4.1.bn1.weight:33
#
# THE HORIZON (analysis/cvt5_registration_derivations.py; scorer header).  K13's
#   complement at 100: beta -12.02 / -12.31 / -12.14, still falling; its r <= 2 pin under
#   five model families 134.1-223.5 (latest: ln, s87).  E_MIN = max(ceil(223.5 + W_POST
#   64.2), ceil(223.5 / 0.75)) = max(288, 298) = 298; E = 300 (next multiple of 10).
#   HEAD's complement pins by magnitude at 62.2-65.8 -> >= 234 post-pin epochs at 300.
#
# THE 100-EPOCH CONTROL IS IN THE SAME RUNS.  Guard 4h proves on BOTH the tree's and
# the live train.py that num_epochs is only the argparse default and the loop bound;
# guard 4h2 proves on the tree's HF.py that SCHED=none's `_sched_factor()` is exactly
# 1.0 at every counter up to 1e7 (incl. 150,000) and that PATCH_VOTEWEIGHT reads no
# step counter; guard 4h3 refuses if a schedule variable is set in this shell.
#
# WALLTIME, PARTITIONS, COST -- MEASURED ON THIS NETWORK'S OWN RUNS
#   cvt1 + cvt2 + cvt3 (49 PlainNet runs on the L4 / A100-MIG / A100 node set of the
#   7-day partitions): 2146-2836 s per 100 epochs.  300 epochs at the worst = 2.363 h.
#   WALL=05:00:00 = 2.12x that, no checkpointing.  5 h > gpu-short's 4:00:00, so
#   slurm_parts_for_wall DROPS gpu-short (and its 2080 Ti nodes) by arithmetic.
#   COST 0.006838265 GPU-h / run-epoch x 12 x 300 = 24.62 GPU-h EXPECTED (without k01
#   18.46); HARD BOUND = WALL x 12 = 60 GPU-h.
#
# RULE 20.  Every ARGS value is ONE shell token.  VOTE_W rides sbatch's --export list and
# CANNOT ride the runner's ENV line: the patch prints `VOTE_W: off` / `VOTE_W: on ...` on
# EVERY run, and that line is what the separate ENV audit and the scorer's G-VOTEW read.
# analysis/argsline_guard.py IS NOT EDITED.
#
# USAGE   bash bin/cVT5_plainhorizon.sh               # dry run: every guard, every
#         bash bin/cVT5_plainhorizon.sh --dry-run     #   composed line printed; submits nothing
#         bash bin/cVT5_plainhorizon.sh --submit      # the ONLY submitting form
# ENV     CVT5_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# the control arms must NOT inherit a VOTE_W through --export=ALL
unset VOTE_W
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cvt1/cifar10
RUNNER=$WS/jobs/run_cifar_cvt1.sh
LIVERUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cvt5
CVT1SAVE=$WS/runs/cvt1
CVT2SAVE=$WS/runs/cvt2

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=300; CTRL_EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=PlainNet18_c100
SEEDS="93 94 95"
MST=1e-3
A0=1e-6
NTENS=53
TOTPAR=11046308
G_K01='scalar'
G_HEAD='sets:1-49,51-53/layer4.1.bn2.weight'
W_K13='layer4.1.bn1.weight:13'
W_K33='layer4.1.bn1.weight:33'
ARMS="k01 HEAD K13 K33"
NJOBS=12
WALL=05:00:00
POST_SHA=e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b
HF_PRE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_POST_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
RUNNER_SHA=d389e8a5f3d85ce7dd4b7640fbcfa2d9075f36eb79c90fe2a7c21727d77c97c3
LADDER_LOG_SHA=0703a161cae3d0bb220976ff9d4b06fa762888f90c5d78a2978c3e9a32cc728c
# the live harness, pinned (guard 3e before, guard 10 after)
PIN_LIVE_BN=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
PIN_LIVE_HF=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
PIN_LIVE_TRAIN=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
PIN_LIVE_LOAD=b52b58a35dcecd80ce6d8f8033b09c20dad9f4827b808f4ea2a828ce88138c46
PIN_LIVE_RUNNER=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT5_plainhorizon_score.py"
CE="$REPO/analysis/corpus_exclusions.py"
CET="$REPO/results/CORPUS-EXCLUSIONS.tsv"
DERIV="$REPO/analysis/cvt5_registration_derivations.py"
SELF="$REPO/bin/cVT5_plainhorizon.sh"
SC2="$REPO/analysis/cVT2_injectladder_score.py"
SC1="$REPO/analysis/cVT1_voteweight_score.py"
TESTL="$REPO/tests/test_voteweight_ladder_realrun.py"
STAGEF="$REPO/bin/cVT1_stage_harness.sh"
PATCHV="$REPO/patches/patch_voteweight.py"
PATCHF="$REPO/patches/patch_plainnet.py"
TESTV="$REPO/tests/test_voteweight.py"
TESTR="$REPO/tests/test_voteweight_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_plainnet.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"
LLOG="$CVT2SAVE/inertness_ladder_realrun.log"

echo "=============================================================="
echo " cvt5 -- PlainNet at $EPOCHS epochs: is GRADED a delay, and does HEAD survive its own pin?"
echo " $DSET / $NET (PlainNet indices)   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs (in-run control at $CTRL_EPOCHS); seeds $SEEDS"
echo "   k01    $G_K01      VOTE_W unset"
echo "   HEAD   $G_HEAD      VOTE_W unset"
for a in K13 K33; do v="W_$a"; echo "   $a    $G_HEAD      VOTE_W=${!v}"; done
echo " 4 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cvt1's, unchanged)   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY (a) K13@300 vs K33@300 and K13@100; (b) RHO = D_HEAD@300 / D_HEAD@100; (c) K33; pin gate AFTER, as stamps"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT5_plainhorizon_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && echo "guard 0b: VOTE_W is UNSET in the launcher's environment (control arms cannot inherit it)" \
                     || guard_fail "guard 0b: VOTE_W is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$CE" "$CET" "$DERIV" "$SELF" "$SC2" "$SC1" "$TESTL" "$STAGEF" "$PATCHV" "$PATCHF" "$TESTV" "$TESTR" "$TESTT" \
         "$TESTF" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$LIVERUNNER" "$HFH" "$HFVW" "$HFPRE" "$BNH" "$HARN/train.py" "$LIVE/train.py" "$LLOG"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cVT5_plainhorizon_score.py analysis/cvt5_registration_derivations.py bin/cVT5_plainhorizon.sh"
REUSEDF="analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv analysis/cVT2_injectladder_score.py analysis/cVT1_voteweight_score.py tests/test_voteweight_ladder_realrun.py bin/cVT1_stage_harness.sh patches/patch_voteweight.py patches/patch_plainnet.py tests/test_voteweight.py tests/test_voteweight_realrun.py tests/test_probe_tensor.py tests/test_plainnet.py bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEDF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEDF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the three new registration files and the fourteen reused ones are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT5_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT5_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT5_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT5_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING its O2 proof, the filtered floor, the horizon and the invocation ----
SELFT=/tmp/cvt5_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
[ "$(grep -c '^  SKIP' "$SELFT")" = 0 ] && echo "guard 1c0: 0 SKIP (every host-only check ran on this host)" \
  || guard_fail "guard 1c0: $(grep -c '^  SKIP' "$SELFT") selftest check(s) SKIPPED on this host"
grep -qE "^  PASS O2 EQUILIBRIUM \+ SURVIVES \(pinned\) +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt5's cells (100 AND 300 ep), and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT5_plainhorizon_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 93, 94 or 95" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
[ "$(grep -c '^  PASS SIGMA_.* re-derives THROUGH filter_rows == frozen' "$SELFT")" = 6 ] \
  && grep -q "^  PASS the UNFILTERED PlainNet sigma is" "$SELFT" \
  && echo "guard 1c5: all six frozen sigmas re-derive THROUGH corpus_exclusions.filter_rows; the unfiltered PlainNet sigma is NOT the literal (OPERATIONS s36)" \
  || guard_fail "guard 1c5: the filtered noise floor was not proved on this host"
grep -q "^  PASS the horizon re-derived from cvt2/cvt3's probe records on this host" "$SELFT" \
  && grep -q "^  PASS K33 premise re-derived" "$SELFT" && grep -q "^  PASS K13 premise re-derived" "$SELFT" \
  && echo "guard 1c6: $(grep "the horizon re-derived" "$SELFT" | sed 's/^  PASS //')" || guard_fail "guard 1c6: the horizon was not re-derived from cvt2/cvt3's records on this host"
grep -q "^  PASS SPEC for k01 / HEAD / K13 / K33 BYTE-IDENTICAL" "$SELFT" && grep -q "^  PASS VOTEW for k01 / HEAD / K13 / K33 BYTE-IDENTICAL" "$SELFT" \
  && grep -q "^  PASS WITNESS for k01 / HEAD / K13 / K33 BYTE-IDENTICAL" "$SELFT" \
  && echo "guard 1c7: SPEC / VOTEW / WITNESS byte-identical to cvt2's registered scorer" || guard_fail "guard 1c7: arm strings not proved identical to cvt2's"
[ "$(grep -c 'FLOOR GATE: no non-anchor arm predicted at the floor' "$SELFT")" = 6 ] \
  && grep -q "^  PASS DELAYED-COLLAPSE .*DECLARED floor account" "$SELFT" && ! grep -q '^  FAIL .*FLOOR GATE' "$SELFT" \
  && echo "guard 1c8: FLOOR GATE passes under the six non-floor accounts; the one declared floor account reads as bounds" \
  || guard_fail "guard 1c8: the floor gate did not pass as registered"
grep -q "^  PASS E WRONG K: K13's records carry x33" "$SELFT" && grep -q "^  PASS E the pin state NEVER moves" "$SELFT" \
  && echo "guard 1c9: a WRONG-K run reaches PATCH-NOT-VERIFIED; the pin state never moves a branch" || guard_fail "guard 1c9: wrong-K / pin-order case did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"
python3 "$CE" --check --runs "$WS/runs" >/tmp/cvt5_ce.$$ 2>&1 \
  && echo "guard 1d: corpus_exclusions.py --check PASS on this host's corpus ($(grep -c . "$CET") TSV lines)" \
  || { guard_fail "guard 1d: corpus_exclusions.py --check FAILED"; tail -6 /tmp/cvt5_ce.$$; }
python3 "$DERIV" "$WS/runs" >/tmp/cvt5_deriv.$$ 2>&1 && grep -q "^FROZEN: PIN_BOUND_R2 223.5  PIN_BOUND_EXACT_TAIL 183.8  E_MIN 298  E 300  HEAD_FIRST_R2_MAX 65.8$" /tmp/cvt5_deriv.$$ \
  && echo "guard 1e: analysis/cvt5_registration_derivations.py on this host: $(tail -1 /tmp/cvt5_deriv.$$) (output sha $(sha /tmp/cvt5_deriv.$$ | cut -c1-16))" \
  || { guard_fail "guard 1e: the derivation did not reproduce the frozen horizon"; tail -3 /tmp/cvt5_deriv.$$; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt5-' "$CSVP" && guard_fail "guard 2: cvt5-* already in the CSV" || echo "guard 2: no cvt5-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
pl = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], pl))
raise SystemExit(1 if hit or not set(pl) <= {"cpl1", "cpl2", "cvt1", "cvt2", "cvt3"} else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt5-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt5-*.out on disk" || guard_fail "guard 2b: $NOUT cvt5-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (93|94|95)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 93/94/95" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 93/94/95"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt5-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt5-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt5-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt5-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt5-* in squeue" || guard_fail "guard 2f: $NQ cvt5-* already queued"
fi
[ ! -e "$SAVE/PROVENANCE.txt" ] && echo "guard 2g: no submit-mode PROVENANCE.txt under $SAVE" || guard_fail "guard 2g: $SAVE/PROVENANCE.txt already exists"

# ---- guard 3/4a: cvt1's isolated tree is consistent (VERIFY mode) and sees CIFAR-100 -----------
bash "$STAGEF" >/tmp/cvt5_stage.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT1_stage_harness.sh (VERIFY mode, unedited) VERIFIES cvt1's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt5_stage.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt1's tree is NOT consistent"; tail -20 /tmp/cvt5_stage.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "$POST_SHA" ] && echo "guard 3b: build_network.py == cpl1's POST_SHA e65e6773...a18b" \
  || guard_fail "guard 3b: build_network.py is not cpl1's POST_SHA"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFVW")" = "$HF_PRE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA 3f2b98e1...; HF.py.pre_voteweight == the LIVE HF.py == HF_PRE_SHA 4732b74a...; .pre_probe_tensor == live" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or the live HF.py moved)"
grep -q PATCH_VOTEWEIGHT "$LIVE/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE HF.py carries PATCH_VOTEWEIGHT" \
  || echo "guard 3d: the LIVE HF.py carries no PATCH_VOTEWEIGHT"
LIVEHASH_BEFORE="$(sha "$LIVE/build_network.py") $(sha "$LIVE/Optimizers/HF.py") $(sha "$LIVE/train.py") $(sha "$LIVE/load_data.py") $(sha "$LIVERUNNER") $(sha "$RUNNER")"
[ "$LIVEHASH_BEFORE" = "$PIN_LIVE_BN $PIN_LIVE_HF $PIN_LIVE_TRAIN $PIN_LIVE_LOAD $PIN_LIVE_RUNNER $RUNNER_SHA" ] \
  && echo "guard 3e: live build_network / HF / train / load_data / run_cifar.sh and the cvt1 runner are the pinned shas" \
  || guard_fail "guard 3e: the live harness hashes are not the pinned ones: $LIVEHASH_BEFORE"
[ "$(sha "$HARN/train.py")" = "$PIN_LIVE_TRAIN" ] && cmp -s "$HARN/load_data.py" "$LIVE/load_data.py" \
  && echo "guard 3f: the tree's train.py == the live train.py (3fea309e...) and load_data.py == live" \
  || guard_fail "guard 3f: the tree's train.py / load_data.py differ from the live ones"

# ---- guard 4b/4e/4f/4g: the patch, proved inert and biting (cvt1's logs; cvt2's ladder log for these strings) ----
python3 "$TESTT" --pre "$HFPRE" --post "$HFVW" --structural-only >/tmp/cvt5_struct.$$ 2>&1 \
   && echo "guard 4b: PATCH_PROBE_TENSOR structural inertness R0 PASSES on HF.py.pre_voteweight (= live)" \
   || { guard_fail "guard 4b: PATCH_PROBE_TENSOR structural R0 FAILED on HF.py.pre_voteweight"; tail -8 /tmp/cvt5_struct.$$; }
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH")
VLOG="$CVT1SAVE/inertness_voteweight.log"
if [ -f "$VLOG" ] && grep -q '^ALL PASS$' "$VLOG" && grep -q "^POST_SHA256 $HFSHA" "$VLOG" && grep -q "^BN_SHA256 $BNSHA" "$VLOG"; then
  echo "guard 4e: cvt1's tests/test_voteweight.py log (V0-V6, CPU) ALL PASS on THIS HF and THIS build_network: $(grep -c '^  PASS' "$VLOG") PASS / $(grep -c '^  FAIL' "$VLOG") FAIL"
else
  guard_fail "guard 4e: no ALL PASS test_voteweight log for HF $HFSHA / BN $BNSHA at $VLOG"

p, how = m.resolve_manifest(sys.argv[2], None)
print("guard 8: with NO --manifest, `%s <runsdir>` resolves the manifest to %s (%s)" % (os.path.basename(sys.argv[1]), p, how))
raise SystemExit(0 if p and os.path.realpath(p) == os.path.realpath(sys.argv[3]) else 1)
PYEOF

# ---- guard 9: the harness the guards tested IS the harness the runner runs ---------
RUN_CIF=$(awk '$1=="cd"{print $2; exit}' "$RUNNER" 2>/dev/null)
[ "$(cd "$HARN" 2>/dev/null && pwd -P)" = "$(cd "$RUN_CIF" 2>/dev/null && pwd -P)" ] \
  && echo "guard 9: the runner cd's into $RUN_CIF -- the SAME tree guards 3/4 tested" \
  || guard_fail "guard 9: guards tested $HARN but the runner cd's into $RUN_CIF"

if [ "$FAIL" = 1 ]; then
  if [ "$SUBMIT" = 1 ]; then echo "!!! SUBMIT ABORTED -- nothing submitted."; exit 2; fi
  echo; echo "!!! ONE OR MORE GUARDS FAILED.  DRY RUN: the composition below still prints, but --submit WOULD ABORT."; echo
fi

# ---- compose, guard each line under RULE 20, then (only with --submit) submit ----
DESIGN=(alg-base=SGDm alg-meta=Lion num-epochs=$EPOCHS batch-size=$BATCH \
        gamma=$GAMMA dataset=$DSET NN-name=$NET weight-decay-meta=0 \
        meta-stepsize=$MST alpha0=$A0)
N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    case $ARM in
      k01)   G=$G_K01;  VW="" ;;
      HEAD)  G=$G_HEAD; VW="" ;;
      K13)   G=$G_HEAD; VW=",VOTE_W=$W_K13" ;;
      K33)   G=$G_HEAD; VW=",VOTE_W=$W_K33" ;;
      *) echo "!!! unknown arm $ARM"; exit 2 ;;
    esac
    RN="cvt5-$ARM-s$S"
    PDIR="$SAVE/probe_$RN"
    CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP_C,HIER=none,SCHED=none,PROBE=$PROBE,PROBE_TENSOR=$PT,PROBE_DIR=$PDIR$VW \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
--gamma $GAMMA --meta-stepsize $MST --alpha0 $A0 --num-epochs $EPOCHS \
--stepsize-groups $G --seed $S --save-directory $SAVE --run-name $RN"
    guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1 \
      || { echo "!!! RULE 20 PRE-CHECK FAILED for $RN"; guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1)); }
    guard_presubmit "$CMD" "stepsize-groups=$G" "seed=$S" "run-name=$RN" >/dev/null 2>&1 \
      || { echo "!!! RULE 20: an axis does not round-trip for $RN"; guard_presubmit "$CMD" "stepsize-groups=$G" "seed=$S" "run-name=$RN"; BAD=$((BAD+1)); }
    NVW=$(printf '%s\n' "$CMD" | grep -o 'VOTE_W=[^ ,]*' | wc -l | tr -d ' ')
    WANTVW=$(printf '%s' "$VW" | sed 's/^,VOTE_W=//')
    case $ARM in
      k01|HEAD) [ "$NVW" = 0 ] || { echo "!!! $RN carries a VOTE_W"; BAD=$((BAD+1)); } ;;
      *)        [ "$NVW" = 1 ] && [ "$(printf '%s\n' "$CMD" | grep -o 'VOTE_W=[^ ,]*')" = "VOTE_W=$WANTVW" ] \
                  || { echo "!!! $RN carries $NVW VOTE_W or the wrong one"; BAD=$((BAD+1)); } ;;
    esac
    NEP=$(printf '%s\n' "$CMD" | grep -o -- '--num-epochs [0-9]*' | wc -l | tr -d ' ')
    [ "$NEP" = 1 ] && printf '%s\n' "$CMD" | grep -q -- "--num-epochs $EPOCHS " \
      || { echo "!!! $RN does not carry exactly one --num-epochs $EPOCHS"; BAD=$((BAD+1)); }
    if [ "$SUBMIT" = 1 ]; then
      mkdir -p "$PDIR"
      eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
    else
      echo "$CMD"; echo; N=$((N+1))
    fi
  done
done
echo "guard 6: $N composed command lines, $BAD failed the RULE 20 / VOTE_W / num-epochs pre-check"
[ "$BAD" = 0 ] || { echo "!!! composed lines are malformed"; exit 2; }
[ "$N" = "$NJOBS" ] || { echo "!!! composed $N lines, the design says $NJOBS"; exit 2; }
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'DRY RUN, NOTHING SUBMITTED' )); $REJ rejected ----"

# ---- guard 10: the live harness is STILL the pinned one --------------------------
LIVEHASH_AFTER="$(sha "$LIVE/build_network.py") $(sha "$LIVE/Optimizers/HF.py") $(sha "$LIVE/train.py") $(sha "$LIVE/load_data.py") $(sha "$LIVERUNNER") $(sha "$RUNNER")"
[ "$LIVEHASH_AFTER" = "$LIVEHASH_BEFORE" ] \
  && echo "guard 10: live harness + runner hashes UNCHANGED across this launcher run" \
  || echo "!!! guard 10: a live harness hash CHANGED during this launcher run: $LIVEHASH_BEFORE -> $LIVEHASH_AFTER"

if [ "$SUBMIT" = 1 ]; then
  grep -qx 'cvt5-' "$REPO/bin/PROTECTED.txt" 2>/dev/null || echo "cvt5-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- cvt5- present in bin/PROTECTED.txt ----"
  guard_postlaunch "$WS/runs" "cvt5-" "${DESIGN[@]}"; RC=$?
  case $RC in
    0) echo "guard 7: RULE 20 post-launch check PASSED against a run's own ARGS line" ;;
    2) echo "guard 7: UNVERIFIED -- no cvt5 job started inside the window.  Re-check with:"
       echo "         python3 $AGF $WS/runs --name cvt5- --batch-consistency --vary stepsize-groups --vary seed --vary run-name" ;;
    *) echo "!!! guard 7: RULE 20 MISMATCH -- scancel THIS batch only:"
       echo "    squeue -h -u $USER_NAME -o '%i %j' | awk '\$2 ~ /^cvt5-/ {print \$1}' | xargs -r scancel" ;;
  esac
  echo
  echo "ENV AUDIT:"
  echo "    grep -h '^ENV:' $WS/runs/cvt5-*.out | sed 's/ PROBE_DIR=[^ ]*//' | sort | uniq -c    # ONE line, x12"
  echo "    grep -h '^PROBE_TENSOR:' $WS/runs/cvt5-*.out | sed 's/ dir=.*//' | sort | uniq -c    # scalar x3, blockwise x9, tensors=53"
  echo "    grep -H '^VOTE_W' $WS/runs/cvt5-*.out                                               # off x6, w=13.0 x3, w=33.0 x3"
  echo
  echo "SCORE IT WITH, AND ONLY WITH (at 12/12 RUN_DONE):"
  echo "    python3 $REPO/analysis/cVT5_plainhorizon_score.py $WS/runs"
else
  echo
  echo "NOTHING WAS SUBMITTED.  The submitting form is:  bash bin/cVT5_plainhorizon.sh --submit"
  [ "$FAIL" = 1 ] && { echo "(dry run ended with guard failures -- exit 2)"; exit 2; }
fi
exit 0
