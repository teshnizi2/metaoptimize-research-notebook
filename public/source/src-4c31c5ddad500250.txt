# =============================================================================

# =============================================================================
# cgn2 (CORRECTIONS 225) isolated {50 layer4.0.bn2.weight, 53

# CAVEAT (225.4(1)): at 100 epochs NO complement had pinned and every arm was
# still climbing -- ISO's complement beta ended at -8.941 / -8.800 / -8.903,
# falling 0.023-0.029 nats per epoch.  On PlainNet (cpl1, 218.4(2)) the ISO
# rescue was a DELAY: the complement pinned at epoch 60-62 and the level froze.
# Is the GN rescue a rescue, or a delayed collapse that 100 epochs cannot see?
#
# =============================================================================
# THE HORIZON, DERIVED FROM cgn2's MEASURED DESCENT -- NOT CHOSEN ROUND
# =============================================================================
# From cgn2's own 1,500 ISO probe records (--selftest RE-DERIVES every number
# below on any host that can see runs/cgn2; guard 1c2 requires that it RAN):
#   ISO beta[0] (the 59-tensor complement), seed 72 / 73 / 74:
#     terminal @ record 499   -8.940540 / -8.800484 / -8.902525
#     OLS slope, last 50 rec  -0.0050150 / -0.0048443 / -0.0057570 nats/record
#     OLS slope, last 100 rec -0.0056757 / -0.0046762 / -0.0057979
#       (x5 = -0.025 / -0.024 / -0.029 per epoch)
#     tail-linear extrapolation (the 190 method, cVH1's extrapolate()):
#       to the exact clamp -15.0     last-50  341.6515 / 355.9504 / 311.8266
#                                    last-100 313.5222 / 365.1491 / 310.3325
#       to r <= 2 (-15 + ln 2)       last-50  314.0088 / 327.3334 / 287.7466
#                                    last-100 289.0972 / 335.5036 / 286.4224
#     last-quarter median alpha/alpha_floor 643.95 / 610.09 / 602.81 -- FREE
#   ISO beta[1] (the isolated trio) sits on the clamp from epoch 22.0/21.4/22.6.
#
# THE DESCENT IS NOT LINEAR, AND WHICH NON-LINEARITY IS UNKNOWN -- REGISTERED:
#   window slopes of ISO's complement, nats/epoch, epochs 20-40 / 40-60 /
#   60-80 / 80-100: seed 72 -0.069 -0.015 -0.010 -0.028; 73 -0.075 -0.016
#   -0.007 -0.023; 74 -0.075 -0.020 -0.013 -0.029.  It SLOWED and then
#   ACCELERATED.  (k01 and CTL decelerate monotonically instead.)
#   The r <= 2 pin epoch of ISO's complement under four model families:
#     tail-linear, last 50 / 100 records (ep 90-100 / 80-100)   286.4 - 335.5
#     linear, last 250 / 350 records     (ep 50-100 / 30-100)   386.8 - 460.4
#     beta = a + c sqrt(t), epochs >= 30                          816 -   902
#     beta = a + c ln(t),   epochs >= 30                       13,592 - 18,966
#   HOLDOUT (fit on epochs 30-80, predict the mean beta over records 475-499):
#   error linear -0.03..+0.10 nats, sqrt +0.06..+0.20, log +0.15..+0.28 -- every
#   family predicts a SLOWER descent than occurred and linear is the least
#   wrong on ISO.  On k01 and CTL the same holdout favours log.  So the pin
#   epoch is UNCERTAIN BY A FACTOR > 2 WITHIN THE LINEAR FAMILY ALONE, and a
#   GN complement that never pins is a live possibility (registered below as
#   the informative UNRESOLVED-NOT-PINNED stamp).
#
#   1. PIN_BOUND    = 365.1491  the LATEST tail-linear exact-clamp extrapolation
#      PIN_BOUND_R2 = 335.5036  the same to r <= 2.  (On BN/VGG this method
#      bounded the measured pin 3 times: ciso2 ISO/ONE, cvh1 ISO.  On GN it is
#      the EARLY end of the model range, disclosed above.)
#   2. W_POST = 64.2: the longest post-pin window measurable in a 100-epoch
#      batch (cvg1 k01, CORRECTIONS 213 / cVH1 header; ciso1 CTRL 63.2, cvi1

#      FROZEN / DECAYS / IMPROVES-AFTER-PIN at READ_BAR; it is NOT enough to
#      exclude a decay slower than READ_BAR per 64 epochs (disclosed).
#   3. E = ceil(max(PIN_BOUND + W_POST, PIN_BOUND_R2 + W_POST)) = ceil(429.3491)

def gn_tensors():
    out = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "GroupNorm"),
           ("bn1.bias", 64, "GroupNorm")]
    cin = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for bi in range(2):
            stride2 = (li > 1 and bi == 0)
            p = "layer%d.%d." % (li, bi)
            a = cin if bi == 0 else w
            out += [(p + "conv1.weight", w * a * 9, "Conv2d"), (p + "bn1.weight", w, "GroupNorm"),
                    (p + "bn1.bias", w, "GroupNorm"), (p + "conv2.weight", w * w * 9, "Conv2d"),
                    (p + "bn2.weight", w, "GroupNorm"), (p + "bn2.bias", w, "GroupNorm")]
            if stride2:
                out += [(p + "shortcut.0.weight", w * a, "Conv2d"),
                        (p + "shortcut.1.weight", w, "GroupNorm"),
                        (p + "shortcut.1.bias", w, "GroupNorm")]
        cin = w
    out += [("linear.weight", 100 * 512, "Linear"), ("linear.bias", 100, "Linear")]
    return out


TENSOR_NAMES = tuple(n for n, _q, _o in gn_tensors())
TENSOR_NUMELS = tuple(q for _n, q, _o in gn_tensors())


def spec_to_groups(spec, names=TENSOR_NAMES):
    """compose a registered spec into 1-based index groups, without torch."""
    if spec == "scalar":
        return [list(range(1, len(names) + 1))]
    if spec == "layerwise":
        return [[i] for i in range(1, len(names) + 1)]
    if not spec.startswith("sets:") or spec[5:].count("/") != 1:
        raise ValueError("unparsable spec %r" % spec)
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            tok = tok.strip()
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            elif tok:
                if tok not in names:
                    raise ValueError("spec names %r, which is not on the model" % tok)
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out

def classify(med_r):
    if med_r is None or med_r != med_r:
        return "UNMEASURED"
    if med_r <= PIN_RATIO_MAX:
        return "PINNED"
    if med_r >= FREE_RATIO_MIN:
        return "FREE"
    return "NEITHER"


def arm_class(classes):
    c = collections.Counter(classes)
    for k in ("PINNED", "FREE"):
        if c.get(k, 0) >= 2:
            return k
    return "NEITHER"


def beta_series(recs, k=0):
    out = []
    for r in recs:
        b = r.get("beta")
        if isinstance(b, list) and len(b) > k:
            out.append(float(b[k]))
    return out


def group_stats(b):
    """step-size-magnitude statistics of ONE beta trajectory."""
    if not b:
        return None
    n = len(b)
    t = b[-max(1, int(n * TAIL_FRAC)):]
    r = [math.exp(v - BETA_LO) for v in t]
    lr2 = BETA_LO + math.log(PIN_RATIO_MAX)
    ft = next((i for i, v in enumerate(b) if v <= BETA_LO + PIN_EPS), None)
    f2 = next((i for i, v in enumerate(b) if v <= lr2), None)
    return {"n": n, "terminal": b[-1], "med_r": statistics.median(r), "max_r": max(r),
            "occ_exact": sum(1 for v in t if v <= BETA_LO + PIN_EPS) / float(len(t)),
            "first_touch": ft, "first_r2": f2}


def ols_slope(y):
    n = len(y)
    if n < 2:
        return float("nan")
    mx, my = (n - 1) / 2.0, sum(y) / n
    den = sum((i - mx) ** 2 for i in range(n))
    return sum((i - mx) * (v - my) for i, v in enumerate(y)) / den


def extrapolate(b, level, window):
    """epoch at which a linear fit of the last `window` records, from the terminal value, reaches `level`."""
    sl = ols_slope(b[-window:])
    if not (sl < 0) or b[-1] <= level:
        return None, sl
    return (len(b) + (b[-1] - level) / (-sl)) / float(REC_PER_EPOCH), sl


def three_way(delta, lo_tag, mid_tag, hi_tag, bar):
    if delta is None:
        return "UNMEASURED"
    if delta <= -bar:
        return lo_tag
    if delta >= bar:
        return hi_tag
    return mid_tag


def decide_primary(d100, de):
    """THE BRANCH on the primary, a pure function of two in-batch contrasts."""
    rho = de / d100 if abs(d100) > 1e-12 else None
    if d100 < D100_MIN:
        return "NO-RESCUE-AT-100", rho
    if de <= REFUTE_BAR:
        return "RESCUE-COLLAPSES", rho
    if rho is not None and rho >= SURVIVE_FRAC and de >= SUPPORT_BAR:
        return "RESCUE-SURVIVES", rho
    return "RESCUE-ATTENUATES", rho

def parse_args_line(a):
    toks = a.split()[1:] if a.startswith("ARGS:") else a.split()
    d, cnt, cur = {}, collections.Counter(), None
    for t in toks:
        if t.startswith("--"):
            cur = t
            cnt[t] += 1
            d[t] = ""
        elif cur is not None:
            d[cur] = (d[cur] + " " + t).strip()
    return d, sorted(f for f, c in cnt.items() if c > 1)


def parse_env_line(e):
    d = {}
    for t in e.split()[1:]:
        if "=" in t:
            k, v = t.split("=", 1)
            d[k] = v
    return d


def load_outs(runsdir):
    runs, dups = {}, collections.Counter()
    for path in sorted(glob.glob(os.path.join(runsdir, PREFIX + "*.out"))):
        m = RUN_RE.match(os.path.basename(path))
        if not m:
            continue
        rn, arm, seed, jid = m.group(1), m.group(2), int(m.group(3)), int(m.group(4))
        dups[rn] += 1
        d = {"run": rn, "arm": arm, "seed": seed, "job": jid, "path": path,
             "args": None, "env": None, "pt": None, "node": None, "n_args": 0, "n_env": 0, "n_pt": 0,
             "train": {}, "test": {}, "done": False, "traceback": False}
        with open(path, errors="replace") as fh:
            for line in fh:
                if line.startswith("ARGS:"):
                    d["args"] = line.rstrip("\n")
                    d["n_args"] += 1
                elif line.startswith("ENV:"):
                    d["env"] = line.rstrip("\n")
                    d["n_env"] += 1
                elif line.startswith("PROBE_TENSOR:"):
                    d["pt"] = line.rstrip("\n")
                    d["n_pt"] += 1
                elif line.startswith("NODE="):
                    d["node"] = line.split("|")[0].strip()
                elif line.startswith("RUN_DONE"):
                    d["done"] = True
                elif "Traceback (most recent call last)" in line:
                    d["traceback"] = True
                else:
                    em = EPOCH_RE.match(line)
                    if em:
                        e = int(em.group(1))
                        d["train"][e] = float(em.group(2))
                        d["test"][e] = float(em.group(3))
        if rn not in runs or jid > runs[rn]["job"]:
            runs[rn] = d
    return runs, dups

def window_mean(acc, end_epoch):
    v = [acc[e] for e in range(end_epoch - 5, end_epoch) if e in acc]
    return (sum(v) / 5.0) if len(v) == 5 else None


def tail_slope(acc, lo, hi):
    xs = [e for e in range(lo, hi + 1) if e in acc]
    if len(xs) < 3:
        return None
    mx = sum(xs) / float(len(xs))
    my = sum(acc[e] for e in xs) / float(len(xs))
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (acc[x] - my) for x in xs) / den if den else None


def decompose_check(rec, groups):
    zt, mt, za, mp = rec.get("z_tensor"), rec.get("m_tensor"), rec.get("z_agg"), rec.get("mom_pre")
    if not (isinstance(zt, list) and isinstance(mt, list) and za and mp):
        return 0, 0, 0.0
    za = za[0] if isinstance(za[0], list) else za
    mp = mp[0] if isinstance(mp[0], list) else mp
    nch = nfa = 0
    worst = 0.0
    for k, g in enumerate(groups):
        if k >= len(za) or k >= len(mp):
            continue
        for arr, ref in ((zt, za[k]), (mt, mp[k])):
            terms = [arr[i - 1] for i in g]
            d = abs(sum(terms) - ref)
            scale = max(abs(ref), sum(abs(x) for x in terms))   # cvh1's scale-aware error
            err = d / scale if scale > 1e-30 else d
            worst = max(worst, err)
            nch += 1
            if err > DECOMP_TOL:
                nfa += 1
    return nch, nfa, worst

    if missing or notdone:
        print("  NO BRANCH IS TAKEN AND NO NUMBER FROM THIS BATCH IS QUOTABLE.")
        return finish("INCOMPLETE", ["MISSING:%d" % len(missing), "NOT-DONE:%d" % len(notdone)]
                      + ["MISSING-%s" % rn[len(PREFIX):] for rn in missing]
                      + ["NOT-DONE-%s" % rn[len(PREFIX):] for rn in notdone], code=2)

    # ---- [G0] provenance -------------------------------------------------------
    print("\n[G0] PROVENANCE -- every run's OWN ARGS / ENV / PROBE_TENSOR line, probe records, PROVENANCE.txt")
    pok = chk(not extra, "no run outside the registered 12", "extra %s" % extra)
    pok &= chk(all(dups[rn] == 1 for rn in want), "no run has more than one .out (no silent resubmission)",
               "dups %s" % {k: v for k, v in dups.items() if v > 1})
    probes = load_probes(runsdir)
    envs = set()
    for rn in want:
        d = runs[rn]
        a, dupf = parse_args_line(d["args"] or "")
        need = dict(REQ_ARGS)
        need.update({"--stepsize-groups": SPEC[d["arm"]], "--seed": str(d["seed"]), "--run-name": rn})
        bad = ["%s=%r" % (k, a.get(k)) for k, v in need.items() if a.get(k) != v]
        pok &= chk(not bad and not dupf and d["n_args"] == 1, "%s ARGS carries every registered flag, none repeated" % rn,
                   "bad %s repeated %s n_args %d" % (bad, dupf, d["n_args"]))
        ev = parse_env_line(d["env"] or "")
        badenv = ["%s=%r" % (k, ev.get(k)) for k, v in REQ_ENV.items() if ev.get(k) != v]
        pdir_ok = str(ev.get("PROBE_DIR", "")).endswith("/%s/probe_%s" % (BATCH, rn))
        pok &= chk(not badenv and pdir_ok and d["n_env"] == 1, "%s ENV line is the registered one" % rn,
                   "bad %s PROBE_DIR %s" % (badenv, ev.get("PROBE_DIR")))
        envs.add(re.sub(r" PROBE_DIR=\S*", "", d["env"] or ""))
        pm = PT_RE.match(d["pt"] or "")
        pok &= chk(bool(pm) and d["n_pt"] == 1 and pm.group(1) == str(PROBE_EVERY) and pm.group(2) == PT_TYPE[d["arm"]]
                   and pm.group(3) == str(N_TENSORS),
                   "%s PROBE_TENSOR: on every=%d type=%s tensors=%d" % (rn, PROBE_EVERY, PT_TYPE[d["arm"]], N_TENSORS),
                   (d["pt"] or "")[:100])
        pr = probes.get(rn)
        if pr is None:
            pok &= chk(False, "%s has probe records" % rn)
            continue
        pok &= chk(len(pr["recs"]) == N_RECORDS, "%s has exactly %d probe records" % (rn, N_RECORDS),
                   "got %d" % len(pr["recs"]))
        want_nb = [N_PARAMS] if d["arm"] == "k01" else (list(TENSOR_NUMELS) if d["arm"] == "kL" else list(PARAMS[d["arm"]]))
        pok &= chk(bool(pr["block"]) and list(pr["block"].get("n_b") or []) == want_nb,
                   "%s block_sizes n_b is the registered arity" % rn)
    pok &= chk(len(envs) == 1, "ONE distinct ENV line across the 12 runs (PROBE_DIR masked)", "%d distinct" % len(envs))
    tot = fail = 0
    worst = 0.0
    for rn in want:
        if runs[rn]["arm"] in M2_ARMS and rn in probes:
            g = spec_to_groups(SPEC[runs[rn]["arm"]])
            for rec in probes[rn]["recs"]:
                c, f, w = decompose_check(rec, g)
                tot, fail, worst = tot + c, fail + f, max(worst, w)
    if tot:
        pok &= chk(fail <= DECOMP_FAIL_MAX * tot, "group sums of z_tensor / m_tensor reproduce z_agg / mom_pre",
                   "%d/%d fail, worst %.3e" % (fail, tot, worst))
    else:
        pok &= chk(False, "decomposition -- no blockwise probe record carries z_tensor")
    pp = resolve_provenance(runsdir)
    if pp is None:
        pok &= chk(False, "PROVENANCE.txt present")
    else:
        prov = read_kv(pp)
        me = file_sha(THIS)
        pok &= chk(prov.get("MODE") == "submit", "PROVENANCE MODE submit", str(prov.get("MODE")))
        pok &= chk(prov.get("BUILD_NETWORK_SHA256") == POST_SHA,
                   "PROVENANCE build_network.py == POST_SHA %s... (the cgn1 isolated tree)" % POST_SHA[:12],
                   str(prov.get("BUILD_NETWORK_SHA256"))[:16])
        pok &= chk(prov.get("SCORER_SHA256") == me, "PROVENANCE SCORER_SHA256 == this scorer %s..." % me[:12],
                   str(prov.get("SCORER_SHA256"))[:16])
    if not pok:
        return finish("UNRESOLVED-PROVENANCE", ["PROVENANCE-FAIL"], code=1)

    # ---- [L] levels, from the raw .out ----------------------------------------
    print("\n[L] LEVELS -- plateau5 TEST and TRAIN at %d and %d epochs, per seed and per arm" % (CTRL_EPOCHS, EPOCHS))
    V = {}
    for arm in ARMS:
        for s in SEEDS:
            d = runs["%s%s-s%d" % (PREFIX, arm, s)]
            V[(arm, s)] = {"P100": window_mean(d["test"], CTRL_EPOCHS), "PE": window_mean(d["test"], EPOCHS),
                           "T100": window_mean(d["train"], CTRL_EPOCHS), "TE": window_mean(d["train"], EPOCHS)}
    A = {}
    for arm in ARMS:
        A[arm] = {k: statistics.mean(V[(arm, s)][k] for s in SEEDS) for k in ("P100", "PE", "T100", "TE")}
        A[arm]["R100"] = max(V[(arm, s)]["P100"] for s in SEEDS) - min(V[(arm, s)]["P100"] for s in SEEDS)
        A[arm]["RE"] = max(V[(arm, s)]["PE"] for s in SEEDS) - min(V[(arm, s)]["PE"] for s in SEEDS)
        sl = [tail_slope(runs["%s%s-s%d" % (PREFIX, arm, s)]["test"], EPOCHS - 20, EPOCHS - 1) for s in SEEDS]
        print("  %-4s TEST  @%d %.4f (%s; range %.3f)   @%d %.4f (%s; range %.3f)"
              % (arm, CTRL_EPOCHS, A[arm]["P100"], " ".join("%.3f" % V[(arm, s)]["P100"] for s in SEEDS), A[arm]["R100"],
                 EPOCHS, A[arm]["PE"], " ".join("%.3f" % V[(arm, s)]["PE"] for s in SEEDS), A[arm]["RE"]))
        print("       TRAIN @%d %.4f (%s)   @%d %.4f (%s)"
              % (CTRL_EPOCHS, A[arm]["T100"], " ".join("%.3f" % V[(arm, s)]["T100"] for s in SEEDS),
                 EPOCHS, A[arm]["TE"], " ".join("%.3f" % V[(arm, s)]["TE"] for s in SEEDS)))
        print("       TEST OLS slope over epochs %d-%d (descriptive): %s pp/ep"
              % (EPOCHS - 20, EPOCHS - 1, " / ".join(fmt(x, 4) for x in sl)))
    ss = df = 0
    for arm in ARMS:
        for k in ("P100", "PE"):
            v = [V[(arm, s)][k] for s in SEEDS]
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
    sigma_in = math.sqrt(ss / df)
    sigma_used = max(SIGMA_FROZEN, sigma_in)
    se = sigma_used * math.sqrt(2.0 / 3.0)
    read_bar = 2.0 * se
    print("  SIGMA_FROZEN %.6f (literal)   SIGMA_INBATCH %.6f (df %d, this batch's own seeds)" % (SIGMA_FROZEN, sigma_in, df))
    print("  SIGMA_USED %.6f   SE_USED %.6f   READ_BAR %.6f" % (sigma_used, se, read_bar))

    # ---- [A] anchors / [D] divergence -----------------------------------------
    print("\n[A] ANCHORS -- the in-run 100-epoch floor and ceiling inside their FROZEN GN bands")
    aok = True
    for arm in ("k01", "kL"):
        lo, hi = ANCHOR[arm] - BAND_HALF, ANCHOR[arm] + BAND_HALF
        aok &= chk(lo <= A[arm]["P100"] <= hi, "%s P100 inside [%.4f, %.4f]" % (arm, lo, hi), "%.4f" % A[arm]["P100"])
    if not aok:
        return finish("UNRESOLVED-ANCHOR", ["ANCHOR-FAIL"])
    print("\n[D] DIVERGENCE -- every arm's seed range <= DEAD_BAR %.1f at both horizons" % DEAD_BAR)
    dok = True
    for arm in ARMS:
        dok &= chk(A[arm]["R100"] <= DEAD_BAR and A[arm]["RE"] <= DEAD_BAR, "%s seed range" % arm,
                   "@%d %.4f  @%d %.4f" % (CTRL_EPOCHS, A[arm]["R100"], EPOCHS, A[arm]["RE"]))
    if not dok:
        return finish("UNRESOLVED-DIVERGED", ["BIMODAL-ARM"])

    # ---- [B] the primary and the branch ----------------------------------------
    print("\n[B] PRIMARY -- all contrasts IN BATCH, both horizons IN RUN")
    D100 = A["ISO"]["P100"] - A["k01"]["P100"]
    DE = A["ISO"]["PE"] - A["k01"]["PE"]
    branch, RHO = decide_primary(D100, DE)
    DT100 = A["ISO"]["T100"] - A["k01"]["T100"]
    DTE = A["ISO"]["TE"] - A["k01"]["TE"]
    branch_tr, RHO_TR = decide_primary(DT100, DTE)
    print("  D_ISO @%d = %+.4f pp (%+.2f SE)   D_ISO @%d = %+.4f pp (%+.2f SE)" % (CTRL_EPOCHS, D100, D100 / se, EPOCHS, DE, DE / se))
    print("  RHO = %s   (SURVIVE needs RHO >= %.2f AND D_ISO@%d >= %.1f; COLLAPSE D_ISO@%d <= %.1f)"
          % (fmt(RHO), SURVIVE_FRAC, EPOCHS, SUPPORT_BAR, EPOCHS, REFUTE_BAR))
    print("  TRAIN beside it: D_ISO(TRAIN) @%d %+.4f  @%d %+.4f  RHO(TRAIN) %s -> the same map gives %s"
          % (CTRL_EPOCHS, DT100, EPOCHS, DTE, fmt(RHO_TR), branch_tr))
    for s in SEEDS:
        d1 = V[("ISO", s)]["P100"] - V[("k01", s)]["P100"]
        d2 = V[("ISO", s)]["PE"] - V[("k01", s)]["PE"]
        print("    seed %d paired: D@%d %+.4f  D@%d %+.4f  rho %s (descriptive)"
              % (s, CTRL_EPOCHS, d1, EPOCHS, d2, fmt(d2 / d1 if abs(d1) > 1e-12 else None)))
    stamps = ["RHO:%s" % fmt(RHO)]
    if branch == "RESCUE-ATTENUATES":
        stamps.append("ATTENUATE-MILD" if (RHO is not None and RHO >= ATTEN_MILD_FRAC) else "ATTENUATE-SEVERE")
    stamps.append("TRAIN-AGREES" if branch_tr == branch else "TRAIN-DISAGREES")
    print("  BRANCH (decided by the primary alone): %s" % branch)

    # ---- [P] the pin state -- AFTER the primary, as stamps ----------------------
    print("\n[P] PIN STATE -- step-size MAGNITUDE over the last quarter, r = alpha/alpha_floor (cvh1's gate)")
    iso = {}
    for s in SEEDS:
        g = group_stats(beta_series(probes["%sISO-s%d" % (PREFIX, s)]["recs"], 0))
        iso[s] = g
        print("  ISO-s%d complement: median r %.4f  max r %.3f  terminal %.4f  first r<=%.0f epoch %s  first touch %s  "
              "exact-clamp dwell %.4f -> %s"
              % (s, g["med_r"], g["max_r"], g["terminal"], PIN_RATIO_MAX,
                 "%.1f" % (g["first_r2"] / float(REC_PER_EPOCH)) if g["first_r2"] is not None else "never",
                 "%.1f" % (g["first_touch"] / float(REC_PER_EPOCH)) if g["first_touch"] is not None else "never",
                 g["occ_exact"], classify(g["med_r"])))
    cls = arm_class([classify(iso[s]["med_r"]) for s in SEEDS])

    if cls == "PINNED":

                if classify(iso[s]["med_r"]) == "PINNED" and iso[s]["first_r2"] is not None]
        latest = max(pins)

        stamps.append("ISO-PIN-EPOCH:%.1f" % statistics.mean(pins))
        stamps.append("POST-PIN-WINDOW:%.1f" % window)

        stamps.append("PIN-INSIDE-BOUND" if latest <= PIN_BOUND_R2 else "PIN-AFTER-BOUND")
        print("  pin epochs (r<=2) %s -> latest %.1f (registered bound %.4f); post-pin window %.1f epochs (W_POST %.1f)"

        if window >= POST_MIN_WINDOW:
            e0 = int(math.ceil(latest)) + 5
            post = statistics.mean(statistics.mean(runs["%sISO-s%d" % (PREFIX, s)]["test"][e] for e in range(e0, e0 + 5))
                                   for s in SEEDS)

            stamps.append(three_way(d_pp, "DECAYS-AFTER-PIN", "FROZEN-AT-PIN", "IMPROVES-AFTER-PIN", read_bar))

    else:
        stamps.append("UNRESOLVED-NOT-PINNED")
        stamps.append("POST-PIN-UNMEASURED")
        print("  the complement did NOT pin in %d epochs (INFORMATIVE; the branch stands on the primary)" % EPOCHS)

                            "ONE-SHARE-FALLS", "ONE-SHARE-HOLDS", "ONE-SHARE-RISES", ONE_SHARE_BAR))

                         for s in SEEDS])
    stamps.append("ONE-COMPLEMENT-" + one_cls)
    h100 = A["ISO"]["P100"] - A["ONE"]["P100"]
    hE = A["ISO"]["PE"] - A["ONE"]["PE"]
    print("  D_HEAD3 = ISO - ONE: @%d %+.4f  @%d %+.4f" % (CTRL_EPOCHS, h100, EPOCHS, hE))
    stamps.append(three_way(hE - h100, "HEAD3-NARROWS", "HEAD3-HOLDS", "HEAD3-WIDENS", read_bar))
    for arm, k in (("k01", 0), ("ONE", 0), ("ONE", 1), ("ISO", 1)):
        gs = [group_stats(beta_series(probes["%s%s-s%d" % (PREFIX, arm, s)]["recs"], k)) for s in SEEDS]
        print("  %s beta[%d] median r %s  terminal %s  first r<=2 epoch %s (descriptive)"

                 " / ".join("%.1f" % (g["first_r2"] / 5.0) if g["first_r2"] is not None else "never" for g in gs)))

    print("\n[S] STAMPS -- none of these moves the branch")
    kl_free = []
    for s in SEEDS:

    print("  ISO TRAIN @%d - @%d = %+.4f pp" % (EPOCHS, CTRL_EPOCHS, d))
    stamps.append("ISO-TRAIN-FROZEN" if abs(d) < read_bar else "ISO-TRAIN-MOVES")
    stamps.append("SIGMA-FROZEN-DOMINATES" if sigma_used == SIGMA_FROZEN else "SIGMA-INBATCH-DOMINATES")
    d = D100 - CGN2_D_ISO_100
    print("  BETWEEN-BATCH, NON-GATING: D_ISO@%d here %.4f vs cgn2's %.4f, delta %+.4f (%+.2f SE)"
          % (CTRL_EPOCHS, D100, CGN2_D_ISO_100, d, d / se))
    stamps.append("REPLICATES-CGN2" if abs(d) < read_bar else "DIFFERS-FROM-CGN2")
    stamps += ["MAGNITUDE-NOT-SEPARATED", "NO-IN-BATCH-CTL", "GN32-ALSO-CHANGES-PER-CHANNEL-INVARIANCE", "ONE-CELL-ONLY"]

    print("\n  REGISTERED POINT PREDICTIONS (descriptive, non-gating), account being tested %s:" % TESTED_ACCOUNT)
    for acc, pred in ACCOUNTS.items():
        cells = []
        for arm in ARMS:
            lo, hi = pred[arm]
            cells.append("%s %s" % (arm, "IN" if lo <= A[arm]["PE"] <= hi else "OUT"))
        lo, hi = pred["RHO"]
        cells.append("RHO %s" % ("IN" if RHO is not None and lo <= RHO <= hi else "OUT"))
        print("    %-13s %s" % (acc, "  ".join(cells)))
    print("    ISO complement pin (r<=2) predicted <= %.1f (tail-linear); long-window linear 386.8-460.4; sqrt/log never"
          % PIN_BOUND_R2)
    return finish(branch, stamps)

CELLKEYS = ["network", "dataset", "granularity", "base", "meta", "meta_stepsize", "alpha0", "gamma",
            "augment", "beta_clip", "batch_size", "epochs_requested", "hier", "lam", "eta_ratio"]
M2N = re.compile(r"^\[\d+,\d+\]$")


def _rows(path):
    if not path or not os.path.isfile(path):
        return None
    with open(path) as fh:
        return [r for r in csv.DictReader(fh) if not str(r.get("run") or "").startswith(PREFIX)]


def _usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
            and (r.get("plateau5") or "").strip() != "")


def _std(r, ep):
    return (r.get("epochs_requested") == str(ep) and r.get("augment") == "1" and r.get("beta_clip") == CLIP_C
            and r.get("meta_stepsize") == MS and r.get("alpha0") == A0 and r.get("batch_size") == "100"
            and r.get("base") == "SGDm" and r.get("meta") == "Lion" and r.get("gamma") == "1"
            and not (r.get("hier") or "").strip())


def _std_anygran100(r):
    return (r.get("epochs_requested") == "100" and r.get("augment") == "1" and r.get("beta_clip") == CLIP_C
            and r.get("meta_stepsize") == MS and r.get("alpha0") == A0 and r.get("batch_size") == "100")


def _pooled(rows, filt, key):
    cells = collections.defaultdict(list)
    for r in rows:
        if _usable(r) and filt(r):
            cells[key(r)].append(float(r["plateau5"]))
    ss, df = 0.0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = sum(v) / len(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
    return (math.sqrt(ss / df) if df else float("nan")), df


def noise_variants(rows):
    """the seven registered definitions, re-derived.  SELFTEST ONLY -- score() never calls this."""
    ck = lambda r: tuple(r.get(k, "") for k in CELLKEYS)
    bg = lambda r: (re.split(r"[-_]", r["run"])[0], r.get("granularity"), r.get("network"), r.get("epochs_requested"))
    GN = lambda r: r.get("network") == NET and r.get("dataset") == DSET
    N = lambda r: r.get("network") == "ResNet18_c100" and r.get("dataset") == "CIFAR100"
    VG = lambda r: r.get("network") == "VGG11_bn_c100"
    return {
        "GN_CELL_100": _pooled(rows, lambda r: GN(r) and _std_anygran100(r), ck),
        "GN_BATCH_100": _pooled(rows, lambda r: GN(r) and _std(r, 100), bg),
        "R18ALL_100": _pooled(rows, lambda r: N(r) and _std_anygran100(r), ck),
        "R250": _pooled(rows, lambda r: N(r) and _std(r, 250), bg),
        "R772": _pooled(rows, lambda r: N(r) and _std(r, 772), bg),
        "R100_M2_NARROW": _pooled(rows, lambda r: N(r) and _std(r, 100) and r.get("epochs_done") == "100"
                                  and bool(M2N.match(r.get("granularity") or "")), bg),
        "VGG_328": _pooled(rows, lambda r: VG(r) and _std(r, 328), bg),
    }


def anchor_levels(rows):
    out = {}
    for arm, gran in (("k01", "scalar"), ("kL", "layerwise")):
        v = [float(r["plateau5"]) for r in rows if _usable(r) and _std_anygran100(r)
             and r.get("network") == NET and r.get("dataset") == DSET and r.get("granularity") == gran]
        out[arm] = (statistics.mean(v), len(v)) if v else (None, 0)
    return out


def _final_of(text):
    for line in text.splitlines():
        if line.startswith("FINAL: "):
            return line[len("FINAL: "):]
    return None


def _line_of(text, key):
    for line in text.splitlines():
        if key in line:
            return line.strip()
    return None


def _run_scorer(runsdir, csv_path=None):
    cmd = [sys.executable, THIS, runsdir]
    if csv_path is not None:
        cmd += ["--csv", csv_path]
    p = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, universal_newlines=True, timeout=1200)
    return p.returncode, p.stdout


def _stress_csv(src_rows, header, dst, gn_vals, own_vals):
    """invented rows: 3 at the GN standard cell (moves every live GN sigma) and 12
    named cgn3-* at cgn3's OWN cells (ResNet18_gn_c100, the four specs, 430 epochs)."""
    tmpl = next(r for r in src_rows if r.get("network") == NET and r.get("granularity") == "scalar" and _usable(r))
    new = []
    for i, v in enumerate(gn_vals):
        r = dict(tmpl)
        r.update(run="zstress-k01-s%d" % (90 + i), job_id=str(9990000 + i), seed=str(90 + i),
                 plateau5="%.4f" % v, plateau="", best_test="", final_test="", dup_group="")
        new.append(r)
    i = 0
    for arm in ARMS:
        for s in SEEDS:
            r = dict(tmpl)
            r.update(run="%s%s-s%d" % (PREFIX, arm, s), job_id=str(9991000 + i), seed=str(s),
                     granularity=SPEC[arm], epochs_requested=str(EPOCHS), epochs_done=str(EPOCHS),
                     plateau5="%.4f" % own_vals[i % len(own_vals)], plateau="", best_test="", final_test="",
                     dup_group="")
            new.append(r)
            i += 1
    with open(dst, "w", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=header, extrasaction="ignore")
        w.writeheader()
        for r in src_rows + new:
            w.writerow(r)


def _cgn2_spec():
    if not os.path.isfile(CGN2_SCORER):
        return None
    m = re.search(r"^SPEC = (\{.*?^\})", open(CGN2_SCORER).read(), re.S | re.M)
    return ast.literal_eval(m.group(1)) if m else None

        fin = _final_of(out)
        finals[name] = (fin, out)
        ok = (fin is not None and fin.split(" | ")[0] == branch and all(t in fin.split(" | ") for t in toks)
              and rc == want_rc.get(branch, 0))
        chk(ok, "SYNTHETIC %s -> %s%s" % (name, branch, (" with " + ",".join(toks)) if toks else ""),
            "rc %d FINAL: %s" % (rc, fin))
        if not ok:
            print("\n".join("      | " + x for x in out.splitlines()[-30:]))
    if finals.get("survive_freeze", (None,))[0]:
        recs = [json.loads(x) for x in open(os.path.join(tmp, "survive_freeze", BATCH, "probe_%sISO-s81" % PREFIX, "probe.jsonl"))]
        st = group_stats(beta_series(recs, 0))
        chk(st["occ_exact"] < 0.50 and classify(st["med_r"]) == "PINNED",
            "the chatter scenario reproduces the ciso2 trigger (dwell %.4f < 0.50) and the magnitude gate PINS it (median r %.4f)"
            % (st["occ_exact"], st["med_r"]))

    print("\n[S6] FROZEN BARS (O2): score() is INVARIANT to the corpus")
    src = inspect.getsource(score)
    chk(not any(t in src for t in ("noise_variants(", "anchor_levels(", "_pooled(", "_rows(")),
        "score()'s source calls no corpus-derived sigma, band or level")
    base_rows = _rows(csv_path)
    if base_rows is None or not any(r.get("network") == NET and r.get("granularity") == "scalar" and _usable(r) for r in base_rows):
        skip("O2 stress corpora", "no GN template row in %s" % csv_path)
    else:
        header = list(csv.DictReader(open(csv_path)).fieldnames)
        sA, sB = os.path.join(tmp, "stress_A.csv"), os.path.join(tmp, "stress_B.csv")
        _stress_csv(base_rows, header, sA, (5.0, 30.0, 20.0), (11.0, 88.0, 3.0, 64.0))
        _stress_csv(base_rows, header, sB, (14.1, 14.2, 40.0), (14.0, 14.0, 99.0))
        liveA = noise_variants(_rows(sA))
        mx = max(v for v, _ in liveA.values() if v == v)
        rb_live = 2.0 * mx * math.sqrt(2.0 / 3.0)
        chk(mx > SIGMA_FROZEN and rb_live > 1.65,
            "the stress corpus is NOT VACUOUS: a live-max sigma would be %.4f -> READ_BAR %.4f > the near_bar margin 1.65,"
            " so a live-reading scorer would stamp CEIL-TRACKS where this one stamps CEIL-ABOVE" % (mx, rb_live))
        for name in ("near_bar", "survive_freeze", "collapse"):
            d = os.path.join(tmp, name)
            base_fin, base_out = finals[name]
            seen = [("one-arg default corpus", base_fin, _line_of(base_out, "SE_USED"))]
            for lab, cp in (("given corpus", csv_path), ("stress A", sA), ("stress B", sB),
                            ("no corpus", os.path.join(tmp, "absent.csv"))):
                rc, out = _run_scorer(d, cp)
                seen.append((lab, _final_of(out), _line_of(out, "SE_USED")))
            chk(len(set(x[1] for x in seen)) == 1 and len(set(x[2] for x in seen)) == 1 and seen[0][1] is not None,
                "O2 FINAL identical across %d corpora on %s" % (len(seen), name),
                "; ".join("%s: %s" % (x[0], x[1]) for x in seen) if len(set(x[1] for x in seen)) != 1 else seen[0][1])

    print("\n[S7] RULE 21 PREMISE: no cgn3 run may exist yet")
    n_out = len(glob.glob(os.path.join(rd, PREFIX + "*.out")))
    chk(n_out == 0, "no cgn3-*.out under %s" % rd, "%d found" % n_out)
    if csv_path and os.path.isfile(csv_path):
        chk(not any(str(r.get("run") or "").startswith(PREFIX) for r in csv.DictReader(open(csv_path))), "no cgn3- row in the CSV")
    if not keep:
        shutil.rmtree(tmp, ignore_errors=True)
    else:
        print("  synthetic runs KEPT at %s" % tmp)
    return _st_end()


def _st_end():
    print("\n" + "=" * 78)
    print("SELFTEST: %d PASS  %d FAIL  %d SKIP" % (_N_PASS, _N_FAIL, _N_SKIP))
    print("=" * 78)
    return 1 if _N_FAIL else 0


def main():
    ap = argparse.ArgumentParser(description="registered scorer for cgn3 (the GN isolation rescue at a long horizon)")
    ap.add_argument("runsdir", nargs="?", default=None,
                    help="the directory holding cgn3-*.out and cgn3/ (the documented single argument)")
    ap.add_argument("--runsdir", dest="runsdir_kw", default=None, help="same, for --selftest")
    ap.add_argument("--csv", default=DEFAULT_CSV,
                    help="defaults to the repo's results/all_runs.csv; score() reads it for a row-count NOTE only")
    ap.add_argument("--manifest", default=None, help="OPTIONAL: defaults to <runsdir>/cgn3/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--keep", action="store_true", help="--selftest: keep the synthetic runs")
    a = ap.parse_args()
    rd = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, rd or ".", keep=a.keep)
    if not rd:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    return score(rd, a.csv, a.manifest)


if __name__ == "__main__":
    sys.exit(main())
