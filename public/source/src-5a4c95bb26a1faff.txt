#!/bin/bash
# =============================================================================
# cJ1_count_balance_ladder.sh -- `cbl1`, THE COUNT LADDER AND THE BALANCE
#                                LADDER, SUBMITTED AS ONE BATCH.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.
# Nothing here touches /cluster/account_a and nothing is submitted to cluster_a.
# cluster_a is holding in489g1 jobs that this script must not go near, and cluster_b
# is holding lsm1 jobs that it must not disturb: every guard and every scancel
# line below is filtered on the `cbl1-` prefix alone.
#
# REGISTERED SCORER: analysis/cJ1_cbl1_score.py, git-committed BEFORE this batch
# is submitted (STANDING RULE 21).  Every bar quoted here is frozen there, and
# that file's --selftest re-derives every archived number from
# results/all_runs.csv, re-derives every partition from its own spec string, and
# asserts each constant against THIS FILE'S own text.
#
# NO CODE PATCH.  Optimizers/HF.py::polish_the_stepsize_groups already parses an
# explicit int list given as the STRING "[a,b,...]" into CONTIGUOUS groups in
# named_parameters() order, and init_meta routes any unrecognised string to
# `blockwise`.  GUARD 4 below re-proves that on the LIVE model at submission
# time and writes PARTITION-MANIFEST.txt, which the scorer REQUIRES.  The shared
# tree is not modified by this batch in any way.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# `hb1` refuted the head/backbone account (CAP_HEAD 0.0412, SPEC -0.1753,
# REFUTED-SMALL) and left two rival readings of the same six numbers:
#   (a) CAPTURE rises with the group COUNT m  (0, 0.2165, 0.6486, 1.0 at
#       m = 1, 2, 6, 62), and
#   (b) at FIXED m = 2, BALANCE matters  ([31,31] 0.2165 vs [60,2] 0.0412 vs
#       [2,60] -0.0028).
# This batch varies ONE at a time, inside a SINGLE submission, so BATCH -- the
# unit of replication (F(62,85)=5.47, p 6.9e-13; SEED is null, F 1.21, p 0.213)
# -- is held constant and every contrast is WITHIN this batch.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  11 arms x 3 seeds = 33 jobs, ONE submission.
#
#   A. COUNT LADDER, balanced partitions of the 62 parameter tensors:
#      c01  scalar                    m= 1   B=1.00000   anchor
#      c02  [31,31]                   m= 2   B=1.00000
#      c04  [16,16,15,15]             m= 4   B=0.99962
#      c08  [8,8,8,8,8,8,7,7]         m= 8   B=0.99923
#      c16  [4 x14,3,3]               m=16   B=0.99859
#      c32  [2 x30,1,1]               m=32   B=0.99729
#      c62  layerwise                 m=62   B=1.00000   anchor
#      Geometric in m: evenly spaced in log2 m, which is the spacing a log-m
#      law would have to be tested at.  m=1 and m=62 are run IN BATCH even
#      though archived values exist -- that is what makes the ladder
#      within-batch rather than a splice.
#
#   B. BALANCE LADDER, fixed m = 2, the cut moved along the tensor order:
#      c02  [31,31]  B=1.00000   (shared with the count ladder)
#      b45  [45,17]  B=0.84742
#      b17  [17,45]  B=0.84742   MIRROR CONTROL of b45
#      b55  [55,7]   B=0.50861
#      b60  [60,2]   B=0.20559   (hb1's `head` arm, re-run in batch)
#
#   B(g) = H(p)/log(m) with p_i = |g_i|/62 is the IMBALANCE STATISTIC, chosen
#   and frozen in the scorer BEFORE any run exists.  b17 exists because balance
#   and cut POSITION are otherwise perfectly collinear at m=2 with contiguous
#   groups; b17 has the SAME balance as b45 and the OPPOSITE cut position, so
#   the registered statistic can fail in batch.
#
#   CIFAR-100 / ResNet18_c100 / SGDm base / Lion meta / ms 1e-3 / alpha0 1e-6 /
#   100 ep / batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / HIER
#   unset / PROBE=0.  Matched cell-for-cell to hb1 and to the archived
#   CIFAR-100 rows, so the two batches compose.
#
#   CAPTURE(arm) = (M(arm) - M(c01)) / (M(c62) - M(c01))  on plateau5, both
#   anchors measured IN BATCH.
#
# PRE-REGISTERED REFUTATION CONDITIONS, frozen in the scorer.  "Capture rises
# with m" is nearly unfalsifiable on its own -- CAPTURE is 0 at m=1 and 1 at
# m=62 BY CONSTRUCTION -- so the count law is three claims with three deaths:
#
#   R1 NONMONOTONE        max adjacent DROP in CAPTURE along the count ladder
#                         > 0.05 (= 2.33 pp = 7.1 SE).  A reversal, not a wobble.
#   R2 BALANCE-DOMINATES  BAL_RANGE >= CNT_RANGE.  If moving the cut at fixed
#                         m=2 buys as much as multiplying m by 31, COUNT is not
#                         the operative variable.  On record from hb1: expected
#                         ~0.22 vs ~0.78, so R2 needs the ratio 3.5x the other
#                         way.
#   R3 FLAT               CAPTURE(c32) - CAPTURE(c02) < 0.10.  Monotone-but-flat
#                         is a REFUTATION, not a confirmation.
#   R4 ANTI-BALANCE       an arm with B < 0.60 beats c02 by > 0.05: the balanced
#                         split is not the m=2 maximum and the balance account
#                         is wrong in SIGN.
#   R5 MIRROR-BREAK       |CAPTURE(b45) - CAPTURE(b17)| > 0.10 at identical
#                         B: normalised entropy is disqualified as the
#                         explanation and is NOT swapped for a better fit.
#
#   COUNT-LAW-HOLDS requires monotone AND span >= 0.10 AND BAL_RANGE < 0.5 x
#   CNT_RANGE.  Anything else in that region is UNRESOLVED, which is a valid
#   deliverable.
#
# MAY NOT CLAIM: anything about CIFAR-10, Tiny-ImageNet, ImageNet-489,
# ResNet-50, alpha0 1e-3, any meta-stepsize but 1e-3, any optimiser pair but
# SGDm+Lion, any hierarchical or shrinkage operator (HIER is unset in every
# arm), or -- specifically -- whether the optimum in m MOVES WITH CLASS COUNT.
# That observation rests on cross-batch CIFAR-10 pools and one CIFAR-100 batch
# cannot settle it.  No optimum in m is named: CAPTURE is pinned to 1 at m=62.
#
# WALLTIME, MEASURED not guessed.  hb1 ran this exact cell on this account: the
# longest of its 18 jobs was 00:46:39 (sacct).  02:30:00 is 3.2x the worst
# observed, matches hb1's own request, and stays inside gpu-short's 4 h ceiling
# so no partition in the list is excluded.
#
# USAGE   bash bin/cJ1_count_balance_ladder.sh            # dry run, submits nothing
#         bash bin/cJ1_count_balance_ladder.sh --submit
# =============================================================================
set -u
SUBMIT=0; [ "${1:-}" = "--submit" ] && SUBMIT=1
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

USER_NAME="${USER:-$(whoami)}"
# NOTE: bin/_lib_guards.sh re-assigns WS at source time from METAOPT_WS with a
# /cluster/account_a default, which would clobber this value and point
# guard_postlaunch at the wrong account's directory (it did exactly that to tn1
# and to hb1).  Exporting METAOPT_WS here is the documented work-around; the
# library is NOT patched while jobs are queued.
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cbl1
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP="-15:-2.3026"; MST=1e-3; ALPHA0=1e-6; EPOCHS=100; BATCH=100
GAMMA=1; AUG=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="0 1 2"
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
WALL=02:30:00

# ---- the registered partitions, written out ONCE and reused everywhere ------
G_C04='[16,16,15,15]'
G_C08='[8,8,8,8,8,8,7,7]'
G_C16='[4,4,4,4,4,4,4,4,4,4,4,4,4,4,3,3]'
G_C32='[2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,2,1,1]'
G_C02='[31,31]'
G_B45='[45,17]'
G_B17='[17,45]'
G_B55='[55,7]'
G_B60='[60,2]'
ARMS="c01 c02 c04 c08 c16 c32 c62 b45 b17 b55 b60"

echo "=============================================================="
echo " cbl1 -- CIFAR-100 COUNT ladder + BALANCE ladder, one batch"
echo " $DSET / $NET  box $CLIP  ms $MST  alpha0 $ALPHA0  AUGMENT=$AUG"
echo " 11 arms x 3 seeds = 33 jobs, one submission"
echo " scorer: analysis/cJ1_cbl1_score.py (registered before any run exists)"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
CSVP="$REPO/results/all_runs.csv"
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- guard 1: the registered scorer exists, is committed, and selftests ------
SC="$REPO/analysis/cJ1_cbl1_score.py"
[ -f "$SC" ] || guard_fail "guard 1a: scorer missing"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  DIRTY=$(git -C "$REPO" status --porcelain -- analysis/cJ1_cbl1_score.py bin/cJ1_count_balance_ladder.sh 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: scorer AND this script are committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires the scorer committed BEFORE any run exists: $DIRTY"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 was verified"
  echo "          on the host that committed the scorer.  Not a pass, not a fail."
fi
python3 "$SC" --selftest >/tmp/cbl1_selftest.$$ 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES" \
  || { guard_fail "guard 1c: scorer selftest FAILED"; tail -25 /tmp/cbl1_selftest.$$; }

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ---------
grep -q '^cbl1-' "$CSVP" && guard_fail "guard 2: cbl1-* already in the CSV" \
                         || echo "guard 2: no cbl1-* name collision in the corpus"
if [ -d "$WS/runs" ]; then
  NOUT=$(ls "$WS/runs"/cbl1-*.out 2>/dev/null | wc -l | tr -d ' ')
  [ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cbl1-*.out on disk" \
                       || guard_fail "guard 2b: $NOUT cbl1-*.out files already exist"
fi

# ---- guard 3: the CIFAR-100 data is actually staged on THIS account ---------
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3: cifar-100-python present under $CIF/data" \
  || guard_fail "guard 3: CIFAR-100 is not staged at $CIF/data/cifar-100-python"

# ---- guard 4: EVERY PARTITION PROVEN ON THE LIVE MODEL, AND WRITTEN DOWN ----
# No assumption about tensor ordering or count: named_parameters() is read from
# the model this batch will actually train, polish_the_stepsize_groups is called
# on it, and the balanced size vectors are RECOMPUTED from the measured tensor
# count rather than taken from this script's header.
mkdir -p "$SAVE"
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"
python3 - "$CIF" "$MANIFEST" "$G_C02" "$G_C04" "$G_C08" "$G_C16" "$G_C32" \
                             "$G_B45" "$G_B17" "$G_B55" "$G_B60" <<'PYEOF' \
  || guard_fail "guard 4: live partition check FAILED"
import sys, os, math
cif, out = sys.argv[1], sys.argv[2]
specs = sys.argv[3:]
sys.path.insert(0, cif)
os.chdir(cif)
from build_network import build_network
from Optimizers.HF import HF
net = build_network("ResNet18_c100", "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
tot = sum(int(p.numel()) for _, p in net.named_parameters())
T = len(nps)
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot]
print("guard 4: live ResNet18_c100 has %d parameter tensors, %d parameters"
      % (T, tot))

def balanced(m, total=T):
    q, r = divmod(total, m)
    return [q + 1] * r + [q] * (m - r)

def B(sizes):
    m = len(sizes)
    if m <= 1:
        return 1.0
    s = float(sum(sizes))
    h = -sum((x / s) * math.log(x / s) for x in sizes)
    return h / math.log(m)

# the count ladder's interior arms must BE the balanced partitions at their m,
# recomputed here from the measured T.  A mismatch aborts the submission.
want = {4: balanced(4), 8: balanced(8), 16: balanced(16), 32: balanced(32),
        2: balanced(2)}
bad = 0
for spec in specs:
    g = HF.polish_the_stepsize_groups(None, spec, nps)
    sizes = [len(x) for x in g]
    assert sum(sizes) == T, "%s does not cover the model" % spec
    lines.append("SPEC %s M %d SIZES %s FIRSTGROUP %s LASTGROUP %s"
                 % (spec, len(g), ",".join(str(s) for s in sizes),
                    " ".join(g[0]), " ".join(g[-1])))
    mark = ""
    if len(g) in want and sizes == want[len(g)]:
        mark = "  <- BALANCED at m=%d" % len(g)
    print("guard 4: %-70s m=%-3d B=%.5f%s"
          % (spec if len(spec) < 70 else spec[:67] + "...", len(g), B(sizes), mark))
    print("         group sizes = %s" % sizes)
    print("         first group = %s" % " ".join(g[0]))
    print("         last  group = %s" % " ".join(g[-1]))
# the count-ladder arms must be exactly the balanced partitions
for m in (2, 4, 8, 16, 32):
    spec = "[" + ",".join(str(x) for x in balanced(m)) + "]"
    if spec not in specs:
        print("!!! guard 4: the balanced partition at m=%d is %s, which this "
              "script does NOT compose" % (m, spec))
        bad += 1
if bad:
    raise SystemExit(1)
print("guard 4: all five count-ladder interior arms ARE the balanced partitions")
# the balance ladder must be fixed at m=2 and ordered in B
bl = [s for s in specs if len(HF.polish_the_stepsize_groups(None, s, nps)) == 2]
print("guard 4: %d m=2 arms: %s" % (len(bl), " ".join(bl)))
for s in bl:
    sz = [len(x) for x in HF.polish_the_stepsize_groups(None, s, nps)]
    print("         %-10s sizes %-10s B=%.5f ratio=%.2f"
          % (s, str(sz), B(sz), float(max(sz)) / float(min(sz))))
assert abs(B([45, 17]) - B([17, 45])) < 1e-12, "the mirror pair must share B"
open(out, "w").write("\n".join(lines) + "\n")
print("guard 4: PARTITION-MANIFEST.txt written to %s" % out)
PYEOF
[ -s "$MANIFEST" ] && echo "guard 4b: manifest is non-empty" || guard_fail "guard 4b: no manifest"

# ---- guard 5: capacity ------------------------------------------------------
FREE=$(df -BG "$WS" 2>/dev/null | tail -1 | awk '{print $4}' | tr -d 'G')
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: ${FREE:-?} GB free on \$WS; $PEND jobs pending, adding 33"
[ "${PEND:-0}" -le 40 ] || guard_fail "guard 5: $PEND pending, over cap"

[ "$FAIL" = 1 ] && { echo "!!! SUBMIT ABORTED -- nothing submitted."; exit 2; }

# ---- compose, guard each line under RULE 20, then submit --------------------
. "$REPO/bin/_lib_guards.sh"
DESIGN=(alg-base=SGDm alg-meta=Lion meta-stepsize=$MST alpha0=$ALPHA0 \
        num-epochs=$EPOCHS batch-size=$BATCH gamma=$GAMMA dataset=$DSET NN-name=$NET)

N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    case $ARM in
      c01) G=scalar ;;
      c02) G=$G_C02 ;;
      c04) G=$G_C04 ;;
      c08) G=$G_C08 ;;
      c16) G=$G_C16 ;;
      c32) G=$G_C32 ;;
      c62) G=layerwise ;;
      b45) G=$G_B45 ;;
      b17) G=$G_B17 ;;
      b55) G=$G_B55 ;;
      b60) G=$G_B60 ;;
      *)   echo "!!! unknown arm $ARM"; exit 2 ;;
    esac
    # NOTE the single quotes around $G below: "[31,31]" is a bash bracket GLOB.
    # Unquoted, inside the eval, it would expand against the CWD if a file named
    # 3 or 1 or , ever existed there.  Quoted, argv reaches train.py literally
    # and the runner's `echo "ARGS: $@"` prints it unchanged for RULE 20.
    CMD="sbatch --job-name=cbl1-$ARM-s$S --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,HIER=none,SCHED=none,PROBE=0 \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
--gamma $GAMMA --meta-stepsize $MST --alpha0 $ALPHA0 --num-epochs $EPOCHS \
--stepsize-groups '$G' --seed $S --save-directory $SAVE --run-name cbl1-$ARM-s$S"
    if guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1; then :; else
      echo "!!! RULE 20 PRE-CHECK FAILED for cbl1-$ARM-s$S"
      guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1))
    fi
    if [ "$SUBMIT" = 1 ]; then eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
    else echo "$CMD"; echo; N=$((N+1)); fi
  done
done
echo "guard 6: $N composed command lines, $BAD failed the RULE 20 pre-check"
[ "$BAD" = 0 ] || { echo "!!! composed lines are malformed"; exit 2; }
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'dry run, nothing submitted' )); $REJ rejected ----"

if [ "$SUBMIT" = 1 ]; then
  grep -qx 'cbl1-' "$REPO/bin/PROTECTED.txt" 2>/dev/null || echo "cbl1-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- cbl1- added to bin/PROTECTED.txt ----"
  # jobs/run_cifar.sh hard-codes #SBATCH --output=$WS/runs/%x-%j.out, so the .out
  # files land in $WS/runs, NOT in --save-directory.
  guard_postlaunch "$WS/runs" "cbl1-" "${DESIGN[@]}"; RC=$?
  case $RC in
    0) echo "guard 7: RULE 20 post-launch check PASSED against a run's own ARGS line" ;;
    2) echo "guard 7: UNVERIFIED -- no cbl1 job started inside the window.  This is NOT"
       echo "         a mismatch.  Re-check once one starts with:"
       echo "         python3 $REPO/analysis/argsline_guard.py $WS/runs --name cbl1-" ;;
    *) echo "!!! guard 7: RULE 20 MISMATCH -- scancel THIS batch only:"
       echo "    squeue -h -u $USER_NAME -o '%i %j' | awk '\$2 ~ /^cbl1-/ {print \$1}' | xargs -r scancel" ;;
  esac
fi
