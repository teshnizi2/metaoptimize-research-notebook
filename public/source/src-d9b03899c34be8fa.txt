set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
# CYCLE 8 HEADLINE FOLLOW-UP.
# ad-l-* (n=2) produced the campaign's FIRST genuine interior optimum:
#   r=0   92.52 +-0.11 | r=0.03 93.01 +-0.17 | r=0.1 93.02 +-0.01 | r=0.3 91.44 +-0.02 | plain 91.12 +-0.26 (n=5)
# Non-monotone, and r=0.1 beats plain on test (+1.90) AND on final_train (99.88 vs 99.66)
# -- a Pareto win, unlike shrink which trades fit for test (train 97.8 at its best cells).
# This block (a) raises the peak cells to n=5, (b) adds r=0.05/0.2 to locate the peak,
# (c) puts n=3 on the shoulders.
# Time: measured 31 min/100ep on L4 -> 01:15:00 (2.4x headroom) and dual-partition so it backfills.
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ sbatch --parsable --job-name=$1-s$3 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 --time=01:15:00 \
  --export="ALL,AUGMENT=1,$C,HIER=additive,ETA_RATIO=$2" \
  jobs/run_cifar.sh $B --stepsize-groups layerwise --seed $3 \
  --save-directory $R/addit --run-name $1_s$3 >/dev/null; }
# (a) peak cells to n=5
for s in 2 3 4; do sub ad-l-r003 0.03 $s; sub ad-l-r01 0.1 $s; done
# (b) locate the peak
for s in 0 1 2; do sub ad-l-r005 0.05 $s; sub ad-l-r02 0.2 $s; done
# (c) shoulders to n=3
sub ad-l-r0  0   2
sub ad-l-r03 0.3 2
sleep 3
echo "AD5_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -cE '^ad-l-(r003|r01|r005|r02|r0|r03)-s[234]|^ad-l-(r005|r02)-s[012]')"
echo "AD_TOTAL=$(squeue -h -u account_b -o '%j' | grep -c '^ad-l-')"
