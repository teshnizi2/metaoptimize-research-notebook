#!/bin/bash
# =============================================================================
# Cycle 49 -- the FREE-BETA FAMILY LADDER.  Runs on cluster_a.  18 jobs.
#
# OPEN QUESTION IT ANSWERS, and it is the one asymmetry left in Direction C.
#   CORRECTIONS 52(2) states the project's result in two halves:
#     (a) per-weight meta-gradients are NOT independent AT THE ADAPTED EQUILIBRIUM --
#         free rho_s(weightwise) = 8.888e-08, reproducing FINDINGS 44.3 to +5.1% -- and
#         that is the regime Adam-mini / Adalayer / SGG actually run in;
#     (b) the STRUCTURE is set by distance from the meta-optimum: channel-scaled off
#         equilibrium, approximately scale-free at it.
#   Half (b)'s OFF-equilibrium leg is now measured in FOUR families (fz3, FINDINGS 48.17).
#   Half (a) -- THE HEADLINE, the sentence the paper is actually for -- and the whole of
#   (b)'s AT-equilibrium leg are measured in exactly ONE: ResNet18 / CIFAR-10.
#   The first reviewer question is therefore not "is it real", it is "is it ResNet18".
#   This batch is the free-beta mirror of fz3: same three families, same three rungs,
#   same seeds, so every frozen number already on disk gets its free partner.
#
#   It also buys, at zero extra cost, the replication that CORRECTIONS 53 says the
#   cycle-49 transfer function needs.  `analysis/probe5_time_ladder.py` measured on R18
#   that the frozen/free gain is ~1 at every scale in Q1 (before beta moves) and then
#   switches on to 23.7x at k=1 / 4.57x at k=775 / 0.61x at layerwise.  That table is one
#   family and it was found POST-HOC.  These runs make it pre-registered and 4-family.
#
# PRE-REGISTERED PREDICTIONS -- score the verdict against these, not against what lands.
#
#   (C0) VALIDITY GATE, batch void per family if it fails.
#        1. beta must MOVE: beta_true_max != beta_true_min on the last record, 18/18.
#           This is the INVERSE of fz3's check and it must be run, not assumed.
#        2. each family's weightwise n_tot must equal its fz3 counterpart EXACTLY --
#           r10 4,903,242 / r34 21,282,122 / c100 11,220,132.  A mismatch means a
#           different network was built and the frozen/free contrast is void.
#        3. n_records == 2000 on every arm.
#
#   (C1) THE HEADLINE GENERALISES.  Steady-half free rho_s at WEIGHTWISE resolves
#        (rho_s > its own rho_min) in all three families, i.e. per-weight meta-gradients
#        are correlated at the adapted equilibrium outside ResNet18/CIFAR-10 too.
#        REFUTATION: any family whose free weightwise rho_s lands BELOW its own rho_min is
#        a BOUND, not an absence (CORRECTIONS 33), and half (a) narrows to the families
#        where it does resolve.  If it fails in 2 of 3, the headline is a ResNet18 result
#        and the paper must say so in its title sentence.
#
#   (C2) THE FILTER'S TRANSFER FUNCTION REPLICATES.  Per family, gain = frozen rho_w /
#        free rho_w on the steady half, frozen taken from the byte-matched fz3 arm:
#          gain(k=1) in [8x, 60x]          (R18 measured 22.9x, and 22.8x independently
#                                           in FINDINGS 44.3 from a different statistic)
#          gain strictly DECREASING in k within each family
#          gain(layerwise) within 3x of 1.0 -- adaptation does nothing at layer scale
#        REFUTATION: gain non-monotone in k in 2 or more families -> "a step-size adapter
#        is a high-pass filter" is a ResNet18 statement and CORRECTIONS 49(2)/52(2) must
#        be narrowed to one architecture.  A single non-monotone family with an
#        unresolved rung in the chain is NOT a refutation -- it is a hole; say so.
#
#   (C3) THE EQUILIBRIUM PROFILE FLATTENS.  Free b1 (the weight->channel leg) lands BELOW
#        its own frozen b1 in 3 of 3, and at or below 0.05 in each:
#              family   frozen b1 (FINDINGS 48.18)   R18 reference
#              r10      0.181                        frozen 0.156 -> free -0.059
#              c100     0.150
#              r34      0.069   <-- thin margin, say so rather than claiming a win
#        REFUTATION: any family with free b1 >= its frozen b1 -> the flattening is
#        ResNet18-specific.  r34's frozen b1 is only 0.069, so a null there is weak
#        evidence either way and must be reported as such, not counted as a pass.
#
#   (C4) EXCHANGEABILITY IS NOT REJECTED AT EQUILIBRIUM.  The parameter-free check
#        (calibrate the one-factor model on k=1 alone, predict every coarser rung) should
#        over-predict by < 5x in 3 of 3, against 13.6x-127x frozen (FINDINGS 48.17).
#        REFUTATION: rejection at >= 10x in any family -> the frozen/free asymmetry that
#        CORRECTIONS 47 built is not general.
#
#   RESOLUTION -- computed BEFORE submission from the fz3 arms already on disk, per
#   CORRECTIONS 33.  Predicted free rho_s = measured frozen rho_s / the R18 gain (22.9x at
#   weightwise, 5.49x nodewise, 0.77x layerwise).  Predicted free rho_min = the measured
#   FROZEN rho_min inflated 3.5x, which is the measured free/frozen rho_min ratio on the
#   R18 weightwise arm (2.90e-08 vs 8.24e-09) -- a free beta raises frac_zero and coarsens
#   the floor, and ignoring that would over-state every margin below.
#
#     family  rung  frozen rho_s  frozen rho_min  pred free rho_s  pred free rho_min  margin
#     r10     w        5.994e-06       2.033e-08        2.62e-07          7.1e-08      3.7x
#     r10     node     1.074e-03       1.063e-05        1.96e-04          3.7e-05      5.3x
#     r10     lay      2.909e-03       2.012e-03        3.78e-03          7.0e-03      0.5x  <-- see below
#     r34     w        8.146e-07       4.705e-09        3.56e-08          1.6e-08      2.2x
#     r34     node     4.269e-04       4.242e-06        7.78e-05          1.5e-05      5.2x
#     r34     lay      9.363e-03       8.938e-04        1.22e-02          3.1e-03      3.9x
#     c100    w        1.002e-06       8.651e-09        4.38e-08          3.0e-08      1.5x  <-- see below
#     c100    node     2.836e-04       6.085e-06        5.17e-05          2.1e-05      2.4x
#     c100    lay      7.706e-03       1.238e-03        1.00e-02          4.3e-03      2.3x
#
#   TWO RUNGS ARE EXPECTED TO BE THIN, AND THEY ARE NAMED HERE SO THAT A NULL IN EITHER
#   CANNOT LATER BE READ AS A DISCOVERY:
#     * c100 WEIGHTWISE, margin ~1.5x.  This is the rung C1 is scored on, so it is the
#       worst place for a thin margin.  A c100 weightwise null is a BOUND at rho_s ~ 3e-8,
#       NOT evidence that CIFAR-100 meta-gradients are independent -- and 3e-8 is itself
#       below the R18 value it would be compared against, so the bound would be
#       uninformative.  Score C1 on r10 and r34 if this happens, and say why.
#     * r10 LAYERWISE, which does not clear its floor even FROZEN (1.45x, the weakest rung
#       in the whole fz3 batch).  Predicted free margin 0.5x, i.e. it will NOT resolve.
#       It is submitted anyway because it costs 2 jobs and it is the k-endpoint of r10's
#       gain chain; C2's monotonicity for r10 is therefore expected to be scored on
#       k=1 -> k=566 alone, which is a HOLE and not a refutation.
#
#   WHY 2 SEEDS AND NOT 3.  fz3 measured seed-to-seed spread of rho_w at 3% (layerwise)
#   to 21% (weightwise) against effects of 16x-265x, and this batch's partner arms are
#   n=2.  Matching n keeps the gain ratio a like-for-like comparison; a third seed would
#   improve the free rho_min by 18% and cost 9 jobs.  The two thin rungs above are thin by
#   1.5x-3.7x, which a sqrt(3/2) is not going to rescue -- the honest fix for those is a
#   longer run, and a longer run would break the byte-matching that makes the gain ratio
#   meaningful in the first place.  So: match fz3, and report bounds where they fall.
#
# DESIGN.  Byte-for-byte the fz3 recipe with ONE field changed, `--alg-meta fixed` ->
# the p7free meta-optimizer, exactly as c48_free_profile_p5.sh did to p5:
#     --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
# 20 epochs, PROBE=5 -> T=2000 records.  Nothing else moves.
#
# NOT INCLUDED, deliberately:
#   * `resnet18_blocks` on ResNet10/ResNet34 -- ResNet18-only, dies with ZeroDivisionError
#     at HF.py:175 (six jobs already lost to this).
#   * `resnet18_blocks` on c100 -- it resolved in NEITHER arm at R18 (rho_s went negative
#     on the steady half in both frozen and free) and it is 2 jobs for a rung that has
#     never once cleared its own floor on this window.  fz3 included it; that was 2 jobs
#     of coverage that produced nothing quotable, and repeating it is not free.
#
# WHY cluster_a.  Account assignment is BY COMPARATOR (CORRECTIONS 46(6)).  Every number this
# batch is differenced against -- the whole fz3 frozen ladder -- is on cluster_a.  The cycle-49
# meta-stepsize ladder goes to cluster_b for the mirror-image reason: its comparators, p5 and
# fr5, are there.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. beta_true_max != beta_true_min on the last record of all 18 (C0.1)
#   2. weightwise n_tot matches fz3 exactly, per family (C0.2)
#   3. sum(neg_counts)/(n_records*n_tot) == time-mean of frac_neg to <1e-6
#   4. python3 analysis/probe5_window.py --selftest && analysis/probe5_time_ladder.py --selftest
#
# USAGE
#   bash c49_free_family_ladder.sh            # dry run: prints every sbatch line
#   bash c49_free_family_ladder.sh --submit   # submits, ONLY if the guards below pass
# =============================================================================
set -u

WS=/cluster/account_a/metaopt                  # cluster_a
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/ff5
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_a
PEND_CAP=40
NJOBS=18
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c49 FREE-beta family ladder + PROBE5 (cluster_a) ==="

if ! grep -q PATCH_PROBE5 "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5 not present in $HF"; exit 1
fi
if ! grep -q PATCH_PROBE5_FIX "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5_FIX not present -- unfixed patch crashes at record 500"; exit 1
fi
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX present  OK"

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 2: FairShare=${FS:-unknown} (informational only), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
sub () {   # sub <family> <nn-name> <dataset> <gran-token> <stepsize-groups> <seed>
  local fam=$1 nn=$2 ds=$3 gtok=$4 groups=$5 s=$6
  local RN="ff5-${fam}-${gtok}-s${s}"
  local PD="$SAVE/probe_${fam}_${gtok}_s${s}"
  N=$((N+1))
  local CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
    --export="ALL,AUGMENT=1,BETA_CLIP=-15:-2.3026,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
    "$RUNNER"
    --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
    --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
    --dataset "$ds" --NN-name "$nn" --batch-size 100
    --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3
    --num-epochs 20 --stepsize-groups "$groups" --seed "$s"
    --save-directory "$SAVE" --run-name "$RN")
  if [ "$SUBMIT" = "1" ]; then "${CMD[@]}"; else printf '%s\n' "${CMD[*]}"; fi
}

for s in 0 1; do
  # ResNet10 / CIFAR-10     frozen partner fz3-r10-*   (b1 = 0.181, span 264.6x)
  sub r10  ResNet10      CIFAR10  lay  layerwise  "$s"
  sub r10  ResNet10      CIFAR10  node nodewise   "$s"
  sub r10  ResNet10      CIFAR10  w    weightwise "$s"
  # ResNet34 / CIFAR-10     frozen partner fz3-r34-*   (b1 = 0.069, span 16.6x)
  sub r34  ResNet34      CIFAR10  lay  layerwise  "$s"
  sub r34  ResNet34      CIFAR10  node nodewise   "$s"
  sub r34  ResNet34      CIFAR10  w    weightwise "$s"
  # ResNet18 / CIFAR-100    frozen partner fz3-c100-*  (b1 = 0.150, span 23.3x)
  sub c100 ResNet18_c100 CIFAR100 lay  layerwise  "$s"
  sub c100 ResNet18_c100 CIFAR100 node nodewise   "$s"
  sub c100 ResNet18_c100 CIFAR100 w    weightwise "$s"
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo SUBMITTED || echo 'dry run, nothing submitted' )) ----"
echo "reduce with:  python3 analysis/probe5_window.py ../probes_ff5       (C1, C3, C4)"
echo "         and: python3 analysis/probe5_time_ladder.py ../probes_ff5 ../probes_fz3   (C2)"
