EPTR_RE = re.compile(r"Epoch\s+(\d+),\s*Train Accuracy:\s*([0-9.]+|nan)\s*%,\s*Test Accuracy:\s*([0-9.]+|nan)")

def mean(v):
    return sum(v) / float(len(v))


def sd(v):
    if len(v) < 2:
        return 0.0
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def med(v):
    v = sorted(x for x in v if x is not None and math.isfinite(x))
    n = len(v)
    if not n:
        return float("nan")
    return v[n // 2] if n % 2 else 0.5 * (v[n // 2 - 1] + v[n // 2])


def file_sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()

def rung_state(d, se, diverged, healthy, boxfree):
    if diverged:
        return "DIVERGED"
    if not healthy:
        return "UNHEALTHY"
    if not boxfree:
        return "BOXBOUND"
    if d <= -NULLBAR and d < -RES * se:
        return "REVERSES"
    if d <= NULLBAR and d + RES * se < POOL_EXCLUDE:
        return "VANISHES"
    if d >= SURV and d > RES * se:
        return "SURVIVES"
    return "UNDECIDED"


def primary_of(s1, s4):
    if s1 in ("DIVERGED", "UNHEALTHY", "BOXBOUND"):
        return "ANCHOR-UNREADABLE"
    if s1 in ("VANISHES", "REVERSES"):
        return "ANCHOR-NOT-REPRODUCED"
    if s4 in ("DIVERGED", "UNHEALTHY", "BOXBOUND"):
        return "UNRESOLVED-%s-W4" % s4
    return "AUDIT-%s" % s4


def scalar_of(k_level, k_div, cmp_ok, t_ch, t_nd, se):
    if k_div:
        return "SCALAR-DIVERGED"
    if k_level < HEALTH_MIN:
        return "SCALAR-COLLAPSED"
    if not cmp_ok:
        return "SCALAR-UNREADABLE"
    tmin = min(t_ch, t_nd)
    if t_ch >= SURV and t_nd >= SURV and t_ch > RES * se and t_nd > RES * se:
        return "SCALAR-BEATS-BEST"
    if tmin - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tmin <= -SURV and tmin < -RES * se:
        return "SCALAR-BELOW-BEST"
    return "SCALAR-UNRESOLVED"


def denom_of(ch_ok, gap, se):
    if not ch_ok:
        return "DENOM-UNREADABLE"
    if gap >= DENOM_BAR and gap > RES * se:
        return "DENOM-HOLDS"
    if gap <= -SURV and gap < -RES * se:
        return "DENOM-REVERSES"
    if gap + RES * se < DENOM_BAR:
        return "DENOM-CLOSES"
    return "DENOM-UNRESOLVED"


def compose_sigma(sigma_in):
    return (SIGMA_PRIOR, "SIGMA_PRIOR_frozen") if SIGMA_PRIOR >= sigma_in else (sigma_in, "SIGMA_INBATCH")


def decide(A, sigma_used):
    """A[arm] = dict(level, train, range, n, boxfree).  Returns (primary, scalar, denom, states, keys)."""
    states, keys = {}, {}
    for r in D.RUNG_IDS:
        ch, nd = A["ch" + r], A["nd" + r]
        d = ch["level"] - nd["level"]
        se = sigma_used * math.sqrt(1.0 / ch["n"] + 1.0 / nd["n"])
        states[r] = rung_state(d, se, ch["range"] > DIVERGED or nd["range"] > DIVERGED,
                               ch["level"] >= HEALTH_MIN and nd["level"] >= HEALTH_MIN,
                               ch["boxfree"] and nd["boxfree"])
        keys["D_" + r] = (d, se)
        keys["DTRAIN_" + r] = ch["train"] - nd["train"]
        k, L = A["k01" + r], A["kL" + r]
        for g in ("ch", "nd"):
            x = A[g + r]
            keys["T%s_%s" % (g, r)] = (k["level"] - x["level"], sigma_used * math.sqrt(1.0 / k["n"] + 1.0 / x["n"]))
        keys["G_" + r] = (L["level"] - k["level"], sigma_used * math.sqrt(1.0 / k["n"] + 1.0 / L["n"]))
        for g in D.GRAIN_IDS:
            x = A[g + r]
            keys["GAP%s_%s" % (g, r)] = (D.SGD_LANDED - x["level"],
                                         math.sqrt(sigma_used ** 2 / x["n"] + D.SGD_LANDED_SD ** 2 / D.SGD_LANDED_N))
    primary = primary_of(states["W1"], states["W4"])
    k4, c4, n4 = A["k01W4"], A["chW4"], A["ndW4"]
    cmp_ok = (c4["level"] >= HEALTH_MIN and n4["level"] >= HEALTH_MIN and c4["range"] <= DIVERGED
              and n4["range"] <= DIVERGED)
    scal = scalar_of(k4["level"], k4["range"] > DIVERGED, cmp_ok, keys["Tch_W4"][0], keys["Tnd_W4"][0],
                     keys["Tch_W4"][1])
    ch_ok = c4["level"] >= HEALTH_MIN and c4["range"] <= DIVERGED
    den = denom_of(ch_ok, keys["GAPch_W4"][0], keys["GAPch_W4"][1])
    d1, se1 = keys["D_W1"]
    d4, se4 = keys["D_W4"]
    keys["DD_W4"] = (d4 - d1, math.sqrt(se1 ** 2 + se4 ** 2))
    return primary, scal, den, states, keys

def parse_args_line(line):
    """-> (pairs in order, repeated flags).  argparse semantics for this harness: every value is ONE token."""
    toks = shlex.split(line[len("ARGS:"):])
    pairs, seen, rep, i = [], set(), [], 0
    while i < len(toks):
        t = toks[i]
        if not t.startswith("--"):
            return None, ["stray token %r" % t]
        if i + 1 >= len(toks) or toks[i + 1].startswith("--"):
            return None, ["flag %s has no value" % t]
        f = t[2:]
        if f in seen:
            rep.append(f)
        seen.add(f)
        pairs.append((f, toks[i + 1]))
        i += 2
    return pairs, rep


def read_out(path):
    """Keeps the FIRST ARGS / ENV line; an epoch printed twice is a STRUCTURE FAILURE here (cgw1's latent F1, 296.6,
    closed in this scorer by construction: `dup_epochs` is gated in score())."""
    L = open(path, errors="replace").read().splitlines()
    info = dict(path=path, args=None, env=None, gpu=None, done=False, eps={}, kinds_on=[], node=None, pt=False,
                dup_epochs=[], n_args=0, n_env=0)
    for ln in L:
        if ln.startswith("ARGS:"):
            info["n_args"] += 1
            if info["args"] is None:
                info["args"] = ln
        elif ln.startswith("ENV:"):
            info["n_env"] += 1
            if info["env"] is None:
                info["env"] = ln
        elif ln.startswith("NODE=") and info["node"] is None:
            info["node"] = ln.split("|")[0].strip()
        elif ln.startswith("NVIDIA") and info["gpu"] is None:
            info["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            info["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            info["pt"] = True
        else:
            m = EPTR_RE.search(ln)
            if m:
                e = int(m.group(1))
                if e in info["eps"]:
                    info["dup_epochs"].append(e)
                info["eps"][e] = (float(m.group(2)), float(m.group(3)))
            k = KIND_ON_RE.match(ln)
            if k:
                info["kinds_on"].append(ln.strip())
    info["complete"] = info["done"] and all(e in info["eps"] for e in range(D.EPOCHS))
    return info


def find_outs(runsdir):
    out = []
    for base in (runsdir, os.path.join(runsdir, PREFIX)):
        out += glob.glob(os.path.join(base, "%s-*.out" % PREFIX))
    return sorted(set(os.path.realpath(p) for p in out))


def probe_path(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, D.probe_dir_name(arm, seed), "probe.jsonl")
        if os.path.exists(p):
            return p
    return None


def read_probe(path):
    """Each record keeps the gate fields and u = median over the record's step sizes of exp(beta) (wd-free)."""
    recs = []
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            r = json.loads(ln)
        except ValueError:
            recs.append(None)
            continue
        if not isinstance(r, dict):
            recs.append(None)
            continue
        x = {k: r.get(k) for k in ("step", "n_beta", "beta_true_min", "beta_true_max", "n_at_lo", "n_at_hi")}
        b = r.get("beta")
        if isinstance(b, list) and b and all(isinstance(v, (int, float)) and math.isfinite(v) for v in b):
            x["nb_list"] = len(b)
            x["u"] = med([math.exp(v) for v in b])
        else:
            x["nb_list"] = None
            x["u"] = None
        recs.append(x)
    return recs


def beta_len_want(m):
    """the record's `beta` list: 1 entry on a scalar arm, 62 per-tensor entries on every other grain (296.5)."""
    return 1 if m == 1 else D.NTENS


def probe_audit(recs, m_want):
    """-> (ok, why, occupancy dict).  G-STRUCT: 10,000 records at steps 0,5,..; n_beta == the arm's m on EVERY record;
    the `beta` list has the grain's length on every record (the dose is read from it)."""
    if recs is None:
        return False, "no probe.jsonl", None
    if len(recs) != D.N_RECORDS:
        return False, "%d records, want %d" % (len(recs), D.N_RECORDS), None
    if any(r is None for r in recs):
        return False, "an unreadable record", None
    steps = [r["step"] for r in recs]
    if steps != list(range(0, D.N_RECORDS * D.PROBE, D.PROBE)):
        return False, "record steps are not 0,5,...,%d" % ((D.N_RECORDS - 1) * D.PROBE), None
    nb = set(r["n_beta"] for r in recs)
    if nb != {m_want}:
        return False, "n_beta %s != the arm's registered m %d (the grain is not what ARGS says)" % (sorted(nb)[:3], m_want), None
    bl = set(r["nb_list"] for r in recs)
    if bl != {beta_len_want(m_want)}:
        return False, "beta list length %s != %d on some record" % (sorted(bl, key=str)[:3], beta_len_want(m_want)), None
    mins = [r["beta_true_min"] for r in recs]
    maxs = [r["beta_true_max"] for r in recs]
    if not (max(maxs) - min(mins) > 1e-9):
        return False, "beta never moved", None
    T = len(recs)
    rec_lo = sum(1 for x in mins if x <= D.LO + BOX_EPS) / float(T)
    rec_hi = sum(1 for x in maxs if x >= D.HI - BOX_EPS) / float(T)
    win = [r for r in recs if r["step"] >= WIN_STEP]
    tot = float(sum(r["n_beta"] for r in win))
    have = all(r.get("n_at_lo") is not None and r.get("n_at_hi") is not None for r in win)
    occ = dict(rec_lo=rec_lo, rec_hi=rec_hi,
               coord_lo_win=(sum(r["n_at_lo"] for r in win) / tot) if have else None,
               coord_hi_win=(sum(r["n_at_hi"] for r in win) / tot) if have else None,
               boxfree=(rec_lo < BOXFREE_MAX and rec_hi < BOXFREE_MAX))
    return True, "", occ


def dose_of_run(recs, wd):
    """-> (plateau median, peak) of the realised per-step shrink u*wd (caw2's definition; DESCRIPTIVE)."""
    allv = [r["u"] * wd for r in recs if r["u"] is not None]
    win = [r["u"] * wd for r in recs if r["u"] is not None and r["step"] >= WIN_STEP]
    return (med(win), max(allv) if allv else float("nan"))


def read_corpus_disclosure(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    sys.path.insert(0, HERE)
    try:
        import corpus_exclusions as ce
    finally:
        sys.path.pop(0)
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    return len(ce.filter_rows(rows))


def resolve_prov(runsdir):
    for p in (os.path.join(runsdir, PREFIX, "PROVENANCE.txt"), os.path.join(runsdir, "PROVENANCE.txt")):
        if os.path.exists(p):
            return p
    return None

def score(runsdir, csvpath=DEFAULT_CSV, out=print):
    P = out
    P("=" * 110)
    P("g3b -- THE COUNT-MATCHED AUDIT, SCALAR, AND THE DENOMINATOR ON CIFAR-100 AT 0.1 AND 5e-4  (CORRECTIONS 299; G3b)")
    P("scorer sha256 %s   design sha256 %s" % (file_sha(os.path.abspath(__file__)), file_sha(os.path.join(HERE, "g3b_design.py"))))
    P("runsdir %s" % runsdir)
    P("=" * 110)
    harness = []

    def hfail(msg):
        harness.append(msg)
        P("  !!! HARNESS %s" % msg)

    found = {}
    for p in find_outs(runsdir):
        m = OUT_RE.match(os.path.basename(p))
        if not m:
            hfail("unparseable file name %s" % os.path.basename(p))
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        if arm not in D.ARMS or seed not in D.SEEDS[arm]:
            hfail("a %s .out that is not a registered run: %s" % (PREFIX, os.path.basename(p)))
            continue
        found.setdefault((arm, seed), []).append((jid, p))
    runs, missing, ignored = {}, [], []
    for (arm, seed) in D.RUNS:
        cands = [(jid, p, read_out(p)) for jid, p in sorted(found.get((arm, seed), []))]
        comp = [c for c in cands if c[2]["complete"]]
        ignored += [os.path.basename(c[1]) for c in cands if not c[2]["complete"]]
        if not comp:
            missing.append(D.run_name(arm, seed))
        elif len(comp) > 1:
            hfail("%s has %d COMPLETE .out files (%s) -- ambiguous" % (D.run_name(arm, seed), len(comp),
                  ", ".join(os.path.basename(c[1]) for c in comp)))
        else:
            runs[(arm, seed)] = comp[0]
    P("COMPLETION  %d / %d registered runs complete (100 epoch lines + RUN_DONE)" % (len(runs), D.NJOBS))
    if ignored:
        P("  ignored (incomplete, a complete twin or none): %s" % ", ".join(sorted(ignored)))
    if missing:
        P("  MISSING / INCOMPLETE: %s" % ", ".join(missing))

    prov = resolve_prov(runsdir)
    if prov is None:
        hfail("G-PROV: no PROVENANCE.txt under %s/%s" % (runsdir, PREFIX))
    else:
        kv = dict(ln.split(" ", 1) for ln in open(prov).read().splitlines() if " " in ln)
        want = {"SCORER_SHA256": file_sha(os.path.abspath(__file__)),
                "DESIGN_SHA256": file_sha(os.path.join(HERE, "g3b_design.py"))}
        for k, v in sorted(want.items()):
            if kv.get(k, "").strip() != v:
                hfail("G-PROV: %s in %s is %r, the file that is running is %s" % (k, prov, kv.get(k), v[:16]))
        P("G-PROV  %s  MODE %s  SCORER %s  DESIGN %s  HF %s" % (prov, kv.get("MODE", "?"), kv.get("SCORER_SHA256", "?")[:16],
          kv.get("DESIGN_SHA256", "?")[:16], kv.get("HF_SHA256", "?")[:16]))
        if kv.get("HF_SHA256", D.HF_LIVE_SHA).strip() != D.HF_LIVE_SHA:
            hfail("G-PROV: HF_SHA256 is not the registered live HF %s" % D.HF_LIVE_SHA[:16])

    per = {}
    gpus = {}
    save_dirs = set()
    for (arm, seed), (jid, path, info) in sorted(runs.items()):
        rn = D.run_name(arm, seed)
        if info["args"] is None or info["n_args"] != 1 or info["n_env"] != 1:
            hfail("G-ARGS %s: %d ARGS lines and %d ENV lines (want exactly 1 each)" % (rn, info["n_args"], info["n_env"]))
            continue
        if info["dup_epochs"]:
            hfail("G-STRUCT %s: epoch line(s) printed twice %s" % (rn, sorted(set(info["dup_epochs"]))[:5]))
            continue
        pairs, rep = parse_args_line(info["args"])
        if pairs is None or rep:
            hfail("G-ARGS %s: %s" % (rn, rep))
            continue
        sd_ = dict(pairs).get("save-directory", "")
        save_dirs.add(sd_)
        want = D.expected_args(arm, seed, sd_)
        if pairs != want:
            diff = [(f, v, dict(want).get(f)) for f, v in pairs if dict(want).get(f) != v]
            miss = [f for f, _v in want if f not in dict(pairs)]
            hfail("G-ARGS %s: %d flags differ from the registered line %s missing %s"
                  % (rn, len(diff), diff[:4], miss[:4]))
            continue
        pdir = os.path.join(sd_, D.probe_dir_name(arm, seed))
        want_env = D.ENV_TEMPLATE.replace("<P>", pdir)
        if info["env"] != want_env:
            hfail("G-ENV %s: ENV line is not the registered one (%r)" % (rn, (info["env"] or "")[:160]))
            continue
        if info["kinds_on"] or info["pt"]:
            hfail("G-KIND %s: prints a patch witness %s" % (rn, (info["kinds_on"] or ["PROBE_TENSOR"])[:2]))
            continue
        pp = probe_path(runsdir, arm, seed)
        recs = read_probe(pp) if pp else None
        ok, why, occ = probe_audit(recs, D.M_OF_ARM[arm])
        if not ok:
            hfail("G-STRUCT %s: %s" % (rn, why))
            continue
        eps = info["eps"]
        per[(arm, seed)] = dict(level=mean([eps[e][1] for e in range(PL_LO, PL_HI + 1)]),
                                train=mean([eps[e][0] for e in range(PL_LO, PL_HI + 1)]), occ=occ, jid=jid,
                                dose=dose_of_run(recs, D.WDF[arm]))
        gpus[info["gpu"]] = gpus.get(info["gpu"], 0) + 1
    if len(save_dirs) > 1:
        hfail("G-ARGS: the runs name %d different save directories %s" % (len(save_dirs), sorted(save_dirs)))
    P("G-ARGS / G-ENV / G-KIND / G-STRUCT  %d of %d complete runs pass (20 flags each, ONE ARGS and ONE ENV line, the "
      "registered ENV line, no patch witness, no epoch printed twice, 10,000 probe records with n_beta == the arm's m "
      "and a beta list of the grain's length)" % (len(per), len(runs)))
    P("G-HW (DISCLOSURE ONLY, never a gate): %s" % ", ".join("%s x%d" % (k, v) for k, v in sorted(gpus.items(), key=str)))
    n_corp = read_corpus_disclosure(csvpath)
    P("corpus: %s  (DISCLOSURE ONLY -- enters no bar, sigma, level, state, contrast or stamp)"
      % ("not readable" if n_corp is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (n_corp, PREFIX)))
    if harness:
        P("\nFINAL HARNESS-FAIL | %d harness failure(s); nothing is read" % len(harness))
        return 3
    if missing:
        P("\nFINAL INCOMPLETE | %d of %d runs missing or incomplete; nothing is read" % (len(missing), D.NJOBS))
        return 2

    A = {}
    ss, df = 0.0, 0
    pooled_arms = []
    P("\nARMS  plateau5 = mean TEST over epochs %d-%d from each run's OWN .out; box occupancy from its OWN probe.jsonl" % (PL_LO, PL_HI))
    P("  %-6s %-4s %-5s %-9s %2s %9s %7s %7s %9s  %-6s %8s %8s %11s %11s" % ("arm", "rung", "wd", "grain", "n", "plateau5", "sd",
      "range", "train5", "box", "rec_lo", "rec_hi", "coordlo_win", "coordhi_win"))
    for a in D.ARMS:
        xs = [per[(a, s)]["level"] for s in D.SEEDS[a]]
        tr = [per[(a, s)]["train"] for s in D.SEEDS[a]]
        occs = [per[(a, s)]["occ"] for s in D.SEEDS[a]]
        bf = all(o["boxfree"] for o in occs)
        m_ = mean(xs)
        if m_ >= HEALTH_MIN:
            ss += sum((x - m_) ** 2 for x in xs)
            df += len(xs) - 1
            pooled_arms.append(a)
        A[a] = dict(level=m_, train=mean(tr), range=max(xs) - min(xs), n=len(xs), boxfree=bf, sd=sd(xs), seeds=xs)
        cl = [o["coord_lo_win"] for o in occs]
        ch_ = [o["coord_hi_win"] for o in occs]
        P("  %-6s %-4s %-5s %-9s %2d %9.4f %7.3f %7.3f %9.4f  %-6s %8.4f %8.4f %11s %11s" % (
            a, D.RUNG_OF[a], D.WD[a], D.SPEC[a], len(xs), m_, A[a]["sd"], A[a]["range"], A[a]["train"],
            "free" if bf else "BOUND", max(o["rec_lo"] for o in occs), max(o["rec_hi"] for o in occs),
            "na" if None in cl else "%.6f" % max(cl), "na" if None in ch_ else "%.6f" % max(ch_)))
        P("         seeds: %s" % ", ".join("s%d %.4f" % (s, x) for s, x in zip(D.SEEDS[a], xs)))
    sig_in = math.sqrt(ss / df) if df else 0.0
    sig, which = compose_sigma(sig_in)
    P("\nSIGMA  O2: max(frozen cell floor, in-batch) -- nothing from the corpus")
    P("  SIGMA_PRIOR_frozen   %.6f  (df %d, filtered, literal; this cell)" % (SIGMA_PRIOR, SIGMA_PRIOR_DF))
    P("  SIGMA_INBATCH        %.6f  (df %d, pooled within the %d arms with mean >= %.0f: %s)"
      % (sig_in, df, len(pooled_arms), HEALTH_MIN, " ".join(pooled_arms)))
    P("  SIGMA_USED           %.6f  (= %s)" % (sig, which))
    P("  SIGMA_DEMO (quoted)  %.6f  (df %d; corpus_exclusions --check SIGMA_R18ALL, the mechanism cell; DESCRIPTIVE)"
      % (SIGMA_DEMO, SIGMA_DEMO_DF))

    primary, scal, den, states, K = decide(A, sig)
    P("\nCONTRASTS (every one WITHIN a rung, IN BATCH unless marked; +/-2 SE is a HALF-WIDTH, not a bound on any effect)")
    for r in D.RUNG_IDS:
        d, se = K["D_" + r]
        P("  D_%s  = ch%s - nd%s = %+.4f pp = %+.2f SE (SE %.6f)  [%+.4f, %+.4f]  TRAIN %+.4f   state %s"
          % (r, r, r, d, d / se, se, d - RES * se, d + RES * se, K["DTRAIN_" + r], states[r]))
    v, se = K["DD_W4"]
    P("  DD_W4 = D_W4 - D_W1 = %+.4f pp = %+.2f SE (SE %.6f)   DESCRIPTIVE" % (v, v / se, se))
    for r in D.RUNG_IDS:
        for g in ("ch", "nd"):
            v, se = K["T%s_%s" % (g, r)]
            P("  T%s_%s = k01%s - %s%s = %+.4f pp = %+.2f SE%s" % (g, r, r, g, r, v, v / se,
              "" if r == "W4" else "   DESCRIPTIVE (k01W1 is predicted collapsed)"))
    for r in D.RUNG_IDS:
        v, se = K["G_" + r]
        P("  G_%s  = kL%s - k01%s = %+.4f pp = %+.2f SE   DESCRIPTIVE (layerwise vs scalar in this cell)" % (r, r, r, v, v / se))
    P("\nDENOMINATOR (BETWEEN-BATCH: tuned SGD + cosine = the landed cdn1 lr 0.1 literal %.4f, n %d, sd %.4f, wd 5e-4)"
      % (D.SGD_LANDED, D.SGD_LANDED_N, D.SGD_LANDED_SD))
    for r in D.RUNG_IDS:
        for g in D.GRAIN_IDS:
            v, se = K["GAP%s_%s" % (g, r)]
            tag = "   GATED (the cdn1-m arm)" if (g, r) == ("ch", "W4") else "   DESCRIPTIVE"
            P("  GAP%s_%s = SGD - %s%s = %+.4f pp = %+.2f SE (SE %.6f)  [%+.4f, %+.4f]%s"
              % (g, r, g, r, v, v / se, se, v - RES * se, v + RES * se, tag))
    best4 = max(D.GRAIN_IDS, key=lambda g: A[g + "W4"]["level"])
    P("  best grain at W4 = %s (%.4f): GAP %+.4f -- DESCRIPTIVE, a max over four arms is biased UP, so this gap is biased DOWN"
      % (D.SPEC_OF_GRAIN[best4], A[best4 + "W4"]["level"], K["GAP%s_W4" % best4][0]))
    P("  landed cdn1 GAP_in (cdn1-m at 0.1, in cdn1's batch) %+.4f; GAPch_W1 here %+.4f (DESCRIPTIVE replication)"
      % (D.CDN1_GAP_IN, K["GAPch_W1"][0]))

    P("\nTHE DOSE (DESCRIPTIVE): realised per-step shrink a*wd = median over the arm's step sizes of exp(beta)*wd per record;"
      " plateau = median over records with step >= %d; arm = median over its seeds.  ch / nd are APPROXIMATE (62"
      " per-tensor mean log step sizes per record)." % WIN_STEP)
    for a in D.ARMS:
        dl = [per[(a, s)]["dose"] for s in D.SEEDS[a]]
        pl, pk = med([x[0] for x in dl]), med([x[1] for x in dl])
        extra = ""
        if D.RUNG_OF[a] == "W4":
            extra = "   = 1/%.0f of lr*wd %.0e, 1/%.0f of a standard recipe's ~%.0e" % (
                LRWD_REF / pl if pl > 0 else float("nan"), LRWD_REF, STD_RECIPE_REF / pl if pl > 0 else float("nan"),
                STD_RECIPE_REF)
        P("  %-6s %-9s wd %-5s plateau %.4e  peak %.4e%s%s" % (a, D.SPEC[a], D.WD[a], pl, pk,
          "  (approx)" if D.GRAIN_OF[a] in ("ch", "nd") else "", extra))

    P("\nSENSITIVITY (DESCRIPTIVE, no token): the primary states re-read at the quoted SIGMA_DEMO %.6f instead of SIGMA_USED:"
      % SIGMA_DEMO)
    _p2, _s2, _d2, st2, _k2 = decide(A, max(sig, SIGMA_DEMO))
    P("  %s" % " ".join("%s-%s" % (r, st2[r]) for r in D.RUNG_IDS))

    P("\nLADDERS (LEVELS, not effects; this cell only)")
    for g in D.GRAIN_IDS:
        P("  %-9s %s" % (D.SPEC_OF_GRAIN[g], " | ".join("%s %.4f" % (D.WD_TOKEN[r], A[g + r]["level"]) for r in D.RUNG_IDS)))
    P("  cgw1 side by side (CIFAR-10, DESCRIPTIVE, cross-batch): D_W1 %+.4f, D_W4 %+.4f, Tch_W4 %+.4f, Tnd_W4 %+.4f"
      % (D.CGW1_D["W1"], D.CGW1_D["W4"], D.CGW1_T4["ch"], D.CGW1_T4["nd"]))

    stamps = list(BOUNDS)
    stamps.append("SIGMA-PRIOR-FROZEN" if which == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH")
    d1 = K["D_W1"][0]
    stamps.append("ANCHOR-MATCHES-POOL" if abs(d1 - D.C100_POOL_D) <= MATCH else "ANCHOR-DIFFERS-FROM-POOL")
    for g in D.GRAIN_IDS:
        if abs(A[g + "W1"]["level"] - D.LANDED_W1[g]) > MATCH:
            stamps.append("W1-LEVEL-DIFFERS-%s" % g)
    stamps.append("K01W1-COLLAPSED" if A["k01W1"]["level"] < HEALTH_MIN else "K01W1-HEALTHY")
    if primary in ("AUDIT-SURVIVES", "AUDIT-REVERSES"):
        stamps.append("TRAIN-AGREES" if (K["DTRAIN_W4"] > 0) == (K["D_W4"][0] > 0) else "TRAIN-DISAGREES")
    g4, gse = K["G_W4"]
    if g4 <= -SURV and g4 < -RES * gse:
        stamps.append("LAYERWISE-BELOW-SCALAR-W4")
    elif g4 >= SURV and g4 > RES * gse:
        stamps.append("LAYERWISE-ABOVE-SCALAR-W4")
    if not A["k01W4"]["boxfree"]:
        stamps.append("SCALAR-W4-BOX-BOUND")
    for a in D.ARMS:
        if A[a]["range"] > DIVERGED:
            stamps.append("DIVERGED-%s" % a)
    stateword = "+".join("%s-%s" % (r, states[r]) for r in D.RUNG_IDS)
    P("\nLICENCE (primary)  %s" % LICENCE[primary])
    P("LICENCE (scalar)   %s" % LICENCE[scal])
    P("LICENCE (denom)    %s" % LICENCE[den])
    P("\nFINAL %s | %s | %s | %s | %s" % (primary, scal, den, stateword, " ".join(stamps)))
    return 0

RESULTS = {"PASS": 0, "FAIL": 0, "SKIP": 0}


def chk(cond, label):
    RESULTS["PASS" if cond else "FAIL"] += 1
    print("  %s %s" % ("PASS" if cond else "FAIL", label))
    return cond


def skip(label):
    RESULTS["SKIP"] += 1
    print("  SKIP %s" % label)


def _curve(level):
    return [min(99.0, level * (1 - math.exp(-(e + 1) / 12.0))) for e in range(D.EPOCHS)]


def _write_out(d, arm, seed, jid, level, save_dir, args_over=None, env_over=None, extra_line=None, done=True,
               n_ep=D.EPOCHS, gpu="NVIDIA L4, 23034 MiB", dup_epoch=None, dup_args=False):
    rn = D.run_name(arm, seed)
    pairs = D.expected_args(arm, seed, save_dir)
    if args_over:
        pairs = args_over(pairs)
    args = "ARGS: " + " ".join("--%s %s" % kv for kv in pairs)
    env = D.ENV_TEMPLATE.replace("<P>", os.path.join(save_dir, D.probe_dir_name(arm, seed)))
    if env_over:
        env = env_over(env)
    c = _curve(level)
    L = ["NODE=[HOST_2f2f29d9e1] | JOB=%s %d | AUGMENT=1" % (rn, jid), args, env, gpu, "",
         "args_base includes unnecessary attributes {'normalizer_param'} ", ""]
    if dup_args:
        L.append(args)
    if extra_line:
        L.append(extra_line)
    for e in range(n_ep):
        te = level if e >= PL_LO - 5 else c[e]
        L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.99, te + 5.0), te))
        if dup_epoch == e:
            L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, 10.0, 1.0))
    L.append("%d  minutes" % 50)
    if done:
        L.append("RUN_DONE")
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(L) + "\n")


def _write_probe(pdir, m, lo=-8.0, hi=-5.0, n=D.N_RECORDS, nb=None, hi_frac=0.0, bval=-7.0, blen=None):
    os.makedirs(pdir, exist_ok=True)
    bl = beta_len_want(m) if blen is None else blen
    bstr = "[" + ",".join(["%.1f" % bval] * bl) + "]"
    with open(os.path.join(pdir, "probe.jsonl"), "w") as fh:
        for i in range(n):
            bmax = D.HI if i >= n * (1 - hi_frac) else hi + 1e-4 * (i % 7)
            fh.write('{"step": %d, "beta": %s, "beta_true_min": %.6f, "beta_true_max": %.6f, "n_beta": %d, '
                     '"n_at_lo": 0, "n_at_hi": %d}\n' % (i * D.PROBE, bstr, lo - 1e-4 * (i % 5), bmax,
                                                         nb if nb is not None else m, 1 if bmax >= D.HI else 0))


JIT = {4: (0.02, 0.03, -0.03, -0.02)}   # zero-mean


def _build(root, template, levels, save_dir="/SAVE/runs/g3b", over=None, drop=(), extra=None, probe_over=None,
           prov_over=None, jit=None):
    os.makedirs(os.path.join(root, PREFIX), exist_ok=True)
    jid = 7000000
    for (arm, seed) in D.RUNS:
        jid += 1
        if (arm, seed) in drop:
            continue
        i = D.SEEDS[arm].index(seed)
        lv = levels[arm] + (JIT[len(D.SEEDS[arm])][i] if jit is None else jit(arm, i))
        kw = dict((over or {}).get((arm, seed), {}))
        _write_out(os.path.join(root, PREFIX), arm, seed, jid, lv, save_dir, **kw)
        dst = os.path.join(root, PREFIX, D.probe_dir_name(arm, seed))
        if probe_over and (arm, seed) in probe_over:
            _write_probe(dst, D.M_OF_ARM[arm], **probe_over[(arm, seed)])
        else:
            os.symlink(os.path.join(template, D.probe_dir_name(arm, seed)), dst)
    for f in (extra or []):
        f(os.path.join(root, PREFIX))
    kv = {"MODE": "submit", "SCORER_SHA256": file_sha(os.path.abspath(__file__)),
          "DESIGN_SHA256": file_sha(os.path.join(HERE, "g3b_design.py")), "HF_SHA256": D.HF_LIVE_SHA}
    kv.update(prov_over or {})
    open(os.path.join(root, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % x for x in kv.items()))
    return root


def _run(root, csvpath=None):
    lines = []
    rc = score(root, csvpath, out=lines.append)
    lines = [x for ln in lines for x in ln.split("\n")]
    fin = [ln for ln in lines if ln.startswith("FINAL ")]
    return rc, (fin[0] if fin else ""), lines


def _levels(w1=(71.96, 70.42, 35.80, 71.08), w4=(69.50, 68.40, 71.50, 68.50)):
    L = {}
    for r, v in (("W1", w1), ("W4", w4)):
        for g, x in zip(D.GRAIN_IDS, v):
            L[g + r] = x
    return L


def _w4(ch=69.50, nd=68.40, k=71.50, kl=68.50):

    ap.add_argument("--runsdir", dest="runsdir_opt")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    a = ap.parse_args()

        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))

        ap.error("runsdir required")
