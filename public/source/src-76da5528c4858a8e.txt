#!/bin/bash
# =============================================================================
# Cycle 61 -- THE PER-COORDINATE SIGN PROBE.  cluster_a (account_a).  8 jobs.
# Tag `cp9-*`.  100 epochs.  Requires PATCH_PROBE6.
#
# WRITTEN WHILE cluster_a WAS DOWN FOR A SIXTH CONSECUTIVE TICK (FINDINGS 61.0) AND
# NOT SUBMITTED.  Registered now so the next reachable tick submits rather than
# designs.  Its own guards re-check every precondition at submit time.
#
# WHY THIS OUTRANKS `rw9`.  Three open items in this campaign, reached from three
# unrelated directions, all reduce to ONE missing instrument:
#
#   (1) `KILLTEST-idea2.md` section 5 caveat 1.  Idea 2's differentiator (ii) is
#       dead at TENSOR granularity (cross-block dependence -0.04pp / +0.03pp, at
#       the null).  Its ONE surviving question is whether the weights INSIDE a
#       tensor cluster across tensor boundaries.  "The probe never writes
#       per-coordinate signs", so this is unanswerable from every run in the
#       campaign.
#   (2) CORRECTIONS 89.7 -- why `conv1` and not `conv2`.  Cycle 61 showed the
#       per-TENSOR z time series IS recoverable from the cumulative moments
#       already on disk (`analysis/c61_z_timecourse.py`, 46/46 selftests) and
#       used it.  The answer was the PRE-REGISTERED NULL: the exception leaves
#       no signature in the tensor MEAN, at any window width from 1000 steps
#       down to 50, and the seed whose row structure inverts by 500x is the
#       LOWEST of its three siblings (FINDINGS 61.4-61.5).  That is expected
#       under H_B -- a spatial mean over rows destroys within-tensor structure
#       by construction -- so the question needs a per-ROW or per-COORDINATE
#       series.  Nothing on disk has one.
#   (3) Direction C, the headline.  "53.1% of 11.17M per-weight meta-gradients
#       agree in sign" is a claim about COORDINATES, but every correlation-
#       structure statistic the campaign owns is at tensor or row granularity
#       (59.3, 59.7, 59.8, 60.2-60.7).  Whether the structure is coordinate-level
#       has never been measured.
#
# `rw9` (18 jobs) decides D2, one question about the weightwise arm.  This batch
# (8 jobs) is the only thing that can move any of the three above, and it is the
# DEFAULT research direction rather than a side question.  `sp8` (cluster_b) and
# `hz9` (cluster_a, 9) keep their places -- they are cheap, validated, and hz9
# decides whether the current headline is a general claim.  Submit sp8, hz9,
# THEN this, THEN rw9.
#
# THE INSTRUMENT.  `patches/patch_probe6_coord.py`, gated on PROBE6=1, writes two
# sidecars beside probe.jsonl and does NOT touch the jsonl schema (CORRECTIONS 18
# is the reason that matters -- a third format generation would need a new guard
# in every existing reducer):
#     coord_signs.npy   [n_rec, k]      int8 in {-1,0,+1}, k=20,000 FIXED coords
#     coord_idx.npy     [k]             int64, the tracked coordinate ids
#     tensor_signs.npy  [n_rec, T, 3]   int64 (n_neg, n_zero, n_total), EXACT
#     coord_meta.json                   n_records / n_tot / k / seed / n_tensors
# Coordinates map to (tensor, row) through `c59_row_premise.build_index`, the map
# that already passed 59.2's gates A0/A1/A2.
#
# VALIDATED AGAINST REAL TORCH BEFORE SUBMISSION, NOT AFTER.  The patch was applied
# to a copy of HF_patched.py and the INSERTED BLOCK ITSELF was exec'd against a
# synthetic ResNet18-shaped z on the ROG offload node (torch 2.6.0, numpy 2.4.6):
# `tests/test_probe6_block.py`, **26/26 pass**, including that the synthetic arm
# reproduces the real n_tot 11,173,962 exactly, that every record's signs equal
# sign(zall[idx]), that the per-tensor split is exact and reconstructs frac_neg to
# 1e-12, that PROBE6=0 writes nothing and sets no optimizer state, and that no
# .tmp file survives.
#
#   THE HARNESS CAUGHT A REAL BUG, AND IT WAS ONE THIS PROJECT HAD ALREADY FIXED
#   ONCE.  The first draft copied PATCH_PROBE5's `np.save(<name>.npy.tmp)` idiom.
#   numpy APPENDS '.npy' to a string path that does not already end in it, so the
#   file lands as `.npy.tmp.npy` and the next line's os.replace raises
#   FileNotFoundError -- exactly the fatal bug `patches/patch_probe5_fix.py`
#   repaired in cycle 47, which would have crashed every job ~5 epochs in after
#   burning 25% of its wallclock.  Fixed the same way (hand numpy an open file
#   OBJECT) and the fix is commented in-place so it is not "simplified" back.
#
# THE OPERATING POINT IS `mx_sig_*`, RE-DERIVED FROM results/all_runs.csv AND NOT
# QUOTED FROM PROSE (STANDING RULE 1).  All 15 mx_sig rows read: SGDm base
# (momentum 0.99, wd 0.1) / Lion meta (momentum 0.99, beta2 0.9, wd 0), CIFAR10,
# ResNet18, batch 100, gamma 1, ms=1e-3, alpha0=1e-6, BETA_CLIP=-15:-2.3026,
# **AUGMENT=1**, 100 epochs, PROBE=25 -> 2000 records.
#   NOTE CORRECTIONS 826 describes mx as "AdamW+Adam".  The CSV says SGDm+Lion on
#   all 15 rows.  The CSV wins (trust data over prose); that prose line is about
#   `PP-*`, not mx, and is flagged for the next tick.
#
# THE 8 JOBS, AND WHY EACH EXISTS.
#   4x cp9-w-a6-s{3,4}, cp9-w-a3-s{3,4}    WEIGHTWISE, the arm the question is
#                                          about.  Two alpha0 rungs because
#                                          CORRECTIONS 12 measures a **16x**
#                                          collapse in weightwise agreement
#                                          between a0=1e-6 and a0=1e-3, so a
#                                          single rung cannot speak for the arm.
#   4x cp9-lay-a6-s{3,4}, cp9-lay-a3-s{3,4} LAYERWISE.  Costs nothing extra and
#                                          `tensor_signs` there is a FREE exact
#                                          replication of the tensor-level result
#                                          the kill-test already has, i.e. an
#                                          internal control on the new instrument
#                                          against a published number.
#   Seeds 3,4 throughout: mx_sig owns seeds 0,1,2 at every granularity, so there
#   is NO collision and these runs are poolable with it.
#
# PRE-REGISTERED DECISION RULE -- copied from KILLTEST-idea2.md section 5, written
# BEFORE the data exists, and NOT to be edited afterwards.  Measure within-tensor
# and across-tensor pairwise sign agreement against the CIRCULAR-SHIFT null (the
# marginal-preserving null of KILLTEST section 3, which is what killed the raw
# across-block excess):
#   K-A  across-tensor dependence <= 0.1 pp above the null  ->  **IDEA 2 IS DEAD
#        OUTRIGHT.**  The sign-agreement statistic belongs in the measurement
#        paper as a characterisation, never as a method.  This is the outcome the
#        tensor-level result predicts and it is registered as the expected one.
#   K-B  > ~0.5 pp AND surviving pairs NOT concentrated at |delta block| = 1  ->
#        Idea 2 has a target and differentiator (ii) survives.
#   K-C  anything between, or concentrated at |delta block| = 1  ->  UNDECIDED.
#        Write UNDECIDED.  Do not round it to a verdict (STANDING RULE 3).
#   A DIRECTION IS REGISTERED DELIBERATELY (the FINDINGS 55.6 convention): the
#   tensor-level measurement points hard at K-A, so a K-A result is a
#   REPLICATION at finer granularity and NOT a discovery, and a K-B result is
#   visibly an overturning of the campaign's own read.
#
# SEPARATELY, AND IT IS THE PART THAT SERVES 89.7: `coord_signs` at the
# weightwise arm gives a per-coordinate sign TIME SERIES, so within-tensor row
# structure can be dated.  That is exploratory, is NOT covered by the decision
# rule above, and must be labelled POST-HOC wherever it is reported.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. epochs_done == 100 == epochs_requested on EVERY job; drop, never rescale.
#   2. coord_meta.json n_records == 2000 on every arm (a short run silently
#      truncates the series exactly as FINDINGS 57.1's truncation did).
#   3. python3 analysis/c61_z_timecourse.py --selftest   (46/46)
#   4. tensor_signs must reconstruct probe.jsonl's frac_neg to 1e-12 on every
#      record -- a free cross-check of the new instrument against the old one.
#      If it fails, the instrument is wrong and nothing else may be read.
#   5. Re-derive every number from results/all_runs.csv, never from this comment.
#
# USAGE
#   bash c61_coord_probe.sh            # dry run: prints every sbatch line
#   bash c61_coord_probe.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

WS=/cluster/account_a/metaopt
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/cp9
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_a
PEND_CAP=40
NJOBS=8
CLIP=-15:-2.3026
EPOCHS=100
WALL=03:50:00
PROBE_EVERY=25
K=20000
WRITE_EVERY=200
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c61 cp9 -- PER-COORDINATE SIGN PROBE (cluster_a) ==="
echo "    weightwise x {a0=1e-6, 1e-3} x s{3,4} + layerwise x same = $NJOBS jobs, ${EPOCHS}ep"
echo "    registered expectation: K-A (across-tensor at the null) => Idea 2 dead outright"
echo

# GUARD 1 -- runner and optimizer source exist and compile.
[ -r "$RUNNER" ] || { echo "GUARD FAIL: $RUNNER not readable"; exit 1; }
[ -r "$HF" ]     || { echo "GUARD FAIL: $HF not readable"; exit 1; }
python3 -c "import py_compile; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 1: runner readable, HF.py readable and compiles  OK"

# GUARD 2 -- THE PATCH MUST BE PRESENT, AND IT MUST BE THE FIXED FORM.
# Without PROBE6 the batch runs to completion and produces NOTHING this cycle
# wants, at full wallclock cost.  This is the c49 guard-1 pattern.
grep -q PATCH_PROBE6 "$HF" || {
  echo "GUARD FAIL: PATCH_PROBE6 absent from $HF."
  echo "            Apply it first:  python3 patches/patch_probe6_coord.py"; exit 1; }
echo "guard 2: PATCH_PROBE6 present  OK"

# GUARD 2b -- the numpy-extension bug must NOT be present in the applied block.
# patch_probe6_coord.py ships the fixed form, but the file on the cluster may be
# an older copy.  Assert the fix by its signature rather than trusting the marker.
if grep -A4 'PATCH_PROBE6' "$HF" >/dev/null 2>&1 && \
   ! grep -q "with open(_t6, 'wb') as _fh6b" "$HF"; then
  echo "GUARD FAIL: the applied PATCH_PROBE6 writes via a STRING path ending in"
  echo "            '.npy.tmp'.  numpy appends '.npy', so os.replace raises"
  echo "            FileNotFoundError at the first dump -- the cycle-47"
  echo "            patch_probe5_fix.py bug.  Re-apply the fixed patch."; exit 1
fi
echo "guard 2b: PATCH_PROBE6 uses the file-object write (cycle-47 bug absent)  OK"

# GUARD 2c -- the block must be INERT unless PROBE6=1, so this batch cannot
# perturb any other running job's numerics.
grep -q "os.environ.get('PROBE6', '') == '1'" "$HF" || {
  echo "GUARD FAIL: PATCH_PROBE6 is not gated on PROBE6=1"; exit 1; }
echo "guard 2c: PATCH_PROBE6 is env-gated (inert without PROBE6=1)  OK"

# GUARD 3 -- THE OPERATING POINT IS DIFFED AGAINST THE CORPUS, NOT ASSERTED.
# H-pooling with mx_sig_* is only valid if this batch sits at mx's exact point.
# Read it from the CSV (STANDING RULE 1), not from the comment above.
python3 - "$CLIP" "$EPOCHS" <<'PYEOF' || exit 1
import csv, os, sys
clip, epochs = sys.argv[1], int(sys.argv[2])
here = os.path.dirname(os.path.abspath(__file__))
csvp = os.path.join(here, "..", "results", "all_runs.csv")
rows = [r for r in csv.DictReader(open(csvp)) if r["run"].startswith("mx_sig_")]
assert len(rows) == 15, f"expected 15 mx_sig rows, found {len(rows)}"
def one(k):
    v = {r[k] for r in rows}
    assert len(v) == 1, f"mx_sig is not homogeneous in {k}: {v}"
    return v.pop()
assert one("base") == "SGDm", one("base")
assert one("meta") == "Lion", one("meta")
assert one("meta_stepsize") == "1e-3", one("meta_stepsize")
assert one("alpha0") == "1e-6", one("alpha0")
assert one("beta_clip") == clip, f'mx_sig clip {one("beta_clip")} != {clip}'
assert one("augment") == "1", "mx_sig is NOT augmented -- do not match it"
assert one("network") == "ResNet18" and one("dataset") == "CIFAR10"
assert int(one("epochs_requested")) == epochs
# seeds 0,1,2 are taken at every granularity; this batch must use 3,4.
seeds = {r["run"].rsplit("_s", 1)[1] for r in rows}
assert seeds == {"0", "1", "2"}, f"mx_sig seeds are {seeds}, not 0/1/2"
print("guard 3: operating point matches mx_sig on 9 fields; mx_sig owns seeds "
      "0/1/2 so cp9's 3/4 cannot collide  OK")
PYEOF

# GUARD 4 -- the design arithmetic, re-run rather than trusted.
python3 - "$PROBE_EVERY" "$EPOCHS" "$K" "$WRITE_EVERY" <<'PYEOF' || exit 1
import sys
every, epochs, k, we = (int(x) for x in sys.argv[1:5])
steps = epochs * 500                      # CIFAR10 / batch 100 -> 500 steps/epoch
recs = steps // every
assert recs == 2000, f"probe would write {recs} records, not the mx-matched 2000"
# storage, per weightwise run
coord_mb = recs * k / 1e6                 # int8
tens_mb = recs * 62 * 3 * 8 / 1e6         # int64
assert coord_mb < 60, f"coord_signs would be {coord_mb:.0f} MB"
# I/O: the whole buffer is rewritten at each dump, so cost is quadratic in dumps
dumps = recs // we
io_gb = sum((i + 1) * we * k for i in range(dumps)) / 1e9
assert io_gb < 1.0, f"rewrite I/O would be {io_gb:.2f} GB per run -- raise WRITE_EVERY"
# RAM held by the in-process buffer
assert recs * k / 1e6 < 60, "the in-RAM sign buffer is too large"
print(f"guard 4: {recs} records (mx-matched); coord_signs {coord_mb:.0f} MB + "
      f"tensor_signs {tens_mb:.1f} MB per run; {dumps} dumps -> {io_gb:.2f} GB "
      f"rewrite I/O; total batch < {8 * (coord_mb + tens_mb) / 1000:.2f} GB  OK")
PYEOF

# GUARD 5 -- queue depth.  8 jobs is inside the ~20-job / 0.5%-RawUsage rule.
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

# GUARD 6 -- the wall-clock request must fit the TIGHTEST partition in $PARTS.
# CORRECTIONS 71: sbatch rejects against the SMALLEST MaxTime among the requested
# partitions, so one over-long --time loses the WHOLE batch.  Read live.
python3 - "$WALL" "$PARTS" <<'PYEOF' || exit 1
import subprocess, sys
def secs(t):
    d, _, rest = t.partition("-")
    if not rest:
        rest, d = d, "0"
    p = [int(x) for x in rest.split(":")]
    while len(p) < 3:
        p.insert(0, 0)
    return int(d) * 86400 + p[0] * 3600 + p[1] * 60 + p[2]
want, parts = sys.argv[1], sys.argv[2].split(",")
try:
    out = subprocess.run(["sinfo", "-h", "-o", "%P %l"], capture_output=True,
                         text=True, timeout=30).stdout
except Exception as e:
    print(f"guard 6: sinfo unavailable ({e}) -- SKIPPED, re-run at submit time")
    sys.exit(0)
lim = {}
for line in out.strip().splitlines():
    if not line.strip():
        continue
    name, _, tl = line.partition(" ")
    name = name.rstrip("*").strip()
    tl = tl.strip()
    if name in parts and tl not in ("infinite", "n/a", ""):
        lim[name] = secs(tl)
if not lim:
    print("guard 6: no finite MaxTime found for the requested partitions -- SKIPPED")
    sys.exit(0)
tight = min(lim, key=lambda k: lim[k])
if secs(want) > lim[tight]:
    print(f"GUARD FAIL: --time={want} exceeds {tight}'s MaxTime "
          f"({lim[tight]}s).  sbatch would reject the WHOLE batch.")
    sys.exit(1)
print(f"guard 6: --time={want} fits the tightest partition {tight} "
      f"({lim[tight]}s)  OK")
PYEOF

mkdir -p "$SAVE" 2>/dev/null || true

N=0
for G in weightwise layerwise; do
  case $G in
    weightwise) TAG=w   ;;
    layerwise)  TAG=lay ;;
  esac
  for A in 1e-6:a6 1e-3:a3; do
    AV=${A%:*}; AT=${A#*:}
    for S in 3 4; do
      RN="cp9-${TAG}-${AT}-s${S}"
      PD="$SAVE/probe_${TAG}_${AT}_s${S}"
      CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time="$WALL"
        --export="ALL,AUGMENT=1,BETA_CLIP=$CLIP,PROBE=$PROBE_EVERY,PROBE6=1,PROBE6_K=$K,PROBE6_WRITE_EVERY=$WRITE_EVERY,PROBE6_SEED=0,PROBE_DIR=$PD"
        "$RUNNER"
        --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
        --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
        --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
        --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 "$AV"
        --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$S"
        --save-directory "$SAVE" --run-name "$RN")
      N=$((N+1))
      if [ "$SUBMIT" = "1" ]; then "${CMD[@]}" >/dev/null; else printf '%s\n' "${CMD[*]}"; fi
    done
  done
done

echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo SUBMITTED || echo 'dry run, nothing submitted' )) ----"
[ "$N" = "$NJOBS" ] || { echo "WARNING: emitted $N jobs, expected $NJOBS"; exit 1; }
echo "PROBE6_SEED is pinned to 0 on every job, so all 8 runs track the SAME"
echo "coordinate ids -- cross-run comparison at coordinate granularity is only"
echo "meaningful because of that.  Do not vary it."
echo
echo "reduce with:  python3 analysis/killtest_idea2.py <probe root>   (after"
echo "              extending it to read coord_signs.npy -- it currently reads"
echo "              only the tensor-level fields)"
echo "cross-check:  tensor_signs must reproduce probe.jsonl frac_neg to 1e-12"
