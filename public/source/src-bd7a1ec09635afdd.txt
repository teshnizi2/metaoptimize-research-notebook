#!/usr/bin/env python3
"""VGG carrier NOMINATION (a HYPOTHESIS, not a result) -- ctd1's convention,
re-implemented independently: L_i = 0.9*m_tensor_i + 0.1*z_tensor_i (Lion
beta1=0.9), ranked on the SCALAR arm's PINNED records (beta[0] at the -15
clamp), exactly as ctd1 did on ResNet.  Reads cvg1's own probe.jsonl only."""
import json, os, re, sys, collections

RUNS = sys.argv[1] if len(sys.argv) > 1 else "/cluster/account_b/metaopt/runs"
CLAMP = -15.0
man = open(os.path.join(RUNS, "cvg1-PARTITION-MANIFEST.txt")).read()
names = {int(i): n for i, n, s in re.findall(r"^TENSOR (\d+) (\S+) (\d+)$", man, re.M)}
sizes = {int(i): int(s) for i, n, s in re.findall(r"^TENSOR (\d+) (\S+) (\d+)$", man, re.M)}

for arm in ("k01", "kL"):
    print(f"\n################ ARM {arm} ################")
    agg = collections.defaultdict(list)      # tensor idx -> list of mean|L| per seed
    share_pos = collections.defaultdict(list)
    for seed in ("s31", "s32", "s33"):
        p = os.path.join(RUNS, "cvg1", f"probe_cvg1-{arm}-{seed}", "probe.jsonl")
        if not os.path.exists(p):
            print(f"  {seed}: MISSING"); continue
        recs = [json.loads(l) for l in open(p)]
        pinned = [r for r in recs if min(r["beta"]) <= CLAMP + 1e-9]
        use = pinned if pinned else recs
        tag = "PINNED" if pinned else "ALL (no pinned record exists)"
        absL = [0.0] * 27
        npos = [0] * 27
        for r in use:
            z, m = r["z_tensor"], r["m_tensor"]
            for i in range(26):
                L = 0.9 * m[i] + 0.1 * z[i]
                absL[i + 1] += abs(L)
                if L > 0: npos[i + 1] += 1
        n = len(use)
        print(f"  {seed}: {len(recs)} records, {len(pinned)} pinned -> ranking on {n} {tag}")
        for i in range(1, 27):
            agg[i].append(absL[i] / n)
            share_pos[i].append(npos[i] / n)
    if not agg: continue
    tot = sum(sum(v) / len(v) for v in agg.values())
    rows = sorted(agg, key=lambda i: -sum(agg[i]) / len(agg[i]))
    print(f"  {'rank':>4} {'idx':>4} {'name':<16}{'numel':>9}{'mean|L_i|':>14}{'share of sum':>14}{'frac L>0':>10}")
    for r, i in enumerate(rows, 1):
        mv = sum(agg[i]) / len(agg[i])
        print(f"  {r:>4} {i:>4} {names[i]:<16}{sizes[i]:>9}{mv:>14.6e}{mv/tot:>14.4f}"
              f"{sum(share_pos[i])/len(share_pos[i]):>10.3f}")
    top4 = set(rows[:4])
    print(f"  TOP-4 = {sorted(top4)} -> names {[names[i] for i in sorted(top4)]}")
    print(f"  Is TOP-4 exactly the four 512-wide BN scales {{14,17,20,23}}? "
          f"{'YES' if top4 == {14,17,20,23} else 'NO'}")
