#!/bin/bash
# =============================================================================
# cVT4_betahold.sh -- `cvt4`: IS TENSOR 50's OWN LARGE STEP SIZE THE LEVER?  ONE SUBMISSION, 18 JOBS
#   (6 arms x 3 seeds), 100 EPOCHS, SEEDS {90,91,92}, PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1
#   ON EVERY ARM.  CORRECTIONS 237.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here touches the other account;
# every guard and scancel line is filtered on `cvt4-`.
#
# REGISTERED SCORER: analysis/cVT4_betahold_score.py, committed BEFORE this batch is submitted (RULE 21).
# Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT4_betahold_score.py <runsdir>
#
# NEW HARNESS CODE, ISOLATED AND OPT-IN.  The batch runs from $WS/harness_cvt4/cifar10 = cvt1's tree
# (unchanged, re-verified by bin/cVT1_stage_harness.sh) + PATCH_BETAHOLD (patches/patch_betahold.py) in HF.py,
# built and verified by bin/cVT4_stage_harness.sh, and the runner $WS/jobs/run_cifar_cvt4.sh (run_cifar.sh with
# the cd line changed).  BETA_HOLD unset == bitwise the unpatched tree; guard 4f requires the inertness / bite log
# (tests/test_betahold.py + tests/test_betahold_realrun.py, bin/cVT4_realrun_inertness.sbatch) on THIS HF.py and
# THIS scorer.
#
# THE ARMS.  PlainNet indices (53 tensors); guard 4 re-derives every one on the LIVE model by NAME, constructs
# each arm's HF with its BETA_HOLD, checks the witness, and writes the manifest.
#   k01         scalar                                BETA_HOLD unset
#   HEAD        sets:1-49,51-53/layer4.1.bn2.weight   BETA_HOLD unset
#   HOLDLOW     HEAD's   BETA_HOLD=layer4.1.bn2.weight:floor
#   HOLDSHARED  HEAD's   BETA_HOLD=layer4.1.bn2.weight:shared
#   HOLDHIGH    HEAD's   BETA_HOLD=layer4.1.bn2.weight:tri:9428
#   HOLDHEAD    HEAD's   BETA_HOLD=layer4.1.bn2.weight:tri:5041
#
# RULE 20.  Every ARGS value is ONE shell token; every held arm's ARGS line is HEAD's except the run name.
# BETA_HOLD rides sbatch's --export list (value class [A-Za-z0-9_.:], no comma) and CANNOT ride the runner's ENV
# line: the patch prints `BETA_HOLD: off` / `BETA_HOLD: on ...` at construction on EVERY run, and that line is what
# the separate ENV audit and the scorer's G-HOLDW gate read.  analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  cvt1 0.692, cvt3 0.674, cvt2 0.757 GPU-h per PlainNet run (sacct).  18 jobs ~= 12.4 GPU-h expected;
# HARD BOUND = WALL x 18 = 54 GPU-h (WALL 03:00:00 = x2 cvt2's slowest run, 5,344 s).
#
# USAGE   bash bin/cVT4_betahold.sh               # dry run: every guard, every composed line printed
#         bash bin/cVT4_betahold.sh --dry-run     #   submits nothing
#         bash bin/cVT4_betahold.sh --submit      # the ONLY submitting form
# ENV     CVT4_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# the control arms must NOT inherit a BETA_HOLD (or a VOTE_W) through --export=ALL
unset VOTE_W BETA_HOLD
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cvt4/cifar10
HARN1=$WS/harness_cvt1/cifar10
RUNNER=$WS/jobs/run_cifar_cvt4.sh
SAVE=$WS/runs/cvt4

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=PlainNet18_c100
SEEDS="90 91 92"
MST=1e-3
A0=1e-6
NTENS=53
TOTPAR=11046308
G_K01='scalar'
G_HEAD='sets:1-49,51-53/layer4.1.bn2.weight'
B_HOLDLOW='layer4.1.bn2.weight:floor'
B_HOLDSHARED='layer4.1.bn2.weight:shared'
B_HOLDHIGH='layer4.1.bn2.weight:tri:9428'
B_HOLDHEAD='layer4.1.bn2.weight:tri:5041'
ARMS="k01 HEAD HOLDLOW HOLDSHARED HOLDHIGH HOLDHEAD"
NJOBS=18
WALL=03:00:00
POST_SHA=e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_PRE_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_POST_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
RUNNER_SHA=c801fc85115f124d82186078a722904a0e8189b94eff44e0daf0e6a7e1c7145d
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT4_betahold_score.py"
SELF="$REPO/bin/cVT4_betahold.sh"
DERIV="$REPO/analysis/cvt4_registration_derivations.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cVT4_realrun_inertness.sbatch"
TESTB="$REPO/tests/test_betahold.py"
TESTBR="$REPO/tests/test_betahold_realrun.py"
STAGE4="$REPO/bin/cVT4_stage_harness.sh"
STAGE1="$REPO/bin/cVT1_stage_harness.sh"
PATCHB="$REPO/patches/patch_betahold.py"
PATCHV="$REPO/patches/patch_voteweight.py"
PATCHF="$REPO/patches/patch_plainnet.py"
TESTV="$REPO/tests/test_voteweight.py"
TESTR="$REPO/tests/test_voteweight_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_plainnet.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFBH="$HARN/Optimizers/HF.py.pre_betahold"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"
ILOG="$SAVE/inertness_betahold.log"

echo "=============================================================="
echo " cvt4 -- is tensor 50's own large step size the lever?  (PATCH_BETAHOLD)"
echo " $DSET / $NET (PlainNet indices)   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01         $G_K01      BETA_HOLD unset"
echo "   HEAD        $G_HEAD      BETA_HOLD unset"
echo "   HOLDLOW     HEAD's      BETA_HOLD=$B_HOLDLOW"
echo "   HOLDSHARED  HEAD's      BETA_HOLD=$B_HOLDSHARED"
echo "   HOLDHIGH    HEAD's      BETA_HOLD=$B_HOLDHIGH"
echo "   HOLDHEAD    HEAD's      BETA_HOLD=$B_HOLDHEAD"
echo " 6 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cvt1's + PATCH_BETAHOLD)   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " CO-PRIMARY: P_HIGH = HEAD - HOLDHIGH; P_COUP = HOLDHIGH - HOLDSHARED; P_LOW = HOLDLOW - HEAD"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT4_betahold_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && echo "guard 0b: VOTE_W and BETA_HOLD are UNSET in the launcher's environment (control arms cannot inherit them)" \
                     || guard_fail "guard 0b: VOTE_W or BETA_HOLD is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$SELF" "$DERIV" "$EXCL" "$EXCLT" "$RRSB" "$TESTB" "$TESTBR" "$STAGE4" "$STAGE1" "$PATCHB" "$PATCHV" "$PATCHF" \
         "$TESTV" "$TESTR" "$TESTT" "$TESTF" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$HFBH" "$HFVW" "$HFPRE" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cVT4_betahold_score.py bin/cVT4_betahold.sh analysis/cvt4_registration_derivations.py bin/cVT4_realrun_inertness.sbatch tests/test_betahold.py tests/test_betahold_realrun.py patches/patch_betahold.py bin/cVT4_stage_harness.sh"
REUSEF="analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv tests/test_voteweight_realrun.py tests/test_voteweight.py tests/test_probe_tensor.py tests/test_plainnet.py patches/patch_voteweight.py patches/patch_plainnet.py bin/cVT1_stage_harness.sh bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the eight registration files and eleven reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT4_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT4_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT4_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT4_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING O2, RULE P, the real-record null and the invocation ----
SELFT=/tmp/cvt4_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
grep -qE "^  PASS O2 OWN-STEP-MAGNITUDE +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt4's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT4_betahold_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 90, 91 or 92" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cVT1's" "$SELFT" && grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cVT3's" "$SELFT" \
  && grep -q "^  PASS HEAD's spec is BYTE-IDENTICAL to cPL2's" "$SELFT" \
  && echo "guard 1c5: HEAD is byte-identical to cvt1's, cvt3's and cpl2's HEAD" || guard_fail "guard 1c5: HEAD not proved byte-identical"
grep -q "^  PASS B RULE P re-run on THIS host's landed records == P_HIGH 9428, P_HEAD 5041" "$SELFT" \
  && echo "guard 1c6: RULE P re-derived P_HIGH 9428 and P_HEAD 5041 from the landed records ON THIS HOST" \
  || guard_fail "guard 1c6: RULE P was not re-derived on this host"
grep -q "^  PASS B G-BITE's Lion half PASSES on all 18 landed" "$SELFT" && grep -q "^  PASS B THE NULL ON REAL RECORDS" "$SELFT" \
  && grep -q "^  PASS B non-vacuity" "$SELFT" \
  && echo "guard 1c7: G-BITE passes 18 landed controls and FAILS all 36 real-record nulls on this host; non-vacuity re-derived" \
  || guard_fail "guard 1c7: the real-record G-BITE checks did not pass"
grep -q "^  PASS SIGMA_PLAIN: the FROZEN literal 0.45826358375406856 IS the FILTERED value" "$SELFT" \
  && grep -q "^  PASS the NAIVE PlainNet floor is" "$SELFT" \
  && echo "guard 1c8: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value, and NOT the naive one" \
  || guard_fail "guard 1c8: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E BROKEN PATCH: every held arm UNHELD" "$SELFT" && grep -q "^  PASS E HOLD ONLY AT PROBE STEPS" "$SELFT" \
  && grep -q "^  PASS E WRONG P" "$SELFT" && grep -q "^  PASS E COMPLEMENT TOUCHED" "$SELFT" \
  && echo "guard 1c9: the broken-patch null, a probe-step-only hold, a wrong P and a touched complement reach PATCH-NOT-VERIFIED" \
  || guard_fail "guard 1c9: G-BITE break cases did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 1d: the derivation reproduces RULE P and the floor on THIS host ----------
DOUT=/tmp/cvt4_deriv.$$
python3 "$DERIV" "$WS/runs" "$CSVP" >"$DOUT" 2>&1 \
  && grep -q "^  REGISTERED: P_HIGH 9428 (HOLDHIGH), P_HEAD 5041 (HOLDHEAD)." "$DOUT" \
  && grep -q "SIGMA_PLAIN   FILTERED 0.4582635837540" "$DOUT" \
  && echo "guard 1d: analysis/cvt4_registration_derivations.py reproduces RULE P and the filtered floor on this host (output sha $(sha "$DOUT"))" \
  || { guard_fail "guard 1d: the derivation did not reproduce RULE P / the floor"; tail -5 "$DOUT"; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt4-' "$CSVP" && guard_fail "guard 2: cvt4-* already in the CSV" || echo "guard 2: no cvt4-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
gn = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], gn))
raise SystemExit(1 if hit or not set(gn) <= {"cpl1", "cpl2", "cvt1", "cvt2", "cvt3"} else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt4-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt4-*.out on disk" || guard_fail "guard 2b: $NOUT cvt4-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (90|91|92)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 90/91/92" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 90/91/92"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt4-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt4-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt4-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt4-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt4-* in squeue" || guard_fail "guard 2f: $NQ cvt4-* already queued"
fi

# ---- guard 3/4a: cvt1's tree unchanged; cvt4's tree consistent and sees CIFAR-100 -----------
bash "$STAGE1" >/tmp/cvt4_stage1.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT1_stage_harness.sh (verify mode) VERIFIES cvt1's tree (the base, unchanged):"; grep -E 'V-h2|V-b2|VERIFIED' /tmp/cvt4_stage1.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt1's tree is NOT consistent"; tail -20 /tmp/cvt4_stage1.$$; }
bash "$STAGE4" >/tmp/cvt4_stage4.$$ 2>&1 \
  && { echo "guard 4a2: bin/cVT4_stage_harness.sh (verify mode) VERIFIES cvt4's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt4_stage4.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a2: cvt4's tree is NOT consistent"; tail -20 /tmp/cvt4_stage4.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "$POST_SHA" ] && echo "guard 3b: build_network.py == cpl1's POST_SHA e65e6773...a18b" \
  || guard_fail "guard 3b: build_network.py is not cpl1's POST_SHA"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFBH")" = "$HF_PRE_SHA" ] && [ "$(sha "$HARN1/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && [ "$(sha "$HFVW")" = "$HF_LIVE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA 84b345ad...; .pre_betahold == cvt1's HF.py == 3f2b98e1...; .pre_voteweight == the LIVE HF.py == 4732b74a...; .pre_probe_tensor == live" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or cvt1's / the live HF.py moved)"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == c801fc85... (run_cifar.sh with the cd line changed)" \
  || guard_fail "guard 3e: $RUNNER is not the pinned cvt4 runner"
grep -q PATCH_BETAHOLD "$LIVE/Optimizers/HF.py" "$HARN1/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE or cvt1's HF.py carries PATCH_BETAHOLD" \
  || echo "guard 3d: neither the LIVE HF.py nor cvt1's carries PATCH_BETAHOLD"

# ---- guard 4b/4f: the optimiser's patches, proved inert; the new patch proved inert and biting ----
python3 "$TESTT" --pre "$HFPRE" --post "$HFVW" --structural-only >/tmp/cvt4_struct.$$ 2>&1 \
   && echo "guard 4b: PATCH_PROBE_TENSOR structural inertness R0 PASSES on HF.py.pre_voteweight (= live)" \
   || { guard_fail "guard 4b: PATCH_PROBE_TENSOR structural R0 FAILED"; tail -8 /tmp/cvt4_struct.$$; }
python3 "$TESTB" --pre "$HFBH" --post "$HFH" --structural-only >/tmp/cvt4_bh0.$$ 2>&1 \
   && echo "guard 4b2: PATCH_BETAHOLD structural B0 PASSES (deleting the four regions == cvt1's HF.py, byte for byte)" \
   || { guard_fail "guard 4b2: PATCH_BETAHOLD structural B0 FAILED"; tail -8 /tmp/cvt4_bh0.$$; }
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH"); SCSHA=$(sha "$SC")
if [ -f "$ILOG" ] && grep -q "^TEST_BETAHOLD_RC 0$" "$ILOG" && grep -q "^REALRUN_RC 0$" "$ILOG" \
   && [ "$(grep -c '^ALL PASS$' "$ILOG")" = 2 ] && [ "$(grep -c '^  FAIL' "$ILOG")" = 0 ] \
   && grep -q "^TREE_POST $HARN HF $HFSHA build_network $BNSHA" "$ILOG" && grep -q "^SCORER_SHA256 $SCSHA$" "$ILOG" \
   && grep -q "^DRIVER_SHA256 $(sha "$TESTBR")$" "$ILOG" && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTR")$" "$ILOG" \
   && grep -q "^$(sha "$TESTB")  tests/test_betahold.py$" "$ILOG" && grep -q "^$(sha "$PATCHB")  patches/patch_betahold.py$" "$ILOG" \
   && grep -q "PASS RR0 unpatched k01 vs itself: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 HEAD BETA_HOLD unset: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 k01  BETA_HOLD empty: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR2 HOLDHIGH   beta\[1\] after EVERY one" "$ILOG" && grep -q "PASS RR3 HOLDLOW's final parameters differ" "$ILOG" \
   && grep -q "PASS RR4 beta IDENTICAL at every step" "$ILOG" && grep -q "PASS B4 HOLDSHARED the WHOLE momentum_meta vector" "$ILOG"; then
  echo "guard 4f: the inertness / bite log (tests/test_betahold.py + tests/test_betahold_realrun.py) ALL PASS on THIS HF, THIS scorer: $(grep -c '^  PASS' "$ILOG") PASS / $(grep -c '^  FAIL' "$ILOG") FAIL (sha $(sha "$ILOG" | cut -c1-8))"
else
  guard_fail "guard 4f: no ALL PASS inertness log for HF $HFSHA and scorer $SCSHA at $ILOG -- sbatch $RRSB, then copy its .out there"
fi

# ---- guard 4c': the launcher's literals == the scorer's (imported, not regexed) ------
python3 - "$SC" "$G_K01" "$G_HEAD" "$B_HOLDLOW" "$B_HOLDSHARED" "$B_HOLDHIGH" "$B_HOLDHEAD" \
          "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$POST_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$RUNNER_SHA" "$STAGE4" "$ARMS" <<'PYEOF' \
