def kspec(k):
    return "[%d,%d]" % (k, N_TENSORS - k)


def arm_spec(arm):
    return kspec(int(arm[1:]))


# =============================================================================
# the corpus

def _csv_rows(path=CSV):
    with open(path) as fh:
        return [r for r in csv.DictReader(fh) if (r.get("superseded") or "0") != "1"]


def _batch_of(run):
    return re.split(r"[-_]", str(run))[0]


def _in_cell(r):
    """The cts1/cpk1 cell, EXACTLY."""
    return (r["network"] == NET and r["dataset"] == DSET
            and r["base"] == BASE_ALG and r["meta"] == META_ALG
            and r["meta_stepsize"] == MS and r["alpha0"] == ALPHA0
            and r["gamma"] == GAMMA and r["augment"] == AUG
            and r["beta_clip"] == CLIP and not (r.get("hier") or "").strip()
            and r["batch_size"] == str(BATCH) and r["epochs_done"] == str(EPOCHS)
            and r["collapsed"] == "0")


def noise_floor_from_csv(path=CSV):
    """Pooled WITHIN (batch x granularity) SD of plateau5, m = 2 arms only."""
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not _in_cell(r) or not M2_RE.match(str(r["granularity"] or "")):
            continue
        try:
            cells[(_batch_of(r["run"]), r["granularity"])].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc


def _prefix_means(prefix, path=CSV):
    g = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not str(r.get("run") or "").startswith(prefix):
            continue
        try:
            g[r["granularity"]].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    return g


def cpk1_cliff2_from_csv(path=CSV):
    """cpk1's own k52 - k55 on plateau5, in pp.  DESCRIPTIVE."""
    g = _prefix_means(CPK1_PREFIX, path)
    a, b = kspec(52), kspec(55)
    if a not in g or b not in g:
        return None
    ma, mb = statistics.mean(g[a]), statistics.mean(g[b])
    c1 = None
    if kspec(49) in g:
        c1 = statistics.mean(g[kspec(49)]) - ma
    return (ma, mb, ma - mb, len(g[a]), len(g[b]), c1)


def cts1_gap_from_csv(path=CSV):
    """cts1's own in-batch layerwise - scalar GAP, in pp.  DESCRIPTIVE."""
    g = _prefix_means(CTS1_PREFIX, path)
    if "scalar" not in g or "layerwise" not in g:
        return None
    lo, hi = statistics.mean(g["scalar"]), statistics.mean(g["layerwise"])
    return (lo, hi, hi - lo)


def premise_no_scl1_rows(path=CSV):
    return sum(1 for r in _csv_rows(path)
               if str(r.get("run") or "").startswith(PREFIX))


# =============================================================================
# the SE arithmetic.  Each contrast gets the SE of ITS OWN linear combination.
# =============================================================================
def se_of_combo(coefs, sigma=SIGMA_W, n=3):
    """SE, in PERCENTAGE POINTS, of sum_i coefs[i]*mean(arm_i), n seeds each."""
    return sigma * math.sqrt(sum(c * c for c in coefs) / float(n))


SE_ARM_DIFF = se_of_combo((1, -1))                     # any DROP_i, and TOTAL
SE_ADJ_STEPS = se_of_combo((1, -2, 1))                 # DROP_i - DROP_{i+1}
SE_END_STEPS = se_of_combo((1, -1, -1, 1))             # DROP1 - DROP3
SE_EVEN_13 = se_of_combo((2 / 3.0, -1, 1 / 3.0))       # DROP1 - TOTAL/3
SE_EVEN_2 = se_of_combo((-1 / 3.0, 1, -1, 1 / 3.0))    # DROP2 - TOTAL/3

def _tokens(line):
    import shlex
    try:
        return shlex.split(line)
    except ValueError:
        return line.split()


def parse_args_line(line):
    """(effective_flags, repeated_flags).  argparse keeps the LAST occurrence."""
    toks = _tokens(line)
    eff, seen = {}, collections.Counter()
    i = 0
    while i < len(toks):
        t = toks[i]
        if t.startswith("--"):
            key = t[2:]
            val = "1"
            if i + 1 < len(toks) and not toks[i + 1].startswith("--"):
                val = toks[i + 1]
                i += 1
            seen[key] += 1
            eff[key] = val
        i += 1
    return eff, sorted(k for k, c in seen.items() if c > 1)


def parse_env_line(line):
    out = {}
    for tok in line.split():
        if "=" in tok:
            k, v = tok.split("=", 1)
            out[k] = v
    return out


def read_out(path):
    args = env = None
    test, train = {}, {}
    try:
        with open(path, "r", errors="replace") as fh:
            for line in fh:
                if args is None:
                    m = ARGS_RE.match(line)
                    if m:
                        args = m.group(1).strip()
                        continue
                if env is None:
                    m = ENV_RE.match(line)
                    if m:
                        env = m.group(1).strip()
                        continue
                m = EP_RE.search(line)
                if m:
                    test[int(m.group(1))] = float(m.group(2))
                m = EPTR_RE.search(line)
                if m:
                    train[int(m.group(1))] = float(m.group(2))
    except IOError:
        return None
    if args is None:
        return None
    eff, dup = parse_args_line(args)
    return {"args": eff, "dup": dup, "argsline": args,
            "env": parse_env_line(env or ""), "envline": env or "",
            "test": test, "train": train, "path": path}


def tail5(series, budget=EPOCHS, w=5):
    """PRIMARY window.  Mean of the LAST 5 epochs; train.py prints 0-indexed."""
    v = [series[e] for e in range(budget - w, budget) if e in series]
    return sum(v) / len(v) if len(v) == w else None


def norm_num(s):
    try:
        return "%g" % float(s)
    except (TypeError, ValueError):
        return None


def collect(runsdir):
    """Every scl1 .out file, with its arm MEASURED from its own ARGS line."""
    recs = []
    pats = [os.path.join(runsdir, PREFIX + "*.out"),
            os.path.join(runsdir, "**", PREFIX + "*.out")]
    seen = set()
    files = []
    for p in pats:
        for f in glob.glob(p, recursive=True):
            if f not in seen:
                seen.add(f)
                files.append(f)
    for p in sorted(files):
        base = os.path.basename(p)
        m = NAME_RE.match(base)
        r = read_out(p)
        if r is None:
            recs.append({"path": p, "base": base, "bad": "no ARGS line"})
            continue
        r["base"] = base
        r["job_id"] = m.group(3) if m else None
        r["name_arm"] = m.group(1) if m else None
        r["name_seed"] = int(m.group(2)) if m else None
        spec = r["args"].get("stepsize-groups")
        r["meas_arm"] = None
        for a in ARMS:
            if spec == arm_spec(a):
                r["meas_arm"] = a
        r["spec"] = spec
        r["clip"] = r["env"].get("BETA_CLIP")
        try:
            r["seed"] = int(r["args"].get("seed"))
        except (TypeError, ValueError):
            r["seed"] = None
        r["p5"] = tail5(r["test"])
        r["t5"] = tail5(r["train"])
        r["n_ep"] = len(r["test"])
        recs.append(r)
    return recs

def read_manifest(path):
    if not path or not os.path.isfile(path):
        return None
    out = {"specs": {}}
    for line in open(path):
        t = line.split()
        if not t:
            continue
        if t[0] == "NUM_PARAM_TENSORS":
            out["ntens"] = int(t[1])
        elif t[0] == "TOTAL_PARAMS":
            out["total"] = int(t[1])
        elif t[0] == "SPEC":
            d = {}
            i = 1
            d["spec"] = t[1]
            i = 2
            while i < len(t):
                key = t[i]
                if key in ("M", "SIZES", "PARAMS"):
                    d[key] = t[i + 1]
                    i += 2
                elif key in ("FIRSTGROUP", "LASTGROUP"):
                    j = i + 1
                    vals = []
                    while j < len(t) and t[j] not in ("FIRSTGROUP", "LASTGROUP",
                                                     "M", "SIZES", "PARAMS"):
                        vals.append(t[j])
                        j += 1
                    d[key] = vals
                    i = j
                else:
                    i += 1
            out["specs"][d["spec"]] = d
    return out

class Report(object):
    def __init__(self):
        self.fail = 0

    def ok(self, label, cond, extra=""):
        tag = "PASS" if cond else "FAIL"
        if not cond:
            self.fail += 1
        print("  [%s] %-64s %s" % (tag, label, extra))
        return bool(cond)

    def note(self, s):
        print("       %s" % s)


def fmt(x, w=9, p=4):
    return ("%*.*f" % (w, p, x)) if x is not None and x == x else " " * (w - 2) + "--"

def verdict(d1, d2, d3):
    """(cliff, split, tensor, mechanism, why).  R1 first, then R2, then R3."""
    if d1 is None or d2 is None or d3 is None:
        return ("UNRESOLVED-INCOMPLETE", "UNRESOLVED", "(none)", "UNRESOLVED",
                "an arm mean is missing")
    drops = (d1, d2, d3)
    total = d1 + d2 + d3
    # ---- R1, CHECKED FIRST ---------------------------------------------------
    if total < CLIFF_BAR:
        return ("CLIFF-NOT-REPRODUCED", "UNRESOLVED", "(none)", "UNRESOLVED",
                "TOTAL = %.4f pp is below CLIFF_BAR %.4f pp (half of cpk1's own "
                "%.4f pp).  There is no cliff here to decompose: NO TENSOR MAY "
                "BE NAMED, the deliverable is UNRESOLVED, and cpk1's or cts1's "
                "arms may NOT be spliced in to rescue it."
                % (total, CLIFF_BAR, CPK1_CLIFF2))
    # ---- R2 -----------------------------------------------------------------
    if all(abs(d - total / 3.0) <= EVEN_BAND for d in drops):
        return ("CLIFF-REPRODUCES", "SPLIT-EVEN", "(none)", "NOT-SHARED",
                "all three drops lie within EVEN_BAND %.4f pp of TOTAL/3 = "
                "%.4f pp.  The second cliff is NOT a single-tensor effect, no "
                "tensor may be named, B1 and B2 both fail, and cts1's "
                "single-tensor result is a ONE-SITE ACCIDENT: CORRECTIONS "
                "147.6's withdrawal STANDS."
                % (EVEN_BAND, total / 3.0))
    # ---- R3 -----------------------------------------------------------------
    order = sorted(range(3), key=lambda i: drops[i], reverse=True)
    lead, runner = order[0], order[1]
    if drops[lead] <= total / 2.0:
        return ("CLIFF-REPRODUCES", "NO-MAJORITY", "(none)", "NOT-SHARED",
                "the leader %s = %.4f pp does not exceed TOTAL/2 = %.4f pp.  "
                "Naming requires the CONJUNCTION of majority AND separation; "
                "no tensor may be named and CORRECTIONS 147.6's withdrawal "
                "STANDS." % (DROP_LABEL[lead], drops[lead], total / 2.0))
    if drops[lead] - drops[runner] <= SEPARATION_BAR:
        return ("CLIFF-REPRODUCES", "NO-MAJORITY", "(none)", "NOT-SHARED",
                "the leader %s = %.4f pp does not beat the runner-up %s = "
                "%.4f pp by SEPARATION_BAR %.4f pp.  No tensor may be named "
                "and CORRECTIONS 147.6's withdrawal STANDS."
                % (DROP_LABEL[lead], drops[lead], DROP_LABEL[runner],
                   drops[runner], SEPARATION_BAR))
    # ---- a tensor may be named ----------------------------------------------
    tensor = MOVES[lead][2]
    if lead == 0:
        mech = "SHARED-MECHANISM"
        why = ("the majority is DROP1, the 512-parameter normalisation SCALE "
               "`%s` leaving the fine group while its own convolution "
               "`layer4.0.shortcut.0.weight` is already coarse -- structurally "
               "the SAME step as cts1's winner `layer4.0.bn2.weight`, at a "
               "DIFFERENT site, in an INDEPENDENT submission.  'One tensor "
               "carries the step' is a PATTERN for this sub-class and "
               "CORRECTIONS 147.6's withdrawal was TOO CAUTIOUS.  B1 HELD, B2 "
               "REFUTED." % tensor)
    else:
        mech = "NOT-SHARED"
        why = ("the majority is %s, `%s` (%d parameters) -- a single tensor, "
               "but NOT the tensor class or structural relation that carried "
               "cliff 1.  The two cliffs do NOT share a mechanism; cts1's "
               "attribution is a ONE-SITE result and CORRECTIONS 147.6's "
               "withdrawal STANDS.  %s"
               % (DROP_LABEL[lead], tensor, MOVES[lead][3],
                  "B2 HELD, B1 REFUTED." if lead == 2
                  else "B1 and B2 BOTH REFUTED."))
    return ("CLIFF-REPRODUCES", "SINGLE-TENSOR-MAJORITY", tensor, mech, why)

def score(runsdir, manifest_path):
    R = Report()
    print("=" * 78)
    print("scl1 -- THE SECOND CLIFF, k = 52 -> k = 55, AT SINGLE-TENSOR")
    print("        RESOLUTION.  registered scorer, run unedited.")
    print("  runs      %s" % runsdir)
    print("  manifest  %s" % manifest_path)
    print("=" * 78)

    recs = collect(runsdir)
    print("\n-- G0 PROVENANCE -------------------------------------------------")
    R.ok("%d scl1 .out files" % N_RUNS, len(recs) == N_RUNS, "%d found" % len(recs))
    bad = [r for r in recs if r.get("bad")]
    R.ok("every .out has an ARGS line", not bad,
         ", ".join(r["base"] for r in bad) if bad else "")
    good = [r for r in recs if not r.get("bad")]
    dups = [r["base"] for r in good if r["dup"]]
    R.ok("no repeated flag on any ARGS line (RULE 20)", not dups, ", ".join(dups))
    jids = set(r["job_id"] for r in good if r["job_id"])
    R.ok("%d distinct job ids" % N_RUNS, len(jids) == N_RUNS, "%d" % len(jids))
    mism = [r["base"] for r in good
            if r["meas_arm"] != r["name_arm"] or r["seed"] != r["name_seed"]]
    R.ok("run NAME agrees with its own ARGS line", not mism, ", ".join(mism))
    wrongcfg = []
    for r in good:
        a = r["args"]
        for k, want in (("dataset", DSET), ("NN-name", NET),
                        ("alg-base", BASE_ALG), ("alg-meta", META_ALG),
                        ("gamma", GAMMA)):
            if a.get(k) != want:
                wrongcfg.append("%s %s=%s" % (r["base"], k, a.get(k)))
        for k, want in (("meta-stepsize", MS), ("alpha0", ALPHA0),
                        ("num-epochs", str(EPOCHS)), ("batch-size", str(BATCH))):
            if norm_num(a.get(k)) != norm_num(want):
                wrongcfg.append("%s %s=%s" % (r["base"], k, a.get(k)))
        if r["env"].get("BETA_CLIP") != CLIP:
            wrongcfg.append("%s BETA_CLIP=%s" % (r["base"], r["env"].get("BETA_CLIP")))
        if r["env"].get("AUGMENT") != AUG:
            wrongcfg.append("%s AUGMENT=%s" % (r["base"], r["env"].get("AUGMENT")))
        if r["env"].get("PROBE", "0") not in ("0", "none"):
            wrongcfg.append("%s PROBE=%s" % (r["base"], r["env"].get("PROBE")))
        if (r["env"].get("HIER") or "none") != "none":
            wrongcfg.append("%s HIER=%s" % (r["base"], r["env"].get("HIER")))
    R.ok("every run's own ARGS **and ENV** matches the registered cell",
         not wrongcfg, "; ".join(wrongcfg[:4]))
    got = collections.Counter(r["meas_arm"] for r in good)
    R.ok("all four arms have 3 seeds", all(got.get(a) == 3 for a in ARMS),
         str(dict(got)))
    envs = set(r["envline"] for r in good)
    R.ok("all 12 runs share ONE ENV line", len(envs) == 1,
         list(envs)[0] if len(envs) == 1 else "%d distinct" % len(envs))
    stripped = set()
    for r in good:
        toks = _tokens(r["argsline"])
        out, i = [], 0
        while i < len(toks):
            if toks[i] in ("--stepsize-groups", "--seed", "--run-name",
                           "--save-directory"):
                i += 2
                continue
            out.append(toks[i])
            i += 1
        stripped.add(" ".join(out))
    R.ok("all 12 ARGS lines collapse to ONE string off-axis", len(stripped) == 1,
         "%d distinct" % len(stripped))
    if R.fail:
        print("\nVOID -- G0 failed.  No number below may be reported.")
        return 2

    print("\n-- G1 THE LIVE-MODEL PARTITION MANIFEST --------------------------")
    man = read_manifest(manifest_path)
    R.ok("PARTITION-MANIFEST.txt is readable", man is not None, manifest_path or "")
    if man is None:
        print("\nVOID -- the partition was not proven on the live model.")
        return 2
    R.ok("the live model has %d parameter tensors" % N_TENSORS,
         man.get("ntens") == N_TENSORS, str(man.get("ntens")))
    R.ok("the live model has %d parameters" % TOTAL_PARAMS,
         man.get("total") == TOTAL_PARAMS, str(man.get("total")))
    tt = dict((i, (n, p)) for i, n, p in TENSOR_TABLE)
    manprob = []
    for k in KGRID:
        d = man["specs"].get(kspec(k))
        if d is None:
            manprob.append("%s absent" % kspec(k))
            continue
        if d.get("M") != "2":
            manprob.append("%s m=%s" % (kspec(k), d.get("M")))
        if d.get("SIZES") != "%d,%d" % (k, N_TENSORS - k):
            manprob.append("%s sizes %s" % (kspec(k), d.get("SIZES")))
        if d.get("PARAMS") != "%d,%d" % SPEC_PARAMS[k]:
            manprob.append("%s params %s" % (kspec(k), d.get("PARAMS")))
        fg, lg = d.get("FIRSTGROUP") or [], d.get("LASTGROUP") or []
        if len(fg) != k or len(lg) != N_TENSORS - k:
            manprob.append("%s group lengths %d/%d" % (kspec(k), len(fg), len(lg)))
        # the last coarse tensor and the first fine tensor, BY NAME
        if fg and k in tt and fg[-1] != tt[k][0]:
            manprob.append("%s coarse ends at %s, not %s" % (kspec(k), fg[-1], tt[k][0]))
        if lg and (k + 1) in tt and lg[0] != tt[k + 1][0]:
            manprob.append("%s fine starts at %s, not %s"
                           % (kspec(k), lg[0], tt[k + 1][0]))
    R.ok("every spec matches the frozen table in sizes, params AND names",
         not manprob, "; ".join(manprob[:4]))

    print("\n-- G1b EACH STEP MOVES EXACTLY ONE TENSOR, AND IT IS THE ONE -----")
    stepprob = []
    for a, b, nm, pc in MOVES:
        da, db = man["specs"].get(kspec(a)), man["specs"].get(kspec(b))
        if not da or not db:
            stepprob.append("k%d->k%d missing" % (a, b))
            continue
        fa, fb = set(da.get("LASTGROUP") or []), set(db.get("LASTGROUP") or [])
        moved = sorted(fa - fb)
        if moved != [nm]:
            stepprob.append("k%d->k%d moves %s" % (a, b, moved))
        pa = int(da["PARAMS"].split(",")[0])
        pb = int(db["PARAMS"].split(",")[0])
        if pb - pa != pc:
            stepprob.append("k%d->k%d moves %d params, not %d" % (a, b, pb - pa, pc))
    R.ok("all three steps move exactly the registered single tensor",
         not stepprob, "; ".join(stepprob[:3]))
    for a, b, nm, pc in MOVES:
        print("       k=%d -> k=%d   %-30s %9d params" % (a, b, nm, pc))
    print("       the three steps move %d params = %.4f %% of the fine group "
          "at k = 52" % (MOVED_PARAMS, MOVED_FRAC_PCT))
    print("       DISCLOSED: cts1's three steps moved 2.6931 %% of ITS fine "
          "group.  The two")
    print("       cliffs are homologous in tensor CLASS, not in moved MASS.")
    if R.fail:
        print("\nVOID -- the partition is not what was registered.")
        return 2

    print("\n-- G2 COMPLETENESS -----------------------------------------------")
    dead = []
    for r in good:
        if r["n_ep"] < EPOCHS:
            dead.append("%s: %d/%d epochs" % (r["base"], r["n_ep"], EPOCHS))
        elif r["p5"] is None or r["p5"] != r["p5"]:
            dead.append("%s: no plateau5" % r["base"])
        elif r["p5"] <= DEAD_BAR:
            dead.append("%s: plateau5 %.4f <= %.2f" % (r["base"], r["p5"], DEAD_BAR))
    R.ok("all 12 runs reached %d epochs and trained" % EPOCHS, not dead,
         "; ".join(dead[:3]))
    if dead:
        print("\nUNRESOLVED-INCOMPLETE -- at least one run did not finish or did "
              "not train.\nThis is a REGISTERED outcome and is NOT scored as "
              "'the cliff vanished'.")
        return 4

    arm = collections.defaultdict(list)
    tarm = collections.defaultdict(list)
    sarm = collections.defaultdict(dict)
    for r in good:
        arm[r["meas_arm"]].append(r["p5"])
        sarm[r["meas_arm"]][r["seed"]] = r["p5"]
        if r["t5"] is not None:
            tarm[r["meas_arm"]].append(r["t5"])

    print("\n-- THE ARMS (plateau5 PRIMARY; train5 reported alongside) --------")
    print("       %-5s %-9s %2s %10s %8s %10s %8s   %s"
          % ("cut", "spec", "n", "plateau5", "sd", "train5", "sd",
             "coarse ends | fine starts"))
    for a in ARMS:
        k = int(a[1:])
        v, t = arm[a], tarm[a]
        d = man["specs"].get(kspec(k), {})
        fg, lg = d.get("FIRSTGROUP") or ["?"], d.get("LASTGROUP") or ["?"]
        print("       %-5s %-9s %2d %10.4f %8.4f %10s %8s   %s | %s"
              % (a, kspec(k), len(v), statistics.mean(v),
                 statistics.stdev(v) if len(v) > 1 else 0.0,
                 "%10.4f" % statistics.mean(t) if t else "        --",
                 "%8.4f" % (statistics.stdev(t) if len(t) > 1 else 0.0) if t else "      --",
                 fg[-1], lg[0]))
    print("       per-seed plateau5: " + " | ".join(
        "%s %s" % (a, "/".join("%.3f" % sarm[a][s] for s in sorted(sarm[a])))
        for a in ARMS))

    print("\n-- G3 NOISE BLOW-UP ----------------------------------------------")
    noisy = ["%s sd %.4f" % (a, statistics.stdev(arm[a])) for a in ARMS
             if statistics.stdev(arm[a]) > NOISY_BAR]
    R.ok("no arm's SD exceeds 3 sigma_w = %.4f pp" % NOISY_BAR, not noisy,
         "; ".join(noisy))
    if noisy:
        print("\nUNRESOLVED-NOISY -- the registered noise floor does not transfer "
              "to this batch,\nso the bars do not apply.")
        return 5

    m = dict((a, statistics.mean(arm[a])) for a in ARMS)
    tm = dict((a, statistics.mean(tarm[a])) for a in ARMS if tarm[a])
    d1 = m["k52"] - m["k53"]
    d2 = m["k53"] - m["k54"]
    d3 = m["k54"] - m["k55"]
    total = m["k52"] - m["k55"]
    td = None
    if len(tm) == 4:
        td = (tm["k52"] - tm["k53"], tm["k53"] - tm["k54"], tm["k54"] - tm["k55"])

    print("\n-- THE THREE SINGLE-TENSOR DROPS ---------------------------------")
    print("       %-6s %-9s %-30s %9s %10s %8s %10s"
          % ("step", "k->k+1", "tensor that moves", "params", "dTEST pp",
             "SE units", "dTRAIN pp"))
    for i, (a, b, nm, pc) in enumerate(MOVES):
        dv = (d1, d2, d3)[i]
        print("       %-6s k%d->k%d    %-30s %9d %10.4f %8.2f %10s"
              % (DROP_LABEL[i], a, b, nm, pc, dv, dv / SE_ARM_DIFF,
                 "%10.4f" % td[i] if td else "        --"))
    print("       %-6s k52->k55  %-30s %9d %10.4f %8.2f %10s"
          % ("TOTAL", "(all three)", MOVED_PARAMS, total, total / SE_ARM_DIFF,
             "%10.4f" % (td[0] + td[1] + td[2]) if td else "        --"))
    print("       cpk1's own second cliff, DESCRIPTIVE, a DIFFERENT batch: "
          "%.4f pp" % CPK1_CLIFF2)

    print("\n-- Q5 THE MATCHED 512-vs-512 CONTROL -----------------------------")
    print("       DROP1 - DROP2 = %.4f pp = %.2f SE   bar %.4f pp   %s"
          % (d1 - d2, (d1 - d2) / SE_ADJ_STEPS, ASYM_BAR,
             "HELD" if abs(d1 - d2) > ASYM_BAR else "REFUTED"))
    print("       `%s` and `%s` hold 512 parameters each, are adjacent in"
          % (MOVES[0][2], MOVES[1][2]))
    print("       named_parameters(), and belong to the same BatchNorm2d.")
    print("       DISCLOSED (CORRECTIONS 147.5d): this control is NESTED, not")
    print("       matched -- DROP2 is the cost of moving the shift GIVEN the")
    print("       scale already went coarse.")

    print("\n-- R1, THEN R2, THEN R3, THEN THE VERDICT ------------------------")
    cliff, split, tensor, mech, why = verdict(d1, d2, d3)
    print("       R1 CLIFF-NOT-REPRODUCED   TOTAL %.4f vs bar %.4f pp   %s"
          % (total, CLIFF_BAR, "FIRES" if total < CLIFF_BAR else "does not fire"))
    if total >= CLIFF_BAR:
        even = all(abs(d - total / 3.0) <= EVEN_BAND for d in (d1, d2, d3))
        print("       R2 SPLIT-EVEN             |DROP_i - %.4f| <= %.4f pp   %s"
              % (total / 3.0, EVEN_BAND, "FIRES" if even else "does not fire"))
        if not even:
            order = sorted(range(3), key=lambda i: (d1, d2, d3)[i], reverse=True)
            dv = (d1, d2, d3)
            print("       R3 NO-MAJORITY            leader %s %.4f vs TOTAL/2 "
                  "%.4f, gap over %s %.4f vs %.4f pp   %s"
                  % (DROP_LABEL[order[0]], dv[order[0]], total / 2.0,
                     DROP_LABEL[order[1]], dv[order[0]] - dv[order[1]],
                     SEPARATION_BAR,
                     "FIRES" if (dv[order[0]] <= total / 2.0
                                 or dv[order[0]] - dv[order[1]] <= SEPARATION_BAR)
                     else "does not fire"))

    print("\n-- THE REGISTERED PREDICTIONS ------------------------------------")
    dv = (d1, d2, d3)
    lead = max(range(3), key=lambda i: dv[i])
    tlead = max(range(3), key=lambda i: td[i]) if td else None
    def mark(b):
        return "HELD" if b else "REFUTED"
    print("       Q1 (B1) largest drop is DROP1                      %s" % mark(lead == 0))
    print("       Q2 (B2) largest drop is DROP3                      %s" % mark(lead == 2))
    print("       Q3 the cliff reproduces (TOTAL >= %.4f)          %s"
          % (CLIFF_BAR, mark(total >= CLIFF_BAR)))
    print("       Q4 the split is NOT even                           %s"
          % mark(not all(abs(d - total / 3.0) <= EVEN_BAND for d in dv)))
    print("       Q5 |DROP1 - DROP2| > %.4f pp                     %s"
          % (ASYM_BAR, mark(abs(d1 - d2) > ASYM_BAR)))
    print("       Q6 TRAIN and TEST agree on the largest step        %s"
          % (mark(tlead == lead) if td else "UNSCORED (no train series)"))
    print("       Q7 DROP3 <= 0 (the big conv does not cost)         %s   "
          "(cts1's DROP3 = -7.707 pp)" % mark(d3 <= 0))

    print("\n-- VERDICT --------------------------------------------------------")
    print("       VERDICT (the CLIFF):     %s" % cliff)
    print("       VERDICT (the SPLIT):     %s" % split)
    print("       VERDICT (the TENSOR):    %s" % tensor)
    print("       VERDICT (the MECHANISM): %s" % mech)
    print("       %s" % why)
    print("\nFINAL: CLIFF %s | SPLIT %s | TENSOR %s | MECHANISM %s"
          % (cliff, split, tensor, mech))
    print("\n       OUT OF SCOPE by registration: CAPTURE (this batch has no "
          "anchors and\n       CAPTURE is not defined in it), the group COUNT "
          "m, resnet18_blocks,\n       CIFAR-10 / Tiny-ImageNet / ImageNet-489 "
          "/ ResNet-50, any other\n       meta-stepsize, alpha0 or optimiser "
          "pair, any hierarchical or shrinkage\n       operator, normalised "
          "entropy (DISQUALIFIED, cbl1 R5), the clamp\n       question at THIS "
          "cut position (cts2 answered it at k=49/50 only), and\n       the "
          "ASYMPTOTIC existence of either cliff (100-epoch snapshot; "
          "CORRECTIONS\n       147.5f showed these arms still rising at epoch "
          "99).")
    print("=" * 78)
    return 0
