"""Leave-one-rung-out robustness of the drift-vs-m log-log slope.
The sqrt(N) model predicts -0.500 over EVERY sub-span; if the refutation
depended on one rung it would not be a refutation."""
import os, sys, glob, math, statistics as st
sys.path.insert(0, "/cluster/account_a/metaopt/bin")
import agree2
from ladder32 import stats_for, ols

FAMS = {"r18": ["blk6","lay","node","w"], "c100": ["blk6","lay","node","w"],
        "r10": ["lay","node","w"], "r34": ["lay","node","w"]}

def pts(root, fam, g, lo, hi):
    out = []
    for d in sorted(glob.glob(os.path.join(root, f"p7-{fam}-{g}-s*"))):
        s = stats_for(d, lo, hi)
        if s and s["drift"] > 0:
            out.append((math.log10(s["ntot"]), math.log10(s["drift"])))
    return out

root = sys.argv[1]
for lo, hi, wname in ((0.50,1.00,"STEADY"), (0.00,0.20,"STARTUP")):
    print(f"\n===== {wname} =====")
    print(f"{'family':7s} {'dropped rung':14s} {'rungs':>5} {'N':>4} {'slope':>9} {'R2':>7}")
    for fam, grans in FAMS.items():
        P = {g: pts(root, fam, g, lo, hi) for g in grans}
        P = {g: v for g, v in P.items() if v}
        def fit(keys):
            xs = [x for g in keys for x, y in P[g]]
            ys = [y for g in keys for x, y in P[g]]
            return ols(xs, ys), len(xs)
        (b, r2), N = fit(list(P))
        print(f"{fam:7s} {'(none)':14s} {len(P):>5} {N:>4} {b:>+9.4f} {r2:>7.3f}")
        for drop in list(P):
            keys = [g for g in P if g != drop]
            if len(keys) < 2: continue
            (b, r2), N = fit(keys)
            print(f"{fam:7s} {drop:14s} {len(keys):>5} {N:>4} {b:>+9.4f} {r2:>7.3f}")
