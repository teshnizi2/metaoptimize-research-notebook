#!/bin/bash
# =============================================================================

# design module and THIS patch.

# THE ARMS (HEAD = sets:1-49,51-53/layer4.1.bn2.weight; HEADPATH = rec:cvt6_headpath; C50 = layer4.1.bn2.weight)

# line of each kind at construction on EVERY run, which the separate ENV audit and the scorer's G-WITNESS read.
# analysis/argsline_guard.py IS NOT EDITED.  WEIGHT DECAY ON idx 50 IS NOT A CSV COLUMN: the 9 masked runs go to
# results/CORPUS-EXCLUSIONS.tsv at landing (261.7).
#
# COST.  cvt9 by sacct: 0.817 GPU-h per PlainNet run.  15 jobs ~= 12.3 GPU-h expected; HARD BOUND = WALL x 15 = 45 GPU-h.
#
# USAGE   bash bin/cWD2_carrierwd.sh [--dry-run]   |   bash bin/cWD2_carrierwd.sh --submit   (the ONLY submitting form)
# ENV     CWD2_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }

echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C   ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"

  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"

  if [ -n "${CWD2_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CWD2_REGISTERED_COMMIT"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CWD2_REGISTERED_COMMIT is unset"

  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"

  || guard_fail "guard 1c4: fresh-seed check did not pass"

   && [ "$(grep -c '^ALL PASS$' "$ILOG")" = 2 ] && [ "$(grep -c '^  FAIL' "$ILOG")" = 0 ] && grep -q "^BATCH cwd2$" "$ILOG" \
   && grep -q "^TREE_POST $HARN HF $(sha "$HFH") build_network $(sha "$BNH")$" "$ILOG" \

   && grep -q "^PATCH_SHA256 $(sha "$PATCHD")$" "$ILOG" && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTR")$" "$ILOG" \
   && grep -q "^$(sha "$TESTD")  tests/test_decaymask.py$" "$ILOG" \
   && grep -q "PASS RR0 parent k01 vs itself: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 HIGHHEADPATH DECAY_MASK empty: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 LOWHEADPATH  DECAY_MASK unset: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR2 HIGHWD0      beta at every step BITWISE its unmasked twin's" "$ILOG" \
   && grep -q "PASS RR3 HIGHWD0      w', h', m' BITWISE the formula" "$ILOG" && grep -q "PASS RR3 NON-VACUITY (test-only" "$ILOG" \
   && grep -q "PASS D3(b) masked tensors" "$ILOG"; then
  echo "guard 4f: the inertness / bite log ALL PASS on THIS HF, THIS design, THIS patch: $(grep -c '^  PASS' "$ILOG") PASS / $(grep -c '^  FAIL' "$ILOG") FAIL (sha $(sha "$ILOG" | cut -c1-8))"

# ---- guard 4c': the launcher's literals == the design module's and the stage script's ------

          "$BN_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$RUNNER_SHA" "$REPLAY_SHA" "$STAGEW" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"

if {"k01": g_k01, "k01WD0": g_k01, "HIGHHEADPATH": g_head, "HIGHWD0": g_head, "LOWWD0": g_head} != Bm.SPEC: bad.append("SPEC")
if {"k01": "", "k01WD0": "", "HIGHHEADPATH": high, "HIGHWD0": high, "LOWWD0": low} != Bm.BETAHOLD: bad.append("BETAHOLD")
if {"k01": "", "k01WD0": "", "HIGHHEADPATH": chp, "HIGHWD0": chp, "LOWWD0": chp} != Bm.COMPHOLD: bad.append("COMPHOLD")

print("guard 4c': ARMS, spec strings, BETA_HOLD / COMP_HOLD / DECAY_MASK values, NET, NTENS, TOTPAR, SEEDS and the tree / runner / replay shas agree in launcher / design / stage script")

# ---- guard 4: THE LIVE-MODEL MANIFEST, built from cwd2's tree, per arm with every witness ----

python3 - "$HARN" "$MTMP" "$DES" <<'PYEOF' || guard_fail "guard 4: live partition / witness / manifest check FAILED"
