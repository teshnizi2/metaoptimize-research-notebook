#!/usr/bin/env python3
# =============================================================================
# cHE1_hdominate_element_score.py -- H-DOMINATE, ELEMENT LEVEL (TRACK A OF THE
# SIXTEENTH-CANDIDATE CYCLE).  ZERO GPU.  NO RUNS OF ITS OWN.
#
# THIS FILE MAKES NO RULE 21 CLAIM.  It re-analyses probe files already on disk
# (`cru1`, ingested at CORRECTIONS 174; `cfr1`/`cfr2`, ingested at 153/155) and
# the corpus CSV, so the wall-clock proof against an sacct Submit time does not
# exist for it.  It claims ONLY commit-before-first-execution.  Precedent: cQ1
# (149), cS1 (159), cZ1 (170), cY2 (174), cHD1 (179).
#
# =============================================================================
# THE HYPOTHESIS, AS THE CYCLE BRIEF STATES IT, BEFORE ANY NUMBER IS READ
# =============================================================================
#   H-DOMINATE (E), element level: in the scalar arm the ONE meta-gradient
#   z = sum_i <h_i, g_i> is a sum over ~11.2M elementwise products h_j*g_j,
#   unnormalised.  The brief's claim: on a substantial fraction of meta-steps
#   the SIGN of that aggregate disagrees with the sign the element-COUNT
#   majority would give, because a minority of large-magnitude products
#   carries the sum; and when they disagree the aggregate is NEGATIVE (beta
#   goes DOWN), which is why beta pins at the -15 floor and scalar fails on
#   CIFAR-100 (test 22.7 / train 22.9 at the standard cell, per the brief).
#
#   THE BRIEF'S RECIPE:  s_agg = sign(dz), dz the differenced per-record group
#   sum recovered from the cumulative z_mean (n = step + 2, CORRECTIONS 179);
#   s_maj = sign(0.5 - frac_neg);  R_E = P[s_agg != s_maj] over records where
#   both are defined.  BARS the brief asks to register: R_E > 0.25 SUBSTANTIAL
#   (domination), R_E < 0.10 REFUTES element-level domination; chance under
#   independent coin flips 0.50; signed tendency at disagreement DOWN vs UP;
#   CIFAR-10 (cfr1/cfr2 scalar arms, where scalar reaches ~88 %) as a control.
#
# =============================================================================
# WHAT THE SOURCE SAYS `frac_neg` IS.  READ OFF THE HARNESS BEFORE THIS FILE
# WAS WRITTEN, AND ASSERTED BY --selftest S2.  THIS DECIDES THE WHOLE TRACK.
# =============================================================================
#   patches/HF_patched.py, block_product (line ~161):
#       if self.stepsize_type == 'scalar':
#           return [sum([(u_*v_).sum() for u_,v_ in zip(u,v)])]
#   -> in the scalar arm `z` is a Python list holding ONE 0-d tensor: the
#      fully reduced aggregate.  The elementwise products h_j*g_j are summed
#      away inside block_product and are never returned.
#
#   patches/HF_patched.py, _probe, PATCH_PROBE2 (line ~461):
#       if self.stepsize_type in ('scalar', 'layerwise', 'blockwise'):
#           zall = z[0].detach().reshape(-1).float()
#       n_tot = zall.numel()
#       frac_neg  = (zall < 0).sum().item() / max(n_tot, 1)
#       frac_zero = (zall == 0).sum().item() / max(n_tot, 1)
#   -> `zall` is the arm's OWN partition vector (PATCH_PROBE5/6/7 say so in
#      their docstrings: "62 entries layerwise, 14,420 nodewise, 11.17M
#      weightwise").  In the SCALAR arm n_tot == 1, and frac_neg is the
#      INDICATOR 1[z_step < 0] of the single aggregate at the record's step.
#      It is INSTANTANEOUS (the z of that one meta-step, not cumulative) and
#      MODEL-WIDE, but it is the sign of the SUM, not a count over elements.
#      frac_zero is 1[z_step == 0].  t_neg/t_zero/t_n (PATCH_PROBE4) are the
#      per-element-of-`z` split -> [1[z<0]], [1[z==0]], [1].  z_sub subsamples
#      k = min(PROBE_SUB, n_tot) = 1 coordinate of zall: the same sign again.
#
#   CONSEQUENCE, REGISTERED HERE, BEFORE THE FIRST EXECUTION:
#   (a) The like-for-like statistic -- sign of the instantaneous aggregate
#       versus sign(0.5 - frac_neg) at the same step -- is IDENTICALLY ZERO in
#       the scalar arm by construction (both are the sign of the same scalar).
#       It is computed and printed as R_E_INST; it is a TAUTOLOGY, and its
#       only role is to confirm the source reading on every file: if it is
#       not 0.0000 on some file the reading above is wrong and the scorer
#       stamps SOURCE-READING-CONTRADICTED.
#   (b) The brief's LITERAL recipe compares sign(dz) -- the 100-step WINDOW
#       sum recovered by differencing -- against frac_neg at the record step,
#       i.e. the sign of the LAST meta-step in that window.  Those are two
#       different TIME objects of the same scalar.  The literal R_E is
#       therefore nonzero in general, but it measures within-window sign
#       VOLATILITY of the aggregate, NOT element domination.  It is computed
#       and printed as R_E_LIT (TEMPORAL), descriptive, no bar, and it may not
#       be quoted as an element-level domination rate.
#   (c) NO element-level statistic exists in any scalar probe on disk: the
#       jsonl carries the aggregate's sign only; the per-coordinate sidecars
#       (PATCH_PROBE5 neg_counts.npy, PATCH_PROBE6 coord_signs.npy /
#       tensor_signs.npy, PATCH_PROBE7 coord_neg_counts.npy) were not enabled
#       on cru1 / cfr1 / cfr2 (`find` over the three run roots returned
#       nothing before registration; the scorer re-runs that search on every
#       probe dir it reads).  The only instrument that COULD deliver (E) at
#       scalar granularity is PATCH_PROBE7 (it counts signs on u*v BEFORE the
#       reduction), and even it stores a cumulative per-coordinate negative
#       COUNT, not a per-record element majority.  The one scalar-trajectory
#       per-step instrument on disk, crn1_stage/probe_scalar.terms.npy, holds
#       the 62 per-TENSOR terms (12 epochs, seed 0): that is level (T), not
#       (E), and is out of this track's scope.
#
#   THE PRIMARY BRANCH IS THEREFORE DETERMINED BY THE SOURCE, NOT BY THE DATA:
#       UNRESOLVED-FRAC-NEG-NOT-THE-RIGHT-STATISTIC
#   sharpened to ELEMENT-STATISTIC-NOT-ON-DISK when the sidecar search on
#   every file is empty and every file has n_tot == 1.  The brief's bars are
#   recorded below so that a future instrument can be scored against them
#   without re-registration; they are NOT applied to R_E_LIT.
#
# =============================================================================
# A SECOND PLACE THE BRIEF SIMPLIFIES THE HARNESS, REGISTERED
# =============================================================================
#   The brief: "scalar's ONE beta moves by -ms*sign(sum_i <h_i,g_i>) each
#   meta-step".  The harness (HF_patched.py Lion_meta_update, line ~640):
#       beta <- (1 - ms*wd)*beta - ms*sign(Lion_beta2*m + (1-Lion_beta2)*z)
#       m    <- momentum_param*m + (1-momentum_param)*z
#   with, on cru1 (bin/cY1_rule11_ms_alpha0.sh lines 466-467) and on cfr1/cfr2
#   (bin/cO1_floor_release_row24.sh 541-542, cO2 likewise), and as train.py's
#   defaults: --momentum-param-meta 0.99 --Lion-beta2-meta 0.9
#   --weight-decay-meta 0.  So the per-meta-step DOWN vote is
#   sign(0.9*m + 0.1*z) with m an EMA(0.99) of z, not sign(z); the two agree
#   only when the momentum term agrees with the instantaneous aggregate.
#   With wd = 0 every meta-step moves beta by EXACTLY ms while off the clip
#   bounds, so over a 100-step window that cannot touch a bound,
#       n_down - n_up = -(beta_j - beta_{j-1}) / ms      (exactly, up to fp32)
#   which recovers the TRUE per-window DOWN-vote count from the recorded
#   beta trajectory.  That is the "signed tendency" in the one form the
#   record supports -- the aggregate's own direction, not the direction at an
#   (unmeasurable) element-level disagreement.  Descriptive, no bar.
#
# =============================================================================
# DISCLOSURE -- WHAT WAS VISIBLE AT REGISTRATION
# =============================================================================
#   * The source facts above (HF_patched.py, patch_probe*.py, cY1/cO1/cO2
#     launchers, train.py argparse defaults on cluster_b, live HF.py sha
#     9abd4318... which 179 records as byte-identical on the block_product
#     branch to cru1's pinned 0d8ee431...).
#   * Schema of every scalar probe: probe_index.json (n_tot 1, k 1), the
#     record count (500), the step layout (0..49900, PROBE=100), t_n == [1],
#     block_sizes n_b (11,220,132 c100 / 11,173,962 c10) on all 42 files; the
#     first three records of ONE file (cru1-sc-m1e-3-a1e-6-s15: frac_neg 1.0,
#     frac_zero 0.0, beta -13.81 -> -13.51 rising, z_std == |z_mean| at step
#     0) were seen while reading the schema.  The absence of any sidecar file
#     under the three roots.
#   * From the record: 174's cru1 numbers (scalar pins the floor; ~26 % of the
#     shared-ms gap), 177's spread/TRAVEL numbers, 179's window-sum recovery
#     and the n = step+2 identity, 153/155's cfr verdicts, the corpus rows'
#     plateau5 / final_train for the 42 runs (printed again below).
#   * NOT computed before the commit: any R_E, any DOWN share, any window
#     sum on any file, any floor-occupancy fraction, any Lion vote count.
#
# =============================================================================
# THE REGISTERED BARS (nominal -- for a future element-level instrument)
# =============================================================================
#   ELEMENT-DOMINATION      R_E >= 0.25 on the standard-cell pool (3 seeds)
#   NO-ELEMENT-DOMINATION   R_E <  0.10
#   INDETERMINATE           otherwise
#   SIGNED TENDENCY at disagreement: DOWN share >= 0.60 -> DOWN; <= 0.40 -> UP;
#                           else MIXED.  H-DOMINATE predicts DOWN.
#   DOMINATION-BUT-NOT-DOWN R_E >= 0.25 and tendency not DOWN
#   DOMINATION-NOT-SPECIFIC CIFAR-10 pool (cfr1-C, the same ms 1e-3 / clip -15
#                           / alpha0 1e-3 as the cru1 m1e-3-a1e-3 rung; the
#                           standard cell's alpha0 is 1e-6, which no CIFAR-10
#                           scalar probe shares) has R_E within 0.05 of the
#                           CIFAR-100 value AND the same tendency
#   ACROSS RUNGS            H-DOMINATE ties domination to floor pinning, so
#                           R_E should be ORDERED like the floor-occupancy
#                           share across the ten cru1 rungs (Spearman > 0);
#                           the brief's own expectation ("R_E rises as ms
#                           falls") is recorded but NOT adopted: at the frozen
#                           rungs ms is too small for beta to reach the floor
#                           at all, so pinning -- and, under the hypothesis,
#                           domination -- should be LOWEST there.
#   None of these is applied to anything this file can compute.
#
# =============================================================================
# THE DESCRIPTIVE COMPANIONS (labelled, no bar, every one per run / per rung /
# per dataset).  They characterise the ONE aggregate's dynamics; none is a
# test of H-DOMINATE (E).
# =============================================================================
#   R_E_INST   P[sign(z_step) != sign(0.5 - frac_neg)]   -- the tautology (a)
#   R_E_LIT    P[sign(window sum) != sign(z at the window's last step)],
#              indeterminate windows excluded (fp32 floor as in 179)  -- (b)
#   DOWN_INST  P[frac_neg == 1] over records: the instantaneous aggregate is
#              negative (Lion would vote DOWN if momentum agreed)
#   DOWN_WIN   P[window sum < 0] over determinate windows
#   LION_DOWN  mean over CLEAN windows of n_down/100 recovered from delta beta;
#              CLEAN = beta_{j-1} more than 100*ms + eps from both clip bounds
#              (so no clipping can occur inside the window); the residual
#              |x - round(x)| and parity failures are printed as the fp32
#              precision diagnostic; UNCERTAIN windows counted, not scored
#   FLOOR      P[n_at_lo == 1] over records (beta at the -15 floor, eps 1e-6
#              as the harness counts it); CEIL likewise for n_at_hi
#   ZERO       count of records with frac_zero == 1 (aggregate exactly 0)
#
#   Recovery checks run on every file (G-checks): stepsize_type scalar;
#   sum(n_b) == the model's parameter count; 500 records at 0..49900; step-0
#   identity z_std == |z_mean| (n = 2, first term zero); STEP-0 SIGN IDENTITY
#   sign(z_mean_0) == the sign frac_neg_0 / frac_zero_0 encode (both are the
#   sign of z_0, so they must agree EXACTLY -- this is the requested
#   verification of the differencing recovery on the step-0 record);
#   t_n == [1]; frac_neg and frac_zero in {0, 1}; n_beta == 1.
#
# USAGE
#   python3 analysis/cHE1_hdominate_element_score.py [--probe-root DIR]... [--csv PATH]

def read_csv(path):
    with open(path, newline="") as fh:
        return list(csv.DictReader(fh))


def scalar_rows(rows):
    out = []
    for r in rows:
        if not r["run"].startswith(BATCH_PREFIXES):
            continue
        if r.get("granularity") != "scalar":
            continue
        if r.get("complete") != "1" or r.get("superseded") != "0":
            continue
        if r.get("hier", "") != "":
            continue
        out.append(r)
    out.sort(key=lambda r: (r["dataset"], r["run"]))
    return out


def parse_clip(s):
    lo, hi = s.split(":")
    return float(lo), float(hi)

def find_probe(roots, run):
    for root in roots:
        if not root or not os.path.isdir(root):
            continue
        for pat in (os.path.join(root, "probe_" + run),
                    os.path.join(root, "*", "probe_" + run)):
            for d in sorted(glob.glob(pat)):
                if os.path.isfile(os.path.join(d, "probe.jsonl")):
                    return d
    return None


def read_probe(d):
    recs = []
    with open(os.path.join(d, "probe.jsonl")) as fh:
        for line in fh:
            line = line.strip()
            if line:
                recs.append(json.loads(line))
    bs = None
    p = os.path.join(d, "block_sizes.json")
    if os.path.isfile(p):
        with open(p) as fh:
            bs = json.load(fh)
    pidx = None
    p = os.path.join(d, "probe_index.json")
    if os.path.isfile(p):
        with open(p) as fh:
            pidx = json.load(fh)
    side = [s for s in SIDECARS if os.path.exists(os.path.join(d, s))]
    return recs, bs, pidx, side


def sgn(x):
    return (x > 0) - (x < 0)


def rec_sign(r):
    """The sign of z at the record's step, as frac_neg / frac_zero encode it."""
    if r["frac_zero"] == 1.0:
        return 0
    return -1 if r["frac_neg"] == 1.0 else +1


def step0_is_n2(r0):
    for m, s in zip(r0["z_mean"], r0["z_std"]):
        if abs(abs(m) - s) > 1e-6 * max(abs(m), 1e-30) + 1e-30:
            return False
    return True


def gcheck(recs, bs, pidx, nparams):
    bad = []
    if bs is None:
        bad.append("no block_sizes.json")
    else:
        if bs.get("stepsize_type") != "scalar":
            bad.append("stepsize_type %r != scalar" % bs.get("stepsize_type"))
        if nparams is not None and sum(bs.get("n_b", [])) != nparams:
            bad.append("sum(n_b) %s != %s" % (sum(bs.get("n_b", [])), nparams))
    if len(recs) != N_RECORDS:
        bad.append("%d records != %d" % (len(recs), N_RECORDS))
    steps = [r.get("step") for r in recs]
    if steps != EXPECTED_STEPS[:len(steps)] or len(steps) != N_RECORDS:
        bad.append("steps are not 0,100,...,49900")
    for r in recs:
        if len(r.get("z_mean", [])) != 1 or len(r.get("beta", [])) != 1:
            bad.append("a record does not carry exactly one z_mean / beta")
            break
        if r.get("t_n") != [1]:
            bad.append("t_n != [1]")
            break
        if r.get("frac_neg") not in (0.0, 1.0) or r.get("frac_zero") not in (0.0, 1.0):
            bad.append("frac_neg/frac_zero not in {0,1}")
            break
        if r.get("n_beta") != 1:
            bad.append("n_beta != 1")
            break
    if pidx is not None and (pidx.get("n_tot") != 1 or pidx.get("k") != 1):
        bad.append("probe_index n_tot/k != 1")
    if recs and not step0_is_n2(recs[0]):
        bad.append("step-0 record fails z_std == |z_mean| (the n = 2 identity)")
    if recs and sgn(recs[0]["z_mean"][0]) != rec_sign(recs[0]):
        bad.append("STEP-0 SIGN IDENTITY fails: sign(z_mean_0) != frac_neg_0's sign")
    return bad


def n_tot_of(recs, pidx):
    if pidx is not None and "n_tot" in pidx:
        return int(pidx["n_tot"])
    if recs:
        return int(sum(recs[0].get("t_n") or [0]))
    return -1

# the statistics, per file
# =============================================================================
def score_file(recs, ms, lo, hi):
    """All per-file statistics.  Returns an OrderedDict of counts/rates."""
    st = collections.OrderedDict()
    n = len(recs)
    # --- instantaneous sign per record
    inst = [rec_sign(r) for r in recs]
    zero = sum(1 for s in inst if s == 0)
    st["records"] = n
    st["zero_records"] = zero
    # R_E_INST: sign(z_step) vs sign(0.5 - frac_neg) -- same scalar, both ways
    n_def = 0
    n_dis = 0
    for r, s in zip(recs, inst):
        s_maj = sgn(0.5 - r["frac_neg"])
        if s == 0 or s_maj == 0:
            continue
        n_def += 1
        if s != s_maj:
            n_dis += 1
    st["RE_inst_n"] = n_def
    st["RE_inst"] = (n_dis / n_def) if n_def else float("nan")
    # DOWN_INST
    st["down_inst"] = sum(1 for s in inst if s < 0) / n if n else float("nan")
    # --- windows: S_j = z_mean_j * (step_j + 2)
    S = [r["z_mean"][0] * (r["step"] + N_OFFSET) for r in recs]
    W = []
    for j in range(1, n):
        w = S[j] - S[j - 1]
        fl = FLOOR_ADDS * FLOAT32_ULP * max(abs(S[j]), abs(S[j - 1]))
        W.append((w, fl))
    det = [(j, w) for j, (w, fl) in enumerate(W, start=1) if abs(w) > fl]
    st["windows"] = len(W)
    st["windows_indet"] = len(W) - len(det)
    st["down_win"] = (sum(1 for _, w in det if w < 0) / len(det)) if det else float("nan")
    # R_E_LIT: sign(window sum) vs sign(z at the window's last step) = inst[j]
    n_def = 0
    n_dis = 0
    n_dis_down = 0
    for j, w in det:
        s_last = inst[j]
        if s_last == 0:
            continue
        n_def += 1
        if sgn(w) != s_last:
            n_dis += 1
            if w < 0:
                n_dis_down += 1
    st["RE_lit_n"] = n_def
    st["RE_lit"] = (n_dis / n_def) if n_def else float("nan")
    st["RE_lit_down_at_dis"] = (n_dis_down / n_dis) if n_dis else float("nan")
    # --- floor / ceiling occupancy from the harness's own counters
    st["floor"] = sum(1 for r in recs if r.get("n_at_lo") == 1) / n if n else float("nan")
    st["ceil"] = sum(1 for r in recs if r.get("n_at_hi") == 1) / n if n else float("nan")
    # --- Lion vote count per CLEAN window from delta beta
    betas = [r["beta"][0] for r in recs]
    clean = []
    uncertain = 0
    resid = []
    parity_fail = 0
    for j in range(1, n):
        b0, b1 = betas[j - 1], betas[j]
        reach = PROBE_EVERY * ms + CLIP_EPS
        if (b0 - lo) > reach and (hi - b0) > reach:
            x = -(b1 - b0) / ms                  # n_down - n_up
            k = int(round(x))
            resid.append(abs(x - k))
            if (k + PROBE_EVERY) % 2 != 0:
                parity_fail += 1
                k += 1 if x > k else -1
            n_down = (PROBE_EVERY + k) / 2.0
            clean.append(n_down / PROBE_EVERY)
        else:
            uncertain += 1
    st["lion_clean"] = len(clean)
    st["lion_uncertain"] = uncertain
    st["lion_down"] = (sum(clean) / len(clean)) if clean else float("nan")
    st["lion_resid_mean"] = (sum(resid) / len(resid)) if resid else float("nan")
    st["lion_resid_max"] = max(resid) if resid else float("nan")
    st["lion_parity_fail"] = parity_fail
    # time course of the Lion vote by decile of the CLEAN windows' position
    dec = [[] for _ in range(10)]
    for j in range(1, n):
        b0, b1 = betas[j - 1], betas[j]
        reach = PROBE_EVERY * ms + CLIP_EPS
        if (b0 - lo) > reach and (hi - b0) > reach:
            x = -(b1 - b0) / ms
            dec[min(9, (j - 1) * 10 // (n - 1))].append((PROBE_EVERY + round(x)) / 2.0 / PROBE_EVERY)
    st["lion_down_deciles"] = [((sum(d) / len(d)) if d else float("nan")) for d in dec]
    st["floor_deciles"] = []
    for q in range(10):
        seg = recs[q * n // 10:(q + 1) * n // 10]
        st["floor_deciles"].append(sum(1 for r in seg if r.get("n_at_lo") == 1) / len(seg) if seg else float("nan"))
    st["beta_first"] = betas[0]
    st["beta_last"] = betas[-1]
    st["beta_min"] = min(betas)
    return st


# =============================================================================
# pooling
# =============================================================================
def pool(stats):
    """Pool per-file stats by summing counts (rates re-derived from counts)."""
    if not stats:
        return None
    p = collections.OrderedDict()
    n = sum(s["records"] for s in stats)
    p["files"] = len(stats)
    p["records"] = n
    p["zero_records"] = sum(s["zero_records"] for s in stats)
    tot = sum(s["RE_inst_n"] for s in stats)
    p["RE_inst"] = (sum(s["RE_inst"] * s["RE_inst_n"] for s in stats if s["RE_inst_n"]) / tot) if tot else float("nan")
    p["down_inst"] = sum(s["down_inst"] * s["records"] for s in stats) / n
    dw = sum(s["windows"] - s["windows_indet"] for s in stats)
    p["down_win"] = (sum(s["down_win"] * (s["windows"] - s["windows_indet"]) for s in stats if s["windows"] - s["windows_indet"]) / dw) if dw else float("nan")
    p["windows_indet"] = sum(s["windows_indet"] for s in stats)
    tot = sum(s["RE_lit_n"] for s in stats)
    p["RE_lit"] = (sum(s["RE_lit"] * s["RE_lit_n"] for s in stats if s["RE_lit_n"]) / tot) if tot else float("nan")
    ndis = sum(round(s["RE_lit"] * s["RE_lit_n"]) for s in stats if s["RE_lit_n"])
    nd = sum(round(s["RE_lit_down_at_dis"] * round(s["RE_lit"] * s["RE_lit_n"])) for s in stats
             if s["RE_lit_n"] and not math.isnan(s["RE_lit_down_at_dis"]))
    p["RE_lit_down_at_dis"] = (nd / ndis) if ndis else float("nan")
    p["floor"] = sum(s["floor"] * s["records"] for s in stats) / n
    p["ceil"] = sum(s["ceil"] * s["records"] for s in stats) / n
    lc = sum(s["lion_clean"] for s in stats)
    p["lion_clean"] = lc
    p["lion_uncertain"] = sum(s["lion_uncertain"] for s in stats)
    p["lion_down"] = (sum(s["lion_down"] * s["lion_clean"] for s in stats if s["lion_clean"]) / lc) if lc else float("nan")
    p["lion_parity_fail"] = sum(s["lion_parity_fail"] for s in stats)
    p["lion_resid_max"] = max(s["lion_resid_max"] for s in stats)
    return p


def fmt(x, d=4):
    if x is None or (isinstance(x, float) and math.isnan(x)):
        return "   nan"
    return "%.*f" % (d, x)

def _synth(zs, ms, lo, hi, probe_every=PROBE_EVERY):
    """Synthetic scalar probe: run the harness's arithmetic on a z sequence.

    zs[t] is z at counter t for t = -1..T-1 (zs[0] is the counter -1 term, 0).
    Returns records in the harness layout with cumulative z_mean, frac_neg,
    frac_zero, beta post-Lion-update post-clamp, n_at_lo/hi, t_n."""
    recs = []
    zsum = 0.0
    zsq = 0.0
    n = 0
    beta = -6.9
    m = 0.0
    for t, z in enumerate(zs, start=-1):
        # meta update (wd = 0), clamp, then probe -- the step() order
        beta = beta - ms * sgn(LION_BETA2 * m + (1 - LION_BETA2) * z)
        m = MOM_META * m + (1 - MOM_META) * z
        beta = min(max(beta, lo), hi)
        zsum += z
        zsq += z * z
        n += 1
        if t % probe_every:
            continue
        mean = zsum / n
        var = zsq / n - mean * mean
        recs.append({"step": t, "beta": [beta], "z_mean": [mean], "z_std": [math.sqrt(max(var, 0.0))],
                     "frac_neg": 1.0 if z < 0 else 0.0, "frac_zero": 1.0 if z == 0 else 0.0,
                     "n_at_lo": 1 if beta <= lo + CLIP_EPS else 0, "n_at_hi": 1 if beta >= hi - CLIP_EPS else 0,
                     "n_beta": 1, "t_neg": [1 if z < 0 else 0], "t_zero": [1 if z == 0 else 0], "t_n": [1]})
    return recs


def selftest():
    fails = []

    def ok(name, cond, detail=""):
        print("   %s  %s%s" % ("PASS" if cond else "FAIL", name, ("  " + detail) if detail else ""))
        if not cond:
            fails.append(name)

    print("cHE1_hdominate_element_score.py --selftest")
    # S1 constants
    ok("S1 500 records at PROBE=100 over 50,000 meta-steps", N_RECORDS == 500 and EXPECTED_STEPS[-1] == 49900)
    ok("S1 parameter counts: ResNet18_c100 11,220,132 / ResNet18 11,173,962",
       PARAMS["ResNet18_c100"] == 11220132 and PARAMS["ResNet18"] == 11173962)

    # S2 the source premises on the repo's own patch files
    hfp = os.path.join(ROOT, "patches", "HF_patched.py")
    prp = os.path.join(ROOT, "patches", "patch_probe.py")
    p5 = os.path.join(ROOT, "patches", "patch_probe5.py")
    if os.path.isfile(hfp) and os.path.isfile(prp):
        hf = open(hfp).read()
        pr = open(prp).read()
        ok("S2 scalar reduction returns ONE fully reduced aggregate",
           "return [sum([(u_*v_).sum() for u_,v_ in zip(u,v)])]" in hf)
        ok("S2 zall is z[0] flattened for scalar/layerwise/blockwise",
           "zall = z[0].detach().reshape(-1).float()" in hf and "n_tot = zall.numel()" in hf)
        ok("S2 frac_neg / frac_zero are counts over zall divided by n_tot",
           "frac_neg = (zall < 0).sum().item() / max(n_tot, 1)" in hf
           and "frac_zero = (zall == 0).sum().item() / max(n_tot, 1)" in hf)
        ok("S2 t_n is the per-element-of-z numel list (PATCH_PROBE4)",
           "_t_n = [int(zi.numel()) for zi in z]" in hf)
        ok("S2 z_sub draws k = min(PROBE_SUB, n_tot) coordinates of zall",
           "_k = min(int(os.environ.get('PROBE_SUB', '20000')), int(n_tot))" in hf)
        ok("S2 counter starts at -1; meta_update, clamp, THEN _probe, THEN counter += 1",
           "self.counter = -1" in hf
           and hf.index("self.meta_update(HtT_gradft)") < hf.index("self.beta[_i].clamp(self._beta_lo, self._beta_hi)")
           < hf.index("self._probe(HtT_gradft)") < hf.index("self.counter+=1"))
        ok("S2 z_mean is a cumulative running mean (never reset)",
           "self._z_sum += zv" in pr and "mean = self._z_sum / max(self._z_n, 1)" in pr
           and "self._z_sum = None" not in pr.split("def _probe(self, z):")[1])
        ok("S2 Lion meta update is (1-ms*wd)*beta - ms*sign(b2*m + (1-b2)*z), m <- mp*m + (1-mp)*z",
           "self.beta[i] = (1-self.args_meta['meta_stepsize']*self.args_meta['weight_decay'])*self.beta[i] - self.args_meta['meta_stepsize'] * torch.sign(self.args_meta['Lion_beta2']*self.momentum_meta[i] + (1-self.args_meta['Lion_beta2'])*HtT_gradft[i])" in hf
           and "self.momentum_meta[i] = self.args_meta['momentum_param'] * self.momentum_meta[i] + (1-self.args_meta['momentum_param'])*HtT_gradft[i]" in hf)
        ok("S2 PATCH_CLIPCOUNT eps is 1e-6", "_ceps = 1e-6" in hf and CLIP_EPS == 1e-6)
        if os.path.isfile(p5):
            ok("S2 PATCH_PROBE5 documents zall as the arm's OWN partition vector",
               "the arm's" in open(p5).read() and "OWN partition vector" in open(p5).read())
    else:
        ok("S2 patch files reachable", False, "patches/HF_patched.py or patch_probe.py missing")
    cy1 = os.path.join(ROOT, "bin", "cY1_rule11_ms_alpha0.sh")
    co1 = os.path.join(ROOT, "bin", "cO1_floor_release_row24.sh")
    for nm, p in (("cY1 (cru1)", cy1), ("cO1 (cfr1)", co1)):
        if os.path.isfile(p):
            s = open(p).read()
            ok("S2 %s passes --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0" % nm,
               "--momentum-param-meta 0.99 --Lion-beta2-meta 0.9" in s and "--weight-decay-meta 0" in s)

    # S3 synthetic recovery: n = step + 2, step-0 identities, R_E_INST == 0
    import random
    rng = random.Random(20260908)
    T = 3000
    zs = [0.0] + [rng.gauss(-0.2, 1.0) for _ in range(T)]
    recs = _synth(zs, 1e-3, -15.0, -2.3026)
    ok("S3 synthetic probe has the harness layout (records at 0,100,..)", [r["step"] for r in recs] == list(range(0, T, 100)))
    ok("S3 step-0 identity z_std == |z_mean| (n = 2, first term 0)", step0_is_n2(recs[0]))
    ok("S3 step-0 sign identity sign(z_mean_0) == frac_neg_0's sign", sgn(recs[0]["z_mean"][0]) == rec_sign(recs[0]))
    S = [r["z_mean"][0] * (r["step"] + N_OFFSET) for r in recs]
    errs = []
    for j in range(1, len(recs)):
        true = sum(zs[recs[j - 1]["step"] + 2: recs[j]["step"] + 2])
        errs.append(abs((S[j] - S[j - 1]) - true))
    ok("S3 differencing with n = step+2 recovers every 100-step window sum", max(errs) < 1e-9, "max err %.2e" % max(errs))
    st = score_file(recs, 1e-3, -15.0, -2.3026)
    ok("S3 R_E_INST is identically 0 when n_tot = 1", st["RE_inst_n"] > 0 and st["RE_inst"] == 0.0)
    gb = gcheck(recs, {"stepsize_type": "scalar", "n_b": [PARAMS["ResNet18_c100"]]}, {"n_tot": 1, "k": 1}, PARAMS["ResNet18_c100"])
    ok("S3 the only G-check failures on the 30-record synthetic are the length ones",
       set(gb) == {"%d records != %d" % (len(recs), N_RECORDS), "steps are not 0,100,...,49900"}, str(gb))
    gb2 = gcheck(recs, {"stepsize_type": "layerwise", "n_b": [1, 2]}, {"n_tot": 62, "k": 62}, PARAMS["ResNet18_c100"])
    ok("S3 a layerwise-typed or n_tot != 1 file FAILS the G-check", "stepsize_type 'layerwise' != scalar" in gb2 and "probe_index n_tot/k != 1" in gb2)
    # S3 R_E_LIT is a genuine temporal statistic: nonzero here, and it is the
    # window-vs-last-step disagreement computed directly
    dis = 0
    n = 0
    for j in range(1, len(recs)):
        w = S[j] - S[j - 1]
        s_last = sgn(zs[recs[j]["step"] + 1])
        if s_last == 0:
            continue
        n += 1
        dis += (sgn(w) != s_last)
    ok("S3 R_E_LIT equals the directly computed window-vs-last-step disagreement", abs(st["RE_lit"] - dis / n) < 1e-12 and st["RE_lit"] > 0,
       "%.4f" % st["RE_lit"])
    # S3 Lion vote recovery from delta beta on clean windows, checked against the truth
    truth = []
    m = 0.0
    votes = []
    for z in zs:
        votes.append(sgn(LION_BETA2 * m + (1 - LION_BETA2) * z))
        m = MOM_META * m + (1 - MOM_META) * z
    for j in range(1, len(recs)):
        seg = votes[recs[j - 1]["step"] + 2: recs[j]["step"] + 2]
        truth.append(sum(1 for v in seg if v > 0) / 100.0)   # DOWN = beta decreases = vote > 0
    ok("S3 Lion DOWN share recovered from delta beta matches the true vote count on every clean window",
       st["lion_clean"] == len(truth) and abs(st["lion_down"] - sum(truth) / len(truth)) < 1e-9 and st["lion_resid_max"] < 1e-9)
    # S3 clipping: a floor-bound trajectory marks windows UNCERTAIN, never scores them
    zs2 = [0.0] + [abs(rng.gauss(1.0, 0.1)) for _ in range(T)]       # z > 0 -> vote DOWN always
    recs2 = _synth(zs2, 1e-3, -7.5, -2.3026)                        # floor 0.6 below start
    st2 = score_file(recs2, 1e-3, -7.5, -2.3026)
    ok("S3 floor-pinned synthetic: floor share > 0.7 and pinned windows are UNCERTAIN, not scored",
       st2["floor"] > 0.7 and st2["lion_uncertain"] >= 20 and st2["lion_clean"] + st2["lion_uncertain"] == len(recs2) - 1)
    ok("S3 floor-pinned synthetic: clean windows vote DOWN 100 %", st2["lion_down"] == 1.0)
    ok("S3 DOWN_INST on a z > 0 sequence is 0 (frac_neg = 0 means z >= 0)", st2["down_inst"] == 0.0)
    stz = score_file(_synth([0.0] * 301, 1e-3, -15, -2.3), 1e-3, -15, -2.3)   # counters -1..299 -> records 0,100,200
    ok("S3 zero records are counted and excluded from R_E_INST", stz["zero_records"] == 3 and math.isnan(stz["RE_inst"]), str(stz["zero_records"]))

    # S4 pooling re-derives rates from counts
    p = pool([st, st2])
    ok("S4 pooled records = sum of files", p["records"] == st["records"] + st2["records"])
    ok("S4 pooled floor share is record-weighted", abs(p["floor"] - (st["floor"] * st["records"] + st2["floor"] * st2["records"]) / p["records"]) < 1e-12)

    # S5 corpus filter
    csvp = os.path.join(ROOT, "results", "all_runs.csv")
    if os.path.isfile(csvp):
        rows = read_csv(csvp)
        sr = scalar_rows(rows)
        ok("S5 corpus has 2,740 rows", len(rows) == 2740, str(len(rows)))
        ok("S5 42 scalar rows: 30 cru1 + 6 cfr1 + 6 cfr2", len(sr) == 42 and sum(r["run"].startswith("cru1") for r in sr) == 30, str(len(sr)))
        ok("S5 every one is SGDm+Lion, 100 epochs, batch 100, augment 1, gamma 1",
           all(r["base"] == "SGDm" and r["meta"] == "Lion" and r["epochs_requested"] == "100"
               and r["batch_size"] == "100" and r["augment"] == "1" and r["gamma"] == "1" for r in sr))
        ok("S5 clip strings parse", all(parse_clip(r["beta_clip"]) in ((-15.0, -2.3026), (-80.0, -2.3026)) for r in sr))

    print("selftest: %s" % ("ALL PASS" if not fails else "FAIL: " + ", ".join(fails)))
    return 0 if not fails else 1

    sys.exit(main(a))
