#!/bin/bash
# =============================================================================
# cVT4_stage_harness.sh -- BUILD OR VERIFY THE ISOLATED `cvt4` HARNESS TREE.
#   (CORRECTIONS 237.)  Touches NO existing file.  Submits nothing.
#
# WHY AN ISOLATED TREE.  `cvt4` needs PATCH_BETAHOLD (patches/patch_betahold.py) in HF.py.  The live shared
# harness must stay byte-untouched, and cvt1's tree ($WS/harness_cvt1, HF.py 3f2b98e1...) is a REGISTERED
# tree three landed batches ran from (cvt1, cvt2, cvt3) and is not edited either.  So cvt4 gets its own
# tree: a byte-identical copy of cvt1's tree with PATCH_BETAHOLD applied on top of cvt1's HF.py, and its own
# runner that differs from jobs/run_cifar.sh in its `cd` line only.
#
# WHAT IT BUILDS (under $METAOPT_WS, every path new and cvt4-namespaced):
#   harness_cvt4/cifar10/                every file of cvt1's tree (data excepted), byte-identical, except HF.py
#   harness_cvt4/cifar10/data  ->        SYMLINK to the live tree's data/ (the same target as cvt1's)
#   harness_cvt4/cifar10/Optimizers/HF.py.pre_betahold   cvt1's HF.py bytes (HF_PRE_SHA 3f2b98e1...)
#   harness_cvt4/cifar10/Optimizers/HF.py                = cvt1's HF.py + PATCH_BETAHOLD (HF_POST_SHA)
#   jobs/run_cifar_cvt4.sh               jobs/run_cifar.sh with EXACTLY the cd line changed
#   harness_cvt4/STAGE-MANIFEST.txt      sha256 of every staged file
#
# USAGE
#   bash bin/cVT4_stage_harness.sh            # VERIFY ONLY (exit 0 iff consistent)
#   bash bin/cVT4_stage_harness.sh --stage    # build it (refuses if it exists)
# =============================================================================
set -u
MODE=verify; [ "${1:-}" = "--stage" ] && MODE=stage
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
SRC=$WS/harness_cvt1/cifar10
HROOT=$WS/harness_cvt4
DEST=$HROOT/cifar10
RUNNER_SRC=$WS/jobs/run_cifar.sh
RUNNER=$WS/jobs/run_cifar_cvt4.sh
PATCH_BH=$REPO/patches/patch_betahold.py
POST_SHA=e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_PRE_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_POST_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
FAIL=0; bad() { echo "!!! STAGE FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

src_files() { (cd "$SRC" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort); }

echo "cVT4_stage_harness: mode=$MODE  WS=$WS"
echo "  source tree $SRC (cvt1's, read only)"
echo "  cvt4 tree   $DEST"
echo "  runner      $RUNNER  (from $RUNNER_SRC)"

if [ "$MODE" = stage ]; then
  [ -e "$HROOT" ] && { echo "!!! $HROOT already exists -- refusing to overwrite. Verify it instead."; exit 2; }
  [ -e "$RUNNER" ] && { echo "!!! $RUNNER already exists -- refusing to overwrite."; exit 2; }
  [ -f "$PATCH_BH" ] || { echo "!!! $PATCH_BH is missing"; exit 2; }
  [ "$(sha "$SRC/Optimizers/HF.py")" = "$HF_PRE_SHA" ] || { echo "!!! cvt1's HF.py sha != pinned $HF_PRE_SHA"; exit 2; }
  [ "$(sha "$SRC/build_network.py")" = "$POST_SHA" ] || { echo "!!! cvt1's build_network.py sha != pinned $POST_SHA"; exit 2; }
  mkdir -p "$DEST" || exit 2
  for f in $(src_files); do
    mkdir -p "$DEST/$(dirname "$f")"; cp -p "$SRC/$f" "$DEST/$f" || exit 2
  done
  ln -s "$(readlink "$SRC/data")" "$DEST/data" || exit 2
  HF_PATH="$DEST/Optimizers/HF.py" python3 "$PATCH_BH" || { echo "!!! patch_betahold failed"; exit 2; }
  sed "s#^cd $LIVE\$#cd $DEST#" "$RUNNER_SRC" > "$RUNNER" || exit 2
  chmod --reference="$RUNNER_SRC" "$RUNNER" 2>/dev/null || chmod 755 "$RUNNER"
  { echo "STAGED_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "SRC $SRC"; echo "DEST $DEST"
    echo "PATCH_BETAHOLD_SHA256 $(sha "$PATCH_BH")"
    for f in $(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort); do
      echo "FILE $(sha "$DEST/$f") $f"; done
    echo "RUNNER $(sha "$RUNNER") $RUNNER"
    echo "RUNNER_SRC $(sha "$RUNNER_SRC") $RUNNER_SRC"; } > "$HROOT/STAGE-MANIFEST.txt"
  echo "staged; now verifying"
fi

# ---------------------------------------------------------------- VERIFY ------
[ -d "$DEST" ] || { echo "!!! no cvt4 tree at $DEST -- run with --stage"; exit 2; }
N=0
for f in $(src_files); do
  [ "$f" = ./Optimizers/HF.py ] && continue
  if [ ! -f "$DEST/$f" ]; then bad "V-a $f missing from the cvt4 tree"; continue; fi
  [ "$(sha "$SRC/$f")" = "$(sha "$DEST/$f")" ] && N=$((N+1)) || bad "V-a $f differs from cvt1's tree"
done
echo "V-a: $N files byte-identical to cvt1's tree (Optimizers/HF.py excepted)"
EXTRA=$(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort | while read -r f; do
          [ -f "$SRC/$f" ] || [ "$f" = ./Optimizers/HF.py.pre_betahold ] || echo "$f"; done)
[ -z "$EXTRA" ] && echo "V-a2: no file in the cvt4 tree beyond cvt1's except Optimizers/HF.py.pre_betahold" || bad "V-a2 extra files: $EXTRA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_betahold")" = "$HF_PRE_SHA" ] && echo "V-h: HF.py.pre_betahold sha == cvt1's HF.py == pinned $HF_PRE_SHA" \
  || bad "V-h HF.py.pre_betahold sha != pinned $HF_PRE_SHA"
[ "$(sha "$DEST/Optimizers/HF.py")" = "$HF_POST_SHA" ] && echo "V-h2: cvt4 HF.py sha == pinned post $HF_POST_SHA" \
  || bad "V-h2 cvt4 HF.py sha $(sha "$DEST/Optimizers/HF.py") != pinned post $HF_POST_SHA"
[ "$(sha "$SRC/Optimizers/HF.py")" = "$HF_PRE_SHA" ] && echo "V-h3: cvt1's HF.py is still $HF_PRE_SHA (untouched)" \
  || bad "V-h3 cvt1's HF.py has moved"
grep -q PATCH_BETAHOLD "$SRC/Optimizers/HF.py" && bad "V-h4 cvt1's HF.py carries PATCH_BETAHOLD" \
  || echo "V-h4: cvt1's HF.py carries NO PATCH_BETAHOLD marker"
grep -q PATCH_BETAHOLD "$LIVE/Optimizers/HF.py" && bad "V-h5 the LIVE HF.py carries PATCH_BETAHOLD" \
  || echo "V-h5: the LIVE HF.py carries NO PATCH_BETAHOLD marker"
[ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] && echo "V-h6: the LIVE HF.py is still $HF_LIVE_SHA (untouched)" \
  || echo "V-h6: NOTE -- the LIVE HF.py has moved ($(sha "$LIVE/Optimizers/HF.py")); the cvt4 tree does not read it."
[ "$(sha "$DEST/build_network.py")" = "$POST_SHA" ] && echo "V-b: cvt4 build_network.py == cpl1's POST_SHA $POST_SHA" \
  || bad "V-b cvt4 build_network.py != $POST_SHA"
[ -L "$DEST/data" ] && [ "$(readlink "$DEST/data")" = "$LIVE/data" ] && echo "V-c: data -> $LIVE/data (symlink, cvt1's target)" \
  || bad "V-c data is not a symlink to $LIVE/data"
[ -d "$DEST/data/cifar-100-python" ] && echo "V-c2: CIFAR-100 visible through the symlink" || bad "V-c2 cifar-100-python not visible"
if [ -f "$RUNNER" ]; then
  DL=$(diff "$RUNNER_SRC" "$RUNNER" | grep '^[<>]')
  D=$(printf '%s\n' "$DL" | grep -c '^[<>]')
  if [ "$D" = 2 ] && echo "$DL" | grep -qx "< cd $LIVE" && echo "$DL" | grep -qx "> cd $DEST"; then
    echo "V-d: $RUNNER differs from run_cifar.sh in EXACTLY one line: cd $LIVE -> cd $DEST"
  else
    bad "V-d the runner diff is not exactly the cd line:"; diff "$RUNNER_SRC" "$RUNNER"
  fi
else
  bad "V-d no runner at $RUNNER"
fi
if python3 "$REPO/tests/test_plainnet.py" --pre "$DEST/build_network.py.pre_plainnet" --post "$DEST/build_network.py" \
     --hf "$DEST/Optimizers/HF.py" --csv "$REPO/results/all_runs.csv" >/tmp/cvt4_stage_tp.$$ 2>&1; then
  echo "V-e: tests/test_plainnet.py passes on the cvt4 tree ($(grep -c '^  PASS' /tmp/cvt4_stage_tp.$$) PASS)"
else
  bad "V-e tests/test_plainnet.py FAILS on the cvt4 tree"; grep '^  FAIL' /tmp/cvt4_stage_tp.$$ | head
fi
rm -f /tmp/cvt4_stage_tp.$$
if python3 "$REPO/tests/test_betahold.py" --pre "$DEST/Optimizers/HF.py.pre_betahold" --post "$DEST/Optimizers/HF.py" \
     --structural-only >/tmp/cvt4_stage_bh.$$ 2>&1; then
  echo "V-f: tests/test_betahold.py B0 (structure: deleting the four regions == pre, byte for byte) passes ($(grep -c '^  PASS' /tmp/cvt4_stage_bh.$$) PASS)"
else
  bad "V-f tests/test_betahold.py B0 FAILS on the cvt4 tree"; grep '^  FAIL' /tmp/cvt4_stage_bh.$$ | head
fi
rm -f /tmp/cvt4_stage_bh.$$
[ "$FAIL" = 0 ] && { echo "cVT4_stage_harness: VERIFIED"; exit 0; } || { echo "cVT4_stage_harness: NOT CONSISTENT"; exit 2; }
