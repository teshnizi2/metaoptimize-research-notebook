# cVT1_voteweight_score.py -- THE REGISTERED SCORER FOR `cvt1`:
#   SEPARATE IDENTITY FROM MAGNITUDE BY INTERVENING ON THE VOTE.  (CORRECTIONS 227)
#   ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS {78,79,80},
#   PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM,
#   run from the ISOLATED tree $WS/harness_cvt1 (PATCH_PLAINNET + PATCH_VOTEWEIGHT).
#
# Committed BEFORE any cvt1 run exists (STANDING RULE 21).  RUN IT UNEDITED
# (RULE 16).  The documented invocation takes ONE argument:
#
#     python3 analysis/cVT1_voteweight_score.py <runsdir>
#
# --manifest and --csv are OPTIONAL and DEFAULTED (the 164.2 pattern).
#
# =============================================================================
# FROZEN BARS (O2) -- READ THIS FIRST
# =============================================================================
# Every bar, sigma, tolerance and threshold score() uses is a LITERAL here.
# score() NEVER reads a bar term from results/all_runs.csv (it prints a row count
# as a disclosure and nothing else).  The ONLY data-dependent term that can enter
# a bar is SIGMA_INBATCH, through compose_sigma(), a max() with the frozen floor
# taking ONE argument.  --selftest section E proves it: the SAME synthetic runs
# give a byte-identical FINAL line (and identical SIGMA/SE/bar/PRIMARY lines)
# against the real corpus, the real corpus + INVENTED rows at THIS batch's own
# cells (PlainNet18_c100 `scalar` -- where k01, MUTE and DOSE all sit, since the
# CSV has no VOTE_W column -- and the HEAD/INJECT `sets:` string) and at the
# ResNet18_c100 cells the frozen sigma came from, invented rows only, and no corpus.
#
# =============================================================================
# THE QUESTION
# =============================================================================
# On PlainNet18_c100 (cpl1/cpl2) isolating layer4.1.bn2.weight (1-based idx 50 of
# 53) alone rescues the scalar collapse (cpl2 HEAD 64.18 vs k01 11.81); its bn1
# twin (idx 47) does not.  On k01's pinned records idx 50 holds 0.6143 of the
# mean-|L| mass, its twin 0.0009 (x691 less), and removing 50's term flips the
# sign of the shared sum on 1.0000 of records.  Isolation does TWO things at once:
# it REMOVES 50's VOTE from the shared sum, and it GIVES 50 ITS OWN STEP SIZE.
# And no carrier-free tensor votes with 50's magnitude, so set-matching cannot say
# whether the rescue is about WHICH tensor (identity) or HOW LOUD its vote is
# (magnitude).  cvt1 intervenes on the vote itself with PATCH_VOTEWEIGHT
# (z_G = sum_i w_i <h_i, g_i>, before the sign; momentum is linear in z so w_i
# reweights the tensor's whole vote).
#
# THE ARMS (every name re-derived on the LIVE 53-tensor model by the launcher)
#   k01     scalar                                   VOTE_W unset     the floor anchor
#   HEAD    sets:1-49,51-53/layer4.1.bn2.weight      VOTE_W unset     cpl2's HEAD, byte-
#           [52,1]  isolates 50                                        identical: POSITIVE CONTROL
#   MUTE    scalar  VOTE_W=layer4.1.bn2.weight:0     50 KEEPS the shared step size; its
#                                                    term's weight is 0 -- it no longer votes
#   DOSE    scalar  VOTE_W=layer4.1.bn2.weight:0.1   as MUTE, weight 0.1
#   INJECT  sets:1-49,51-53/layer4.1.bn2.weight      HEAD's grouping, PLUS the twin's term in
#           VOTE_W=layer4.1.bn1.weight:691           the complement multiplied by K = 691
#
# WHY K IS A FIXED FACTOR, NOT A PER-STEP MATCH TO THE LIVE TERM (CORRECTIONS 227.3)
#   * K = 691 is 226.4(3)'s figure: on cpl2 k01's 956 pinned records the pooled
#     mean-|L| ratio of idx 50 to idx 47 is 690.71 (per seed 641.0 / 696.2 / 743.0).
#     Measured in EVERY phase of k01's trajectory, the PER-RECORD median |L_50|/|L_47|
#     is 704 (epochs 0-5), 691 (5-17), 743 (17-36, the decline), 696 (36-100): a fixed
#     x691 reproduces the carrier's typical per-step magnitude in every phase within 8 %.
#   * A per-step match to 50's LIVE term in the INJECT run itself is broken by the
#     very isolation INJECT keeps: in cpl2's HEAD runs 50's own beta pins at the
#     clamp at epoch 22-23, and from then on |z_50|/|z_47| is 0.12-0.21 (not ~700).
#     "Match 50's live term" would SHRINK the twin's vote x5-9 for 78 % of training
#     -- the opposite of an injection (on HEAD's epoch 50-100 records it flips the
#     complement's Lion sign on 0.81-0.92 of records, almost all of them DOWN -> UP).
#   (analysis/cvt1_registration_derivations.py reproduces every number here.)
#   * Linear in z, so the momentum sees exactly K times the twin's term too; a
#     per-step ratio is nonlinear in the momentum.  Deterministic: one literal.
#   * The twin's own SIGN is preserved (w > 0).  On k01 z_47's sign agrees with z_50's
#     on 0.996 of records in epochs 17-36 and 1.000 after (0.960 in 0-5; 0.550 in
#     5-17, where 50 holds ~1.5 % of the mass and decides no record's sign).
#   What it cannot do: reproduce 50's term record-by-record.  Disclosed.
#
# WHAT DOSE = 0.1 CAN SHOW (registered, 227.3)
#   On cpl2 k01's records from epoch 17, the critical weight w* at which 50's
#   weighted term stops flipping the shared sign is q01 0.51 / median 0.60-0.63:
#   0.1 is far below it on EVERY record.  Under a SIGN-THRESHOLD vote account DOSE
#   therefore predicts MUTE's level, not a partial one; a partial DOSE says the
#   threshold moved along the intervened trajectory (graded), and DOSE at k01 says
#   a 10x smaller vote still collapses -- which contradicts the threshold account.
#
# =============================================================================
# THE REGISTERED ACCOUNTS (every arm's predicted LEVEL; all UNSURE)
# =============================================================================
# Anchors (cpl2, between batch, non-gating): k01 11.8120, HEAD 64.1767, kL 68.9380.
# K01_PL 8-16 | FREE_PL 56-72 | PART_PL 20-50 | COLLAPSE_PL 8-30.
#   VOTE-MAGNITUDE          removing 50's vote is the rescue; a vote of 50's size
#                           from the twin collapses the complement again.
#                           MUTE free, DOSE free (threshold), INJECT collapse.
#                                               -> VOTE-MAGNITUDE  (+ DOSE-AT-MUTE)
#   VOTE-MAGNITUDE-GRADED   as above, DOSE partial.   -> VOTE-MAGNITUDE (+ DOSE-PARTIAL)
#   IDENTITY-BEYOND-VOTE    removing 50's vote rescues, but an equally loud vote from
#                           a different tensor does NOT collapse: 50's vote matters
#                           because it is 50's.  MUTE, DOSE, INJECT free.
#                                               -> IDENTITY-BEYOND-VOTE
#   OWN-STEP-SIZE           the rescue needs 50 on its OWN step size; silencing its
#                           vote does nothing; injection harmless.  MUTE, DOSE at k01;
#                           INJECT free.        -> OWN-STEP-SIZE
#   STEP-AND-VOTE           own step size needed to rescue AND a loud vote suffices to
#                           collapse.  MUTE, DOSE at k01; INJECT collapse.
#                                               -> STEP-SIZE-NEEDED-VOTE-SUFFICES
#   BROKEN-PATCH (null)     the weights never reach the sum.  PREDICTS IDENTICALLY TO
#                           OWN-STEP-SIZE ON EVERY ARM (MUTE = DOSE = k01, INJECT = HEAD)
#                           -- DECLARED.  It is separated NOT by levels but by G-BITE,
#                           which reads the runs' own probe records: z_agg must equal
#                           sum_i w_i z_tensor_i at every record, and must NOT equal the
#                           unweighted sum where the weighted term is not negligible.
#                           A failure -> FINAL: PATCH-NOT-VERIFIED, no level is read.
#
# =============================================================================
# THE NOISE FLOOR -- RE-DERIVED AT REGISTRATION WITH `cvt1-` EXCLUDED, FROZEN
# =============================================================================
# results/all_runs.csv at 2,956 rows (sha 5c82c06c...), pooled within-cell SD of
# plateau5 at the standard cell (100 ep, AUGMENT=1, BETA_CLIP -15:-2.3026, ms 1e-3,
# alpha0 1e-6, batch 100; superseded / collapsed / incomplete dropped):
#   SIGMA_R18ALL  0.694442846939599     ResNet18_c100, every granularity (df 209)
#   SIGMA_PLAIN   0.39545713778624325   PlainNet18_c100 = cpl1 + cpl2 (df 23, 7 cells)
#   SIGMA_PRIOR := max = 0.694442846939599;  SE_ARM_DIFF at the prior 0.567010 pp.
#
# =============================================================================
# PRIMARY STATISTICS (co-primary; both IN BATCH; bounded in neither direction)
# =============================================================================
#   P_VOTE   = plateau5(HEAD) - plateau5(MUTE)    removing the vote vs own step size
#              VOTE-MAG / IDENTITY -16..+16 | OWN-STEP / STEP-AND-VOTE +40..+64
#   P_INJECT = plateau5(HEAD) - plateau5(INJECT)  does a carrier-sized vote re-collapse?
#              VOTE-MAG / STEP-AND-VOTE +26..+64 | IDENTITY / OWN-STEP -16..+16
# KEY SECONDARIES: D_HEAD = HEAD - k01 (positive control), D_MUTE = MUTE - k01,
#   D_DOSE = DOSE - k01, D_INJ = INJECT - k01.  DOSE_FRAC = D_DOSE / D_MUTE DESCRIPTIVE.
#
# =============================================================================
# THE BARS, FROZEN (pp; SE at the frozen prior 0.567010 in brackets)
# =============================================================================
#   GAP_MIN       20.0 [35.27]  D_HEAD >= 20 before any vote contrast is read
#   K01_MAX       20.0          k01 mean must be <= 20 (the collapse reproduces)

#   NULL_BAR       2.0 [ 3.53]  D_arm <= 2 is AT k01

#   COLLAPSE_BAR  10.0 [17.64]  HEAD - INJECT >= 10 -> the injection re-collapses
#   DIVERGED_BAR   5.0          within-arm seed range
#   FLOOR_MIN     15.0 / CEIL_MAX 90.0   max arm mean -> HARNESS-UNSOUND
#   G-BITE: BITE_TOL 1e-4 (relative to sum_i |z_tensor_i|); INFORM 1e-2; >= 50
#           informative records on every intervened run; 500 records on every run.
#
# =============================================================================
# HARNESS GATES (-> HARNESS-UNSOUND, rc 1)
# =============================================================================
#   G-ARGS   every run's OWN ARGS line (exactly one, no repeated flag) has its arm's
#            registered stepsize-groups, PlainNet18_c100, CIFAR100, 100 ep, batch

#   G-ENV    one distinct ENV line (PROBE_DIR stripped) == cpl2's; one

#   G-VOTEW  exactly one `VOTE_W:` line per run, == WITNESS[arm], byte for byte.
#   G-STRUCT PARTITION-MANIFEST.txt BYTE-IDENTICAL to synthetic_manifest_text().
#   G-PROV   PROVENANCE.txt: MODE submit; BUILD_NETWORK_SHA256 == POST_SHA;
#            HF_SHA256 == HF_POST_SHA (the PATCH_VOTEWEIGHT tree); SCORER_SHA256 == this file.
#   G-BITE   (-> PATCH-NOT-VERIFIED, rc 1) see above; read from <runsdir>/cvt1/probe_cvt1-<arm>-s<seed>/.
#
# =============================================================================
# THE BRANCH MAP, FROZEN (decide(); first match wins)
# =============================================================================
#   INCOMPLETE / HARNESS-UNSOUND / PATCH-NOT-VERIFIED   (above)
#   UNRESOLVED-DIVERGED             an arm's seed range > 5

# STAMPS (never the branch): HARNESS-CLEAN; PATCH-BITES (G-BITE passed: the

def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def file_sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()


# =============================================================================
# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.
# =============================================================================
def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

# the frozen manifest, composed from ResNet-18's layout (no torch).  The
# launcher's guard 4d proves the LIVE manifest BYTE-IDENTICAL to this text.

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists (each sorted); no HF.py import."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names):
    """1-based groups in beta order for an arm's spec."""
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest INTEGER job id among COMPLETE runs wins; a
    truncated resubmission cannot shadow a complete run."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, any_gran=False):
    return ((any_gran or r.get("granularity") in ("scalar", "layerwise"))
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra, any_gran=False):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, any_gran) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm


def net_is(net):
    return lambda r: r.get("network") == net and r.get("dataset") == DSET


def prefix_arm_mean(rows, prefix, gran, net):
    v = [float(r["plateau5"]) for r in rows if r.get("run", "").startswith(prefix + "-")
         and usable(r) and std_cell(r, True) and r.get("network") == net and r.get("granularity") == gran]
    return (mean(v), len(v)) if v else (None, 0)

def load_probe(runsdir, arm, seed):
    """-> list of records, or None if the file is absent.  Unparseable lines are
    counted as None entries so a garbled file cannot pass the record-count check."""
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            return recs
    return None


def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail dict).  Pure function of the records."""
    names = [n for n, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    W = WEIGHTS[arm]
    d = {"n": 0, "bad_rec": 0, "bad_z": 0, "bad_m": 0, "worst_z": 0.0, "worst_m": 0.0,
         "informative": 0, "unweighted_match_on_informative": 0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for r in recs:
        ok = (r is not None and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS
              and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
              and isinstance(r.get("z_agg"), list) and len(r["z_agg"]) == 1 and isinstance(r["z_agg"][0], list)
              and len(r["z_agg"][0]) == len(groups)
              and isinstance(r.get("mom_pre"), list) and len(r["mom_pre"]) == 1 and isinstance(r["mom_pre"][0], list)
              and len(r["mom_pre"][0]) == len(groups))
        if not ok:
            d["bad_rec"] += 1
            continue
        try:
            zt = [float(x) for x in r["z_tensor"]]
            mt = [float(x) for x in r["m_tensor"]]
            za = [float(x) for x in r["z_agg"][0]]
            ma = [float(x) for x in r["mom_pre"][0]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in zt + mt + za + ma):
            d["bad_rec"] += 1
            continue
        d["n"] += 1
        sz = sum(abs(x) for x in zt) + 1e-30
        sm = sum(abs(x) for x in mt) + 1e-30
        wz = [sum(W.get(i, 1.0) * zt[i - 1] for i in g) for g in groups]
        wm = [sum(W.get(i, 1.0) * mt[i - 1] for i in g) for g in groups]
        uz = [sum(zt[i - 1] for i in g) for g in groups]
        ez = max(abs(p - q) for p, q in zip(wz, za)) / sz
        em = max(abs(p - q) for p, q in zip(wm, ma)) / sm
        d["worst_z"], d["worst_m"] = max(d["worst_z"], ez), max(d["worst_m"], em)
        d["bad_z"] += ez > BITE_TOL
        d["bad_m"] += em > BITE_TOL
        if W and max(abs(p - q) for p, q in zip(wz, uz)) / sz > INFORM:
            d["informative"] += 1
            d["unweighted_match_on_informative"] += max(abs(p - q) for p, q in zip(uz, za)) / sz <= BITE_TOL
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] == N_RECORDS
          and d["bad_z"] == 0 and d["bad_m"] == 0)
    if W:
        ok = ok and d["informative"] >= MIN_INFORMATIVE and d["unweighted_match_on_informative"] == 0
    return ok, d

def beta_summary(recs, coord):
    vals = [(r["step"], r["beta"][coord]) for r in recs
            if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord
            and isinstance(r.get("step"), int)]
    if not vals:
        return None
    at = [v <= CLAMP_LO + 1e-4 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    return {"last": vals[-1][1], "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH),
            "frac_clamp": sum(at) / float(len(at)), "pin_epoch": pin, "n": len(vals)}


def descriptive_probe_readout(runsdir):
    print("\nDESCRIPTIVE, NON-GATING -- beta trajectories from the runs' own probe.jsonl")
    print("  (cpl2: k01 pinned at epoch ~36; HEAD's complement at 98.2/98.8/never, its isolated beta at ~22.)")
    for a in ARMS:
        for s in SEEDS:
            recs = load_probe(runsdir, a, s)
            if not recs:
                continue
            for coord, lab in ((0, "beta[0]" if PT_TYPE[a] == "scalar" else "complement beta[0]"),
                               (1, "isolated beta[1]")):
                if PT_TYPE[a] == "scalar" and coord == 1:
                    continue
                bs = beta_summary(recs, coord)
                if bs:
                    print("  %-6s s%d  %-19s last %8.3f  peak %8.3f @ep %5.1f  at-clamp frac %.3f  pinned-from-epoch %s"
                          % (a, s, lab, bs["last"], bs["peak"], bs["peak_epoch"], bs["frac_clamp"],
                             "never" if bs["pin_epoch"] is None else "%.1f" % bs["pin_epoch"]))
    print("\nDESCRIPTIVE, NON-GATING -- the vote on THIS batch's records (L_i = b2*m_i + (1-b2)*z_i, raw terms)")
    for a, lab in (("k01", "k01 pinned records"), ("MUTE", "MUTE records, epoch >= 17")):
        for s in SEEDS:
            recs = load_probe(runsdir, a, s) or []
            sel = [r for r in recs if isinstance(r, dict) and isinstance(r.get("m_tensor"), list)
                   and len(r["m_tensor"]) == NTENS and isinstance(r.get("z_tensor"), list)
                   and len(r["z_tensor"]) == NTENS and isinstance(r.get("beta"), list) and r["beta"]
                   and isinstance(r.get("step"), int)
                   and ((r["beta"][0] <= CLAMP_LO + 1e-4) if a == "k01" else r["step"] >= 17 * STEPS_PER_EPOCH)]
            if not sel:
                continue
            m50 = m47 = 0.0
            flip = 0
            for r in sel:
                b2 = r.get("pt_b2", 0.9)
                L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
                m50 += abs(L[IDX_HEAD - 1])
                m47 += abs(L[IDX_TWIN - 1])
                S = sum(L)
                if a == "k01":
                    Sr = S - L[IDX_HEAD - 1]
                else:
                    S = S - L[IDX_HEAD - 1]          # what MUTE's harness summed
                    Sr = S + L[IDX_HEAD - 1]         # adding 50's vote back
                flip += ((S > 0) - (S < 0)) != ((Sr > 0) - (Sr < 0))
            print("  %-26s s%d  %3d records  mean|L50|/mean|L47| %s  %s %.4f"
                  % (lab, s, len(sel), fmt(m50 / m47 if m47 else None, 1),
                     "removing 50 flips" if a == "k01" else "adding 50 back flips", flip / float(len(sel))))

# THE BRANCH MAP -- a pure function of the in-batch means and one flag
# =============================================================================
def decide(M, diverged=False):
    D_HEAD = M["HEAD"] - M["k01"]
    D_MUTE = M["MUTE"] - M["k01"]
    P_VOTE = M["HEAD"] - M["MUTE"]
    P_INJECT = M["HEAD"] - M["INJECT"]
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if D_HEAD < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED"
    if -P_INJECT > MATCH_BAR:
        return "INJECT-ABOVE-HEAD"
    inj_holds = abs(P_INJECT) <= MATCH_BAR
    inj_collapse = P_INJECT >= COLLAPSE_BAR
    if D_MUTE >= RESCUE_BAR:
        if P_VOTE > MATCH_BAR:
            return "VOTE-REMOVAL-PARTIAL"
        if inj_collapse:
            return "VOTE-MAGNITUDE"
        if inj_holds:
            return "IDENTITY-BEYOND-VOTE"
        return "VOTE-MAGNITUDE-ATTENUATED"
    if D_MUTE <= NULL_BAR:
        if inj_holds:
            return "OWN-STEP-SIZE"
        if inj_collapse:
            return "STEP-SIZE-NEEDED-VOTE-SUFFICES"
        return "OWN-STEP-SIZE-ATTENUATED"
    return "MUTE-WEAK"


def dose_stamp(M):
    if M["DOSE"] - M["k01"] <= NULL_BAR:
        return "DOSE-AT-K01"
    if abs(M["MUTE"] - M["DOSE"]) <= MATCH_BAR:
        return "DOSE-AT-MUTE"
    return "DOSE-PARTIAL"

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    print("=" * 78)
    print(" cvt1 -- SEPARATE IDENTITY FROM MAGNITUDE BY INTERVENING ON THE VOTE  (PlainNet18_c100)")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-6s %-38s VOTE_W=%s" % (a, SPEC[a], VOTEW[a] or "(unset)"))
    print(" PRIMARY (co-primary): P_VOTE = HEAD - MUTE;  P_INJECT = HEAD - INJECT.")
    print("          plateau5 = mean TEST over epochs %d..%d of each run's own .out." % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2); the corpus cannot move them.")
    print("=" * 78)
    rows = read_corpus(csvpath)
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)"
          % ("not readable" if rows is None else "%d rows, %s- excluded" % (len(rows), PREFIX)))

    runs, rnotes = read_runs(runsdir)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"

                missing.append("%s-s%d" % (a, s))

def _e_probe_records(arm, weights=None, n=N_RECORDS, seed=0):
    """synthetic PROBE_TENSOR records consistent with `weights` (default: the arm's own)."""
    import random
    rnd = random.Random(1000 + seed)
    W = WEIGHTS[arm] if weights is None else weights
    names = [nm for nm, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    out = []
    for k in range(n):
        zt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        mt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        zt[IDX_HEAD - 1] = 5e-2 * (1 + 0.1 * rnd.random())
        mt[IDX_HEAD - 1] = 4e-2
        zt[IDX_TWIN - 1] = 1e-4 * (1 + 0.1 * rnd.random())
        mt[IDX_TWIN - 1] = 8e-5
        za = [sum(W.get(i, 1.0) * zt[i - 1] for i in g) for g in groups]
        ma = [sum(W.get(i, 1.0) * mt[i - 1] for i in g) for g in groups]
        b = max(-15.0, -6.0 - 0.02 * k)
        out.append({"step": 100 * k, "beta": [b] * len(groups), "pt_b2": 0.9, "pt_mp": 0.99,
                    "z_tensor": zt, "m_tensor": mt, "z_agg": [za], "mom_pre": [ma],
                    "beta_pre": [[b] * len(groups)]})
    return out

             env_break=None, spec_break=None, vw_break=None, manifest=None, prov=None,

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

    m = re.search(r"^SPEC = (\{.*?^\})", open(p).read(), re.S | re.M)

                           stderr=subprocess.STDOUT, universal_newlines=True, timeout=600)
        fl = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 2 and fl and fl[-1].startswith("FINAL: INCOMPLETE"),
            "... and on a dropped run exits 2 with FINAL: INCOMPLETE", "rc %d" % p.returncode)
    except Exception as e:
        chk(False, "subprocess invocation", "%s: %s" % (type(e).__name__, e))

    print("\nF. THE BRANCH MAP IS TOTAL -- 4,000 random in-batch means, every branch registered")
    import random
    rnd = random.Random(20260916)
    seen = set()
    for _ in range(4000):
        m = dict((a, rnd.uniform(1.0, 80.0)) for a in ARMS)
        seen.add(decide(m, rnd.random() < 0.05))
        dose_stamp(m)
    chk(seen <= set(LICENSE), "every branch decide() returns has a registered licence text", "%d distinct" % len(seen))
    reached = set(r[1].split(" |")[0][len("FINAL: "):] for r in results.values())
    chk(reached | seen == set(LICENSE), "every registered branch is reached by section E or the sweep",
        "missing %s" % sorted(set(LICENSE) - reached - seen))

    print("\nG. RULE 21 PREMISE: no cvt1 run may exist yet")
    runs, _n = read_runs(runsdir) if runsdir else ({}, [])
    chk(not runs, "no cvt1-*.out under %s" % runsdir, "%d found" % len(runs))
    if csvpath and os.path.exists(csvpath):
        chk(not any(r.get("run", "").startswith(PREFIX + "-") for r in csv.DictReader(open(csvpath))),
            "no cvt1- row in the CSV")
    shutil.rmtree(tmp, ignore_errors=True)

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))
    for f in GATE_FAIL:
        print("   FAIL", f)
    return 1 if GATE_FAIL else 0
