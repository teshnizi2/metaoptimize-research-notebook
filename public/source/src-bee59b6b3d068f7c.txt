def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def file_sha(path):

# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.
# =============================================================================
def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


# =============================================================================
# the frozen manifest, composed from ResNet-18's definition (no torch).  The
# launcher's guard 4d proves the LIVE manifest BYTE-IDENTICAL to this text.

def gn_tensors():
    out = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "GroupNorm"),
           ("bn1.bias", 64, "GroupNorm")]
    cin = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for bi in range(2):
            stride2 = (li > 1 and bi == 0)
            p = "layer%d.%d." % (li, bi)
            a = cin if bi == 0 else w
            out += [(p + "conv1.weight", w * a * 9, "Conv2d"), (p + "bn1.weight", w, "GroupNorm"),
                    (p + "bn1.bias", w, "GroupNorm"), (p + "conv2.weight", w * w * 9, "Conv2d"),
                    (p + "bn2.weight", w, "GroupNorm"), (p + "bn2.bias", w, "GroupNorm")]
            if stride2:
                out += [(p + "shortcut.0.weight", w * a, "Conv2d"),
                        (p + "shortcut.1.weight", w, "GroupNorm"),
                        (p + "shortcut.1.bias", w, "GroupNorm")]
        cin = w
    out += [("linear.weight", 100 * 512, "Linear"), ("linear.bias", 100, "Linear")]
    return out


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists (each sorted), re-implemented
    HERE so the registered index sets are checked without importing HF.py."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))

# run readers -- the RAW .out files are PRIMARY

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  Among several files for one (arm, seed) the highest
    INTEGER job id among COMPLETE runs wins; a truncated resubmission cannot shadow
    a complete run.  -> (runs, notes)"""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, any_gran=False):
    return ((any_gran or r.get("granularity") in ("scalar", "layerwise"))
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra, any_gran=False):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, any_gran) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm


def net_is(net):
    return lambda r: r.get("network") == net and r.get("dataset") == DSET


def prefix_arm_mean(rows, prefix, gran, net):
    v = [float(r["plateau5"]) for r in rows if r.get("run", "").startswith(prefix + "-")
         and usable(r) and std_cell(r, True) and r.get("network") == net and r.get("granularity") == gran]
    return (mean(v), len(v)) if v else (None, 0)

def load_probe(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    continue
                if isinstance(r.get("beta"), list) and isinstance(r.get("step"), int):
                    recs.append(r)
            return recs
    return None


def beta_summary(recs, coord):
    vals = [(r["step"], r["beta"][coord]) for r in recs if len(r["beta"]) > coord]
    if not vals:
        return None
    at = [v <= CLAMP_LO + 1e-4 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    return {"last": vals[-1][1], "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH),
            "frac_clamp": sum(at) / float(len(at)), "pin_epoch": pin, "n": len(vals)}


NOM_SETS = {"50_53_59": [50, 53, 59], "47_48_56": [47, 48, 56], "50": [50], "59": [59]}


def postpeak_nomination(recs):
    """217.6's DISCLOSED post-peak method on k01 records (1-based sets)."""
    ok = [r for r in recs if isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
          and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS and len(r["beta"]) == 1]
    if not ok:
        return None
    ipk = max(range(len(ok)), key=lambda i: ok[i]["beta"][0])
    post = ok[ipk + 1:]
    if not post:
        return None
    mass = [0.0] * NTENS
    flips = dict((k, 0) for k in NOM_SETS)
    down = 0
    for r in post:
        b2 = r.get("pt_b2", 0.9)
        L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
        S = sum(L)
        s0 = (S > 0) - (S < 0)
        down += s0 > 0            # a positive L votes beta DOWN (Lion subtracts ms*sign)
        for i, v in enumerate(L):
            mass[i] += abs(v)
        for k, idx in NOM_SETS.items():
            Sr = S - sum(L[i - 1] for i in idx)
            if ((Sr > 0) - (Sr < 0)) != s0:
                flips[k] += 1
    tm = sum(mass) or 1.0
    return {"n": len(post), "share": [x / tm for x in mass],
            "flip": dict((k, v / float(len(post))) for k, v in flips.items()),
            "down": down / float(len(post))}

# =============================================================================
def decide(M, trains_ok=True, diverged=False):
    D_GAP = M["kL"] - M["k01"]
    D_ISO = M["ISO"] - M["k01"]
    D_CTL = M["CTL"] - M["k01"]
    D_ONE = M["ONE"] - M["k01"]
    DELTA_ID = M["ISO"] - M["CTL"]
    if not trains_ok:
        return "GN-UNTRAINABLE"
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if D_GAP < GAP_MIN:
        return "GAP-NOT-REPRODUCED-GN"
    rI, rC, rO = D_ISO >= RESCUE_BAR, D_CTL >= RESCUE_BAR, D_ONE >= RESCUE_BAR
    if not (rI or rC or rO):
        return "NO-RESCUE-GN"
    if not rI:
        return "UNEXPECTED-PATTERN"
    if DELTA_ID <= -IDENTITY_BAR:
        return "CONTROL-DOMINATES-GN"
    if DELTA_ID >= IDENTITY_BAR:
        if rO:
            return "IDENTITY-TRANSFERS-GN"
        if D_ONE <= NULL_BAR:
            return "IDENTITY-TRANSFERS-ONE-NULL-GN"
        return "IDENTITY-TRANSFERS-ONE-PARTIAL-GN"
    if rC and abs(DELTA_ID) <= NULL_BAR:
        return "CLASS-OPERATIVE-GN"
    return "IDENTITY-ATTENUATED-GN"

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def _spec_from_scorer(fname, key):
    p = os.path.join(os.path.dirname(os.path.abspath(__file__)), fname)
    if not os.path.exists(p):
        return None
    import ast
    m = re.search(r"^SPEC = (\{.*?^\})", open(p).read(), re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)

    iso3, ctl3 = parse_sets(SPEC["ISO"], names)[1], parse_sets(SPEC["CTL"], names)[1]
    chk(len(iso3) == len(ctl3) == 3 and all(numel[i - 1] == 512 for i in iso3 + ctl3)
        and all(names[i - 1].startswith("layer4.") for i in iso3 + ctl3)
        and not set(iso3) & set(ctl3), "ISO and CTL: disjoint, cardinality 3/3, numel 1536/1536, all layer4")
    chk([names[i - 1].endswith(".bias") for i in ctl3] == [False, True, False],
        "CTL contains ONE GN shift (48 layer4.0.bn1.bias) -- matched on numel, NOT class (stamped)")
    chk(set(parse_sets(SPEC["ONE"], names)[1]) < set(iso3), "ONE's tensor (50) is a member of ISO's triple")
    chk(set(iso3) == {50, 53, 59} and {47, 56} <= set(ctl3),
        "ISO = 217.6's nominated {50,53,59}; CTL contains 217.6's matched pair {47,56}")
    live_man = os.path.join(runsdir, PREFIX, "PARTITION-MANIFEST.txt") if runsdir else ""
    if live_man and os.path.exists(live_man):
        chk(open(live_man).read() == synthetic_manifest_text(),
            "B the LIVE manifest guard 4 wrote is BYTE-IDENTICAL to synthetic_manifest_text()")
    else:
        note("no live manifest under %s/%s -- the launcher's guard 4d makes the byte comparison" % (runsdir, PREFIX))

    print("\nC. CORPUS PREMISES, re-derived (NOTES on drift -- under O2 no drift reaches a bar)")
    rows = read_corpus(csvpath)
    if rows is None:
        skip("corpus premises -- %s not readable" % csvpath)
    else:
        sha = file_sha(csvpath)
        reg = sha == REG_CORPUS_SHA256
        note("corpus %d rows (cgn2 excluded), sha %s... -- %s" % (len(rows), sha[:12],
             "THE REGISTRATION CORPUS" if reg else "not the registration corpus: drift is NOTED, never failed"))
        for nm, (s, df_, nc, nm_), frozen in (
                ("SIGMA_R18ALL", pooled_sigma(rows, net_is(REF_NET), True), SIGMA_R18ALL),
                ("SIGMA_GNC100", pooled_sigma(rows, net_is(NET), True), SIGMA_GNC100)):
            print("       %s live %s  df %s  cells %s  members %s" % (nm, fmt(s, 12), df_, nc, nm_))
            if reg:
                chk(s is not None and abs(s - frozen) < 1e-12, "%s re-derives == frozen %.15g" % (nm, frozen))
            else:
                note("%s live %s vs frozen %.15g -- feeds no bar (O2)" % (nm, fmt(s, 12), frozen))
        for pre, lab, net, frozen in (("cgn1", "scalar", NET, R_CGN1["k01"]), ("cgn1", "layerwise", NET, R_CGN1["kL"]),
                                      ("ciso1", SPEC["ISO"], REF_NET, R_CISO1["ISO"]),
                                      ("ciso1", SPEC["ONE"], REF_NET, R_CISO1["ONE"]),
                                      ("cdep1", SPEC["CTL"], REF_NET, R_CDEP1["DEPTH"]),
                                      ("cdep1", "scalar", REF_NET, R_CDEP1["k01"])):
            v, n = prefix_arm_mean(rows, pre, lab, net)
            if v is not None and abs(v - frozen) < 5e-5:
                chk(n == 3, "frozen reference %s %-10s re-derives == %.4f (n=3)" % (pre, lab[:10], frozen))
            else:
                note("frozen reference %s %s: %.4f, live %s (non-gating)" % (pre, lab[:12], frozen, fmt(v)))
        n_gn = [r["run"] for r in rows if r.get("network") == NET]
        chk(all(x.startswith("cgn1-") for x in n_gn), "%s appears ONLY in cgn1's rows (%d)" % (NET, len(n_gn)))
        chk(not any(r.get("seed") in ("72", "73", "74") for r in rows), "ZERO corpus rows carry seed 72, 73 or 74")

    print("\nD. THE FLOOR TABLE (164.6).  Chance %.2f pp, ceiling 100.  Levels UNSURE." % CHANCE)
    print("   'at-k01' = predicted at the IN-BATCH k01 LEVEL (not a clamp: cgn1's GN scalar never pinned)")
    lo_all = 100.0
    for acc, pred in ACCOUNTS.items():
        print("  %-9s" % acc)
        for a in ARMS:
            lo, hi = pred[a]
            role = ("the in-batch floor ANCHOR" if a == "k01" else
                    "at-k01" if pred[a] == pred["k01"] else
                    "OFF k01 by >= %.0f pp" % (lo - pred["k01"][1]))
            print("      %-4s %4.0f-%-4.0f  above chance %+5.1f..%+5.1f  below ceiling %5.1f..%5.1f  %s"
                  % (a, lo, hi, lo - CHANCE, hi - CHANCE, 100 - hi, 100 - lo, role))
            lo_all = min(lo_all, lo)
        dlo = pred["ISO"][0] - pred["CTL"][1]
        dhi = pred["ISO"][1] - pred["CTL"][0]
        print("      PRIMARY DELTA_ID predicted %+.0f..%+.0f  (hard range +-%.0f)" % (dlo, dhi, 100 - CHANCE))
        chk(dhi <= (100 - CHANCE) - 25 and dlo >= -(100 - CHANCE) + 25,
            "%-9s DELTA_ID's predicted interval is >= 25 pp inside BOTH hard bounds (bounded in neither direction)" % acc)
        off = [a for a in BLOCKWISE_ARMS if pred[a] != pred["k01"]]
        if acc in ("DIFFUSE", "H-BROKEN"):
            chk(not off, "%-9s puts every m=2 arm at k01 -- declared: NO-RESCUE cannot confirm it" % acc)
        else:
            chk(bool(off) and all(pred[a][0] - pred["k01"][1] >= 2 * RESCUE_BAR for a in off),
                "%-9s has an off-k01 m=2 arm >= %.0f pp clear of k01 (kills H-BROKEN in batch)" % (acc, 2 * RESCUE_BAR),
                ",".join(off))
        mid = dict((a, (pred[a][0] + pred[a][1]) / 2.0) for a in ARMS)
        chk(decide(mid) == EXPECTED_BRANCH[acc], "%-9s midpoint prediction lands in %s" % (acc, EXPECTED_BRANCH[acc]),
            decide(mid))
    chk(lo_all - CHANCE >= 8.0, "the LOWEST level ANY account predicts for ANY arm is >= 8 pp above chance",
        "%.1f pp" % (lo_all - CHANCE))
    chk(ACCOUNTS["DIFFUSE"] == ACCOUNTS["H-BROKEN"], "DIFFUSE and H-BROKEN predict IDENTICALLY on every arm (declared)")
    chk(all(pred["kL"][0] - pred["k01"][1] >= GAP_MIN + 5 for pred in ACCOUNTS.values()),
        "every account predicts D_GAP >= GAP_MIN + 5 (the carrier branches are reachable)")

    print("\nE. THE REAL score() ON SYNTHETIC RUNS (invented numbers; evidence of nothing)")
    tmp = tempfile.mkdtemp(prefix="cgn2_selftest_")
    bnman = synthetic_manifest_text().replace("GroupNorm", "BatchNorm2d")
    cases = [
        ("T-IDENT", {}, {}, "IDENTITY-TRANSFERS-GN"),
        ("T-SET", {"ONE": 14.6}, {}, "IDENTITY-TRANSFERS-ONE-NULL-GN"),
        ("one-partial", {"ONE": 20.0}, {}, "IDENTITY-TRANSFERS-ONE-PARTIAL-GN"),
        ("CLASS", {"CTL": 49.5, "ONE": 45.0}, {}, "CLASS-OPERATIVE-GN"),
        ("DIFFUSE/H-BROKEN", {"ISO": 15.0, "ONE": 14.2}, {}, "NO-RESCUE-GN"),
        ("gap not reproduced", {"kL": 32.0, "ISO": 31.0}, {}, "GAP-NOT-REPRODUCED-GN"),
        ("untrainable", {"k01": 14.0, "kL": 25.0, "ISO": 24.0, "CTL": 14.1, "ONE": 14.2}, {}, "GN-UNTRAINABLE"),
        ("unexpected", {"ISO": 15.5, "CTL": 40.0}, {}, "UNEXPECTED-PATTERN"),
        ("control dominates", {"ISO": 30.0, "CTL": 50.0}, {}, "CONTROL-DOMINATES-GN"),
        ("identity attenuated", {"ISO": 45.0, "CTL": 36.0}, {}, "IDENTITY-ATTENUATED-GN"),
        ("near-bar ONE just rescues", {"ONE": 14.5 + RESCUE_BAR + 0.05}, {}, "IDENTITY-TRANSFERS-GN"),
        ("near-bar ONE just misses", {"ONE": 14.5 + RESCUE_BAR - 0.05}, {}, "IDENTITY-TRANSFERS-ONE-PARTIAL-GN"),
        ("near-bar gap", {"kL": 14.5 + GAP_MIN - 0.05, "ISO": 33.0}, {}, "GAP-NOT-REPRODUCED-GN"),
        ("diverged", {}, {"over": {("kL", 74): 40.0}}, "UNRESOLVED-DIVERGED"),
        ("harness: every arm tiny", dict((a, 3.0) for a in ARMS), {}, "HARNESS-UNSOUND"),
        ("harness: AUGMENT=0 on one run", {}, {"env_break": ("CTL", 74)}, "HARNESS-UNSOUND"),
        ("harness: ARGS spec != arm", {}, {"spec_break": ("ISO", 73)}, "HARNESS-UNSOUND"),
        ("harness: ARGS name the BN net", {}, {"net_break": ("ONE", 72)}, "HARNESS-UNSOUND"),
        ("harness: manifest is a BN net", {}, {"manifest": bnman}, "HARNESS-UNSOUND"),
        ("harness: provenance is the live tree", {}, {"prov": {"bn": PRE_SHA}}, "HARNESS-UNSOUND"),
        ("harness: provenance dry-run", {}, {"prov": {"mode": "dry-run"}}, "HARNESS-UNSOUND"),
        ("harness: provenance names another scorer", {}, {"prov": {"scorer": "0" * 64}}, "HARNESS-UNSOUND"),
        ("a DROPPED run", {}, {"drop": ("ONE", 73)}, "INCOMPLETE"),
        ("a TRUNCATED higher-jid resubmission", {}, {"shadow": ("ISO", 72)}, "IDENTITY-TRANSFERS-GN"),
    ]
    results = {}
    for lab, lv, kw, want in cases:
        d = os.path.join(tmp, re.sub(r"\W+", "_", lab))
        os.makedirs(d)
        _e_build(d, levels=lv, **kw)
        rc, fin, out = _e_score(d, csvpath)
        results[lab] = (rc, fin, out, d)
        chk(fin.startswith("FINAL: %s |" % want), "E %-40s -> %s" % (lab, want), fin[:80])
    rc, fin, out, dpath = results["T-IDENT"]
    chk(rc == 0 and "HARNESS-CLEAN" in fin and "H-BROKEN-EXCLUDED" in fin and "K01-IS-A-LEVEL-NOT-A-CLAMP" in fin
        and "HORIZON-100-ONLY" in fin and "RESCUE-PERSISTENCE-NOT-TESTED" in fin,
        "E T-IDENT exits 0 with HARNESS-CLEAN, H-BROKEN-EXCLUDED and the unconditional stamps")
    chk(results["a DROPPED run"][0] == 2, "E a DROPPED run exits 2 (INCOMPLETE)")
    chk(results["harness: AUGMENT=0 on one run"][0] == 1, "E a harness failure exits 1")
    chk(results["a TRUNCATED higher-jid resubmission"][1] == results["T-IDENT"][1],
        "E a truncated higher-jid resubmission does not shadow a complete run")
    chk("H-BROKEN-NOT-EXCLUDED" in results["DIFFUSE/H-BROKEN"][1]
        and "FLOOR-READINGS-ARE-BOUNDS" in results["DIFFUSE/H-BROKEN"][1],
        "E NO-RESCUE stamps H-BROKEN-NOT-EXCLUDED and FLOOR-READINGS-ARE-BOUNDS")
    chk("DECOMPOSITION-BY-SUBTRACTION" in results["T-SET"][1], "E ONE-NULL stamps DECOMPOSITION-BY-SUBTRACTION")
    d = os.path.join(tmp, "probe")
    os.makedirs(d)
    _e_build(d)
    _e_probe(d)
    rc, fin, out = _e_score(d, csvpath)
    chk(fin == results["T-IDENT"][1] and "post-peak records" in out and "pinned-from-epoch never" in out,
        "E never-pinned probe records: the readout runs, reports 'never', and FINAL does not move")
    d2 = os.path.join(tmp, "probe_garbled")
    os.makedirs(d2)
    _e_build(d2)
    _e_probe(d2)
    open(os.path.join(d2, PREFIX, "probe_cgn2-k01-s72", "probe.jsonl"), "w").write("{not json\n\x00\x01\n")
    rc, fin, out = _e_score(d2, csvpath)
    chk(rc == 0 and fin == results["T-IDENT"][1], "E a GARBLED probe file changes nothing in FINAL or rc")

    print("\n   O2 INVARIANCE: the SAME synthetic runs against four corpora")
    variants = _e_csv_variants(csvpath if csvpath and os.path.exists(csvpath) else None, tmp)
    for case in ("T-IDENT", "DIFFUSE/H-BROKEN"):
        ref_fin, ref_bars = None, None
        dd = results[case][3]
        for lab in ("real", "real+invented", "minimal-invented", "no-corpus"):
            if lab not in variants:
                skip("O2 %s corpus not available on this host" % lab)

            rc, fin, out = _e_score(dd, variants[lab])

            if ref_fin is None:
                ref_fin, ref_bars = fin, bars
            chk(fin == ref_fin and bars == ref_bars and len(bars) == 4,
                "O2 %-16s %-17s FINAL and SIGMA/SE/bars/PRIMARY lines are IDENTICAL" % (case, lab), fin[:44])
    if "real+invented" in variants:
        a_ = pooled_sigma(read_corpus(variants["real"]), net_is(NET), True)[0]

            "O2 non-vacuity: the invented rows MOVE the live GN corpus sigma (%.4f -> %.4f) and FINAL did not move"

    print("\n   THE DOCUMENTED ONE-ARGUMENT INVOCATION, as a subprocess")
    me = os.path.abspath(__file__)
    try:
        p = subprocess.run([sys.executable, me, dpath], stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                           universal_newlines=True, timeout=300)
        fl = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 0 and fl and fl[-1] == results["T-IDENT"][1],
            "`python3 %s <runsdir>` exits 0 with the in-process FINAL line" % os.path.basename(me),
            "rc %d" % p.returncode)
        p = subprocess.run([sys.executable, me, results["a DROPPED run"][3]], stdout=subprocess.PIPE,

        chk(p.returncode == 2 and fl and fl[-1].startswith("FINAL: INCOMPLETE"),
            "... and on a dropped run exits 2 with FINAL: INCOMPLETE", "rc %d" % p.returncode)
    except Exception as e:
        chk(False, "subprocess invocation", "%s: %s" % (type(e).__name__, e))

    print("\nF. THE BRANCH MAP IS TOTAL -- 4,000 random in-batch means, every branch registered")
    import random
    rnd = random.Random(20260916)
    seen = set()
    for _ in range(4000):
        m = dict((a, rnd.uniform(1.0, 80.0)) for a in ARMS)
        seen.add(decide(m, rnd.random() > 0.05, rnd.random() < 0.05))
    chk(seen <= set(LICENSE), "every branch decide() returns has a registered licence text", "%d distinct" % len(seen))
    reached = set(r[1].split(" |")[0][len("FINAL: "):] for r in results.values())

        "missing %s" % sorted(set(LICENSE) - reached - seen))

    runs, _n = read_runs(runsdir) if runsdir else ({}, [])

        chk(not any(r.get("run", "").startswith(PREFIX + "-") for r in csv.DictReader(open(csvpath))),
            "no cgn2- row in the CSV")

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))

def main():
    ap = argparse.ArgumentParser(description="cgn2 registered scorer.  Documented invocation: "
                                             "python3 analysis/cGN2_gn_isolation_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cgn2/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
    sys.exit(main())
