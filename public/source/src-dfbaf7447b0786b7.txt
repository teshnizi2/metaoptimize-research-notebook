def f32(x):
    return struct.unpack("<f", struct.pack("<f", x))[0]


B0 = f32(math.log(1e-6))


def mean(v):
    return math.fsum(v) / len(v)


def sdev(v):
    m = mean(v)
    return math.sqrt(math.fsum((x - m) ** 2 for x in v) / (len(v) - 1))


def ols(xs, ys):
    xm, ym = mean(xs), mean(ys)
    return math.fsum((x - xm) * (y - ym) for x, y in zip(xs, ys)) / math.fsum((x - xm) ** 2 for x in xs)


def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def sgn(x):
    return (x > 0) - (x < 0)


def tri_raw(n, P):
    """237.3(1): u(n) = min(n-1, P) - max(0, n-1-P) for n >= 1, else 0; clamp(b0 + ms u) (double)."""
    u = 0 if n <= 0 else min(n - 1, P) - max(0, n - 1 - P)
    return min(HI, max(LO, B0 + MS * u))


def tri(n, P):
    return f32(tri_raw(n, P))


def sched50(arm, n, cut=None):
    """249.2 / 249.4: 50's applied value at update n: floor, or tri:P, or tri:9428 inside [n0, n1) and the floor outside."""
    mode, P = HOLD50[arm]
    if mode == "floor":
        return f32(LO)
    if arm in WINDOWS:
        n0, n1 = WINDOWS[arm]
        if cut is not None:
            n0, n1 = (0, cut) if arm == "EARLY" else (cut, None)
        inside = n >= n0 and (n1 is None or n < n1)
        if not inside:
            return f32(LO)
    return tri(n, P)


class Replay(object):
    """242.3(2): b0 for n < n_0; v_k + (v_{k+1} - v_k)(n - n_k)/(n_{k+1} - n_k) on [n_k, n_{k+1}); v_last for
    n >= n_last; clamped; float32."""

    def __init__(self, path):
        raw = open(path, "rb").read()
        self.sha = hashlib.sha256(raw).hexdigest()
        d = json.loads(raw.decode("utf-8"))
        self.id = d["id"]
        self.n = [int(x) for x in d["n"]]
        self.v = [float(x) for x in d["v"]]

    def __call__(self, n):
        kn, kv = self.n, self.v
        if n < kn[0]:
            x = B0
        elif n >= kn[-1]:
            x = kv[-1]
        else:
            k = (n - kn[0]) // (kn[1] - kn[0])
            while kn[k] > n:
                k -= 1
            while kn[k + 1] <= n:
                k += 1
            x = kv[k] + (kv[k + 1] - kv[k]) * (n - kn[k]) / float(kn[k + 1] - kn[k])
        return f32(min(HI, max(LO, x)))


def lion(bp, m, z):
    """clamp(beta_pre - ms*sign(b2*mom_pre + (1-b2)*z_agg)); tie iff |L| <= TIE_REL of its parts."""
    p, q = B2 * m, (1.0 - B2) * z
    L = p + q
    return max(LO, min(HI, bp - MS * sgn(L))), abs(L) <= TIE_REL * (abs(p) + abs(q))


def witness_bh(arm):
    if arm not in HOLD50:
        return "BETA_HOLD: off"
    mode, P = HOLD50[arm]
    w = "BETA_HOLD: on type=blockwise group=1 groupsize=1 name=%s mode=%s" % (NAME50, mode)
    if mode == "floor":
        return w + " value=" + repr(LO)
    return w + " P=%d b0=%s ms=%s lo=%s hi=%s peak=%s" % (P, repr(B0), repr(MS), repr(LO), repr(HI),
                                                          repr(min(HI, max(LO, B0 + MS * P))))


def witness_ch(arm, rep):

def plainnet_tensors():
    """PlainNet18_c100 = ResNet18_c100 without the 9 shortcut tensors (249.2 / 239 manifest), rebuilt here."""
    t = [("conv1.weight", 3 * 64 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    cin = 64
    for L, w in ((1, 64), (2, 128), (3, 256), (4, 512)):
        for b in (0, 1):
            ci = cin if b == 0 else w
            p = "layer%d.%d." % (L, b)
            t.append((p + "conv1.weight", ci * w * 9, "Conv2d"))
            t.append((p + "bn1.weight", w, "BatchNorm2d"))
            t.append((p + "bn1.bias", w, "BatchNorm2d"))
            t.append((p + "conv2.weight", w * w * 9, "Conv2d"))
            t.append((p + "bn2.weight", w, "BatchNorm2d"))
            t.append((p + "bn2.bias", w, "BatchNorm2d"))
        cin = w
    t.append(("linear.weight", 512 * 100, "Linear"))
    t.append(("linear.bias", 100, "Linear"))
    return t

def parse_out(path):
    r = {"args": [], "env": [], "pt": [], "vw": [], "bh": [], "ch": [], "wh": [], "ep": {}, "done": False, "tb": 0,
         "dup": 0}
    for ln in open(path, errors="replace"):
        s = ln.rstrip("\n")
        if s.startswith("ARGS:"):
            r["args"].append(s)
        elif s.startswith("ENV:"):
            r["env"].append(s)
        elif s.startswith("PROBE_TENSOR:"):
            r["pt"].append(s)
        elif s.startswith("VOTE_W"):
            r["vw"].append(s)
        elif s.startswith("BETA_HOLD"):
            r["bh"].append(s)
        elif s.startswith("COMP_HOLD"):
            r["ch"].append(s)
        elif s.startswith("WINDOW_HOLD"):
            r["wh"].append(s)
        elif s.startswith("Epoch "):
            p = s.split(",")
            e = int(p[0].split()[1])
            tr = float(p[1].split(":")[1].replace("%", "").strip())
            te = float(p[2].split(":")[1].replace("%", "").strip())
            r["dup"] += e in r["ep"]
            r["ep"][e] = (tr, te)
        elif s.strip() == "RUN_DONE":
            r["done"] = True
        if "Traceback (most recent call last)" in s:
            r["tb"] += 1
    return r


def flags_of(line):
    out = []
    for t in line[len("ARGS:"):].split():
        if t.startswith("--"):
            out.append([t[2:], None])
        elif out:
            out[-1][1] = t if out[-1][1] is None else out[-1][1] + " " + t
    return out


def groups_of(spec, names):
    if spec == "scalar":
        return [list(range(1, len(names) + 1))]
    gs = []
    for g in spec[len("sets:"):].split("/"):
        idx = []
        for tok in g.split(","):
            if tok.isdigit():
                idx.append(int(tok))
            elif "-" in tok and all(x.isdigit() for x in tok.split("-")):
                a, b = tok.split("-")
                idx.extend(range(int(a), int(b) + 1))
            else:
                idx.append(names.index(tok) + 1)
        gs.append(sorted(idx))
    return gs


def load_recs(pdir):
    out = []
    for x in open(os.path.join(pdir, "probe.jsonl")):
        if not x.strip():
            continue
        try:
            v = json.loads(x)
        except ValueError:
            v = None
        out.append(v if isinstance(v, dict) else None)
    return out


def plateau_of(path):
    r = parse_out(path)
    return mean([r["ep"][e][1] for e in range(95, 100)])


def first_stays(vals, pred):
    for e in range(len(vals)):
        if all(pred(v) for v in vals[e:]):
            return e
    return None
