#!/bin/bash
# =============================================================================
# Cycle 48 -- SUPERSEDES bin/c43_frozen_ladder.sh.  Runs on cluster_a (not cluster_b).
#
# OPEN QUESTION IT ANSWERS
#   Cycle 48 measured, for the first time with a heterogeneity-CORRECTED independence
#   floor, the scale profile of per-weight meta-gradient correlation:
#
#       steady half, ResNet18/CIFAR-10, beta FROZEN, alpha0=1e-3, T=2000 records
#         k=1       (weightwise)  rho_w^implied = 3.200e-06
#         k=775     (nodewise)                  = 1.133e-06
#         k=180,225 (layerwise)                 = 4.620e-08      span 69.3x
#
#   That is the c44 pre-registration's outcome (b) -- "the correlation length is REAL" --
#   and it is the project's structural result.  It currently rests on ONE architecture,
#   ONE dataset and 2 seeds.  If it is a ResNet18/CIFAR-10 artefact the claim collapses to
#   a single cell, exactly as FINDINGS 42.4 would have if the frozen exponent had never
#   been carried past R18.  This batch is that generalisation.
#
#   It ALSO carries c43's original question -- the frozen N_eff exponent `s` on
#   R10/R34/CIFAR-100 -- at zero extra cost, because the runs are the same runs.  c43 could
#   not be submitted as written: it exported PROBE=100 and no PROBE5=1, so its 30 runs
#   could not have carried the floor correction and could not have been corrected
#   afterwards (c43 header, amended cycle 47).  This fixes that.
#
# PRE-REGISTERED PREDICTIONS -- write the verdict against these, not against what lands.
#   (A0) SANITY, batch void if it fails.  Each family's n_weights is taken from ITS OWN
#        weightwise arm's `neg_counts.json` n_tot -- NEVER from block_sizes.json, which
#        reports the R18 parameter count for nodewise too (CORRECTIONS 16).  H <= 1 on
#        every rung, and H(weightwise) >= H(layerwise) to within resolution.
#   (A1) THE PROFILE GENERALISES.  On the STEADY HALF, every family shows rho_w^implied
#        falling >= 10x from k=1 to its own layerwise k.
#        REFUTATION: any family flat to within 3x -> the correlation length is a
#        ResNet18/CIFAR-10 artefact and the structural claim must be withdrawn to one cell.
#        This is the outcome that would most change the paper.
#   (A2) THE EXPONENT IS ARCHITECTURE-INSENSITIVE.  Fitting rho_w ~ k^-b over the resolved
#        rungs, b lands in [0.2, 0.6] in all three families and the across-family spread in
#        b is smaller than the within-family rung-to-rung spread.
#        REFUTATION: b varying more than 2x across families -> the profile is an
#        architecture property, a different and weaker claim than a correlation length.
#   (A3) c43's ORIGINAL prediction, carried forward unchanged: every frozen family lands
#        with `s` well below its own free-beta value -- R10 free 0.912 -> frozen <= 0.75;
#        R34 free 1.012 -> <= 0.80; C100 free 0.911 -> <= 0.75; and the frozen exponents
#        agree across families to within [0.55, 0.75].
#        REFUTATION: any frozen family at s >= 0.85 means the frozen/free gap of FINDINGS
#        42.4 is ResNet18-specific.
#
#   RESOLUTION -- state it next to any null (CORRECTIONS 33's standing rule).  At T=2000
#   (steady half T=1000) the measured rho_min on the R18 batch was
#       weightwise n~1.1e7   8.5e-09      nodewise n~1.4e4   7.1e-06
#       layerwise  n~62      1.3e-03      blk6      n=6      >8.6e-02
#   The blk6 rung DID NOT resolve on the steady half of the R18 batch (its rho_s went
#   negative, -4.6e-2 +-2.3e-2) and is NOT expected to resolve here.  It is included only
#   because it costs 2 jobs and it does resolve on the STARTUP window.
#
# WINDOW.  Report the STEADY HALF.  Cycle 48 measured that the full-run window inflates
# rho_s ~4.2x through the startup transient, and that the steady half reproduces FINDINGS
# 44.3's independently-measured frozen value 1.925e-6 to within 3-8% on a byte-matched
# config.  The profile verdict (b) holds in ALL THREE windows (span 53.4x full / 69.3x
# steady / 520x startup), so no window choice is a selection -- but quote the steady half.
#
# DESIGN.  Byte-for-byte the `p5-*-a3` recipe (cluster_b, cycle 47) with ONLY --NN-name,
# --dataset and --stepsize-groups changed.  --alg-meta fixed freezes beta, so the partition
# cannot affect the parameter update and every arm of a family runs the identical optimizer.
#   Recovered p5-w-a3-s0 ARGS:
#     --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
#     --alg-meta fixed --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
#     --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3 --num-epochs 20
#     --stepsize-groups weightwise --seed 0
#   Recovered ENV: AUGMENT=1 BETA_CLIP=-15:-2.3026 PROBE=5 (HIER/LAM/ETA_RATIO UNSET --
#   exporting the literals `HIER=none LAM=na` kills the job; see CONTINUE-HERE gotchas).
#
# WHY cluster_a AND NOT cluster_b.  c43 specified cluster_b to be same-account as `fz-*`.  That is
# the wrong pairing for THIS batch: its comparators are the FREE-beta families p7free
# (`p7-r18-*`, `p7-c100-*`), which ran on cluster_a.  Running here makes the frozen-vs-free
# contrast same-account, which is the contrast A3 is about.  PATCH_PROBE5 +
# PATCH_PROBE5_FIX were applied to cluster_a's HF.py this cycle and
# `tests/test_probe5_block.py` passes 10/10 against the live file.
#
# NOT INCLUDED, deliberately: `resnet18_blocks` on ResNet10/ResNet34 -- it is ResNet18-only
# and dies with ZeroDivisionError at HF.py:175 (six jobs already lost to this).  Those two
# families therefore span lay->node->w (3 rungs), which is the full resolvable range anyway.
#
# SEEDS: 2, not c43's 3.  Measured seed-to-seed spread of rho_w on the R18 batch is 3% at
# layerwise and 21% at weightwise, against a 69x effect.  The third seed buys nothing and
# costs 10 jobs.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. every record has beta_true_max == beta_true_min  (beta really is frozen)
#   2. neg_counts.json exists with n_records == 2000 and n_tot matching the arm
#   3. sum(neg_counts)/(n_records*n_tot) equals the time-mean of frac_neg to <1e-6
#   4. python3 analysis/probe5_floor.py --selftest  &&  tests/test_twochannel.py
#
# USAGE
#   bash c48_frozen_ladder_p5.sh            # dry run: prints every sbatch line
#   bash c48_frozen_ladder_p5.sh --submit   # submits, ONLY if the guards below pass
# =============================================================================
set -u

WS=/cluster/account_a/metaopt                  # cluster_a
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/fz3
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_a
PEND_CAP=40
NJOBS=20
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c48 frozen ladder + PROBE5 (cluster_a) ==="

# ---- guard 1: the patch must be present, or 20 jobs return nothing (FINDINGS 47.5) ----
if ! grep -q PATCH_PROBE5 "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5 not present in $HF"; exit 1
fi
if ! grep -q PATCH_PROBE5_FIX "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5_FIX not present -- unfixed patch crashes at record 500"; exit 1
fi
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX present  OK"

# ---- guard 2: batch-size rule (CORRECTIONS 37; the 0.35 FairShare floor is RETIRED) ----
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 2: FairShare=${FS:-unknown} (informational only), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
sub () {   # sub <family> <nn-name> <dataset> <gran-token> <stepsize-groups> <seed>
  local fam=$1 nn=$2 ds=$3 gtok=$4 groups=$5 s=$6
  local RN="fz3-${fam}-${gtok}-s${s}"
  local PD="$SAVE/probe_${fam}_${gtok}_s${s}"
  N=$((N+1))
  local CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
    --export="ALL,AUGMENT=1,BETA_CLIP=-15:-2.3026,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
    "$RUNNER"
    --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
    --alg-meta fixed --dataset "$ds" --NN-name "$nn" --batch-size 100
    --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3
    --num-epochs 20 --stepsize-groups "$groups" --seed "$s"
    --save-directory "$SAVE" --run-name "$RN")
  if [ "$SUBMIT" = "1" ]; then "${CMD[@]}"; else printf '%s\n' "${CMD[*]}"; fi
}

for s in 0 1; do
  # ResNet10 / CIFAR-10    (free-beta comparator s = 0.912 +-0.028, n=2)
  sub r10  ResNet10      CIFAR10  lay  layerwise       "$s"
  sub r10  ResNet10      CIFAR10  node nodewise        "$s"
  sub r10  ResNet10      CIFAR10  w    weightwise      "$s"
  # ResNet34 / CIFAR-10    (free-beta comparator s = 1.012 +-0.006, n=2)
  sub r34  ResNet34      CIFAR10  lay  layerwise       "$s"
  sub r34  ResNet34      CIFAR10  node nodewise        "$s"
  sub r34  ResNet34      CIFAR10  w    weightwise      "$s"
  # ResNet18 / CIFAR-100   (free-beta comparator s = 0.911 +-0.021, n=10)
  sub c100 ResNet18_c100 CIFAR100 blk6 resnet18_blocks "$s"
  sub c100 ResNet18_c100 CIFAR100 lay  layerwise       "$s"
  sub c100 ResNet18_c100 CIFAR100 node nodewise        "$s"
  sub c100 ResNet18_c100 CIFAR100 w    weightwise      "$s"
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo SUBMITTED || echo 'dry run, nothing submitted' )) ----"
echo "reduce with:  python3 analysis/probe5_window.py <probe root>   (steady half)"
echo "         and: python3 analysis/neff_ladder.py fz3              (the A3 exponent)"
