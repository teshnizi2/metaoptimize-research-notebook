def _fin(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool) and math.isfinite(x)


def _med(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return float("nan")
    return v[n // 2] if n % 2 else 0.5 * (v[n // 2 - 1] + v[n // 2])


def _sg(x):
    return (x > 0) - (x < 0)


# =============================================================================
# THE LION VOTE DECOMPOSITION (the harness-null that the meta update is Lion on the recorded trace), scalar arms only
# =============================================================================
def vote_terms(r):
    try:
        mp, b2 = float(r["pt_mp"]), float(r["pt_b2"])
        z, m = r["z_tensor"], r["m_tensor"]
        mom, za = float(r["mom_pre"][0][0]), float(r["z_agg"][0][0])
    except (KeyError, TypeError, IndexError, ValueError):
        return None
    if len(z) != NTENS or len(m) != NTENS or not all(_fin(x) for x in z) or not all(_fin(x) for x in m):
        return None
    L = [b2 * mi + (1.0 - b2) * zi for mi, zi in zip(m, z)]
    app = b2 * mom + (1.0 - b2) * za
    if not (_fin(app) and all(_fin(x) for x in L)):
        return None
    return L, app


def decomposition(recs):
    """-> (n_usable, n_bad_sum, n_bad_sign, n_nonfinite)."""
    n = bad = bads = nonf = 0
    for r in recs or []:
        v = vote_terms(r) if isinstance(r, dict) else None
        if v is None:
            nonf += 1
            continue
        L, app = v
        s = sum(L)
        scale = sum(abs(x) for x in L)
        n += 1
        bad += not (abs(s - app) <= DECOMP_TOL * scale or abs(s - app) <= 1e-30)
        bads += _sg(s) != _sg(app) and abs(app) > DECOMP_TOL * scale
    return n, bad, bads, nonf


# =============================================================================
# THE ROUTE AUDIT: PATCH_DECAYROUTE's own dr_* keys, per run (a HARNESS-NULL)
# =============================================================================
def route_audit(recs, arm):
    """-> (ok, detail).  Pure function of the records.  Every record: dr_mode == the arm's mode, dr_lam == its LAMBDA
    (None off alpha_indep), dr_n == step + 2, dr_shrink_meas / dr_shrink_applied / dr_trace_decay finite numbers; then
    the mode's own checks on the MEASURED shrink (see the tolerances above)."""
    mode, lam, wd = D.MODE[arm], D.LAM_OF[arm], D.WD[arm]
    d = {"n": 0, "bad_rec": 0, "bad_mode": 0, "bad_lam": 0, "bad_n": 0, "missing": 0, "bad_shrink": 0, "bad_coef": 0,
         "checked": 0, "worst": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for r in recs:
        if not (isinstance(r, dict) and isinstance(r.get("step"), int)):
            d["bad_rec"] += 1
            continue
        d["n"] += 1
        if r.get("dr_mode") != mode:
            d["bad_mode"] += 1
        rl = r.get("dr_lam", "absent")
        if not ((lam is None and rl is None) or (lam is not None and _fin(rl) and rl == lam)):
            d["bad_lam"] += 1
        if r.get("dr_n") != r["step"] + 2:
            d["bad_n"] += 1
        me, ap, td = r.get("dr_shrink_meas"), r.get("dr_shrink_applied"), r.get("dr_trace_decay")
        if not (_fin(me) and _fin(ap) and _fin(td)):
            d["missing"] += 1
            continue
        if mode == "shrink_only":
            if abs(td - 1.0) > EXACT_TOL or not ap > 0:
                d["bad_coef"] += 1
            if ap >= SHRINK_CHECK_MIN:
                d["checked"] += 1
                if D.SPEC[arm] == "scalar":
                    dev = abs(me / ap - 1.0)
                    d["worst"] = max(d["worst"], dev)
                    d["bad_shrink"] += dev > S_REL_TOL
                else:
                    b = r.get("beta")
                    if not (isinstance(b, list) and b and all(_fin(x) for x in b)):
                        d["bad_shrink"] += 1
                        continue
                    lo, hi = wd * math.exp(min(b)), wd * math.exp(max(b))
                    ok = lo * (1 - LW_REL_TOL) - LW_ABS_TOL <= me <= hi * (1 + LW_REL_TOL) + LW_ABS_TOL
                    d["bad_shrink"] += not ok
        elif mode == "trace_only":
            d["checked"] += 1
            d["worst"] = max(d["worst"], abs(me))
            d["bad_shrink"] += abs(me) > T_ABS_MAX
            if ap != 0.0 or not (1.0 - wd * 0.1 - EXACT_TOL <= td <= 1.0 + EXACT_TOL):
                d["bad_coef"] += 1
        else:
            d["checked"] += 1
            dev = abs(me / lam - 1.0)
            d["worst"] = max(d["worst"], dev)
            d["bad_shrink"] += dev > I_REL_TOL
            if abs(ap - lam) > EXACT_TOL * lam or abs(td - (1.0 - lam)) > EXACT_TOL:
                d["bad_coef"] += 1
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] == N_RECORDS and d["bad_mode"] == 0 and
          d["bad_lam"] == 0 and d["bad_n"] == 0 and d["missing"] == 0 and d["bad_shrink"] == 0 and d["bad_coef"] == 0)
    return ok, d


# =============================================================================
# THE MEASURED SHRINK, TRACE AND DOSE (descriptive), from the arm's own records
# =============================================================================
def measure_seed(recs):
    """-> dict: shrink_onset (median dr_shrink_meas over ONSET_LO <= step < ONSET_HI), shrink_peak, shrink_cum
    (sum of PROBE * dr_shrink_meas), td_onset (median dr_trace_decay in the window) and its horizon 1/(1 - td), turn
    (first record with applied Lion sign +1), floor_share (last 20 % of records with min beta <= FLOOR_BETA)."""
    sh, sho, tdo, late, turn = [], [], [], [], None
    last0 = 4 * N_RECORDS * D.PROBE // 5
    for r in recs or []:
        if not (isinstance(r, dict) and isinstance(r.get("step"), int)):
            continue
        me, td = r.get("dr_shrink_meas"), r.get("dr_trace_decay")
        if _fin(me):
            sh.append(me)
            if ONSET_LO <= r["step"] < ONSET_HI:
                sho.append(me)
        if _fin(td) and ONSET_LO <= r["step"] < ONSET_HI:
            tdo.append(td)
        b = r.get("beta")
        if r["step"] >= last0 and isinstance(b, list) and b and all(_fin(x) for x in b):
            late.append(min(b) <= FLOOR_BETA)
        if turn is None and "z_agg" in r:
            v = vote_terms(r)
            if v is not None and _sg(v[1]) == 1:
                turn = r["step"]
    tdm = _med(tdo)
    return {"shrink_onset": _med(sho), "shrink_peak": max(sh) if sh else float("nan"),
            "shrink_cum": D.PROBE * sum(sh) if sh else float("nan"), "td_onset": tdm,
            "horizon_onset": (1.0 / (1.0 - tdm)) if _fin(tdm) and tdm < 1.0 else float("inf"),
            "turn": turn, "floor_share": (sum(late) / len(late)) if late else float("nan")}


def dose_word(rho):
    if not _fin(rho):
        return "UNREAD"
    return "BELOW" if rho < DOSE_BAR_LO else "REACHED"

# THE STATES AND THE TWO LADDERS

def cell_state(Lseeds, kseeds):
    L, k = mean(Lseeds), mean(kseeds)
    if L < REF_MIN or (max(Lseeds) - min(Lseeds)) > DIVERGED_BAR:
        return "UNREADABLE"
    ncol = sum(1 for x in kseeds if x <= R50 * L)
    if 1 <= ncol <= len(kseeds) - 1:
        return "SPLIT"
    if k <= R50 * L:
        return "COLLAPSE"
    if L - k < GAP_BAR:
        return "NOGAP"
    return "PARTIAL"


def decide_route(s, t):
    """(S state, T state) -> ROUTE token.  Pure; first match; the last line is the default, so it is exhaustive."""
    if s == "COLLAPSE" and t == "COLLAPSE":
        return "ROUTE-EITHER-SUFFICES"
    if s == "COLLAPSE" and t == "NOGAP":
        return "ROUTE-IS-WEIGHT-SHRINK"
    if t == "COLLAPSE" and s == "NOGAP":
        return "ROUTE-IS-TRACE"
    if s == "NOGAP" and t == "NOGAP":
        return "ROUTE-NEITHER-ALONE"
    if s == "COLLAPSE":
        return "ROUTE-WEIGHT-SHRINK-SUFFICES-TRACE-UNRESOLVED"
    if t == "COLLAPSE":
        return "ROUTE-TRACE-SUFFICES-SHRINK-UNRESOLVED"
    if "UNREADABLE" in (s, t):
        return "ROUTE-REFERENCE-UNREADABLE"
    return "ROUTE-PARTIAL"


def decide_indep(i):
    """I state -> INDEP token.  Pure; exhaustive."""
    if i == "COLLAPSE":
        return "INDEP-COLLAPSES"
    if i == "NOGAP":
        return "INDEP-NOGAP"
    if i == "UNREADABLE":
        return "INDEP-UNREADABLE"
    return "INDEP-PARTIAL"


def decide(states):
    return decide_route(states["S"], states["T"]), decide_indep(states["I"])

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 ladders taken, 1 harness-unsound, 2 incomplete."""
    print("=" * 78)
    print(" crd1 -- THE COLLAPSE ROUTE (ICML-PLAN 1.6).  At the mechanism cell (wd 0.1): weight shrink, shortened trace,")
    print("         or neither alone?  And does the collapse survive alpha-INDEPENDENT decay (LAMBDA %s)?" % D.LAMBDA_TOKEN)
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1  SGDm(0.99)+Lion"
          % (D.NET, D.DSET, EPOCHS, ",".join(str(s) for s in SEEDS), D.AUG, D.CLIP, D.PROBE))
    for a in ARMS:
        print("   %-4s cell %s %-9s wd %-3s DECAY_ROUTE=%-22s %s" % (a, D.CELL_OF[a], D.SPEC[a], D.WDTOK[a], D.ROUTE[a], D.ROLE[a]))
    print(" Every state is WITHIN its cell.  plateau5 = mean TEST over epochs %d..%d.  BARS ARE FROZEN LITERALS (O2)."
          % (EPOCHS - 5, EPOCHS - 1))
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = C.read_runs(runsdir, PREFIX, ARMS, EPOCHS)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        print("  NOTE %s" % n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        for ln in LICENSE_ROUTE["INCOMPLETE"]:
            print("  " + ln)
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    [HARNESS-NULL] every run's OWN ARGS line == its arm's registered flags (wd 0.1 on S / T, 0 on I)")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = C.parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = dict(D.args_pairs(a, s, "<save>"))
            bad = [k for k, v in want.items() if k != "save-directory" and f.get(k, [None])[-1] != v]
            extra = sorted(set(f) - set(want))
            sv = f.get("save-directory", [""])[-1] or ""
            ok = not dup and not bad and not extra and sv.rstrip("/").endswith("/" + PREFIX)
            hchk(ok, "G-ARGS %s-s%d (%s wd %s)" % (a, s, D.SPEC[a], D.WDTOK[a]),
                 "" if ok else "repeated %s; wrong %s; unregistered %s; save %s"
                 % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad], extra, sv))

         "G-ENV one distinct ENV line (PROBE_DIR stripped), one per run", "%d distinct" % len(envs))
    hchk(envs == [D.ENV_EXPECTED], "G-ENV the ENV line is cvt1 ... cwd5 / csh1's, byte for byte (PROBE_DIR stripped)")

    print("\nG-PT      ONE PROBE_TENSOR line per run == its arm's witness (grain, 62 tensors, Lion meta)")
    for s in SEEDS:
        for a in ARMS:
            pt = runs[(a, s)]["pt"]
            ok = len(pt) == 1 and pt[0].startswith(D.pt_witness_prefix(a) + " dir=")
            hchk(ok, "G-PT %s-s%d" % (a, s), "" if ok else "got %r" % ([x[:150] for x in pt],))

    print("\nG-WITNESS [HARNESS-NULL] EXACTLY ONE `DECAY_ROUTE:` line == the arm's registered witness (mode, wd, LAMBDA,"
          " its float32, gamma); NO hold / mask witness line (the cdr1 tree has none of those patches)")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            drl = dr_lines(os.path.join(runsdir, r["file"]))
            others = dict((k, r[k]) for k in C.KINDS if r[k])
            ok = drl == [D.dr_witness(a)] and not others
            hchk(ok, "G-WITNESS %s-s%d (%s)" % (a, s, D.ROUTE[a]), "" if ok else "DECAY_ROUTE %r; other %r" % (drl, others))

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == crd1_design.manifest_text(), byte for byte")

                               ("HF_PRE_SHA256", D.HF_PRE_SHA, "HF.py.pre_decayroute == the live HF.py"),
                               ("RUNNER_SHA256", D.RUNNER_SHA, "runner == run_cifar_cdr1.sh, UNCHANGED"),
                               ("TRAIN_SHA256", D.TRAIN_SHA, "train.py == the tree's"),
                               ("PATCH_SHA256", D.PATCH_SHA, "patches/patch_decayroute.py == 305's"),
                               ("SCORER_SHA256", me, "SCORER == this file"),
                               ("DESIGN_SHA256", file_sha(D.__file__), "DESIGN == the crd1_design.py this scorer loaded"),
                               ("CSH1_DESIGN_SHA256", file_sha(D.H.__file__), "the reused csh1_design.py this scorer loaded"),
                               ("CWD_DESIGN_SHA256", file_sha(D.W.__file__), "the reused cwd_design.py this scorer loaded"),
                               ("COMMON_SHA256", file_sha(C.__file__), "COMMON == the cwd_common.py this scorer loaded")):
            hchk(prov.get(key) == want, "G-PROV %s" % lab, "%s=%s" % (key, str(prov.get(key))[:16]))

    print("\nG-PROBE   [HARNESS-NULL] %d records per run; the step-size count is the GRAIN's (1 or %d); on the scalar"
          " arms the per-tensor Lion vote sums to the harness's applied term (within %g of sum|L_i|) with its sign"
          % (N_RECORDS, NTENS, DECOMP_TOL))
    recs_all, nonfin = {}, {}
    for s in SEEDS:
        for a in ARMS:
            recs = C.load_probe(runsdir, PREFIX, a, s)
            recs_all[(a, s)] = recs
            if recs is None:
                hchk(False, "G-PROBE %s-s%d probe.jsonl present" % (a, s))
                continue
            want_nb = 1 if D.SPEC[a] == "scalar" else NTENS
            badrec = sum(1 for r in recs if not (isinstance(r, dict) and isinstance(r.get("step"), int)
                                                   and isinstance(r.get("beta"), list) and len(r["beta"]) == want_nb
                                                   and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS))
            line = "%d records, %d malformed / wrong step-size count" % (len(recs), badrec)
            ok = len(recs) == N_RECORDS and badrec == 0
            if D.SPEC[a] == "scalar":
                n, bsum, bsign, nf = decomposition(recs)
                nonfin[(a, s)] = nf
                line += "; Lion decomposition: %d usable, %d off, %d wrong sign, %d non-finite (disclosed)" % (n, bsum, bsign, nf)
                ok = ok and bsum == 0 and bsign == 0 and n > 0
            hchk(ok, "G-PROBE %s-s%d (%s, %d step size(s))" % (a, s, D.SPEC[a], want_nb), line)

    print("\nG-ROUTE   [HARNESS-NULL] PATCH_DECAYROUTE's OWN record on every run: dr_mode / dr_lam / dr_n == step + 2 on"
          " all %d records, and the MEASURED shrink is the mode's: shrink_only |meas/applied - 1| <= %g (scalar) or inside"

        contr[lab] = M[x] - M[y]
        print("  %-10s = %+.4f pp [%+.4f, %+.4f]" % (lab, contr[lab], contr[lab] - 2 * se, contr[lab] + 2 * se))
    print("\nBETWEEN-BATCH, NON-GATING: against the landed OFF premise cwd5 W1 (scalar %.4f, layerwise %.4f; state %s)"
          % (CWD5_W1["scalar"], CWD5_W1["layerwise"], cell_state([CWD5_W1["layerwise"]] * 3, [CWD5_W1["scalar"]] * 3)))

def _e_curve(level):
    return [level if e >= EPOCHS - 5 else level * (1 - math.exp(-(e + 1) / 12.0)) for e in range(EPOCHS)]


def _e_write(d, arm, s, jid, level, n_ep=EPOCHS, run_done=True, env=_E_ENV, args=None, pt=None, drl=None, extra=()):
    lines = ["NODE=[HOST_2f2f29d9e1] | JOB=%s-%s-s%d %d | AUGMENT=1" % (PREFIX, arm, s, jid),
             "ARGS: %s" % (args if args is not None else D.args_string(arm, s, _E_SAVE)), env, "NVIDIA L4, 23034 MiB"]
    lines += list(drl if drl is not None else [D.dr_witness(arm)])
    lines += list(extra)
    lines.append(pt if pt is not None else "%s dir=/x/probe_%s-%s-s%d" % (D.pt_witness_prefix(arm), PREFIX, arm, s))
    for e, v in enumerate(_e_curve(level)[:n_ep]):
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(v + 20, 99.9), v))
    if run_done:
        lines.append("RUN_DONE")
    open(os.path.join(d, "%s-%s-s%d-%d.out" % (PREFIX, arm, s, jid)), "w").write("\n".join(lines) + "\n")


def _e_records(arm, seed, nb=None, decomp_break=False, beta=None, floor=False, rng=None, dr=True, dr_forge=None):
    """500 synthetic records under Lion algebra, with PATCH_DECAYROUTE's dr_* keys consistent with the arm's mode."""
    rng = rng or random.Random(seed * 7 + len(arm))
    mp, b2 = 0.99, 0.9
    nb = nb if nb is not None else (1 if D.SPEC[arm] == "scalar" else NTENS)
    mode, lam, wd = D.MODE[arm], D.LAM_OF[arm], D.WD[arm]
    out = []
    for k in range(N_RECORDS):
        L = [rng.gauss(0, 1e-6) for _ in range(NTENS)]
        m = [rng.gauss(0, 1e-6) for _ in range(NTENS)]
        z = [(Li - b2 * mi) / (1 - b2) for Li, mi in zip(L, m)]
        mom, za = sum(m), sum(z)
        if decomp_break:
            za = -za - 1e-3
        bv = beta if beta is not None else (-15.0 if (floor and k >= 400) else -5.0)
        bl = [bv + (0.01 * j if nb > 1 else 0.0) for j in range(nb)]
        step = k * D.PROBE
        r = {"step": step, "beta": bl, "z_tensor": z, "m_tensor": m, "mom_pre": [[mom]], "z_agg": [[za]],
             "pt_mp": mp, "pt_b2": b2, "beta_pre": [list(bl)]}
        if dr:
            av = [math.exp(x) for x in bl]
            if mode == "shrink_only":
                ap = wd * sum(av) / len(av)
                me, td = wd * _med(av) * (1 + 1e-4), 1.0
            elif mode == "trace_only":
                ap, me, td = 0.0, 1e-9, 1.0 - wd * sum(av) / len(av)
            else:
                ap, me, td = lam, lam * (1 + 2e-4), 1.0 - lam
            r.update({"dr_mode": mode, "dr_lam": lam, "dr_n": step + 2, "dr_shrink_meas": me,
                      "dr_shrink_applied": ap, "dr_trace_decay": td})
            if dr_forge == "mode":
                r["dr_mode"] = "trace_only" if mode != "trace_only" else "shrink_only"
            elif dr_forge == "n":
                r["dr_n"] = step + 1
            elif dr_forge == "shrink":
                r["dr_shrink_meas"] = (me * 1.2) if mode != "trace_only" else 1e-5
            elif dr_forge == "missing" and k == 250:
                r["dr_shrink_meas"] = None
            elif dr_forge == "lam":
                r["dr_lam"] = 5e-5 if lam is not None else 3.15e-4
        out.append(r)
    return out


def _e_probe_templates(root):
    t = {}
    os.makedirs(root, exist_ok=True)

def _e_build(d, tpl, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, prov_over=None, man_break=False,
             rec_over=None):
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    lv = dict(_E_BASE)
    lv.update(levels or {})
    jid = 7000000
    for s, j in zip(SEEDS, jit):
        for a in ARMS:
            jid += 1
            if drop == (a, s):
                continue
            L = lv[a] if not isinstance(lv[a], (list, tuple)) else lv[a][SEEDS.index(s)]
            kw = dict((over or {}).get((a, s), {}))
            _e_write(d, a, s, jid, L + (0 if isinstance(lv[a], (list, tuple)) else j), **kw)
            pd = os.path.join(d, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s))
            os.makedirs(pd, exist_ok=True)
            if rec_over and (a, s) in rec_over:
                with open(os.path.join(pd, "probe.jsonl"), "w") as fh:
                    for r in rec_over[(a, s)]:
                        fh.write(json.dumps(r) + "\n")
            else:
                os.symlink(tpl[(a, s)], os.path.join(pd, "probe.jsonl"))
    man = D.manifest_text() + ("X\n" if man_break else "")
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(man)
    prov = _e_prov()
    prov.update(prov_over or {})
    open(os.path.join(d, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % kv for kv in prov.items()))

RES = {"PASS": 0, "FAIL": 0, "SKIP": 0}


def chk(cond, label, extra=""):
    print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
    RES["PASS" if cond else "FAIL"] += 1
    return bool(cond)


def skip(label, extra=""):
    print("  SKIP %s%s" % (label, ("   " + extra) if extra else ""))
    RES["SKIP"] += 1


def _find_out(runsdir, name_prefix):
    if not runsdir or not os.path.isdir(runsdir):
        return None
    for fn in sorted(os.listdir(runsdir)):
        if fn.startswith(name_prefix) and fn.endswith(".out"):
            return os.path.join(runsdir, fn)
    return None

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--csv", default=os.path.join(REPO, "results", "all_runs.csv"))
    ap.add_argument("--manifest", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir required")
    rc, _final = score(a.runsdir, a.csv, a.manifest)
    raise SystemExit(rc)


if __name__ == "__main__":
