cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
# Standard guard ON (published, prevents the numerical artifact) so we measure hierarchy, not overflow
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"
CLIP="BETA_CLIP=-15:-2.3026"
sub () { # name granularity extra_env seed
  sbatch --parsable --job-name=$1-s$4 --partition=gpu-l4-24g --gres=gpu:l4:1 --time=03:00:00 \
    --export=ALL,AUGMENT=1,$CLIP,$3 \
    jobs/run_cifar.sh $B --stepsize-groups $2 --seed $4 --save-directory $R/hier --run-name $1_s$4 >/dev/null
}
for s in 0 1; do
  # M0 shrink on weightwise -- the proposal, applied where the failure is
  sub hs-w-lam001  weightwise "HIER=shrink,LAM=0.01" $s
  sub hs-w-lam01   weightwise "HIER=shrink,LAM=0.1"  $s
  sub hs-w-lam05   weightwise "HIER=shrink,LAM=0.5"  $s
  # M1 additive on weightwise
  sub ha-w-r01     weightwise "HIER=additive,ETA_RATIO=0.1" $s
  sub ha-w-r03     weightwise "HIER=additive,ETA_RATIO=0.3" $s
  # M0 shrink on layerwise -- can hierarchy beat the best plain arm?
  sub hs-l-lam01   layerwise  "HIER=shrink,LAM=0.1"  $s
done
sleep 2
echo "HIER_SWEEP_SUBMITTED=$(squeue -h -u account_a -o '%j' | grep -cE '^(hs|ha)-')"
