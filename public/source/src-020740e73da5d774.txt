#   ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS {132,133,134},
#   PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM,

# --manifest and --csv are OPTIONAL and DEFAULTED.
#
# FROZEN BARS (O2): every bar, sigma and tolerance score() uses is a LITERAL here; the only data-dependent terms are the
# in-batch arm means and SIGMA_INBATCH through compose_sigma(), a max() with the frozen floor.  The noise floor literal
# was read through corpus_exclusions.filter_rows (OPERATIONS 36); the selftest re-derives it.
#
# THE ARMS (cvt9's cell exactly: SGDm 0.99 / wd 0.1 + Lion 0.99/0.9, ms 1e-3, alpha0 1e-6, clip -15:-2.3026, aug 1, bs 100)
#   k01           scalar                                                                  floor anchor
#   k01WD0        scalar, DECAY_MASK=layer4.1.bn2.weight                                  does the SCALAR collapse need WD on 50?
#   HIGHHEADPATH  HEAD's grouping, 50 on tri:9428, complement on rec:cvt6_headpath        the held-step stall (cvt6/cvt9 replicate)
#   HIGHWD0       HIGHHEADPATH + DECAY_MASK=layer4.1.bn2.weight                           THE KEY ARM
#   LOWWD0        HEAD's grouping, 50 at the floor, complement on HEADPATH, DECAY_MASK=50 CONTROL (CTL) under the same mask
#   In the three held arms every applied step size is exogenous (open loop): the mask changes ONLY 50's weight update
#   (its meta trace h feeds a vote that the holds overwrite).  In k01WD0 it changes 50's update AND its trace h (both
#   routes; not separable here).
#
# ACCOUNTS (UNSURE levels; K01 8-16, FREE 56-72, PART 20-50).  Every account: k01 K01, HIGHHEADPATH K01, LOWWD0 FREE.
#   ROUTE   STEP-ROUTE (HIGHWD0 K01: the damage does not need WD on 50) | WD-ROUTE (HIGHWD0 FREE: removing WD on 50
#           removes the damage) | BOTH-ROUTES (HIGHWD0 PART)
#   SCALAR  SCALAR-WITHOUT-CARRIER-WD (k01WD0 K01) | SCALAR-NEEDS-CARRIER-WD (k01WD0 FREE) | SCALAR-PARTIAL (PART)
#   BROKEN-MASK null (DECAY_MASK never bites: every masked arm = its unmasked twin) = STEP-ROUTE | SCALAR-WITHOUT-
#           CARRIER-WD by level, separated ONLY by G-BITE's DECAY_MASK audit (declared; the selftest shows landed cvt9
#           records read as the masked arms FAIL it).
#
# BRANCH MAP (decide(); first match wins):  UNRESOLVED-DIVERGED (a seed range > 5) -> SCALAR-NOT-COLLAPSED (k01 > 20)
#   -> CONTROL-FAILED (LOWWD0 - k01 < 20) -> REPLICATE-FAILED (HIGHHEADPATH > k01 + 2) -> TWO tokens:
#   ROUTE:  WD-ROUTE (HIGHWD0 >= CTL - 5) | STEP-ROUTE (HIGHWD0 <= HIGHHEADPATH + 2) | BOTH-ROUTES
#   SCALAR: SCALAR-NEEDS-CARRIER-WD (k01WD0 >= CTL - 5) | SCALAR-WITHOUT-CARRIER-WD (k01WD0 <= k01 + 2) | SCALAR-PARTIAL
# CO-PRIMARY (in batch; bounded in neither direction): P_WD = HIGHWD0 - HIGHHEADPATH (what removing WD on 50 gives back);
#   P_LEFT = LOWWD0 - HIGHWD0 (damage left without WD on 50); P_SC = k01WD0 - k01.
# KEY: D_CTL = LOWWD0 - k01; R_HIGH = LOWWD0 - HIGHHEADPATH.  DESCRIPTIVE: F_WD = P_WD / R_HIGH.
# G-BITE (-> PATCH-NOT-VERIFIED, before any level): cvt9's registered bite_check (analysis/cVT9_dosewindow_score.py,
#   imported UNEDITED) on every run read as its cvt9 twin (Lion on k01 / k01WD0; 50 on its schedule and the complement
#   on HEADPATH in beta AND beta_pre at every record, bh_* / ch_* audits, no hold key on k01 / k01WD0), AND the
#   DECAY_MASK record audit (analysis/cwd_common.dm_audit): masked arms carry dm_n == step+2, dm_skipped == dm_n * 1,
#   dm_masked 1, a positive dm_wdterm; unmasked arms carry no dm_* key.
# plateau5 PRIMARY (epochs 95..99 of each run's own .out); TRAIN beside TEST; the dm_* readout after FINAL is DESCRIPTIVE.

def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d

def decide(M, diverged=False):
    """M: arm -> mean plateau5."""
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M[CTL] - M["k01"] < GAP_MIN:
        return "CONTROL-FAILED"
    if M["HIGHHEADPATH"] > M["k01"] + NULL_BAR:
        return "REPLICATE-FAILED"
    if M["HIGHWD0"] >= M[CTL] - MATCH_BAR:
        route = "WD-ROUTE"
    elif M["HIGHWD0"] <= M["HIGHHEADPATH"] + NULL_BAR:
        route = "STEP-ROUTE"
    else:
        route = "BOTH-ROUTES"
    if M["k01WD0"] >= M[CTL] - MATCH_BAR:
        scal = "SCALAR-NEEDS-CARRIER-WD"
    elif M["k01WD0"] <= M["k01"] + NULL_BAR:
        scal = "SCALAR-WITHOUT-CARRIER-WD"
    else:
        scal = "SCALAR-PARTIAL"
    return "%s | %s" % (route, scal)

def bite_run(arm, recs):
    """-> (ok, detail9, detaildm).  cvt9's registered bite_check on the run read as its cvt9 twin, AND the dm audit."""
    ok9, d9 = S9.bite_check(B.AS_CVT9[arm], recs)
    okd, dd = C.dm_audit(recs, K_MASKED[arm], N_RECORDS)
    return ok9 and okd, d9, dd


def dm_readout(runsdir, runs):
    print("\nDESCRIPTIVE (non-gating) -- PATCH_DECAYMASK's own readout on the masked arms: ||w_50|| (dm_norm), min |w_50|,")
    print("entries |w| < 1e-3, at records 0 / 50 / 95 / 200 / 499 (epochs 0 / 10 / 19 / 40 / 99.8); arm means over seeds")
    for a in MASKED:
        rows = []
        for s in SEEDS:
            recs = C.load_probe(runsdir, PREFIX, a, s) or []
            rows.append(recs)
        parts = []
        for k in (0, 50, 95, 200, 499):
            v = [r[k] for r in rows if len(r) > k and isinstance(r[k], dict) and isinstance(r[k].get("dm_norm"), list)]
            if not v:
                parts.append("rec %d n/a" % k)
                continue
            nr = mean([x["dm_norm"][0] for x in v if C._finite(x["dm_norm"][0])] or [float("nan")])
            am = mean([x["dm_absmin"] for x in v if C._finite(x["dm_absmin"])] or [float("nan")])
            smv = mean([x["dm_small"] for x in v])
            parts.append("rec %d: norm %.4g absmin %.3g small %.1f" % (k, nr, am, smv))
        print("  %-8s %s" % (a, " | ".join(parts)))

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    print("=" * 78)
    print(" cwd2 -- PLAINNET TENSOR 50: DOES THE HELD LARGE STEP ACT THROUGH COUPLED WEIGHT DECAY ON THAT SCALE?  (PATCH_DECAYMASK)")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-12s %-38s BETA_HOLD %-30s COMP_HOLD %-18s DECAY_MASK %s" % (a, SPEC[a], B.BETAHOLD[a] or "(unset)",
              B.COMPHOLD[a] or "(unset)", B.DMASK[a] or "(unset)"))
    print(" CO-PRIMARY: P_WD = HIGHWD0 - HIGHHEADPATH; P_LEFT = LOWWD0 - HIGHWD0; P_SC = k01WD0 - k01.")
    print(" plateau5 = mean TEST over epochs %d..%d of each run's own .out.  BARS ARE FROZEN LITERALS (O2)." % (EPOCHS - 5, EPOCHS - 1))
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = C.read_runs(runsdir, PREFIX, ARMS, EPOCHS)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    every run's OWN ARGS line says what its file name registers")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = C.parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s),
                    "momentum-param-base": "0.99", "weight-decay-base": "0.1", "momentum-param-meta": "0.99",
                    "Lion-beta2-meta": "0.9", "weight-decay-meta": "0", "gamma": "1"}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad]) if (dup or bad) else "")

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(" ".join(t for t in ln.split() if not t.startswith("PROBE_DIR=")) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()), "G-ENV one distinct ENV line, one per run",
         "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
    hchk(envs == [ENV_EXPECTED], "G-ENV the ENV line is cvt1's ... cvt9's, byte for byte (PROBE_DIR stripped)")
    badpt = [(a, s) for (a, s), r in sorted(runs.items())
             if len(r["pt"]) != 1 or not r["pt"][0].startswith("PROBE_TENSOR: on every=%d type=%s tensors=%d " % (PROBE, PT_TYPE[a], NTENS))]
    hchk(not badpt, "G-ENV every run printed ONE `PROBE_TENSOR: on every=%d type=<its arm's> tensors=%d`" % (PROBE, NTENS),
         "bad: %s" % badpt[:5] if badpt else "%d runs" % len(runs))

    print("\nG-WITNESS one line of each kind per run == the arm's registered witness (VOTE_W / BETA_HOLD / COMP_HOLD / WINDOW_HOLD / DECAY_MASK)")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            for kind, want in (("VOTE_W", B.WITNESS_VW), ("BETA_HOLD", B.WITNESS_BH[a]), ("COMP_HOLD", B.WITNESS_CH[a]),
                               ("WINDOW_HOLD", B.WITNESS_WH[a]), ("DECAY_MASK", B.WITNESS_DM[a])):
                hchk(r[kind] == [want], "G-WITNESS %s %s-s%d" % (kind, a, s), "" if r[kind] == [want] else "got %r" % ([x[:160] for x in r[kind][:2]],))

            "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 "
            "--weight-decay-meta 0 --dataset CIFAR100 --NN-name %s --batch-size 100 "
            "--max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-6 "
            "--num-epochs 100 --stepsize-groups %s --seed %d "
            "--save-directory /cluster/account_b/metaopt/runs/cwd2 --run-name %s" % (NET, spec or SPEC[arm], s, rn))
    lines = ["NODE=[HOST_45cba27b74] | JOB=%s %d | AUGMENT=1" % (rn, jid), "ARGS: " + args, env, "NVIDIA L4, 23034 MiB",

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None, env_break=None,
             spec_break=None, dm_break=None, bh_break=None, extra_break=None, manifest=None, prov=None, probe_break=None,
             probe_drop=None):
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(manifest if manifest is not None else D.manifest_text(B))
    _e_prov(d, prov)
    jid = 9800000
    pb = probe_break or {}
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = "layerwise" if spec_break == (a, s) else None
            dm = dm_break[1] if dm_break and dm_break[0] == (a, s) else None
            bh = bh_break[1] if bh_break and bh_break[0] == (a, s) else None
            extra = extra_break[1] if extra_break and extra_break[0] == (a, s) else None
            _e_write(d, a, s, jid, L, env=env, spec=spec, dm=dm, bh=bh, extra=extra)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--csv", default=S9.DEFAULT_CSV)
    ap.add_argument("--manifest", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir is required")
    rc, _fin = score(a.runsdir, a.csv, a.manifest)
    raise SystemExit(rc)
