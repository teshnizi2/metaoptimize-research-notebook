import sys, os, re, csv, glob, math, json, collections

HERE = os.path.dirname(os.path.abspath(__file__))
REPO = os.path.dirname(HERE)
RUNS = [os.path.join(REPO, "..", "runs"), os.path.join(REPO, "..", "runs_cluster_b")]
CSV = os.path.join(REPO, "results", "all_runs.csv")
DOCS = os.path.join(REPO, "docs")

# The config axes held FIXED when forming a granularity contrast.  `epochs_done` is in the
# key as well as `epochs_requested`: a cell that mixes a finished 20-epoch run with a
# truncated one would compare different windows, which is the very confound under test.
CELL_KEY = ["network", "dataset", "base", "meta", "meta_stepsize", "alpha0", "gamma",
            "augment", "beta_clip", "hier", "lam", "eta_ratio", "batch_size",
            "epochs_requested", "epochs_done"]

SHORT_MAX = 20      # the stratum where plateau_of's k=20 window swallows the whole run
LONG_MIN = 25       # the stratum where tests[-20:] really is a tail (A2's negative control)


# ---------------------------------------------------------------- primitives

def mean(xs):
    return sum(xs) / len(xs)


def sem(xs):
    """Standard error of the mean.  n=1 has NO measured spread; returning 0.0 would let a
    single-seed arm claim an infinitely tight interval and win the resolution gate for
    free, so n<2 is reported as None and any contrast touching it is UNRESOLVED."""
    if len(xs) < 2:
        return None
    m = mean(xs)
    var = sum((x - m) ** 2 for x in xs) / (len(xs) - 1)
    return math.sqrt(var / len(xs))


def window_mean(curve, k):
    """Mean of the last k epochs.  When k >= len(curve) this is the WHOLE-RUN MEAN -- the
    defect itself -- and it is returned rather than refused, because reproducing what
    aggregate.py actually computes is the point of A1."""
    if not curve:
        return None
    return sum(curve[-k:]) / len(curve[-k:])


def resolved(m_a, s_a, m_b, s_b):
    """65.6 / 94.3's resolution gate: |m_a - m_b| > 2*sqrt(sem_a^2 + sem_b^2).
    Either sem being None (n<2) makes the contrast UNRESOLVED, never resolved-by-default."""
    if s_a is None or s_b is None:
        return False
    return abs(m_a - m_b) > 2.0 * math.sqrt(s_a ** 2 + s_b ** 2)


def argmax_resolved(arms):
    """arms: {name: [values]}.  Returns (winner, margin, runner_up) if the top arm beats its
    runner-up by the resolution gate, else (None, margin, runner_up).  Fewer than 2 arms is
    never a resolved argmax."""
    if len(arms) < 2:
        return None, 0.0, None
    stats = sorted(((mean(v), sem(v), k) for k, v in arms.items()), reverse=True,
                   key=lambda t: t[0])
    (m1, s1, k1), (m2, s2, k2) = stats[0], stats[1]
    return (k1 if resolved(m1, s1, m2, s2) else None), m1 - m2, k2


def family_of(run_name):
    """Job families are named <family>-<arm>-<seed> (or with underscores).  The family is
    the first token.  `p6f` and `p6` are DIFFERENT families and must not be merged, so this
    splits on the separator only -- it never strips trailing characters."""
    return run_name.replace("_", "-").split("-")[0]


def sign(x, eps=1e-12):
    return 0 if abs(x) < eps else (1 if x > 0 else -1)


# ---------------------------------------------------------------- raw .out parsing

ARGS_RE = re.compile(r"^ARGS: (.+)$", re.M)
TEST_RE = re.compile(r"Test Accuracy: ([0-9.]+)")
ENV_RE = re.compile(r"^ENV: (.+)$", re.M)


def parse_args_line(line):
    toks, out, i = line.split(), {}, 0
    while i < len(toks):
        if toks[i].startswith("--"):
            k = toks[i][2:]
            v = toks[i + 1] if i + 1 < len(toks) and not toks[i + 1].startswith("--") else "1"
            out[k] = v
            i += 2
        else:
            i += 1
    return out


def parse_out(path):
    """Minimal re-implementation of aggregate.py's parse, deliberately NOT importing it:
    the whole point is a readout independent of the module whose column is under test.
    Returns the raw curve plus the config axes needed to form a cell key."""
    txt = open(path, errors="replace").read()
    m = ARGS_RE.search(txt)
    if not m:
        return None
    a = parse_args_line(m.group(1))
    curve = [float(x) for x in TEST_RE.findall(txt)]
    if not curve:
        return None
    env = ENV_RE.search(txt)
    e = dict(kv.split("=", 1) for kv in env.group(1).split()) if env else {}
    base = os.path.basename(path)
    return {
        "run": a.get("run-name", base),
        "job_id": (re.search(r"-(\d+)\.out$", base) or [None, ""])[1],
        "path": path,
        "curve": curve,
        "epochs_done": len(curve),
        "network": a.get("NN-name", "?"), "dataset": a.get("dataset", "?"),
        "batch_size": a.get("batch-size", ""),
        "granularity": a.get("stepsize-groups", "?"),
        "base": a.get("alg-base", a.get("optimizer", "?")),
        "meta": a.get("alg-meta", "?"), "meta_stepsize": a.get("meta-stepsize", ""),
        "alpha0": a.get("alpha0", ""), "gamma": a.get("gamma", ""),
        "augment": e.get("AUGMENT") or (re.search(r"AUGMENT=(\d)", txt) or [None, "?"])[1],
        "beta_clip": e.get("BETA_CLIP", "none"),
        "hier": (e.get("HIER", "") if e.get("HIER") != "none" else ""),
        "lam": e.get("LAM", "na"), "eta_ratio": e.get("ETA_RATIO", "na"),
        "epochs_requested": a.get("num-epochs", ""),
        "seed": a.get("seed", ""),
    }


def load_raw():
    rows = []
    for d in RUNS:
        for f in glob.glob(os.path.join(d, "**", "*.out"), recursive=True):
            r = parse_out(f)
            if r:
                rows.append(r)
    # Resubmission reuses --run-name (OPERATIONS): keep the longest run per name, exactly
    # as aggregate.py's superseded flag does.  T12 asserts the two agree.
    by_name = collections.defaultdict(list)
    for r in rows:
        by_name[r["run"]].append(r)
    live = []
    for name, group in by_name.items():
        best = max(group, key=lambda r: r["epochs_done"])
        for r in group:
            r["superseded"] = 0 if r is best else 1
        live.extend(group)
    return live


# ---------------------------------------------------------------- selftests

def selftests():
    T = []

    def ok(name, cond, detail=""):
        T.append((name, bool(cond), detail))

    # --- window arithmetic -------------------------------------------------
    c20 = list(range(1, 21))                      # 1..20, mean 10.5
    ok("T1 window_mean k=20 on a 20-epoch curve IS the whole-run mean",
       window_mean(c20, 20) == mean(c20) == 10.5, f"{window_mean(c20,20)}")
    ok("T2 window_mean k=5 on the same curve is the tail",
       window_mean(c20, 5) == 18.0, f"{window_mean(c20,5)}")
    ok("T3 the two windows differ by 7.5 on this curve -- the defect has a size",
       abs(window_mean(c20, 20) - window_mean(c20, 5)) == 7.5)
    c100 = list(range(1, 101))
    ok("T4 on a 100-epoch curve k=20 is a genuine tail, not the whole run",
       window_mean(c100, 20) == 90.5 and window_mean(c100, 20) != mean(c100))
    ok("T5 window_mean tolerates k > len (returns whole-run mean, does not raise)",
       window_mean([1.0, 2.0], 20) == 1.5)
    ok("T6 window_mean of an empty curve is None, never 0.0",
       window_mean([], 5) is None)

    # --- sem / resolution gate --------------------------------------------
    ok("T7 sem of n=1 is None, NOT 0.0 (a single seed may not win the gate for free)",
       sem([5.0]) is None)
    ok("T8 sem of identical values is 0.0", sem([3.0, 3.0, 3.0]) == 0.0)
    ok("T9 sem matches the closed form on a known sample",
       abs(sem([1.0, 2.0, 3.0]) - (1.0 / math.sqrt(3))) < 1e-12,
       f"{sem([1.0,2.0,3.0])}")
    ok("T10 resolution gate fires when the margin exceeds 2*sqrt(sum of sem^2)",
       resolved(10.0, 0.1, 9.0, 0.1) is True)
    ok("T11 resolution gate does NOT fire on a margin inside the band",
       resolved(10.0, 1.0, 9.9, 1.0) is False)
    ok("T11b gate treats a None sem as UNRESOLVED, in both slots",
       resolved(10.0, None, 1.0, 0.1) is False and resolved(10.0, 0.1, 1.0, None) is False)

    # --- argmax ------------------------------------------------------------
    a = {"lay": [10.0, 10.1, 9.9], "node": [5.0, 5.1, 4.9], "wt": [1.0, 1.1, 0.9]}
    w, marg, ru = argmax_resolved(a)
    ok("T13 argmax_resolved picks the clear winner and names the runner-up",
       w == "lay" and ru == "node" and abs(marg - 5.0) < 1e-9, f"{w},{marg},{ru}")
    b = {"lay": [10.0, 11.0, 9.0], "node": [10.1, 11.1, 9.1]}
    w2, _, _ = argmax_resolved(b)
    ok("T14 argmax_resolved returns None when the top two overlap", w2 is None, f"{w2}")
    ok("T15 a single arm is never a resolved argmax",
       argmax_resolved({"lay": [1.0, 2.0]})[0] is None)
    c = {"lay": [10.0], "node": [1.0, 1.1, 0.9]}
    ok("T16 an n=1 arm cannot win the gate even with a huge margin",
       argmax_resolved(c)[0] is None)

    # --- family parsing ----------------------------------------------------
    ok("T17 family_of splits on the separator only",
       family_of("ml5-lay-m3-s0") == "ml5" and family_of("ml5_lay_m3_s0") == "ml5")
    ok("T18 family_of keeps p6f and p6 distinct",
       family_of("p6f-a-s0") == "p6f" and family_of("p6-a-s0") == "p6")
    ok("T19 sign() is a true three-way sign",
       (sign(1.0), sign(-1.0), sign(0.0)) == (1, -1, 0))
    ok("T19b sign() calls a sub-epsilon difference a TIE, not a direction",
       sign(1e-15) == 0)

    # --- the parser, against a run verified by hand in FINDINGS 67.4 -------
    raw = load_raw()
    ok("T20 the raw corpus parses at all", len(raw) > 1500, f"{len(raw)} .out parsed")
    by = {r["run"]: r for r in raw if not r["superseded"]}
    ml5 = by.get("ml5-lay-m3-s0")
    if ml5:
        p20, p5 = window_mean(ml5["curve"], 20), window_mean(ml5["curve"], 5)
        ok("T21 ml5-lay-m3-s0 reproduces FINDINGS 67.4's k=20 value 74.822",
           abs(p20 - 74.822) < 5e-4, f"{p20:.4f}")
        ok("T22 ml5-lay-m3-s0 reproduces FINDINGS 67.4's k=5 value 86.898",
           abs(p5 - 86.898) < 5e-4, f"{p5:.4f}")
        ok("T23 ml5-lay-m3-s0 is 20 epochs, so k=20 IS the whole curve",
           ml5["epochs_done"] == 20 and abs(p20 - mean(ml5["curve"])) < 1e-12)
    else:
        ok("T21 ml5-lay-m3-s0 present in the local corpus", False, "MISSING")

    # --- CSV join ----------------------------------------------------------
    csvrows = list(csv.DictReader(open(CSV)))
    ok("T24 the CSV is the 1707-run corpus the docs describe",
       len(csvrows) == 1707, f"{len(csvrows)}")
    csv_live = {r["run"] for r in csvrows if r["superseded"] != "1"}
    raw_live = {r["run"] for r in raw if not r["superseded"]}
    ok("T12 locally re-derived superseded flags agree with the CSV's",
       csv_live == raw_live,
       f"csv_only={len(csv_live-raw_live)} raw_only={len(raw_live-csv_live)}")

    # A1's object must actually exist in the CSV before A1 is meaningful
    short_csv = [r for r in csvrows if r["superseded"] != "1" and r["plateau"]
                 and int(r["epochs_done"]) <= SHORT_MAX]
    ok("T25 the affected stratum is the 380 runs FINDINGS 67.4's W4 counted",
       len(short_csv) == 380, f"{len(short_csv)}")

    # --- cell construction -------------------------------------------------
    cells = build_cells(raw)
    ok("T26 the affected stratum yields the 22 matched cells the header claims",
       len(cells) == 22, f"{len(cells)}")
    npair = sum(len(c["arms"]) * (len(c["arms"]) - 1) // 2 for c in cells)
    ok("T27 those cells carry the 97 pairwise contrasts the header claims",
       npair == 97, f"{npair}")
    ok("T28 every arm in every scored cell has n>=2 (the gate needs a spread)",
       all(len(v) >= 2 for c in cells for v in c["arms"].values()))
    ok("T29 every scored cell holds granularity as the ONLY free axis",
       all(len({tuple(r[k] for k in CELL_KEY) for arm in c["arms_rows"].values()
                for r in arm}) == 1 for c in cells))
    ok("T30 every run in every scored cell is inside the affected stratum",
       all(r["epochs_done"] <= SHORT_MAX for c in cells
           for arm in c["arms_rows"].values() for r in arm))

    npass = sum(1 for _, p, _ in T if p)
    for name, p, detail in T:
        print(f"  [{'PASS' if p else 'FAIL'}] {name}" + (f"   ({detail})" if detail else ""))
    print(f"\n{npass}/{len(T)} selftests pass")
    return npass == len(T)


# ---------------------------------------------------------------- cell construction

def build_cells(raw, k_arm_min=2):
    """Matched cells inside the affected stratum: everything in CELL_KEY held fixed,
    `granularity` free, every arm at n >= k_arm_min, at least 2 arms."""
    short = [r for r in raw if not r["superseded"] and r["epochs_done"] <= SHORT_MAX]
    groups = collections.defaultdict(lambda: collections.defaultdict(list))
    for r in short:
        groups[tuple(r[k] for k in CELL_KEY)][r["granularity"]].append(r)
    cells = []
    for key, arms in groups.items():
        keep = {g: rs for g, rs in arms.items() if len(rs) >= k_arm_min}
        if len(keep) < 2:
            continue
        cells.append({
            "key": dict(zip(CELL_KEY, key)),
            "families": sorted({family_of(r["run"]) for rs in keep.values() for r in rs}),
            "arms_rows": keep,
            "arms": {g: [window_mean(r["curve"], 20) for r in rs] for g, rs in keep.items()},
        })
    cells.sort(key=lambda c: (c["key"]["network"], c["key"]["meta"],
                              c["key"]["meta_stepsize"], c["families"]))
    return cells


def arm_values(cell, k):
    return {g: [window_mean(r["curve"], k) for r in rs] for g, rs in cell["arms_rows"].items()}


# ---------------------------------------------------------------- the score

def score():
    raw = load_raw()
    csvrows = list(csv.DictReader(open(CSV)))
    out = {}
    print("=" * 88)
    print("C68 -- BLAST RADIUS OF THE k=20 `plateau` DEFECT.  Registered before scoring.")
    print("=" * 88)

    # ---------------- A1 / A2 : mechanical gate --------------------------
    bycsv = {r["run"]: r for r in csvrows if r["superseded"] != "1"}
    byraw = {r["run"]: r for r in raw if not r["superseded"]}

    a1_ok = a1_n = 0
    a1_worst = 0.0
    for name, cr in bycsv.items():
        if not cr["plateau"] or int(cr["epochs_done"]) > SHORT_MAX:
            continue
        rr = byraw.get(name)
        if not rr:
            continue
        a1_n += 1
        d = abs(float(cr["plateau"]) - mean(rr["curve"]))
        a1_worst = max(a1_worst, d)
        a1_ok += d <= 1e-3
    a1_frac = a1_ok / a1_n if a1_n else 0.0
    print(f"\nA1  short stratum (<= {SHORT_MAX} ep): recorded plateau == whole-run mean in "
          f"{a1_ok}/{a1_n} = {100*a1_frac:.2f}%  (worst |diff| {a1_worst:.4f} pp)")
    print(f"    -> {'PASS' if a1_frac >= 0.995 else 'FAIL'} (bar 99.5%)")
    out["A1"] = {"n": a1_n, "agree": a1_ok, "frac": a1_frac, "worst": a1_worst}

    a2_ok = a2_n = 0
    a2_d5 = []
    for name, cr in bycsv.items():
        if not cr["plateau"] or int(cr["epochs_done"]) < LONG_MIN:
            continue
        rr = byraw.get(name)
        if not rr:
            continue
        a2_n += 1
        a2_ok += abs(float(cr["plateau"]) - window_mean(rr["curve"], 20)) <= 1e-3
        a2_d5.append(abs(float(cr["plateau"]) - window_mean(rr["curve"], 5)))
    a2_frac = a2_ok / a2_n if a2_n else 0.0
    a2_med = sorted(a2_d5)[len(a2_d5) // 2] if a2_d5 else 0.0
    print(f"\nA2  long stratum (>= {LONG_MIN} ep): recorded plateau == mean(last 20) in "
          f"{a2_ok}/{a2_n} = {100*a2_frac:.2f}%   [bar 99.5%]")
    print(f"    median |recorded - mean(last 5)| = {a2_med:.3f} pp   [bar > 0.05 pp]")
    a2_pass = a2_frac >= 0.995 and a2_med > 0.05
    print(f"    -> {'PASS' if a2_pass else 'FAIL'}")
    out["A2"] = {"n": a2_n, "frac": a2_frac, "median_d5": a2_med}

    if not (a1_frac >= 0.995 and a2_pass):
        print("\nMECHANICAL GATE FAILED -- B/C NOT SCORED, as registered.")
        return out

    # ---------------- B1 : argmax stability ------------------------------
    cells = build_cells(raw)
    print(f"\n{'='*88}\nB1  ARGMAX STABILITY over {len(cells)} matched cells "
          f"(granularity free, everything else fixed)\n{'='*88}")
    hdr = (f"{'family':<12}{'net':<15}{'meta':<7}{'ms':<7}{'a0':<7}"
           f"{'argmax k=20':<16}{'argmax k=5':<16}{'verdict'}")
    print(hdr); print("-" * len(hdr))
    both, agree, rows_b1 = 0, 0, []
    for c in cells:
        v20, v5 = arm_values(c, 20), arm_values(c, 5)
        w20, m20, _ = argmax_resolved(v20)
        w5, m5, _ = argmax_resolved(v5)
        if w20 and w5:
            both += 1
            agree += (w20 == w5)
            verdict = "AGREE" if w20 == w5 else "*** FLIP ***"
        elif w20 and not w5:
            verdict = "res->UNRES"
        elif w5 and not w20:
            verdict = "UNRES->res"
        else:
            verdict = "unres both"
        k = c["key"]
        print(f"{'+'.join(c['families']):<12}{k['network']:<15}{k['meta']:<7}"
              f"{k['meta_stepsize']:<7}{k['alpha0']:<7}"
              f"{(w20 or '-- unresolved'):<16}{(w5 or '-- unresolved'):<16}{verdict}")
        rows_b1.append({"families": c["families"], **{kk: k[kk] for kk in
                        ("network", "meta", "meta_stepsize", "alpha0")},
                        "w20": w20, "w5": w5, "verdict": verdict})
    frac = agree / both if both else float("nan")
    print(f"\nB1  doubly-resolved cells: {both}/{len(cells)};  argmax agrees "
          f"{agree}/{both} = {100*frac:.1f}%")
    b1 = ("CONFINED (a)" if frac >= 0.90 else
          "SYSTEMIC (b)" if frac < 0.75 else "UNDECIDED (c)")
    print(f"    -> {b1}   [registered: >=90% CONFINED, <75% SYSTEMIC]")
    out["B1"] = {"cells": len(cells), "both_resolved": both, "agree": agree,
                 "frac": frac, "branch": b1, "rows": rows_b1}

    # ---------------- B2 / B3 / B4 / B5 : pairwise ------------------------
    flips = pairs = res20 = res5 = res20_unres5 = 0
    d20s, d5s = [], []
    flip_detail = []
    for c in cells:
        v20, v5 = arm_values(c, 20), arm_values(c, 5)
        names = sorted(v20)
        for i in range(len(names)):
            for j in range(i + 1, len(names)):
                a, b = names[i], names[j]
                pairs += 1
                d20 = mean(v20[a]) - mean(v20[b])
                d5 = mean(v5[a]) - mean(v5[b])
                d20s.append(abs(d20)); d5s.append(abs(d5))
                if sign(d20) != sign(d5):
                    flips += 1
                    flip_detail.append((c["families"], c["key"]["network"],
                                        c["key"]["meta_stepsize"], a, b, d20, d5))
                r20 = resolved(mean(v20[a]), sem(v20[a]), mean(v20[b]), sem(v20[b]))
                r5 = resolved(mean(v5[a]), sem(v5[a]), mean(v5[b]), sem(v5[b]))
                res20 += r20; res5 += r5
                res20_unres5 += (r20 and not r5)

    print(f"\n{'='*88}\nB2  SIGN STABILITY over all {pairs} pairwise contrasts\n{'='*88}")
    print(f"    sign flips k=20 -> k=5: {flips}/{pairs} = {100*flips/pairs:.1f}%"
          f"   [registered: >10% => the window materially reorders the stratum]")
    for f in flip_detail[:15]:
        print(f"      {'+'.join(f[0]):<10} {f[1]:<15} ms={f[2]:<6} {f[3]} vs {f[4]}: "
              f"k20 {f[5]:+.3f} -> k5 {f[6]:+.3f}")
    if len(flip_detail) > 15:
        print(f"      ... and {len(flip_detail)-15} more")
    out["B2"] = {"pairs": pairs, "flips": flips, "frac": flips / pairs}

    print(f"\nB3  RESOLUTION COLLAPSE")
    print(f"    resolved at k=20: {res20}/{pairs} = {100*res20/pairs:.1f}%")
    print(f"    resolved at k=5 : {res5}/{pairs} = {100*res5/pairs:.1f}%")
    cf = res20_unres5 / res20 if res20 else float("nan")
    print(f"    resolved@20 but UNRESOLVED@5: {res20_unres5}/{res20} = {100*cf:.1f}%"
          f"   [registered: >30% => the stratum's resolving power is a window artifact]")
    out["B3"] = {"res20": res20, "res5": res5, "collapse": res20_unres5, "frac": cf}

    print(f"\nB4  POWER CONTROL (registered to KILL B1-B3 if it fails)")
    p5f = res5 / pairs if pairs else 0.0
    print(f"    contrasts resolved under k=5: {res5}/{pairs} = {100*p5f:.1f}%   [bar >=10%]")
    print(f"    -> {'PASS -- k=5 retains discriminating power' if p5f >= 0.10 else 'FAIL -- B1-B3 UNINTERPRETABLE'}")
    out["B4"] = {"frac_res5": p5f, "pass": p5f >= 0.10}

    md20 = sorted(d20s)[len(d20s) // 2]
    md5 = sorted(d5s)[len(d5s) // 2]
    print(f"\nB5  EFFECT-SIZE SHRINKAGE (descriptive)")
    print(f"    median |delta| k=20: {md20:.3f} pp;  k=5: {md5:.3f} pp;  "
          f"ratio {md20/md5 if md5 else float('inf'):.2f}x")
    out["B5"] = {"median_d20": md20, "median_d5": md5}

    # ---------------- C1 : the CIFAR-100 gain ----------------------------
    print(f"\n{'='*88}\nC1  THE CIFAR-100 GRANULARITY GAIN UNDER BOTH WINDOWS\n{'='*88}")
    c1 = []
    for c in cells:
        if "c100" not in c["key"]["network"] and c["key"]["dataset"] != "CIFAR100":
            continue
        v20, v5 = arm_values(c, 20), arm_values(c, 5)
        names = sorted(v20, key=lambda g: -mean(v20[g]))
        best, worst = names[0], names[-1]
        g20 = mean(v20[best]) - mean(v20[worst])
        g5 = mean(v5[best]) - mean(v5[worst])
        rel = abs(g5 - g20) / g20 * 100 if g20 else float("nan")
        print(f"    {'+'.join(c['families']):<10} {c['key']['network']:<15} "
              f"meta={c['key']['meta']:<6} n={ {g: len(v) for g, v in v20.items()} }")
        for g in names:
            print(f"       {g:<18} k20 {mean(v20[g]):7.3f}   k5 {mean(v5[g]):7.3f}")
        print(f"       spread ({best} - {worst}):  k20 {g20:7.3f} pp   k5 {g5:7.3f} pp"
              f"   -> {rel:.1f}% relative move {'*** FLAG ***' if rel > 20 else ''}")
        c1.append({"families": c["families"], "net": c["key"]["network"],
                   "gain20": g20, "gain5": g5, "rel": rel})
    out["C1"] = c1

    # ---------------- C2 : citation audit --------------------------------
    print(f"\n{'='*88}\nC2  CITATION AUDIT OF THE AFFECTED FAMILIES\n{'='*88}")
    short = [r for r in raw if not r["superseded"] and r["epochs_done"] <= SHORT_MAX
             and r["run"] in bycsv and bycsv[r["run"]]["plateau"]]
    fams = collections.Counter(family_of(r["run"]) for r in short)
    doctxt = ""
    for f in glob.glob(os.path.join(DOCS, "*.md")):
        doctxt += open(f, errors="replace").read()
    cited, orphan = [], []
    scored_fams = {f for c in cells for f in c["families"]}
    print(f"{'family':<10}{'n runs':<9}{'in docs/':<10}{'scored here':<13}")
    print("-" * 42)
    for f, n in fams.most_common():
        hit = len(re.findall(rf"\b{re.escape(f)}[-_ ]", doctxt))
        (cited if hit else orphan).append((f, n))
        print(f"{f:<10}{n:<9}{(str(hit)+' hits') if hit else 'ORPHAN':<10}"
              f"{'yes' if f in scored_fams else 'no':<13}")
    print(f"\n    cited in docs/: {len(cited)} families / {sum(n for _, n in cited)} runs")
    print(f"    ORPHANS       : {len(orphan)} families / {sum(n for _, n in orphan)} runs"
          f"  -> {[f for f, _ in orphan]}")
    out["C2"] = {"cited": cited, "orphan": orphan}

    print(f"\n{'='*88}\nSUMMARY: A1 {'PASS' if a1_frac>=0.995 else 'FAIL'} | "
          f"A2 {'PASS' if a2_pass else 'FAIL'} | B1 {b1} | "
          f"B2 flips {100*flips/pairs:.1f}% | B3 collapse {100*cf:.1f}% | "
          f"B4 {'PASS' if p5f>=0.10 else 'FAIL'}\n{'='*88}")
    return out
