#!/bin/bash
# cRT1_rule20.sh -- CORRECTIONS 300: RULE 20 at current coverage for `crt1` (a copy of cGW1_rule20.sh with crt1's table;
# the ENV half is analysis/crt1_rule20_envaudit.py).  READ-ONLY: no accuracy number is opened.
# analysis/argsline_guard.py is used UNEDITED from the stage.
#   * `meta-stepsize`, `alpha0` and `stepsize-groups` VARY BY ARM, so the batch-consistency half is given
#     `--vary meta-stepsize --vary alpha0 --vary stepsize-groups` as well as `--vary seed --vary run-name`; the per-run
#     half then pins ALL THREE to the arm's own registered tokens, and `weight-decay-base` to 5e-4 on every run, so a run
#     carrying another config's token is a PER-RUN VIOLATION.
#   * the ENV line carries none of the three; the ENV half checks the single registered ENV line and the OFF witnesses.
#   bash bin/cRT1_rule20.sh            (run from ~/stage_crt1 on cluster_b)
set +u
source /etc/profile >/dev/null 2>&1
module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1
source /cluster/account_b/metaopt/envs/mo/bin/activate
export METAOPT_WS=/cluster/account_b/metaopt
WS=$METAOPT_WS; R=$WS/runs; P=crt1
S="$(cd "$(dirname "$0")/.." && pwd)"
G=$S/analysis/argsline_guard.py
echo "RULE20 run utc=$(date -u +%FT%TZ) host=$(hostname) batch=$P guard_sha=$(sha256sum $G | cut -c1-16)"
HF="--expect optimizer=HF --expect alg-base=SGDm --expect momentum-param-base=0.99 --expect weight-decay-base=5e-4 --expect alg-meta=Lion --expect momentum-param-meta=0.99 --expect Lion-beta2-meta=0.9 --expect weight-decay-meta=0 --expect dataset=CIFAR10 --expect batch-size=100 --expect max-time=999:00:00 --expect gamma=1"
NET=ResNet18
echo "  .out files: $(ls $R/$P-*.out 2>/dev/null | wc -l)   with ARGS: $(grep -l -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | wc -l)   squeue: $(squeue -h -u account_b -o '%j %T' | awk -v p="^$P-" '$1 ~ p {print $2}' | sort | uniq -c | tr '\n' ' ')"
OUT=$(python3 $G $R --name $P- --batch-consistency --strict --vary seed --vary run-name --vary meta-stepsize --vary alpha0 --vary stepsize-groups 2>&1); RC=$?
echo "  BATCH-CONSISTENCY --name $P- --vary seed --vary run-name --vary meta-stepsize --vary alpha0 --vary stepsize-groups -> rc=$RC  $(echo "$OUT" | grep -E '^argsline_guard:|^VERDICT|no candidate|NONE carried|ACROSS-RUN|^    --' | tr '\n' ' ')"
n=0; bad=0; noargs=0
for f in $(ls $R/$P-*.out 2>/dev/null); do
  b=$(basename $f .out); rn=${b%-*}; rest=${rn#$P-}; arm=${rest%-s*}; seed=${rest##*-s}
  grep -q -m1 '^ARGS:' "$f" || { noargs=$((noargs+1)); continue; }
  n=$((n+1))
  case $arm in
    *M2) MS=1e-4; A0=1e-3 ;;
    *A2) MS=1e-4; A0=1e-2 ;;
    *M3) MS=3e-4; A0=1e-3 ;;
    *M1) MS=3e-5; A0=1e-3 ;;
    *) bad=$((bad+1)); echo "  PER-RUN VIOLATION $b: UNKNOWN CONFIG"; continue ;;
  esac
  case $arm in
    ch??) GR=chunk777 ;; nd??) GR=nodewise ;; k01??) GR=scalar ;;
    *) bad=$((bad+1)); echo "  PER-RUN VIOLATION $b: UNKNOWN GRAIN"; continue ;;
  esac
  case " 176 177 178 " in *" $seed "*) ;; *) bad=$((bad+1)); echo "  PER-RUN VIOLATION $b: seed $seed is not registered"; continue ;; esac
  E="$HF --expect meta-stepsize=$MS --expect alpha0=$A0 --expect NN-name=$NET --expect num-epochs=100 --expect stepsize-groups=$GR"
  O=$(python3 $G "$f" $E --expect seed=$seed --expect save-directory=$R/$P --expect run-name=$rn 2>&1); RCX=$?
  NFG=$(echo "$O" | grep -o 'no repeated flag ([0-9]* flags)' | grep -o '[0-9]*')
  if [ $RCX != 0 ] || [ "$NFG" != 20 ]; then bad=$((bad+1)); echo "  PER-RUN VIOLATION $b rc=$RCX flags=$NFG want 20 (ms=$MS a0=$A0 grain=$GR)"; echo "$O" | grep -E '!!!|DESIGN|declared' | sed 's/^/      /'; fi
done
echo "  PER-RUN --expect ($P, each run pinned to its OWN config's meta step size and alpha0, its OWN grain, wd 5e-4): $n runs with an ARGS line checked, $bad violations, $noargs .out without ARGS yet"
echo "  COVERAGE across the started runs (read from the runs' ARGS lines; registered counts first):"
echo "    registered meta-stepsize: 1e-4 x18, 3e-4 x9, 3e-5 x9 | alpha0: 1e-3 x27, 1e-2 x9 | weight-decay-base: 5e-4 x36"
grep -h -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | grep -o -- '--meta-stepsize [^ ]*' | sort | uniq -c | sed 's/^/    /'
grep -h -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | grep -o -- '--alpha0 [^ ]*' | sort | uniq -c | sed 's/^/    /'
grep -h -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | grep -o -- '--weight-decay-base [^ ]*' | sort | uniq -c | sed 's/^/    /'
echo "  GRAIN COVERAGE (registered: 12 each):"
grep -h -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | grep -o -- '--stepsize-groups [^ ]*' | sort | uniq -c | sed 's/^/    /'
python3 $S/analysis/crt1_rule20_envaudit.py $R | sed 's/^/  /'
echo "RULE20 end utc=$(date -u +%FT%TZ)"
