#!/bin/bash
# =============================================================================
# c83_gen_c100_screen.sh -- `gc1`, DOES THE TAIL PRESCRIPTION SURVIVE A SECOND
# DATASET?  AN EXPLICIT SCREEN WITH A PRE-REGISTERED STOPPING RULE.
# Runs on cluster_a ONLY (needs PATCH_CHUNKWISE; cluster_b does not carry it).
# 8 jobs: {nodewise, chunk771} x seeds 0-3, ResNet18_c100 / CIFAR-100, ms=1e-3,
# 100 ep, AUGMENT=1, probe ON.  NO NEW PATCH -- chunkwise already ships, and the
# CIFAR-100 data is already staged (cifar-100-python.tar.gz, 169,001,437 B; and
# load_data.py dispatches CIFAR100).
#
# THIS IS A SCREEN, NOT A REPLICATION, AND IT IS RANKED THIRD FOR A REASON.
#   The batch buys an OPTION on CIFAR-100, not an answer about it.  Its own
#   pre-registered STOPPING RULE (S1 below) can only ever say "spend 30 more jobs"
#   or "close CIFAR-100 as unaffordable".  It CANNOT say "D does not generalise".
#   That sentence is forbidden to this batch by its own arithmetic and the ban is
#   registered here, before the data exists, in the DOES-NOT-CLAIM section.
#
# WHY IT EXISTS
#   The paper is the tail prescription: D = chunk777 - nodewise, count-matched at
#   m = 14,421 vs 14,420 on ResNet18/CIFAR-10, reading +0.485 / +0.581 / +0.697 /
#   +0.727 across two meta-stepsizes and two disjoint seed sets, none below +0.48.
#   Every one of those readings is ResNet18/CIFAR-10.  There is not one chunkwise
#   or nodewise1d run on ANY other dataset or architecture in 1,891 rows.  A
#   reviewer's first question is whether the effect is a property of partitions or
#   a property of one net on one dataset, and the campaign currently cannot answer.
#
# WHY THE SCREEN IS AFFORDABLE AND THE FULL TEST IS NOT -- costed, not asserted.
#   Measured within-cell sd of plateau5 on the HEALTHY ResNet18_c100 100-ep cells
#   (guard 2e re-derives all of this from the CSV at submit time):
#       layerwise @a0=1e-3  0.447 (n=8)   layerwise @a0=1e-6  0.471 (n=8)
#       weightwise          0.309 (n=3)   nodewise            0.084 (n=3)
#     -> pooled 0.419 on 18 df.  The scalar (22.560) and blk6 (51.584) cells are in
#        a COLLAPSED regime and are excluded -- their sd measures a collapse, not
#        seed noise.  That is 2-3x ResNet18/CIFAR-10's 0.157 at 0.65x the cost per
#        run (0.40 min/epoch measured over n=22, vs 0.62).
#   So se(D) at n=4 v 4 is 0.419*sqrt(2/4) = **0.296 pp**, and:
#       detecting the CIFAR-10-sized D = +0.5 at t = 3 needs n = 13 per arm
#         -> 30 jobs, ~20 GPU-h, for ONE reading;
#       detecting D = +1.0 at t = 3 needs n = 4  -> 8 jobs, 5.3 GPU-h.  THIS BATCH.
#   The screen is therefore powered for ONE hypothesis only: that the effect is
#   AMPLIFIED on CIFAR-100.  That is a real prior, not a hope -- re-derived from
#   the CSV by guard 2d at submit time:
#       layerwise - scalar  =  +47.034 pp on ResNet18_c100/CIFAR-100
#       layerwise - scalar  =   +3.357 pp on ResNet18/CIFAR-10
#     -> granularity effects run **14.0x larger** on CIFAR-100 at the same
#        ms=1e-3, alpha0=1e-3, 100 ep, AUGMENT=1, same box.
#   If D scales with granularity effects at all, +0.5 becomes >= 1.0 and 8 jobs
#   resolve it at t >= 3.4.  If it does not, the screen is silent by construction
#   and the honest deliverable is the registered sentence
#   "the CIFAR-100 noise floor forbids this question at reasonable cost",
#   which under this project's own rules is a valid result.
#
# THE TWO ARMS.  ONE MATCHED-COUNT PAIR, MEASURED NOT DERIVED.
#     node  nodewise   m = 14,600  |  ch  chunk771  m = 14,595   (5 groups apart)
#   777 and 2325 are ResNet18/CIFAR-10 constants found by search against THAT net's
#   m; they are wrong on every other network.  K=771 was found by binary search on
#   the (monotone non-increasing) chunk count for ResNet18_c100 and is MEASURED
#   from the ALLOCATED beta by guard 4 before anything is submitted.
#   THE -5 GROUP RESIDUAL IS QUANTIFIED, NOT WAVED AWAY.  The count slope,
#   re-derived from the CSV by guard 2f, is +0.491 pp per decade of m (chunk777
#   n=9 92.580 -> chunk2325 n=6 92.812 over 0.4732 decades).  -5 groups is
#   1.487e-4 decades = **0.000073 pp** of bias, 680x below the 0.05 pp resolution
#   floor and smaller than the +1-group residual chunk777 itself already carries.
#   STRUCTURAL MATCH TO THE CIFAR-10 CONTRAST, measured on the login node:
#   ResNet18_c100 has 62 tensors, 41 of them 1-D, and the largest 1-D tensor is
#   512 elements < K=771, so ALL 41 1-D TENSORS GET EXACTLY ONE CHUNK.  This is
#   the same comparator as chunk777 on ResNet18, and it is NOT what would happen
#   on ResNet50 (largest 1-D tensor 2048 > any matching K, so the comparator would
#   split the very tensors whose merging is the prescription).
#   The tail being merged is the same shape too: 9,700 size-1 nodewise groups =
#   66.44% of m over 0.0865% of the weights, against ResNet18/CIFAR-10's
#   9,610 / 66.64% / 0.0860%.
#
# WHY ms=1e-3, AND WHAT IT COSTS.  ms=1e-3 is the ONLY meta stepsize with any
#   100-epoch CIFAR-100 data at all -- every c100 row in the corpus is at 1e-3 --
#   so it is the only stepsize at which the fresh nodewise arm can double as a
#   batch-offset diagnostic against an existing cell (S2).  THE PRICE, registered
#   in advance: ms=1e-3 is not known to be either arm's argmax on this dataset,
#   there is no CIFAR-100 ms ladder in the corpus, and RULE 11 is therefore NOT
#   satisfied.  On ResNet18/CIFAR-10 tuning each rung to its own ms shrank the
#   granularity ladder by ~86%.  This batch is a SHARED-ms contrast and says so.
#
# ================= THE BOX, AND WHY A BIND IS THE EXPECTED OUTCOME ===========
#   BETA_CLIP=-15:-2.3026 is NOT a free choice here.  Every 100-ep CIFAR-100 run
#   in the corpus, including the 71.415 nodewise cell S2 reads against, sits in
#   exactly this box; changing it would destroy the only comparator the batch has.
#   And it WILL bind.  Registered now, with the arithmetic, not discovered later:
#     * HF.py's Lion meta update with --weight-decay-meta 0 is
#         beta <- beta - ms*sign(.),  so |dbeta| = ms exactly, every update.
#       beta_0 = ln(1e-3) = -6.907755;  T = 100 ep x 500 steps = 50,000 updates;
#       ms*T = 50.0 nats.  The reachable interval is [-56.91, +43.09], which
#       STRICTLY CONTAINS the box.  The ceiling -2.3026 sits 4.6052 nats above
#       beta_0, i.e. reachable after 4,606 updates = **epoch 9.3 at the earliest**.
#     * Empirically: `ar1` ran four arms x 3 seeds in THIS EXACT BOX at ms=3e-4 and
#       bound on 12/12 (rec_lo 0.4521-0.4597), which VOIDED its A3 field gate.
#       ms here is 3.3x LARGER.
#   SO THE REGISTRATION IS:
#     (a) A BIND DOES NOT VOID S1.  S1 is accuracy-only, within-batch, and both
#         arms sit in the SAME box -- exactly the condition under which every D the
#         campaign quotes (mm1/pp1/ar1/cc1) was measured.  The box is a shared
#         condition of the contrast, not a confound in it.  gc1's numbers are
#         statements about a BOXED optimizer and must always carry the box.
#     (b) NO FIELD STATISTIC MAY BE READ FROM THIS BATCH.  There is no N_eff/m gate,
#         no rho gate, no sign-agreement gate, and the scorer refuses to compute
#         one.  The probe is carried as INSTRUMENTATION for (c), not as a gate.
#     (c) **S0.4, THE ONE THING THE BOX CAN STILL VOID: OCCUPANCY ASYMMETRY.**
#         nodewise and chunk771 have near-identical m but DIFFERENT group-size
#         distributions, so they need not press the guard equally, and an
#         asymmetric clamp is a real alternative explanation for any D.  BAR,
#         fixed here: |rec_lo(ch) - rec_lo(node)| <= 0.10 AND the same for rec_hi,
#         on cell means.  If either exceeds 0.10, S1 is DEMOTED to
#         **UNINTERPRETABLE -- ASYMMETRIC GUARD**, the screen returns NO READING,
#         and the stopping rule resolves to CLOSE-UNRESOLVED.  It does NOT resolve
#         to "no effect".  A guard artefact and a null are different findings and
#         are kept different here.
# =============================================================================
#
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
#
#   S0    VALIDITY.  epochs_done == epochs_requested == 100; collapsed == 0;
#         n_records == 10000 on every probe dir; plateau5 present.
#   S0.2  n_beta EXACT on EVERY record: node 14600, ch 14595.  BOTH MEASURED by
#         guard 4 from the ALLOCATED beta on the real built ResNet18_c100 before
#         anything is submitted -- never from a formula, never from a comment.
#   S0.3  THE INSTRUMENT FIRED.  neg_counts.json present, n_tot == n_beta, and the
#         npy shape READ FROM ITS HEADER == (n_tot,).  (Header, never file size --
#         that inference produced c74's false VOID on 12/12 arms.)
#   S0.4  **BOX OCCUPANCY.  SCORED FIRST, PER SEED, NEVER POOLED.**  A bind is
#         EXPECTED (see above) and does not void S1.  What voids S1 is ASYMMETRY
#         between the arms: bar 0.10 absolute on rec_lo and on rec_hi.
#
#   S1    **THE PRIMARY.  THE SCREEN.  FIVE-WAY, SYMMETRIC ON |D|, WITHIN-BATCH,
#         COUNT-MATCHED.**
#         D_c100 = plateau5(chunk771) - plateau5(nodewise), n=4 v 4, se ~ 0.30.
#         The gate is the threshold on the DIFFERENCE OF MEANS.  Both arms are in
#         ONE batch at ONE seed set, so a paired reading is admissible under the
#         amended campaign standard and the point estimate is identical either way
#         in a balanced design; the se printed is the conservative Welch one.
#           D >= +1.0        -> **SCREEN PASSES.  PROCEED** to a powered n=15 cell.
#                (t >= 3.4 at the registered se.)  The amplification prior holds
#                and CIFAR-100 becomes affordable.
#           (+0.30, +1.0)    -> **SCREEN NEGATIVE -- CLOSE AS UNAFFORDABLE.**  D may
#                well be real and CIFAR-10-sized here; this batch cannot tell, and
#                resolving it would cost 30 jobs / ~20 GPU-h for one reading.  This
#                is the branch the batch was designed to be able to return, and
#                under the project's own rules it is a valid deliverable.
#           [-0.30, +0.30]   -> **NO AMPLIFICATION.**  The 14x amplification prior
#                is refuted FOR THIS CONTRAST.  **STILL NOT A NULL** -- at se 0.30
#                this band does not exclude the CIFAR-10 D of +0.5.  Close as
#                unaffordable, and say WHY in exactly those words.
#           (-1.0, -0.30)    -> **SCREEN NEGATIVE, REVERSED SIGN -- CLOSE.**
#           D <= -1.0        -> **REVERSAL AT SCREEN STRENGTH.  PROCEED.**  The
#                stopping rule is on |D| and is registered symmetric: a reversal
#                this large is as much worth 30 jobs as a confirmation, and is
#                reported as an inversion of the tail prescription on a second
#                dataset, never as noise.
#         PER-SEED SIGN CHECK, k/4, reported beside the mean.  It is a VALID binary
#         gate here and only here: both arms are in the SAME submission batch at
#         the SAME seeds, which is the one configuration STANDING RULE 14 leaves
#         untouched.  It is SUPPORTING, not the gate.
#
#   S2    **THE SECONDARY.  A DECLARED RULE-14 CROSS-BATCH DIAGNOSTIC.**
#         nodewise here vs the existing 100-ep ResNet18_c100 cell at the identical
#         15-axis signature: c100f-node-s{0,1,2}, plateau5 71.415 +-0.084 (n=3).
#         **WELCH, UNPAIRED, NEVER POOLED WITH IT.**  Bar +-0.50 pp, the campaign's
#         cross-batch reproducibility bar, and the row additionally carries the
#         measured irreducible cross-batch floor sqrt(2)*sd_batch = 0.20-0.29 pp,
#         which does not shrink with n.  It is ALSO cross-ACCOUNT: the c100f cell
#         ran on account_b and this batch runs on account_a (cluster_b lacks
#         PATCH_CHUNKWISE, so there is no choice).  Account was measured null
#         (F=1.19, sd 0.038) and that is stated, not assumed away.
#         **S2 CANNOT VOID S1.**  S1 is a within-batch difference and any offset
#         common to its two arms cancels exactly.
#
#   S3    THE NOISE METER.  **DESCRIPTIVE, NO DIRECTION REGISTERED.**  Per-arm sd
#         here vs the 0.419 pooled planning sd.  Its ONLY job is to re-cost the
#         PROCEED branch: if sd lands materially above 0.419 the n=15 cell must be
#         re-costed BEFORE it is submitted, and if materially below, n falls.  It
#         cannot gate S1 and is printed after it.
#
#   S4    THE STRUCTURE METER.  **DESCRIPTIVE.**  The size-1 tail census on
#         ResNet18_c100 (9,700 groups, 66.44% of m, 0.0865% of weights) beside
#         ResNet18/CIFAR-10's (9,610 / 66.64% / 0.0860%), and the 41/41 one-chunk
#         property.  All MEASURED by guard 4.  It records that the two datasets'
#         contrasts are structurally the same object; it tests nothing.
#
# WHAT THIS BATCH EXPLICITLY DOES NOT CLAIM
#   * **IT CANNOT CLAIM D FAILS TO GENERALISE.**  se(D) = 0.30, so a return of
#     D = 0.00 and a return of D = +0.50 (the CIFAR-10 value) are 1.7 se apart and
#     this batch does not separate them.  Any sentence of the form "the tail
#     prescription does not hold on CIFAR-100" is FORBIDDEN to gc1 regardless of
#     what lands.  The ONLY negative sentence it may write is the registered
#     "the CIFAR-100 noise floor forbids this question at reasonable cost".
#   * It does not test the MECHANISM.  There is no nodewise1d arm and no chunk2293
#     arm here, so the tail-removed leg G -- the contrast that located the effect
#     in the degenerate size-1 tail on CIFAR-10 -- is entirely unmeasured on
#     CIFAR-100.  Registered as unaffordable: at sd 0.419 the +-0.15 NULL band that
#     G's CIFAR-10 reading turns on would need n=34 per arm.
#   * It does not satisfy RULE 11.  ms=1e-3 is a shared, untuned stepsize chosen
#     for the S2 diagnostic; neither arm's CIFAR-100 argmax is measured or claimed.
#   * It does not generalise across ARCHITECTURE.  It moves the dataset axis with
#     the architecture family held; ResNet34/50 and ResNet34_c100 are untouched,
#     and ResNet50 is structurally excluded from this comparator entirely (its
#     largest 1-D tensor is 2048, above any count-matching K).
#   * Its arms are BOXED, at ms=1e-3, in a box that is expected to bind.  No field
#     statistic may be quoted from them.
#   * n=4, one architecture, one dataset, one stepsize, one box.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 tests/test_chunkwise.py            (CPU; asserts a config equals itself first)
#   2. python3 tests/test_granularity.py          (CPU; covers the nodewise arm)
#   3. python3 analysis/c52_boxfree.py --selftest
#   4. python3 analysis/c83_gc1_score.py --selftest
#   5. `gc1` registered in c55_neff_noise.BOXES BEFORE scoring, from THIS CLIP=.
#   6. plateau5 (mean of the last 5 test epochs) is PRIMARY.  The CSV `plateau`
#      column is mean-of-last-20 and is NOT comparable across budgets.
#
# USAGE
#   bash c83_gen_c100_screen.sh            # dry run
#   bash c83_gen_c100_screen.sh --submit   # submits, ONLY if every guard passes
# =============================================================================
set -u

SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

GUARD_FAILS=0
guard_fail() {
  GUARD_FAILS=$((GUARD_FAILS + 1))
  echo "!!! GUARD FAIL: $1"
  if [ "$SUBMIT" = "1" ]; then
    echo "!!! SUBMIT ABORTED -- nothing was submitted."
    exit 1
  fi
  echo "    (dry run: continuing so the emitter can be validated off-cluster."
  echo "     THIS BATCH MUST NOT BE SUBMITTED UNTIL THIS GUARD PASSES ON cluster_a.)"
}

USER_NAME="${USER:-$(whoami)}"
WS=/cluster/account_a/metaopt
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/gc1
WALL=04:00:00
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
ACCOUNT_EXPECT=account_a      # cluster_a only: cluster_b lacks PATCH_CHUNKWISE

NJOBS=8
SEEDS="0 1 2 3"
CLIP=-15:-2.3026               # the box EVERY 100-ep c100 run in the corpus sits in
MST=1e-4                       # **REVISED AT CLOSEOUT.  WAS 1e-3.**  See below.
                               # ms=1e-3 was chosen because it is the only ms with
                               # existing 100-ep CIFAR-100 data, which bought a
                               # cross-batch offset diagnostic against c100f-node.
                               # It is the WRONG choice and the diagnostic was not
                               # worth it.  At ms=1e-3 the Lion identity with
                               # --weight-decay-meta 0 makes beta reachable over
                               # [-56.91, +43.09], an interval that STRICTLY CONTAINS
                               # the box, with the ceiling reachable from epoch 9.3 --
                               # so ~90% of training runs CLAMPED.  Every R18 anchor
                               # this screen is read against (mm1/pp1/cc1) sits at
                               # ms=1e-4, where rec_lo measured 0.0000 on 24/24 arms,
                               # and fa1 established that this box class CHANGES THE
                               # OPTIMISER rather than merely the instrument (F3
                               # resolved at -0.177).  A clamped-regime D is therefore
                               # NOT THE SAME ESTIMAND as the box-free D it screens
                               # against, and comparability to the anchors is the
                               # entire point of a generalisation screen.  At ms=1e-4
                               # travel is 5.0000 nats: FLOOR unreachable by 3.09 nats,
                               # CEILING reachable only from epoch 92.1 (disclosed).
                               # COST OF THE MOVE, stated: the ms=1e-3 batch-offset
                               # diagnostic is GONE, and no CIFAR-100 run exists at
                               # ms=1e-4 at all, so the LEVEL both arms land at is
                               # unpredicted.  The CONTRAST does not need a predicted
                               # level -- it is a within-batch difference -- but a
                               # REGIME FAILURE is now possible and the scorer aborts
                               # to CLOSE-UNRESOLVED on one rather than reading a
                               # difference between two broken arms.
ALPHA0=1e-3
GAMMA=1
BATCH=100
EPOCHS=100
STEPS_PER_EPOCH=500            # 50,000 train images / batch 100
WRITE_EVERY=500
NET=ResNet18_c100              # <-- paired with DSET below; see the uc5 lesson
DSET=CIFAR100
CHUNK_K=771                    # m(771) = 14,595 = m(nodewise) - 5
M_NODE=14600
M_CHUNK=14595
N_PARAMS=11220132
N_1D=41                        # 1-D tensors on ResNet18_c100
MAX_1D=512                     # largest 1-D tensor; MUST be < CHUNK_K
TAIL_GROUPS=9700               # size-1 nodewise groups = the tail being merged
NUM_CLASSES=100                # the uc5 check: a 100-class head for 100-class labels

# Registered planning constants, all re-derived from the CSV by guard 2 at submit time.
SD_PLAN=0.226                  # **REVISED AT CLOSEOUT.  WAS 0.419.**  Pooled over the
                               # FINE arms only (nodewise sd 0.084 n=3, weightwise 0.309
                               # n=3) -- the arms structurally like the ones this batch
                               # runs.  The old 0.419 pooled five granularities and was
                               # 78% LAYERWISE by weight, an arm gc1 does not run.
SE_D_PLAN=0.160                # = SD_PLAN * sqrt(2/4).  A CIFAR-10-sized D = +0.5 lands
                               # at t = 3.1, so the screen IS powered for the campaign's
                               # own point prior -- which is precisely what the old
                               # fixed 1.0 pp stopping rule denied.
D_GO=1.0                       # RETAINED AS A DESCRIPTIVE SCALE ONLY.  It is NO LONGER
                               # the stopping rule; see guard 2e and the scorer.
D_NULL=0.30                    # the no-amplification band
SLOPE_PP_DECADE=0.491          # count slope; -5 groups => 0.000073 pp of bias
AMP_MIN=3.0                    # refuse if c100/c10 granularity amplification < 3x
SD_CAP=0.60                    # refuse if the measured healthy sd exceeds this
OFFSET_BAR=0.50                # S2's cross-batch bar
ASYM_BAR=0.10                  # S0.4's occupancy-asymmetry bar

REPO="$(cd "$(dirname "$0")/.." && pwd)"
CSVP="$REPO/results/all_runs.csv"

echo "=============================================================="
echo "  gc1 -- DOES THE TAIL PRESCRIPTION SURVIVE A SECOND DATASET?"
echo "    A SCREEN.  $NET / $DSET, ms=$MST, ${EPOCHS}ep, AUGMENT=1"
echo "    node(m=$M_NODE) vs chunk${CHUNK_K}(m=$M_CHUNK), seeds $SEEDS = $NJOBS jobs"
echo "    BETA_CLIP=$CLIP  (EXPECTED TO BIND -- see the header; a bind does not"
echo "                      void S1, an ARM ASYMMETRY > $ASYM_BAR does)"
echo "    S1 (PRIMARY, THE SCREEN): D = chunk${CHUNK_K} - nodewise, se ~ $SE_D_PLAN"
echo "       |D| >= $D_GO  PROCEED to a powered n=15 cell (t >= 3.4)"
echo "       $D_NULL < |D| < $D_GO  SCREEN NEGATIVE -> CLOSE AS UNAFFORDABLE"
echo "       |D| <= $D_NULL  NO AMPLIFICATION -> CLOSE.  **STILL NOT A NULL.**"
echo "    FORBIDDEN TO THIS BATCH: 'D does not generalise to CIFAR-100'."
echo "=============================================================="

# --- GUARD 0 -- the runner and the CSV exist -------------------------------
if [ ! -f "$CSVP" ]; then
  guard_fail "guard 0: no CSV at $CSVP"
else
  echo "guard 0a: CSV present ($(wc -l < "$CSVP") lines)"
fi
if [ ! -f "$RUNNER" ]; then
  guard_fail "guard 0: no runner at $RUNNER"
else
  echo "guard 0b: runner present at $RUNNER"
fi

# --- GUARD 1a-c -- the probe patch is present AND ENABLED (STANDING RULE 19) -
python3 - "$WS" "$REPO" "$0" <<'PYEOF' || guard_fail "guard 1a-c: probe wiring"
import os, re, sys
ws, repo, me = sys.argv[1], sys.argv[2], sys.argv[3]
hf = os.path.join(ws, "MetaOptimize", "codes", "Supervised_tasks", "MetaOptimize",
                  "cifar10", "Optimizers", "HF.py")
if not os.path.exists(hf):
    sys.exit("GUARD FAIL: %s absent" % hf)
src = open(hf).read()
if not re.search(r"os\.environ\.get\(\s*['\"]PROBE5['\"].*?\)\s*==\s*['\"]1['\"]", src):
    sys.exit("GUARD FAIL: HF.py's PROBE5 gate is not the =='1' form this script exports for")
print("guard 1a: HF.py gates the sign-count writer on PROBE5 == '1'")
mine = [l for l in open(me).read().splitlines()
        if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
if len(mine) != 1:
    sys.exit("GUARD FAIL: expected exactly 1 --export line with PROBE_DIR, found %d" % len(mine))
for need in ("PROBE=5", "PROBE5=1", "PROBE5_WRITE_EVERY="):
    if need not in mine[0]:
        sys.exit("GUARD FAIL: export line missing %r -- that is bf8's bug" % need)
print("guard 1b: this script exports PROBE=5 AND PROBE5=1 AND PROBE5_WRITE_EVERY")
tw0 = os.path.join(repo, "bin", "c74_tuned_weightwise_probe.sh")
ref = [l for l in open(tw0).read().splitlines()
       if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
def keys(l):
    return sorted(k.split("=")[0] for k in re.search(r'"ALL,([^"]+)"', l).group(1).split(","))
# tw0's line is the CONFIRMED-FIRING reference: every key it carries must still be
# here (losing one is bf8's bug).  Extra keys are allowed ONLY on the registered pin
# list -- HIER/SCHED/ETA_RATIO/LAM, which guard 1f pins because --export=ALL would
# otherwise inherit them from the submitting shell.
PINNED_EXTRAS = {"HIER", "SCHED", "ETA_RATIO", "LAM"}
missing = set(keys(ref[0])) - set(keys(mine[0]))
if missing:
    sys.exit("GUARD FAIL: export line has LOST key(s) %s from tw0's CONFIRMED-FIRING "
             "line -- that is bf8's bug" % sorted(missing))
extra = set(keys(mine[0])) - set(keys(ref[0]))
if extra - PINNED_EXTRAS:
    sys.exit("GUARD FAIL: export line carries unregistered extra key(s) %s"
             % sorted(extra - PINNED_EXTRAS))
print("guard 1c: export keys are tw0's CONFIRMED-FIRING set plus the registered "
      "pins %s" % sorted(extra))
PYEOF

# --- GUARD 1d -- THE PATCH THIS BATCH NEEDS, AND THE DISPATCH IS EXACT-MATCH -

    """plateau5 at THIS batch's exact 15-axis signature.  Complete, uncollapsed,

      "on the REALISED statistic: CLOSE-CONFIRMED if D is separated from zero at |t| >= "
      "3; PROCEED if D is unresolved but the realised se is under 0.30; "
      "CLOSE-UNAFFORDABLE only if the realised se EXCEEDS 0.30.  The %.1f pp figure is "
      "retained only as the descriptive scale it always was." % d_go)
print("guard 2e: at this sd a POWERED n=15/arm cell would cost 30 jobs; the screen is "
      "8 jobs and may NOT report a null -- see the scorer's forbidden-sentence list.")
if sd_meas > sd_cap:
    sys.exit("GUARD FAIL: measured fine-arm sd %.3f exceeds the %.2f cap; se(D) = %.3f. "
             "Re-cost before submitting." % (sd_meas, sd_cap, se_d))

c2325 = cell('ResNet18', 'CIFAR10', 'chunk2325', ms='1e-4')
if min(len(c777), len(c2325)) < 3:
    sys.exit("GUARD FAIL: the count slope cannot be re-derived (n = %d, %d)"
             % (len(c777), len(c2325)))
dec = math.log10(14421) - math.log10(4851)
slope = (statistics.mean(c2325) - statistics.mean(c777)) / dec
print("guard 2f: count slope re-derived = %+.3f pp/decade (chunk777 %.3f n=%d -> "
      "chunk2325 %.3f n=%d over %.4f decades); registered %+.3f"
      % (slope, statistics.mean(c777), len(c777), statistics.mean(c2325), len(c2325),
         dec, slope_reg))
if abs(slope - slope_reg) > 0.5 * abs(slope_reg):
    sys.exit("GUARD FAIL: the count slope moved from %+.3f to %+.3f; the -5 group bias "
             "must be re-derived before this batch's matched-count claim is made"
             % (slope_reg, slope))

    sys.exit("GUARD FAIL: %d gc1-* runs already in the CSV: %s" % (len(clash), clash[:5]))
print("guard 2g: no gc1-* run name exists in the CSV (%d rows scanned)" % len(rd))

if want != sorted(set(want)):
    sys.exit("GUARD FAIL: the seed list %r has duplicates" % seeds)
if len(want) != 4:
    sys.exit("GUARD FAIL: this batch is registered at 4 seeds, got %d" % len(want))
print("guard 2g: seeds %s, 4 distinct, matching the registered n=4 power calculation" % want)

(
  module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1
  # shellcheck disable=SC1091
  source "$WS/envs/mo/bin/activate"

           "$N_PARAMS" "$N_1D" "$MAX_1D" "$TAIL_GROUPS" "$SLOPE_PP_DECADE" <<'PYEOF' || exit 1
import math, sys, torch
sys.path.insert(0, ".")
from build_network import build_network
from Optimizers.HF import HF
(NET, K, want_node, want_chunk) = (sys.argv[1], int(sys.argv[2]),
                                   int(sys.argv[3]), int(sys.argv[4]))
MST, A0 = float(sys.argv[5]), float(sys.argv[6])
(NCLS, NPAR, N1D, MAX1D, TAIL, SLOPE) = (int(sys.argv[7]), int(sys.argv[8]),
                                         int(sys.argv[9]), int(sys.argv[10]),
                                         int(sys.argv[11]), float(sys.argv[12]))

BASE = {"alg": "SGDm", "weight_decay": 0.1, "momentum_param": 0.99}
META = {"alg": "Lion", "meta_stepsize": MST, "momentum_param": 0.99,
        "Lion_beta2": 0.9, "weight_decay": 0}

class NullWriter:
    def add_scalar(self, *a, **k):
        pass

def build(gran):
    torch.manual_seed(0)
    net = build_network(NET, "cpu")
    return HF(net, stepsize_groups=gran, alpha0=A0, args_base=dict(BASE),
              args_meta=dict(META), gamma=1, writer=NullWriter())

def m_of(opt):
    return int(sum(int(b.numel()) for b in opt.beta))

net = build_network(NET, "cpu")
shapes = [tuple(p.shape) for p in net.parameters()]

# 4a -- THE uc5 CHECK, from the built network, not from the name.
ncls = shapes[-1][0]
if ncls != NCLS:
    sys.exit("GUARD FAIL: %s builds a %d-class head, not %d.  THIS IS THE uc5 BUG."
             % (NET, ncls, NCLS))
npar = sum(int(torch.tensor(list(s)).prod()) for s in shapes)
if npar != NPAR:
    sys.exit("GUARD FAIL: %s has %d parameters, not the %d registered" % (NET, npar, NPAR))
print("guard 4a: %s builds a %d-class head with %d parameters, %d tensors"
      % (NET, ncls, npar, len(shapes)))

# 4b -- m from the ALLOCATED beta, both arms.
got = {}
for name, want in (("nodewise", want_node), ("chunk%d" % K, want_chunk)):
    m = m_of(build(name))
    got[name] = m
    print("guard 4b: %-12s ALLOCATED beta m=%-8d (registered %d)" % (name, m, want))
    if m != want:
        sys.exit("GUARD FAIL: m(%s) is %d, not the %d registered.  K=%d was found by "
                 "search against THIS network; a rebuild has moved it." % (name, m, want, K))

# 4c -- the count match, and its bias at the MEASURED slope.
d = abs(got["chunk%d" % K] - got["nodewise"])
if d > 5:
    sys.exit("GUARD FAIL: the arms are %d groups apart, above the 5 registered" % d)
bias = SLOPE * abs(math.log10(got["nodewise"]) - math.log10(got["chunk%d" % K]))
print("guard 4c: the arms are %d groups apart (m=%d vs %d) = %.3e decades = **%.6f pp** "
      "of count bias at %+.3f pp/decade -- %.0fx below the 0.05 pp resolution floor"
      % (d, got["chunk%d" % K], got["nodewise"],
         abs(math.log10(got["nodewise"]) - math.log10(got["chunk%d" % K])), bias, SLOPE,
         0.05 / bias if bias else float("inf")))

# 4d -- THE STRUCTURAL MATCH TO THE CIFAR-10 CONTRAST.  Every 1-D tensor must get
# exactly ONE chunk, or the comparator SPLITS the tensors whose merging is the
# prescription -- which is precisely why ResNet50 is excluded from this design.
ones = [s for s in shapes if len(s) == 1]
if len(ones) != N1D:
    sys.exit("GUARD FAIL: %s has %d 1-D tensors, not the %d registered" % (NET, len(ones), N1D))
mx = max(s[0] for s in ones)
if mx != MAX1D:
    sys.exit("GUARD FAIL: the largest 1-D tensor is %d, not the %d registered" % (mx, MAX1D))
if mx >= K:
    sys.exit("GUARD FAIL: the largest 1-D tensor (%d) is >= K (%d), so the chunk arm "
             "would SPLIT 1-D tensors and would not be the ResNet18/CIFAR-10 comparator"
             % (mx, K))
chunks_on_1d = sum(math.ceil(s[0] / K) for s in ones)
if chunks_on_1d != len(ones):
    sys.exit("GUARD FAIL: chunk%d puts %d groups on %d 1-D tensors, not one each"
             % (K, chunks_on_1d, len(ones)))
tail = sum(s[0] for s in ones)
if tail != TAIL:
    sys.exit("GUARD FAIL: nodewise has %d size-1 groups on %s, not the %d registered -- "
             "the network changed under the design" % (tail, NET, TAIL))
print("guard 4d: all %d 1-D tensors get EXACTLY ONE chunk (max 1-D numel %d < K=%d); "
      "the merged size-1 tail is %d groups = %.2f%% of m over %.4f%% of the weights "
      "(ResNet18/CIFAR-10: 9,610 = 66.64%% over 0.0860%%)"
      % (len(ones), mx, K, tail, 100.0 * tail / got["nodewise"], 100.0 * tail / npar))
PYEOF
) || guard_fail "guard 4: beta allocation on the LIVE tree"

import csv, sys, os, statistics as st
csvp, net, wall, epochs, K = sys.argv[1:6]
def secs(t):
    q = [int(x) for x in t.split(":")]
    return q[0] * 3600 + q[1] * 60 + q[2]
wall_min = secs(wall) / 60.0
R = list(csv.DictReader(open(csvp)))
def wc(pred):
    out = []
    for r in R:
        if not pred(r):
            continue
        try:
            out.append(float(r["wallclock_min"]))
        except Exception:
            pass
    return out
tgt = wc(lambda r: r["network"] == net and r["epochs_done"] == epochs)
if len(tgt) < 10:
    sys.exit("GUARD FAIL: fewer than 10 %s %s-ep runs to project from" % (net, epochs))
worst = max(tgt)
base = wc(lambda r: r["network"] == "ResNet18" and r["dataset"] == "CIFAR10"
          and r["epochs_done"] == "100" and r["granularity"] == "layerwise")
if len(base) < 20:
    sys.exit("GUARD FAIL: cannot measure the ResNet18 layerwise reference rate")
bm = st.median(base)
ratios = {}
for gran, proxy in (("nodewise", "nodewise"), ("chunk%s" % K, "chunk777")):
    v = wc(lambda r, p=proxy: r["network"] == "ResNet18" and r["dataset"] == "CIFAR10"
           and r["epochs_done"] == "100" and r["granularity"] == p)
    if len(v) < 3:
        sys.exit("GUARD FAIL: no ResNet18 cost proxy for %s (proxy %s)" % (gran, proxy))
    ratios[gran] = st.median(v) / bm
rmax = max(ratios.values())

      % (net, epochs, worst, len(tgt), {k: round(v, 3) for k, v in ratios.items()}))
print("guard 7: BINDING ARM is %s at %.3fx -> projected worst %.0f min against a %.0f "
