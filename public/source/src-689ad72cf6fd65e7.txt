#!/bin/bash
# l262_rule20.sh <csv1> -- CORRECTIONS 262: RULE 20 at FULL coverage for `csv1` (a copy of ~/l260_rule20.sh, sha
# 48f4edc63f2c4c7123e897ab93c1e3b29af5915d5cc8201af61ecbfa4046ac36, with the batch table swapped and the ENV audit
# pointed at l262_envaudit.py).  READ-ONLY.  analysis/argsline_guard.py is UNEDITED (RULE 16), taken from
# ~/stage_csv1, the stage at csv1's registered commit ca435b6b.
set +u
source /etc/profile >/dev/null 2>&1
module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1
source /cluster/account_b/metaopt/envs/mo/bin/activate
export METAOPT_WS=/cluster/account_b/metaopt
WS=$METAOPT_WS; R=$WS/runs; P=$1
G=$HOME/stage_csv1/analysis/argsline_guard.py
echo "RULE20 run utc=$(date -u +%FT%TZ) host=$(hostname) batch=$P guard_sha=$(sha256sum $G | cut -c1-16)"
HF="--expect optimizer=HF --expect alg-base=SGDm --expect momentum-param-base=0.99 --expect weight-decay-base=0.1 --expect alg-meta=Lion --expect momentum-param-meta=0.99 --expect Lion-beta2-meta=0.9 --expect weight-decay-meta=0 --expect dataset=CIFAR100 --expect batch-size=100 --expect max-time=999:00:00 --expect gamma=1"
spec_of() {
  case "$1:$2" in
    csv1:k01|csv1:INERT|csv1:SHADOWLOW|csv1:NAIVELOW|csv1:MUTE) echo scalar ;;
    csv1:HEAD) echo 'sets:1-49,51-53/layer4.1.bn2.weight' ;;
    *) echo UNKNOWN ;;
  esac
}
case $P in csv1) NET=PlainNet18_c100 ;; *) echo "bad batch"; exit 2 ;; esac
echo "  .out files: $(ls $R/$P-*.out 2>/dev/null | wc -l)   with ARGS: $(grep -l -m1 '^ARGS:' $R/$P-*.out 2>/dev/null | wc -l)   squeue: $(squeue -h -u account_b -o '%j %T' | awk -v p="^$P-" '$1 ~ p {print $2}' | sort | uniq -c | tr '\n' ' ')"
OUT=$(python3 $G $R --name $P- --batch-consistency --strict --vary stepsize-groups --vary seed --vary run-name 2>&1); RC=$?
echo "  BATCH-CONSISTENCY --name $P- --vary stepsize-groups --vary seed --vary run-name -> rc=$RC  $(echo "$OUT" | grep -E '^argsline_guard:|^VERDICT|no candidate|NONE carried|ACROSS-RUN|^    --' | tr '\n' ' ')"
n=0; bad=0; noargs=0
for f in $(ls $R/$P-*.out 2>/dev/null); do
  b=$(basename $f .out); rn=${b%-*}; rest=${rn#$P-}; arm=${rest%-s*}; seed=${rest##*-s}
  grep -q -m1 '^ARGS:' "$f" || { noargs=$((noargs+1)); continue; }
  n=$((n+1))
  E="$HF --expect meta-stepsize=1e-3 --expect alpha0=1e-6 --expect NN-name=$NET --expect num-epochs=100 --expect stepsize-groups=$(spec_of $P $arm)"
  O=$(python3 $G "$f" $E --expect seed=$seed --expect save-directory=$R/$P --expect run-name=$rn 2>&1); RCX=$?
  NFG=$(echo "$O" | grep -o 'no repeated flag ([0-9]* flags)' | grep -o '[0-9]*')
  if [ $RCX != 0 ] || [ "$NFG" != 20 ]; then bad=$((bad+1)); echo "  PER-RUN VIOLATION $b rc=$RCX flags=$NFG want 20"; echo "$O" | grep -E '!!!|DESIGN|declared' | sed 's/^/      /'; fi
done
echo "  PER-RUN --expect ($P): $n runs with an ARGS line checked, $bad violations, $noargs .out without ARGS yet"
python3 $HOME/l262_envaudit.py $P $R | sed 's/^/  /'
EARC=${PIPESTATUS[0]}
echo "  ENV AUDIT rc=$EARC"
echo "RULE20 end utc=$(date -u +%FT%TZ)"
