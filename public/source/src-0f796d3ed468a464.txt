def travel(ms):
    return META_STEPS * float(ms)


def reach(ms, a0):
    b0 = math.log(float(a0))
    T = travel(ms)
    du = BETA_HI - b0
    r = T / du
    st = ("FROZEN" if r < FROZEN_RATIO else "STARVED" if r < 1.0
          else "ADEQUATE" if r < 3.0 else "FREE")
    return T, du, r, st


def ceil_spread(ms, a0):
    """Arithmetic ceiling of max_i beta_i - min_i beta_i after META_STEPS."""
    b0 = math.log(float(a0))
    T = travel(ms)
    return min(b0 + T, BETA_HI) - max(b0 - T, BETA_LO)


def rank_avg(xs):
    idx = sorted(range(len(xs)), key=lambda i: xs[i])
    r = [0.0] * len(xs)
    i = 0
    while i < len(idx):
        j = i
        while j + 1 < len(idx) and xs[idx[j + 1]] == xs[idx[i]]:
            j += 1
        avg = (i + j) / 2.0 + 1.0
        for k in range(i, j + 1):
            r[idx[k]] = avg
        i = j + 1
    return r


def pearson(x, y):
    n = len(x)
    if n < 2:
        return float("nan")
    mx, my = sum(x) / n, sum(y) / n
    sxx = sum((a - mx) ** 2 for a in x)
    syy = sum((b - my) ** 2 for b in y)
    if sxx == 0 or syy == 0:
        return float("nan")
    return sum((a - mx) * (b - my) for a, b in zip(x, y)) / math.sqrt(sxx * syy)


def spearman(x, y):
    return pearson(rank_avg(x), rank_avg(y))


def fmean(v):
    return statistics.mean(v) if v else float("nan")

def resolve(explicit, candidates, probe=True):
    if explicit:
        return explicit, "given explicitly"
    for d in candidates:
        if probe:
            if os.path.isdir(d) and glob.glob(os.path.join(d, "probe_*")):
                return d, "DEFAULT %s" % d
        else:
            if os.path.isdir(d) and glob.glob(os.path.join(d, "cru1-*.out")):
                return d, "DEFAULT %s" % d
    return None, "no default contains the expected files"


def read_probe(path, ms):
    """One run.  Returns dict with per-record spreads, terminal state, gates."""
    recs = []
    g1_bad = 0
    g2_bad = 0
    n_beta_bad = 0
    beta0 = None
    with open(path, errors="ignore") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            try:
                j = json.loads(line)
            except ValueError:
                continue
            b = j.get("beta")
            if not isinstance(b, list) or not b:
                continue
            b = [float(v) for v in b]
            step = int(j.get("step", -1))
            if beta0 is None:
                beta0 = list(b)
            S = max(b) - min(b)
            tmin, tmax = j.get("beta_true_min"), j.get("beta_true_max")
            if tmin is not None and tmax is not None:
                if abs(S - (float(tmax) - float(tmin))) > TOL_S:
                    g1_bad += 1
            nb = j.get("n_beta")
            if nb is not None and int(nb) != len(b):
                n_beta_bad += 1
            bound = float(ms) * step + TOL_B
            if len(b) == len(beta0):
                if any(abs(bi - b0i) > bound for bi, b0i in zip(b, beta0)):
                    g2_bad += 1
            recs.append((step, b, S))
    if not recs:
        return None
    step_T, b_T, S_T = recs[-1]
    disp = max(abs(bi - b0i) for bi, b0i in zip(b_T, beta0)) if len(b_T) == len(beta0) else float("nan")
    return {"n": len(recs), "step_T": step_T, "beta0": beta0, "b_T": b_T,
            "S_T": S_T, "min_T": min(b_T), "max_T": max(b_T),
            "S_mean": fmean([r[2] for r in recs]),
            "S_at": {s: next((r[2] for r in recs if r[0] == s), None) for s in CHECK_STEPS},
            "traj": [(r[0], min(r[1]), max(r[1]), r[2]) for r in recs],
            "disp_T": disp, "g1_bad": g1_bad, "g2_bad": g2_bad,
            "n_beta_bad": n_beta_bad, "k": len(b_T)}


def read_probe_index(dirpath):
    p = os.path.join(dirpath, "probe_index.json")
    if not os.path.isfile(p):
        return None
    try:
        return json.load(open(p))
    except Exception:
        return None


def read_csv_gaps(csvpath):
    """-> {(ms,a0): {"sc":[p5...], "lay":[p5...], "sc_tr":[...], "lay_tr":[...]}}, n_rows"""
    cells = collections.defaultdict(lambda: {"sc": [], "lay": [], "sc_tr": [], "lay_tr": []})
    n = 0
    if not os.path.isfile(csvpath):
        return cells, -1
    for r in csv.DictReader(open(csvpath)):
        if not r.get("run", "").startswith("cru1-"):
            continue
        g = r.get("granularity", "")
        arm = "sc" if g == "scalar" else "lay" if g == "layerwise" else None
        if arm is None:
            continue
        try:
            p5 = float(r["plateau5"])
        except (KeyError, ValueError):
            continue
        n += 1
        key = (r["meta_stepsize"], r["alpha0"])
        cells[key][arm].append(p5)
        try:
            cells[key][arm + "_tr"].append(float(r["final_train"]))
        except (KeyError, ValueError):
            pass
    return cells, n


def read_out_tail5(outroot):
    """-> {(arm,ms,a0,seed): tail5 train mean} using cY1's EPTR_RE."""
    out = {}
    if not outroot:
        return out
    for f in glob.glob(os.path.join(outroot, "cru1-*.out")):
        m = OUT_RE.match(os.path.basename(f))
        if not m:
            continue
        d = {}
        for line in open(f, errors="ignore"):
            mm = EPTR_RE.search(line)
            if mm:
                d[int(mm.group(1))] = float(mm.group(2))
        v = [d[e] for e in range(EPOCHS - 5, EPOCHS) if e in d]
        if len(v) == 5:
            out[(m.group(1), m.group(2), m.group(3), int(m.group(4)))] = sum(v) / 5.0
    return out

# =============================================================================
def report(probe_root, out_root, csvpath):
    print("=" * 78)
    print("cY3_cru1_spread.py -- ACCOUNT F's PREMISE, TESTED ON cru1's OWN PROBE.")
    print("NO BRANCH FOR cru1.  cru1's verdict is whatever cY1_cru1_score.py returns.")
    print("probe root : %s" % probe_root)
    print("out root   : %s" % (out_root or "NONE (train tail-5 = SKIP)"))
    print("csv        : %s" % csvpath)
    print("=" * 78)

    # ---- census (P1) --------------------------------------------------------
    dirs = sorted(d for d in glob.glob(os.path.join(probe_root, "probe_*"))
                  if os.path.isfile(os.path.join(d, "probe.jsonl")))
    runs, unclassified = {}, []
    g0_bad = []
    for d in dirs:
        m = RUN_RE.match(os.path.basename(d))
        if not m:
            n = sum(1 for _ in open(os.path.join(d, "probe.jsonl"), errors="ignore"))
            unclassified.append((os.path.basename(d), n))
            continue
        arm, ms, a0, seed = m.group(1), m.group(2), m.group(3), int(m.group(4))
        pi = read_probe_index(d)
        if pi is not None:
            pn = tuple(int(v) for v in pi.get("param_numels", ()))
            if pn != TENSOR_NUMEL:
                g0_bad.append(os.path.basename(d))
        r = read_probe(os.path.join(d, "probe.jsonl"), ms)
        if r is None:
            unclassified.append((os.path.basename(d), 0))
            continue
        runs[(arm, ms, a0, seed)] = r

    print("\n[CENSUS]  probe dirs with probe.jsonl: %d   parseable+nonempty: %d   "
          "UNCLASSIFIED: %d" % (len(dirs), len(runs), len(unclassified)))
    for name, n in unclassified:
        print("    UNCLASSIFIED %s (%d records)" % (name, n))
    lay = {k: v for k, v in runs.items() if k[0] == "lay"}
    sc = {k: v for k, v in runs.items() if k[0] == "sc"}
    print("  layerwise runs %d   scalar runs %d" % (len(lay), len(sc)))

    # ---- gates --------------------------------------------------------------
    print("\n[G0] probe_index.json param_numels == TENSOR_NUMEL (naming alignment):"
          " %s" % ("PASS (all %d indexed runs)" % sum(
              1 for d in dirs if read_probe_index(d) is not None)
              if not g0_bad else "FAIL in %d runs: %s" % (len(g0_bad), g0_bad[:5])))
    k_bad = [k for k, v in lay.items() if v["k"] != 62] + [k for k, v in sc.items() if v["k"] != 1]
    print("[G0b] layerwise beta list has 62 entries, scalar has 1: %s"
          % ("PASS" if not k_bad else "FAIL %s" % k_bad[:5]))
    g1 = sum(v["g1_bad"] for v in runs.values())
    nb = sum(v["n_beta_bad"] for v in runs.values())
    print("[G1] S(beta list) == beta_true_max - beta_true_min within %g: %s "
          "(mismatching records %d; n_beta mismatches %d)"
          % (TOL_S, "PASS" if g1 == 0 and nb == 0 else "FAIL", g1, nb))
    g2 = sum(v["g2_bad"] for v in runs.values())
    print("[G2] |beta_i(t)-beta_i(0)| <= ms*step + %g for every coordinate, both "
          "arms: %s (violating records %d of %d)"
          % (TOL_B, "PASS" if g2 == 0 else "FAIL -- EVERY TRAVEL NUMBER IN 174 IS SUSPECT",
             g2, sum(v["n"] for v in runs.values())))
    steps_T = collections.Counter(v["step_T"] for v in runs.values())
    print("[G3] terminal probe step per run: %s" % dict(steps_T))

    # ---- Q1 per-run table ---------------------------------------------------
    print("\n[Q1] PER-RUN TERMINAL SPREAD (layerwise) AND SCALAR POSITION IN THE ENVELOPE")
    print("%-6s %-6s %-4s %-8s %8s %8s %8s %8s %8s | %9s %7s %8s %7s"
          % ("ms", "a0", "seed", "reach", "S_T", "S_mean", "min_T", "max_T",
             "exp(S_T)", "beta_sc_T", "POS_T", "FRAC<sc", "POSmean"))
    q1 = {}
    for key in sorted(lay, key=lambda k: (k[2], float(k[1]), k[3])):
        arm, ms, a0, seed = key
        v = lay[key]
        T, du, r, st = reach(ms, a0)
        s = sc.get(("sc", ms, a0, seed))
        if s is not None:
            bsc = s["b_T"][0]
            pos = (bsc - v["min_T"]) / (v["max_T"] - v["min_T"]) if v["max_T"] > v["min_T"] else float("nan")
            fb = sum(1 for b in v["b_T"] if b < bsc) / float(len(v["b_T"]))
            # time-mean position: align by step
            posm = []
            sc_by_step = {}
            for step, mn, mx, S in s["traj"]:
                sc_by_step[step] = mn  # scalar: min == max == its one beta
            for step, mn, mx, S in v["traj"]:
                if step in sc_by_step and mx > mn:
                    posm.append((sc_by_step[step] - mn) / (mx - mn))
            posmean = fmean(posm)
        else:
            bsc = pos = fb = posmean = float("nan")
        q1[key] = (v["S_T"], bsc, pos, fb, posmean)
        print("%-6s %-6s %-4d %-8s %8.4f %8.4f %8.4f %8.4f %8.2f | %9.4f %7.3f %8.3f %7.3f"
              % (ms, a0, seed, st, v["S_T"], v["S_mean"], v["min_T"], v["max_T"],
                 math.exp(v["S_T"]), bsc, pos, fb, posmean))

    # ---- Q1 trajectory by cell ---------------------------------------------
    print("\n[Q1] SPREAD TRAJECTORY, CELL MEAN OVER SEEDS, AT CHECKPOINT STEPS")
    print("%-6s %-6s %-8s " % ("ms", "a0", "reach") + " ".join("%8d" % s for s in CHECK_STEPS))
    cells = collections.defaultdict(list)
    for key, v in lay.items():
        cells[(key[1], key[2])].append(v)
    for c in sorted(cells, key=lambda c: (c[1], float(c[0]))):
        T, du, r, st = reach(*c)
        vals = []
        for s in CHECK_STEPS:
            xs = [v["S_at"][s] for v in cells[c] if v["S_at"][s] is not None]
            vals.append("%8.4f" % fmean(xs) if xs else "%8s" % "-")
        print("%-6s %-6s %-8s " % (c[0], c[1], st) + " ".join(vals))

    # ---- Q2 -----------------------------------------------------------------
    print("\n[Q2] THE PREMISE TEST AT THE FROZEN RUNGS.  FRAC_CEIL = S_T / CEIL,")
    print("     CEIL = min(beta0+TRAVEL, BETA_HI) - max(beta0-TRAVEL, BETA_LO).")
    print("     BAR: < %.2f NEAR-ZERO | >= %.2f SUBSTANTIAL | else INTERMEDIATE"
          % (NEAR_ZERO_MAX, SUBSTANTIAL_MIN))
    print("%-6s %-6s %-8s %3s %7s %7s %8s %9s %8s %9s %9s  %s"
          % ("ms", "a0", "reach", "n", "TRAVEL", "CEIL", "S_T", "FRAC_CEIL",
             "exp(S_T)", "disp/T", "POS_T", "token"))
    tokens = {}
    spread_cell = {}
    for c in sorted(cells, key=lambda c: (c[1], float(c[0]))):
        T, du, r, st = reach(*c)
        C = ceil_spread(*c)
        S = [v["S_T"] for v in cells[c]]
        frac = fmean(S) / C
        spread_cell[c] = fmean(S)
        disp = fmean([v["disp_T"] / T for v in cells[c]])
        posT = fmean([q1[("lay", c[0], c[1], k[3])][2] for k in lay if (k[1], k[2]) == c
                      and not math.isnan(q1[("lay", c[0], c[1], k[3])][2])])
        tok = ("NEAR-ZERO" if frac < NEAR_ZERO_MAX else
               "SUBSTANTIAL" if frac >= SUBSTANTIAL_MIN else "INTERMEDIATE")
        tokens[c] = tok
        flag = "  <-- F rung" if c in F_RUNGS else ("  (frozen, not an F rung)" if st == "FROZEN" else "")
        print("%-6s %-6s %-8s %3d %7.2f %7.3f %8.4f %9.4f %8.2f %9.3f %9.3f  %s%s"
              % (c[0], c[1], st, len(S), T, C, fmean(S), frac, math.exp(fmean(S)),
                 disp, posT, tok if st == "FROZEN" else "-", flag))
    ft = [tokens.get(c) for c in F_RUNGS]
    if all(t == "SUBSTANTIAL" for t in ft):
        premise = "PREMISE-FALSE"
    elif all(t == "NEAR-ZERO" for t in ft):
        premise = "PREMISE-STANDS"
    elif any(t is None for t in ft):
        premise = "PREMISE-UNTESTABLE (an F rung has no layerwise probe)"
    else:
        premise = "PREMISE-MIXED"
    print("\nPREMISE VERDICT over F's registered rungs %s: %s   [%s]"
          % (list(F_RUNGS), premise, " | ".join("%s/%s:%s" % (c[0], c[1], t) for c, t in zip(F_RUNGS, ft))))

    # ---- Q3 -----------------------------------------------------------------
    print("\n[Q3] DOSE-RESPONSE.  gap = mean plateau5(lay) - mean plateau5(sc) from the CSV.")
    gaps, ncsv = read_csv_gaps(csvpath)
    if ncsv < 0:
        print("  CSV not found -> Q3 SKIP")
    else:
        print("  cru1 CSV rows read (scalar+layerwise, plateau5 present): %d" % ncsv)
    print("%-6s %-6s %-8s %7s %8s %8s %9s %9s %8s %7s %7s"
          % ("ms", "a0", "reach", "TRAVEL", "sc p5", "lay p5", "gap", "gap/SE", "Spread", "SAT", "n_sc/lay"))
    rows = []
    missing = 0
    for c in sorted(set(list(cells) + list(gaps)), key=lambda c: (c[1], float(c[0]))):
        T, du, r, st = reach(*c)
        g = gaps.get(c)
        if not g or not g["sc"] or not g["lay"] or c not in spread_cell:
            missing += 1
            print("%-6s %-6s %-8s %7.2f %8s %8s %9s   MISSING (csv sc %d lay %d, probe %s)"
                  % (c[0], c[1], st, T, "-", "-", "-",
                     len(g["sc"]) if g else 0, len(g["lay"]) if g else 0, c in spread_cell))
            continue
        gap = fmean(g["lay"]) - fmean(g["sc"])
        rows.append((c, T, gap, spread_cell[c]))
        print("%-6s %-6s %-8s %7.2f %8.4f %8.4f %9.4f %9.2f %8.4f %7.3f %3d/%-3d"
              % (c[0], c[1], st, T, fmean(g["sc"]), fmean(g["lay"]), gap, gap / SE_GAP,
                 spread_cell[c], spread_cell[c] / CLIP_WIDTH, len(g["sc"]), len(g["lay"])))
    print("  cells dropped from Q3 as MISSING: %d" % missing)
    for a0 in sorted(set(c[1] for c, _, _, _ in rows), key=float):
        sub = [x for x in rows if x[0][1] == a0]
        if len(sub) < 3:
            print("  ladder a0=%s: %d cells, too few for a rank correlation" % (a0, len(sub)))
            continue
        gv = [x[2] for x in sub]
        print("  ladder a0=%s (%d rungs): rho(gap,Spread)=%+.3f  rho(gap,TRAVEL)=%+.3f  "
              "[within-ladder: Spread can at best TIE, see header]"
              % (a0, len(sub), spearman(gv, [x[3] for x in sub]), spearman(gv, [x[1] for x in sub])))
    if len(rows) >= 4:
        gv = [x[2] for x in rows]
        rs = spearman(gv, [x[3] for x in rows])
        rt = spearman(gv, [x[1] for x in rows])
        ps = pearson(gv, [x[3] for x in rows])
        pt = pearson(gv, [math.log10(x[1]) for x in rows])
        tok = ("SPREAD-TRACKS-BETTER" if rs - rt > RHO_MARGIN else
               "TRAVEL-TRACKS-BETTER" if rt - rs > RHO_MARGIN else "TIE")
        print("  POOLED over %d cells (THE REGISTERED COMPARISON): rho(gap,Spread)=%+.3f  "
              "rho(gap,TRAVEL)=%+.3f  -> %s   [Pearson r: Spread %+.3f, log10 TRAVEL %+.3f]"
              % (len(rows), rs, rt, tok, ps, pt))
    else:
        print("  POOLED: fewer than 4 cells -> SKIP")

    # ---- Q4 -----------------------------------------------------------------
    print("\n[Q4] WHICH LAYERS SIT AT THE TOP AND BOTTOM OF THE SPREAD AT THE FROZEN RUNGS")
    tail5 = read_out_tail5(out_root)
    print("  .out files with a full tail-5 train series: %d" % len(tail5))
    frozen_cells = [c for c in sorted(cells, key=lambda c: (c[1], float(c[0]))) if reach(*c)[3] == "FROZEN"]
    for c in frozen_cells:
        g = gaps.get(c) or {"sc": [], "lay": [], "sc_tr": [], "lay_tr": []}
        print("\n  -- cell ms=%s a0=%s %s --" % (c[0], c[1], "(F rung)" if c in F_RUNGS else "(frozen, not an F rung)"))
        print("     TRAIN: CSV final_train  sc %s  lay %s ;  tail-5 from .out  sc %s  lay %s"
              % ("%.2f" % fmean(g["sc_tr"]) if g["sc_tr"] else "-",
                 "%.2f" % fmean(g["lay_tr"]) if g["lay_tr"] else "-",
                 "%.2f" % fmean([tail5[k] for k in tail5 if k[0] == "sc" and k[1:3] == c]) if any(k[0] == "sc" and k[1:3] == c for k in tail5) else "SKIP",
                 "%.2f" % fmean([tail5[k] for k in tail5 if k[0] == "lay" and k[1:3] == c]) if any(k[0] == "lay" and k[1:3] == c for k in tail5) else "SKIP"))
        print("     TEST plateau5:  sc %s  lay %s"
              % ("%.4f" % fmean(g["sc"]) if g["sc"] else "-", "%.4f" % fmean(g["lay"]) if g["lay"] else "-"))
        per_coord = collections.defaultdict(list)
        for key in sorted(k for k in lay if (k[1], k[2]) == c):
            v = lay[key]
            b = v["b_T"]
            order = sorted(range(len(b)), key=lambda i: b[i])
            names = TENSOR_NAMES if len(b) == len(TENSOR_NAMES) else tuple("i%d" % i for i in range(len(b)))
            for i, bi in enumerate(b):
                per_coord[i].append(bi)
            print("     seed %d  S_T %.4f  bottom-3: %s | top-3: %s"
                  % (key[3], v["S_T"],
                     ", ".join("%s(%.3f)" % (names[i], b[i]) for i in order[:3]),
                     ", ".join("%s(%.3f)" % (names[i], b[i]) for i in order[-3:][::-1])))
        if per_coord:
            k = len(per_coord)
            names = TENSOR_NAMES if k == len(TENSOR_NAMES) else tuple("i%d" % i for i in range(k))
            cm = [fmean(per_coord[i]) for i in range(k)]
            order = sorted(range(k), key=lambda i: cm[i])
            print("     CELL-MEAN beta per coordinate: bottom-3 %s | top-3 %s"
                  % (", ".join("%s(%.3f)" % (names[i], cm[i]) for i in order[:3]),
                     ", ".join("%s(%.3f)" % (names[i], cm[i]) for i in order[-3:][::-1])))
            n_below = sum(1 for x in cm if x < math.log(float(c[1])))
            print("     coordinates whose cell-mean terminal beta is BELOW beta0=%.4f: %d of %d; "
                  "ABOVE: %d" % (math.log(float(c[1])), n_below, k, k - n_below))

    print("\n[HOW TO READ THIS, REGISTERED IN ADVANCE]")
    print("  * PREMISE-FALSE means: at the rungs F was registered on, the layerwise arm")
    print("    realises a between-layer spread that is a substantial fraction of the")
    print("    largest spread arithmetic allows, so 'the granularity variable has almost")
    print("    nothing to act on' is not true there, and the sentence 'the gap is NOT a")
    print("    step-size-adaptation phenomenon' does NOT follow from F-FAILS.  F-FAILS")
    print("    itself (the measured gaps against the 1.0021 bar) is untouched.")
    print("  * PREMISE-STANDS means the opposite and 174.8 is left exactly as written.")
    print("  * Q3 is descriptive with 10 cells; SPREAD-TRACKS-BETTER is consistent with,")
    print("    not proof of, spread being the operative variable.")
    print("  * Nothing here moves cY1's FINAL line.")
    return 0

# selftest -- touches NO probe file and NO cru1 CSV row
# =============================================================================
def selftest():
    P = F = 0

    def chk(ok, label, detail=""):
        nonlocal P, F
        if ok:
            P += 1
            print("  PASS  %s %s" % (label, detail))
        else:
            F += 1
            print("  FAIL  %s %s" % (label, detail))

    print("cY3_cru1_spread.py --selftest")
    chk(META_STEPS == 50000, "S1 meta-step count", "-> %d" % META_STEPS)
    cy1 = os.path.join(HERE, "cY1_cru1_score.py")
    if os.path.isfile(cy1):
        import importlib.util
        spec = importlib.util.spec_from_file_location("cy1", cy1)
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        chk(dict(m.LADDER) == LADDER and m.META_STEPS == META_STEPS
            and (m.BETA_LO, m.BETA_HI) == (BETA_LO, BETA_HI)
            and abs(m.SIGMA_W - SIGMA_W) < 1e-9 and abs(m.SE_GAP - SE_GAP) < 1e-9
            and abs(m.BAR_GAP - BAR_GAP) < 1e-9 and m.FROZEN_RATIO == FROZEN_RATIO,
            "S2 design constants and SIGMA_W/SE_GAP/BAR_GAP match the REGISTERED cY1",
            "(SE_GAP %.6f BAR %.6f)" % (SE_GAP, BAR_GAP))
        chk(m.EPTR_RE.pattern == EPTR_RE.pattern, "S2b EPTR_RE identical to cY1's")
        src = open(cy1).read()
        chk("GRANULARITY-INOPERATIVE-AT-FROZEN" in src and "almost nothing to act on" in src,
            "S3 the premise this file tests is present verbatim in cY1")
    else:
        chk(False, "S2 cY1_cru1_score.py not found next to this file", cy1)
    chk(len(TENSOR_NAMES) == 62 and len(TENSOR_NUMEL) == 62 and sum(TENSOR_NUMEL) == 11220132,
        "S4 62 names, 62 numels, 11,220,132 params", "")
    fr = [(ms, a0) for a0 in LADDER for ms in LADDER[a0] if reach(ms, a0)[3] == "FROZEN"]
    chk(set(fr) == {("1e-5", "1e-3"), ("3e-5", "1e-3"), ("1e-4", "1e-6")},
        "S5 the FROZEN rungs reproduce", "-> %s" % fr)
    chk(all(c in fr for c in F_RUNGS), "S5b F's rungs are FROZEN rungs")
    chk(abs(ceil_spread("1e-5", "1e-3") - 1.0) < 1e-9 and abs(ceil_spread("3e-5", "1e-3") - 3.0) < 1e-9,
        "S6 CEIL at F's rungs = 2*TRAVEL = 1.0 and 3.0 (clip does not bind)",
        "-> %.4f %.4f" % (ceil_spread("1e-5", "1e-3"), ceil_spread("3e-5", "1e-3")))
    chk(abs(ceil_spread("1e-4", "1e-6") - (math.log(1e-6) + 5.0 - BETA_LO)) < 1e-9,
        "S6b CEIL at (1e-4,1e-6) is clip-bound", "-> %.4f" % ceil_spread("1e-4", "1e-6"))
    chk(abs(spearman([1, 2, 3, 4], [10, 20, 30, 40]) - 1.0) < 1e-12
        and abs(spearman([1, 2, 3, 4], [40, 30, 20, 10]) + 1.0) < 1e-12
        and abs(spearman([1, 2, 3, 4, 5], [1, 3, 2, 5, 4]) - 0.8) < 1e-12,
        "S7 Spearman on known vectors", "")
    chk(rank_avg([3, 1, 3]) == [2.5, 1.0, 2.5], "S7b average ranks on ties")
    # a synthetic run exercising G1/G2 and the spread
    import tempfile
    d = tempfile.mkdtemp()
    p = os.path.join(d, "probe.jsonl")
    with open(p, "w") as fh:
        b0 = [-6.9] * 62
        for step in (0, 100, 200):
            b = [b0[i] + (1e-5 * step if i % 2 else -1e-5 * step) for i in range(62)]
            fh.write(json.dumps({"step": step, "beta": b, "beta_true_min": min(b),
                                 "beta_true_max": max(b), "n_beta": 62}) + "\n")
    r = read_probe(p, "1e-5")
    chk(r is not None and r["n"] == 3 and abs(r["S_T"] - 0.004) < 1e-9 and r["g1_bad"] == 0
        and r["g2_bad"] == 0 and abs(r["disp_T"] - 0.002) < 1e-9,
        "S8 synthetic run: spread 0.004 at step 200, gates clean", "")
    r2 = read_probe(p, "1e-6")   # too small an ms -> G2 must fire at step 200
    # (step 100 moves 1e-3 against a bound of 1e-4 + TOL_B = 1.1e-3: inside;
    #  step 200 moves 2e-3 against 1.2e-3: violates.  Exactly one record.)
    chk(r2 is not None and r2["g2_bad"] == 1, "S8b G2 fires when ms is understated", "-> %d" % r2["g2_bad"])
    print("SELFTEST: %d PASS, %d FAIL" % (P, F))
    return F


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--probe-root", default=None, help="OPTIONAL override")
    ap.add_argument("--out-root", default=None, help="OPTIONAL override (dir holding cru1-*.out)")
    ap.add_argument("--csv", default=DEFAULT_CSV, help="OPTIONAL override")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    if a.selftest:
        return 1 if selftest() else 0
    root, how = resolve(a.probe_root, DEFAULT_PROBE_ROOTS, probe=True)
    if not root:
        print("no probe root found (%s)" % how)
        return 1
    oroot, _ = resolve(a.out_root, DEFAULT_OUT_ROOTS, probe=False)
    return report(root, oroot, a.csv)
