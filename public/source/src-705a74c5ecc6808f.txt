#
# =============================================================================
# THE QUESTION
# =============================================================================
# `in489g1` (CORRECTIONS 144) measured a 47.4 pp CLIFF between two ADJACENT
# rungs of the granularity ladder at 489 classes: m = 6 is dead (~1.4 pp, at
# scalar's ~1.0 pp) and m = 62 reaches ~48.9 pp.  Nobody has measured anything
# in between.  The transition therefore sits somewhere in the OPEN INTERVAL
# (6, 62) and this batch's ONE job is to bracket it.
#
# The comparison target is `cbl1` (CORRECTIONS 143), the CIFAR-100 count ladder
# over BALANCED partitions of the same 62 parameter tensors.  Its interior arms
# at m = 8, 16, 32 are re-used here BYTE-FOR-BYTE (selftest asserts the spec
# strings against the CSV's own `granularity` column), so the two ladders sit on
# a SHARED GRID and the cross-dataset statement is a direct comparison rather
# than an interpolation.
#
# =============================================================================
# THE DESIGN.  7 arms x 2 seeds = 14 jobs, ONE submission, so BATCH -- the unit
# of replication in this corpus -- is held constant and every contrast is WITHIN
# this batch.  BOTH ANCHORS ARE IN BATCH; nothing is spliced from in489g1.
#
#   label  --stepsize-groups                        m     role
#   g01    scalar                                    1    ANCHOR, capture zero
#   g08    [8,8,8,8,8,8,7,7]                         8    balanced; cbl1 grid
#   g16    [4 x14,3,3]                              16    balanced; cbl1 grid
#   g32    [2 x30,1,1]                              32    balanced; cbl1 grid
#   g45    [2 x17,1 x28]                            45    balanced; bisects
#                                                          (32,62) in log space
#   g62    layerwise                                62    ANCHOR, capture one
#   p08    [45,3,3,3,2,2,2,2]                        8    POSITION CONTROL at
#                                                          m matched to g08
#   seed in {0, 1}
#
# WHY THESE RUNGS.  The interval to be searched is (6, 62), i.e. 3.37 log2
# units wide.  m = 8, 16, 32 are the cbl1 grid and are kept for the shared-grid
# comparison; that grid leaves (32, 62) as a 0.95-log2 gap, which is exactly
# where BOTH pre-registered models below put the crossing, so m = 45 is added to
# bisect it (log2 45/32 = 0.49, log2 62/45 = 0.46).  m = 4 and below are NOT
# run: in489g1 already measured CAPTURE ~= 0.008 at m = 6, so a rung below that
# buys no information about a crossing at 0.5.
#
# WHY n = 2 AND NOT n = 3, STATED AS A CHOICE AND NOT AN ACCIDENT.  Seed is
# statistically null in this corpus -- selftest re-derives F(11,1976) = 0.152,
# p = 0.999 for SEED as a factor after removing cell means over 310 config
# cells / 1,988 rows -- while BATCH is not, which is why every contrast here is
# in-batch and why the seed budget is the one that gets cut.  in489g1's own
# measured within-arm SD of the primary statistic is 0.1254 pp (pooled, df 8)
# against an anchor gap of 48.15 pp, so at n = 2 the SE of a CAPTURE value is
# 0.0026 -- 192 SE below the 0.5 threshold this batch must bracket and 19 SE
# below the smallest bar used anywhere in this file.  Spending the same
# ~350 GPU-h on 5 arms x 3 seeds instead would have bought a statistic that is
# already 200x over-powered while DELETING two rungs of the curve that is the
# entire question.  Arms beat seeds here; that is the trade and it is on record.
#
# COST, SIZED FROM THE MEASURED RECORD AND NOT FROM A GUESS.  in489g1's twelve
# runs took 21:41:30 to 1-07:48:52 (sacct), ~299 GPU-h in total, and its slowest
# runs were slow from PARTITION CONTENTION and not from their arm: its
# blockwise arm ran at 675 img/s on an idle L4 (identical to layerwise) and at
# 436 img/s under load.  14 runs at the same per-run cost is ~350 GPU-h.  Wall
# 34:00:00 with --max-time 32:00:00 is carried over unchanged: it produced 0
# TIMEOUT and 0 FAILED across all 12 in489g1 runs, truncating the three
# contended ones at 80-81 epochs, which the common horizon below absorbs.
#
# =============================================================================
# THE PRIMARY STATISTIC, FROZEN
# =============================================================================
#   W5(run)   = mean TEST accuracy over the last 5 SCORED epochs, where "scored"
#               means epochs 0 .. ESTAR-1 and ESTAR is the COMMON HORIZON.
#   W5T(run)  = the same on TRAIN accuracy.  ALWAYS REPORTED ALONGSIDE.
#   M(arm)    = mean W5 over that arm's seeds.
#   The CSV `plateau` column is BANNED as primary and is never read by this
#   file.  `best_test` is never read.  On ImageNet-489, W5 at ESTAR is called
#   "window5@ESTAR" and NEVER a plateau unless ESTAR >= 60.
#
#   CAPTURE(arm) = (M(arm) - M(g01)) / (M(g62) - M(g01)),  BOTH ANCHORS IN BATCH.
#   TRAINCAP(arm) = the same on W5T.
#   CAPTURE is 0 at g01 and 1 at g62 BY CONSTRUCTION.  No optimum in m is named
#   by this file and none may be read out of it.
#
#   M50 = the m at which the piecewise interpolant of CAPTURE, LINEAR IN log2 m
#   over the COUNT LADDER (g01, g08, g16, g32, g45, g62), FIRST crosses 0.5.
#   It is reported as a BRACKET of adjacent rungs plus the interpolated point.
#   The BRACKET is the result; the point estimate is a convenience inside it and
#   may never be quoted without its bracket.
#
# =============================================================================
# THE PRE-REGISTERED PREDICTION.  Stated before submission, on record, and
# NEVER GATING -- a missed prediction voids nothing, it is just reported.
# =============================================================================
# Two families, both pinned to CAPTURE(1) = 0 and CAPTURE(62) = 1 by
# construction, both calibrated on the ONE interior point that exists:
# in489g1's CAPTURE(m=6) = 0.008487 (cross-batch, DESCRIPTIVE ONLY, re-derived
# by selftest from the CSV).
#
#   MODEL A, power in log m:  CAPTURE(m) = (log m / log 62)^p.
#     CIFAR-100 (cbl1) best-fit p = 0.902, i.e. essentially the LOG law, and its
#     own crossing is M50 = 9.30 in the bracket (8, 16).
#     ImageNet-489 calibrated on CAPTURE(6) = 0.008487 gives p = 5.716, a curve
#     six times steeper in log m, and M50 = 38.70  ->  bracket (32, 45).
#   MODEL B, linear in log2 m from (6, 0.008487) to (62, 1):  M50 = 19.10
#     ->  bracket (16, 32).   (The geometric mean of the interval, 19.29, is
#     the same answer to two figures and is the naive default.)
#
# REGISTERED PREDICTION: **CAPTURE crosses 0.5 at M50 in [16, 62]**, point
# estimate 38.70 under Model A, with Model B's 19.10 recorded as the live rival.
# Both models say the crossing is ABOVE m = 16; that shared consequence is the
# part of the prediction that can cheaply fail.
#
# =============================================================================
# WHAT WOULD REFUTE A SMOOTH MONOTONE TRANSITION.  Five deaths, frozen.
# =============================================================================
# R1 NONMONOTONE     max adjacent DROP in CAPTURE along the count ladder
#                    > 0.05.  At the measured capture SE of 0.0026 that is 19
#                    SE, so it is a reversal and not a wobble.  If it fires,
#                    "CAPTURE rises smoothly with m" is REFUTED and M50 is not
#                    reported as a crossing.
# R2 POSITION-DOMINATES  |CAPTURE(p08) - CAPTURE(g08)| > 0.10 at IDENTICAL
#                    m = 8.  This is the in-batch rival that can kill this
#                    file's own registered statistic, and it exists because
#                    cbl1's mirror control killed cbl1's (CORRECTIONS 143:
#                    "cut POSITION is [the operative variable]").  If it fires,
#                    m is NOT the operative variable at 489 classes, the count
#                    statistic is **DISQUALIFIED AND IS NOT REPLACED BY A
#                    BETTER-FITTING ONE**, and no M50 may be reported as a
#                    property of m.  The bracket may still be reported as a
#                    property of THIS partition policy, labelled as such.
# R3 CLIFF           the adjacent RISE in CAPTURE inside the crossing bracket
#                    > 0.60.  Then the ladder is a step and not a transition:
#                    the bracket stands, the interpolated M50 does NOT, and the
#                    result is reported as a CLIFF.
# R4 LOW-END-NOT-FLAT  CAPTURE(g08) > 0.25.  The premise that in489g1's
#                    m = 6 point (0.008) extends to a flat low end fails, the
#                    crossing is below the registered interval, and the
#                    prediction above is a directional MISS.
# R5 TRAIN-TEST DIVERGENCE  max over rungs of |TRAINCAP - CAPTURE| > 0.15.
#                    The coarse-arm failure has been TRAIN-SIDE on all three
#                    datasets measured so far (CIFAR-100 train 22.8 vs 99.0;
#                    Tiny-ImageNet 9.76 vs 88.67; ImageNet-489 1.04 vs 44.5).
#                    If the two ladders come apart here, that is a FINDING and
#                    is reported as one, not smoothed over.
#
# =============================================================================
# THE GATES
# =============================================================================
# G0 PROVENANCE, ABSOLUTELY GATING (RULE 20).  Every fact about a run is read
#    from that run's OWN ARGS / ENV / GROUPS / DATA lines.  The arm is
#    identified by the --stepsize-groups STRING, NEVER by the run name.  Any
#    repeated flag VOIDS that run: argparse keeps the last occurrence and two
#    whole batches of this campaign were silently voided that way.
# G1 BROKEN-RUN FLOOR, AND WHY IT DELIBERATELY DIFFERS FROM cI1's.
#    The registered in489g1 scorer voided any arm below FLOOR_W5 = 5.0 pp as
#    G1_NOT_LEARNED, which correctly refused ITS primaries: there, scalar and
#    blockwise were TREATMENT arms and comparing two dead arms means nothing.
#    HERE THE SCALAR ARM IS NOT A TREATMENT -- IT IS THE ZERO OF THE CAPTURE
#    SCALE.  in489g1 has already MEASURED that zero at ~1.03 pp, so importing a
#    5.0 pp floor would void the denominator of the only statistic this batch
#    computes and would return UNRESOLVED for every possible dataset, which is
#    not a scorer.  This departure is registered BEFORE any run exists, with its
#    reason, and it is NOT a relaxation dressed up: the floor is replaced, not
#    removed, by two stricter-where-it-matters conditions --
#      G1a a run whose W5 is <= CHANCE (0.2045 pp) or non-finite is VOID as
#          BROKEN.  A run that trains 80 epochs and reports ~1 pp on a
#          489-class task is a MEASUREMENT; a run at or below chance is a fault.
#      G1b SCALE GATE: the batch is scorable only if M(g62) >= 5.0 pp (the same
#          20x-chance floor cI1 used, applied to the arm that must LEARN) AND
#          the anchor gap M(g62) - M(g01) >= 10.0 pp.  Otherwise
#          UNRESOLVED_NO_SCALE and NO capture is printed at all.
# G2 m FROM THE LIVE TREE.  Each run prints GROUPS: with m read off its live
#    beta.  m must equal this file's expectation for that spec string and must
#    be identical across the seeds of an arm.
# G3 OPTIMIZER PROVENANCE.  HF_SOURCE must resolve inside the campaign's
#    maintained cifar10 tree, the only implementation that can build a
#    non-scalar arm at all.
# G4 DESIGN CONSISTENCY.  Every non-axis flag identical across all runs;
#    --stepsize-groups and --seed are the ONLY axes.  DATASET imagenet489,
#    classes 489, train_images 627329, val_images 24450.
# G5 FULL EPOCHS.  EPOCH_DEF must report short_epoch=False.
# G6 COUNT-LADDER COMPLETENESS.  M50 requires BOTH anchors and at least the
#    three cbl1-grid rungs live.  A ladder with a hole is reported as a ladder
#    with a hole and no M50 is interpolated across the hole.
#
# THE COMMON HORIZON, as in cI1 and for the same reason (no checkpoints, a
# flushed per-epoch line, real truncation risk on a contended partition):
#   ESTAR = min over live runs of the number of scored epochs; every arm is read
#   at the SAME ESTAR so unequal run lengths cannot bias any contrast.
#     ESTAR >= 60         full-budget; W5 may be called plateau5.
#     30 <= ESTAR < 60    verdict allowed, STAMPED short_budget, called
#                         window5@ESTAR and NEVER a plateau.
#     ESTAR < 30          UNRESOLVED_INSUFFICIENT_BUDGET at any bar.
#   AND THE BUDGET WARNING IS REGISTERED, NOT DISCOVERED LATER: CORRECTIONS 1
#   shows this ordering INVERTING with budget on CIFAR-10.  A crossing at ESTAR
#   is a crossing AT THAT BUDGET ONLY, and the full epoch-wise capture curve is
#   always printed so the report can carry the curve and not only the endpoint.
#
# WHAT THIS FILE MAY NEVER CLAIM.  Anything about ImageNet-1k; about any
# partition policy other than the CONTIGUOUS BALANCED one it runs (and note the
# convention puts the LARGER groups FIRST, so the tail is always the finer end
# -- that is a confound with cbl1's cut-position finding, it is inherited
# DELIBERATELY so the two ladders remain comparable, and R2 is the in-batch
# test of it); about CIFAR-10, ResNet-50, any meta-stepsize but 1e-3, any
# optimiser pair but SGDm+Lion, or any hierarchical or shrinkage operator
# (HIER is unset in every arm).
# =============================================================================

def balanced(m, total=T_TENSORS):
    """The contiguous balanced partition of `total` tensors into m groups, the
    LARGER groups first.  This is cbl1's own convention, reproduced so the two
    ladders share a policy as well as a grid."""
    q, r = divmod(total, m)
    return [q + 1] * r + [q] * (m - r)


def spec_of(sizes):
    return '[' + ','.join(str(x) for x in sizes) + ']'


# ---- parsing (identical contract to cI1_in489g1_score.py) -------------------
def _tokenise(argstr):
    toks = argstr.split()
    eff, order, i = {}, [], 0
    while i < len(toks):
        t = toks[i]
        if t.startswith('--'):
            flag, val = t[2:], ''
            if i + 1 < len(toks) and not toks[i + 1].startswith('--'):
                val = toks[i + 1]
                i += 1
            order.append(flag)
            eff[flag] = val
        i += 1
    return eff, order


def parse_out(path):
    r = {'path': path, 'name': os.path.basename(path), 'void': [], 'epochs': []}
    try:
        txt = open(path, errors='replace').read()
    except OSError as e:
        r['void'].append('UNREADABLE %s' % e)
        return r

    m = re.search(r'^ARGS:(.*)$', txt, re.M)
    if not m:
        r['void'].append('NO_ARGS_LINE')
        return r
    eff, order = _tokenise(m.group(1))
    r['args'] = eff
    dups = sorted({f for f in order if order.count(f) > 1})
    if dups:
        r['void'].append('REPEATED_FLAG %s' % ','.join(dups))

    m = re.search(r'^ENV:(.*)$', txt, re.M)
    r['env'] = dict(re.findall(r'(\w+)=(\S+)', m.group(1))) if m else {}
    if not m:
        r['void'].append('NO_ENV_LINE')

    m = re.search(r'^GROUPS:.*?\bm=(\d+)', txt, re.M)
    r['m'] = int(m.group(1)) if m else None
    m = re.search(r'^GROUPS:.*?\bn_param_tensors=(\d+)', txt, re.M)
    r['n_tensors'] = int(m.group(1)) if m else None
    m = re.search(r'^HF_SOURCE:\s*(\S+)', txt, re.M)
    r['hf_source'] = m.group(1) if m else None
    m = re.search(r'^DATA:.*?classes=(\d+)\s+train_images=(\d+)\s+val_images=(\d+)',
                  txt, re.M)
    r['data'] = ({'classes': int(m.group(1)), 'train_images': int(m.group(2)),
                  'val_images': int(m.group(3))} if m else None)
    m = re.search(r'^EPOCH_DEF:.*?short_epoch=(\w+)', txt, re.M)
    r['short_epoch'] = (m.group(1) == 'True') if m else None

    # RULE 20: the arm is the --stepsize-groups STRING, never the run name.
    r['spec'] = eff.get('stepsize-groups')
    r['arm'] = SPEC_TO_LABEL.get(r['spec'])
    r['seed'] = eff.get('seed')
    for e, tr, te in re.findall(
            r'^Epoch (\d+), Train Accuracy: ([0-9.]+) %, Test Accuracy: ([0-9.]+) %',
            txt, re.M):
        r['epochs'].append((int(e), float(tr), float(te)))
    r['epochs'].sort()
    r['n_ep'] = len(r['epochs'])
    r['done'] = 'RUN_DONE' in txt
    r['stopped'] = bool(re.search(r'^STOP: wall budget reached', txt, re.M))
    return r


def w5(run, estar, col=2):
    """col 2 = TEST, col 1 = TRAIN.  Mean over the last 5 epochs before estar."""
    v = [t[col] for t in run['epochs'] if t[0] < estar]
    return st.mean(v[-5:]) if len(v) >= 5 else None


# ---- gates ------------------------------------------------------------------
def apply_gates(runs):
    for r in runs:
        if r.get('void') and 'NO_ARGS_LINE' in r['void']:
            continue
        if r.get('arm') is None:
            r['void'].append('UNKNOWN_SPEC %r' % r.get('spec'))
        if r.get('short_epoch') is not False:
            r['void'].append('G5_SHORT_EPOCH short_epoch=%s' % r.get('short_epoch'))
        if r.get('m') is None:
            r['void'].append('G2_NO_GROUPS_LINE')
        elif r.get('arm') and r['m'] != EXPECT_M[r['arm']]:
            r['void'].append('G2_M_MISMATCH arm=%s m=%s expected=%s'
                             % (r['arm'], r['m'], EXPECT_M[r['arm']]))
        if r.get('n_tensors') is not None and r['n_tensors'] != T_TENSORS:
            r['void'].append('G2_TENSOR_COUNT %s != %d' % (r['n_tensors'], T_TENSORS))
        hs = r.get('hf_source') or ''
        if 'cifar10' not in hs or 'Optimizers/HF.py' not in hs:
            r['void'].append('G3_HF_PROVENANCE %r' % hs)
        if r.get('data') != EXPECT_DATA:
            r['void'].append('G4_DATA %r' % (r.get('data'),))
        if r.get('env', {}).get('DATASET') != 'imagenet489':
            r['void'].append('G4_DATASET %r' % r.get('env', {}).get('DATASET'))

    live = [r for r in runs if not r['void']]
    incon = []
    if live:
        allflags = set()
        for r in live:
            allflags |= set(r['args'])
        for f in sorted(allflags - AXIS_FLAGS):
            vals = {r['args'].get(f) for r in live}
            if len(vals) > 1:
                incon.append((f, sorted(str(v) for v in vals)))
        for k in ('BETA_CLIP', 'AUGMENT', 'HIER'):
            vals = {r['env'].get(k) for r in live}
            if len(vals) > 1:
                incon.append(('ENV:' + k, sorted(str(v) for v in vals)))
        for a in set(r['arm'] for r in live):
            ms = {r['m'] for r in live if r['arm'] == a}
            if len(ms) > 1:
                incon.append(('m[%s]' % a, sorted(str(v) for v in ms)))
    return runs, incon


# ---- statistics -------------------------------------------------------------
def pooled_sd(S):
    """Pooled within-arm SD over every arm with >= 2 seeds."""
    ss, df = 0.0, 0
    for arr in S.values():
        if len(arr) >= 2:
            mu = st.mean(arr)
            ss += sum((x - mu) ** 2 for x in arr)
            df += len(arr) - 1
    return (math.sqrt(ss / df), df) if df > 0 else (None, 0)


def contrast(M, S, a, b):
    """(diff, t) for arm a minus arm b, t from THIS batch's own seed spread,
    pooled across every arm that has >= 2 seeds so an arm reduced to n = 1 by a
    lost run still gets an interval."""
    if a not in M or b not in M:
        return None, None
    d = M[a] - M[b]
    sp, df = pooled_sd(S)
    if sp is None or sp <= 0:
        return d, None
    se = sp * math.sqrt(1.0 / len(S[a]) + 1.0 / len(S[b]))
    return d, (d / se if se > 0 else None)


def crossing(caps, rungs, level=0.5):
    """First crossing of `level`, interpolated LINEARLY IN log2 m between the
    two bracketing rungs.  Returns (m_lo, m_hi, m50, rise) or None."""
    pts = [(EXPECT_M[a], caps[a]) for a in rungs if a in caps]
    pts.sort()
    for i in range(len(pts) - 1):
        (ma, ca), (mb, cb) = pts[i], pts[i + 1]
        if ca < level <= cb:
            t = (level - ca) / (cb - ca)
            lo, hi = math.log2(ma), math.log2(mb)
            return ma, mb, 2.0 ** (lo + t * (hi - lo)), cb - ca
    return None


# ---- report -----------------------------------------------------------------
def score(paths):
    runs = [parse_out(p) for p in paths]
    runs, incon = apply_gates(runs)
    live = [r for r in runs if not r['void']]

    print('=' * 78)
    print('cI2_in489g2_score.py -- ImageNet-489 granularity-TRANSITION locator')
    print('  ImageNet-489: 489 of 1000 classes, 627,329 train, 24,450 val')
    print('  (50/class, labels RECOVERED FROM PER-IMAGE XML ANNOTATIONS).')
    print('  chance top-1 = 100/489 = %.4f%%.  NOT comparable to ImageNet-1k.' % CHANCE)
    print('  REGISTERED PREDICTION: M50 in [%.0f, %.0f], point %.2f (Model A),'
          % (PRED_M50_LO, PRED_M50_HI, PRED_M50_POINT))
    print('                         rival %.2f (Model B).  NEVER GATING.' % PRED_M50_RIVAL)
    print('=' * 78)

    print('\nG0 PROVENANCE -- %d file(s), %d live, %d void'
          % (len(runs), len(live), len(runs) - len(live)))
    for r in sorted(runs, key=lambda r: r['name']):
        print('  %s%-40s arm=%-4s m=%-6s seed=%-3s ep=%-4s%s'
              % ('LIVE ' if not r['void'] else 'VOID ', r['name'], r.get('arm'),
                 r.get('m'), r.get('seed'), r.get('n_ep'),
                 '' if not r['void'] else '  <- ' + '; '.join(r['void'])))
    if incon:
        print('\nG4 REFUSES -- non-axis flags differ across the batch:')
        for f, v in incon:
            print('    %-28s %s' % (f, v))
        print('\nVERDICT: UNRESOLVED (G4 design inconsistency; RULE 20)')
        return 'UNRESOLVED'
    if not live:
        print('\nVERDICT: UNRESOLVED (no live run)')
        return 'UNRESOLVED'
    print('\nG4 PASS -- every non-axis flag identical across the %d live runs.' % len(live))

    estar = min(r['n_ep'] for r in live)
    print('\nCOMMON HORIZON  ESTAR = %d   (per-run scored epochs: %s)'
          % (estar, ', '.join('%s.s%s=%d' % (r['arm'], r['seed'], r['n_ep'])
                              for r in sorted(live, key=lambda r: (r['arm'], r['seed'])))))
    if estar < ESTAR_MIN:
        print('  ESTAR < %d -> no verdict at any bar.' % ESTAR_MIN)
        print('\nVERDICT: UNRESOLVED_INSUFFICIENT_BUDGET')
        return 'UNRESOLVED_INSUFFICIENT_BUDGET'
    short = estar < ESTAR_FULL
    label = 'window5@%d' % estar if short else 'plateau5'
    if short:
        print('  %d <= ESTAR < %d -> STAMPED short_budget; the primary statistic'
              % (ESTAR_MIN, ESTAR_FULL))
        print('  is "%s" and MUST NOT be called a plateau.' % label)

    # G1a broken-run floor
    S, ST = {}, {}
    for r in live:
        te, tr = w5(r, estar, 2), w5(r, estar, 1)
        if te is None or not math.isfinite(te) or te <= CHANCE:
            r['void'].append('G1a_BROKEN %s=%s <= chance %.4f' % (label, te, CHANCE))
            continue
        S.setdefault(r['arm'], []).append(te)
        ST.setdefault(r['arm'], []).append(tr)
    live = [r for r in live if not r['void']]
    for r in runs:
        if r['void'] and any(v.startswith('G1a') for v in r['void']):
            print('  G1a VOID %s -- %s' % (r['name'], r['void'][-1]))

    M = {a: st.mean(v) for a, v in S.items()}
    MT = {a: st.mean(v) for a, v in ST.items()}
    sp, spdf = pooled_sd(S)

    print('\nARM MEANS -- %s, TEST and TRAIN, both always reported' % label)
    print('  %-4s %-8s %5s   %10s %8s   %10s %8s' %
          ('arm', 'spec', 'm', 'TEST', 'sd', 'TRAIN', 'sd'))
    for a in [x for x in LADDER if x in M] + [x for x in SPECS if x not in LADDER and x in M]:
        sd = st.stdev(S[a]) if len(S[a]) > 1 else float('nan')
        sdt = st.stdev(ST[a]) if len(ST[a]) > 1 else float('nan')
        nm = SPECS[a] if len(SPECS[a]) <= 8 else SPECS[a][:6] + '..'
        print('  %-4s %-8s %5d   %10.4f %8.4f   %10.4f %8.4f  n=%d'
              % (a, nm, EXPECT_M[a], M[a], sd, MT[a], sdt, len(S[a])))
    if sp is not None:
        print('  pooled within-arm SD = %.4f pp (df %d);  in489g1 measured %.4f'
              % (sp, spdf, POOLED_SD_IN489G1))

    # G1b scale gate
    if ANCHOR_LO not in M or ANCHOR_HI not in M:
        print('\nG1b REFUSES -- an anchor is missing (%s / %s present: %s / %s)'
              % (ANCHOR_LO, ANCHOR_HI, ANCHOR_LO in M, ANCHOR_HI in M))
        print('\nVERDICT: UNRESOLVED_NO_SCALE')
        return 'UNRESOLVED_NO_SCALE'
    gap = M[ANCHOR_HI] - M[ANCHOR_LO]
    print('\nG1b SCALE GATE   M(%s) = %.4f (floor %.1f)   gap = %.4f (min %.1f)'
          % (ANCHOR_HI, M[ANCHOR_HI], LAY_FLOOR, gap, ANCHOR_GAP_MIN))
    if M[ANCHOR_HI] < LAY_FLOOR or gap < ANCHOR_GAP_MIN:
        print('  the capture denominator is not a real gap -- NO capture is printed.')
        print('\nVERDICT: UNRESOLVED_NO_SCALE')
        return 'UNRESOLVED_NO_SCALE'
    print('  PASS -- capture is on a real scale.')
    gapT = MT[ANCHOR_HI] - MT[ANCHOR_LO]

    caps = {a: (M[a] - M[ANCHOR_LO]) / gap for a in M}
    capsT = {a: (MT[a] - MT[ANCHOR_LO]) / gapT for a in MT} if gapT != 0 else {}

    print('\nTHE COUNT LADDER (balanced contiguous partitions of %d tensors)' % T_TENSORS)
    print('  CAPTURE = (M(arm) - M(%s)) / (M(%s) - M(%s)); 0 and 1 BY CONSTRUCTION.'
          % (ANCHOR_LO, ANCHOR_HI, ANCHOR_LO))
    print('  %-4s %5s   %10s %9s   %10s %9s' % ('arm', 'm', 'TEST', 'CAPTURE',
                                                'TRAIN', 'TRAINCAP'))
    rung = [a for a in LADDER if a in caps]
    for a in rung:
        print('  %-4s %5d   %10.4f %+9.4f   %10.4f %+9.4f'
              % (a, EXPECT_M[a], M[a], caps[a], MT[a], capsT.get(a, float('nan'))))
    print('  capture SE at this batch\'s own spread ~ %.5f'
          % ((sp * math.sqrt(1.0) / gap) if sp else float('nan')))

    # G6 completeness
    holes = [a for a in LADDER if a not in caps]
    grid_ok = all(a in caps for a in GRID_CBL1)
    if holes:
        print('\nG6 -- the count ladder has holes at %s; no M50 is interpolated '
              'across a hole.' % ','.join(holes))

    # R1 monotonicity
    drops = [(rung[i], rung[i + 1], caps[rung[i + 1]] - caps[rung[i]])
             for i in range(len(rung) - 1)]
    worst_drop = min([d for _, _, d in drops], default=0.0)
    r1 = worst_drop < -DROP_BAR
    print('\nR1 MONOTONICITY  worst adjacent change = %+.4f (bar: a drop worse '
          'than %-.2f refutes)' % (worst_drop, -DROP_BAR))
    for a, b, d in drops:
        print('    %s -> %s   %+.4f' % (a, b, d))
    print('    %s' % ('R1 FIRES: NONMONOTONE -- a smooth monotone transition is REFUTED.'
                      if r1 else 'R1 does not fire: the ladder is monotone within the bar.'))

    # R2 position control
    r2 = False
    if 'p08' in caps and 'g08' in caps:
        d, t = contrast(M, S, 'p08', 'g08')
        dc = caps['p08'] - caps['g08']
        r2 = abs(dc) > SHAPE_BAR
        print('\nR2 POSITION CONTROL AT MATCHED m = 8')
        print('    g08 %-34s CAPTURE %+.4f' % (SPECS['g08'], caps['g08']))
        print('    p08 %-34s CAPTURE %+.4f' % (SPECS['p08'], caps['p08']))
        print('    dCAPTURE = %+.4f   dTEST = %+.4f pp   t = %s   (bar |dCAP| > %.2f)'
              % (dc, d, ('%+.2f' % t) if t is not None else 'na', SHAPE_BAR))
        if r2:
            print('    R2 FIRES: POSITION-DOMINATES.  Two partitions with IDENTICAL m')
            print('    differ by %.4f in capture, so m is NOT the operative variable at'
                  % abs(dc))
            print('    489 classes.  The count statistic is DISQUALIFIED and is NOT')
            print('    replaced by a better-fitting one.  No M50 is a property of m.')
        else:
            print('    R2 does not fire: at m = 8 the partition SHAPE moves capture by')
            print('    less than %.2f, so the count reading survives its in-batch rival.'
                  % SHAPE_BAR)
    else:
        print('\nR2 CANNOT BE EVALUATED -- the position control is not live.')

    # R4 low end
    r4 = 'g08' in caps and caps['g08'] > LOWEND_BAR
    print('\nR4 LOW END       CAPTURE(g08) = %s (bar > %.2f refutes a flat low end)'
          % (('%+.4f' % caps['g08']) if 'g08' in caps else 'na', LOWEND_BAR))
    if r4:
        print('    R4 FIRES: the crossing is BELOW the registered interval.')

    # R5 train/test divergence
    div = [(a, capsT[a] - caps[a]) for a in rung if a in capsT]
    worst = max(div, key=lambda x: abs(x[1])) if div else (None, 0.0)
    r5 = abs(worst[1]) > TRAIN_DIVERGE_BAR
    print('\nR5 TRAIN vs TEST worst |TRAINCAP - CAPTURE| = %.4f at %s (bar %.2f)'
          % (abs(worst[1]), worst[0], TRAIN_DIVERGE_BAR))
    print('    %s' % ('R5 FIRES: the two ladders come apart -- the coarse-arm failure '
                      'is no longer purely train-side.  THAT IS A FINDING.' if r5 else
                      'R5 does not fire: the train ladder tracks the test ladder, as on '
                      'CIFAR-100, Tiny-ImageNet and in489g1.'))

    # the crossing
    cr = crossing(caps, rung) if not holes or grid_ok else None
    print('\nM50 -- WHERE CAPTURE CROSSES 0.5')
    r3 = False
    if cr is None:
        top = max((caps[a] for a in rung if a != ANCHOR_HI), default=float('nan'))
        print('    NO CROSSING at any interior rung (max interior CAPTURE = %+.4f).' % top)
        print('    The transition sits in the LAST bracket, between m = %d and m = 62.'
              % max(EXPECT_M[a] for a in rung if a != ANCHOR_HI))
    else:
        ma, mb, m50, rise = cr
        r3 = rise > CLIFF_BAR
        print('    BRACKET  (m = %d, m = %d]   rise across it %+.4f' % (ma, mb, rise))
        print('    interpolated (linear in log2 m)  M50 = %.2f' % m50)
        if r3:
            print('    R3 FIRES: the rise across the bracket exceeds %.2f, so this is a'
                  % CLIFF_BAR)
            print('    CLIFF, not a transition.  The BRACKET stands; the point estimate')
            print('    M50 = %.2f DOES NOT and must not be quoted.' % m50)
        else:
            print('    R3 does not fire: the bracket rise is under %.2f, so the'
                  % CLIFF_BAR)
            print('    interpolated point is admissible INSIDE ITS BRACKET.')
        hit = PRED_M50_LO <= m50 <= PRED_M50_HI
        print('    REGISTERED PREDICTION [%.0f, %.0f]: %s   (point %.2f vs predicted '
              '%.2f, rival %.2f)  -- NEVER GATING'
              % (PRED_M50_LO, PRED_M50_HI, 'HIT' if hit else 'MISS', m50,
                 PRED_M50_POINT, PRED_M50_RIVAL))

    # cross-dataset, DESCRIPTIVE ONLY
    print('\nSHARED GRID WITH cbl1 (CIFAR-100) -- CROSS-BATCH, CROSS-DATASET,')
    print('DESCRIPTIVE ONLY.  It gates nothing and may not be promoted over any')
    print('in-batch number above.  Same balanced spec strings, byte-for-byte.')
    ref = cbl1_ladder()
    if ref:
        print('    %5s   %12s   %12s' % ('m', 'CIFAR-100', 'ImageNet-489'))
        for a in LADDER:
            m_ = EXPECT_M[a]
            c1 = ref.get(m_)
            print('    %5d   %12s   %12s'
                  % (m_, ('%+.4f' % c1) if c1 is not None else '--',
                     ('%+.4f' % caps[a]) if a in caps else '--'))
        print('    CIFAR-100 crossing: M50 = %.2f (bracket 8-16).' % M50_C100)

    # verdict
    print('\n' + '=' * 78)
    if r2:
        v = 'COUNT-DISQUALIFIED-POSITION-DOMINATES'
    elif r1:
        v = 'NONMONOTONE-SMOOTH-TRANSITION-REFUTED'
    elif cr is None:
        v = 'TRANSITION-ABOVE-45-BRACKET-45-62'
    elif r3:
        v = 'TRANSITION-CLIFF-BRACKET-%d-%d' % (cr[0], cr[1])
    else:
        v = 'TRANSITION-LOCATED-BRACKET-%d-%d' % (cr[0], cr[1])
    if short:
        v += '-SHORT_BUDGET'
    print('VERDICT: %s' % v)
    if r4:
        print('  plus R4: the low end is NOT flat; the registered prediction is a MISS.')
    if r5:
        print('  plus R5: TRAIN and TEST ladders diverge -- report it as a finding.')
    print('SCOPE: ImageNet-489 only; contiguous BALANCED partitions with the larger')
    print('groups first; ms 1e-3, alpha0 1e-6, SGDm+Lion, HIER unset, %s.' % label)
    print('=' * 78)
    return v


# ---- archived reference ------------------------------------------------------
def cbl1_ladder():
    """cbl1's CIFAR-100 capture ladder, RE-DERIVED from results/all_runs.csv at
    call time.  Never a literal."""
    if not os.path.exists(CSV):
        return None
    by = {}
    for r in csv.DictReader(open(CSV)):
        if not r['run'].startswith('cbl1-'):
            continue
        by.setdefault(r['run'].split('-')[1], []).append(float(r['plateau5']))
    if 'c01' not in by or 'c62' not in by:
        return None
    s, l = st.mean(by['c01']), st.mean(by['c62'])
    lab = {1: 'c01', 2: 'c02', 4: 'c04', 8: 'c08', 16: 'c16', 32: 'c32', 62: 'c62'}
    return {m: (st.mean(by[a]) - s) / (l - s) for m, a in lab.items() if a in by}


# ---- selftest ---------------------------------------------------------------
def selftest():
    n = [0]

    def chk(cond, msg):
        n[0] += 1
        if not cond:
            print('  FAIL %s' % msg)
            raise SystemExit(1)
        print('  ok   %s' % msg)

    print('cI2_in489g2_score.py --selftest')
    chk(os.path.exists(CSV), 'corpus present at %s' % CSV)
    rows = list(csv.DictReader(open(CSV)))
    chk(len(rows) > 2000, 'corpus has %d rows' % len(rows))

    # RULE 21 premise
    chk(not [r for r in rows if r['run'].startswith('in489g2')],
        'RULE 21: ZERO in489g2 rows in the corpus at registration')

    # partitions re-derived, not quoted
    chk(spec_of(balanced(8)) == SPEC_G08, 'balanced(8) re-derives %s' % SPEC_G08)
    chk(spec_of(balanced(16)) == SPEC_G16, 'balanced(16) re-derives SPEC_G16')
    chk(spec_of(balanced(32)) == SPEC_G32, 'balanced(32) re-derives SPEC_G32')
    chk(spec_of(balanced(45)) == SPEC_G45, 'balanced(45) re-derives SPEC_G45')
    for a in ('g08', 'g16', 'g32', 'g45'):
        sz = [int(x) for x in SPECS[a][1:-1].split(',')]
        chk(sum(sz) == T_TENSORS and len(sz) == EXPECT_M[a],
            '%s covers %d tensors in %d groups' % (a, T_TENSORS, EXPECT_M[a]))
    psz = [int(x) for x in SPEC_P08[1:-1].split(',')]
    chk(sum(psz) == T_TENSORS and len(psz) == 8,
        'p08 covers %d tensors in 8 groups (position control at matched m)' % T_TENSORS)
    chk(psz != balanced(8), 'p08 is NOT the balanced partition at m = 8')

    # the shared grid is byte-identical to cbl1's OWN granularity strings
    gr = {r['granularity'] for r in rows if r['run'].startswith('cbl1-')}
    for a in GRID_CBL1:
        chk(SPECS[a] in gr, '%s spec is byte-identical to a cbl1 arm: %s' % (a, SPECS[a]))
    chk(SPEC_G45 not in gr, 'g45 is new -- it bisects the (32,62) bracket cbl1 left open')
    chk(SPEC_P08 not in gr, 'p08 is new')

    # in489g1 re-derivation
    i4 = [r for r in rows if r['run'].startswith('in489g1-')]
    chk(len(i4) == 12, 'in489g1 has 12 rows')
    by = {}
    for r in i4:
        by.setdefault(r['granularity'], []).append(float(r['plateau5']))
    s, l = st.mean(by['scalar']), st.mean(by['layerwise'])
    gap = l - s
    cap6 = (st.mean(by['resnet18_blocks']) - s) / gap
    chk(abs(gap - ANCHOR_GAP_IN489G1) < 5e-4,
        'in489g1 anchor gap re-derives to %.4f pp' % gap)
    chk(abs(cap6 - CAP6_IN489) < 5e-6,
        'in489g1 CAPTURE(m=6) re-derives to %.6f' % cap6)
    ss, df = 0.0, 0
    for v in by.values():
        mu = st.mean(v)
        ss += sum((x - mu) ** 2 for x in v)
        df += len(v) - 1
    sd = math.sqrt(ss / df)
    chk(abs(sd - POOLED_SD_IN489G1) < 5e-5,
        'in489g1 pooled within-arm SD re-derives to %.4f pp (df %d)' % (sd, df))
    chk(abs(sd * math.sqrt(1.0 / 2 + 1.0 / 2) / gap - CAPTURE_SE_N2) < 5e-5,
        'capture SE at n = 2 re-derives to %.5f' % (sd / gap))
    chk(0.5 / (sd * math.sqrt(1.0) / gap) > 150,
        'n = 2 leaves the 0.5 crossing threshold %.0f capture-SE away'
        % (0.5 / (sd / gap)))

    # the two registered models, re-derived from that one point
    x6 = math.log(6) / math.log(62)
    p = math.log(cap6) / math.log(x6)
    chk(abs(p - P_IN489) < 5e-4, 'Model A exponent re-derives to p = %.4f' % p)
    m50a = 62 ** (0.5 ** (1.0 / p))
    chk(abs(m50a - PRED_M50_POINT) < 5e-2,
        'Model A crossing re-derives to M50 = %.2f' % m50a)
    m50b = 6 * (62 / 6.0) ** ((0.5 - cap6) / (1 - cap6))
    chk(abs(m50b - PRED_M50_RIVAL) < 5e-2,
        'Model B crossing re-derives to M50 = %.2f' % m50b)
    chk(PRED_M50_LO <= m50a <= PRED_M50_HI and PRED_M50_LO <= m50b <= PRED_M50_HI,
        'both models land inside the registered interval [%.0f, %.0f]'
        % (PRED_M50_LO, PRED_M50_HI))
    chk(m50a > 16 and m50b > 16,
        'the cheap shared consequence: BOTH models put the crossing above m = 16')

    # cbl1 re-derivation
    ref = cbl1_ladder()
    chk(ref is not None and len(ref) == 7, 'cbl1 ladder re-derives 7 rungs')
    chk(abs(ref[1]) < 1e-12 and abs(ref[62] - 1) < 1e-12,
        'cbl1 capture is pinned 0 at m=1 and 1 at m=62 BY CONSTRUCTION')
    best = min(((pp / 1000.0,
                 sum((ref[m] - (math.log(m) / math.log(62)) ** (pp / 1000.0)) ** 2
                     for m in (2, 4, 8, 16, 32)))
                for pp in range(200, 6000)), key=lambda z: z[1])
    chk(abs(best[0] - P_C100) < 2e-3,
        'CIFAR-100 best-fit exponent re-derives to p = %.3f (vs %.3f here)'
        % (best[0], p))
    cc = crossing({a: ref[EXPECT_M[a]] for a in LADDER if EXPECT_M[a] in ref},
                  [a for a in LADDER if EXPECT_M[a] in ref])
    chk(cc is not None and abs(cc[2] - M50_C100) < 5e-2,
        'CIFAR-100 crossing re-derives to M50 = %.2f in bracket (%d,%d)'
        % (cc[2], cc[0], cc[1]))
    chk(cc[2] < PRED_M50_LO,
        'CIFAR-100 crosses BELOW the interval predicted for ImageNet-489 '
        '(%.2f < %.0f): the ladder is predicted to move RIGHT with class count'
        % (cc[2], PRED_M50_LO))

    # SEED is null in this corpus -- the justification for n = 2, re-derived
    import collections
    def key(r):
        return tuple(r[k] for k in ['dataset', 'network', 'granularity', 'base',
                                    'meta', 'meta_stepsize', 'alpha0', 'beta_clip',
                                    'augment', 'epochs_requested', 'batch_size',
                                    'hier', 'gamma'])
    cells = collections.defaultdict(list)
    for r in rows:
        if r.get('window_ok') != '1' or r.get('complete') != '1':
            continue
        if r.get('superseded') not in ('0', '', 'None'):
            continue
        try:
            v = float(r['plateau5'])
        except (ValueError, KeyError):
            continue
        cells[key(r)].append((r['seed'], v))
    use = {k: v for k, v in cells.items()
           if len(v) >= 2 and len({s for s, _ in v}) >= 2}
    resid = []
    for v in use.values():
        mu = st.mean(x for _, x in v)
        resid += [(s, x - mu) for s, x in v]
    bys = collections.defaultdict(list)
    for s, x in resid:
        bys[s].append(x)
    bys = {s: v for s, v in bys.items() if len(v) >= 2}
    grand = st.mean(x for _, x in resid)
    k_ = len(bys)
    N = sum(len(v) for v in bys.values())
    ssb = sum(len(v) * (st.mean(v) - grand) ** 2 for v in bys.values())
    ssw = sum(sum((x - st.mean(v)) ** 2 for x in v) for v in bys.values())
    F = (ssb / (k_ - 1)) / (ssw / (N - k_))
    chk(F < 1.0, 'SEED is null after removing cell means: F(%d,%d) = %.3f over '
                 '%d cells / %d rows -- this is why the seed budget was cut to 2'
                 % (k_ - 1, N - k_, F, len(use), N))

    # bar sanity
    chk(LAY_FLOOR >= 20 * CHANCE, 'the learn floor is >= 20x chance (%.4f)' % CHANCE)
    chk(DROP_BAR > 10 * CAPTURE_SE_N2, 'R1 bar is > 10 capture-SE')
    chk(SHAPE_BAR > 10 * CAPTURE_SE_N2, 'R2 bar is > 10 capture-SE')
    chk(0 < LOWEND_BAR < 0.5 < CLIFF_BAR, 'R3/R4 bars straddle the 0.5 threshold')
    chk(len(LADDER) == 6 and len(SPECS) == 7,
        'the design is 6 count-ladder rungs + 1 position control = 7 arms')
    chk(EXPECT_M['p08'] == EXPECT_M['g08'],
        'the position control is at m MATCHED to a count-ladder rung')

    print('\nSELFTEST %d/%d PASS' % (n[0], n[0]))
    return 0


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--runs', default='/cluster/account_a/metaopt/runs')
    ap.add_argument('--glob', default='in489g2-*.out')
    ap.add_argument('--selftest', action='store_true')
    a = ap.parse_args()
    if a.selftest:
        return selftest()
    paths = sorted(glob.glob(os.path.join(a.runs, a.glob)) +
                   glob.glob(os.path.join(a.runs, 'in489g2', a.glob)))
    if not paths:
        print('no in489g2 .out files under %s' % a.runs)
