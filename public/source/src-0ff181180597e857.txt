#!/bin/bash
# =============================================================================
# Cycle 47 -- ARM C: the robustness comparison the paper's claim actually needs.
#
# OPEN QUESTION IT ANSWERS
#   Cycles 45/46 measured MetaOptimize's alpha0 robustness (arm B) against a
#   GENUINELY FIXED learning rate (arm A).  That is the parent paper's own framing
#   and arm B wins it easily -- span 3.3pp vs 21.9pp over the shared sub-grid.
#   But nobody ships a constant LR.  The competitor a practitioner would actually
#   pick is a tuned schedule, and the campaign ALREADY HAS THAT SWEEP: `SW_*`
#   (cycle 43) is AdamW + horizon-matched cosine (COS_TOTAL=50000) over peak LR
#   1e-5..1e-2, at the identical network / dataset / batch size / augmentation /
#   budget / ACCOUNT.  Adding it as arm C costs nothing and it changes the verdict
#   (FINDINGS 47.1):
#
#     shared sub-grid {1e-5,1e-4,3e-4,1e-3,1e-2}   A fixed   B meta   C cosine
#       peak plateau                                91.796   93.350    94.028
#       worst plateau                               69.898   90.095    83.884
#       width <=1pp of own best (decades)            0.477    0.000     0.523
#       width <=2pp of own best                      1.000    2.000     2.000
#       width above ABSOLUTE 90                      0.477    3.000     2.000
#
#   So MetaOptimize is NOT flatter than a tuned cosine in the scale-free sense
#   (loses at 1pp, ties at 2pp and 3pp) and its peak is 0.678pp lower on-grid.
#   Its ONE surviving advantage is the ABSOLUTE floor: it holds >=90 over 3 decades
#   where the cosine holds it over 2.
#
#   THE PROBLEM THIS BATCH FIXES: arm C has no cell at 1e-6 or 1e-1, and those are
#   exactly the two grid points where arm B's advantage is claimed.  The head-to-head
#   is therefore being decided on 5 of 7 points, with the 2 missing ones chosen by
#   which arm happened to be swept first.  That is not a comparison, it is a censor.
#
# PRE-REGISTERED PREDICTIONS (write the verdict against these, not whatever lands)
#   (C1) NEARLY FORCED, and stated so it is not scored as a success.  Arm C at peak
#        LR 1e-6 warms up linearly to 1e-6 over 10,000 steps and then cosine-decays
#        to ~0, so its MEAN lr is strictly below arm A's constant 1e-6.  Predict
#        C(1e-6) <= A(1e-6) = 68.680 +-0.530, i.e. below 70.  REFUTATION: C(1e-6) > 85
#        would mean warmup+decay recovers from a 1e-6 peak, which is mechanically
#        surprising and would delete MetaOptimize's bottom-end advantage outright.
#   (C2) THE INFORMATIVE ONE.  At 1e-1 the warmup is doing real work that arm A does
#        not get (A(1e-1) = 13.896).  Predict C(1e-1) in [20, 85].  If C(1e-1) >= 90
#        the cosine's >=90 width grows to 3 decades and TIES arm B, deleting the last
#        advantage.  If C(1e-1) < 20 the warmup buys nothing at the top and arm A and
#        arm C fail the same way there.
#   (C3) THE DECIDING ARITHMETIC, computable in advance.  Under C1 and C2, arm C's
#        width above 90 STAYS at 2.0 decades [1e-4..1e-2].  Arm B's grows from 3.0 to
#        4.0 decades, because B(1e-6) = 91.396 is already measured and is >= 90 and
#        adjacent to the existing [1e-5..1e-2] run.  So the pre-registered full-grid
#        result is  B 4.0 decades vs C 2.0 decades above 90 -- MetaOptimize's genuine
#        advantage, quantified, and worth exactly 2 decades of alpha0.
#        REFUTATION of that sentence: C(1e-1) >= 90, or B's 1e-6 cell moving below 90
#        when its third seed lands.
#
#   RESOLUTION.  Cell sd on arm C is 0.086-0.170pp at n=2, so a 2-seed cell has
#   s.e. ~0.1pp.  Differences below ~0.3pp between arm C cells are NOT resolvable
#   here and must not be used to move a band edge.  Which is precisely why the
#   1e-4 cell is being topped up -- see the seed plan.
#
# SEED PLAN, and every job names the question it answers
#   1e-6 x s{0,1,2}   NEW grid point -- C1/C3, the bottom extreme
#   1e-1 x s{0,1,2}   NEW grid point -- C2/C3, the top extreme
#   1e-4 x s{2,3}     DECISION-CRITICAL.  C(1e-4) = 92.962 against a 1pp band edge of
#                     93.028: it misses by 0.066pp, which is BELOW the 0.1pp s.e. of
#                     the cell.  Arm C's `width <=1pp` is 0.523 or 1.000 decades on a
#                     coin flip.  n=2 -> n=4 resolves it.
#   3e-4 x s2         n=2 -> 3, the other 1pp band edge
#   1e-3 x s2         n=2 -> 3, arm C's on-grid peak
#   = 10 jobs, ~30-70 min each (measured from SW_*: 30 min on a fast node, 69 on a
#   2080ti).  ~0.1% of RawUsage, inside the revised cycle-44 budget rule.
#
# DESIGN.  Byte-matched to `SW_*` -- recovered from SW-3e-4-s0's own ARGS:/ENV: lines:
#     ARGS: --optimizer AdamW --alpha0 3e-4 --dataset CIFAR10 --NN-name ResNet18
#           --batch-size 100 --max-time 999:00:00 --gamma 1 --num-epochs 100 --seed 0
#     ENV:  AUGMENT=1 BETA_CLIP=none HIER=none LAM=na ETA_RATIO=na
#           COS_TOTAL=50000 COS_WARMUP=default
#   `BETA_CLIP=none HIER=none LAM=na ETA_RATIO=na` is what run_cifar.sh ECHOES for an
#   UNSET variable -- exporting those literals kills the job (CONTINUE-HERE gotchas).
#   So this batch exports ONLY  ALL,AUGMENT=1,COS_TOTAL=50000  and deliberately leaves
#   COS_WARMUP unset so the ENV line reads `COS_WARMUP=default`, byte-matching SW_*.
#   Verified against the live source this cycle (build_optimizer.py:60-62): the default
#   is 10000 steps = 20 epochs of linear warmup, then cosine over the 50,000-step
#   horizon.  This is the same arm, not a near-match.
#
# MUST RUN ON cluster_a.  `SW_*` ran on account_a; arm A did too.  Keeping arm C
# single-account removes the confound that cost CORRECTIONS 10 a whole claim.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. every i3c .out must echo  COS_TOTAL=50000 COS_WARMUP=default  -- if it says
#      COS_TOTAL=default the export did not take and the run is a 422000-step cosine,
#      i.e. no decay at all, i.e. silently arm A with a warmup.
#   2. `--optimizer AdamW` must appear in ARGS (NOT `--optimizer HF`), else it is arm B.
#   3. python3 analysis/idea3_threearm.py --selftest  -> 22/22 PASS
#
# USAGE
#   bash c47_idea3_armC_cosine.sh                        # dry run, submits nothing
#   bash c47_idea3_armC_cosine.sh --submit               # submits if guards pass
#   bash c47_idea3_armC_cosine.sh --submit --force-fairshare
# =============================================================================
set -uo pipefail

SUBMIT=0
FORCE_FS=0
for a in "$@"; do
  [ "$a" = "--submit" ] && SUBMIT=1
  [ "$a" = "--force-fairshare" ] && FORCE_FS=1
done

USER_NAME=account_a
ROOT=/cluster/account_a/metaopt
RUNNER=$ROOT/jobs/run_cifar.sh
OUT=$ROOT/runs/i3c
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
TLIM=03:50:00

FS_FLOOR=0.35
PEND_CAP=40          # the operator's hard limit, not the withdrawn 0.35-fairshare floor

echo "=== c47 ARM C (AdamW + horizon-matched cosine) -- 10 jobs ==="

# ---- guard 1: the runner must exist -------------------------------------------------
if [ ! -f "$RUNNER" ]; then
  echo "GUARD FAIL: runner not found at $RUNNER"
  [ "$SUBMIT" = "1" ] && exit 1
fi

# ---- guard 2: FairShare, WITHDRAWN as a floor but still reported ---------------------
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
FS_OK=$(awk -v a="${FS:-0}" -v b="$FS_FLOOR" 'BEGIN{print (a+0>=b+0)?1:0}')
echo "FairShare = ${FS:-unknown} (nominal floor $FS_FLOOR, FS_OK=$FS_OK)"
if [ "$FS_OK" != "1" ] && [ "$FORCE_FS" = "1" ]; then
  echo "  --force-fairshare: OVERRIDDEN.  CORRECTIONS 35 measured PriorityDecayHalfLife"
  echo "  = 14 days, so the floor cannot be waited out; CORRECTIONS 37 replaces it with a"
  echo "  BATCH-SIZE rule.  This batch is 10 jobs of ~40 min = ~0.1% of RawUsage."
fi

# ---- guard 3: pending cap (the real hard limit) --------------------------------------
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "pending  = $PEND (cap $PEND_CAP)"
PEND_OK=$(awk -v a="$PEND" -v b="$PEND_CAP" 'BEGIN{print (a+0<=b+0)?1:0}')

if [ "$SUBMIT" = "1" ]; then
  if [ "$PEND_OK" != "1" ]; then
    echo "GUARD FAIL: pending $PEND exceeds cap $PEND_CAP -- refusing to submit"; exit 1
  fi
  if [ "$FS_OK" != "1" ] && [ "$FORCE_FS" != "1" ]; then
    echo "GUARD FAIL: FairShare below floor and --force-fairshare not given"; exit 1
  fi
  mkdir -p "$OUT"
fi

# armC <alpha0> <token> <seed>
armC () {
  local a0="$1" tok="$2" s="$3"
  local name="i3c-${tok}-s${s}"
  local cmd=(sbatch --job-name="$name" --partition="$PARTS" --time="$TLIM"
    --export=ALL,AUGMENT=1,COS_TOTAL=50000
    "$RUNNER"
    --optimizer AdamW --alpha0 "$a0" --seed "$s"
    --num-epochs 100 --NN-name ResNet18 --dataset CIFAR10 --batch-size 100
    --max-time 999:00:00 --gamma 1
    --save-directory "$OUT" --run-name "$name")
  if [ "$SUBMIT" = "1" ]; then "${cmd[@]}"; else printf '%s\n' "${cmd[*]}"; fi
}

N=0
# the two NEW grid points -- C1 / C2 / C3
for s in 0 1 2; do armC 1e-6 1e6 "$s"; N=$((N+1)); done
for s in 0 1 2; do armC 1e-1 1e1 "$s"; N=$((N+1)); done
# the band-edge top-ups -- resolution, see the seed plan
for s in 2 3; do armC 1e-4 1e4 "$s"; N=$((N+1)); done
armC 3e-4 3e4 2; N=$((N+1))
armC 1e-3 1e3 2; N=$((N+1))

echo "=== $N jobs $([ "$SUBMIT" = "1" ] && echo SUBMITTED || echo 'printed (dry run -- nothing submitted)') ==="
