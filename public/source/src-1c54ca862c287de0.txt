#!/usr/bin/env python3
"""SYNTHETIC cvk1 run directory for a scoring-path smoke test of the REGISTERED,
UNEDITED scorer.  Every number here is INVENTED and is evidence of nothing.
Structure copied from the harness (jobs/run_cifar.sh lines 18-24 and a real
cpk1 PROBE=0 .out): NODE line, ARGS line, ENV line, GPU line, warnings, Epoch
lines 0..99, '<n>  minutes', RUN_DONE.  File name <jobname>-<jobid>.out, as the
runner's `#SBATCH --output=.../runs/%x-%j.out` writes it.

usage: cvk1_smoke_gen.py <outdir> <manifest-to-copy> [--drop ARM-SEED] [--dup ARM-SEED]
"""
import math, os, random, shutil, sys

out, man = sys.argv[1], sys.argv[2]
drop = sys.argv[sys.argv.index("--drop") + 1] if "--drop" in sys.argv else None
dup = sys.argv[sys.argv.index("--dup") + 1] if "--dup" in sys.argv else None
WS = "/cluster/account_b/metaopt"
SPEC = {"k01": "scalar", "kL": "layerwise", "k13": "[13,13]", "k16": "[16,10]",
        "k19": "[19,7]", "k20": "[20,6]", "k21": "[21,5]", "k22": "[22,4]",
        "k23": "[23,3]"}
ARMS = ["k01", "kL", "k13", "k16", "k19", "k20", "k21", "k22", "k23"]
# INVENTED plateau targets (pp); a shape chosen only so every scorer path runs.
LEVEL = {"k01": 35.0, "kL": 66.0, "k13": 47.0, "k16": 54.0, "k19": 60.0,
         "k20": 58.5, "k21": 59.0, "k22": 64.0, "k23": 37.0}
ENV = ("ENV: AUGMENT=1 BETA_CLIP=-15:-2.3026 HIER=none LAM=na ETA_RATIO=na "
       "COS_TOTAL=default COS_WARMUP=default SCHED=none SCHED_TOTAL=none "
       "SCHED_WARMUP=none SCHED_MIN=none PROBE=0 PROBE_DIR=none EB_RHO=na EB_LOG=0")
os.makedirs(os.path.join(out, "cvk1"), exist_ok=True)
shutil.copy(man, os.path.join(out, "cvk1", "PARTITION-MANIFEST.txt"))
rng = random.Random(20260910)
jid = 9900000
made = []


def write(arm, s, jid):
    rn = "cvk1-%s-s%d" % (arm, s)
    args = ("--optimizer HF --alg-base SGDm --momentum-param-base 0.99 "
            "--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 "
            "--Lion-beta2-meta 0.9 --weight-decay-meta 0 --dataset CIFAR100 "
            "--NN-name VGG11_bn_c100 --batch-size 100 --max-time 999:00:00 --gamma 1 "
            "--meta-stepsize 1e-3 --alpha0 1e-6 --num-epochs 100 --stepsize-groups %s "
            "--seed %d --save-directory %s/runs/cvk1 --run-name %s"
            % (SPEC[arm], s, WS, rn))
    L = LEVEL[arm] + rng.uniform(-0.6, 0.6)
    lines = ["NODE=[HOST_45cba27b74] | JOB=%s %d | AUGMENT=1" % (rn, jid), "ARGS: " + args, ENV,
             "NVIDIA A100 80GB PCIe, 81920 MiB", "",
             "args_base includes unnecessary attributes {'normalizer_param'} ", "",
             "Files already downloaded and verified", "Files already downloaded and verified"]
    for e in range(100):
        te = L * (1 - math.exp(-(e + 1) / 12.0)) + rng.uniform(-0.3, 0.3)
        tr = te - 0.4 + rng.uniform(-0.3, 0.3)
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, tr, te))
    lines += ["27  minutes", "RUN_DONE"]
    fn = os.path.join(out, "%s-%d.out" % (rn, jid))
    open(fn, "w").write("\n".join(lines) + "\n")
    made.append(fn)


for s in (55, 56, 57):
    for arm in ARMS:
        jid += 1
        if drop == "%s-s%d" % (arm, s):
            continue
        write(arm, s, jid)
        if dup == "%s-s%d" % (arm, s):
            write(arm, s, jid + 1000)   # a later resubmission; the scorer keeps the higher job id
print("SYNTHETIC: %d .out files + manifest under %s (INVENTED NUMBERS)" % (len(made), out))
