#!/bin/bash
# =============================================================================
# cRT1_stage_harness.sh -- BUILD OR VERIFY THE ISOLATED `crt1` HARNESS TREE.
#   (CORRECTIONS 300; a copy of cGW1_stage_harness.sh with crt1 names -- the SAME pinned bytes.)  Touches NO existing file.  Submits nothing.  Applies NO patch.
#
# WHY AN ISOLATED TREE AND NOT THE LIVE ONE.  `crt1` needs no new code at all -- its axes are the
# existing `train.py --meta-stepsize` and `--alpha0` floats.  But a PENDING Slurm job imports whatever HF.py /
# train.py is on disk at the moment it STARTS, and other tracks share this workspace.  A byte copy
# freezes, for all 36 jobs, exactly the bytes the launcher's guards tested.
#
# WHAT IT BUILDS (under $METAOPT_WS, every path new and crt1-namespaced):
#   harness_crt1/cifar10/            a copy of the live harness's *.py files (train.py, load_data.py,
#                                    tin_data.py, build_network.py, Optimizers/*.py), BYTE-IDENTICAL
#                                    to the live tree, and each equal to the pinned sha below
#   harness_crt1/cifar10/data  ->    SYMLINK to the live tree's data/ (no copy)
#   jobs/run_cifar_crt1.sh           jobs/run_cifar.sh with EXACTLY ONE line changed: the `cd`
#   harness_crt1/STAGE-MANIFEST.txt  sha256 of every staged file
#
# USAGE
#   bash bin/cRT1_stage_harness.sh            # VERIFY ONLY (exit 0 iff consistent)
#   bash bin/cRT1_stage_harness.sh --stage    # build it (refuses if it exists), then verify
# =============================================================================
set -u
MODE=verify; [ "${1:-}" = "--stage" ] && MODE=stage
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HROOT=$WS/harness_crt1
DEST=$HROOT/cifar10
RUNNER_SRC=$WS/jobs/run_cifar.sh
RUNNER=$WS/jobs/run_cifar_crt1.sh
# the pinned bytes (== analysis/crt1_design.py; the launcher's guard 4c' checks the two agree)
HF_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
TRAIN_SHA=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
BUILDOPT_SHA=25a899b3745e9d66fbf630795a1202c6075e2067daffa077e4a54c28e54c7ec2
LOADDATA_SHA=b52b58a35dcecd80ce6d8f8033b09c20dad9f4827b808f4ea2a828ce88138c46
TINDATA_SHA=9e0f3322e5025e966aa18e6ac39c1d1b347c9ce2bf6e50423f301e965a6eecb7
RUNNER_SRC_SHA=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
FAIL=0; bad() { echo "!!! STAGE FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

live_py() { (cd "$LIVE" && find . -name '*.py' -not -path '*/__pycache__/*' -not -path './data/*' | sort); }

echo "cRT1_stage_harness: mode=$MODE  WS=$WS"
echo "  live tree   $LIVE"
echo "  crt1 tree   $DEST"
echo "  runner      $RUNNER  (from $RUNNER_SRC)"

if [ "$MODE" = stage ]; then
  [ -e "$HROOT" ] && { echo "!!! $HROOT already exists -- refusing to overwrite. Verify it instead."; exit 2; }
  [ -e "$RUNNER" ] && { echo "!!! $RUNNER already exists -- refusing to overwrite."; exit 2; }
  for pr in "Optimizers/HF.py $HF_SHA" "train.py $TRAIN_SHA" "build_network.py $BN_SHA" \
            "Optimizers/build_optimizer.py $BUILDOPT_SHA" "load_data.py $LOADDATA_SHA" "tin_data.py $TINDATA_SHA"; do
    set -- $pr
    [ "$(sha "$LIVE/$1")" = "$2" ] || { echo "!!! live $1 is $(sha "$LIVE/$1"), not the pinned $2 -- refusing to stage"; exit 2; }
  done
  [ "$(sha "$RUNNER_SRC")" = "$RUNNER_SRC_SHA" ] || { echo "!!! $RUNNER_SRC moved from the pinned $RUNNER_SRC_SHA"; exit 2; }
  mkdir -p "$DEST" || exit 2
  for f in $(live_py); do
    mkdir -p "$DEST/$(dirname "$f")"; cp -p "$LIVE/$f" "$DEST/$f" || exit 2
  done
  ln -s "$LIVE/data" "$DEST/data" || exit 2
  sed "s#^cd $LIVE\$#cd $DEST#" "$RUNNER_SRC" > "$RUNNER" || exit 2
  chmod --reference="$RUNNER_SRC" "$RUNNER" 2>/dev/null || chmod 755 "$RUNNER"
  { echo "STAGED_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "LIVE $LIVE"; echo "DEST $DEST"
    for f in $(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' | sort); do
      echo "FILE $(sha "$DEST/$f") $f"; done
    echo "RUNNER $(sha "$RUNNER") $RUNNER"
    echo "RUNNER_SRC $(sha "$RUNNER_SRC") $RUNNER_SRC"; } > "$HROOT/STAGE-MANIFEST.txt"
  echo "staged; now verifying"
fi

# ---------------------------------------------------------------- VERIFY ------
[ -d "$DEST" ] || { echo "!!! no crt1 tree at $DEST -- run with --stage"; exit 2; }
# V-h: the six harness files carry the PINNED bytes (this binds even if the live tree later moves)
for pr in "Optimizers/HF.py $HF_SHA" "train.py $TRAIN_SHA" "build_network.py $BN_SHA" \
          "Optimizers/build_optimizer.py $BUILDOPT_SHA" "load_data.py $LOADDATA_SHA" "tin_data.py $TINDATA_SHA"; do
  set -- $pr
  [ "$(sha "$DEST/$1")" = "$2" ] && echo "V-h: $1 == pinned $(echo "$2" | cut -c1-16)" || bad "V-h $1 is $(sha "$DEST/$1"), not the pinned $2"
done
# V-a: every staged .py matches the manifest written at stage time (no edit since)
N=0
while read -r tag s f; do
  [ "$tag" = FILE ] || continue
  case "$f" in *.py) ;; *) continue ;; esac
  [ "$(sha "$DEST/$f")" = "$s" ] && N=$((N+1)) || bad "V-a $f changed since staging"
done < "$HROOT/STAGE-MANIFEST.txt"
echo "V-a: $N staged .py files unchanged since staging (STAGE-MANIFEST.txt)"
# V-a1: vs the LIVE tree (a NOTE only if the live tree moved; V-h is what binds)
M=0; MOV=""
for f in $(live_py); do
  [ -f "$DEST/$f" ] || { MOV="$MOV $f(new-in-live)"; continue; }
  [ "$(sha "$LIVE/$f")" = "$(sha "$DEST/$f")" ] && M=$((M+1)) || MOV="$MOV $f"
done
[ -z "$MOV" ] && echo "V-a1: all $M live harness .py files byte-identical in the crt1 tree" \
              || echo "V-a1: NOTE -- the live tree has moved since staging:$MOV (not a failure; the crt1 tree keeps the pinned bytes)"
# V-a2: no PATCH marker whose gate could fire, and no .py the stage did not copy
EXTRA=$(cd "$DEST" && find . -name '*.py' -not -path '*/__pycache__/*' -not -path './data/*' | sort | while read -r f; do grep -q " $f\$" "$HROOT/STAGE-MANIFEST.txt" || echo "$f"; done)
[ -z "$EXTRA" ] && echo "V-a2: no .py in the crt1 tree that the stage did not copy" || bad "V-a2 extra .py: $EXTRA"
for mk in PATCH_BETAHOLD PATCH_GROUPHOLD PATCH_RESTHOLD PATCH_VOTEWEIGHT PATCH_DECAYMASK; do
  grep -q "$mk" "$DEST/Optimizers/HF.py" && bad "V-a3 HF.py carries $mk" || true
done
echo "V-a3: HF.py carries none of PATCH_BETAHOLD / GROUPHOLD / RESTHOLD / VOTEWEIGHT / DECAYMASK"
# V-c: data is a symlink to the live data, CIFAR-10 visible
[ -L "$DEST/data" ] && [ "$(readlink "$DEST/data")" = "$LIVE/data" ] && echo "V-c: data -> $LIVE/data (symlink)" || bad "V-c data is not a symlink to $LIVE/data"
[ -d "$DEST/data/cifar-10-batches-py" ] && echo "V-c2: cifar-10-batches-py visible through the symlink" || bad "V-c2 cifar-10-batches-py not visible"
# V-d: the runner differs from the PINNED run_cifar.sh in EXACTLY the cd line
[ "$(sha "$RUNNER_SRC")" = "$RUNNER_SRC_SHA" ] && echo "V-d0: $RUNNER_SRC == pinned $(echo $RUNNER_SRC_SHA | cut -c1-16)" || bad "V-d0 $RUNNER_SRC moved"
if [ -f "$RUNNER" ]; then
  D=$(diff "$RUNNER_SRC" "$RUNNER" | grep -c '^[<>]')
  DL=$(diff "$RUNNER_SRC" "$RUNNER" | grep '^[<>]')
  if [ "$D" = 2 ] && echo "$DL" | grep -qx "< cd $LIVE" && echo "$DL" | grep -qx "> cd $DEST"; then
    echo "V-d: $RUNNER differs from run_cifar.sh in EXACTLY one line: cd $LIVE -> cd $DEST"
  else
    bad "V-d the runner diff is not exactly the cd line:"; diff "$RUNNER_SRC" "$RUNNER"
  fi
  grep -q "^RUNNER $(sha "$RUNNER") $RUNNER\$" "$HROOT/STAGE-MANIFEST.txt" && echo "V-d2: runner unchanged since staging ($(sha "$RUNNER" | cut -c1-16))" || bad "V-d2 runner changed since staging"
else
  bad "V-d no runner at $RUNNER"
fi
[ "$FAIL" = 0 ] && { echo "cRT1_stage_harness: VERIFIED"; exit 0; } || { echo "cRT1_stage_harness: NOT CONSISTENT"; exit 2; }
