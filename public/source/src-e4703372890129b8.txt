def _sem(v):
    return statistics.stdev(v) / math.sqrt(len(v)) if len(v) > 1 else float("nan")


def _t(a, b):
    se = math.sqrt(_sem(a) ** 2 + _sem(b) ** 2)
    if not (se > 0):
        return float("nan")
    return (statistics.mean(a) - statistics.mean(b)) / se


def band(d):
    """THE DESCRIPTIVE BAND ON THE POINT ESTIMATE.  Symmetric on |D|.
    **THIS IS NO LONGER THE STOPPING RULE.**  See resolve()."""
    if d >= D_GO:
        return "PROCEED"
    if d > D_NULL:
        return "SCREEN-NEGATIVE"
    if d >= -D_NULL:
        return "NO-AMPLIFICATION"
    if d > -D_GO:
        return "SCREEN-NEGATIVE"
    return "PROCEED-REVERSAL"


def resolve(d, se, sd_arms=None):
    """**THE STOPPING RULE, REWRITTEN AT CLOSEOUT.  IT RESOLVES ON THE REALISED
    STATISTIC, NOT ON A FIXED THRESHOLD.**

    The original rule was: |D| >= 1.0 pp -> PROCEED, else CLOSE AS UNAFFORDABLE.
    That rule classified the campaign's OWN point prior for D (+0.5805 +- 0.0939)
    as a NEGATIVE -- its modal outcome was a false close.  Worse, the 1.0 pp
    threshold was derived from a '14x amplification' prior computed as layerwise
    minus scalar on CIFAR-100, where the scalar arm reads 22.6% on a 100-class
    task: a FAILED ARM, not an amplified granularity effect, and silent about a
    count-matched contrast between two fine partitions that both sit near 71%.

    The replacement asks the only question a screen can honestly answer -- did we
    measure anything? -- in this order:

      REGIME FAILURE   either arm's within-arm sd exceeds SD_REGIME, or an arm has
                       a collapsed/truncated seed.  -> CLOSE-UNRESOLVED.  NOT a
                       negative: at ms=1e-4 no CIFAR-100 data exists, so both arms
                       landing in a broken regime is a live possibility and it
                       must not be read as "no effect".
      CLOSE-CONFIRMED  |D| separated from zero at |t| >= T_CONFIRM.  The tail
                       prescription is present on a second dataset.
      PROCEED          D unresolved but the REALISED se is under SE_AFFORD, so a
                       powered cell is worth buying.
      CLOSE-UNAFFORDABLE  the realised se EXCEEDS SE_AFFORD.  This is the ONLY
                       branch that closes CIFAR-100, and it closes it on measured
                       noise, never on a small point estimate."""
    if sd_arms is not None and max(sd_arms) > SD_REGIME:
        return "CLOSE-UNRESOLVED (REGIME FAILURE)"
    if se and se > 0 and abs(d) / se >= T_CONFIRM:
        return "CLOSE-CONFIRMED"
    if se and se <= SE_AFFORD:
        return "PROCEED"
    return "CLOSE-UNAFFORDABLE"


def proceeds(d):
    """The OLD fixed-threshold rule, RETAINED ONLY so the selftest can assert that
    it is no longer what decides the batch.  Do not call it for a verdict."""
    return abs(d) >= D_GO


def plateaus(csv_path, prefix):
    """plateau5 by seed for a gc1 arm.  Validity-filtered per S0."""
    out = {}
    with open(csv_path) as fh:
        for r in csv.DictReader(fh):
            if not r["run"].startswith(prefix):
                continue
            if int(r["epochs_done"] or 0) != EPOCHS:
                continue
            if r.get("epochs_requested") and int(r["epochs_requested"]) != EPOCHS:
                continue
            if r.get("collapsed") not in ("", "0"):
                continue
            if not r["plateau5"]:
                continue
            out[int(r["seed"])] = float(r["plateau5"])
    return out


def comparator(csv_path):
    """S2's cell.  Read by RUN NAME, so it can never be silently pooled with gc1."""
    out = []
    with open(csv_path) as fh:
        for r in csv.DictReader(fh):
            if not r["run"].startswith(COMPARATOR_PREFIX):
                continue
            if int(r["epochs_done"] or 0) != EPOCHS or not r["plateau5"]:
                continue
            out.append(float(r["plateau5"]))
    return sorted(out)


def arm_of_dir(d):
    parts = os.path.basename(d).split("_")
    return parts[1] if len(parts) > 2 and parts[0] == "probe" else None


def seed_of_dir(d):
    base = os.path.basename(d)
    tail = base.rsplit("_s", 1)
    try:
        return int(tail[1])
    except (IndexError, ValueError):
        return None
