set -e
module load Python/3.10.4-GCCcore-11.3.0
source /cluster/account_a/metaopt/envs/mo/bin/activate
cd /cluster/account_a/metaopt/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10

nvidia-smi --query-gpu=name,memory.total --format=csv,noheader
