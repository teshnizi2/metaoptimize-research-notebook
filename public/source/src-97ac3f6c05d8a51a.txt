#!/bin/bash
# =============================================================================
# cIS1_isolate_carriers.sh -- `ciso1`, H-ISOLATE: THE CAUSAL TEST OF ctd1's
#   THREE-TENSOR DOMINATION.  ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds),
#   100 EPOCHS, SEEDS {21,22,23}, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account, nothing is submitted to `cluster_a`, and every guard
# and scancel line below is filtered on the `ciso1-` prefix alone.
#
# REGISTERED SCORER: analysis/cIS1_ciso1_isolate_score.py, git-committed
# BEFORE this batch is submitted (STANDING RULE 21; the proof is the commit
# time against the earliest sacct Submit, stated in CORRECTIONS with the
# margin).  The hypothesis, every bar (in pp AND in SE from a noise floor
# re-derived at registration with ciso1's own rows excluded from every
# reader), the partitions, the branch map and the point predictions are
# frozen there; --selftest re-derives every corpus premise and, on this host,
# the ctd1 premise from the probe records.
#
# THE DOCUMENTED SCORER INVOCATION -- no second argument to forget:
#     python3 analysis/cIS1_ciso1_isolate_score.py <runsdir>
#
# -----------------------------------------------------------------------------
# THE ARMS (m = 2 group ORDER: the big group is beta[0], the isolated group
# beta[1]).  1-BASED indices, verified BY NAME on the live model by guard 4
# below, which prints the full manifest per arm.
#   k01   scalar                                       floor anchor, in batch
#   k62   layerwise                                    ceiling anchor, in batch
#   ISO   sets:<59 others>/50,53,59  [59,3]            layer4.0.bn2.weight,
#                                                      layer4.0.shortcut.1.weight,
#                                                      layer4.1.bn2.weight
#   CTRL  sets:<59 others>/20,23,29  [59,3]            layer2.0.bn2.weight,
#                                                      layer2.0.shortcut.1.weight,
#                                                      layer2.1.bn2.weight
#         -- the carriers' STRUCTURAL HOMOLOGUES two stages earlier (block-0
#         bn2 scale, block-0 shortcut BN scale, block-1 bn2 scale).  Same class,
#         same count, same roles, same group sizes, measured non-involvement on
#         ctd1 (carrier share 0.0000/0.0000/0.0008).  NOT numel-matched (128 vs
#         512): a numel-matched non-carrier triple does not exist on this model
#         (512-wide BN scales live only in layer4, which has TWO non-carriers).
#   ONE   sets:<61 others>/50        [61,1]            layer4.0.bn2.weight alone
#         -- the dose arm: on ctd1's pinned records the trio dominates the rest
#         on 0.98 but 50 alone and the pair {53,59} each on 0.0000, so a static
#         reading says ONE should also rescue; the corpus prefix ladder (k50
#         30.34, k53 22.08) says the collapse builds up one carrier at a time.
#         3 jobs, ~2.6 GPU-h, decides between those two readings.
#
# The isolated group is written by NAME and the big group by INDEX RANGES, so
# an off-by-one in either half is a duplicate or a gap and PATCH_NAMESETS raises
# at parse time (guard 4 composes every arm on the live model before anything
# is submitted).
#
# RULE 20.  Every value below is ONE shell token of [A-Za-z0-9_.,:/-].
# analysis/argsline_guard.py IS NOT EDITED.  BETA_CLIP, PROBE and PROBE_TENSOR
# are ENVIRONMENT variables; the first two are audited from the runner's ENV
# line, PROBE_TENSOR from the `PROBE_TENSOR: on` line the patch prints.
#
# PARTITIONS: composed by bin/_lib_guards.sh's slurm_parts_for_wall
# (CORRECTIONS 178); gpu-short is included at WALL=03:00:00 and the source-time
# hook aborts if PARTS/WALL disagree.  WALL is 1.9x ctd1's slowest run (95 min
# on a 2080ti node).
#
# USAGE   bash bin/cIS1_isolate_carriers.sh            # dry run, submits nothing
#         bash bin/cIS1_isolate_carriers.sh --submit
# =============================================================================
set -u

USER_NAME="${USER:-$(whoami)}"

REPO="$(cd "$(dirname "$0")/.." && pwd)"

HFLIVE="$CIF/Optimizers/HF.py"

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1

NET=ResNet18_c100
SEEDS="21 22 23"
MST=1e-3
A0=1e-6
NTENS=62

# ---- the registered specs, LITERAL.  The scorer holds the same five strings;
# ---- guard 4c proves the two agree so neither can drift.
G_K01='scalar'
G_K62='layerwise'
G_ISO='sets:1-49,51-52,54-58,60-62/layer4.0.bn2.weight,layer4.0.shortcut.1.weight,layer4.1.bn2.weight'
G_CTRL='sets:1-19,21-22,24-28,30-62/layer2.0.bn2.weight,layer2.0.shortcut.1.weight,layer2.1.bn2.weight'
G_ONE='sets:1-49,51-62/layer4.0.bn2.weight'
ARMS="k01 k62 ISO CTRL ONE"
WALL=03:00:00
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

echo "=============================================================="
echo " ciso1 -- H-ISOLATE, THE CAUSAL TEST, STANDARD CIFAR-100 CELL"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01  $G_K01"
echo "   k62  $G_K62"
echo "   ISO  $G_ISO"
echo "   CTRL $G_CTRL"
echo "   ONE  $G_ONE"
echo " 5 arms x 3 seeds = 15 jobs, ONE submission"
echo " partitions $PARTS   walltime $WALL"
echo " scorer: analysis/cIS1_ciso1_isolate_score.py (registered before any run exists)"
echo " DOCUMENTED INVOCATION: python3 analysis/cIS1_ciso1_isolate_score.py <runsdir>"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
CSVP="$REPO/results/all_runs.csv"
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV at $CSVP"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- guard 1: the registered scorer exists, is committed, and selftests -----
SC="$REPO/analysis/cIS1_ciso1_isolate_score.py"
PATCHN="$REPO/patches/patch_namesets.py"
TESTN="$REPO/tests/test_namesets.py"
PATCHT="$REPO/patches/patch_probe_tensor.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTB="$REPO/tests/test_probe_tensor_blockwise.py"
for f in "$SC" "$PATCHN" "$TESTN" "$PATCHT" "$TESTT" "$TESTB"; do
  [ -f "$f" ] || guard_fail "guard 1a: missing $f"
done
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  DIRTY=$(git -C "$REPO" status --porcelain -- \
      analysis/cIS1_ciso1_isolate_score.py bin/cIS1_isolate_carriers.sh \
      tests/test_probe_tensor_blockwise.py patches/patch_namesets.py tests/test_namesets.py \
      patches/patch_probe_tensor.py tests/test_probe_tensor.py 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: scorer, patches, tests AND this script are committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires them committed BEFORE any run exists: $DIRTY"
  DEL=$(git -C "$REPO" diff HEAD --numstat -- analysis/ 2>/dev/null | awk '$2+0>0{print}')
  [ -z "$DEL" ] && echo "guard 1b2: no uncommitted DELETIONS under analysis/ (RULE 16)" \
               || guard_fail "guard 1b2: analysis/ has uncommitted deletions: $DEL"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 is verified on"
  echo "          the committing host by wall clock against the earliest sacct Submit."
fi
python3 "$SC" --selftest --runsdir "$WS/runs" >/tmp/ciso1_selftest.$$ 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' /tmp/ciso1_selftest.$$) checks, $(grep -c '^  FAIL' /tmp/ciso1_selftest.$$) failures, $(grep -c '^  SKIP' /tmp/ciso1_selftest.$$) skipped)" \
  || { guard_fail "guard 1c: scorer selftest FAILED"; tail -40 /tmp/ciso1_selftest.$$; }
grep -q 'ctd1 passes the in-batch PREMISE rule' /tmp/ciso1_selftest.$$ && grep -q '^  PASS ctd1 pinned: tensor 50 alone' /tmp/ciso1_selftest.$$ \
  && echo "guard 1c2: the ctd1 PREMISE section ran on this host's probe records (not skipped)" \
  || guard_fail "guard 1c2: the ctd1 PREMISE section did not run -- ctd1 probes not reachable under $WS/runs/ctd1"
echo "guard 1c: scorer sha256 $( (sha256sum "$SC" 2>/dev/null || shasum -a 256 "$SC") | awk '{print $1}')"

# ---- guard 1d: PATCH_NAMESETS re-proved FUNCTIONALLY on the LIVE tree ---------
# The registered test's N0 demands byte-identity between --pre and --post minus
# the two namesets regions.  HF.py.pre_namesets predates PATCH_REDNORM and
# PATCH_PROBE_TENSOR, so N0 cannot pass against it any more.  DISCLOSED: --pre
# is built HERE as the live file with the two namesets regions deleted by the
# test's own recipe, so N0 is trivially true and the load-bearing checks are
# N1-N3 (backward compatibility over the corpus specs, sets:1-49/50-62 ==
# [49,13] exactly, init_meta end to end) on the LIVE code.  Neither test file
# is edited.
if [ -f "$TESTN" ] && [ -f "$HFLIVE" ]; then
  PRE_PY=/tmp/ciso1_hf_live_minus_namesets.$$.py
  python3 - "$HFLIVE" "$PRE_PY" <<'PYEOF' || guard_fail "guard 1d: could not strip the namesets regions from the live file"
import sys
src = open(sys.argv[1]).read()
i = src.find("# --- PATCH_NAMESETS: `sets:")
j = src.find("class HF():\n")
assert 0 <= i < j, (i, j)
s = src[:i] + src[j:]
ins2 = ("        # --- PATCH_NAMESETS: the ONLY new branch.  Every pre-existing spec form\n"
        "        # fails this test and reaches the original body byte-unchanged. ---\n"
        "        if isinstance(stepsize_groups, str) and stepsize_groups.startswith(_NS_PREFIX):\n"
        "            return _namesets_parse(stepsize_groups, net_param_names_and_size)\n")
assert s.count(ins2) == 1
s = s.replace(ins2, "", 1)
assert "PATCH_NAMESETS" not in s
open(sys.argv[2], "w").write(s)
print("guard 1d: --pre := live minus the two namesets regions (%d -> %d bytes)" % (len(src), len(s)))
PYEOF
  if python3 "$TESTN" --pre "$PRE_PY" --post "$HFLIVE" --cifar-dir "$CIF" \
       --argsline-guard "$REPO/analysis/argsline_guard.py" >/tmp/ciso1_namesets.$$ 2>&1; then
    echo "guard 1d: tests/test_namesets.py PASSES on the LIVE tree (N1-N3 load-bearing, N0 by construction)"
    tail -3 /tmp/ciso1_namesets.$$ | sed 's/^/          /'
  else
    guard_fail "guard 1d: tests/test_namesets.py FAILED on the LIVE tree"; tail -30 /tmp/ciso1_namesets.$$
  fi
  rm -f "$PRE_PY"
else
  guard_fail "guard 1d: cannot run the namesets proof -- $TESTN or $HFLIVE missing"
fi

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ---------
grep -q '^ciso1-' "$CSVP" && guard_fail "guard 2: ciso1-* already in the CSV" \
                          || echo "guard 2: no ciso1-* name collision in the corpus"
if [ -d "$WS/runs" ]; then
  NOUT=$(ls "$WS/runs"/ciso1-*.out 2>/dev/null | wc -l | tr -d ' ')
  [ "${NOUT:-0}" = 0 ] && echo "guard 2b: no ciso1-*.out on disk" \
                       || guard_fail "guard 2b: $NOUT ciso1-*.out files already exist"
fi
NDUP=$(python3 - "$CSVP" "$NET" <<'PYEOF'
import sys, csv
n = 0
for r in csv.DictReader(open(sys.argv[1])):
    if r.get("network") == sys.argv[2] and str(r.get("seed")) in ("21", "22", "23"):
        n += 1
print(n)
PYEOF
)
[ "${NDUP:-1}" = 0 ] && echo "guard 2c: ZERO $NET rows anywhere carry seed 21, 22 or 23" \
                     || guard_fail "guard 2c: $NDUP $NET rows already carry seed 21/22/23"

# ---- guard 3: the CIFAR-100 data is staged on THIS account ------------------
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3: cifar-100-python present under $CIF/data" \
  || guard_fail "guard 3: CIFAR-100 is not staged at $CIF/data/cifar-100-python"

# ---- guard 4a-e: THE LIVE SOURCE CARRIES BOTH PATCHES, PROVED INERT ---------
mkdir -p "$SAVE"
[ -f "$HFLIVE" ] || guard_fail "guard 4: no live HF.py at $HFLIVE"
HFSHA=$( (sha256sum "$HFLIVE" 2>/dev/null || shasum -a 256 "$HFLIVE") | awk '{print $1}')
grep -q 'PATCH_NAMESETS' "$HFLIVE" && echo "guard 4a: live HF.py carries PATCH_NAMESETS" \
                                   || guard_fail "guard 4a: live HF.py lacks PATCH_NAMESETS -- apply patches/patch_namesets.py first"
NPT=$(grep -c 'PATCH_PROBE_TENSOR' "$HFLIVE" 2>/dev/null); NPT=${NPT:-0}
[ "$NPT" = 3 ] && echo "guard 4a: live HF.py carries PATCH_PROBE_TENSOR (3 markers)" \
               || guard_fail "guard 4a: live HF.py has $NPT PATCH_PROBE_TENSOR markers, the patch inserts exactly 3"
PRE="$HFLIVE.pre_probe_tensor"
[ -f "$PRE" ] && echo "guard 4b: pre-patch backup present: $PRE" || guard_fail "guard 4b: no pre-patch backup at $PRE"
python3 "$TESTT" --pre "$PRE" --post "$HFLIVE" --structural-only >/tmp/ciso1_struct.$$ 2>&1 \
   && echo "guard 4c: structural inertness R0 PASSES on the live file" \
   || { guard_fail "guard 4c: structural inertness R0 FAILED on the live file"; tail -20 /tmp/ciso1_struct.$$; }
ILOG="$WS/runs/ctd1/inertness_test.log"
if [ -f "$ILOG" ] && grep -q '^ALL PASS$' "$ILOG" && grep -q "^POST_SHA256 $HFSHA" "$ILOG"; then
  echo "guard 4d: ctd1's CPU inertness test (R0-R5, scalar+layerwise) ALL PASS on THIS live sha ($ILOG)"
else
  guard_fail "guard 4d: no ALL PASS inertness log for sha $HFSHA at $ILOG"
fi
BLOG="$SAVE/inertness_blockwise.log"
if [ -f "$BLOG" ] && grep -q '^ALL PASS$' "$BLOG" && grep -q "^POST_SHA256 $HFSHA" "$BLOG"; then
  echo "guard 4e: the BLOCKWISE inertness test (B0-B5 on the three sets: arms) ALL PASS on THIS live sha ($BLOG)"
  for G in "$G_ISO" "$G_CTRL" "$G_ONE"; do
    grep -qF -- "$G" "$BLOG" || guard_fail "guard 4e: $BLOG does not mention the spec $G"
  done
else
  guard_fail "guard 4e: no ALL PASS blockwise inertness log for sha $HFSHA at $BLOG -- run:
    { echo POST_SHA256 $HFSHA; python3 $TESTB --pre $PRE --post $HFLIVE --cifar-dir $CIF --work /tmp/ciso1_bw; } 2>&1 | tee $BLOG"
fi

# ---- guard 4c': the literal specs here and in the scorer agree --------------
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"
MANIFEST2="$WS/runs/ciso1-PARTITION-MANIFEST.txt"
python3 - "$SC" "$G_K01" "$G_K62" "$G_ISO" "$G_CTRL" "$G_ONE" <<'PYEOF' \
  || guard_fail "guard 4c': the launcher's specs and the scorer's SPEC table disagree"
import sys, re, ast
sc = open(sys.argv[1]).read()
mine = dict(zip(("k01", "k62", "ISO", "CTRL", "ONE"), sys.argv[2:7]))
m = re.search(r"^SPEC = (\{.*?^\})", sc, re.S | re.M)
if not m:
    print("!!! guard 4c': could not find SPEC in the scorer"); raise SystemExit(1)
theirs = ast.literal_eval(m.group(1))
bad = 0
for k in mine:
    if mine[k] != theirs.get(k):
        print("!!! guard 4c': %s launcher=%r scorer=%r" % (k, mine[k], theirs.get(k))); bad += 1
if bad:
    raise SystemExit(1)
print("guard 4c': all 5 spec strings are byte-identical in the launcher and the scorer")
PYEOF

# ---- guard 4: THE LIVE-MODEL MANIFEST, printed per arm with tensor NAMES ----
python3 - "$CIF" "$MANIFEST" "$NTENS" "$TOTPAR" "$CLIP_C" "$EPOCHS" "$BATCH" \
         "$G_ISO" "$G_CTRL" "$G_ONE" <<'PYEOF' \
  || guard_fail "guard 4: live partition / design / source check FAILED"
import sys, os, inspect, re
cif, out, ntens, totpar = sys.argv[1], sys.argv[2], int(sys.argv[3]), int(sys.argv[4])
clip_c, epochs, bsz = sys.argv[5], int(sys.argv[6]), int(sys.argv[7])
specs = dict(zip(("ISO", "CTRL", "ONE"), sys.argv[8:11]))
sys.path.insert(0, cif); os.chdir(cif)
import torch
from build_network import build_network
from Optimizers.HF import HF
net = build_network("ResNet18_c100", "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]
numel = [int(p.numel()) for _, p in net.named_parameters()]
T, tot = len(nps), sum(numel)
bad = 0
print("guard 4: live ResNet18_c100 -- %d parameter tensors, %d parameters" % (T, tot))
if T != ntens or tot != totpar:
    print("!!! guard 4: live model is %d / %d, the batch assumes %d / %d" % (T, tot, ntens, totpar)); raise SystemExit(1)
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot, "CLIP_C %s" % clip_c, "EPOCHS %d" % epochs,
         "META_STEPS %d" % (epochs * (50000 // bsz))]
for i in range(T):
    lines.append("TENSOR %d %s %d" % (i + 1, names[i], numel[i]))
CAR = ("layer4.0.bn2.weight", "layer4.0.shortcut.1.weight", "layer4.1.bn2.weight")
CTL = ("layer2.0.bn2.weight", "layer2.0.shortcut.1.weight", "layer2.1.bn2.weight")
want_iso = {"ISO": (50, 53, 59), "CTRL": (20, 23, 29), "ONE": (50,)}
want_sizes = {"ISO": [59, 3], "CTRL": [59, 3], "ONE": [61, 1]}
print("guard 4: THE CARRIERS BY NAME ON THE LIVE MODEL (1-based):")
for n in CAR + CTL:
    print("         %2d  %-30s %5d params" % (names.index(n) + 1, n, numel[names.index(n)]))
if tuple(names.index(n) + 1 for n in CAR) != (50, 53, 59):
    print("!!! guard 4: the carriers are not at 50/53/59 on the live model"); bad += 1
if tuple(names.index(n) + 1 for n in CTL) != (20, 23, 29):
    print("!!! guard 4: the controls are not at 20/23/29 on the live model"); bad += 1
groups = {}
for arm in ("ISO", "CTRL", "ONE"):
    g = HF.polish_the_stepsize_groups(None, specs[arm], nps)
    sizes = [len(x) for x in g]
    idx1 = tuple(names.index(n) + 1 for n in g[1])
    pc = [sum(numel[names.index(n)] for n in x) for x in g]
    print("guard 4: %-4s %s" % (arm, specs[arm]))
    print("         m=%d  sizes %s  params %s  isolated (1-based) %s" % (len(g), sizes, pc, idx1))
    print("         GROUP0 (beta[0], %d tensors): %s" % (len(g[0]), " ".join(g[0])))
    print("         GROUP1 (beta[1], %d tensors): %s" % (len(g[1]), " ".join(g[1])))
    if len(g) != 2 or sizes != want_sizes[arm] or idx1 != want_iso[arm] or sorted(g[0] + g[1]) != sorted(names):
        print("!!! guard 4: %s does not compose to the registered partition" % arm); bad += 1
    groups[arm] = g
    lines.append("ARMSPEC %s SPEC %s M %d SIZES %s PARAMS %s GROUP0 %s GROUP1 %s"
                 % (arm, specs[arm], len(g), ",".join(str(s) for s in sizes), ",".join(str(s) for s in pc),
                    " ".join(g[0]), " ".join(g[1])))
    lines.append("ISOLATED %s %s" % (arm, ",".join(str(i) for i in idx1)))
    # end to end through init_meta: routes to blockwise, beta has exactly 2 entries, no rednorm
    o = HF.__new__(HF); o.num_layers = T; o._device = torch.device("cpu")
    HF.init_meta(o, specs[arm], nps, 1e-6)
    shapes = [tuple(b.shape) for b in o.beta]
    ok = o.stepsize_type == "blockwise" and [len(x) for x in o.param_groups_indices] == sizes and shapes == [(2,)]
    print("         init_meta: stepsize_type %s  group sizes %s  beta shapes %s  rednorm %s"
          % (o.stepsize_type, [len(x) for x in o.param_groups_indices], shapes, getattr(o, "_rednorm", None)))
    if not ok or getattr(o, "_rednorm", False):
        print("!!! guard 4: %s does not route to a 2-coordinate blockwise beta without rednorm" % arm); bad += 1
sd = set(groups["ISO"][1]) ^ set(groups["CTRL"][1])
sd0 = set(groups["ISO"][0]) ^ set(groups["CTRL"][0])
if sd == set(CAR) | set(CTL) and sd0 == sd and len(groups["ISO"][1]) == len(groups["CTRL"][1]) == 3:
    print("guard 4d: ISO ^ CTRL == EXACTLY {%s} -- three names swapped at identical sizes [59,3]" % ", ".join(sorted(sd)))
else:
    print("!!! guard 4d: ISO ^ CTRL = %s" % sorted(sd)); bad += 1
if set(groups["ISO"][1]) - set(groups["ONE"][1]) == set(CAR[1:]) and groups["ONE"][1] == [CAR[0]]:
    print("guard 4d: ONE isolates layer4.0.bn2.weight alone; ISO adds layer4.0.shortcut.1.weight and layer4.1.bn2.weight")
else:
    print("!!! guard 4d: ONE/ISO relation broken"); bad += 1
lines.append("SYMDIFF ISO CTRL %s" % ",".join(sorted(sd)))
# ---- guard 4f: the update rule and the code paths, off the live source ------
src = inspect.getsource(HF.Lion_meta_update)
m = re.search(r"torch\.sign\(self\.args_meta\['Lion_beta2'\]\*self\.momentum_meta\[i\]\s*\+\s*"
              r"\(1-self\.args_meta\['Lion_beta2'\]\)\*HtT_gradft\[i\]\)", src)
print("guard 4f: Lion_meta_update subtracts ms*sign(b2*momentum_meta + (1-b2)*z): %s" % bool(m))
if not m: bad += 1
sbp = inspect.getsource(HF.block_product)
bw = "return [torch.tensor([sum([(u[i]*v[i]).sum() for i in group_indices]) for group_indices in self.param_groups_indices])]"
print("guard 4f: block_product's blockwise branch is the UNNORMALISED per-group sum: %s" % (bw in sbp))
if bw not in sbp: bad += 1
sim = inspect.getsource(HF.init_meta)
print("guard 4f: _rednorm is set only by a `tn:` prefix: %s" % ("startswith('tn:')" in sim))
if "startswith('tn:')" not in sim: bad += 1
sst = inspect.getsource(HF.step)
i_bp = sst.find("HtT_gradft = self.block_product(self.h_condenced, g)")
i_pt = sst.find("self._pt_capture(self.h_condenced, g, HtT_gradft)")
i_bu = sst.find("self.base_update(net,g)")
print("guard 4f: capture sits after block_product and before base_update: %s" % (0 <= i_bp < i_pt < i_bu))
if not (0 <= i_bp < i_pt < i_bu): bad += 1
if bad:
    raise SystemExit(1)
open(out, "w").write("\n".join(lines) + "\n")
