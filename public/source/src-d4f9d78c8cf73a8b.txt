class NoBiasNet(nn.Module):
    """A net whose parameters are ALL ndim >= 2 (V12's nodewise1d == nodewise check)."""

    def __init__(self):
        super().__init__()
        self.c1 = nn.Conv2d(3, 8, 3, padding=1, bias=False)
        self.c2 = nn.Conv2d(8, 16, 3, stride=2, padding=1, bias=False)
        self.c3 = nn.Conv2d(16, 16, 1, bias=False)
        self.fc = nn.Linear(16 * 16 * 16, 10, bias=False)

    def forward(self, x):
        x = torch.relu(self.c1(x))
        x = torch.relu(self.c2(x))
        x = torch.relu(self.c3(x))
        return self.fc(x.reshape(x.shape[0], -1))


class MixedRagged(nn.Module):

    V9 needs ragged final chunks on the ndim>=2 side AND one-D tensors on both sides of
    the selection, which the real ResNet18 has but is far too large to check explicitly.
    """

    def __init__(self):

        self.w1 = nn.Parameter(torch.arange(10 * 7, dtype=torch.float32).reshape(10, 7) * 0.01)

        self.b1 = nn.Parameter(torch.arange(6, dtype=torch.float32).clone())
        self.b2 = nn.Parameter(torch.arange(4, dtype=torch.float32).clone() * 0.5)

def make_opt(net, groups, meta_stepsize=None):
    meta = dict(META)
    if meta_stepsize is not None:
        meta["meta_stepsize"] = meta_stepsize
    return HF(net, stepsize_groups=groups, alpha0=ALPHA0, args_base=dict(BASE),
              args_meta=meta, gamma=1, writer=NullWriter())


def _beta_flat(opt):
    """beta as ONE flat vector, so partitions holding it in different list shapes are
    still comparable.  nodewise stores 62 length-shape[0] tensors and chunkvecns41 stores
    62 flat ones of the same lengths -- both flatten to the same thing when the
    partitions are equivalent."""
    return torch.cat([b.detach().float().reshape(-1).cpu() for b in opt.beta])


def run_steps_uncached(groups, k=STEPS, seed=0, meta_stepsize=None, net_factory=fresh_net):
    """Return (weights, flat beta).  BETA is where granularity actually lives, so the
    equivalences are asserted on it as well as on the weights."""
    net = net_factory(seed)
    opt = make_opt(net, groups, meta_stepsize)
    crit = nn.CrossEntropyLoss()
    g = torch.Generator().manual_seed(1234)
    for _ in range(k):
        x = torch.randn(8, 3, 32, 32, generator=g).to(DEV)
        y = torch.randint(0, 10, (8,), generator=g).to(DEV)
        loss = crit(net(x), y)
        opt.step(net, loss)
    return ([p.detach().float().cpu().clone() for p in net.parameters()],
            _beta_flat(opt))


_CACHE = {}


def run_steps(groups, k=STEPS, seed=0, meta_stepsize=None, net_factory=fresh_net):
    """MEMOISED run_steps.  V0 PROVES, before anything below runs, that the same
    configuration executed twice is BITWISE identical on this device; a cached result is
    therefore provably the same object a fresh run would produce, and re-executing 16
    duplicate ResNet18 configurations would buy nothing but wall-clock.  **V0 itself
    calls run_steps_uncached twice, by construction**, so the determinism claim is never
    served out of the cache it justifies."""
    key = (groups, k, seed, meta_stepsize, net_factory.__name__)
    if key not in _CACHE:
        _CACHE[key] = run_steps_uncached(groups, k, seed, meta_stepsize, net_factory)
    return _CACHE[key]


def compare(name, ra, rb, tol=1e-6, want_equal=True):
    """Compare BOTH the weights and the flat beta of two runs."""
    wa, ba = ra
    wb, bb = rb
    w_worst = max((x - y).abs().max().item() for x, y in zip(wa, wb))
    b_worst = (ba - bb).abs().max().item() if ba.numel() == bb.numel() else float("inf")
    worst = max(w_worst, b_worst)
    ok = (worst <= tol) if want_equal else (worst > tol)
    print(f"  {'PASS' if ok else 'FAIL'}  {name:56s} w={w_worst:.3e} beta={b_worst:.3e}")
    if not ok:
        FAILS.append(name)
    return ok


def compare_w(name, ra, rb, tol=1e-6, want_equal=True):
    """WEIGHTS ONLY.  Used where the two arms hold beta in genuinely different-sized
    partitions, so a beta comparison would be trivially satisfied by the size mismatch
    rather than by the dynamics."""
    w_worst = max((x - y).abs().max().item() for x, y in zip(ra[0], rb[0]))
    ok = (w_worst <= tol) if want_equal else (w_worst > tol)
    print(f"  {'PASS' if ok else 'FAIL'}  {name:56s} max|dW| = {w_worst:.3e}")
    if not ok:
        FAILS.append(name)
    return ok


def check(name, cond, detail=""):
    print(f"  {'PASS' if cond else 'FAIL'}  {name:56s} {detail}")
    if not cond:
        FAILS.append(name)
    return cond


def group_sizes(opt):
    """Every group's SIZE, per tensor, derived from the allocated chunk vector."""
    out = []
    for K, ne in zip(opt.cvec_K, opt.cvec_numel):
        c = (ne + K - 1) // K
        out.extend([K] * (c - 1) + [ne - K * (c - 1)])
    return out

     for p in net.parameters()]
v = [torch.ones_like(p) * 2.0 for p in net.parameters()]
got = opt.block_product(u, v)
check("chunk vector is per-tensor", list(opt.cvec_K) == [4, 4, 1, 4],
      f"got {list(opt.cvec_K)}, expected [4, 4, 1, 4] (w1,w2 at K2=4; b1 singletons; b2 one group)")
check("chunk counts", [int(t.numel()) for t in got] == [18, 4, 6, 1],
      f"got {[int(t.numel()) for t in got]}, expected [18, 4, 6, 1]")
ok_all = True
for li, (ui, vi) in enumerate(zip(u, v)):
    flat = (ui * vi).reshape(-1)
    n = flat.numel()
    K = opt.cvec_K[li]
    for ci in range(got[li].numel()):
        lo, hi = ci * K, min((ci + 1) * K, n)          # REAL members only
        want = float(flat[lo:hi].sum())
        have = float(got[li][ci])
        if abs(want - have) > 1e-5:
            ok_all = False
            print(f"      tensor {li} chunk {ci}: got {have}, explicit sum {want}")
check("every chunk == explicit sum over its REAL members", ok_all,
      "(a zero-pad must contribute 0, never rescale a ragged chunk)")

# ---- V9b: THE SAME EXPLICIT CHECK, ON THE REAL ResNet18, ON AN INTERIOR RUNG ----
# Added at cycle-84 review.  V1/V2/V4 use the SYMBOLIC ND specs 'n'/'w', and V3 is at
# k=0 (an already-existing arm), so before this block the combination an interior rung
# actually executes -- a NUMERIC K2, WITH injected one-D singletons, WITH ragged tails
# on the ndim>=2 side -- had no explicit block_product check on the real network at all.
# It was covered only by m-counting, the size census, the alpha map and divergence.
# A beta_to_alpha / block_product MISMATCH would be a SILENT SCIENTIFIC CORRUPTION, not
# a crash: beta index i would scale one set of weights while being driven by the
# gradient product of a different set.  So both directions are now checked against each
# other on the real net, not merely against the same cvec_K list.
print("\nV9b block_product on the REAL ResNet18 at an interior rung (chunkvec914s21)")
_net = fresh_net()
_opt = make_opt(_net, "chunkvec914s21")
_names = [n for n, _ in _net.named_parameters()]
_ps = [p for _, p in _net.named_parameters()]
_u = [torch.arange(p.numel(), dtype=torch.float32, device=DEV).view_as(p) * 1e-4 + 1.0
      for p in _ps]
_v = [torch.ones_like(p) * 0.5 for p in _ps]
_got = _opt.block_product(_u, _v)
# The hard cases, chosen deliberately: the smallest conv (ragged at K2=914), the
# largest conv (many whole chunks), a SELECTED one-D tensor (K = 1), and an UNSELECTED
# one-D tensor (K = numel, i.e. exactly one group).
_sel = {}
for i, (nm, p) in enumerate(zip(_names, _ps)):
    if nm == "conv1.weight":                       _sel["smallest conv (ragged)"] = i
    if p.numel() == max(x.numel() for x in _ps):   _sel["largest conv"] = i
for i, nm in enumerate(_names):
    if _ps[i].dim() == 1 and int(_opt.cvec_K[i]) == 1 and "SELECTED one-D (K=1)" not in _sel:
        _sel["SELECTED one-D (K=1)"] = i
    if _ps[i].dim() == 1 and int(_opt.cvec_K[i]) == _ps[i].numel() \
            and "UNSELECTED one-D (K=numel)" not in _sel:
        _sel["UNSELECTED one-D (K=numel)"] = i
check("all four hard cases exist on this rung", len(_sel) == 4,
      f"found {sorted(_sel)}")
_bad = []
for label, i in sorted(_sel.items()):
    flat = (_u[i] * _v[i]).reshape(-1)
    n = flat.numel()
    K = int(_opt.cvec_K[i])
    exp_chunks = (n + K - 1) // K
    if int(_got[i].numel()) != exp_chunks:
        _bad.append(f"{label}: {int(_got[i].numel())} chunks, expected {exp_chunks}")
        continue
    for ci in range(exp_chunks):
        lo, hi = ci * K, min((ci + 1) * K, n)      # REAL members only
        want = float(flat[lo:hi].sum())
        have = float(_got[i][ci])
        if abs(want - have) > max(1e-4, 1e-6 * abs(want)):
            _bad.append(f"{label} chunk {ci}: got {have}, explicit sum {want}")
            break
    check(f"{label}: {_names[i]} numel {n} K {K} -> {exp_chunks} chunks, each == its "
          f"explicit member sum", not any(label in b for b in _bad))
# The two directions must agree WITH EACH OTHER, not merely with cvec_K: the alpha map
# expands beta by repeat_interleave, so the number of groups it consumes per tensor
# must equal the number of chunks block_product produces per tensor.
_alpha_groups = [int((p.numel() + int(k) - 1) // int(k))
                 for p, k in zip(_ps, _opt.cvec_K)]
check("beta_to_alpha's group count == block_product's chunk count, PER TENSOR",
      _alpha_groups == [int(t.numel()) for t in _got],
      "a mismatch here would silently drive beta i from the wrong gradients")
check("no explicit-sum discrepancy anywhere on the interior rung", not _bad,
      "; ".join(_bad[:3]))
