#!/bin/bash
# =============================================================================
# cVT8_doseroute.sh -- `cvt8`: ResNet18_c100 -- IS cvt7's PARTIAL LOSS A DOSE EFFECT, AND IS IT DIRECT OR THROUGH THE
#   COMPLEMENT?  ONE SUBMISSION, 21 JOBS (7 arms x 3 seeds), 100 EPOCHS, SEEDS {102,103,104}, ResNet18_c100 / CIFAR-100,
#   ciso1's cell, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.  CORRECTIONS 248.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here touches the other account; every guard
# and scancel line is filtered on `cvt8-`.
#
# REGISTERED SCORER: analysis/cVT8_doseroute_score.py, committed BEFORE this batch is submitted (RULE 21).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT8_doseroute_score.py <runsdir>
#
# NEW HARNESS CODE, ISOLATED AND OPT-IN.  The batch runs from $WS/harness_cvt8/cifar10 = cvt7's tree (the LIVE ResNet
# harness + PATCH_VOTEWEIGHT + PATCH_BETAHOLD + PATCH_GROUPHOLD) + PATCH_RESTHOLD (patches/patch_resthold.py) + the replay
# file Optimizers/cvt8_isopath.json, built and verified by bin/cVT8_stage_harness.sh, and the runner
# $WS/jobs/run_cifar_cvt8.sh (run_cifar.sh with the cd line changed).  Guard 4f requires the inertness / bite log
# (tests/test_resthold.py + tests/test_resthold_realrun.py, bin/cVT8_realrun_inertness.sbatch) on THIS HF.py and scorer.
#
# THE ARMS (ISO's grouping except k01; <C> = layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight)
#   k01          scalar  GROUP_HOLD unset       REST_HOLD unset
#   ISO          ISO's   GROUP_HOLD unset       REST_HOLD unset
#   HOLDHIGH     ISO's   GROUP_HOLD=<C>:tri:8609 REST_HOLD unset
#   HOLDBIG      ISO's   GROUP_HOLD=<C>:tri:9428 REST_HOLD unset
#   HIGHISOPATH  ISO's   GROUP_HOLD=<C>:tri:8609 REST_HOLD=rec:cvt8_isopath
#   BIGISOPATH   ISO's   GROUP_HOLD=<C>:tri:9428 REST_HOLD=rec:cvt8_isopath
#   LOWISOPATH   ISO's   GROUP_HOLD=<C>:floor    REST_HOLD=rec:cvt8_isopath
#
# RULE 20.  Every ARGS value is ONE shell token; every held arm's ARGS line is ISO's except the run name.  GROUP_HOLD and
# REST_HOLD ride sbatch's --export list and CANNOT ride the runner's ENV line: the patches print their witness lines at
# construction on EVERY run, and those lines are what the separate ENV audit and the scorer's G-GHOLD / G-RHOLD read.
# analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  cvt7 by sacct: 49,896 s / 15 = 0.92 GPU-h per 100-epoch ResNet18_c100 run including [HOST_3c1b003035] (~0.71 off it).
# 21 jobs ~= 19.3 GPU-h expected (~15 off [HOST_3c1b003035]); HARD BOUND = WALL x 21 = 63 GPU-h (WALL 03:00:00; cvt7's slowest 5,650 s).
#
# USAGE   bash bin/cVT8_doseroute.sh               # dry run: every guard, every composed line printed; submits nothing
#         bash bin/cVT8_doseroute.sh --submit      # the ONLY submitting form
# ENV     CVT8_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
unset VOTE_W BETA_HOLD GROUP_HOLD COMP_HOLD REST_HOLD
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
H7=$WS/harness_cvt7/cifar10
HARN=$WS/harness_cvt8/cifar10
RUNNER=$WS/jobs/run_cifar_cvt8.sh
SAVE=$WS/runs/cvt8

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="102 103 104"
MST=1e-3
A0=1e-6
NTENS=62
TOTPAR=11220132
G_K01='scalar'
G_ISO='sets:1-49,51-52,54-58,60-62/layer4.0.bn2.weight,layer4.0.shortcut.1.weight,layer4.1.bn2.weight'
CARR='layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight'
GH_HIGH="$CARR:tri:8609"
GH_BIG="$CARR:tri:9428"
GH_LOW="$CARR:floor"
RH_ISO="rec:cvt8_isopath"
ARMS="k01 ISO HOLDHIGH HOLDBIG HIGHISOPATH BIGISOPATH LOWISOPATH"
NJOBS=21
WALL=03:00:00
LIVE_BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_PRE_SHA=2396f2be8102724c120ea5271d2d84acf3835ffb9a5c7947fc282121cf2a588f
HF_POST_SHA=5197dc2eecbbcb6feb815f3798b3455171e6ad614deac913be5df48dd9ede1a9
ISOPATH_SHA=08ab25f3a329cb260bb39fb72f3299c021e7169bf612fa27d539166296e28e70
RUNNER_SHA=1dbe4ff52eccbbb1c9936f4a687ed88172dd41c48b14ae60eb353b45f766f5e0
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT8_doseroute_score.py"
SELF="$REPO/bin/cVT8_doseroute.sh"
DERIV="$REPO/analysis/cvt8_registration_derivations.py"
DERIV4="$REPO/analysis/cvt4_registration_derivations.py"
DERIV6="$REPO/analysis/cvt6_registration_derivations.py"
DERIV7="$REPO/analysis/cvt7_registration_derivations.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cVT8_realrun_inertness.sbatch"
TESTR8="$REPO/tests/test_resthold.py"
TESTRR8="$REPO/tests/test_resthold_realrun.py"
STAGE8="$REPO/bin/cVT8_stage_harness.sh"
PATCHR="$REPO/patches/patch_resthold.py"
ISOJ="$REPO/patches/cvt8_isopath.json"
TESTJ="$REPO/patches/cvt8_testdown.json"
TESTG="$REPO/tests/test_grouphold.py"
TESTGR="$REPO/tests/test_grouphold_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
SC7="$REPO/analysis/cVT7_grouphold_score.py"
SC6="$REPO/analysis/cVT6_complementpath_score.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFRH="$HARN/Optimizers/HF.py.pre_resthold"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"
ILOG="$SAVE/inertness_resthold.log"

echo "=============================================================="
echo " cvt8 -- ResNet18_c100: dose x route on the carrier group  (PATCH_GROUPHOLD + PATCH_RESTHOLD)"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C  ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"
echo "   k01          $G_K01"
echo "   ISO          $G_ISO"
echo "   HOLDHIGH     ISO's  GROUP_HOLD=$GH_HIGH"
echo "   HOLDBIG      ISO's  GROUP_HOLD=$GH_BIG"
echo "   HIGHISOPATH  ISO's  GROUP_HOLD=$GH_HIGH  REST_HOLD=$RH_ISO"
echo "   BIGISOPATH   ISO's  GROUP_HOLD=$GH_BIG  REST_HOLD=$RH_ISO"
echo "   LOWISOPATH   ISO's  GROUP_HOLD=$GH_LOW  REST_HOLD=$RH_ISO"
echo " 7 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN   runner $RUNNER   partitions $PARTS   walltime $WALL"
echo " CO-PRIMARY: P_DOSE = HOLDHIGH - HOLDBIG; P_ROUTE = HIGHISOPATH - HOLDHIGH; P_ROUTE_BIG = BIGISOPATH - HOLDBIG"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT8_doseroute_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${GROUP_HOLD+x}" ] && [ -z "${COMP_HOLD+x}" ] && [ -z "${REST_HOLD+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD, GROUP_HOLD, COMP_HOLD and REST_HOLD are UNSET in the launcher's environment" \
  || guard_fail "guard 0b: an intervention switch is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" || guard_fail "guard 0: no CSV at $CSVP"

for f in "$SC" "$SELF" "$DERIV" "$DERIV4" "$DERIV6" "$DERIV7" "$EXCL" "$EXCLT" "$RRSB" "$TESTR8" "$TESTRR8" "$STAGE8" "$PATCHR" "$ISOJ" "$TESTJ" \
         "$TESTG" "$TESTGR" "$TESTT" "$SC7" "$SC6" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$HFRH" "$HFVW" "$HFPRE" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

NEWF="analysis/cVT8_doseroute_score.py bin/cVT8_doseroute.sh analysis/cvt8_registration_derivations.py bin/cVT8_realrun_inertness.sbatch tests/test_resthold.py tests/test_resthold_realrun.py patches/patch_resthold.py patches/cvt8_isopath.json patches/cvt8_testdown.json bin/cVT8_stage_harness.sh jobs/run_cifar_cvt8.sh"
REUSEF="analysis/cvt4_registration_derivations.py analysis/cvt6_registration_derivations.py analysis/cvt7_registration_derivations.py analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv tests/test_grouphold.py tests/test_grouphold_realrun.py tests/test_probe_tensor.py analysis/cVT7_grouphold_score.py analysis/cVT6_complementpath_score.py bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the eleven registration files and twelve reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT8_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT8_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT8_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT8_REGISTERED_COMMIT=<sha>"
  fi
fi
cmp -s "$RUNNER" "$REPO/jobs/run_cifar_cvt8.sh" && echo "guard 1b2: the archived jobs/run_cifar_cvt8.sh is BYTE-IDENTICAL to the runner the batch uses" \
  || guard_fail "guard 1b2: jobs/run_cifar_cvt8.sh in the repo differs from $RUNNER"

# ---- guard 1c: the scorer selftest, INCLUDING O2, RULE I, the real-record nulls and the invocation ----
SELFT=/tmp/cvt8_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION\|Traceback' "$SELFT" | head -20; tail -5 "$SELFT"; }
grep -qE "^  PASS O2 DOSE-FULL x DIRECT x BIG-DIRECT +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt8's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT8_doseroute_score.py <runsdir>. exits 0" "$SELFT" && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 102, 103 or 104" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cIS1's" "$SELFT" && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cDP1's" "$SELFT" \
  && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cIS2's" "$SELFT" && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cVT7's" "$SELFT" \
  && grep -q "^  PASS B cvt7's registered scorer: HOLDHIGH's string and witness" "$SELFT" && grep -q "^  PASS B cvt6's registered scorer: P_BIG" "$SELFT" \
  && echo "guard 1c5: ISO byte-identical to ciso1's / cdep1's / ciso2's / cvt7's; HOLDHIGH is cvt7's string; P_BIG is cvt6's schedule" \
  || guard_fail "guard 1c5: the byte-identity checks did not pass"
grep -q "^  PASS B RULE I re-run on THIS host: the eligible runs are exactly the 12 registered sources" "$SELFT" \
  && grep -q "^  PASS B RULE I re-run on THIS host's landed records reproduces the replay file BYTE FOR BYTE" "$SELFT" \
  && grep -q "^  PASS B RULE P (243.3) re-run on THIS host == P_HIGH 8609" "$SELFT" \
  && echo "guard 1c6: RULE I (12 eligible runs, the replay file byte for byte) and RULE P (8609) re-derived ON THIS HOST" \
  || guard_fail "guard 1c6: RULE I / RULE P were not re-derived on this host"
grep -q "^  PASS B G-BITE PASSES on all 24 landed" "$SELFT" && grep -q "^  PASS B every landed ISO run read as each of the 5 held arms FAILS" "$SELFT" \
  && grep -q "^  PASS B cvt7's 3 landed HOLDHIGH runs PASS G-BITE" "$SELFT" && grep -q "^  PASS B THE BROKEN-REST_HOLD NULL ON REAL RECORDS" "$SELFT" \
  && grep -q "^  PASS B cvt7 HOLDHIGH read as HOLDBIG FAILS" "$SELFT" \
  && echo "guard 1c7: G-BITE passes 24 landed controls and cvt7's HOLDHIGH; FAILS 60 real-record carrier nulls, 6 REST_HOLD nulls, 3 wrong-dose reads" \
  || guard_fail "guard 1c7: the real-record G-BITE checks did not pass"
grep -q "^  PASS SIGMA_R18ALL: the FROZEN literal 0.68815302799238276 IS the FILTERED value" "$SELFT" \
  && echo "guard 1c8: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value on the registration corpus" \
  || guard_fail "guard 1c8: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E BROKEN REST_HOLD: every forced complement UNHELD" "$SELFT" && grep -q "^  PASS E HOLD ONLY AT PROBE STEPS: complement" "$SELFT" \
  && grep -q "^  PASS E WRONG DOSE: HOLDBIG" "$SELFT" && grep -q "^  PASS E WRONG REPLAY" "$SELFT" && grep -q "^  PASS E FREE COMPLEMENT TOUCHED" "$SELFT" \
  && echo "guard 1c9: the broken-REST_HOLD null, a probe-step-only hold, a wrong dose, a wrong replay and a touched free complement reach PATCH-NOT-VERIFIED" \
  || guard_fail "guard 1c9: G-BITE break cases did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 1d: the derivation reproduces RULE I, RULE P and the floor on THIS host ----------
DOUT=/tmp/cvt8_deriv.$$
python3 "$DERIV" "$WS/runs" "$CSVP" >"$DOUT" 2>&1 \
  && grep -q "^  candidates 21; ELIGIBLE 12: ciso1-ISO-s21 ciso1-ISO-s22 ciso1-ISO-s23 ciso2-ISO-s31 ciso2-ISO-s32 ciso2-ISO-s33 cdep1-ISO-s24 cdep1-ISO-s25 cdep1-ISO-s26 cvt7-ISO-s99 cvt7-ISO-s100 cvt7-ISO-s101$" "$DOUT" \
  && grep -q "^  REPLAY FILE cvt8_isopath.json  13126 bytes  sha256 $ISOPATH_SHA$" "$DOUT" \
  && grep -q "^  RULE P (243.3) re-run on this host: P_HIGH 8609, P_ISO 5153  (registered 8609 / 5153: True)" "$DOUT" \
  && grep -q "SIGMA_R18ALL  FILTERED 0.68815302799238[0-9]* (df 215" "$DOUT" \
  && echo "guard 1d: analysis/cvt8_registration_derivations.py reproduces RULE I, the replay sha, RULE P and the filtered floor on this host (output sha $(sha "$DOUT"))" \
  || { guard_fail "guard 1d: the derivation did not reproduce RULE I / RULE P / the floor"; grep -E "ELIGIBLE|REPLAY FILE|RULE P|SIGMA_R18ALL" "$DOUT"; tail -3 "$DOUT"; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt8-' "$CSVP" && guard_fail "guard 2: cvt8-* already in the CSV" || echo "guard 2: no cvt8-* row in the corpus"
python3 - "$CSVP" "$SEEDS" <<'PYEOF' || guard_fail "guard 2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
raise SystemExit(1 if hit else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt8-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt8-*.out on disk" || guard_fail "guard 2b: $NOUT cvt8-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (102|103|104)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 102/103/104" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 102/103/104"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt8-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt8-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt8-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt8-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt8-* in squeue" || guard_fail "guard 2f: $NQ cvt8-* already queued"
fi

# ---- guard 3/4a: the cvt8 tree consistent (cvt7's tree and the LIVE tree unchanged) ------------
bash "$STAGE8" >/tmp/cvt8_stage8.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT8_stage_harness.sh (verify mode) VERIFIES cvt8's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt8_stage8.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt8's tree is NOT consistent"; tail -20 /tmp/cvt8_stage8.$$; }
rm -f /tmp/cvt8_stage8.$$
bash "$REPO/bin/cVT7_stage_harness.sh" >/tmp/cvt8_stage7.$$ 2>&1 \
  && echo "guard 4a1: bin/cVT7_stage_harness.sh (registered, unedited, verify mode) VERIFIES cvt7's tree unchanged" \
  || { guard_fail "guard 4a1: cvt7's tree is NOT consistent"; tail -10 /tmp/cvt8_stage7.$$; }
rm -f /tmp/cvt8_stage7.$$
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" || guard_fail "guard 3: CIFAR-100 not visible"
[ "$(sha "$BNH")" = "$LIVE_BN_SHA" ] && [ "$(sha "$LIVE/build_network.py")" = "$LIVE_BN_SHA" ] \
  && echo "guard 3b: build_network.py == the LIVE one c7998883..." || guard_fail "guard 3b: build_network.py is not the LIVE one"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFRH")" = "$HF_PRE_SHA" ] && [ "$(sha "$H7/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && [ "$(sha "$HFVW")" = "$HF_LIVE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA 5197dc2e...; .pre_resthold == cvt7's 2396f2be...; .pre_voteweight == the LIVE 4732b74a..." \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones"
[ "$(sha "$HARN/Optimizers/cvt8_isopath.json")" = "$ISOPATH_SHA" ] && [ "$(sha "$ISOJ")" = "$ISOPATH_SHA" ] \
  && echo "guard 3f: the tree's and the repo's cvt8_isopath.json == ISOPATH_SHA 08ab25f3..." || guard_fail "guard 3f: replay file sha"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == 1dbe4ff5... (run_cifar.sh with the cd line changed)" \
  || guard_fail "guard 3e: $RUNNER is not the pinned cvt8 runner"
grep -q PATCH_RESTHOLD "$LIVE/Optimizers/HF.py" "$H7/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE or cvt7's HF.py carries PATCH_RESTHOLD" \
  || echo "guard 3d: neither the LIVE nor cvt7's HF.py carries PATCH_RESTHOLD"

# ---- guard 4b/4f: the patches, proved inert; the new patch proved inert and biting ----
python3 "$TESTT" --pre "$HFPRE" --post "$HFVW" --structural-only >/tmp/cvt8_struct.$$ 2>&1 \
   && echo "guard 4b: PATCH_PROBE_TENSOR structural inertness R0 PASSES on HF.py.pre_voteweight (= live)" \
   || { guard_fail "guard 4b: PATCH_PROBE_TENSOR structural R0 FAILED"; tail -8 /tmp/cvt8_struct.$$; }
python3 "$TESTR8" --pre "$HFRH" --post "$HFH" --structural-only >/tmp/cvt8_rh0.$$ 2>&1 \
