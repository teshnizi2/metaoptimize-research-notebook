#!/usr/bin/env python3
"""g3b_live_check.py -- `g3b` (CORRECTIONS 299): the spec strings and group counts RE-DERIVED ON THE LIVE MODEL.

Run on cluster_b (CPU ONLY -- model and optimizer CONSTRUCTION, no forward pass, no GPU) from the launcher's guard 4:
    python3 analysis/g3b_live_check.py <tree cifar10 dir> <out manifest path> <cwd1 tree HF.py>
For every one of the 8 arms it parses the arm's OWN registered ARGS tokens with the tree's OWN `train.parse_args`,
builds the network with the tree's OWN `build_network` and the optimizer with the tree's OWN `build_optimizer`, and
reads back: the stepsize type, the group count m = sum(numel(beta)), the weight decay the optimizer holds, the bound
base update, the tensor count and the parameter count.  It then writes the LIVE manifest in the design's own format
and exits 1 unless it is BYTE-IDENTICAL to g3b_design.manifest_text().

guard 4e (cgw1 291.3, carried over): the SOURCE TEXT of `SGDm_base_update` in the tree's HF.py is compared, by AST source
segment and without importing it, with the same method in harness_cwd1's HF.py -- the file whose real-GPU bite log
(cwd5, RW2) proved the passed weight decay reaches the update bitwise at 0.1 / 1e-2 / 1e-3 / 5e-4.  That identity is
what lets this batch inherit the premise without a proof job of its own.
"""
import importlib.util
import os
import sys

tree, out, cwd1_hf = sys.argv[1], sys.argv[2], sys.argv[3]
HERE = os.path.dirname(os.path.abspath(__file__))
spec = importlib.util.spec_from_file_location("g3b_design", os.path.join(HERE, "g3b_design.py"))
D = importlib.util.module_from_spec(spec)
spec.loader.exec_module(D)

os.chdir(tree)
sys.path.insert(0, tree)
import torch  # noqa: E402
import argparse  # noqa: E402
import ast  # noqa: E402
# train.py runs code at module level after its __main__ block, so it is NOT imported: its OWN `parse_args` function
# is lifted out of the file by the AST, byte for byte, and executed in a namespace holding argparse.
_src = open(os.path.join(tree, "train.py")).read()
_fn = [n for n in ast.parse(_src).body if isinstance(n, ast.FunctionDef) and n.name == "parse_args"][0]
_ns = {"argparse": argparse}
exec(compile(ast.Module(body=[_fn], type_ignores=[]), os.path.join(tree, "train.py"), "exec"), _ns)


class train:  # noqa: N801
    parse_args = staticmethod(_ns["parse_args"])


from build_network import build_network  # noqa: E402
from Optimizers.build_optimizer import build_optimizer  # noqa: E402

bad = 0


def _method_src(path, cls, meth):
    src = open(path).read()
    for n in ast.parse(src).body:
        if isinstance(n, ast.ClassDef) and n.name == cls:
            for f in n.body:
                if isinstance(f, ast.FunctionDef) and f.name == meth:
                    return ast.get_source_segment(src, f)
    return None


_a = _method_src(os.path.join(tree, "Optimizers", "HF.py"), "HF", "SGDm_base_update")
_b = _method_src(cwd1_hf, "HF", "SGDm_base_update")
if _a is None or _a != _b:
    print("!!! guard 4e: SGDm_base_update in %s is NOT the source text of %s's" % (tree, cwd1_hf))
    bad += 1
else:
    print("guard 4e: SGDm_base_update source text (%d chars, sha256 %s) is IDENTICAL in this tree and in %s"
          % (len(_a), D.hashlib.sha256(_a.encode()).hexdigest()[:16], cwd1_hf))
    print("          ('delta = a * (self.momentum_base[i] + self.args_base['weight_decay']*w.data)' present: %s)"
          % ("self.args_base['weight_decay']*w.data" in _a))
live_m, ntens, totpar = {}, set(), set()
for a in D.ARMS:
    toks = []
    for k, v in D.expected_args(a, D.SEEDS[a][0], "/tmp/g3b_livecheck"):
        toks += ["--" + k, v]
    sys.argv = ["train.py"] + toks
    args = train.parse_args()
    args.device = torch.device("cpu")
    torch.manual_seed(0)
    net = build_network(args.NN_name, args.device)
    opt = build_optimizer(net, args, None)
    m = sum(int(b.numel()) for b in opt.beta)
    nt = len(list(net.parameters()))
    tp = sum(p.numel() for p in net.parameters())
    wd = opt.args_base["weight_decay"]
    bu = getattr(opt.base_update, "__name__", "?")
    ok = (m == D.M_OF_ARM[a] and abs(wd - D.WDF[a]) == 0.0 and bu == "SGDm_base_update"
          and args.stepsize_groups == D.SPEC[a] and args.weight_decay_base == D.WDF[a]
          and opt.args_meta["meta_stepsize"] == float(D.MST) and float(args.alpha0) == float(D.A0)
          and args.dataset == D.DSET and args.NN_name == D.NET)
    print("guard 4: %-6s spec %-9s type %-10s m %6d (registered %6d)  wd %r (registered %r)  base_update %s  %s"
          % (a, args.stepsize_groups, opt.stepsize_type, m, D.M_OF_ARM[a], wd, D.WDF[a], bu, "OK" if ok else "!!! MISMATCH"))
    bad += not ok
    live_m.setdefault(D.GRAIN_OF[a], set()).add(m)
    ntens.add(nt)
    totpar.add(tp)
if len(ntens) != 1 or len(totpar) != 1:
    print("!!! guard 4: tensor / parameter counts differ across arms: %s %s" % (ntens, totpar))
    bad += 1
nt, tp = ntens.pop(), totpar.pop()
L = ["BATCH %s" % D.PREFIX, "NETWORK %s" % D.NET, "DATASET %s" % D.DSET, "NUM_PARAM_TENSORS %d" % nt,
     "TOTAL_PARAMS %d" % tp, "CLIP_C %s" % D.CLIP, "MS %s ALPHA0 %s" % (D.MST, D.A0), "EPOCHS %d" % D.EPOCHS,
     "META_STEPS %d" % (D.EPOCHS * D.STEPS_PER_EPOCH), "PROBE %d N_RECORDS %d" % (D.PROBE, D.N_RECORDS),
     "WD_LADDER %s" % ",".join("%s=%s" % (r, D.WD_TOKEN[r]) for r in D.RUNG_IDS)]
for g, spec_, _m in D.GRAINS:
    ms = sorted(live_m[g])
    L.append("GRAIN %s SPEC %s M %s" % (g, spec_, ",".join(str(x) for x in ms)))
for a in D.ARMS:
    L.append("ARM %s RUNG %s WD %s SPEC %s M %s SEEDS %s" % (a, D.RUNG_OF[a], D.WD[a], D.SPEC[a],
             ",".join(str(x) for x in sorted(live_m[D.GRAIN_OF[a]])), ",".join(str(s) for s in D.SEEDS[a])))
L.append("NJOBS %d" % D.NJOBS)
txt = "\n".join(L) + "\n"
if txt != D.manifest_text():
    a_, b_ = txt.splitlines(), D.manifest_text().splitlines()
    for k in range(max(len(a_), len(b_))):
        if k >= len(a_) or k >= len(b_) or a_[k] != b_[k]:
            print("!!! guard 4d: live manifest differs at line %d:\n    live      %s\n    design    %s"
                  % (k + 1, a_[k] if k < len(a_) else "", b_[k] if k < len(b_) else ""))
            break
    bad += 1
else:
    print("guard 4d: live %s: %d tensors, %d parameters; the LIVE manifest is BYTE-IDENTICAL to "
          "g3b_design.manifest_text() (%d bytes)" % (D.NET, nt, tp, len(txt)))
if bad:
    raise SystemExit(1)
open(out, "w").write(txt)
