def chk(cond, label, extra=""):

def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def fmt(x, nd=4):
    return "n/a" if x is None else ("%.*f" % (nd, x))


def file_sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()

def build_tensor_list():
    """ResNet-18 for 100 classes, in torch's named_parameters() order.
    -> list of (name, numel, module_type, ndim), 1-based position = list index + 1."""
    t = []

    def conv(name, cout, cin, k):
        t.append((name, cout * cin * k * k, "Conv2d", 4))

    def bn(prefix, c, with_bias=True):
        t.append((prefix + ".weight", c, "BatchNorm2d", 1))
        if with_bias:
            t.append((prefix + ".bias", c, "BatchNorm2d", 1))

    conv("conv1.weight", 64, 3, 3)
    bn("bn1", 64)
    cin = 64
    for li, cout in enumerate((64, 128, 256, 512), start=1):
        for bi in (0, 1):
            pre = "layer%d.%d" % (li, bi)
            stride_block = (bi == 0 and cin != cout)
            conv(pre + ".conv1.weight", cout, cin, 3)
            bn(pre + ".bn1", cout)
            conv(pre + ".conv2.weight", cout, cout, 3)
            bn(pre + ".bn2", cout)
            if stride_block:
                conv(pre + ".shortcut.0.weight", cout, cin, 1)
                bn(pre + ".shortcut.1", cout)
            cin = cout
    t.append(("linear.weight", 512 * N_CLASSES, "Linear", 2))
    t.append(("linear.bias", N_CLASSES, "Linear", 1))
    return t


def normscale_by_rule(tensors):
    """the patch's own rule: 1-D `*.weight`.  -> list of 1-based positions."""
    return [i + 1 for i, (n, _, _, d) in enumerate(tensors) if d == 1 and n.endswith(".weight")]


def normscale_by_module_type(tensors):
    """independent route: every BatchNorm2d `.weight`, by module type."""
    return [i + 1 for i, (n, _, m, _) in enumerate(tensors) if m == "BatchNorm2d" and n.endswith(".weight")]


def dm_witness_text(tensors, idx):
    return ("DECAY_MASK: on base=SGDm wd=%s spec=normscale masked=%d of=%d numel=%d idx=%s names=%s"
            % (WD_BASE, len(idx), len(tensors), sum(tensors[i - 1][1] for i in idx),
               ",".join(str(i) for i in idx), ",".join(tensors[i - 1][0] for i in idx)))


def manifest_text(tensors, idx):
    L = ["NETWORK %s" % NET,
         "NUM_PARAM_TENSORS %d" % len(tensors),
         "TOTAL_PARAMS %d" % sum(x[1] for x in tensors),
         "CLIP_C %s" % CLIP,
         "EPOCHS %d" % EPOCHS,
         "META_STEPS %d" % META_STEPS,
         "WD_BASE %s" % WD_BASE]
    for i, (n, ne, m, d) in enumerate(tensors, start=1):
        L.append("TENSOR %d %s %d %s ndim=%d" % (i, n, ne, m, d))
    L.append("NORMSCALE %s" % ",".join("%d:%s" % (i, tensors[i - 1][0]) for i in idx))
    wit = dm_witness_text(tensors, idx)
    for a in ARMS:
        L.append("ARMSPEC %s SPEC %s TYPE %s" % (a, SPEC[a], PT_TYPE[a]))
        L.append("DECAYMASK %s %s" % (a, DMASK[a] or "off"))
        L.append("DWITNESS %s %s" % (a, wit if DMASK[a] else "DECAY_MASK: off"))
    return "\n".join(L) + "\n"

def _after(s, tag):
    i = s.find(tag)
    return None if i < 0 else s[i + len(tag):]


def parse_epoch_line(ln):
    """'Epoch 7, Train Accuracy: 1.17 %, Test Accuracy: 1.30 %' -> (7, 1.17, 1.30); None if not one."""
    s = ln.strip()
    if not s.startswith("Epoch "):
        return None
    parts = s.split(",")
    if len(parts) < 3:
        return None
    head = parts[0][len("Epoch "):].strip()
    if not head or not all(c in "0123456789" for c in head):
        return None
    tr = _after(parts[1], "Train Accuracy:")
    te = _after(parts[2], "Test Accuracy:")
    if tr is None or te is None:
        return None
    tr, te = tr.strip().rstrip("%").strip(), te.strip().rstrip("%").strip()
    try:
        return int(head), float(tr), float(te)
    except ValueError:
        return None


def parse_run(path):
    txt = open(path, errors="replace").read()
    lines = txt.splitlines()
    eps = {}
    for ln in lines:
        e = parse_epoch_line(ln)
        if e is not None:
            eps[e[0]] = (e[1], e[2])
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "plateau5": None, "train5": None}
    for k in KINDS:
        rec[k] = [ln for ln in lines if ln.startswith(k)]
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"] and not rec["traceback"]
                       and rec["plateau5"] is not None)
    return rec


def split_out_name(fn):
    """'cwd1-k01NWD-s128-5045343.out' -> ('k01NWD', 128, 5045343), else None.  No regex."""
    if not fn.endswith(".out"):
        return None
    stem = fn[:-4]
    f = stem.split("-")
    if len(f) != 4 or f[0] != PREFIX or f[1] not in ARMS:
        return None
    if not f[2].startswith("s") or not f[2][1:].isdigit() or not f[3].isdigit():
        return None
    return f[1], int(f[2][1:]), int(f[3])


def read_runs(runsdir):
    cand = {}
    for fn in sorted(os.listdir(runsdir)):
        k = split_out_name(fn)
        if k is None:
            continue
        arm, seed, jid = k
        r = parse_run(os.path.join(runsdir, fn))
        r.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(r)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        got[k] = max(comp or lst, key=lambda r: r["job"])
    return got


def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def load_probe(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (PREFIX, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            return recs
    return None


def groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    return [float(x) for x in v[0]]


def lion_natural(beta_pre, mom_pre, z):
    a, b = B2 * mom_pre, (1.0 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(BETA_LO, min(BETA_HI, beta_pre - MS * s)), tie


def lion_check(recs, ng):
    d = {"n": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "nonfinite": 0, "worst": 0.0}
    if recs is None:
        return False, d
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        bpre, mom, z = groupvec(r, "beta_pre", ng), groupvec(r, "mom_pre", ng), groupvec(r, "z_agg", ng)
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        if r["step"] != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        for g in range(ng):
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst"] = max(d["worst"], e)
            d["bad_lion"] += e > BH_TOL
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_step"] == 0 and d["bad_lion"] == 0)
    return ok, d


def _finite(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool) and math.isfinite(x)


def dm_audit(recs, k):
    d = {"n": 0, "bad_rec": 0, "bad_n": 0, "bad_masked": 0, "bad_skipped": 0, "bad_types": 0,
         "keys_on_unmasked": 0, "nonfinite": 0, "pos_wdterm": 0, "last": None}
    if recs is None:
        return False, d
    prev = -1
    for r in recs:
        if r is None or not isinstance(r.get("step"), int):
            d["bad_rec"] += 1
            continue
        d["n"] += 1
        has = [key for key in r if key.startswith("dm_")]
        if k == 0:
            d["keys_on_unmasked"] += bool(has)
            continue
        s = r["step"]
        dn = r.get("dm_n")
        if not (isinstance(dn, int) and dn == s + 2 and dn > prev):
            d["bad_n"] += 1
        prev = dn if isinstance(dn, int) else prev
        if r.get("dm_masked") != k:
            d["bad_masked"] += 1
        if not (isinstance(dn, int) and r.get("dm_skipped") == dn * k):
            d["bad_skipped"] += 1
        wt, nr, am, sm = r.get("dm_wdterm"), r.get("dm_norm"), r.get("dm_absmin"), r.get("dm_small")
        types_ok = (isinstance(wt, (int, float)) and not isinstance(wt, bool) and isinstance(nr, list) and len(nr) == k
                    and all(isinstance(x, (int, float)) and not isinstance(x, bool) for x in nr)
                    and isinstance(am, (int, float)) and not isinstance(am, bool) and isinstance(sm, int) and sm >= 0
                    and sorted(has) == sorted(DM_KEYS))
        if not types_ok:
            d["bad_types"] += 1
            continue
        if not (_finite(wt) and all(_finite(x) for x in nr) and _finite(am)):
            d["nonfinite"] += 1
            continue
        if wt < 0 or am < 0:
            d["bad_types"] += 1
            continue
        d["pos_wdterm"] += wt > 0
        d["last"] = {"dm_n": dn, "dm_skipped": r.get("dm_skipped"), "dm_wdterm": wt, "dm_absmin": am, "dm_small": sm}
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] == N_RECORDS)
    if k == 0:
        ok = ok and d["keys_on_unmasked"] == 0
    else:
        ok = (ok and d["bad_n"] == 0 and d["bad_masked"] == 0 and d["bad_skipped"] == 0
              and d["bad_types"] == 0 and d["pos_wdterm"] > 0)
    return ok, d


def hold_keys_present(recs):
    if recs is None:
        return None
    return sum(1 for r in recs if isinstance(r, dict) and any(key.startswith(HOLD_PREFIXES) for key in r))

# 4. THE CORPUS (one DISCLOSURE line only)
# =============================================================================
def tsv_keys(path):
    lines = [ln.rstrip("\n") for ln in open(path) if ln.strip() and not ln.startswith("#")]
    head = lines[0].split("\t")
    out = set()
    for ln in lines[1:]:
        f = ln.split("\t")
        if len(f) != len(head):
            raise ValueError("bad arity")
        e = dict(zip(head, f))
        out.add((e["run"], e["job_id"]))
    return out


def corpus_disclosure(csvpath, tsvpath):
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    k = tsv_keys(tsvpath)
    kept = [r for r in rows if (r.get("run"), r.get("job_id")) not in k]
    return len(rows), len(kept)

def decide(M, diverged):
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M["kLNWD"] - M["k01"] < GAP_MIN:
        return "REFERENCE-FAILED"
    if M["k01NWD"] >= M["kLNWD"] - MATCH_BAR:
        return "COLLAPSE-VANISHES"
    if M["k01NWD"] <= M["k01"] + NULL_BAR:
        return "COLLAPSE-PERSISTS"
    return "COLLAPSE-REDUCED"

    for nm, (x, y), cp in (("P_NWD", ("k01NWD", "k01"), P_NWD), ("G_NWD", ("kLNWD", "k01NWD"), G_NWD)):
        R("  PRIMARY  %-6s = %-7s - %-7s = %+.4f pp = %+.2f SE   (TRAIN %+.4f)   seeds %s"
          % (nm, x, y, cp[0], cp[0] / se, cp[1],
             " / ".join("%+.3f" % (runs[(x, s)]["plateau5"] - runs[(y, s)]["plateau5"]) for s in SEEDS)))
    R("  KEY      D_REF  = kLNWD - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)"
      % (D_REF, D_REF / se, Tm["kLNWD"] - Tm["k01"]))
    R("  DESCR    RATIO  = k01NWD / kLNWD = %.4f  (census R50 collapse iff <= %.2f)" % (RATIO, R50))
    R("  DESCR    F_NWD  = P_NWD / D_REF  = %.4f" % F_NWD)
    R("  bars: GAP_MIN %.1f (%.2f SE) | MATCH %.1f pp (%.2f SE) | NULL %.1f pp (%.2f SE) | RESCUE %.1f pp (%.2f SE)"
      % (GAP_MIN, GAP_MIN / se, MATCH_BAR, MATCH_BAR / se, NULL_BAR, NULL_BAR / se, RESCUE_BAR, RESCUE_BAR / se))
    R("  state bars: VANISHES iff k01NWD >= %.4f ; PERSISTS iff k01NWD <= %.4f"
      % (M["kLNWD"] - MATCH_BAR, M["k01"] + NULL_BAR))
    R("")
    R("G-DIVERGE (a branch condition, not a harness gate)")
    R("  %s every arm's seed range <= %.1f pp" % ("PASS" if not div else "FAIL", DIVERGED_BAR))
    R("")
    R("=" * 78)
    R(final)
    R("=" * 78)
    R("")
    R("BETWEEN-BATCH, NON-GATING (frozen literals; NO other batch enters any cwd1 bar or contrast):")
    for b_, dct in R_BETWEEN:
        R("  %s: %s" % (b_, "  ".join("%s %.2f" % kv for kv in dct)))
    R("  cwd1 {%s}: %s" % (",".join(str(s) for s in SEEDS), "  ".join("%s %.4f" % (arm, M[arm]) for arm in ARMS)))
    R("")
    R("DESCRIPTIVE (non-gating) -- PATCH_DECAYMASK's readout on the masked arms: ||w|| of the three ctd1 carriers")
    R("(%s) and the median over the 20 scales; min |w| over the 20; entries |w| < 1e-3; records 0 / 50 / 200 / 499; "
      "seed means" % ", ".join(CARRIERS))
    names20 = [T[i - 1][0] for i in by_rule]
    pos20 = [names20.index(c) for c in CARRIERS]
    for arm in ("k01NWD", "kLNWD"):
        rows = [bite[(arm, s)][0] or [] for s in SEEDS]
        parts = []
        for k in (0, 50, 200, 499):
            v = [r[k] for r in rows if len(r) > k and isinstance(r[k], dict)
                 and isinstance(r[k].get("dm_norm"), list) and len(r[k]["dm_norm"]) == 20]
            car = [mean([x["dm_norm"][p] for x in v]) for p in pos20]
            med = mean([sorted(x["dm_norm"])[10] for x in v])
            parts.append("rec %d: carriers %s median %.3g absmin %.3g small %.1f"
                         % (k, "/".join("%.3g" % c for c in car), med,
                            mean([x["dm_absmin"] for x in v]), mean([x["dm_small"] for x in v])))
        R("  %-7s %s" % (arm, " | ".join(parts)))
    R("")
    R("WHAT THIS DECIDES -- as registered, before any run existed:")
    for ln in LICENCE_VANISHES:
        R("  " + ln)
    R("")
    R("WHAT IT DOES NOT LICENSE, UNCONDITIONALLY:")
    for ln in NOT_LICENSED:
        R(ln)

    P("\n" + "=" * 100)
    P("REPLICA -- the registered scorer's WHOLE output, rebuilt independently (its two host-path lines")
    P("written as <RUNSDIR>/...; the scorer's corpus DISCLOSURE line re-derived at commit %s)" % CORPUS_COMMIT[:7])
    P("=" * 100)
    for ln in REPLICA:
        P(ln)
    P("=" * 100)

    # ---- 9. diff against the committed scorer stdouts ---------------------------------------------
    P("\n[DIFF] the replica against each committed scorer stdout (its runsdir replaced by <RUNSDIR>)")
    for path in a.scorer_out:
        got = open(path).read().splitlines()
        base = os.path.basename(path)
        # find and normalise the scorer's own runsdir prefix from its manifest line
        rdpref = None
        for ln in got:
            if ln.startswith("  manifest: ") and ln.endswith("PARTITION-MANIFEST.txt (%s)" % mhow):
                rdpref = ln[len("  manifest: "):-len("/%s/PARTITION-MANIFEST.txt (%s)" % (PREFIX, mhow))]
        norm = [ln.replace(rdpref, "<RUNSDIR>") if rdpref else ln for ln in got]
        same = unexpected = disclosure = 0
        for i in range(max(len(norm), len(REPLICA))):
            g = norm[i] if i < len(norm) else "<missing>"
            w = REPLICA[i] if i < len(REPLICA) else "<missing>"
            if g == w:
                same += 1
            elif g.startswith("corpus: ") and w.startswith("corpus: "):
                disclosure += 1
                P("      %s line %d is the DISCLOSURE line against a different corpus:" % (base, i + 1))
                P("        scorer   %s" % g)
                P("        replica  %s" % w)
            else:
                unexpected += 1
                P("      %s line %d DIFFERS:" % (base, i + 1))
                P("        scorer   %s" % g)
                P("        replica  %s" % w)
        P("    %s: %d lines, %d byte-identical, %d disclosure-only, %d unexpected"
          % (base, len(norm), same, disclosure, unexpected))
        chk(unexpected == 0 and len(norm) == len(REPLICA),
            "%s is reproduced line for line by the independent parser" % base)
        for ln in got:
            if ln.startswith("FINAL: "):
                chk(ln == final, "%s's FINAL line is re-derived exactly" % base)

    # ---- 10. DESCRIPTIVE, outside every gate ------------------------------------------------------
    P("\n" + "=" * 100)
    P("DESCRIPTIVE / UNSURE -- gating nothing, licensed by nothing; boundaries stated on every line")
    P("=" * 100)
    P("\n[D1] TEST (TRAIN) after 15 / 30 / 50 / 100 completed epochs.  BOUNDARY: 'after N epochs' = the")
    P("     .out line `Epoch N-1` (epochs are 0-based, so epoch 99 is the 100th).  Per-arm mean over")
    P("     seeds 128/129/130, single epochs, NOT plateau5.")
    P("  %-7s %s" % ("arm", "  ".join("%18s" % ("after %d ep" % n) for n in (15, 30, 50, 100))))
    for arm in ARMS:
        cells = []
        for n in (15, 30, 50, 100):
            e = n - 1
            te = mean([runs[(arm, s)]["eps"][e][1] for s in SEEDS])
            tr = mean([runs[(arm, s)]["eps"][e][0] for s in SEEDS])
            cells.append("%18s" % ("%.4f (%.4f)" % (te, tr)))
        P("  %-7s %s" % (arm, "  ".join(cells)))
    P("     per seed, after 100 epochs (epoch 99):")
    for arm in ARMS:
        P("       %-7s %s" % (arm, "  ".join("s%d %.2f (%.2f)" % (s, runs[(arm, s)]["eps"][99][1],
                                                                  runs[(arm, s)]["eps"][99][0]) for s in SEEDS)))

    P("\n[D2] the step-size (beta) trajectory.  BOUNDARY: probe records every 100 meta steps, 500 records")
    P("     per run, record r == meta step 100r, r in [0, 499]; beta is log step size, clip [-15, -2.3026].")
    P("     k01 / k01NWD are SHARED (1 group); kLNWD is LAYERWISE (62 groups) and is summarised over groups.")
    for arm in ARMS:
        for s in SEEDS:
            recs = bite[(arm, s)][0]
            if NG[arm] == 1:
                b = [r["beta"][0] for r in recs]
                atfloor = sum(1 for x in b if x <= BETA_LO + 1e-6) / len(b)
                first = next((r["step"] for r in recs if r["beta"][0] <= BETA_LO + 1e-6), None)
                P("  %-7s s%d  shared beta: rec0 %.4f  rec50 %.4f  rec200 %.4f  rec499 %.4f  | max %.4f  min %.4f  "
                  "frac at the -15 floor %.4f  first floor record at meta step %s"
                  % (arm, s, b[0], b[50], b[200], b[499], max(b), min(b), atfloor, first))
            else:
                def stat(r):
                    v = r["beta"]
                    return min(v), mean(v), max(v), sum(1 for x in v if x <= BETA_LO + 1e-6)
                for k in (0, 50, 200, 499):
                    lo, mu, hiq, nf = stat(recs[k])
                    P("  %-7s s%d  rec %3d: 62-group beta  min %.4f  mean %.4f  max %.4f  at floor %d/62"
                      % (arm, s, k, lo, mu, hiq, nf))
    P("     carrier groups of the LAYERWISE arm (group index = tensor position - 1):")
    for s in SEEDS:
        recs = bite[("kLNWD", s)][0]
        for c in CARRIERS:
            gi = [i for i, (n, _, _, _) in enumerate(T) if n == c][0]
            b = [r["beta"][gi] for r in recs]
            P("       kLNWD s%d %-28s rec0 %.4f  rec200 %.4f  rec499 %.4f  min %.4f  max %.4f"
              % (s, c, b[0], b[200], b[499], min(b), max(b)))

    P("\n[D3] the skipped weight-decay term and the scales' norms (PATCH_DECAYMASK's own record).")
    P("     BOUNDARY: dm_wdterm is the magnitude of the coupled WD term the mask REMOVED at that record;")
    P("     dm_norm is the per-tensor ||w|| of the 20 masked scales; dm_absmin the smallest |w| among them.")
    for arm in ("k01NWD", "kLNWD"):
        for s in SEEDS:
            recs = bite[(arm, s)][0]
            P("  %-7s s%d  dm_wdterm rec0 %.6e  rec200 %.6e  rec499 %.6e | dm_absmin rec0 %.4f rec499 %.4f | "
              "dm_small rec499 %d" % (arm, s, recs[0]["dm_wdterm"], recs[200]["dm_wdterm"], recs[499]["dm_wdterm"],
                                      recs[0]["dm_absmin"], recs[499]["dm_absmin"], recs[499]["dm_small"]))

    P("\n[D4] the cmo1 comparison (BETWEEN BATCH, DESCRIPTIVE -- different seeds, different batch, no")
    P("     shared arm was run twice; nothing here is a within-batch contrast and no bar reads it).")
    P("     CORRECTIONS 264: cmo1 W0k01 71.5600 (base weight decay 0 EVERYWHERE + the meta trace),")
    P("     cmo1 k01 22.7887, L_W0 +48.7713 pp.")
    P("  cwd1 k01NWD %.4f (WD 0 on the 20 BatchNorm scales only)  vs  cmo1 W0k01 71.5600   difference %+.4f pp"
      % (M["k01NWD"], M["k01NWD"] - 71.5600))
    P("  cwd1 P_NWD %+.4f pp  vs  cmo1 L_W0 +48.7713 pp   -> the norm-scale mask recovers %.2f %% of the"
      % (P_NWD[0], 100.0 * P_NWD[0] / 48.7713))
    P("  whole-network W0 effect, on this between-batch reading only.")

    P("\nVIOLATIONS %d" % len(VIOL))
    for v in VIOL:
        P("  ! %s" % v)
    sys.stdout.write("\n".join(OUT) + "\n")
    print("HOSTPATH runsdir      %s" % os.path.abspath(rd))
    print("HOSTPATH corpus csv   %s" % os.path.abspath(a.csv))
    print("HOSTPATH corpus tsv   %s" % os.path.abspath(a.tsv))
    for path in a.scorer_out:
        print("HOSTPATH scorer out   %s" % os.path.abspath(path))
    return 0 if not VIOL else 1
