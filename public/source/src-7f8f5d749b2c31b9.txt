def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

def read_corpus(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    return [r for r in csv.DictReader(open(csvpath))
            if not r.get("run", "").startswith("cvi1-")]


def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and r.get("plateau5", "").strip() != "")


def std_cell(r):
    return (r.get("granularity") in ("scalar", "layerwise")
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra=lambda r: True):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r[k] for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm, top = 0.0, 0, 0, 0, []
    for k, v in cells.items():
        if len(v) < 2:
            continue
        m = mean(v)
        s = sum((x - m) ** 2 for x in v)
        ss += s
        df += len(v) - 1
        nc += 1
        nm += len(v)
        top.append((s, k[0], k[2], len(v)))
    top.sort(reverse=True)
    return (math.sqrt(ss / df) if df else None), df, nc, nm, ss, top


def corpus_gaps(rows):
    by = {}
    for r in rows:
        if usable(r) and std_cell(r):
            by.setdefault((r["network"], r["dataset"]), {}) \
              .setdefault(r["granularity"], []).append(float(r["plateau5"]))
    out = {}
    for k, v in by.items():
        if "scalar" in v and "layerwise" in v:
            s, l = mean(v["scalar"]), mean(v["layerwise"])
            out[k] = (s, l, l - s, len(v["scalar"]), len(v["layerwise"]))
    return out

def read_runs(runsdir):
    got = {}
    if not runsdir or not os.path.isdir(runsdir):
        return got
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), m.group(3)
        txt = open(os.path.join(runsdir, fn), errors="replace").read()
        eps = sorted((int(a), float(b), float(c)) for a, b, c in EPTR_RE.findall(txt))
        rec = {
            "file": fn, "job": jid, "arm": arm, "seed": seed,
            "n_epochs": len(eps),
            "traceback": "Traceback (most recent call last)" in txt,
            "run_done": "RUN_DONE" in txt,
            "env": [ln for ln in txt.splitlines() if ENV_RE.match(ln)],
            "pt": [ln for ln in txt.splitlines() if ln.startswith("PROBE_TENSOR:")],
            "plateau5": None, "train5": None,
        }
        tail = [e for e in eps if e[0] >= EPOCHS - 5]
        if len(tail) == 5:
            rec["plateau5"] = mean([e[2] for e in tail])
            rec["train5"] = mean([e[1] for e in tail])
        prev = got.get((arm, seed))
        if prev is None or jid > prev["job"]:
            got[(arm, seed)] = rec
    return got


def read_manifest(path):
    if not path or not os.path.exists(path):
        return None
    d = {"tensors": [], "armspec": {}, "isolated": {}, "raw": {}}
    for ln in open(path):
        p = ln.split()
        if not p:
            continue
        if p[0] == "TENSOR" and len(p) >= 4:
            d["tensors"].append((int(p[1]), p[2], int(p[3])))
        elif p[0] == "ARMSPEC" and len(p) >= 3:
            d["armspec"][p[1]] = " ".join(p[2:])
        elif p[0] == "ISOLATED" and len(p) >= 3:
            d["isolated"][p[1]] = p[2]
        elif len(p) >= 2:
            d["raw"][p[0]] = " ".join(p[1:])
    return d


def resolve_manifest(runsdir, explicit):
    """DEFAULT resolution, so the documented ONE-argument invocation cannot be
    got wrong (the 164.2 pattern).  Returns (path, how)."""
    if explicit:
        return explicit, "explicit --manifest"
    for cand, how in (
        (os.path.join(runsdir, "cvi1", "PARTITION-MANIFEST.txt"), "<runsdir>/cvi1/"),
        (os.path.join(runsdir, "cvi1-PARTITION-MANIFEST.txt"), "<runsdir>/cvi1-"),
    ):
        if os.path.exists(cand):
            return cand, how
    return None, "not found"

# the nomination re-derivation -- reads cvg1's OWN probe records, shares no
# code with cVG1's scorer.  Used by --selftest; SKIPs when unreachable.
# =============================================================================
def rederive_nomination(runsdir):
    """-> (share, flip, npinned) or None.  share/flip keyed as in NOM_*."""
    base = None
    for cand in (os.path.join(runsdir, "cvg1"), runsdir):
        if cand and os.path.isdir(cand) and any(
                d.startswith("probe_cvg1-k01-s") for d in os.listdir(cand)):
            base = cand
            break
    if base is None:
        return None
    dirs = sorted(d for d in os.listdir(base) if d.startswith("probe_cvg1-k01-s"))
    if not dirs:
        return None
    # 1-based index -> name, from THIS scorer's frozen manifest expectation
    share_acc, flip_acc, npin = {}, {}, 0
    SETS = {"23": [22], "20": [19], "14": [13], "17": [16],
            "14_17_20": [13, 16, 19], "22": [21], "25": [24]}
    nseeds = 0
    for d in dirs:
        p = os.path.join(base, d, "probe.jsonl")
        if not os.path.exists(p):
            continue
        recs = []
        for ln in open(p):
            ln = ln.strip()
            if ln:
                try:
                    recs.append(json.loads(ln))
                except ValueError:
                    pass
        pin = [r for r in recs
               if r.get("n_beta", 0) > 0 and r.get("n_at_lo", 0) == r.get("n_beta")
               and isinstance(r.get("m_tensor"), list)
               and len(r["m_tensor"]) == NTENS]
        if not pin:
            continue
        nseeds += 1
        npin += len(pin)
        mass = [0.0] * NTENS
        fl = dict((k, 0) for k in SETS)
        for r in pin:
            b2 = r.get("pt_b2", 0.9)
            L = [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])]
            S = sum(L)
            s0 = 1 if S > 0 else (-1 if S < 0 else 0)
            for i, v in enumerate(L):
                mass[i] += abs(v)
            for k, idxs in SETS.items():
                Sr = S - sum(L[i] for i in idxs)
                sr = 1 if Sr > 0 else (-1 if Sr < 0 else 0)
                if sr != s0:
                    fl[k] += 1
        tm = sum(mass) or 1.0
        for i in range(NTENS):
            share_acc[i] = share_acc.get(i, 0.0) + mass[i] / tm
        for k in SETS:
            flip_acc[k] = flip_acc.get(k, 0.0) + fl[k] / len(pin)
    if not nseeds:
        return None
    share = dict((i, v / nseeds) for i, v in share_acc.items())
    flip = dict((k, v / nseeds) for k, v in flip_acc.items())
    return share, flip, npin, nseeds

def score(runsdir, csvpath, manifest_path):
    print("=" * 78)
    print(" cvi1 -- THE CAMPAIGN'S FIRST ISOLATION INTERVENTION OFF ResNet")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-4s %s" % (a, SPEC[a]))
    print(" PRIMARY: DELTA_ID = plateau5(ISO) - plateau5(CTL), both 3-seed")
    print("          IN-BATCH means from the runs' OWN .out files.")
    print("          plateau5 = mean TEST accuracy over epochs %d..%d."
          % (EPOCHS - 5, EPOCHS - 1))
    print("          TRAIN is printed beside TEST at every arm.")
    print("=" * 78)

    rows = read_corpus(csvpath)
    runs = read_runs(runsdir)

    # ---- G-STRUCT -----------------------------------------------------------
    print("\nG-STRUCT  the live model and both partitions are what the batch claims")
    mp, how = resolve_manifest(runsdir, manifest_path)
    man = read_manifest(mp)
    if man is None:
        skip("G-STRUCT: no PARTITION-MANIFEST.txt (%s)" % how)
    else:
        print("  manifest: %s (%s)" % (mp, how))
        names = [t[1] for t in man["tensors"]]
        numel = dict((t[0], t[2]) for t in man["tensors"])
        idxof = dict((t[1], t[0]) for t in man["tensors"])
        chk(len(man["tensors"]) == NTENS, "G-STRUCT %d parameter tensors" % NTENS,
            str(len(man["tensors"])))
        chk(sum(numel.values()) == TOTPAR, "G-STRUCT %d parameters" % TOTPAR,
            str(sum(numel.values())))
        w512 = [(i, n) for i, n, q in man["tensors"] if q == 512 and n.endswith(".weight")]
        chk(tuple(i for i, _ in w512) == BN512_IDX
            and tuple(n for _, n in w512) == BN512_NAMES,
            "G-STRUCT the four 512-wide BN scales are %s at %s"
            % (",".join(BN512_NAMES), ",".join(str(i) for i in BN512_IDX)),
            str(w512))
        for a, (nm, ix, nu) in ISOLATED.items():
            chk(idxof.get(nm) == ix and numel.get(ix) == nu,
                "G-STRUCT %s isolates %s at 1-based %d, numel %d" % (a, nm, ix, nu),
                "idx %s numel %s" % (idxof.get(nm), numel.get(idxof.get(nm))))
        chk(not any("shortcut" in n for n in names),
            "G-STRUCT NO parameter name contains `shortcut` (no residual projection)")
        for a in BLOCKWISE_ARMS:
            spec = man["armspec"].get(a, "")
            m = re.search(r"SIZES (\S+) PARAMS (\S+)", spec)
            if not m:
                chk(False, "G-STRUCT %s ARMSPEC line present" % a, spec[:60])
                continue
            sizes = [int(x) for x in m.group(1).split(",")]
            pars = [int(x) for x in m.group(2).split(",")]
            chk(sizes == WANT_SIZES and pars == WANT_PARAMS,
                "G-STRUCT %s composes to sizes %s params %s" % (a, WANT_SIZES, WANT_PARAMS),
                "sizes %s params %s" % (sizes, pars))
            chk(SPEC[a] in spec, "G-STRUCT %s ARMSPEC carries the registered spec string" % a)
        chk(man["armspec"].get("ISO", "x") != man["armspec"].get("CTL", "y"),
            "G-STRUCT ISO and CTL are DIFFERENT partitions")

    # ---- G-SOUND / G-ENV -----------------------------------------------------
    print("\nG-SOUND   12 runs, %d epochs each, RUN_DONE, no traceback" % EPOCHS)
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            if r is None:
                chk(False, "G-SOUND %s-s%d present" % (a, s), "missing")
                continue
            chk(r["n_epochs"] == EPOCHS and not r["traceback"]
                and r["plateau5"] is not None and r["run_done"],
                "G-SOUND %s-s%d" % (a, s),
                "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                % (r["n_epochs"], r["run_done"], r["traceback"],
                   fmt(r["plateau5"]), fmt(r["train5"])))

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln)
                      for r in runs.values() for ln in r["env"]))
    chk(len(envs) == 1, "G-ENV one distinct ENV line across all runs",
        "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
        for tok in ("AUGMENT=%s" % AUG, "BETA_CLIP=%s" % CLIP, "PROBE=%d" % PROBE):
            chk(tok in ln, "G-ENV ENV line carries %s" % tok)
    pts = [ln for r in runs.values() for ln in r["pt"]]
    chk(len(runs) > 0 and len(pts) == len(runs) and all(PT_RE.match(p) for p in pts),
        "G-ENV every run printed a matching `PROBE_TENSOR: on ... tensors=%d` line" % NTENS,
        "%d lines / %d runs" % (len(pts), len(runs)))
    tcnt = {}
    for p in pts:
        m = PT_RE.match(p)
        if m:
            tcnt[m.group(1)] = tcnt.get(m.group(1), 0) + 1
    if tcnt:
        print("        PROBE_TENSOR types: %s" % tcnt)
        chk(tcnt.get("scalar") == 3 and tcnt.get("layerwise") == 3
            and tcnt.get("blockwise") == 6,
            "G-ENV types are scalar x3, layerwise x3, blockwise x6", str(tcnt))

    # ---- the levels ----------------------------------------------------------
    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH")
    arm_v, arm_t = {}, {}
    for a in ARMS:
        v = [runs[(a, s)]["plateau5"] for s in SEEDS
             if (a, s) in runs and runs[(a, s)]["plateau5"] is not None]
        t = [runs[(a, s)]["train5"] for s in SEEDS
             if (a, s) in runs and runs[(a, s)]["train5"] is not None]
        arm_v[a], arm_t[a] = v, t
        if v:
            print("  %-4s TEST %.4f  sd %s  range %.3f  n=%d   TRAIN %s"
                  % (a, mean(v), fmt(sd(v), 3), max(v) - min(v), len(v), fmt(mean(t))))
            print("       %s" % SPEC[a])
            print("       seeds: %s"
                  % ", ".join("s%d %.4f" % (s, runs[(a, s)]["plateau5"])
                              for s in SEEDS if (a, s) in runs
                              and runs[(a, s)]["plateau5"] is not None))
        else:
            print("  %-4s NO USABLE RUN   %s" % (a, SPEC[a]))

    if not all(len(arm_v[a]) == len(SEEDS) for a in ARMS):
        print("\nFINAL: INCOMPLETE -- %s"
              % ", ".join("%s %d/%d" % (a, len(arm_v[a]), len(SEEDS)) for a in ARMS))
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2

    # ---- the noise floor ------------------------------------------------------
    print("\nFLOOR     SIGMA, monotone-conservative under corpus drift")
    ss = sum(sum((x - mean(arm_v[a])) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    s_nar_live = s_vgg_live = None
    if rows:
        s_nar_live = pooled_sigma(rows, lambda r: r["network"] == "ResNet18_c100"
                                  and r["dataset"] == "CIFAR100")[0]
        s_vgg_live = pooled_sigma(rows, lambda r: r["network"] == NET)[0]
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    if s_nar_live:
        cands.append(("SIGMA_NARROW_live", s_nar_live))
    if s_vgg_live:
        cands.append(("SIGMA_VGG_live", s_vgg_live))
    for nm, v in cands:
        print("  %-20s %.6f" % (nm, v))
    which, sigma_used = max(cands, key=lambda kv: kv[1])
    se = sigma_used * math.sqrt(2.0 / 3.0)
    print("  SIGMA_USED           %.6f  (= %s; max is registered and conservative)"
          % (sigma_used, which))
    print("  SE_ARM_DIFF          %.6f  = SIGMA_USED * sqrt(2/3)  (df_inbatch %d)"
          % (se, df))

    # ---- G-FLOOR / G-CEIL / G-DIVERGE ------------------------------------------
    print("\nG-FLOOR / G-CEIL / G-DIVERGE")
    hi = max(mean(arm_v[a]) for a in ARMS)
    lo = min(mean(arm_v[a]) for a in ARMS)
    floor_ok = chk(hi >= FLOOR_MIN,
                   "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE),
                   "max %.4f, margin over chance %+.4f pp" % (hi, hi - CHANCE))
    chk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)
    div = []
    for a in ARMS:
        rng = max(arm_v[a]) - min(arm_v[a])
        if not chk(rng <= DIVERGED_BAR,
                   "G-DIVERGE %s seed range <= %.2f pp" % (a, DIVERGED_BAR),
                   "range %.4f" % rng):
            div.append(a)
    print("  margins over chance: %s"
          % ", ".join("%s %+.4f" % (a, mean(arm_v[a]) - CHANCE) for a in ARMS))
    print("  lowest arm mean %.4f  (registered floor prediction was ~%.1f)" % (lo, CVG1_K01))

    # ---- the contrasts ---------------------------------------------------------
    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    DELTA_ID = M["ISO"] - M["CTL"]
    D_ISO = M["ISO"] - M["k01"]
    D_CTL = M["CTL"] - M["k01"]
    D_GAP = M["kL"] - M["k01"]
    DELTA_ID_TR = T["ISO"] - T["CTL"]
    print("\nCONTRASTS (every one WITHIN batch; no other batch's arm appears)")
    print("  PRIMARY  DELTA_ID = ISO - CTL = %.4f - %.4f = %+.4f pp = %+.2f SE"
          % (M["ISO"], M["CTL"], DELTA_ID, DELTA_ID / se))
    print("           DELTA_ID(TRAIN)               = %+.4f pp" % DELTA_ID_TR)
    print("  D_ISO    ISO - k01  = %+.4f pp = %+.2f SE" % (D_ISO, D_ISO / se))
    print("  D_CTL    CTL - k01  = %+.4f pp = %+.2f SE" % (D_CTL, D_CTL / se))
    print("  D_GAP    kL  - k01  = %+.4f pp = %+.2f SE" % (D_GAP, D_GAP / se))
    rec = D_ISO / D_GAP if abs(D_GAP) > 1e-9 else None
    rec_c = D_CTL / D_GAP if abs(D_GAP) > 1e-9 else None
    print("  DESCRIPTIVE ONLY (bounded statistics, NOT primary):")
    print("           RECOVERY     = D_ISO/D_GAP = %s" % fmt(rec))
    print("           RECOVERY_CTL = D_CTL/D_GAP = %s" % fmt(rec_c))
    print("  bars: RESCUE_BAR %+.1f pp (%.2f SE) | IDENTITY_BAR %+.1f pp (%.2f SE) | "
          "NULL_BAR %+.1f pp (%.2f SE)"
          % (RESCUE_BAR, RESCUE_BAR / se, IDENTITY_BAR, IDENTITY_BAR / se,
             NULL_BAR, NULL_BAR / se))

    # ---- the branch -------------------------------------------------------------
    print("\n" + "=" * 78)
    if not floor_ok:
        branch = "HARNESS-UNSOUND"
    elif div:
        branch = "UNRESOLVED-DIVERGED"
    elif D_ISO < RESCUE_BAR:
        branch = "NO-RESCUE-VGG"
    elif DELTA_ID >= IDENTITY_BAR:
        branch = "IDENTITY-OPERATIVE-VGG"
    elif DELTA_ID <= -IDENTITY_BAR:
        branch = "CONTROL-DOMINATES"
    elif D_CTL >= RESCUE_BAR and abs(DELTA_ID) <= NULL_BAR:
        branch = "CLASS-OPERATIVE-VGG"
    else:
        branch = "IDENTITY-ATTENUATED"

    stamps = ["GATES-CLEAN" if not GATE_FAIL else "GATES-%d-FAIL" % len(GATE_FAIL)]
    stamps.append("MAGNITUDE-NOT-SEPARATED")
    stamps.append("NOT-A-ONE-VARIABLE-ABLATION")
    stamps.append("CARDINALITY-ONE-vs-THREE")
    stamps.append("MISTUNING-NOT-EXCLUDED")
    stamps.append("SIGMA-%s" % which.replace("SIGMA_", "").replace("_", "-").upper())
    stamps.append("TRAIN-AGREES" if (DELTA_ID_TR > 0) == (DELTA_ID > 0) else "TRAIN-DISAGREES")
    if branch == "IDENTITY-ATTENUATED":
        stamps.append("MILD" if (rec_c is not None and rec and rec_c / rec <= 0.50) else "SEVERE")
    if abs(M["ISO"] - M["kL"]) <= NULL_BAR:
        stamps.append("ISO-TRACKS-KL")
    elif M["ISO"] < M["kL"]:
        stamps.append("ISO-BELOW-KL")
    else:
        stamps.append("ISO-ABOVE-KL")
    stamps.append("CTL-AT-FLOOR" if abs(D_CTL) <= NULL_BAR else "CTL-ABOVE-FLOOR")
    stamps.append("GAP-REPLICATES-CVG1" if abs(D_GAP - CVG1_D) <= 5.0
                  else "GAP-DIFFERS-FROM-CVG1")
    print("FINAL: %s | %s" % (branch, " | ".join(stamps)))
    print("=" * 78)
    print("BETWEEN-BATCH, NON-GATING: cvg1 measured k01 %.4f / kL %.4f / D %+.4f at this"
          % (CVG1_K01, CVG1_KL, CVG1_D))
    print("  exact cell on seeds {31,32,33}.  cvi1 is on {41,42,43}.  BATCH is the unit of")
    print("  replication (F(62,85)=5.47); NO cvg1 arm enters ANY cvi1 contrast above.")
    if GATE_FAIL:
        print("\nGATE FAILURES (%d): %s" % (len(GATE_FAIL), "; ".join(GATE_FAIL)))
        print("A failed gate SUSPENDS the branch; it is NOT an adverse result.")

    print("\nWHAT THIS DECIDES -- as registered, before any run existed:")
    if branch == "IDENTITY-OPERATIVE-VGG":
        print("  Isolating ONE 512-parameter BatchNorm scale rescues the scalar collapse on")
        print("  a SECOND architecture family, and its numel-, width-, class- and")
        print("  depth-adjacent-matched twin does NOT.  The isolation result is no longer")
        print("  ResNet-only: 195.8's caveat is lifted FOR THE IDENTITY CLAIM.  Note the")
        print("  carrier CARDINALITY differs across families (3 on ResNet18, 1 on VGG11_bn),")
        print("  so what generalises is the MECHANISM, not the tensor list.")
    elif branch == "CLASS-OPERATIVE-VGG":
        print("  ANY deep 512-wide BatchNorm scale rescues on VGG; tensor IDENTITY is NOT")
        print("  operative here.  Either cdep1's ResNet IDENTITY-OPERATIVE is")
        print("  architecture-specific, or the operative variable is POSITION/CLASS rather")
        print("  than identity and cdep1's [59,3] control was matched on the wrong axis.")
        print("  This is a result AGAINST the campaign's current reading.  Report it as one.")
    elif branch == "NO-RESCUE-VGG":
        print("  194.5's nomination is REFUTED BY INTERVENTION.  The mean|L| / remainder-sign")
        print("  statistic that picked ResNet's carriers does NOT predict intervention off")
        print("  ResNet, and the campaign's nomination METHOD loses its claim to generality.")
        print("  Every sentence that leans on a nominated carrier set must say so.")
    elif branch == "CONTROL-DOMINATES":
        print("  Unexpected under EVERY registered account.  Report and stop.")
    elif branch == "IDENTITY-ATTENUATED":
        print("  A rescue is present and identity moves it, but by less than half this")
        print("  cell's own gap.  Identity is CONTRIBUTORY, not decisive, on VGG.")
    elif branch == "HARNESS-UNSOUND":
        print("  All arms under %.2f pp: a statement about the harness, not architecture." % FLOOR_MIN)
    else:
        print("  A bimodal arm must be reported, not averaged.  No branch is taken.")

    print("\nWHAT IT DOES NOT LICENSE, UNCONDITIONALLY:")
    print("  * NOT identity vs MAGNITUDE.  bn8 carries %.4f of the mean|L| mass and bn7"
          % NOM_SHARE["bn8.weight"])
    print("    %.4f; no carrier-free 512-wide VGG BN scale can be magnitude-matched to it."
          % NOM_SHARE["bn7.weight"])
    print("  * NOTHING about residual connections.  VGG differs from ResNet in depth,")
    print("    channel schedule and downsampling too.  PlainNet18 is still not built.")
    print("  * RULE 11: no argmax is located on VGG.  D_ISO is a LOWER bound.")
    return 1 if GATE_FAIL else 0

def selftest(csvpath, runsdir):
    print("cVI1 --selftest: the frozen premises, the nomination, the noise floor,")
    print("the bars, the branch map and the design.  Every corpus reader EXCLUDES")
    print("the `cvi1-` prefix, so each premise is invariant under this batch's own ingest.\n")

    print("ARITHMETIC AND REGEX")
    chk(abs(SIGMA_PRIOR * math.sqrt(2 / 3) - SE_PRIOR) < 1e-6,
        "SE_PRIOR = SIGMA_PRIOR*sqrt(2/3)", "%.6f" % (SIGMA_PRIOR * math.sqrt(2 / 3)))
    chk(SIGMA_PRIOR == max(SIGMA_VGG, SIGMA_NARROW),
        "SIGMA_PRIOR = max(SIGMA_VGG, SIGMA_NARROW), NOT SIGMA_WIDE (disclosed departure)")
    chk(IDENTITY_BAR > RESCUE_BAR > NULL_BAR > 0, "IDENTITY_BAR > RESCUE_BAR > NULL_BAR > 0")
    chk(FLOOR_MIN > CHANCE, "FLOOR_MIN is above chance", "%.2f > %.2f" % (FLOOR_MIN, CHANCE))
    chk(EPTR_RE.search("Epoch 99, Train Accuracy: 35.65 %, Test Accuracy: 35.68 %") is not None,
        "epoch-line regex matches the harness's print")
    chk(OUT_RE.match("cvi1-ISO-s41-1234567.out") is not None
        and OUT_RE.match("cvi1-CTL-s43-1.out") is not None
        and OUT_RE.match("cvg1-k01-s31-1.out") is None
        and OUT_RE.match("ciso2-ISO-s31-1.out") is None,
        "run-file regex matches cvi1 ONLY (not cvg1, not ciso2)")
    chk(PT_RE.match("PROBE_TENSOR: on every=100 type=blockwise tensors=26 meta_alg=Lion "
                    "momentum_param=0.99 Lion_beta2=0.9 dir=/x") is not None,
        "PROBE_TENSOR regex matches a 26-tensor BLOCKWISE VGG line")
    chk(PT_RE.match("PROBE_TENSOR: on every=100 type=blockwise tensors=62 meta_alg=Lion "
                    "momentum_param=0.99 Lion_beta2=0.9 dir=/x") is None,
        "PROBE_TENSOR regex REJECTS a 62-tensor ResNet line")

    print("\nTHE DESIGN")
    chk(set(SPEC) == set(ARMS) and len(ARMS) == 4, "four arms")
    chk(SPEC["k01"] == "scalar" and SPEC["kL"] == "layerwise",
        "the anchors are scalar and layerwise, IN BATCH")
    chk(SPEC["ISO"].startswith("sets:") and SPEC["CTL"].startswith("sets:"),
        "both intervention arms use the sets: grammar")
    chk(SPEC["ISO"] != SPEC["CTL"], "ISO and CTL are different partitions")
    a, b = SPEC["ISO"].split("/")[-1], SPEC["CTL"].split("/")[-1]
    chk("," not in a and "," not in b,
        "each intervention arm isolates EXACTLY ONE tensor", "%s vs %s" % (a, b))
    chk(ISOLATED["ISO"][2] == ISOLATED["CTL"][2] == 512,
        "ISO and CTL isolate the SAME numel (512) -- a numel-matched twin")
    chk(a in BN512_NAMES and b in BN512_NAMES,
        "both isolated tensors are among the four 512-wide BN scales -- class- and width-matched")
    chk(abs(ISOLATED["ISO"][1] - ISOLATED["CTL"][1]) == 3,
        "they are ADJACENT in the BN-scale ordering (1-based %d vs %d)"
        % (ISOLATED["ISO"][1], ISOLATED["CTL"][1]))
    chk(sum(WANT_SIZES) == NTENS and WANT_SIZES == [NTENS - 1, 1],
        "both partitions are [%d,1] and total %d tensors" % (NTENS - 1, NTENS))
    chk(sum(WANT_PARAMS) == TOTPAR, "the partition's parameter totals sum to %d" % TOTPAR)
    chk(len(set(SEEDS)) == 3 and min(SEEDS) >= 41,
        "three FRESH seeds %s" % (SEEDS,))

    print("\nTHE PRIMARY IS BOUNDED IN NEITHER DIRECTION")
    print("  DELTA_ID = plateau5(ISO) - plateau5(CTL): a difference of two free accuracies.")
    print("  Registered accounts place it at ~+25..+31 (IDENTITY-OPERATIVE), ~0")
    print("  (CLASS-OPERATIVE, NO-RESCUE) and negative-without-limit (CONTROL-DOMINATES).")
    print("  RECOVERY = D_ISO/D_GAP IS bounded at 1 and would sit AT its ceiling under the")
    print("  account being confirmed -- which is exactly why it is DESCRIPTIVE, not primary.")
    chk(True, "the primary is a DIFFERENCE, the bounded ratio is descriptive only")

    print("\nCORPUS PREMISES, re-derived (never quoted):")
    rows = read_corpus(csvpath)
    if rows is None:
        skip("corpus premises -- %s not readable" % csvpath)
    else:
        note("corpus rows with `cvi1-` excluded", str(len(rows)))
        cnt = {}
        for r in rows:
            cnt[r["network"]] = cnt.get(r["network"], 0) + 1
        note("network census", ", ".join("%s %d" % kv for kv in sorted(cnt.items())))
        chk(cnt.get(NET, 0) >= 6, "%s is IN the corpus (cvg1 landed): %d rows" % (NET, cnt.get(NET, 0)))
        chk(not any(r["run"].startswith("cvi1-") for r in rows),
            "ZERO cvi1 rows survive the exclusion (the reader works)")

        # constructive proof that a concurrent 250-epoch ingest cannot move any premise
        e250 = [r for r in rows if r.get("epochs_requested") == "250"]
        chk(all(not std_cell(r) for r in e250),
            "CONSTRUCTIVE: all %d 250-epoch rows (incl. any ciso2 ingest) are EXCLUDED by "
            "std_cell -- the noise floor cannot move under them" % len(e250))

        sV, dV, cV, mV, _s, _t = pooled_sigma(rows, lambda r: r["network"] == NET)
        sN, dN, cN, mN, _s, _t = pooled_sigma(
            rows, lambda r: r["network"] == "ResNet18_c100" and r["dataset"] == "CIFAR100")
        sW, dW, cW, mW, ssW, tW = pooled_sigma(rows)
        print("       SIGMA_VGG    %s df %s cells %s members %s"
              % (fmt(sV, 6), dV, cV, mV))
        print("       SIGMA_NARROW %s df %s cells %s members %s"
              % (fmt(sN, 6), dN, cN, mN))
        print("       SIGMA_WIDE   %s df %s cells %s members %s"
              % (fmt(sW, 6), dW, cW, mW))
        chk(sV is not None and abs(sV - SIGMA_VGG) < 1e-6,
            "SIGMA_VGG re-derived == frozen %.6f (invariant: only a VGG batch moves it)"
            % SIGMA_VGG, fmt(sV, 6))
        for nm, live, frozen in (("SIGMA_NARROW", sN, SIGMA_NARROW),
                                 ("SIGMA_WIDE", sW, SIGMA_WIDE)):
            if live is None:
                continue
            if abs(live - frozen) < 1e-6:
                chk(True, "%s re-derived == frozen %.6f" % (nm, frozen))
            else:
                note("%s HAS DRIFTED: frozen %.6f, live %.6f -- NOT a failure; SIGMA_USED "
                     "takes the max at score time, so drift can only WIDEN the bars in SE"
                     % (nm, frozen, live))
        if tW and ssW:
            frac = tW[0][0] / ssW
            print("       SIGMA_WIDE's top cell: %s / %s n=%d, %.2f%% of total SS"
                  % (tW[0][1], tW[0][2], tW[0][3], 100 * frac))
            chk(frac > 0.95,
                "SIGMA_WIDE is >95%% ONE bimodal cell -- the disclosed reason it is NOT used",
                "%.4f" % frac)
        chk(sW > SIGMA_PRIOR, "the naive max() precedent would have picked WIDE -- the "
                              "departure is MATERIAL, not cosmetic")

        g = corpus_gaps(rows)
        print("       scalar -> layerwise gaps at the standard cell:")
        for k in sorted(g):
            s, l, d, ns, nl = g[k]
            print("         %-16s %-14s %7.3f (n%3d) -> %7.3f (n%3d)   D %+8.3f"
                  % (k[0], k[1], s, ns, l, nl, d))
        vg = g.get((NET, DSET))
        if vg:
            chk(abs(vg[0] - CVG1_K01) < 5e-3 and abs(vg[1] - CVG1_KL) < 5e-3
                and abs(vg[2] - CVG1_D) < 5e-3,
                "THIS CELL's own levels re-derive == frozen (k01 %.4f, kL %.4f, D %+.4f)"
                % (CVG1_K01, CVG1_KL, CVG1_D),
                "%.4f / %.4f / %+.4f" % vg[:3])
            chk(IDENTITY_BAR <= vg[2] / 2.0 + 1e-9,
                "IDENTITY_BAR (%.1f) <= HALF this cell's own measured gap" % IDENTITY_BAR,
                "half-gap %.4f" % (vg[2] / 2.0))
            chk(RESCUE_BAR <= vg[2] / 3.0 + 1e-9,
                "RESCUE_BAR (%.1f) <= A THIRD of this cell's own measured gap" % RESCUE_BAR,
                "third-gap %.4f" % (vg[2] / 3.0))
        unc = [(v[2], k) for k, v in g.items() if k[1] != "CIFAR10"]
        if unc:
            mn, mnk = min(unc)
            note("cVG1's anchor -- a quarter of the smallest UNCOMPRESSED corpus gap -- is "
                 "now %.3f, and the smallest uncompressed gap is %s/%s ITSELF (%.3f).  When "
                 "cVG1 froze that anchor the minimum was ResNet18_tin's +40.975 and VGG was "
                 "not in the corpus.  APPLIED TO THIS BATCH THE ANCHOR IS CIRCULAR, so it is "
                 "reported and NOT used as a bar."
                 % (mn / 4.0, mnk[0], mnk[1], mn))
            chk(RESCUE_BAR >= mn / 4.0,
                "RESCUE_BAR (%.1f) is at least as STRICT as cVG1's quarter-anchor (%.3f) -- "
                "a stricter rescue bar makes NO-RESCUE-VGG EASIER to trigger, i.e. it is "
                "conservative AGAINST the account this batch would like to confirm"
                % (RESCUE_BAR, mn / 4.0))
            chk(RESCUE_BAR >= 15.0 * SIGMA_PRIOR * math.sqrt(2.0 / 3.0),
                "RESCUE_BAR is >= 15 SE at the frozen prior -- far outside seed noise",
                "%.2f SE" % (RESCUE_BAR / (SIGMA_PRIOR * math.sqrt(2.0 / 3.0))))

        print("\n       FRESH SEEDS -- verified against the WHOLE corpus, not assumed:")
        used = sorted(set(int(r["seed"]) for r in rows if str(r.get("seed", "")).strip().isdigit()))
        print("         seeds present anywhere: %s" % used)
        chk(not (set(SEEDS) & set(used)),
            "ZERO rows ANYWHERE carry seed %s" % ", ".join(str(s) for s in SEEDS))

    print("\nTHE NOMINATION, re-derived from cvg1's OWN probe records (no shared code):")
    nom = rederive_nomination(runsdir) if runsdir else None
    if nom is None:
        skip("nomination re-derivation -- cvg1 probe records not reachable under %s" % runsdir,
             "(the launcher REQUIRES this section to have run on the cluster host)")
    else:
        share, flip, npin, nseeds = nom
        print("       %d pinned scalar records over %d seeds" % (npin, nseeds))
        idxname = dict((v[1], k) for k, v in ISOLATED.items())
        order = sorted(share, key=lambda i: -share[i])
        for r_, i in enumerate(order[:4], 1):
            print("         rank %2d  idx %2d  share %.4f" % (r_, i + 1, share[i]))
        want_idx = {"bn8.weight": 23, "bn7.weight": 20, "bn5.weight": 14,
                    "bn6.weight": 17, "conv8.weight": 22}
        for nm, frozen in sorted(NOM_SHARE.items(), key=lambda kv: -kv[1]):
            live = share.get(want_idx[nm] - 1)
            chk(live is not None and abs(live - frozen) < NOM_TOL,
                "share(%s) re-derives == frozen %.4f" % (nm, frozen), fmt(live))
        chk(order[0] == 23 - 1, "bn8.weight (idx 23) is RANK 1 by mean|L| mass",
            "rank-1 idx %d" % (order[0] + 1))
        for k, frozen in sorted(NOM_FLIP.items()):
            live = flip.get(k)
            chk(live is not None and abs(live - frozen) < NOM_TOL,
                "remainder-sign flip fraction for {%s} == frozen %.4f" % (k, frozen), fmt(live))
        chk(abs(flip.get("23", 0) - 1.0) < NOM_TOL
            and abs(flip.get("14_17_20", 1) - 0.0) < NOM_TOL,
            "REMOVING bn8 ALONE flips the remainder's sign on ~all records; removing the "
            "OTHER THREE 512-wide BN scales flips NOTHING -- this is what picks the arms")

    print("\nRULE 21 PREMISE: no cvi1 run may exist yet")
    runs = read_runs(runsdir) if runsdir else {}
    chk(not runs, "no cvi1-*.out under %s" % runsdir, "%d found" % len(runs))
    if rows is not None and csvpath and os.path.exists(csvpath):
        chk(not any(r["run"].startswith("cvi1-") for r in csv.DictReader(open(csvpath))),
            "no cvi1- row in the CSV")

    print("\nTHE FLOOR GATE (164.6), stated before any run exists:")
    print("  chance %.2f pp; ceiling 100 pp; FLOOR_MIN %.2f; CEIL_MAX %.2f"
          % (CHANCE, FLOOR_MIN, CEIL_MAX))
    print("  k01 ~ %.1f, kL ~ %.1f (cvg1's OWN measurement at this exact cell)"
          % (CVG1_K01, CVG1_KL))
    print("  ISO ~ 55-66 (IDENTITY/CLASS-OPERATIVE) or ~ %.0f (NO-RESCUE)" % CVG1_K01)
    print("  CTL ~ 55-66 (CLASS-OPERATIVE) or ~ %.0f (IDENTITY-OPERATIVE)" % CVG1_K01)
    chk(CVG1_K01 - CHANCE >= 25.0,
        "the LOWEST level ANY registered account predicts for ANY arm is >= 25 pp clear of "
        "chance", "%.4f pp margin" % (CVG1_K01 - CHANCE))
    chk(100.0 - CVG1_KL >= 25.0,
        "the HIGHEST level ANY registered account predicts is >= 25 pp clear of the ceiling",
        "%.4f pp margin" % (100.0 - CVG1_KL))
    print("  NO ARM IS PREDICTED AT OR NEAR THE FLOOR OR THE CEILING UNDER ANY ACCOUNT.")
    print("  The only route to a floored batch is total training failure -> G-FLOOR.")

    print("\nTHE BRANCH MAP IS TOTAL -- every (D_ISO, DELTA_ID, D_CTL) lands somewhere:")
    cases = [
        (5.0, 0.0, 1.0, "NO-RESCUE-VGG"),
        (30.0, 28.0, 1.0, "IDENTITY-OPERATIVE-VGG"),
        (28.0, 0.5, 27.5, "CLASS-OPERATIVE-VGG"),
        (12.0, -20.0, 32.0, "CONTROL-DOMINATES"),
        (25.0, 8.0, 17.0, "IDENTITY-ATTENUATED"),
        (25.0, 3.0, 22.0, "IDENTITY-ATTENUATED"),
    ]
    for d_iso, d_id, d_ctl, want in cases:
        if d_iso < RESCUE_BAR:
            got = "NO-RESCUE-VGG"
        elif d_id >= IDENTITY_BAR:
            got = "IDENTITY-OPERATIVE-VGG"
        elif d_id <= -IDENTITY_BAR:
            got = "CONTROL-DOMINATES"
        elif d_ctl >= RESCUE_BAR and abs(d_id) <= NULL_BAR:
            got = "CLASS-OPERATIVE-VGG"
        else:
            got = "IDENTITY-ATTENUATED"
        chk(got == want, "branch(D_ISO %+.1f, DELTA_ID %+.1f, D_CTL %+.1f) -> %s"
            % (d_iso, d_id, d_ctl, want), got)

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))
    for f in GATE_FAIL:
        print("   FAIL", f)
    return 1 if GATE_FAIL else 0


def main():
    ap = argparse.ArgumentParser(
        description="cvi1 registered scorer.  Documented invocation: "
                    "python3 analysis/cVI1_vggiso_identity_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="",
                    help="defaults to <runsdir>/cvi1/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    return score(runsdir, a.csv, a.manifest)
