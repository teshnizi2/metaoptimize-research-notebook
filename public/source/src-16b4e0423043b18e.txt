            single-draw alignment measurement in the literature is a statement

            permutation DISTRIBUTION or not at all.

            RESOLUTION.**  pp1's confound inflated no variance it did not
            already carry, and A may be read as a statement about the
            permutation distribution rather than about one draw.  **REGISTERED
            AS A NULL WITH A STATED MDE, NEVER AS PROOF OF ZERO** -- the MDE on
            sigma_perm is printed with it.
          The same two-way reading is registered for F_seed, and it is read
          against the corpus METHODS finding that SEED is null (F 1.21, p 0.213):
          agreement is a reproduction, disagreement is reportable.

  T3    THE PER-DRAW ALIGNMENT CONTRASTS.  A_p = mean_s (y[p][s] - node[s]) for

        and whether the design's promised se ~0.09 was met.  A DESIGN RECEIPT,
        NOT A FINDING.  No threshold is applied to it.

  T5    **A ON THE SCALE OF D.  CROSS-BATCH AND DESCRIPTIVE.  NO VERDICT.**
        The whole point of the second headline is "alignment carries none of D",
        so A's CI has to be read against D.  `rp1` HAS NO chunk777 ARM, by
        design -- it is the alignment leg only -- so D is imported from the two
        batches that measured it WITHIN batch at this exact configuration
        (pp1 seeds 0-2, cc1 seeds 3-5) and is RE-DERIVED FROM THE CSV AT SCORE
        TIME, never quoted from prose.  The +-0.25 pp unmodelled cross-batch
        offset applies to the IMPORT and is printed with it.  Every number in T5
        is labelled CROSS-BATCH AND DESCRIPTIVE and none of them carries a gate.

WHAT THIS SCORER WILL NOT DO
  * It will not report a NULL as "equivalence" unless the 95% CI is inside the
    band.  That is the entire T1b fix and it is not negotiable at read time.
  * It will not drop an arm, a seed or a draw post hoc for any reason, including

  * It will not present any verdict as holding at each partition's own argmax.

  * It will not use the CSV `plateau` column.  plateau5 is PRIMARY (house rule);

  * It will not pool rp1's A with pp1's A.  They are different batches and A is a

BOX REGISTRATION.  `rp1` is deliberately NOT added to `c55_neff_noise.BOXES`:
it registers no field / N_eff gate, so it needs no root there, and c55's

# --- t / F tail probabilities, stdlib only ----------------------------------
def _betacf(a, b, x):
    """Continued fraction for the incomplete beta (Lentz).  Stdlib only."""
    MAXIT, FPMIN, EPSC = 300, 1e-300, 3e-16
    qab, qap, qam = a + b, a + 1.0, a - 1.0
    c = 1.0
    d = 1.0 - qab * x / qap
    if abs(d) < FPMIN:
        d = FPMIN
    d = 1.0 / d
    h = d
    for m in range(1, MAXIT + 1):
        m2 = 2 * m
        aa = m * (b - m) * x / ((qam + m2) * (a + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        h *= d * c
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        de = d * c
        h *= de
        if abs(de - 1.0) < EPSC:
            break
    return h


def betai(a, b, x):
    """Regularised incomplete beta I_x(a, b)."""
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    lbeta = (math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b)
             + a * math.log(x) + b * math.log(1.0 - x))
    front = math.exp(lbeta)
    if x < (a + 1.0) / (a + b + 2.0):
        return front * _betacf(a, b, x) / a
    return 1.0 - math.exp(lbeta) * _betacf(b, a, 1.0 - x) / b


def t_two_sided_p(t, df):
    """P(|T_df| >= |t|)."""
    if df <= 0:
        return float("nan")
    if not math.isfinite(t):
        return 0.0
    return betai(df / 2.0, 0.5, df / (df + t * t))


def f_upper_p(f, df1, df2):
    """P(F_{df1,df2} >= f)."""
    if f <= 0:
        return 1.0
    if not math.isfinite(f):
        return 0.0
    return betai(df2 / 2.0, df1 / 2.0, df2 / (df2 + df1 * f))


def t_quantile(p, df):
    """Two-sided quantile: t such that P(|T_df| >= t) == 1 - p.  Bisection."""
    target = 1.0 - p
    lo, hi = 0.0, 400.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if t_two_sided_p(mid, df) > target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def f_quantile(p, df1, df2):
    """f such that P(F >= f) == 1 - p.  Bisection."""
    target = 1.0 - p
    lo, hi = 0.0, 1e6
    for _ in range(300):
        mid = 0.5 * (lo + hi)
        if f_upper_p(mid, df1, df2) > target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def _sem(v):
    return statistics.stdev(v) / math.sqrt(len(v)) if len(v) > 1 else 0.0


# --- run / probe naming -----------------------------------------------------
def run_name(arm, seed):

def probe_dirname(arm, seed):
    return "probe_%s_%s_s%d" % (arm, BATCH, seed)


def parse_rp1_dirname(base):
    """'probe_p101_rp1_s7' -> ('p101', 7).  (None, None) on anything else.

    The arm token is required to be a REGISTERED rp1 arm and the batch token is
    required to be rp1, so a pp1 or cc1 directory with the same arm token can
    never enter -- the failure mode c77's parser was written to refuse.
    """
    parts = base.split("_")
    if not parts or parts[0] != "probe":
        return None, None
    parts = parts[1:]
    seed = None
    if parts and parts[-1].startswith("s") and parts[-1][1:].isdigit():
        seed = int(parts[-1][1:])
        parts = parts[:-1]
    if BATCH not in parts:
        return None, None
    arm = next((p for p in parts if p in M_OF_ARM), None)
    if arm is None or seed is None:
        return None, None
    return arm, seed


# --- V0 ---------------------------------------------------------------------
def gate_V0(d, arm, row=None):
    R = records(d)
    want = M_OF_ARM.get(arm)
    n_ok = len(R) == N_RECORDS
    nb = {r.get("n_beta") for r in R}
    nb_ok = (nb == {want}) if want else False
    mins = [r["beta_true_min"] for r in R]
    maxs = [r["beta_true_max"] for r in R]
    moved = (max(maxs) - min(mins)) > 1e-9
    ep_ok, ep_d, ep_r = None, None, None
    if row is not None:
        ep_d, ep_r = row.get("epochs_done", ""), row.get("epochs_requested", "")
        ep_ok = (ep_d == str(EPOCHS) and ep_r == str(EPOCHS))
    return dict(arm=arm, n_records=len(R),
                n_beta=sorted(x for x in nb if x is not None),
                n_ok=n_ok, nb_ok=nb_ok, moved=moved, ep_ok=ep_ok,
                epochs_done=ep_d, epochs_requested=ep_r,
                span=max(maxs) - min(mins),
                ok=(n_ok and nb_ok and moved and (ep_ok is not False)))

                    ("n_tot %s != %s" % (n_tot, want) if not tot_ok
                     else "npy shape %s != (%s,)" % (shape, n_tot)))

# --- T1 ---------------------------------------------------------------------
def score_T1(grid, node):
    """**THE PRIMARY.**  grid[p][s] -> plateau5; node[s] -> plateau5.

    Paired WITHIN run seed: d_s = mean_p grid[p][s] - node[s].  n = #seeds.
    Bands are pp1's P2 bands, transcribed unchanged.  Computed from its own two
    arms only; no other gate is consulted and none can reorder it.
    """
    seeds = [s for s in RUN_SEEDS
             if s in node and all(s in grid.get(p, {}) for p in PERM_ARMS)]
    complete = (len(seeds) == len(RUN_SEEDS))
    if not seeds:
        return dict(verdict="NO DATA", A=None, n=0, seeds=[], complete=False,
                    null_at=T1_NULL, strong_at=T1_STRONG, reading="")
    d = [statistics.mean([grid[p][s] for p in PERM_ARMS]) - node[s] for s in seeds]
    A = statistics.mean(d)
    se = _sem(d)
    df = len(d) - 1
    t = (A / se) if se > 0 else float("inf")
    if A > T1_STRONG + EPS:
        v = "ALIGNMENT HURTS"
        reading = ("grouping by OUTPUT CHANNEL is WORSE than grouping the same "
                   "number of same-sized ARBITRARY subsets of the same layer, "
                   "now with the permutation draw decoupled from the run seed.  "
                   "IT IS A WITHIN-LAYER STATEMENT: permnode permutes inside each "
                   "tensor and says nothing about LAYER boundaries.")
    elif A > T1_NULL + EPS:
        v = "UNDECIDED"
        reading = ("landed in the band registered UNDECIDED IN ADVANCE "
                   "(+%.2f, +%.2f]; it is NOT read as either verdict"
                   % (T1_NULL, T1_STRONG))
    elif A >= -T1_NULL - EPS:
        v = "NULL"
        reading = ("ALIGNMENT IS NOT THE CARRIER.  Randomising membership while "
                   "holding the per-tensor size multiset EXACTLY changes nothing "
                   "-- and now that is measured over THREE permutation draws with "
                   "the draw decoupled from the init, at n=6 paired seeds.  "
                   "WHETHER IT IS AN EQUIVALENCE CLAIM IS DECIDED BY T1b, NOT "
                   "HERE.  It remains a WITHIN-LAYER statement.")
    elif A > -T1_STRONG + EPS:
        v = "UNDECIDED"
        reading = ("landed in the band registered UNDECIDED IN ADVANCE "
                   "[-%.2f, -%.2f); it is NOT read as either verdict"
                   % (T1_STRONG, T1_NULL))
    else:
        v = "ALIGNMENT HELPS"
        reading = ("channel structure is a genuine benefit -- randomising "
                   "membership at identical sizes COSTS accuracy, over three "
                   "independent draws.  A WITHIN-LAYER statement.")
    return dict(verdict=v, A=A, se=se, t=t, df=df, n=len(d), d=d, seeds=seeds,
                complete=complete, p=t_two_sided_p(t, df) if df > 0 else float("nan"),
                perm_mean=statistics.mean([grid[p][s] for p in PERM_ARMS
                                           for s in seeds]),
                node_mean=statistics.mean([node[s] for s in seeds]),
                null_at=T1_NULL, strong_at=T1_STRONG, reading=reading)


# --- T1b --------------------------------------------------------------------
def score_T1b(A, se, df, band=T1_NULL):
    """THE EQUIVALENCE RULE AND THE POWER ARITHMETIC.  Printed either way.

    A NULL is only an EQUIVALENCE claim when the whole two-sided 95% CI is inside
    the band.  pp1's band (0.15) was NARROWER than pp1's realised se (0.157), so
    pp1's NULL could never have been an equivalence claim -- that is the
    registration defect this gate exists to expose, in both batches.
    """
    if A is None or se is None:
        return dict(verdict="NO DATA", ci=None, half=None, mde_z=None, mde_t=None,
                    band=band, defect=None, reading="")
    tcrit = t_quantile(0.95, df) if df > 0 else float("inf")
    half = tcrit * se
    ci = (A - half, A + half)
    inside = (ci[0] >= -band - EPS) and (ci[1] <= band + EPS)
    within_band = abs(A) <= band + EPS
    if within_band and inside:
        v = "EQUIVALENT TO ZERO WITHIN +-%.2f" % band
        reading = ("the entire 95%% CI [%+.3f, %+.3f] lies inside the registered "
                   "NULL band, so this IS an equivalence claim at +-%.2f pp and "
                   "may be written as one." % (ci[0], ci[1], band))
    elif within_band:
        v = "CONSISTENT WITH NULL, UNDERPOWERED"
        reading = ("the point estimate is inside the band but the 95%% CI "
                   "[%+.3f, %+.3f] is not, so the data are CONSISTENT with zero "
                   "and equally consistent with an effect up to %.3f pp.  This "
                   "MUST NOT be written as equivalence, and pp1's NULL had "
                   "exactly this status without saying so."
                   % (ci[0], ci[1], max(abs(ci[0]), abs(ci[1]))))
    else:
        v = "OUT OF BAND"
        reading = ("|A| exceeds the band, so the equivalence question does not "
                   "arise; T1's own verdict governs.")
    defect = dict(pp1_band=band, pp1_se=PP1_SE, defective=(PP1_SE > band),
                  ratio=PP1_SE / band)
    return dict(verdict=v, ci=ci, half=half, tcrit=tcrit, se=se, df=df,
                mde_z=MDE_Z * se, mde_t=(t_quantile(0.95, df)
                                         + t_quantile(0.60, df)) * se,
                band=band, defect=defect, reading=reading)


# --- T2 ---------------------------------------------------------------------
def score_T2(grid):
    """**THE TWO-WAY DECOMPOSITION.**  Balanced 3 x 6, one observation per cell.

    Additive random-effects model; the residual is interaction + noise and the
    two are NOT separable at one observation per cell.  Stated, not hidden.
    """
    ps = [p for p in PERM_ARMS if p in grid]
    ss = [s for s in RUN_SEEDS if all(s in grid.get(p, {}) for p in ps)]
    if len(ps) != len(PERM_ARMS) or len(ss) != len(RUN_SEEDS):
        return dict(ok=False, why=("grid is %dx%d, not %dx%d -- the registered "
                                   "estimator is the BALANCED one and this "
                                   "scorer will not substitute another"
                                   % (len(ps), len(ss), len(PERM_ARMS),
                                      len(RUN_SEEDS))))
    a, b = len(ps), len(ss)
    y = {(p, s): grid[p][s] for p in ps for s in ss}
    gm = sum(y.values()) / (a * b)
    pm = {p: sum(y[(p, s)] for s in ss) / b for p in ps}
    sm = {s: sum(y[(p, s)] for p in ps) / a for s in ss}
    ss_perm = b * sum((pm[p] - gm) ** 2 for p in ps)
    ss_seed = a * sum((sm[s] - gm) ** 2 for s in ss)
    ss_res = sum((y[(p, s)] - pm[p] - sm[s] + gm) ** 2 for p in ps for s in ss)
    ss_tot = sum((v - gm) ** 2 for v in y.values())
    df_p, df_s, df_r = a - 1, b - 1, (a - 1) * (b - 1)
    ms_p, ms_s, ms_r = ss_perm / df_p, ss_seed / df_s, ss_res / df_r
    # A ZERO residual is degenerate, not infinitely significant: an exactly flat
    # grid has no draw effect either, and F must read 0 there, not inf.
    if ms_r > 0:
        f_p, f_s = ms_p / ms_r, ms_s / ms_r
    else:
        f_p = float("inf") if ms_p > 0 else 0.0
        f_s = float("inf") if ms_s > 0 else 0.0
    p_p = f_upper_p(f_p, df_p, df_r)
    p_s = f_upper_p(f_s, df_s, df_r)
    raw_perm = (ms_p - ms_r) / b
    raw_seed = (ms_s - ms_r) / a
    v_perm, v_seed, v_e = max(raw_perm, 0.0), max(raw_seed, 0.0), ms_r
    if p_p < ALPHA:
        v_perm_txt = "THE PERMUTATION DRAW IS A REAL EFFECT"
        perm_reading = (
            "which arbitrary same-size partition you happen to draw CHANGES "
            "plateau5.  pp1 therefore folded a REAL effect into its seed "
            "variance, every single-draw alignment measurement is a statement "
            "about that draw, and the alignment null must be stated over the "
            "permutation DISTRIBUTION or not at all.")
    else:
        v_perm_txt = "THE PERMUTATION DRAW IS EXCHANGEABLE AT THIS RESOLUTION"
        perm_reading = (
            "the three draws are not distinguishable against the residual, so "
            "pp1's confound inflated no variance it did not already carry and A "
            "may be read as a statement about the permutation DISTRIBUTION.  "
            "**REGISTERED AS A NULL WITH A STATED MDE, NEVER AS PROOF OF ZERO** "
            "-- the detectable sigma_perm is printed with it.")
    if p_s < ALPHA:
        v_seed_txt = "RUN SEED IS A REAL EFFECT HERE"
        seed_reading = ("this DISAGREES with the corpus METHODS finding that "
                        "SEED is null (F %.2f, p %.3f) and is reportable as a "
                        "disagreement, not smoothed over."
                        % (CORPUS_SEED_F, CORPUS_SEED_P))
    else:
        v_seed_txt = "RUN SEED IS NULL HERE"
        seed_reading = ("this REPRODUCES the corpus METHODS finding that SEED is "
                        "null (F %.2f, p %.3f), from the inside of a batch "
                        "designed for a different question."
                        % (CORPUS_SEED_F, CORPUS_SEED_P))
    fcrit_p = f_quantile(1 - ALPHA, df_p, df_r)
    mde_sigma_perm = math.sqrt(max((fcrit_p - 1.0) * ms_r / b, 0.0))
    return dict(ok=True, a=a, b=b, gm=gm, perm_means=pm, seed_means=sm,
                ss_perm=ss_perm, ss_seed=ss_seed, ss_res=ss_res, ss_tot=ss_tot,
                df_perm=df_p, df_seed=df_s, df_res=df_r,
                ms_perm=ms_p, ms_seed=ms_s, ms_res=ms_r,
                f_perm=f_p, f_seed=f_s, p_perm=p_p, p_seed=p_s,
                var_perm=v_perm, var_seed=v_seed, var_e=v_e,
                raw_var_perm=raw_perm, raw_var_seed=raw_seed,
                truncated_perm=(raw_perm < 0), truncated_seed=(raw_seed < 0),
                sd_perm=math.sqrt(v_perm), sd_seed=math.sqrt(v_seed),
                sd_e=math.sqrt(v_e), fcrit_perm=fcrit_p,
                mde_sigma_perm=mde_sigma_perm,
                verdict_perm=v_perm_txt, reading_perm=perm_reading,
                verdict_seed=v_seed_txt, reading_seed=seed_reading)


# --- T3 ---------------------------------------------------------------------
def score_T3(grid, node):
    """PER-DRAW ALIGNMENT CONTRASTS.  NO INDEPENDENT VERDICT (A = mean_p A_p)."""
    out = {}
    for p in PERM_ARMS:
        ss = [s for s in RUN_SEEDS if s in node and s in grid.get(p, {})]
        if not ss:
            continue
        d = [grid[p][s] - node[s] for s in ss]
        out[p] = dict(A=statistics.mean(d), se=_sem(d), n=len(d),
                      t=(statistics.mean(d) / _sem(d)) if _sem(d) > 0
                      else float("inf"))
    if not out:
        return dict(per_draw={}, spread=None, straddles=None,
                    reading="NO DATA")
    vals = [v["A"] for v in out.values()]
    spread = max(vals) - min(vals)

    def band_of(x):
        if x > T1_STRONG + EPS:
            return "HURTS"
        if x > T1_NULL + EPS:
            return "UND+"
        if x >= -T1_NULL - EPS:
            return "NULL"
        if x > -T1_STRONG + EPS:
            return "UND-"
        return "HELPS"
    bands = {p: band_of(v["A"]) for p, v in out.items()}
    straddles = len(set(bands.values())) > 1
    return dict(per_draw=out, spread=spread, bands=bands, straddles=straddles,
                reading=("**NO INDEPENDENT VERDICT.**  A = mean_p A_p "
                         "algebraically, so scoring the three separately would "
                         "score one degree of freedom three times.  They are "
                         "printed so a reader can see whether a single draw, run "
                         "pp1-style, would have returned a different verdict from "
                         "the same experiment."))


# --- T4 ---------------------------------------------------------------------
def score_T4(se_here):
    """THE DESIGN RECEIPT.  No threshold, no verdict."""
    if se_here is None:
        return dict(reading="NO DATA", se=None)
    return dict(se=se_here, pp1_se=PP1_SE, ratio=se_here / PP1_SE,
                target=TARGET_SE, met=(se_here <= TARGET_SE),
                reading=("A DESIGN RECEIPT, NOT A FINDING.  pp1's realised se on "
                         "the same contrast was %.3f at n=3 with the permutation "
                         "confounded; the design promised ~%.2f here.  No "
                         "threshold is applied to this number."
                         % (PP1_SE, TARGET_SE)))


# --- T5 ---------------------------------------------------------------------
def derive_D(rows):
    """RE-DERIVE the imported D from the CSV.  Never quoted from prose."""
    out = []
    for tag, chp, ndp in D_SOURCES:
        ch = [float(r["plateau5"]) for k, r in rows.items()
              if k.startswith(chp) and r.get("plateau5", "").strip()]
        nd = [float(r["plateau5"]) for k, r in rows.items()
              if k.startswith(ndp) and r.get("plateau5", "").strip()]
        if ch and nd:
            out.append(dict(tag=tag, D=statistics.mean(ch) - statistics.mean(nd),
                            n_ch=len(ch), n_node=len(nd)))
    if not out:
        return dict(ok=False, D=None, parts=[])
    return dict(ok=True, D=statistics.mean(o["D"] for o in out), parts=out)


def score_T5(A, ci, D):
    """**CROSS-BATCH AND DESCRIPTIVE.  NO VERDICT, NO GATE.**"""
    if A is None or ci is None or D is None or D == 0:
        return dict(ok=False, reading="NO DATA")
    return dict(ok=True, D=D, A_frac=A / D,
                ci_frac=(ci[0] / D, ci[1] / D), offset=CROSS_BATCH_OFFSET,
                reading=("**CROSS-BATCH AND DESCRIPTIVE.**  rp1 has NO chunk777 "
                         "arm by design, so D is IMPORTED from the two batches "
                         "that measured it WITHIN batch at this exact "
                         "configuration and is RE-DERIVED from the CSV above.  "
                         "The +-%.2f pp unmodelled cross-batch offset applies to "
                         "the IMPORT, so these percentages carry no gate and no "
                         "verdict." % CROSS_BATCH_OFFSET))

       (r4["df_perm"], r4["df_seed"], r4["df_res"])
       == (DF_PERM, DF_SEED, DF_RESID))
    g_flat = {p: {s: 92.0 for s in RUN_SEEDS} for p in PERM_ARMS}

       r5["ok"] and r5["ss_tot"] < 1e-18)

       not score_T2({p: {s: 1.0 for s in RUN_SEEDS[:-1]}
                     for p in PERM_ARMS})["ok"])
    ck("T2 refuses a grid missing a draw",
       not score_T2({p: {s: 1.0 for s in RUN_SEEDS}
                     for p in PERM_ARMS[:-1]})["ok"])

       "will not" in score_T2({})["why"])
    # a grid with real residual, so the variance components and F have content
    import random

    r6 = score_T2(g_noise)
    ck("variance components are non-negative after truncation",
       r6["var_perm"] >= 0 and r6["var_seed"] >= 0 and r6["var_e"] >= 0)
    ck("a truncated component is FLAGGED as truncated",
       (r6["var_perm"] == 0.0) == (r6["raw_var_perm"] < 0))
    ck("F_perm is MS_perm / MS_resid",
       abs(r6["f_perm"] - r6["ms_perm"] / r6["ms_res"]) < 1e-12)
    ck("p_perm is the (2,10) upper tail of F_perm",
       abs(r6["p_perm"] - f_upper_p(r6["f_perm"], 2, 10)) < 1e-12)
    # A pure-RESIDUAL grid: a rotating (+1, 0, -1) pattern gives every draw and
    # every seed the same mean, so ALL the variance is residual.  It is the exact
    # complement of g_pure_perm and it is what makes both T2 branches testable.
    g_latin = {p_: {s_: 92.0 + [1.0, 0.0, -1.0][(i + j) % 3]
                    for j, s_ in enumerate(RUN_SEEDS)}
               for i, p_ in enumerate(PERM_ARMS)}
    r_lat = score_T2(g_latin)

       r_lat["ok"] and r_lat["ss_perm"] < 1e-18 and r_lat["ss_seed"] < 1e-18
       and r_lat["ss_res"] > 0)
    ck("both T2 branches are reachable and mutually exclusive",
       (score_T2(g_pure_perm)["verdict_perm"]
        == "THE PERMUTATION DRAW IS A REAL EFFECT")
       and (r_lat["verdict_perm"]
            == "THE PERMUTATION DRAW IS EXCHANGEABLE AT THIS RESOLUTION"))
    ck("an exactly flat grid is EXCHANGEABLE, not infinitely significant",
       score_T2(g_flat)["verdict_perm"]
       == "THE PERMUTATION DRAW IS EXCHANGEABLE AT THIS RESOLUTION"
       and score_T2(g_flat)["f_perm"] == 0.0)

       "NEVER AS PROOF OF ZERO" in r_lat["reading_perm"])
    ck("the seed branch is read against the corpus METHODS finding",

    ck("the MDE on sigma_perm is reported and non-negative",
       score_T2(g_noise)["mde_sigma_perm"] >= 0)

    # --- T3
    node_t = {s: 92.0 for s in RUN_SEEDS}
    grid_t = {"p101": {s: 92.4 for s in RUN_SEEDS},
              "p202": {s: 92.0 for s in RUN_SEEDS},
              "p303": {s: 91.6 for s in RUN_SEEDS}}

    ck("T3 detects a straddle across bands", r7["straddles"] is True)

       score_T3({p: {s: 92.0 for s in RUN_SEEDS} for p in PERM_ARMS},

       "NO INDEPENDENT VERDICT" in r7["reading"])

    # --- the dirname parser, including what it must REFUSE
    ck("probe_node_rp1_s6 -> ('node', 6)",
       parse_rp1_dirname("probe_node_rp1_s6") == ("node", 6))

    ck("a cc1 dir is refused", parse_rp1_dirname("probe_node_cc1_s3") == (None, None))
    ck("pp1's perm dir is refused", parse_rp1_dirname("probe_perm_pp1_s0")
       == (None, None))
    ck("an UNREGISTERED arm token is refused",
       parse_rp1_dirname("probe_ch_rp1_s6") == (None, None))
    ck("a non-probe directory is refused",
       parse_rp1_dirname("Tensorboard_outputs") == (None, None))
    ck("run_name composes rp1-p202-s9", run_name("p202", 9) == "rp1-p202-s9")

            json.dump({"n_tot": n_tot}, open(os.path.join(d, "neg_counts.json"), "w"))
            np.save(os.path.join(d, "neg_counts.npy"),
                    np.zeros(arr_n if arr_n is not None else n_tot, dtype=dtype))
            return d
        ck("int32 array of the right length PASSES V0.3",

        ck("V0.3 reads the true shape from the header",

           == (M_OF_ARM["node"],))
        ck("int64 also passes on length",

                           arr_n=M_OF_ARM["p303"] - 1), "p303")["ok"])

        dm = os.path.join(td, probe_dirname("node", 7))
        os.makedirs(dm)
        ck("absent neg_counts.json FAILS V0.3 (bf8's failure)",
           not gate_V03(dm, "node")["ok"])

    # --- the discipline this scorer registered against itself
    ck("the scorer refuses equivalence without a CI inside the band",
       "unless the 95% CI is inside" in __doc__)
    ck("the scorer refuses post-hoc arm dropping",
       "will not drop an arm" in __doc__)
    ck("the scorer refuses an unbalanced T2", "will not score T2" in __doc__)
    ck("the within-layer limit is documented", "WITHIN-LAYER statement" in __doc__)
    ck("the architecture-is-irrelevant sentence is explicitly refused",

    ck("the off-argmax limit is documented", "own 3e-4" in __doc__)

    print("selftest: %d/%d %s" % (p, n, "PASS" if p == n else "FAIL"))

        print("    D could not be re-derived from the CSV -- T5 is not printed.")
    else:
        for part in dref["parts"]:
            print("      D(%s) = %+.3f pp   (within batch, n=%d vs %d, re-derived "
                  "from the CSV)" % (part["tag"], part["D"], part["n_ch"],
                                     part["n_node"]))
        print("      D pooled over %d batches = %+.3f pp"
              % (len(dref["parts"]), dref["D"]))
        if s1["A"] is not None:
            b = score_T1b(s1["A"], s1["se"], s1["df"])
            s5 = score_T5(s1["A"], b["ci"], dref["D"])
            print("      A / D = %+.1f%%;  95%% CI = [%+.1f%%, %+.1f%%] of D"
                  % (100 * s5["A_frac"], 100 * s5["ci_frac"][0],
                     100 * s5["ci_frac"][1]))
            print("      pp1's CI on the same ratio was [-55%, +51%] of D.")
            print("    %s" % s5["reading"])

    print("\n" + "=" * 78)
    if s1["A"] is not None:
        print("VERDICT  T1 %s%s | T1b %s | T2 draw: %s"
              % ("BOX-BOUND, " if any_bound else "", s1["verdict"],
                 score_T1b(s1["A"], s1["se"], s1["df"])["verdict"],
                 s2["verdict_perm"] if s2["ok"] else "NOT SCORED"))
    else:
        print("VERDICT  NO DATA")
    print("=" * 78)
    return 0


if __name__ == "__main__":
    sys.exit(main())
