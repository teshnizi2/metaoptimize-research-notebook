set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
# Cycle 7 §2: the two hierarchical operators are NOT the same knob.
#
# M0 shrink  : beta <- beta - lam*(beta - mean)          -> changes the SHARED beta's drift rate
# M1 additive: beta <- beta_prev + dm + r*(d - dm)       -> mean(beta_new) = mean(beta_prev) + dm,
#                                                           EXACTLY independent of r.
# So additive has no direct channel to the drift rate and acts ONLY through the
# per-group spread. This ladder measures both readouts (beta_true_max-min = spread,
# d(beta)/dstep = drift) across r, at the alpha0 and arm where ad-l-* runs at 100 epochs.
#
# PREDICTIONS ON RECORD:
#  (a) mean-beta drift is ~equal across all r  (falsified if it moves monotonically with r)
#  (b) spread grows monotonically with r; r=0 gives spread == 0 exactly
#  (c) r=1.0 reproduces p2-lay-plain (drift 6.8e-4 over 0-1k, spread 10.8, 74.29 @ 20ep)
#  (d) the clip (-15) binds only at large r -- watch beta_true_min
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 20 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ sbatch --parsable --job-name=$1 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 --time=00:45:00 \
  --export="ALL,AUGMENT=1,$C,HIER=additive,ETA_RATIO=$2,PROBE=100,PROBE_DIR=$R/bdrift4/$1" \
  jobs/run_cifar.sh $B --stepsize-groups layerwise --seed 0 \
  --save-directory $R/bdrift4 --run-name $1 >/dev/null; }
sub p4-ad-r0    0
sub p4-ad-r003  0.03
sub p4-ad-r01   0.1
sub p4-ad-r03   0.3
sub p4-ad-r1    1.0
sleep 3
echo "P4_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -c '^p4-')"
echo "TOTAL_QUEUE=$(squeue -h -u account_b | wc -l)"
