# THE DOCUMENTED SCORER INVOCATION -- no second argument to forget:
#     python3 analysis/cVI1_vggiso_identity_score.py <runsdir>
#
# -----------------------------------------------------------------------------
# THE QUESTION.  195.8 drew the line this batch attacks: cvg1 lifted the ResNet
# scope caveat FOR THE GAP AND FOR THE GAP ONLY.  No isolation arm has EVER run
# off ResNet, so ciso1's rescue, cdep1's IDENTITY-OPERATIVE and ciso2's horizon
# are all still ResNet-only sentences.  cvi1 runs the intervention on VGG.
#
# THE ARMS.  m = 2 group ORDER: the big group is beta[0], the isolated group
# beta[1].  1-BASED indices, RE-DERIVED BY NAME AND NUMEL on the live model by
# guard 4, which prints the full manifest per arm.  NOTHING here is inherited
# from prose -- 187.4 records a spec string that was WRONG as first written.
#   k01   scalar                            floor anchor, IN BATCH
#   kL    layerwise                         ceiling anchor, IN BATCH
#   ISO   sets:1-22,24-26/bn8.weight   [25,1]  isolates tensor 23, 512 params
#   CTL   sets:1-19,21-26/bn7.weight   [25,1]  isolates tensor 20, 512 params
#
# WHY CTL IS A CONTROL ResNet CANNOT BUY.  ISO and CTL isolate ONE tensor each,
# of the SAME numel (512), the SAME width, the SAME class (a BatchNorm scale)
# and ADJACENT depth; their group sizes are byte-identical [25,1] and their
# spec strings differ in ONE tensor name.  On ResNet18 this is impossible: the
# carrier set there has cardinality THREE and only TWO 512-wide non-carrier BN
# scales exist, which is why cdep1 had to build a [59,3] set matched on totals.
# VGG11_bn has FOUR 512-wide BN scales -- one nominated carrier (bn8) and THREE
# matched non-carriers (bn5/bn6/bn7) -- so the twin is exact.
#
# WHAT PICKS bn8.  cvg1's OWN PROBE_TENSOR records, re-derived by the scorer's
# --selftest with no shared code: on the scalar arm's pinned records, bn8.weight
# holds 0.5783 of the mean|L| mass (rank 1 of 26; bn7 0.0021, bn5/bn6 ~0.0000),
# and removing bn8 ALONE flips the remainder sum's sign on 1.0000 of records
# while removing bn5+bn6+bn7 together flips 0.0000.  guard 1c2 REQUIRES that
# section to have RUN on this host, not to have been skipped.
#
# WHY BOTH ANCHORS ARE CARRIED.  BATCH is the unit of replication
# (F(62,85)=5.47; SEED is null, F(11,1976)=0.152).  D_ISO and D_CTL are
# differences against k01 and RECOVERY's denominator is kL - k01, so k01 and kL
# must be IN BATCH.  Splicing cvg1's arms in would be forbidden.  The free
# by-product is a BETWEEN-BATCH concordance readout against cvg1 at disjoint
# seeds, which the scorer stamps and explicitly does NOT gate on.
#
# -----------------------------------------------------------------------------
# WALLTIME AND PARTITIONS -- MEASURED ON THIS BATCH'S OWN PREDECESSOR
# cvg1's 6 runs, 100 epochs each, same net, same dataset, same cell:
#   [HOST_e8d809bd94] (L4)        19-22 min  = 11.4-13.2 s/epoch
#   [HOST_fd0c0ddbe6] (A100)      28 min     = 16.8 s/epoch
#   [HOST_3c1b003035] (2080 Ti)   61-63 min  = 36.6-37.8 s/epoch   <- worst observed
#   corpus-worst on the 2080ti nodes (CORRECTIONS 152's table) 70.8 s/epoch
# 100 epochs at the corpus-worst 70.8 s/epoch is 1.97 h.  WALL=03:00:00 is
# 1.52x that and 2.86x cvg1's own worst observed run, and it FITS gpu-short's
# 4:00:00 MaxTime -- so bin/_lib_guards.sh's composer ADDS gpu-short and the
# batch reaches its independent 12-GPU pool.  PARTS is never hard-coded here;
# it is composed by slurm_parts_for_wall and re-checked by the source-time hook
# (CORRECTIONS 178), so PARTS and WALL cannot disagree.
# COST: 12 jobs x 100 epochs.  cvg1-mixed-pool expectation ~35 min/job ->
#       7.0 GPU-h;  cvg1-worst 63 min -> 12.6 GPU-h;  corpus-worst 70.8 s/ep ->
#       23.6 GPU-h;  HARD BOUND = WALL x 12 = 36 GPU-h.
# QUEUE DEPTH IS CHECKED BEFORE COMMITTING by guard 5b, not assumed.
#
# RULE 20.  Every value below is ONE shell token of [A-Za-z0-9_.,:/-].
# analysis/argsline_guard.py IS NOT EDITED.  BETA_CLIP, PROBE and PROBE_TENSOR
# are ENVIRONMENT variables; the first two are audited from the runner's ENV

grep -q "^  PASS ZERO rows ANYWHERE carry seed 41, 42, 43" /tmp/cvi1_selftest.$$ \
  && echo "guard 1c3: the scorer's OWN fresh-seed check ran and passed" \
  || guard_fail "guard 1c3: the scorer's fresh-seed check did not pass"
echo "guard 1c: scorer sha256 $( (sha256sum "$SC" 2>/dev/null || shasum -a 256 "$SC") | awk '{print $1}')"

# ---- guard 1d: PATCH_VGGBN is present and PROVED ADDITIVE on the LIVE tree ---

import sys, re, ast
sc = open(sys.argv[1]).read()
mine = dict(zip(("k01", "kL", "ISO", "CTL"), sys.argv[2:6]))

  || guard_fail "guard 4: live partition / design / source check FAILED"
import sys, os, inspect, re
cif, out, ntens, totpar = sys.argv[1], sys.argv[2], int(sys.argv[3]), int(sys.argv[4])
clip_c, epochs, bsz, net_name = sys.argv[5], int(sys.argv[6]), int(sys.argv[7]), sys.argv[8]
specs = dict(zip(("ISO", "CTL"), sys.argv[9:11]))
sys.path.insert(0, cif); os.chdir(cif)
import torch
from build_network import build_network
from Optimizers.HF import HF
net = build_network(net_name, "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]
numel = [int(p.numel()) for _, p in net.named_parameters()]
T, tot = len(nps), sum(numel)
bad = 0
print("guard 4: live %s -- %d parameter tensors, %d parameters" % (net_name, T, tot))
if T != ntens or tot != totpar:
    print("!!! guard 4: live model is %d / %d, the batch assumes %d / %d"
          % (T, tot, ntens, totpar)); raise SystemExit(1)
if any("shortcut" in n for n in names):
    print("!!! guard 4: a parameter name contains `shortcut` -- this is not a plain conv stack"); bad += 1
else:
    print("guard 4: NO parameter name contains `shortcut` (no residual projection anywhere)")
w512 = [(i + 1, names[i]) for i in range(T) if numel[i] == 512 and names[i].endswith(".weight")]
print("guard 4: the 512-wide BN scales on the LIVE model (1-based): %s" % w512)
if [x[0] for x in w512] != [14, 17, 20, 23] or \
   [x[1] for x in w512] != ["bn5.weight", "bn6.weight", "bn7.weight", "bn8.weight"]:
    print("!!! guard 4: the four 512-wide BN scales are not at 14/17/20/23 on the live model"); bad += 1
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot, "NETWORK %s" % net_name,
         "CLIP_C %s" % clip_c, "EPOCHS %d" % epochs,
         "META_STEPS %d" % (epochs * (50000 // bsz))]
for i in range(T):
    lines.append("TENSOR %d %s %d" % (i + 1, names[i], numel[i]))
# k01 / kL never reach polish_the_stepsize_groups -- init_meta routes 'scalar' and
# 'layerwise' by exact string match.  They are checked END TO END through init_meta
# on the live class instead, which is the stronger check.
for arm, spec, want_type, want_shape in (("k01", "scalar", "scalar", ()),
                                         ("kL", "layerwise", "layerwise", (T,))):
    o = HF.__new__(HF); o.num_layers = T; o._device = torch.device("cpu")
    HF.init_meta(o, spec, nps, 1e-6)
    shapes = [tuple(b.shape) for b in o.beta]
    print("guard 4: %-4s %-10s init_meta: stepsize_type %s  beta shapes %s  rednorm %s"
          % (arm, spec, o.stepsize_type, shapes, getattr(o, "_rednorm", None)))
    if o.stepsize_type != want_type or shapes != [want_shape] or getattr(o, "_rednorm", False):
        print("!!! guard 4: %s does not route to a %s beta of shape %s without rednorm"
              % (arm, want_type, want_shape)); bad += 1
    sizes = [T] if arm == "k01" else [1] * T
    pars = [tot] if arm == "k01" else list(numel)
    lines.append("ARMSPEC %s SPEC %s M %d SIZES %s PARAMS %s GROUP0 %s GROUP1 %s"
                 % (arm, spec, 1 if arm == "k01" else T,
                    ",".join(str(x) for x in sizes), ",".join(str(x) for x in pars),
                    " ".join(names), "-"))
want_iso = {"ISO": (23,), "CTL": (20,)}
want_name = {"ISO": ("bn8.weight",), "CTL": ("bn7.weight",)}
want_sizes = {"ISO": [25, 1], "CTL": [25, 1]}
want_par = {"ISO": [tot - 512, 512], "CTL": [tot - 512, 512]}
groups = {}
for arm in ("ISO", "CTL"):
    g = HF.polish_the_stepsize_groups(None, specs[arm], nps)
    sizes = [len(x) for x in g]
    idx1 = tuple(names.index(n) + 1 for n in g[1])
    pc = [sum(numel[names.index(n)] for n in x) for x in g]
    print("guard 4: %-4s %s" % (arm, specs[arm]))
    print("         m=%d  sizes %s  params %s  isolated (1-based) %s  isolated names %s"
          % (len(g), sizes, pc, idx1, tuple(g[1])))
    print("         GROUP0 (beta[0], %d tensors): %s" % (len(g[0]), " ".join(g[0])))
    print("         GROUP1 (beta[1], %d tensors): %s" % (len(g[1]), " ".join(g[1])))
    if len(g) != 2 or sizes != want_sizes[arm] or idx1 != want_iso[arm] \
       or tuple(g[1]) != want_name[arm] or pc != want_par[arm] \
       or sorted(g[0] + g[1]) != sorted(names):
        print("!!! guard 4: %s does not compose to the registered partition" % arm); bad += 1
    groups[arm] = g
    lines.append("ARMSPEC %s SPEC %s M %d SIZES %s PARAMS %s GROUP0 %s GROUP1 %s"
                 % (arm, specs[arm], len(g), ",".join(str(s) for s in sizes),
