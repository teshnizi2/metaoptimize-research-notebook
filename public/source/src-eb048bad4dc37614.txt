set -e
cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
# CYCLE 9. The BUDGET CONTROL for the M1-additive interior optimum.
#
# Standing caveat from cycle 8: "The additive peak has no budget control. Cycle 8 sec.5
# predicts the peak LOCATION moves with the budget (spread helps early, hurts late)."
# The ladder at 100 epochs (n=5/5/2) is:
#   plain 91.23 | r=0 92.52 | r=0.03 92.96 | r=0.05 93.41 | r=0.1 92.86 | r=0.3 91.44
# If the peak is a fit-rate artefact it should MOVE TOWARD r=0 (less spread) at 300
# epochs; if it is a genuine optimum of the operator it should stay at ~0.05.
#
# alpha0=1e-6 -- NOT 1e-3 like ext300. This block must be a CONTINUATION of the
# ad-l-* ladder (gotcha 19: train.py has no lr scheduler, so epochs 1-100 of these
# runs are the identical computation to the 100-epoch cells). Changing alpha0 would
# make it a different experiment instead of an extension.
#
# --time from measurement: layerwise 300ep on L4 = 91-94 min (0.31 min/ep, n=4).
# 03:00:00 is ~1.9x headroom AND inside the gpu-short 4h cap (gotcha 12).
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 300 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ # 1=name 2=hier-env 3=seed 4=nice
  sbatch --parsable --job-name=$1-s$3 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 \
    --time=03:00:00 --nice=$4 --export="ALL,AUGMENT=1,$C,$2" \
    jobs/run_cifar.sh $B --stepsize-groups layerwise --seed $3 \
    --save-directory $R/e3a --run-name $1_s$3 >/dev/null; }
# seed 0 sweeps the whole ladder first (nice=0); seed 1 backfills behind it (nice=2000)
for s in 0 1; do
  n=0; [ $s -eq 1 ] && n=2000
  sub e3a-lplain "HIER="                        $s $n
  sub e3a-r0     "HIER=additive,ETA_RATIO=0"    $s $n
  sub e3a-r005   "HIER=additive,ETA_RATIO=0.05" $s $n
  sub e3a-r01    "HIER=additive,ETA_RATIO=0.1"  $s $n
done
sleep 5
echo "E3A_SUBMITTED=$(squeue -h -u account_a -o '%j' | grep -c '^e3a-')"
squeue -h -u account_a -o '%i %j %T %r %P' | grep '^.* e3a-'
