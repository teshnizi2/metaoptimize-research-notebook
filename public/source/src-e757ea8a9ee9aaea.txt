def _fin(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool) and math.isfinite(x)


# =============================================================================
# THE VOTE: per-tensor terms of the applied meta momentum, the decomposition check, DOM_C
# =============================================================================
def vote_terms(r, meta_alg):
    """-> (L list, applied scalar) for ONE scalar-arm record, or None if the record lacks the keys or is non-finite."""
    try:
        mp, b2 = float(r["pt_mp"]), float(r["pt_b2"])
        z, m = r["z_tensor"], r["m_tensor"]
        mom, za = float(r["mom_pre"][0][0]), float(r["z_agg"][0][0])
    except (KeyError, TypeError, IndexError, ValueError):
        return None
    if len(z) != NTENS or len(m) != NTENS or not all(_fin(x) for x in z) or not all(_fin(x) for x in m):
        return None
    if meta_alg == "Adam":
        L = [mp / (1.0 - mp) * mi + zi for mi, zi in zip(m, z)]
        app = mp * mom + za
    else:
        L = [b2 * mi + (1.0 - b2) * zi for mi, zi in zip(m, z)]
        app = b2 * mom + (1.0 - b2) * za
    if not (_fin(app) and all(_fin(x) for x in L)):
        return None
    return L, app


def _sg(x):
    return (x > 0) - (x < 0)


def decomposition(recs, meta_alg):
    """-> (n_usable, n_bad_sum, n_bad_sign, n_nonfinite).  sum_i L_i must equal the harness's applied term."""
    n = bad = bads = nonf = 0
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            nonf += 1
            continue
        L, app = v
        s = sum(L)
        scale = sum(abs(x) for x in L)
        n += 1
        # SCALE-relative: the tolerance is DECOMP_TOL of the vote's total mass sum_i |L_i|, not of the (possibly
        # near-cancelling) net term.  Landed cwd5 scalar records: worst |sum - applied| / sum|L| = 8.5e-7.
        bad += not (abs(s - app) <= DECOMP_TOL * scale or abs(s - app) <= 1e-30)
        bads += _sg(s) != _sg(app) and abs(app) > DECOMP_TOL * scale
    return n, bad, bads, nonf


def fit_carriers(recs, meta_alg, k=3):
    """top-k tensors by signed contribution sum_r s_r * L_i,r (s_r = the applied sign), on the FIT records only."""
    acc = [0.0] * NTENS
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            continue
        L, app = v
        s = _sg(app)
        for i, x in enumerate(L):
            acc[i] += s * x
    return tuple(sorted(sorted(range(NTENS), key=lambda i: (-acc[i], i))[:k]))


def dom_c(recs, meta_alg, Cset):
    """-> (share, n): share of usable records on which C alone fixes the applied sign."""
    n = d = 0
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            continue
        L, app = v
        s = _sg(app)
        sc = sum(L[i] for i in Cset)
        rest = sum(abs(L[i]) for i in range(NTENS) if i not in Cset)
        n += 1
        d += (s != 0 and _sg(sc) == s and abs(sc) > rest)
    return (d / n if n else float("nan")), n


def dom_reading(recs_by_seed, meta_alg):
    """C fitted on FIT_SEED's records, DOM_C scored on SCORE_SEEDS only; plus the DECLARED carriers on all seeds."""
    Cx = fit_carriers(recs_by_seed.get(FIT_SEED), meta_alg)
    held = [r for s in SCORE_SEEDS for r in (recs_by_seed.get(s) or [])]
    share, n = dom_c(held, meta_alg, Cx)
    alls = [r for s in SEEDS for r in (recs_by_seed.get(s) or [])]
    dshare, dn = dom_c(alls, meta_alg, CARRIER_IDX0)
    return {"C": Cx, "dom": share, "n": n, "decl_dom": dshare, "decl_n": dn}

def cell_state(Lseeds, kseeds):
    L, k = mean(Lseeds), mean(kseeds)
    if L < REF_MIN or (max(Lseeds) - min(Lseeds)) > DIVERGED_BAR:
        return "UNREADABLE"
    ncol = sum(1 for x in kseeds if x <= R50 * L)
    if 1 <= ncol <= len(kseeds) - 1:
        return "SPLIT"
    if k <= R50 * L:
        return "COLLAPSE"
    if L - k < GAP_BAR:
        return "NOGAP"
    return "PARTIAL"


def decide(states, k01_mean):
    """states: {"A1": s, "A2": s}.  -> branch token.  Pure function."""
    if k01_mean > K01_MAX:
        return "CONTROL-NOT-COLLAPSED"
    if states["A1"] == "UNREADABLE":
        return "UNRESOLVED-REFERENCE-A1"
    vals = [states[c] for c in D.CELLS]
    if "COLLAPSE" in vals:
        return "COLLAPSE-UNDER-STANDARD-RECIPE"
    if "SPLIT" in vals or "PARTIAL" in vals:
        return "PARTIAL"
    if states["A2"] == "UNREADABLE":
        return "UNRESOLVED-REFERENCE-A2"
    return "NO-COLLAPSE"


def gate_word(branch, states, doms):
    """doms: {arm: dom_reading} for the scalar arms incl. K01 (None if unreadable)."""
    if branch in ("PARTIAL", "NO-COLLAPSE"):
        return "GATE-DOES-NOT-FIRE"
    if branch != "COLLAPSE-UNDER-STANDARD-RECIPE":
        return "GATE-UNREADABLE"
    kc = doms.get(D.CONTROL)
    if kc is None or not (kc["dom"] >= DOM_BAR):
        return "GATE-UNREADABLE-DOMC-CONTROL"
    fired = [c for c in D.CELLS if states[c] == "COLLAPSE" and doms.get(D.SCALAR_OF[c]) is not None
             and doms[D.SCALAR_OF[c]]["dom"] >= DOM_BAR]
    if fired:
        return "GATE-FIRES-%s" % "+".join(fired)
    return "GATE-COLLAPSE-WITHOUT-DOMINANCE"

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound, 2 incomplete."""
    print("=" * 78)
    print(" caw1 -- THE STANDARD-RECIPE GATE.  Does the shared-step-size collapse appear with an AdamW base + Adam")
    print("         meta (the parent's pairing) at alpha-scaled decay 0.1 and 1e-2, or is it an SGDm(0.99) corner case?")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (D.NET, D.DSET, EPOCHS, ",".join(str(s) for s in SEEDS), D.AUG, D.CLIP, D.PROBE))
    for a in ARMS:
        print("   %-4s %-2s %-9s wd %-5s  %s" % (a, D.PAIR_OF[a], D.SPEC[a], D.WD[a], dict((r[0], r[6]) for r in D.ARM_TABLE)[a]))
    print(" CO-PRIMARY: G_A1 = AL1 - AS1.  Every state is WITHIN its cell.  plateau5 = mean TEST over epochs %d..%d."
          % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2).  ALPHA-INDEPENDENT DECAY IS NOT TESTED.")
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = C.read_runs(runsdir, PREFIX, ARMS, EPOCHS)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        print("  NOTE %s" % n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        for ln in LICENSE["INCOMPLETE"]:
            print("  " + ln)
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    [HARNESS-NULL] every run's OWN ARGS line == its arm's registered flags (pairing, decay, grain)")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = C.parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = dict(D.args_pairs(a, s, "<save>"))
            bad = [k for k, v in want.items() if k != "save-directory" and f.get(k, [None])[-1] != v]
            extra = sorted(set(f) - set(want))

def _e_curve(level):
    return [level if e >= EPOCHS - 5 else level * (1 - math.exp(-(e + 1) / 12.0)) for e in range(EPOCHS)]


def _e_write(d, arm, s, jid, level, n_ep=EPOCHS, run_done=True, env=_E_ENV, args=None, pt=None, offs=None):
    lines = ["NODE=[HOST_2f2f29d9e1] | JOB=%s-%s-s%d %d | AUGMENT=1" % (PREFIX, arm, s, jid),
             "ARGS: %s" % (args if args is not None else D.args_string(arm, s, _E_SAVE)), env, "NVIDIA L4, 23034 MiB"]
    lines += list(offs if offs is not None else D.OFF_LINES)
    lines.append(pt if pt is not None else "%s dir=/x/probe_%s-%s-s%d" % (D.pt_witness_prefix(arm), PREFIX, arm, s))
    for e, v in enumerate(_e_curve(level)[:n_ep]):
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(v + 20, 99.9), v))
    if run_done:
        lines.append("RUN_DONE")
    open(os.path.join(d, "%s-%s-s%d-%d.out" % (PREFIX, arm, s, jid)), "w").write("\n".join(lines) + "\n")


def _e_records(arm, seed, dominate=False, nb=None, decomp_break=False, rng=None):
    """500 synthetic PROBE_TENSOR records.  dominate: the three declared carriers carry the applied sign and outweigh
    the rest.  Terms are built so that z_i = L_i and m_i = 0 -> the decomposition is exact."""
    rng = rng or random.Random(seed * 7 + len(arm))
    pair = D.PAIR_OF[arm]
    mp, b2 = (0.9, 1.0) if pair == "AW" else (0.99, 0.9)
    nb = nb if nb is not None else (1 if D.SPEC[arm] == "scalar" else NTENS)
    out = []
    for k in range(N_RECORDS):
        L = [rng.gauss(0, 1e-6) for _ in range(NTENS)]
        if dominate:
            for i in CARRIER_IDX0:
                L[i] = 1e-4
        if pair == "AW":
            z = list(L)
            m = [0.0] * NTENS
            za = sum(L)
        else:
            z = [x / (1 - b2) for x in L]
            m = [0.0] * NTENS
            za = sum(z)
        if decomp_break:
            za = -za
        out.append({"step": k * D.PROBE, "beta": [-13.8] * nb, "z_tensor": z, "m_tensor": m, "mom_pre": [[0.0]],
                    "z_agg": [[za]], "pt_mp": mp, "pt_b2": b2, "beta_pre": [[-13.8] * nb]})
    return out


def _e_probe_templates(root):

def _e_build(d, tpl, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, dom=(), prov_over=None, man_break=False,
             rec_over=None):
    """a complete synthetic runsdir.  levels: arm -> level; over: (arm, seed) -> dict of _e_write kwargs;
    dom: arms whose records are dominated by the carriers; rec_over: (arm, seed) -> list of records."""
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    lv = dict(_E_BASE)
    lv.update(levels or {})
    jid = 7000000
    for s, j in zip(SEEDS, jit):
        for a in ARMS:
            jid += 1
            if drop == (a, s):
                continue
            L = lv[a] if not isinstance(lv[a], (list, tuple)) else lv[a][SEEDS.index(s)]
            kw = dict((over or {}).get((a, s), {}))
            _e_write(d, a, s, jid, L + (0 if isinstance(lv[a], (list, tuple)) else j), **kw)
            pd = os.path.join(d, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s))
            os.makedirs(pd, exist_ok=True)
            if rec_over and (a, s) in rec_over:
                with open(os.path.join(pd, "probe.jsonl"), "w") as fh:
                    for r in rec_over[(a, s)]:
                        fh.write(json.dumps(r) + "\n")
            else:
                os.symlink(tpl[(a, s, a in dom)], os.path.join(pd, "probe.jsonl"))
    man = D.manifest_text() + ("X\n" if man_break else "")
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(man)
    prov = {"MODE": "submit", "BUILD_NETWORK_SHA256": D.BN_SHA, "HF_SHA256": D.HF_SHA, "RUNNER_SHA256": D.RUNNER_SHA,
            "SCORER_SHA256": file_sha(os.path.abspath(__file__)), "DESIGN_SHA256": file_sha(D.__file__),
            "CWD_DESIGN_SHA256": file_sha(D.W.__file__), "COMMON_SHA256": file_sha(C.__file__)}
    prov.update(prov_over or {})
    open(os.path.join(d, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % kv for kv in prov.items()))

def _find_out(runsdir, name_prefix):
    if not runsdir or not os.path.isdir(runsdir):

    for fn in sorted(os.listdir(runsdir)):

    return None

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--csv", default=os.path.join(REPO, "results", "all_runs.csv"))
    ap.add_argument("--manifest", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir required")
    rc, _final = score(a.runsdir, a.csv, a.manifest)
    raise SystemExit(rc)


if __name__ == "__main__":
    main()
