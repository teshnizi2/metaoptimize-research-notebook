#
#   python3 analysis/crt1_attack_indep.py <runsdir> <scorer-log> [<scorer-log> ...] --csv P --tsv P [--rows P]
#
#   <runsdir>  $WS/runs on cluster_b or cluster_a-backup/runs_cluster_b on the Mac: the 36 `crt1-*.out` files at its top level

#   --rows     results/crt1_exclusion_rows_PROPOSED.tsv: the 36 ARGS_WD_BASE rows this landing owes, checked row
#              for row against a rebuild from the runs' own file names and ARGS lines.

# INDEPENDENCE, deliberately (the cwd2...cgw1 pattern):
#   * it imports NOTHING from the repo: not cRT1_retune_score.py, crt1_design.py, corpus_exclusions.py,

#

#   [0] sync integrity: one digest over every run file's name and bytes (equal on both hosts <=> the sync is exact).
#   [1] the design rebuilt from 300.3 / 300.4: 12 arms, 36 runs, the grid, the grain counts.
#   [2] every run's ARGS line rebuilt flag by flag (20 flags, in order) and its ENV line byte for byte; each run

#   [5] line-by-line agreement with each committed log.
#   [6] THE REALISED PER-STEP SHRINK a*wd per arm against a standard SGD recipe (DESCRIPTIVE; 296.4's caveat).
#   [7] what bounds the verdict: the margins of the registered reading, and DESCRIPTIVE sensitivities (sigma choice,

#   [8] the 36 exclusion rows owed at ingest, rebuilt and compared with the PROPOSED file.

def avg(v):
    return sum(v) / float(len(v))


def sdev(v):
    if len(v) < 2:
        return 0.0
    m = avg(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def med(v):
    s = sorted(v)
    n = len(s)
    return s[n // 2] if n % 2 else 0.5 * (s[n // 2 - 1] + s[n // 2])

def is_num(t):
    if t == "nan":
        return True
    if not t or any(c not in "0123456789." for c in t):
        return False
    return t.count(".") <= 1 and t != "."


def parse_epoch(line):
    """`Epoch N, Train Accuracy: X %, Test Accuracy: Y %` anywhere in the line -> (N, X, Y) or None."""
    i = line.find("Epoch")
    while i >= 0:
        rest = line[i + len("Epoch"):]
        parts = rest.split(",")
        if len(parts) >= 3 and rest[:1] in (" ", "\t"):
            n = parts[0].strip()
            tr = parts[1].strip()
            te = parts[2].strip()
            if n.isdigit() and tr.startswith("Train Accuracy:") and te.startswith("Test Accuracy:"):
                x = tr[len("Train Accuracy:"):].strip()
                if x.endswith("%"):
                    x = x[:-1].strip()
                y = te[len("Test Accuracy:"):].strip().split()
                y = y[0] if y else ""
                if y.endswith("%"):
                    y = y[:-1]
                if is_num(x) and is_num(y):
                    return int(n), float(x), float(y)
        i = line.find("Epoch", i + 1)
    return None


def kind_on(line):
    c = line.find(":")
    if c < 2:
        return False
    head = line[:c]
    if not head[0].isupper() or any(ch not in "ABCDEFGHIJKLMNOPQRSTUVWXYZ_" for ch in head):
        return False
    tail = line[c + 1:]
    if not tail.startswith(" on"):
        return False
    nxt = tail[3:4]
    return nxt == "" or not (nxt.isalnum() or nxt == "_")


def read_out(path):
    r = dict(args=None, env=None, gpu=None, done=False, ep={}, dup=0, kinds=0, pt=False, nargs=0, nenv=0, minutes=None)
    for ln in open(path, errors="replace").read().splitlines():
        if ln.startswith("ARGS:"):
            r["nargs"] += 1
            if r["args"] is None:
                r["args"] = ln
        elif ln.startswith("ENV:"):
            r["nenv"] += 1
            if r["env"] is None:
                r["env"] = ln
        elif ln.startswith("NVIDIA") and r["gpu"] is None:
            r["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            r["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            r["pt"] = True
        elif ln.startswith("NODE="):
            pass
        else:
            e = parse_epoch(ln)
            if e:
                if e[0] in r["ep"]:
                    r["dup"] += 1
                else:
                    r["ep"][e[0]] = (e[1], e[2])
            if kind_on(ln):
                r["kinds"] += 1
            w = ln.split()
            if len(w) == 2 and w[1] == "minutes" and w[0].isdigit():
                r["minutes"] = int(w[0])
    r["complete"] = r["done"] and all(k in r["ep"] for k in range(N_EPOCHS))
    return r


def split_args(line):
    toks = line[len("ARGS:"):].split()
    if len(toks) % 2:
        return None
    pairs = []
    for i in range(0, len(toks), 2):
        if not toks[i].startswith("--") or toks[i + 1].startswith("--"):
            return None
        pairs.append((toks[i][2:], toks[i + 1]))
    return pairs


def read_probe(path):
    recs, bad = [], 0
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            d = json.loads(ln)
        except ValueError:
            bad += 1
            continue
        if not isinstance(d, dict):
            bad += 1
            continue
        recs.append(d)
    return recs, bad


def rec_med_shrink(b):
    """the median over the record's step sizes of exp(beta) * wd (caw2's definition, 290)."""
    e = sorted(math.exp(x) for x in b)
    n = len(e)
    return (e[n // 2] if n % 2 else 0.5 * (e[n // 2 - 1] + e[n // 2])) * WD


def csv_split(line):
    out, cur, q, i = [], [], False, 0
    while i < len(line):
        c = line[i]
        if q:
            if c == '"':
                if i + 1 < len(line) and line[i + 1] == '"':
                    cur.append('"')
                    i += 1
                else:
                    q = False
            else:
                cur.append(c)
        elif c == '"':
            q = True
        elif c == ",":
            out.append("".join(cur))
            cur = []
        else:
            cur.append(c)
        i += 1
    out.append("".join(cur))
    return out


def corpus_count(csvp, tsvp):
    if not (csvp and tsvp and os.path.exists(csvp) and os.path.exists(tsvp)):
        return None
    ex, head = set(), None
    for ln in open(tsvp):
        ln = ln.rstrip("\n")
        if not ln.strip() or ln.startswith("#"):
            continue
        f = ln.split("\t")
        if head is None:
            head = f
            continue
        d = dict(zip(head, f))
        ex.add((d.get("run"), d.get("job_id")))
    L = open(csvp).read().splitlines()
    h = csv_split(L[0])
    ir, ij = h.index("run"), h.index("job_id")
    n = 0
    for ln in L[1:]:
        if not ln:
            continue
        f = csv_split(ln)
        if f[ir].startswith(BATCH + "-"):
            continue
        if (f[ir], f[ij]) in ex:
            continue
        n += 1
    return n


def sha_file(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        while True:
            b = fh.read(1 << 22)
            if not b:
                break
            h.update(b)
    return h.hexdigest()


def emax_quad(k, n=200000, lo=-10.0, hi=10.0):
    """E[max of k iid N(0,1)] = integral of x * k * phi(x) * Phi(x)^(k-1), composite Simpson."""
    hstep = (hi - lo) / n
    tot = 0.0
    for i in range(n + 1):
        x = lo + i * hstep
        f = x * k * math.exp(-0.5 * x * x) / math.sqrt(2 * math.pi) * (0.5 * (1 + math.erf(x / math.sqrt(2)))) ** (k - 1)
        w = 1 if i in (0, n) else (4 if i % 2 else 2)
        tot += w * f
    return tot * hstep / 3.0

def pick(cands, key):
    best = None
    for c, x in cands:
        if best is None or x[key] > best[1][key]:
            best = (c, x)
    return best[0] if best else None


def read_word(k_ok, k_lvl, p_ok, t_ch, t_nd, se, b):
    if not k_ok:
        return "SCALAR-DIVERGED"
    if k_lvl < HEALTH:
        return "SCALAR-COLLAPSED"
    if not p_ok:
        return "UNREADABLE"
    if t_ch - b >= SURV and t_nd - b >= SURV and t_ch - b > RES * se and t_nd - b > RES * se:
        return "SCALAR-BEATS-BEST"
    tm = min(t_ch, t_nd)
    if tm - b - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tm + b <= -SURV and tm + b < -RES * se:
        return "PARTITION-ABOVE-SCALAR"
    return "UNRESOLVED"


def cfg_word(k_lvl, k_div, ok, t_ch, t_nd, se):
    if k_div:
        return "DIVERGED"
    if k_lvl < HEALTH:
        return "COLLAPSED"
    if not ok:
        return "UNREADABLE"
    tm = min(t_ch, t_nd)
    if t_ch >= SURV and t_nd >= SURV and t_ch > RES * se and t_nd > RES * se:
        return "BEATS"
    if tm - RES * se >= -SURV:
        return "TIES"
    if tm <= -SURV and tm < -RES * se:
        return "BELOW"
    return "UNRESOLVED"


def branch_word(t, o):
    if t in GATES:
        return "BRANCH-UNREADABLE"
    if t == "PARTITION-ABOVE-SCALAR":
        return "HEADLINE-REFUTED-BY-RETUNE"
    if t == "SCALAR-BEATS-BEST":
        return "NOT-A-TUNING-ARTEFACT" if o == "SCALAR-BEATS-BEST" else "SELECTION-DEPENDENT"
    if t == "SCALAR-TIES-BEST":
        return "WEAKENED-TO-TIE"
    return "RETUNE-UNDECIDED"


def selected(A, sel, sig, b):
    k = sel["k01"]
    if k is None:
        return "SCALAR-DIVERGED", None, None, None
    kx = A["k01" + k]
    if sel["ch"] is None or sel["nd"] is None:
        return read_word(True, kx["level"], False, 0, 0, 1, b), None, None, None
    cx, nx = A["ch" + sel["ch"]], A["nd" + sel["nd"]]
    t_ch, t_nd = kx["level"] - cx["level"], kx["level"] - nx["level"]
    se = sig * math.sqrt(1.0 / kx["n"] + 1.0 / cx["n"])
    ok = cx["level"] >= HEALTH and nx["level"] >= HEALTH
    return read_word(True, kx["level"], ok, t_ch, t_nd, se, b), t_ch, t_nd, se


def decide(A, sig):
    b = EMAX4_TYPED * sig / math.sqrt(len(SEEDS))
    st, so = {}, {}
    for g in G_IDS:
        cands = [(c, A[g + c]) for c in CFG_IDS if A[g + c]["range"] <= DIVG]
        st[g] = pick(cands, "train")
        so[g] = pick(cands, "level")
    rt = selected(A, st, sig, b)
    ro = selected(A, so, sig, b)
    cw, T = {}, {}
    for c in CFG_IDS:
        k, ch, nd = A["k01" + c], A["ch" + c], A["nd" + c]
        t_ch, t_nd = k["level"] - ch["level"], k["level"] - nd["level"]
        se = sig * math.sqrt(1.0 / k["n"] + 1.0 / ch["n"])
        ok = ch["level"] >= HEALTH and nd["level"] >= HEALTH and ch["range"] <= DIVG and nd["range"] <= DIVG
        cw[c] = cfg_word(k["level"], k["range"] > DIVG, ok, t_ch, t_nd, se)
        T[c] = (t_ch, t_nd, se, ch["level"] - nd["level"])
    return b, st, so, rt, ro, cw, T

def main(argv):
    csvp = tsvp = rowsp = None
    pos, i = [], 0
    while i < len(argv):
        if argv[i] in ("--csv", "--tsv", "--rows"):
            if argv[i] == "--csv":
                csvp = argv[i + 1]
            elif argv[i] == "--tsv":
                tsvp = argv[i + 1]
            else:
                rowsp = argv[i + 1]
            i += 2
            continue
        pos.append(argv[i])
        i += 1
    if len(pos) < 2:
        print("usage: crt1_attack_indep.py <runsdir> <scorer-log> [...] --csv P --tsv P [--rows P]")
        return 2
    runsdir, logs = pos[0], pos[1:]
    sub = os.path.join(runsdir, BATCH)

    say("=" * 114)
    say(" crt1_attack_indep -- AN INDEPENDENT RE-DERIVATION OF THE REGISTERED crt1 SCORER'S OUTPUT (CORRECTIONS 300)")
    say(" No repo module imported, no regular expression, every literal re-typed.  Host-independent output.")
    say("=" * 114)

    # ---- locate the run files -----------------------------------------------------------------------------------
    outs = {}
    stray = []
    for base in (runsdir, sub):
        if not os.path.isdir(base):
            continue
        for fn in sorted(os.listdir(base)):
            if not (fn.startswith(BATCH + "-") and fn.endswith(".out")):
                continue
            parts = fn[:-len(".out")].split("-")
            if len(parts) != 4 or parts[0] != BATCH or not parts[2].startswith("s") or not parts[3].isdigit():
                stray.append(fn)
                continue
            a, s, j = parts[1], parts[2][1:], parts[3]
            if a not in ARMS or not s.isdigit() or int(s) not in SEEDS:
                stray.append(fn)
                continue
            key = (a, int(s))
            p = os.path.realpath(os.path.join(base, fn))
            outs.setdefault(key, [])
            if p not in [q for _j, q in outs[key]]:
                outs[key].append((int(j), p))
    probes = {}
    for a in ARMS:
        for s in SEEDS:
            p = os.path.join(sub, "probe_" + rname(a, s), "probe.jsonl")
            probes[(a, s)] = p if os.path.exists(p) else None

    # [0] ----------------------------------------------------------------------------------------------------------
    say("")
    say("[0] SYNC INTEGRITY -- one digest over every run file's name and bytes, in registered run order")
    h = hashlib.sha256()
    nfiles, nbytes = 0, 0
    for a in ARMS:
        for s in SEEDS:
            for _j, p in sorted(outs.get((a, s), [])):
                h.update(os.path.basename(p).encode())
                h.update(sha_file(p).encode())
                nfiles += 1
                nbytes += os.path.getsize(p)

                                                                   "met" if min(t_ch, t_nd) - b - RES * se >= -SURV else "missed"))
    say("    PARTITION-ABOVE needs min T + B <= -0.30:  min T + B = %+.4f  (%.4f pp away)" % (
        min(t_ch, t_nd) + b, min(t_ch, t_nd) + b + SURV))
    say("    the lower edges (T - B - 2 SE): ch %+.4f, nd %+.4f;  the upper edges (T + B + 2 SE): ch %+.4f, nd %+.4f" % (
        t_ch - b - RES * se, t_nd - b - RES * se, t_ch + b + RES * se, t_nd + b + RES * se))
    say("  DESCRIPTIVE 1 -- the reading hinges on O2's sigma.  At the FROZEN floor instead of the in-batch sigma:")

    say("    B %.6f, SE %.6f: ch T-B %+.4f vs 2 SE %.4f, nd T-B %+.4f  -> the same rule would print TUNED-%s" % (
        bf, sef, t_ch - bf, RES * sef, t_nd - bf, wf))
    say("    NOT the registered reading (O2 takes the larger sigma).  The in-batch sigma exceeds the floor by %.1f %%;" % (
        100 * (sig_in / SIGMA_FLOOR - 1)))
    say("    per-arm sd: %s" % "  ".join("%s %.3f" % (a, A[a]["sd"]) for a in ARMS))
    s_sel = math.sqrt(avg([A[g + st[g]]["sd"] ** 2 for g in G_IDS]))
    say("    pooled sd of the three SELECTED arms alone: %.6f (df 6) -- also above the floor" % s_sel)
    say("    the sigma at which the TUNED ch contrast would clear 2 SE net of B (T_ch = B(s) + 2 SE(s)): s = %.6f" % (
        t_ch / (EMAX4_TYPED / math.sqrt(3) + RES * math.sqrt(2.0 / 3))))
    say("  DESCRIPTIVE 2 -- the every-config reading (NO selection, so NO B): %s; at A2 alone T_ch %+.4f = %.2f SE."
        % ("+".join("%s-%s" % (c, cw[c]) for c in CFG_IDS), T["A2"][0], T["A2"][0] / T["A2"][2]))
    say("    Scalar's POINT estimate is above both partitions at every config; the TIE is a bound from B and 2 SE, not a")
    say("    measured zero (TIES-ARE-BOUNDS).")
    say("  DESCRIPTIVE 3 -- per seed at the selected config A2 (plateau5):")

                                         A[g + "A2"]["level"]))
    say("    same-seed paired k01 - ch: %s;  k01 - nd: %s" % (

        " ".join("s%d %+.2f" % (s, A["k01A2"]["xs"][i] - A["ndA2"]["xs"][i]) for i, s in enumerate(SEEDS))))

        say("    drop seed %d: T_ch %+.4f  T_nd %+.4f  (2 v 2)" % (
            s, k - avg([A["chA2"]["xs"][j] for j in keep]), k - avg([A["ndA2"]["xs"][j] for j in keep])))
    say("  DESCRIPTIVE 4 -- the grid edge: every grain selects A2 (alpha0 1e-2), the TOP of a TWO-point alpha0 axis")

        T["M2"][0] - T["A2"][0], T["M2"][1] - T["A2"][1]))
    say("    (ALPHA0-TWO-POINTS); whether a larger alpha0 closes the gap further is UNSURE.")

    say("    coordinate sits at the upper box edge -2.3026 on that fraction of records); ndA2 %.4f, k01A2 %.4f.  The box is"
        % (A["ndA2"]["hi"], A["k01A2"]["hi"]))
    say("    a stamp, never a gate: the question is the tuned level at the harness's fixed box.")
    say("  DESCRIPTIVE 6 -- the selector: train5 at A2 ch %.4f nd %.4f k01 %.4f; next-best config's train5 margin ch %+.4f,"
        % (A["chA2"]["train"], A["ndA2"]["train"], A["k01A2"]["train"],
           A["chA2"]["train"] - max(A["ch" + c]["train"] for c in CFG_IDS if c != "A2")))

        % (A["ndA2"]["train"] - max(A["nd" + c]["train"] for c in CFG_IDS if c != "A2"),
           A["k01A2"]["train"] - max(A["k01" + c]["train"] for c in CFG_IDS if c != "A2")))

    say("  FLOOR / CEILING (164.6):")

        say("    %-6s level %.4f  above HEALTH_MIN 80 by %+.4f  above chance 10 by %.4f  below 100 by %.4f" % (
