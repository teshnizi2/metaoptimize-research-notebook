#!/usr/bin/env python3
# =============================================================================
# cT1_tin_score.py -- THE REGISTERED SCORER FOR `tn1`, THE TINY-IMAGENET
#                     GRANULARITY LADDER.
#
# STANDING RULE 21: this file is git-committed BEFORE any tn1 run exists.  Every
# bar, band and refusal below is frozen here.  selftest() asserts each constant
# against bin/cT1_tin_granularity.sh's OWN TEXT, RE-DERIVES every archived
# number from results/all_runs.csv rather than quoting prose, and asserts that
# the corpus contains ZERO Tiny-ImageNet rows at registration time.
#
# STANDING RULE 16: run this file UNEDITED and quote its verdict.  Passing its
# documented arguments is not editing it.  Once a tn1 run exists it is frozen.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# The parent paper's motivating anomaly is a SCALE anomaly: finer step-size
# granularity beats a scalar on small vision tasks and stops beating it on
# ImageNet (Sharifnassab, Salehkaleybar & Sutton, ICML 2025, section 7.3).  Every
# one of this campaign's 2,174 archived runs is at CIFAR resolution with 10 or
# 100 classes, so the campaign has never observed the axis on which the anomaly
# is claimed to live.  ImageNet-1k is blocked on DATA, not code: the PI tree
# holds 489 of 1000 train classes and an unlabelled 50,000-image validation set,
# and is invisible from this account in any case.
#
# Tiny-ImageNet-200 moves BOTH of the axes that separate CIFAR from ImageNet --
# input resolution 32 -> 64 and class count 10 -> 200 -- and needs no
# credentials and no terms-of-access acceptance.  It is the credential-free
# scale probe.  It is NOT ImageNet and no number here may be compared with a
# published ImageNet-1k number.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  12 jobs, ONE submission, so BATCH -- the unit of replication in
# this corpus (F(62,85)=5.47, p 6.9e-13; SEED is null, F 1.21, p 0.213) -- is
# held constant and every contrast below is WITHIN this batch.
#
#   arm  scal    TinyImageNet / ResNet18_tin / --stepsize-groups scalar
#        lay     TinyImageNet / ResNet18_tin / --stepsize-groups layerwise
#        node    TinyImageNet / ResNet18_tin / --stepsize-groups nodewise
#        c10lay  CIFAR10      / ResNet18     / --stepsize-groups layerwise
#                              <- the TREE GATE, not a science arm
#   seed in {0, 1, 2}
#   SGDm base + Lion meta / meta-stepsize 1e-3 / alpha0 1e-6 / 100 epochs /
#   batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / HIER unset.
#
# WHY A CIFAR-10 CONTROL IS IN A TINY-IMAGENET BATCH.  PATCH_TINYIMAGENET edits
# two files that EVERY run in this workspace imports (load_data.py,
# build_network.py) while 60 jobs of two other decisive batches were queued.
# tests/tin_equivalence.py proves the patch is a bitwise no-op offline; the
# c10lay arm proves it on the live tree, in production, with the same optimizer,
# by reproducing an archived arm mean.  If it misses, the Tiny-ImageNet numbers
# are not reportable and this file says so.
#
# -----------------------------------------------------------------------------
# THE PRIMARY STATISTIC, FROZEN
# -----------------------------------------------------------------------------
#   plateau5 = mean of the LAST 5 test epochs.  The CSV `plateau` column
#   (mean-of-last-20) is BANNED as primary and is never read by this file;
#   `best_test` is never read.  M(arm) = mean plateau5 over that arm's seeds.
#
#   GRAN = M(lay)  - M(scal)      PRIMARY
#   NODE = M(node) - M(lay)       SECONDARY
#
# -----------------------------------------------------------------------------
# THE BARS, FROZEN, AND WHY THESE NUMBERS
# -----------------------------------------------------------------------------
# Re-derived by selftest() from results/all_runs.csv over the archived cell
# ResNet18 / CIFAR10 / SGDm+Lion / meta-stepsize 1e-3 / alpha0 1e-6 /
# BETA_CLIP -15:-2.3026 / AUGMENT=1 / 100 epochs / batch 100 / gamma 1 /
# HIER unset / collapsed 0:
#
#     scalar     87.879 pp   n=12  over 5 batches
#     layerwise  90.893 pp   n=17  over 6 batches
#     nodewise   91.739 pp   n= 8  over 3 batches
#     GRAN_CIFAR = layerwise - scalar    = +3.014 pp
#     NODE_CIFAR = nodewise  - layerwise = +0.846 pp
#     pooled within-cell SD of plateau5  =  0.1774 pp  (df 23, 12 cells)
#
#   REPLICATES   GRAN >= +1.507  (HALF the archived CIFAR-10 lift)  AND t >= +2
#                -> granularity's advantage survives 64px and 200 classes.
#   VANISHES     |GRAN| <= 0.50
#                -> it does not survive.  This is the parent paper's ImageNet
#                   anomaly appearing at Tiny-ImageNet scale, and it is a
#                   POSITIVE finding, not a failure.
#   REVERSES     GRAN <= -0.50 AND t <= -2
#                -> scalar beats layerwise, the strong form of the anomaly.
#   otherwise    UNRESOLVED.
#
# HONESTY ABOUT POWER, REGISTERED IN ADVANCE SO IT CANNOT BE DISCOVERED
# AFTERWARDS.  The 0.1774 pp noise floor is a CIFAR-10 number.  Tiny-ImageNet's
# run-to-run SD has never been measured by anyone here, and a 200-class task at
# lower absolute accuracy will plausibly be noisier.  So the t in every rule
# above is computed from THIS BATCH's own 3-seed spread, never from the archived
# floor, and a wide in-batch spread will correctly produce UNRESOLVED rather
# than a verdict.  Concretely, with 3 seeds per arm:
#     SD = 1x CIFAR-10's   SE(diff) = 0.145 pp   archived lift t = 20.8
#     SD = 4x              SE(diff) = 0.579 pp   archived lift t =  5.2,
#                                                half-size lift t =  2.6
#     SD = 6x              SE(diff) = 0.869 pp   half-size lift t =  1.7 -> UNRESOLVED
# So this batch is powered for an effect of roughly the archived size at any
# plausible Tiny-ImageNet noise level, and is NOT powered for a half-size effect
# if Tiny-ImageNet turns out to be six times noisier than CIFAR-10.  Registered
# here in advance so that a wide result cannot afterwards be called a null.
#
#   NODE:  FINER_HELPS  NODE >= +0.423 and t >= +2
#          FINER_HURTS  NODE <= -0.423 and t <= -2
#          NODE_NULL    |NODE| <= 0.25
#          otherwise    UNRESOLVED.
#
# -----------------------------------------------------------------------------
# THE GATES
# -----------------------------------------------------------------------------
# G0  PROVENANCE, ABSOLUTELY GATING (STANDING RULE 20).  Every fact about a run
#     is read from that run's OWN ARGS and ENV lines.  The arm is identified by
#     (--dataset, --NN-name, --stepsize-groups), NEVER by the run name.  Any
#     repeated flag VOIDS that run, because argparse keeps the last occurrence
#     and two whole batches were silently voided that way.
# G1  CHANCE FLOOR.  Tiny-ImageNet chance is 100/200 = 0.500%.  A run whose
#     plateau5 is below 5.0 has not learned 200 classes and almost certainly has
#     a broken label map or loader; it is VOID, not a data point.
# G2  TREE GATE.  The in-batch c10lay arm must land within 0.60 pp of the
#     archived CIFAR-10 plain-layerwise grand mean, RE-DERIVED from the CSV at
#     score time.  A miss means PATCH_TINYIMAGENET moved the CIFAR path:
#     VOID-PATCHBROKE, and no Tiny-ImageNet number may be reported.
# G3  COMPLETENESS.  A run counts only if it reached >= 95 of its 100 requested
#     epochs AND its last 5 test epochs are all present.  Truncated runs are
#     dropped and the arm carries an INCOMPLETE suffix.
# G4  THE PRIMARY.
#
# -----------------------------------------------------------------------------
# WHAT THIS BATCH MAY NOT CLAIM
# -----------------------------------------------------------------------------
# Anything about ImageNet-1k (this is 200 classes at 64px, not 1000 at 224px);
# any hierarchical operator (HIER is unset in every arm -- this batch measures
# the GRANULARITY ladder, not shrinkage); any alpha0 but 1e-6; any meta-stepsize
# but 1e-3; any optimizer pair but SGDm+Lion; any architecture but ResNet-18;
# any budget but 100 epochs; any granularity but the three run here.  Three
# rungs cannot locate an optimum over granularity.
#
# USAGE
#     python3 analysis/cT1_tin_score.py --selftest
#     python3 analysis/cT1_tin_score.py --runs /cluster/account_b/metaopt/runs
# =============================================================================
import argparse
import collections
import csv
import glob
import math
import os
import re
import statistics
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
REPO = os.path.dirname(HERE)
CSV = os.path.join(REPO, "results", "all_runs.csv")
BATCH_SH = os.path.join(REPO, "bin", "cT1_tin_granularity.sh")

# ---- FROZEN REGISTRATION ----------------------------------------------------
PREFIX = "tn1-"
NET_TIN, DSET_TIN = "ResNet18_tin", "TinyImageNet"
NET_C10, DSET_C10 = "ResNet18", "CIFAR10"
MS, ALPHA0, CLIP, AUG = "1e-3", "1e-6", "-15:-2.3026", "1"
EPOCHS, BATCH, GAMMA = 100, 100, "1"
BASE_ALG, META_ALG = "SGDm", "Lion"
SEEDS = (0, 1, 2)
ARMS = collections.OrderedDict([          # tag -> (dataset, network, granularity)
    ("scal",   (DSET_TIN, NET_TIN, "scalar")),
    ("lay",    (DSET_TIN, NET_TIN, "layerwise")),
    ("node",   (DSET_TIN, NET_TIN, "nodewise")),
    ("c10lay", (DSET_C10, NET_C10, "layerwise")),
])
TIN_ARMS = ("scal", "lay", "node")
CTRL_ARM = "c10lay"

GRAN_CIFAR = 3.014          # archived layerwise - scalar, alpha0 1e-6
NODE_CIFAR = 0.846          # archived nodewise  - layerwise
REPL_BAR = 1.507            # HALF of GRAN_CIFAR
VANISH_BAR = 0.50
NODE_BAR = 0.423            # HALF of NODE_CIFAR
NODE_NULL_BAR = 0.25
T_BAR = 2.0
TREE_BAND = 0.60
ARCH_LAY_C10 = 90.893       # archived plain layerwise grand mean, alpha0 1e-6
ARCH_SCAL_C10 = 87.879
ARCH_NODE_C10 = 91.739
SIGMA_W = 0.1774            # pooled within-cell SD of plateau5, df 23, 12 cells
SIGMA_DF = 23
CHANCE_TIN = 0.5            # 100/200 %
FLOOR = 5.0

EP_RE = re.compile(r"Epoch\s+(\d+).*?Test Accuracy:\s*([0-9.]+)", re.I)
ARGS_RE = re.compile(r"^\s*ARGS:\s*(.*)$")
ENV_RE = re.compile(r"^\s*ENV:\s*(.*)$")


# ---- re-derivation from the corpus.  Nothing here is quoted from prose. -----
def _csv_rows(path=CSV):
    with open(path) as fh:
        return [r for r in csv.DictReader(fh) if (r.get("superseded") or "0") != "1"]


def _batch_of(run):
    return re.split(r"[-_]", str(run))[0]


def _base_cell(r):
    """plateau5 of a row IFF it sits in the archived comparison cell."""
    try:
        p5 = float(r["plateau5"])
    except (TypeError, ValueError):
        return None
    if not (r["network"] == NET_C10 and r["dataset"] == DSET_C10
            and r["base"] == BASE_ALG and r["meta"] == META_ALG
            and r["meta_stepsize"] == MS and r["beta_clip"] == CLIP
            and r["augment"] == AUG and r["epochs_done"] == str(EPOCHS)
            and r["collapsed"] == "0" and r["batch_size"] == str(BATCH)
            and r["gamma"] == GAMMA and r["alpha0"] == ALPHA0
            and not (r.get("hier") or "").strip()):
        return None
    return p5


def archived_granularity(path=CSV):
    """{granularity: (mean, n, distinct batches)} in the archived cell."""
    acc = collections.defaultdict(list)
    for r in _csv_rows(path):
        p5 = _base_cell(r)
        if p5 is None:
            continue
        if r["granularity"] in ("scalar", "layerwise", "nodewise"):
            acc[r["granularity"]].append((p5, _batch_of(r["run"])))
    return {k: (statistics.mean(x for x, _ in v), len(v), len(set(b for _, b in v)))
            for k, v in acc.items()}


def archived_noise(path=CSV):
    """Pooled WITHIN-CELL SD of plateau5 over (batch x granularity) cells."""
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        p5 = _base_cell(r)
        if p5 is None:
            continue
        if r["granularity"] in ("scalar", "layerwise", "nodewise"):
            cells[(_batch_of(r["run"]), r["granularity"])].append(p5)
    ss, df, n = 0.0, 0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            n += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, n


def premise_no_tin_rows(path=CSV):
    """RULE 21: the corpus must hold NO Tiny-ImageNet row at registration."""
    return sum(1 for r in _csv_rows(path)
               if "tiny" in (r.get("dataset") or "").lower()
               or "_tin" in (r.get("network") or "").lower())


# ---- reading the runs.  The arm of a run is MEASURED from its own lines. ----
def _tokens(line):
    import shlex
    try:
        return shlex.split(line)
    except ValueError:
        return line.split()


def parse_args_line(line):
    """(effective_flags, repeated_flags).  argparse keeps the LAST occurrence."""
    toks = _tokens(line)
    occ, i, n = [], 0, len(toks)
    while i < n:
        t = toks[i]
        if t.startswith("--") and len(t) > 2:
            if "=" in t:
                f, v = t.split("=", 1)
                occ.append((f.lstrip("-"), v))
                i += 1
                continue
            vals, j = [], i + 1
            while j < n and not (toks[j].startswith("--") and len(toks[j]) > 2):
                vals.append(toks[j])
                j += 1
            occ.append((t.lstrip("-"), " ".join(vals)))
            i = j
            continue
        i += 1
    eff, seen = collections.OrderedDict(), collections.OrderedDict()
    for f, v in occ:
        eff[f] = v
        seen.setdefault(f, []).append(v)
    return eff, dict((f, v) for f, v in seen.items() if len(v) > 1)


def parse_env_line(line):
    out = {}
    for tok in line.split():
        if "=" in tok:
            k, v = tok.split("=", 1)
            out[k] = v
    return out


def read_out(path):
    args = env = None
    ser = {}
    try:
        with open(path, "r", errors="replace") as fh:
            for line in fh:
                if args is None:
                    m = ARGS_RE.match(line)
                    if m:
                        args = m.group(1).strip()
                        continue
                if env is None:
                    m = ENV_RE.match(line)
                    if m:
                        env = m.group(1).strip()
                        continue
                m = EP_RE.search(line)
                if m:
                    ser[int(m.group(1))] = float(m.group(2))
    except IOError:
        return None
    if args is None:
        return None
    eff, dup = parse_args_line(args)
    return {"args": eff, "dup": dup, "argsline": args,
            "env": parse_env_line(env or ""), "series": ser, "path": path}


def plateau5(series, budget=EPOCHS, w=5):
    """PRIMARY.  Mean of the LAST 5 test epochs; train.py prints 0-indexed."""
    v = [series[e] for e in range(budget - w, budget) if e in series]
    return sum(v) / len(v) if len(v) == w else None


def norm_num(s):
    try:
        return "%g" % float(s)
    except (TypeError, ValueError):
        return None


def arm_of(a):
    """The arm a run ACTUALLY ran, from its own flags.  None if it is not ours."""
    key = (a.get("dataset"), a.get("NN-name"), a.get("stepsize-groups"))
    for tag, want in ARMS.items():
        if key == want:
            return tag
    return None


# ---- G0: provenance ---------------------------------------------------------
def gate0(recs):
    kept, voided = [], []
    for r in recs:
        a, e = r["args"], r["env"]
        why = []
        if r["dup"]:
            why.append("REPEATED FLAG %s" % sorted(r["dup"]))
        tag = arm_of(a)
        if tag is None:
            why.append("arm not in the registered design (dataset=%s NN=%s groups=%s)"
                       % (a.get("dataset"), a.get("NN-name"), a.get("stepsize-groups")))
        for flag, want in (("meta-stepsize", MS), ("alpha0", ALPHA0),
                           ("num-epochs", str(EPOCHS)), ("batch-size", str(BATCH)),
                           ("gamma", GAMMA), ("alg-base", BASE_ALG),
                           ("alg-meta", META_ALG)):
            got = a.get(flag)
            if norm_num(want) is not None:
                same = norm_num(got) == norm_num(want)
            else:
                same = got == want
            if not same:
                why.append("%s=%s (want %s)" % (flag, got, want))
        if e.get("AUGMENT") != AUG:
            why.append("AUGMENT=%s" % e.get("AUGMENT"))
        if e.get("BETA_CLIP") != CLIP:
            why.append("BETA_CLIP=%s" % e.get("BETA_CLIP"))
        if (e.get("HIER") or "none") != "none":
            why.append("HIER=%s (this batch is the GRANULARITY ladder; HIER must be unset)"
                       % e.get("HIER"))
        r["arm"] = tag
        r["seed"] = a.get("seed")
        (voided if why else kept).append((r, why))
    return kept, voided


def mean_se(v):
    m = statistics.mean(v)
    if len(v) < 2:
        return m, float("nan")
    return m, statistics.stdev(v) / math.sqrt(len(v))


def contrast(a, b):
    """(diff, se, t, df) for mean(a) - mean(b), independent 3-seed arms."""
    ma, mb = statistics.mean(a), statistics.mean(b)
    if len(a) < 2 or len(b) < 2:
        return ma - mb, float("nan"), float("nan"), 0
    va, vb = statistics.variance(a), statistics.variance(b)
    se = math.sqrt(va / len(a) + vb / len(b))
    t = (ma - mb) / se if se > 0 else float("inf")
    return ma - mb, se, t, len(a) + len(b) - 2


def gran_verdict(d, t):
    if d >= REPL_BAR and (t >= T_BAR or math.isnan(t)) and not math.isnan(t):
        return "REPLICATES"
    if abs(d) <= VANISH_BAR:
        return "VANISHES"
    if d <= -VANISH_BAR and not math.isnan(t) and t <= -T_BAR:
        return "REVERSES"
    return "UNRESOLVED"


def node_verdict(d, t):
    if d >= NODE_BAR and not math.isnan(t) and t >= T_BAR:
        return "FINER_HELPS"
    if d <= -NODE_BAR and not math.isnan(t) and t <= -T_BAR:
        return "FINER_HURTS"
    if abs(d) <= NODE_NULL_BAR:
        return "NODE_NULL"
    return "UNRESOLVED"


# ---- selftest ---------------------------------------------------------------
def selftest():
    fails = []

    def ok(label, cond, extra=""):
        print("  %-62s %s %s" % (label, "PASS" if cond else "**FAIL**", extra))
        if not cond:
            fails.append(label)

    print("cT1_tin_score selftest")
    print("-- the registration is re-derived, not quoted ------------------")
    g = archived_granularity()
    for k in ("scalar", "layerwise", "nodewise"):
        ok("archived %s arm present in the CSV" % k, k in g,
           ("mean %.3f n=%d batches=%d" % g[k]) if k in g else "")
    if all(k in g for k in ("scalar", "layerwise", "nodewise")):
        gran = g["layerwise"][0] - g["scalar"][0]
        node = g["nodewise"][0] - g["layerwise"][0]
        ok("GRAN_CIFAR constant matches the CSV", abs(gran - GRAN_CIFAR) < 0.02,
           "re-derived %.3f, frozen %.3f" % (gran, GRAN_CIFAR))
        ok("NODE_CIFAR constant matches the CSV", abs(node - NODE_CIFAR) < 0.02,
           "re-derived %.3f, frozen %.3f" % (node, NODE_CIFAR))
        ok("REPL_BAR is exactly half GRAN_CIFAR", abs(REPL_BAR - GRAN_CIFAR / 2) < 0.01)
        ok("NODE_BAR is exactly half NODE_CIFAR", abs(NODE_BAR - NODE_CIFAR / 2) < 0.01)
        ok("ARCH_LAY_C10 matches the CSV", abs(g["layerwise"][0] - ARCH_LAY_C10) < 0.02,
           "%.3f" % g["layerwise"][0])
        ok("ARCH_SCAL_C10 matches the CSV", abs(g["scalar"][0] - ARCH_SCAL_C10) < 0.02,
           "%.3f" % g["scalar"][0])
        ok("ARCH_NODE_C10 matches the CSV", abs(g["nodewise"][0] - ARCH_NODE_C10) < 0.02,
           "%.3f" % g["nodewise"][0])
    sd, df, nc = archived_noise()
    ok("SIGMA_W matches the CSV", abs(sd - SIGMA_W) < 0.01,
       "re-derived %.4f df=%d cells=%d, frozen %.4f df=%d" % (sd, df, nc, SIGMA_W, SIGMA_DF))
    ok("SIGMA_DF matches the CSV", df == SIGMA_DF, "%d" % df)

    print("-- RULE 21: no Tiny-ImageNet run may exist yet ------------------")
    n_tin = premise_no_tin_rows()
    ok("corpus holds ZERO Tiny-ImageNet rows", n_tin == 0, "found %d" % n_tin)

    print("-- the arithmetic this batch's power rests on -------------------")
    se_arch = SIGMA_W * math.sqrt(2.0 / 3.0)
    ok("SE(difference of two 3-seed means) at the archived floor",
       abs(se_arch - 0.1449) < 0.002, "%.4f pp" % se_arch)
    ok("archived lift would land at t = %.1f at that floor" % (GRAN_CIFAR / se_arch),
       GRAN_CIFAR / se_arch > 10)
    se4 = 4 * se_arch
    ok("at 4x that SD the archived lift still clears t=2",
       GRAN_CIFAR / se4 >= T_BAR, "t = %.2f" % (GRAN_CIFAR / se4))
    ok("at 4x that SD a HALF-size effect still clears t=2",
       REPL_BAR / se4 >= T_BAR, "t = %.2f" % (REPL_BAR / se4))
    se6 = 6 * se_arch
    ok("at 6x that SD a HALF-size effect does NOT clear t=2 (declared limit)",
       REPL_BAR / se6 < T_BAR, "t = %.2f" % (REPL_BAR / se6))
    ok("Tiny-ImageNet chance is 0.5 percent", abs(CHANCE_TIN - 100.0 / 200.0) < 1e-9)
    ok("the floor is 10x chance", abs(FLOOR - 10 * CHANCE_TIN) < 1e-9)

    print("-- the verdict function is exercised in every direction ---------")
    ok("REPLICATES at the archived effect", gran_verdict(3.0, 8.0) == "REPLICATES")
    ok("REPLICATES exactly at the bar", gran_verdict(REPL_BAR, 2.0) == "REPLICATES")
    ok("UNRESOLVED just below the bar", gran_verdict(REPL_BAR - 0.01, 9.0) == "UNRESOLVED")
    ok("UNRESOLVED at the bar with a weak t", gran_verdict(REPL_BAR, 1.9) == "UNRESOLVED")
    ok("VANISHES at zero", gran_verdict(0.0, 0.0) == "VANISHES")
    ok("VANISHES at +0.50", gran_verdict(0.50, 5.0) == "VANISHES")
    ok("UNRESOLVED in the gap", gran_verdict(1.0, 9.0) == "UNRESOLVED")
    ok("REVERSES", gran_verdict(-1.2, -4.0) == "REVERSES")
    ok("a big negative with a weak t is UNRESOLVED", gran_verdict(-1.2, -0.5) == "UNRESOLVED")
    ok("FINER_HELPS", node_verdict(0.9, 3.0) == "FINER_HELPS")
    ok("FINER_HURTS", node_verdict(-0.9, -3.0) == "FINER_HURTS")
    ok("NODE_NULL", node_verdict(0.1, 0.4) == "NODE_NULL")

    print("-- the parsers, on adversarial input ----------------------------")
    eff, dup = parse_args_line("--alg-meta Lion --dataset TinyImageNet --alg-meta RMSProp")
    ok("a repeated flag is DETECTED", "alg-meta" in dup, str(dup))
    ok("argparse's last-wins is reproduced", eff["alg-meta"] == "RMSProp")
    eff2, dup2 = parse_args_line("--dataset TinyImageNet --NN-name ResNet18_tin "
                                 "--stepsize-groups layerwise")
    ok("no false positive on a clean line", not dup2)
    ok("arm_of reads the arm from the flags", arm_of(eff2) == "lay")
    ok("arm_of rejects a foreign arm",
       arm_of({"dataset": "CIFAR10", "NN-name": "ResNet18", "stepsize-groups": "nodewise"}) is None)
    ok("plateau5 needs all 5 tail epochs",
       plateau5({e: 50.0 for e in range(95, 100)}) == 50.0
       and plateau5({e: 50.0 for e in range(96, 100)}) is None)
    ok("plateau5 ignores everything but the tail",
       abs(plateau5(dict([(e, 1.0) for e in range(95)] + [(e, 7.0) for e in range(95, 100)])) - 7.0) < 1e-9)

    print("-- the batch script's own text must match this registration -----")
    if not os.path.exists(BATCH_SH):
        ok("bin/cT1_tin_granularity.sh exists", False, BATCH_SH)
    else:
        txt = open(BATCH_SH).read()
        for label, needle in (
                ("run-name prefix tn1-", "tn1-"),
                ("dataset TinyImageNet", "TinyImageNet"),
                ("network ResNet18_tin", "ResNet18_tin"),
                ("CIFAR-10 tree-gate arm", "c10lay"),
                ("meta-stepsize 1e-3", "MST=1e-3"),
                ("alpha0 1e-6", "ALPHA0=1e-6"),
                ("100 epochs", "EPOCHS=100"),
                ("batch 100", "BATCH=100"),
                ("box -15:-2.3026", "CLIP=\"-15:-2.3026\""),
                ("AUGMENT=1", "AUG=1"),
                ("SGDm base", "--alg-base SGDm"),
                ("Lion meta", "--alg-meta Lion"),
                ("HIER=none", "HIER=none"),
                ("seeds 0 1 2", 'SEEDS="0 1 2"'),
        ):
            ok("batch script declares %s" % label, needle in txt)

    print("=" * 70)
    if fails:
        print("SELFTEST FAIL -- %d of %d checks failed" % (len(fails), len(fails)))
        for f in fails:
            print("   %s" % f)
        return 1
    print("SELFTEST PASS")
    return 0


# ---- the score ---------------------------------------------------------------
def score(runs_dir):
    paths = sorted(glob.glob(os.path.join(runs_dir, "**", PREFIX + "*.out"), recursive=True))
    print("cT1_tin_score -- `tn1`, the Tiny-ImageNet granularity ladder")
    print("runs dir: %s" % runs_dir)
    print("%d .out files match %s*" % (len(paths), PREFIX))
    if not paths:
        print("VERDICT: NO DATA -- the batch has not produced any output yet.")
        return 3
    recs = [r for r in (read_out(p) for p in paths) if r]
    print("%d carry an ARGS line" % len(recs))

    print("\n-- G0 PROVENANCE (RULE 20): the arm is read from each run's own flags --")
    kept, voided = gate0(recs)
    for r, why in voided:
        print("   VOID %-28s %s" % (os.path.basename(r["path"]), "; ".join(why)))
    print("   %d runs pass provenance, %d VOID" % (len(kept), len(voided)))
    if not kept:
        print("VERDICT: VOID -- no run survived provenance.")
        return 2

    print("\n-- G3 COMPLETENESS: >= 95/100 epochs and all 5 tail epochs present --")
    cells = collections.defaultdict(list)
    incomplete = []
    for r, _ in kept:
        done = (max(r["series"]) + 1) if r["series"] else 0
        p5 = plateau5(r["series"])
        if p5 is None or done < 0.95 * EPOCHS:
            incomplete.append((os.path.basename(r["path"]), done))
            continue
        cells[r["arm"]].append((p5, r["seed"], os.path.basename(r["path"]), done))
    for n, d in incomplete:
        print("   DROPPED %-30s %d epochs" % (n, d))
    print("   %d complete runs in %d arms" % (sum(len(v) for v in cells.values()), len(cells)))

    print("\n-- G1 CHANCE FLOOR: Tiny-ImageNet chance = %.3f%%, floor = %.1f --"
          % (CHANCE_TIN, FLOOR))
    below = []
    for arm in TIN_ARMS:
        for p5, s, n, _ in cells.get(arm, []):
            if p5 < FLOOR:
                below.append((n, p5))
    for n, p5 in below:
        print("   VOID %-30s plateau5 %.3f is below the floor" % (n, p5))
    if below:
        print("VERDICT: VOID -- a Tiny-ImageNet run did not learn.  Suspect the "
              "label map or the cache, not the optimizer.")
        return 2
    print("   every Tiny-ImageNet run clears the floor")

    print("\n-- the arms ----------------------------------------------------")
    print("   %-8s %-6s %-9s %-9s  %s" % ("arm", "n", "mean", "se", "seeds"))
    means = {}
    for arm in ARMS:
        v = cells.get(arm, [])
        if not v:
            print("   %-8s %-6s %-9s %-9s" % (arm, 0, "--", "--"))
            continue
        p5s = [x for x, _, _, _ in v]
        m, se = mean_se(p5s)
        means[arm] = p5s
        print("   %-8s %-6d %-9.3f %-9.3f  %s" % (arm, len(v), m, se,
              " ".join("s%s=%.2f" % (s, x) for x, s, _, _ in sorted(v, key=lambda z: str(z[1])))))

    print("\n-- G2 TREE GATE: does the patched tree still reproduce CIFAR-10? --")
    g = archived_granularity()
    if "layerwise" not in g:
        print("   the archived comparator is missing from the CSV -- cannot gate.")
        return 2
    arch = g["layerwise"][0]
    if CTRL_ARM not in means:
        print("   NO CONTROL RUNS -- the tree gate cannot be evaluated.")
        print("VERDICT: UNVERIFIED -- Tiny-ImageNet numbers are NOT reportable "
              "without the c10lay control.")
        return 2
    mc = statistics.mean(means[CTRL_ARM])
    d = mc - arch
    print("   in-batch c10lay %.3f   archived plain layerwise %.3f (n=%d, %d batches)"
          % (mc, arch, g["layerwise"][1], g["layerwise"][2]))
    print("   difference %+.3f pp, band +/-%.2f" % (d, TREE_BAND))
    if abs(d) > TREE_BAND:
        print("VERDICT: VOID-PATCHBROKE -- PATCH_TINYIMAGENET moved the CIFAR-10 "
              "path.  No Tiny-ImageNet number here may be reported.")
        return 2
    print("   PASS -- the patched tree reproduces the archived CIFAR-10 arm")

    print("\n-- G4 THE PRIMARY ----------------------------------------------")
    rc = 0
    if "lay" in means and "scal" in means:
        dg, se, t, df = contrast(means["lay"], means["scal"])
        v = gran_verdict(dg, t)
        print("   GRAN = M(lay) - M(scal) = %+.3f pp   se %.3f   t %+.2f (df %d)"
              % (dg, se, t, df))
        print("   bars: REPLICATES >= %+.3f with t >= %.1f | VANISHES |d| <= %.2f | "
              "REVERSES <= %.2f with t <= %.1f" % (REPL_BAR, T_BAR, VANISH_BAR,
                                                   -VANISH_BAR, -T_BAR))
        print("   archived CIFAR-10 comparator: %+.3f pp" % GRAN_CIFAR)
        print("   VERDICT (granularity at Tiny-ImageNet scale): %s" % v)
        if v == "UNRESOLVED":
            rc = max(rc, 1)
    else:
        print("   GRAN cannot be formed -- an arm is missing.")
        rc = 2
    if "node" in means and "lay" in means:
        dn, sen, tn, dfn = contrast(means["node"], means["lay"])
        vn = node_verdict(dn, tn)
        print("   NODE = M(node) - M(lay) = %+.3f pp   se %.3f   t %+.2f (df %d)"
              % (dn, sen, tn, dfn))
        print("   archived CIFAR-10 comparator: %+.3f pp" % NODE_CIFAR)
        print("   VERDICT (finer than layerwise): %s" % vn)
    else:
        print("   NODE cannot be formed -- an arm is missing.")

    print("\n-- SCOPE -------------------------------------------------------")
    print("   Tiny-ImageNet-200 is 200 classes at 64x64.  It is NOT ImageNet-1k and")
    print("   no number above may be compared with a published ImageNet-1k result.")
    print("   HIER is unset in every arm: this is the GRANULARITY ladder, and it")
    print("   says nothing about any hierarchical or shrinkage operator.")
    return rc


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs", default="/cluster/account_b/metaopt/runs")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    sys.exit(selftest() if a.selftest else score(a.runs))


if __name__ == "__main__":
    main()
