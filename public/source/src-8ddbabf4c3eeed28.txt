set -e
cd /cluster/account_b/metaopt
R=/cluster/account_b/metaopt/runs
CLIP="BETA_CLIP=-15:-2.3026"
# The paper's exact CIFAR-10 cell: base AdamW + meta Adam. Guard ON as standard.
B="--optimizer HF --alg-base AdamW --normalizer-param-base 0.999 --momentum-param-base 0.9 --weight-decay-base 0.1 --alg-meta Adam --normalizer-param-meta 0.999 --momentum-param-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"

sub () { # 1=name 2=granularity 3=alpha0 4=seed
  sbatch --parsable --job-name=$1-s$4 --partition=gpu-l4-24g --gres=gpu:l4:1 --time=03:00:00 \
    --export="ALL,AUGMENT=1,$CLIP" \
    jobs/run_cifar.sh $B --alpha0 $3 --stepsize-groups $2 --seed $4 --save-directory $R/alpha0 --run-name $1_s$4 >/dev/null
}

for s in 0 1 2; do
  for cell in "scal:scalar" "blk6:resnet18_blocks" "layer:layerwise"; do
    tag=${cell%%:*}; gran=${cell##*:}
    sub a0-$tag-1e3 $gran 1e-3 $s
    sub a0-$tag-1e4 $gran 1e-4 $s
    sub a0-$tag-1e6 $gran 1e-6 $s   # matched guarded control vs the unguarded gate2 runs
  done
done
sleep 3
echo "A0_SUBMITTED=$(squeue -h -u account_b -o '%j' | grep -c '^a0')"
echo "TOTAL_QUEUED=$(squeue -h -u account_b | wc -l)"
