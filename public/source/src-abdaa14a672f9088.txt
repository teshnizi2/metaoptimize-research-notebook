def file_sha(path):

def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d

    """the GPU name from the run's own nvidia-smi line (`<name>, <n> MiB`), or None.  DISCLOSURE ONLY."""

def lion_natural(beta_pre, mom_pre, z):
    return S9.lion_natural(beta_pre, mom_pre, z)


def lion_check(recs, ng):
    d = {"n": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "nonfinite": 0, "worst": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre, mom, z = S9._groupvec(r, "beta_pre", ng), S9._groupvec(r, "mom_pre", ng), S9._groupvec(r, "z_agg", ng)
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        if r["step"] != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        for g in range(ng):
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst"] = max(d["worst"], e)
            d["bad_lion"] += e > BH_TOL
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS and d["bad_step"] == 0
          and d["bad_lion"] == 0)
    return ok, d


def bite_run(arm, recs):
    okl, dl = lion_check(recs, NG[arm])
    hk = C.hold_keys_present(recs)
    okd, dd = C.dm_audit(recs, K_MASKED[arm], N_RECORDS)
    return okl and okd and hk == 0, dl, dd, hk

def state(M, x):
    if M[x] >= M["NWD"] - MATCH_BAR:
        return "REC"
    if M[x] <= M["k01"] + NULL_BAR:
        return "NULL"
    return "PART"


def account_of(c, t, t2):
    if c == "REC" and t == "NULL":
        return "CARRIER-DECAY-SUFFICES" if t2 == "NULL" else "CARRIER-DECAY-SUFFICES-CLASS-OPEN"
    if c == "REC" and t == "PART":
        return "SPECIFICITY-PARTIAL"
    if c == "REC" and t == "REC":
        return "ANY-THREE-SCALES"
    if c == "PART" and t == "NULL":
        return "CARRIER-DECAY-PARTIAL"
    if c == "PART" and t == "PART":
        return "PARTIAL-BOTH"
    if c == "NULL" and t == "NULL":
        return "NEEDS-MORE-THAN-THREE"
    return "CONTROL-EXCEEDS"


def decide(M, diverged=False):
    """-> (token, state word or None)."""
    if diverged:
        return "UNRESOLVED-DIVERGED", None
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED", None
    if M["NWD"] - M["k01"] < GAP_MIN:
        return "REPLICATE-FAILED", None
    c, t, t2 = state(M, "CARWD0"), state(M, "CTLWD0"), state(M, "CTL2WD0")
    return account_of(c, t, t2), "CAR-%s+CTL-%s+CTL2-%s" % (c, t, t2)

def dm_readout(runsdir):
    print("\nDESCRIPTIVE (non-gating) -- PATCH_DECAYMASK's readout on the masked arms: ||w|| per masked tensor (seed means) and")
    print("min |w|, entries |w| < 1e-3, at records 0 / 50 / 200 / 499.  k01 carries no readout (the mask is off there).")
    for a in MASKED:
        rows = [C.load_probe(runsdir, PREFIX, a, s) or [] for s in SEEDS]
        parts = []
        k = K_MASKED[a]
        for r_ in (0, 50, 200, 499):
            v = [r[r_] for r in rows if len(r) > r_ and isinstance(r[r_], dict) and isinstance(r[r_].get("dm_norm"), list)
                 and len(r[r_]["dm_norm"]) == k]
            if not v:
                parts.append("rec %d n/a" % r_)
                continue
            if k <= 3:
                nrm = "/".join("%.3g" % mean([x["dm_norm"][p] for x in v]) for p in range(k))
            else:
                names = [B.TENSORS[i][0] for i in D.dm_indices("normscale", B.TENSORS)]
                pos = [names.index(c) for c in B.CARRIERS]
                nrm = "carriers %s median %.3g" % ("/".join("%.3g" % mean([x["dm_norm"][p] for x in v]) for p in pos),
                                                     mean([sorted(x["dm_norm"])[len(x["dm_norm"]) // 2] for x in v]))

def _e_records(arm, seed, ng=None, touch=False, add_bh=False, nodm=False, adddm=False, dmforge=None, n=N_RECORDS, k=None):
    rnd = random.Random(7500 + seed + 17 * ARMS.index(arm))
    ng = ng or NG[arm]
    out = []
    for kk in range(n):
        s = 100 * kk
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        z = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [S9.f32(max(S9.LO, min(S9.HI, -13.8 + 0.3 * math.sin(kk / 30.0 + g) - 0.004 * kk))) for g in range(ng)]
        beta = [S9.f32(lion_natural(bpre[g], mom[g], z[g])[0]) for g in range(ng)]
        if touch:
            beta[0] = S9.f32(beta[0] + 0.002)
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99, "beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [z]}
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[0], "bh_held": beta[0]})
        out.append(rec)
    kk_ = K_MASKED[arm] if k is None else k
    if (kk_ and not nodm) or adddm:
        out = C.add_dm_keys(out, kk_ or 3, forge=dmforge)
    return out

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None, env_break=None,
             spec_break=None, dm_break=None, extra_break=None, manifest=None, prov=None, probe_break=None, probe_drop=None,
             gpu=None):
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(manifest if manifest is not None else D.manifest_text(B))
    _e_prov(d, prov)
    jid = 9750000
    pb = probe_break or {}
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = "layerwise" if spec_break == (a, s) else None
            dm = dm_break[1] if dm_break and dm_break[0] == (a, s) else None
            extra = extra_break[1] if extra_break and extra_break[0] == (a, s) else None
            g = (gpu or {}).get((a, s), "NVIDIA L4, 23034 MiB")
            _e_write(d, a, s, jid, L, env=env, spec=spec, dm=dm, extra=extra, gpu=g)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)


def _e_csv_variants(src, dst_dir):
    out = {"no-corpus": os.path.join(dst_dir, "does_not_exist.csv")}
    base = {"dataset": DSET, "base": "SGDm", "meta": "Lion", "meta_stepsize": MST, "alpha0": A0, "gamma": "1",
            "augment": AUG, "beta_clip": CLIP, "batch_size": str(BATCH), "epochs_requested": str(EPOCHS), "hier": "none",
            "lam": "na", "eta_ratio": "na", "superseded": "0", "collapsed": "0", "complete": "1"}
    inv = []
    for i, (gran, v) in enumerate([("scalar", 1.0), ("scalar", 70.0), ("scalar", 35.0), ("layerwise", 3.0), ("layerwise", 95.0),
                                   ("layerwise", 40.0)]):
        inv.append(dict(base, network=NET, granularity=gran, run="zstress-%d-s%d" % (i, 900 + i),
                        job_id=str(9990000 + i), seed=str(900 + i), plateau5="%.4f" % v))
    if src and os.path.exists(src):
        rows = list(csv.DictReader(open(src)))
        hdr = list(rows[0].keys())
        out["real"] = src
        p = os.path.join(dst_dir, "stress.csv")
        w = csv.DictWriter(open(p, "w", newline=""), fieldnames=hdr, extrasaction="ignore")
        w.writeheader()
        for r in rows:
            w.writerow(r)
        for r in inv:
            w.writerow(dict((k, r.get(k, "")) for k in hdr))
        out["real+invented"] = p
    hdr = ["run", "job_id", "seed", "superseded", "collapsed", "complete", "plateau5"] + S9.CELLKEYS
    p = os.path.join(dst_dir, "minimal.csv")
    w = csv.DictWriter(open(p, "w", newline=""), fieldnames=hdr, extrasaction="ignore")
    w.writeheader()
    for r in inv:
        w.writerow(r)
    out["minimal-invented"] = p
    return out


def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()


def _mid(pl):
    return (pl[0] + pl[1]) / 2.0

def main():

    ap.add_argument("runsdir", nargs="?", default="")

    ap.add_argument("--manifest", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))

        ap.error("runsdir is required")
