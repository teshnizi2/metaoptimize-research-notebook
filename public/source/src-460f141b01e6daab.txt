# =============================================================================
# THE ONE CHANGE -- an ms-aware G-DECOMP.  Byte-for-byte cCV0's decomposition() except the sign bar.
# core.decomposition() is NOT edited (RULE 16: the core is registered and sha-pinned by G-CORE).
# =============================================================================
def ulp32(x):
    """The gap between consecutive float32 values at |x|.  For |x| in [2^e, 2^(e+1)) it is 2^(e-23);
    math.frexp gives x = m * 2^E with 0.5 <= |m| < 1, so e = E - 1 and ulp32 = 2^(E - 24)."""
    x = abs(float(x))
    if x == 0.0 or not math.isfinite(x):
        return 2.0 ** -149
    _m, e = math.frexp(x)
    u = math.ldexp(1.0, e - 24)
    return u if u > 0.0 else 2.0 ** -149


def sign_tol(beta_pre, beta_post, ms):
    """TOL = SIGN_ABS_BAR + SIGN_ULP_K * ulp32(beta) / ms, at the larger |beta| of the pair."""
    return SIGN_ABS_BAR + SIGN_ULP_K * ulp32(max(abs(beta_pre), abs(beta_post))) / ms


def decomposition(recs, groups1, ms):
    """-> dict of counts.  groups1: the run's registered partition (1-based index lists).
    Identical to core.decomposition() except that the sign sub-test's absolute 1e-3 bar is replaced by
    sign_tol(beta_pre, beta_post, ms).  Also returns the worst |q - sign| and the widest bar applied."""
    g0 = [[i - 1 for i in g] for g in groups1]
    n = bad_shape = bad_z = bad_m = n_sign = bad_sign = clamped = bad_shared = 0
    worst_z = worst_m = worst_q = 0.0
    tol_max = 0.0
    for r in recs:
        n += 1
        try:
            z, m = r["z_tensor"], r["m_tensor"]
            za, ma = r["z_agg"][0], r["mom_pre"][0]
            bpre, bpost = r["beta_pre"][0], r["beta"]
            b2 = float(r.get("pt_b2", core.B2_REG))
        except (KeyError, IndexError, TypeError, ValueError):
            bad_shape += 1
            continue
        if not (len(z) == len(m) == core.N_TENSORS
                and len(za) == len(ma) == len(bpre) == len(bpost) == len(g0)):
            bad_shape += 1
            continue
        sz = sum(abs(v) for v in z) + 1e-30
        sm = sum(abs(v) for v in m) + 1e-30
        dz = max(abs(sum(z[i] for i in g) - za[k]) for k, g in enumerate(g0)) / sz
        dm = max(abs(sum(m[i] for i in g) - ma[k]) for k, g in enumerate(g0)) / sm
        worst_z, worst_m = max(worst_z, dz), max(worst_m, dm)
        bad_z += dz > core.TOL_Z
        bad_m += dm > core.TOL_M
        if len(g0) == 1:
            L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
            Lh = b2 * ma[0] + (1 - b2) * za[0]
            if abs(sum(L) - Lh) > 1e-4 * (sum(abs(x) for x in L) + 1e-30) and core.sgn(sum(L)) != core.sgn(Lh):
                bad_shared += 1
        for k in range(len(g0)):
            if not (core.BETA_LO + 1e-9 < bpost[k] < core.BETA_HI - 1e-9):
                clamped += 1
                continue
            n_sign += 1
            Lh = b2 * ma[k] + (1 - b2) * za[k]
            q = -(bpost[k] - bpre[k]) / ms
            resid = abs(q - core.sgn(Lh))
            tol = sign_tol(bpre[k], bpost[k], ms)
            worst_q = max(worst_q, resid)
            tol_max = max(tol_max, tol)
            bad_sign += resid > tol
    return {"n": n, "bad_shape": bad_shape, "bad_z": bad_z, "bad_m": bad_m, "worst_z": worst_z, "worst_m": worst_m,
            "n_sign": n_sign, "bad_sign": bad_sign, "clamped": clamped, "bad_shared": bad_shared,
            "worst_q": worst_q, "tol_max": tol_max}


def decomposition_ok(d):
    """core.decomposition_ok()'s rule verbatim (DECOMP_FAIL_MAX / SIGN_FAIL_MAX are the core's literals)."""
    n = max(d["n"], 1)
    return (d["n"] > 0 and d["bad_shape"] == 0 and d["bad_z"] / n <= core.DECOMP_FAIL_MAX
            and d["bad_m"] / n <= core.DECOMP_FAIL_MAX and d["n_sign"] > 0
            and d["bad_sign"] / max(d["n_sign"], 1) <= core.SIGN_FAIL_MAX
            and d["bad_shared"] / n <= core.DECOMP_FAIL_MAX)

def nom_word(dom, top3):
    if dom is None or top3 is None:
        return None
    if dom >= DOM_HI and top3 >= TOP3_BAR:
        return "NOMINATED"
    if dom <= DOM_LO:
        return "NOT-NOMINATED"
    return "NOMINATION-PARTIAL"


def arm_word(d, kind):
    if kind == "ISO":
        return "ISO-RESCUES" if d >= RESCUE_BAR else ("ISO-NO-RESCUE" if d <= NULL_BAR else "ISO-PARTIAL")
    return "CTL-RESCUES" if d >= RESCUE_BAR else ("CTL-NULL" if d <= NULL_BAR else "CTL-PARTIAL")


def decide(M):
    """M: {'lev': {arm: mean}, 'rng': {arm: range}, 'runs': [plateau5...], 'dom': DOM_C, 'top3': TOP3_C}
    -> (branch, stamps).  Reads no record."""
    stamps = []
    if any(M["rng"][a] > DIVERGED_BAR for a in ARMS) or any(not (LEVEL_MIN <= x <= LEVEL_MAX) for x in M["runs"]):
        return "UNRESOLVED-DIVERGED", stamps
    nom = nom_word(M["dom"], M["top3"])
    if M["lev"]["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED", ["UNBRANCHED-%s" % nom]
    d_iso = M["lev"]["ISO"] - M["lev"]["k01"]
    d_ctl = M["lev"]["CTL"] - M["lev"]["k01"]
    iso, ctl = arm_word(d_iso, "ISO"), arm_word(d_ctl, "CTL")
    if iso == "ISO-NO-RESCUE" or ctl == "CTL-NULL":
        stamps.append("FLOOR-READINGS-ARE-BOUNDS")
    if M["lev"]["ISO"] < M["lev"]["CTL"]:
        stamps.append("CTL-ABOVE-ISO")
    return "%s+%s+%s" % (nom, iso, ctl), stamps


def all_branches():
    return ["UNRESOLVED-DIVERGED", "SCALAR-NOT-COLLAPSED"] + ["%s+%s+%s" % t for t in
                                                             itertools.product(NOM_WORDS, ISO_WORDS, CTL_WORDS)]


def licence(branch):
    if branch in ("UNRESOLVED-DIVERGED",):
        return "NONE: a seed range > 5 pp or a run outside [5, 90]; no sentence"
    if branch == "SCALAR-NOT-COLLAPSED":
        return "NONE about carriers: k01 > 34 pp, the cell is not an R50 collapse in this batch"
    nom, iso, ctl = branch.split("+")
    head = HEADLINE.get(branch)
    body = "; ".join((NOM_SENT[nom], ISO_SENT[iso], CTL_SENT[ctl]))
    return ("%s. %s." % (head, body)) if head else (body + ".")

def score(runsdir, csv_path=core.DEFAULT_CSV, quiet=False):
    """-> (final_line, rc).  rc 0 scored, 1 gate failed, 2 incomplete."""
    led = core.Ledger(quiet=quiet)
    P = (lambda *a: None) if quiet else print
    P("=" * 100)
    P("cST2 -- `cst1`: the carrier contrast at meta step 3e-4 (FROZEN SUCCESSOR of cST1; ms-aware G-DECOMP).  "
      "runsdir %s" % runsdir)
    P("=" * 100)
    runs, notes = core.read_outs(runsdir, BATCH, ARMS)
    for n in notes:
        led.note(n)

    # ---- INCOMPLETE ------------------------------------------------------------------------------------------------
    P("\n[G-COMPLETE] every (arm, seed) has a COMPLETE .out and 500 probe records")
    probes = {}
    complete = True
    for a in ARMS:
        for s in SEEDS:
            r = runs.get((a, s))
            ok = led.chk(r is not None and r["complete"], "%s-s%d: complete .out (RUN_DONE, epochs 0..99, finite "
                                                          "plateau5)" % (a, s), gate="COMPLETE")
            pr = core.read_probe(runsdir, BATCH, a, s)
            probes[(a, s)] = pr
            ok2 = led.chk(pr is not None and len(pr["recs"]) >= core.N_RECORDS,
                          "%s-s%d: >= %d probe records" % (a, s, core.N_RECORDS),
                          "%d" % (len(pr["recs"]) if pr else 0), gate="COMPLETE")
            complete &= ok and ok2
    if not complete:
        final = "FINAL: INCOMPLETE | %d/%d complete" % (sum(1 for k, v in runs.items() if v["complete"]), N_RUNS)
        P("\n" + final)
        return final, 2

    # ---- HARNESS gates ---------------------------------------------------------------------------------------------
    P("\n[G-CORE / G-PROV / G-STRUCT] the code and the harness that ran")
    led.chk(core.core_sha() == CORE_SHA, "G-CORE: cCV0_carrier_vote_core.py sha256 == registered", core.core_sha(),
            gate="HARNESS")
    prov = core.read_kv(os.path.join(runsdir, BATCH, "PROVENANCE.txt"))
    led.chk(prov is not None, "G-PROV: <runsdir>/%s/PROVENANCE.txt present" % BATCH, gate="HARNESS")
    prov = prov or {}
    for key, want in (("MODE", "submit"), ("HF_SHA256", core.HF_LIVE_SHA), ("BUILD_NETWORK_SHA256",
                                                                            core.BUILD_NETWORK_SHA),
                      ("TRAIN_PY_SHA256", core.TRAIN_PY_SHA), ("RUNNER_SHA256", core.RUNNER_LIVE_SHA),
                      ("CORE_SHA256", CORE_SHA), ("SCORER_SHA256", REG_SCORER_SHA)):
        led.chk(prov.get(key) == want, "G-PROV: %s == %s" % (key, want), "got %s" % prov.get(key), gate="HARNESS")
    man = None
    mp = os.path.join(runsdir, BATCH, "PARTITION-MANIFEST.txt")
    if os.path.isfile(mp):
        man = open(mp).read()
    led.chk(man == manifest_text(), "G-STRUCT: PARTITION-MANIFEST.txt byte-identical to manifest_text() "
                                    "(%d bytes)" % len(manifest_text()), gate="HARNESS")

    P("\n[G-ARGS / G-ENV / G-PT / G-FOREIGN / G-HEADER / G-RECORDS] every run's own lines")
    for a in ARMS:
        for s in SEEDS:
            r, pr, tag = runs[(a, s)], probes[(a, s)], "%s-s%d" % (a, s)
            led.chk(len(r["args"]) == 1 and args_ok(r["args"][0], a, s), "G-ARGS %s: one ARGS line == registered"
                    % tag, gate="HARNESS")
            env_ok = len(r["env"]) == 1
            pdir = None
            if env_ok:
                m = re.search(r" PROBE_DIR=(\S+)", r["env"][0])
                pdir = m.group(1) if m else None
                env_ok = (re.sub(r" PROBE_DIR=\S+", "", r["env"][0]) == "ENV: " + core.ENV_FIXED and pdir is not None
                          and pdir.endswith("/runs/%s/probe_%s-%s" % (BATCH, BATCH, tag)))
            led.chk(env_ok, "G-ENV %s: one ENV line == registered (modulo PROBE_DIR, which is the run's own)" % tag,
                    gate="HARNESS")
            m = core.PT_RE.match(r["pt"][0]) if len(r["pt"]) == 1 else None
            led.chk(m is not None and m.group(1) == PT_TYPE[a] and m.group(2) == pdir,
                    "G-PT %s: one PROBE_TENSOR line, type=%s, dir == PROBE_DIR" % (tag, PT_TYPE[a]), gate="HARNESS")
            led.chk(not r["foreign"], "G-FOREIGN %s: no intervention-kind line (the live tree ran)" % tag,
                    "; ".join(r["foreign"])[:120], gate="HARNESS")
            h = pr["hdr"] or {}
            led.chk(h.get("PROBE_TENSOR") == 1 and h.get("every") == core.PROBE_EVERY and h.get("stepsize_type") ==
                    PT_TYPE[a] and h.get("n_tensors") == core.N_TENSORS and tuple(h.get("param_numels", ())) ==
                    core.NUMEL[NET] and h.get("meta_alg") == "Lion" and h.get("Lion_beta2") == 0.9,
                    "G-HEADER %s: probe_tensor.json registered (%s numels)" % (tag, NET), gate="HARNESS")
            led.chk(pr["bad"] == 0 and [x.get("step") for x in pr["recs"]] == list(range(0, core.META_STEPS,
                                                                                          core.PROBE_EVERY)),
                    "G-RECORDS %s: 500 parseable records at steps 0..49,900" % tag, gate="HARNESS")
    if led.fails("HARNESS"):
        final = "FINAL: HARNESS-UNSOUND | %d harness check(s) failed: %s" % (len(led.fails("HARNESS")),
                                                                            led.fails("HARNESS")[0])
        P("\n" + final)
        return final, 1

    P("\n[G-SPEC] the partition that RAN, read off the records (the BROKEN-SPEC null's only separator)")
    for a in ARMS:
        for s in SEEDS:
            recs = probes[(a, s)]["recs"]
            nc = set(len(x.get("beta", [])) for x in recs)
            nz = set(len((x.get("z_agg") or [[]])[0]) for x in recs)
            led.chk(nc == {N_COORDS[a]} and nz == {N_COORDS[a]}, "G-SPEC %s-s%d: every record carries %d beta / z_agg "
                                                                 "coordinate(s)" % (a, s, N_COORDS[a]),
                    "beta %s z_agg %s" % (sorted(nc), sorted(nz)), gate="SPEC")
    if led.fails("SPEC"):
        final = "FINAL: SPEC-NOT-APPLIED | %s" % led.fails("SPEC")[0]
        P("\n" + final)
        return final, 1

    P("\n[G-DECOMP] per-group sums of the capture vs the harness; applied step vs sign(L)")
    P("  tolerance (THE ONE CHANGE from cST1): |-(beta - beta_pre)/ms - sign(L)| <= %.1e + %.1f*ulp32(beta)/ms "
      "= %.4e at ms %g, |beta| < 16" % (SIGN_ABS_BAR, SIGN_ULP_K, SIGN_ABS_BAR + SIGN_ULP_K * ULP32_MAX_IN_CLIP / MS,
                                        MS))
    P("  cST1 applied an absolute %.1e; the float32 bound 0.5*ulp32/ms is %.4e here and exceeds it for "
      "ms < %.6e" % (PRED_SIGN_BAR, 0.5 * ULP32_MAX_IN_CLIP / MS, MS_CROSSOVER))
    for a in ARMS:
        for s in SEEDS:
            d = decomposition(probes[(a, s)]["recs"], core.parse_sets(SPEC[a]), MS)
            led.chk(decomposition_ok(d), "G-DECOMP %s-s%d" % (a, s),
                    "shape %d z %d m %d (worst %.1e / %.1e) sign %d/%d clamped %d shared %d | worst|q-s| %.3e "
                    "bar %.3e" % (
                        d["bad_shape"], d["bad_z"], d["bad_m"], d["worst_z"], d["worst_m"], d["bad_sign"],
                        d["n_sign"], d["clamped"], d["bad_shared"], d["worst_q"], d["tol_max"]), gate="DECOMP")
    if led.fails("DECOMP"):
        final = "FINAL: UNRESOLVED-DECOMPOSITION | %s" % led.fails("DECOMP")[0]
        P("\n" + final)
        return final, 1

    # ---- NOMINATION (k01 records only) -----------------------------------------------------------------------------
    P("\n[NOMINATION] k01's records only, pooled over seeds (the registered check)")
    for s in SEEDS:
        core.describe_stats("k01-s%d" % s, core.vote_stats(probes[("k01", s)]["recs"]), out=P)
    nst = core.vote_stats([x for s in SEEDS for x in probes[("k01", s)]["recs"]])
    core.describe_stats("k01 POOLED", nst, out=P)
    P("  calibration at ms 1e-3 (registration): scalar DOM_C %.3f-%.3f pooled %.4f; layerwise 0.0000"
      % (CALIB_SCALAR_DOM + (CALIB_SCALAR_POOLED,)))

    # ---- LEVELS ----------------------------------------------------------------------------------------------------
    P("\n[LEVELS] plateau5 (train5) per run; arm means; contrasts in pp and SE (SE_PRIOR %.6f)" % SE_PRIOR)
    lev, rng, trn, allruns = {}, {}, {}, []
    for a in ARMS:
        v = [runs[(a, s)]["p5"] for s in SEEDS]
        t = [runs[(a, s)]["t5"] for s in SEEDS]
        lev[a], rng[a], trn[a] = core.mean(v), max(v) - min(v), core.mean(t)
        allruns += v
        P("  %-4s %s  mean %.4f  range %.3f  | train %s mean %.4f" % (a, " ".join("%.3f" % x for x in v), lev[a],
                                                                      rng[a], " ".join("%.3f" % x for x in t), trn[a]))
    d_iso, d_ctl, p = lev["ISO"] - lev["k01"], lev["CTL"] - lev["k01"], lev["ISO"] - lev["CTL"]
    for nm, d in (("D_ISO = ISO - k01", d_iso), ("D_CTL = CTL - k01", d_ctl), ("P = ISO - CTL", p)):
        P("  %-20s %+8.4f pp = %+8.2f SE" % (nm, d, d / SE_PRIOR))
    ins = [runs[(a, s)]["p5"] - lev[a] for a in ARMS for s in SEEDS]
    sig_in = math.sqrt(sum(x * x for x in ins) / (len(ins) - len(ARMS)))
    P("  in-batch sigma %.4f (df %d); SIGMA_PRIOR %.4f -> %s" % (sig_in, len(ins) - len(ARMS), SIGMA_PRIOR,
                                                                 "SIGMA-PRIOR-FROZEN"))

    M = {"lev": lev, "rng": rng, "runs": allruns, "dom": nst["DOM_C"], "top3": nst["TOP3_C"]}
    branch, stamps = decide(M)
    tw = None
    if branch not in ("UNRESOLVED-DIVERGED", "SCALAR-NOT-COLLAPSED"):
        tw = "%s+%s" % (arm_word(trn["ISO"] - trn["k01"], "ISO"), arm_word(trn["CTL"] - trn["k01"], "CTL"))
        stamps.append("TRAIN-AGREES" if tw == "+".join(branch.split("+")[1:]) else "TRAIN-DISAGREES(%s)" % tw)
    stamps += ["DOM_C=%.4f" % nst["DOM_C"], "TOP3_C=%.4f" % nst["TOP3_C"],
               "PINNED-FRAC=%s" % core.fmt(nst["PINNED_FRAC"]), "HARNESS-CLEAN", "SPEC-APPLIED", "DECOMPOSITION-OK",
               "ONE-CELL-MS-3E-4", "ALPHA0-1E-6-ONLY", "HORIZON-100-ONLY", "ONE-NETWORK-RESNET18_C100",
               "CTL-HAS-A-BIAS-MEMBER", "NO-LAYERWISE-IN-BATCH", "NOMINATION-FROM-K01-RECORDS-ONLY",
               "SIGMA-PRIOR-FROZEN"]

    # ---- DESCRIPTIVE ----------------------------------------------------------------------------------------------
    P("\n[DESCRIPTIVE, gating nothing] the counterfactual shared vote inside ISO and CTL; beta per group")
    for a in ("ISO", "CTL"):
        core.describe_stats("%s POOLED" % a, core.vote_stats([x for s in SEEDS for x in probes[(a, s)]["recs"]]), out=P)
    for a in ARMS:
        for s in SEEDS:
            recs = probes[(a, s)]["recs"]
            for c in range(N_COORDS[a]):
                b = [x["beta"][c] for x in recs]
                pk = max(range(len(b)), key=lambda i: b[i])
                P("  beta %s-s%d g%d: peak %.3f at step %d; terminal %.3f; share at floor %.3f"
                  % (a, s, c, b[pk], recs[pk]["step"], b[-1], sum(1 for x in b if x <= core.BETA_LO + 1e-6) / len(b)))
    rows = core.read_corpus(csv_path, exclude_prefix=BATCH)
    if rows is not None:
        P("  corpus anchors (non-gating, filtered): %s" % "; ".join("%s %.4f" % kv for kv in ANCHOR.items()))

    P("\nBRANCH   %s" % branch)
    P("LICENCE  %s" % licence(branch))
    P("NOT LICENSED  %s" % NOT_LICENSED)
    final = "FINAL: %s | %s" % (branch, " | ".join(stamps))
    P("\n" + final)
    return final, 0

def _f32(x):
    return struct.unpack("f", struct.pack("f", x))[0]


def quantise_beta(recs, ms, base=-13.8, span=0.35, corrupt=None, corrupt_from=None):
    """A COPY of `recs` whose beta_pre / beta are what the LIVE float32 harness posts:
        beta_pre[k]  = float32(base + span * i / len(recs))          (spread across a float32 binade)
        beta[k]      = float32(beta_pre[k] - float32(ms) * sign(L))  (round-to-nearest, as torch does)
    `corrupt`: None, 'flip' (apply the step in the WRONG direction) or 'zero' (apply no step at all),
    from record index `corrupt_from` on.  z / m / z_agg / mom_pre are left untouched, so only the sign
    sub-test can see the difference."""
    b2 = core.B2_REG
    msf = _f32(ms)
    out = []
    for i, r in enumerate(recs):
        q = dict(r)
        za, ma = r["z_agg"][0], r["mom_pre"][0]
        bpre, bpost = [], []
        for k in range(len(za)):
            bp = _f32(base + span * (i / float(max(len(recs) - 1, 1))) + 0.0007 * k)
            s = core.sgn(b2 * ma[k] + (1.0 - b2) * za[k])
            bad = corrupt is not None and corrupt_from is not None and i >= corrupt_from
            if bad and corrupt == "flip":
                nb = _f32(bp + msf * s)
            elif bad and corrupt == "zero":
                nb = bp
            else:
                nb = _f32(bp - msf * s)
            bpre.append(bp)
            bpost.append(nb)
        q["beta_pre"] = [bpre]
        q["beta"] = bpost
        out.append(q)
    return out


def _synth_batch(root, levels, dom, rng, spec_break=None, decomp_break=None, drop=None, no_done=None, args_mut=None,
                 env_mut=None, pt_mut=None, foreign=None, hdr_mut=None, short=None, prov_mut=None, man_mut=None,
                 seed_jitter=None, kind_else="nodom", cache=None, extra_trunc=None, ws="/cluster/account_b/metaopt",
                 recs_mut=None):
    os.makedirs(os.path.join(root, BATCH), exist_ok=True)
    job = 7000000
    for a in ARMS:
        for s in SEEDS:
            job += 1
            if drop == (a, s):
                continue
            groups = core.parse_sets(SPEC[a])
            share = dom if a == "k01" else 0.0
            key = (a, share, kind_else, spec_break == (a, s), decomp_break == (a, s), short == (a, s))
            recs = cache.get(key) if (cache is not None and recs_mut is None) else None
            if recs is None:
                recs = core.synth_records(groups, share, rng, MS, kind_else=kind_else,
                                          break_decomp=decomp_break == (a, s),
                                          n_coords=1 if spec_break == (a, s) else None)
                if short == (a, s):
                    recs = recs[:-1]
                if cache is not None and recs_mut is None:
                    cache[key] = recs
            if recs_mut is not None:
                recs = recs_mut(recs)
            lvl = levels[a] + (seed_jitter.get((a, s), 0.0) if seed_jitter else 0.0)
            pdir = os.path.join(ws, "runs", BATCH, "probe_%s-%s-s%d" % (BATCH, a, s))
            env = core.ENV_FIXED.replace("PROBE=100", "PROBE=100 PROBE_DIR=" + pdir)
            if env_mut == (a, s):
                env = env.replace("AUGMENT=1", "AUGMENT=0")
            al = args_line(a, s, ws)
            if args_mut == (a, s):
                al = al.replace("--meta-stepsize 3e-4", "--meta-stepsize 1e-3")
            run = core.write_synth_run(root, BATCH, a, s, job, al, PT_TYPE[a] if pt_mut != (a, s) else "scalar",
                                       lvl, recs=recs, drop_run_done=no_done == (a, s), env=env,
                                       hdr_override={"param_numels": list(core.NUMEL["ResNet18"])}
                                       if hdr_mut == (a, s) else None)
            # the PT line must name the run's PROBE_DIR (the synthetic writer names its local dir): rewrite it
            outp = os.path.join(root, "%s-%d.out" % (run, job))
            txt = open(outp).read()
            txt = re.sub(r"(PROBE_TENSOR: on .* dir=)\S+", lambda m: m.group(1) + pdir, txt)
            if foreign and foreign[0] == (a, s):
                txt = txt.replace("\nEpoch 0,", "\n%s\nEpoch 0," % foreign[1], 1)
            open(outp, "w").write(txt)
            if extra_trunc == (a, s):
                job2 = job + 500
                # a later, truncated resubmission (higher job id, 40 epochs, no RUN_DONE); recs=None leaves the
                # complete run's probe records in place
                core.write_synth_run(root, BATCH, a, s, job2, al, PT_TYPE[a], 1.0, recs=None, n_epochs=40,

            "SCORER_SHA256": REG_SCORER_SHA}
    if prov_mut:
        prov.update(prov_mut)
    with open(os.path.join(root, BATCH, "PROVENANCE.txt"), "w") as f:

            f.write("%s %s\n" % (k, v))
    mt = manifest_text()
    if man_mut:

    led.chk(d_new["bad_sign"] == 0 and decomposition_ok(d_new) and d_new["worst_q"] <= 0.5 * ULP32_MAX_IN_CLIP / MS,
            "the FIX: cST2's ms-aware bar accepts every one of them, worst |q-s| %.3e inside the bound %.3e"
            % (d_new["worst_q"], 0.5 * ULP32_MAX_IN_CLIP / MS))
    q1e3 = quantise_beta(core.synth_records(core.parse_sets("scalar"), 0.8, random.Random(1), 1e-3), 1e-3)
    led.chk(core.decomposition_ok(core.decomposition(q1e3, core.parse_sets("scalar"), 1e-3))
            and decomposition_ok(decomposition(q1e3, core.parse_sets("scalar"), 1e-3)),
            "at ms 1e-3 (cct1's cell) the OLD and the NEW bar BOTH pass the same quantised records -- the successor "
            "changes nothing where the predecessor was right")
    led.chk(decomposition(quantise_beta(qrecs, MS, corrupt="flip", corrupt_from=100),
                          core.parse_sets("scalar"), MS)["bad_sign"] == 400,
            "a FLIPPED applied step on 400 of 500 records is still caught by the new bar (|q-s| = 2)")
    led.chk(not decomposition_ok(decomposition(quantise_beta(qrecs, MS, corrupt="zero", corrupt_from=400),
                                               core.parse_sets("scalar"), MS)),
            "a MISSING applied step on 100 of 500 records is still caught by the new bar (|q-s| = 1)")
    led.chk(decomposition_ok(decomposition(quantise_beta(qrecs, MS, corrupt="flip", corrupt_from=495),
                                           core.parse_sets("scalar"), MS))
            and not decomposition_ok(decomposition(quantise_beta(qrecs, MS, corrupt="flip", corrupt_from=494),
                                                   core.parse_sets("scalar"), MS)),
            "the core's 1 %-of-coordinates rule is UNCHANGED: 5/500 faults pass, 6/500 fail")
    run_case("float32-quantised betas (the real harness) -> scored", "NOMINATED+ISO-RESCUES+CTL-NULL |", 0,
             recs_mut=lambda r: quantise_beta(r, MS))
    run_case("float32-quantised + flipped step on k01-s112..CTL-s114", "UNRESOLVED-DECOMPOSITION", 1,
             recs_mut=lambda r: quantise_beta(r, MS, corrupt="flip", corrupt_from=0))

    print("\n[E2] O2 invariance: injected corpus rows at cst1's own cells move the filtered sigma, not FINAL")
    root = os.path.join(tmp, "o2")
    _synth_batch(root, base, 0.8, rng, cache=cache)
    f0, _ = score(root, csv_path, quiet=True)
    inj = os.path.join(tmp, "inject.csv")
    if csv_path and os.path.isfile(csv_path):
        rr = list(csv.DictReader(open(csv_path)))
        head = list(rr[0].keys())
        with open(inj, "w", newline="") as f:
            w = csv.DictWriter(f, fieldnames=head)
            w.writeheader()
            for r in rr:
                w.writerow(r)
            for k in range(11):
                r = dict(rr[0])
                r.update({"run": "zzinject-%d" % k, "job_id": str(9900000 + k), "network": NET, "dataset": DSET,
                          "meta_stepsize": "1e-3", "alpha0": "1e-6", "granularity": SPEC["ISO"],
                          "plateau5": str(1 + 9 * k), "superseded": "0", "collapsed": "0", "complete": "1",
                          "epochs_requested": "100", "augment": "1", "beta_clip": "-15:-2.3026", "batch_size": "100",
                          "hier": ""})
                w.writerow(r)
        s_inj = core.pooled_sigma(core.read_corpus(inj), NET, DSET)[0]
        f1, _ = score(root, inj, quiet=True)
        f2, _ = score(root, None, quiet=True)
        led.chk(f0 == f1 == f2 and abs(s_inj - SIGMA_PRIOR) > 1.0, "FINAL identical on the corpus, the injected "
                                                                   "corpus (filtered sigma %.2f) and no corpus" % s_inj)
    shutil.rmtree(root, ignore_errors=True)

    print("\n[E3] the one-argument subprocess invocation")
    root = os.path.join(tmp, "sub")
    _synth_batch(root, base, 0.8, rng, cache=cache)
    p = subprocess.run([sys.executable, os.path.abspath(__file__), root], capture_output=True, text=True)
    led.chk(p.returncode == 0 and "FINAL: NOMINATED+ISO-RESCUES+CTL-NULL" in p.stdout, "`%s <runsdir>` exit 0"
            % os.path.basename(__file__))
    for f in os.listdir(root):
        if f.startswith("cst1-CTL-s114"):
            os.remove(os.path.join(root, f))
    p = subprocess.run([sys.executable, os.path.abspath(__file__), root], capture_output=True, text=True)
    led.chk(p.returncode == 2 and "FINAL: INCOMPLETE" in p.stdout, "a dropped run: exit 2, FINAL: INCOMPLETE")
    shutil.rmtree(tmp, ignore_errors=True)

    fails = led.fails()
    n_pass = sum(1 for ok, _l, _g in led.rows if ok)
    print("\nSELFTEST %s: %d PASS / %d FAIL" % ("PASSES" if not fails else "FAILS", n_pass, len(fails)))
    for f in fails:
        print("  FAILED: %s" % f)
    return 0 if not fails else 1


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt")
    ap.add_argument("--csv", default=core.DEFAULT_CSV)

    sys.exit(rc)
