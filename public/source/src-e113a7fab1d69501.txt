# THE ARMS (group 1 is the held group of every held arm; every hold is tri:9428 = PlainNet's dose)
#   k01        scalar                                        ISO        [59,3] free
#   HOLDBIG3   [59,3]  GROUP_HOLD=<50+53+59>:tri:9428        ONE50      [61,1] {50} free (ciso1 / cdep1 ONE)

#   ONE59      [61,1] {59} free                              ONE59BIG   [61,1]  GROUP_HOLD=layer4.1.bn2.weight:tri:9428
#   ONE53      [61,1] {53} free                              ONE53BIG   [61,1]  GROUP_HOLD=layer4.0.shortcut.1.weight:tri:9428
#   ISOSPLIT   [59,1,2] {50} held / {53,59} free             GROUP_HOLD=layer4.0.bn2.weight:tri:9428
#
# RULE 20.  Every ARGS value is ONE shell token.  GROUP_HOLD rides sbatch's --export list and CANNOT ride the runner's ENV
# line: PATCH_GROUPHOLD prints its witness line at construction on EVERY run, and that line is what the separate ENV audit
# and the scorer's G-GHOLD read.  analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  cvt8 by sacct: 17.30 GPU-h / 21 = 0.82 GPU-h per 100-epoch ResNet18_c100 run.  30 jobs ~= 24.6 GPU-h expected;
# HARD BOUND = WALL x 30 = 90 GPU-h (WALL 03:00:00).
#
# USAGE   bash bin/cVT10_onevsthree.sh               # dry run: every guard, every composed line printed; submits nothing
#         bash bin/cVT10_onevsthree.sh --submit      # the ONLY submitting form
# ENV     CVT10_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
unset VOTE_W BETA_HOLD GROUP_HOLD COMP_HOLD REST_HOLD WINDOW_HOLD
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"

WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"

HARN=$WS/harness_cvt8/cifar10
RUNNER=$WS/jobs/run_cifar_cvt8.sh
SAVE=$WS/runs/cvt10

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"

ARMS="k01 ISO HOLDBIG3 ONE50 ONE50BIG ONE59 ONE59BIG ONE53 ONE53BIG ISOSPLIT"

LIVE_BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d

RUNNER_SHA=1dbe4ff52eccbbb1c9936f4a687ed88172dd41c48b14ae60eb353b45f766f5e0
# shellcheck disable=SC1091

SC="$REPO/analysis/cVT10_onevsthree_score.py"

echo "   ONE53     $G_ONE53"
echo "   ONE53BIG  ONE53's  GROUP_HOLD=$GH_53"

for f in "$SC" "$SELF" "$BSB" "$TESTB" "$TESTGR" "$STAGE8" "$STAGE7" "$SC8" "$SC7" "$SC1" "$SCD" "$EXCL" "$EXCLT" "$LIBF" "$AGF" \
         "$CSVP" "$ARCH8" "$RUNNER" "$HFH" "$HFRH" "$BNH"; do

  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT10_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"

  fi

  || guard_fail "guard 1b2: jobs/run_cifar_cvt8.sh in the repo differs from $RUNNER"

# ---- guard 1c: the scorer selftest, INCLUDING O2, the real-record nulls and the invocation ----

  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION\|Traceback' "$SELFT" | head -20; tail -5 "$SELFT"; }

  || guard_fail "guard 1c1: the selftest skipped a check on this host"

  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt10's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT10_onevsthree_score.py <runsdir>. exits 0" "$SELFT" && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry a seed of the block 120-123" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"

  && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cIS1's ISO" "$SELFT" \
  && grep -q "^  PASS B cvt8's registered scorer: HOLDBIG's GROUP_HOLD string, GROUP_HOLD witness and REST_HOLD witness == HOLDBIG3's" "$SELFT" \
  && grep -q "^  PASS B cvt8's tree (HF, runner, build_network), its names and its ENV line are this batch's" "$SELFT" \
  && echo "guard 1c5: ONE50 byte-identical to ciso1's / cdep1's ONE; ISO to ciso1's; HOLDBIG3 is cvt8's HOLDBIG; cvt8's tree" \
  || guard_fail "guard 1c5: the byte-identity checks did not pass"
grep -q "^  PASS B G-BITE PASSES on the 6 landed ONE runs" "$SELFT" && grep -q "^  PASS B THE BROKEN-SINGLETON-HOLD NULL ON REAL RECORDS" "$SELFT" \
  && grep -q "^  PASS B cvt8's 3 landed HOLDBIG runs PASS G-BITE as HOLDBIG3" "$SELFT" && grep -q "^  PASS B cvt8's HOLDHIGH (tri:8609) read as HOLDBIG3 FAILS" "$SELFT" \
  && grep -q "^  PASS B G-BITE PASSES on cvt8's 6 landed k01 / ISO runs" "$SELFT" \
  && echo "guard 1c7: G-BITE passes 6 landed ONE, 6 cvt8 controls, 3 cvt8 HOLDBIG; FAILS 12 real-record singleton nulls and 3 wrong-dose reads" \
  || guard_fail "guard 1c7: the real-record G-BITE checks did not pass"

  && echo "guard 1c8: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value on the registration corpus" \
  || guard_fail "guard 1c8: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E BROKEN SINGLETON HOLD: every one-tensor arm unheld" "$SELFT" && grep -q "^  PASS E HOLD ONLY AT PROBE STEPS" "$SELFT" \
  && grep -q "^  PASS E WRONG DOSE: ONE50BIG on tri:8609" "$SELFT" && grep -q "^  PASS E UNHELD GROUP TOUCHED: ISOSPLIT" "$SELFT" \
  && echo "guard 1c9: the broken-singleton null, a probe-step-only hold, a wrong dose and a touched unheld group reach PATCH-NOT-VERIFIED" \
  || guard_fail "guard 1c9: G-BITE break cases did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt10-' "$CSVP" && guard_fail "guard 2: cvt10-* already in the CSV" || echo "guard 2: no cvt10-* row in the corpus"
python3 - "$CSVP" <<'PYEOF' || guard_fail "guard 2c: corpus premise failed"

PYEOF
NOUT=$(ls "$WS/runs"/cvt10-*.out 2>/dev/null | wc -l | tr -d ' ')

                      || guard_fail "guard 2d: $NSEED .out files already ran seed 120-123"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt10-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt10-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt10-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt10-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt10-* in squeue" || guard_fail "guard 2f: $NQ cvt10-* already queued"

# ---- guard 3/4a: cvt8's tree unchanged (cvt7's tree too) --------------------------------
bash "$STAGE8" >/tmp/cvt10_stage8.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT8_stage_harness.sh (registered, UNEDITED, verify mode) VERIFIES cvt8's tree unchanged:"; grep -E 'V-|VERIFIED' /tmp/cvt10_stage8.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt8's tree is NOT consistent"; tail -20 /tmp/cvt10_stage8.$$; }

  && echo "guard 4a1: bin/cVT7_stage_harness.sh (registered, unedited, verify mode) VERIFIES cvt7's tree unchanged" \
  || { guard_fail "guard 4a1: cvt7's tree is NOT consistent"; tail -10 /tmp/cvt10_stage7.$$; }
rm -f /tmp/cvt10_stage7.$$
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" || guard_fail "guard 3: CIFAR-100 not visible"
[ "$(sha "$BNH")" = "$LIVE_BN_SHA" ] && [ "$(sha "$LIVE/build_network.py")" = "$LIVE_BN_SHA" ] \
  && echo "guard 3b: build_network.py == the LIVE one c7998883..." || guard_fail "guard 3b: build_network.py is not the LIVE one"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFRH")" = "$HF_CVT7_SHA" ] && [ "$(sha "$H7/Optimizers/HF.py")" = "$HF_CVT7_SHA" ] \
  && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] \
  && echo "guard 3c: HF.py == cvt8's 5197dc2e...; .pre_resthold == cvt7's 2396f2be...; the LIVE HF.py is 4732b74a..." \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == 1dbe4ff5... (cvt8's runner, reused unchanged)" \
  || guard_fail "guard 3e: $RUNNER is not the pinned cvt8 runner"
grep -q PATCH_GROUPHOLD "$LIVE/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE HF.py carries PATCH_GROUPHOLD" \
  || echo "guard 3d: the LIVE HF.py carries no PATCH_GROUPHOLD"

# ---- guard 4f: the real-run bite log on THIS HF and THIS scorer ----
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH"); SCSHA=$(sha "$SC")
if [ -f "$BLOG" ] && grep -q "^BITE_RC 0$" "$BLOG" && [ "$(grep -c '^ALL PASS$' "$BLOG")" = 1 ] && [ "$(grep -c '^  FAIL' "$BLOG")" = 0 ] \
   && grep -q "^TREE $HARN HF $HFSHA build_network $BNSHA$" "$BLOG" && grep -q "^SCORER_SHA256 $SCSHA$" "$BLOG" \
   && grep -q "^DRIVER_SHA256 $(sha "$TESTB")$" "$BLOG" && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTGR")$" "$BLOG" \
   && [ "$(grep -c "PASS RB1 .* the scorer's OWN bite_check PASSES on the 30 real records" "$BLOG")" = 5 ] \
   && [ "$(grep -c "PASS RB1 .* printed exactly ONE GROUP_HOLD line == the scorer's witness" "$BLOG")" = 5 ] \

   && grep -q "PASS RB2 SPLITFREE \[59,1,2\]: 3 groups at every record" "$BLOG" \
   && [ "$(grep -c "PASS RB3 .* the held group sits at -15 after every step" "$BLOG")" = 2 ] \
   && grep -q "PASS RB0 ONE50 free vs itself: every parameter and buffer after the last step BITWISE-IDENTICAL" "$BLOG" \
   && [ "$(grep -c "PASS RB4 .* raises ValueError at construction" "$BLOG")" = 3 ]; then
  echo "guard 4f: the real-run bite log (tests/test_cvt10_hold_realrun.py) ALL PASS on THIS HF, THIS scorer: $(grep -c '^  PASS' "$BLOG") PASS / $(grep -c '^  FAIL' "$BLOG") FAIL (sha $(sha "$BLOG" | cut -c1-8))"

# ---- guard 4: THE LIVE-MODEL MANIFEST, built from cvt8's tree, per arm with NAMES and GROUP_HOLD ----

python3 - "$HARN" "$MTMP" "$NTENS" "$TOTPAR" "$NET" "$SC" "$G_K01" "$G_ISO" "$G_ONE50" "$G_ONE59" "$G_ONE53" "$G_SPLIT" \
          "$GH_3" "$GH_50" "$GH_59" "$GH_53" "$ARMS" "$SEEDS" <<'PYEOF' || guard_fail "guard 4: live partition / hold / literal check FAILED"

# guard 4c': the launcher's literals == the scorer's
