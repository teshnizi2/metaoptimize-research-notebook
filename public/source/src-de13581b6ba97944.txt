#!/bin/bash
# =============================================================================
# cA1_g4_ms_ladder.sh -- `g4m`, THE r x META-STEPSIZE LADDER.
#
# NO CODE CHANGE.  Nothing in the live tree is touched.  This batch varies two
# knobs that already exist: the additive operator's ETA_RATIO and train.py's
# --meta-stepsize.
#
# REGISTERED SCORER: analysis/cA1_g4_score.py, committed BEFORE this batch is
# submitted (STANDING RULE 21).  Every bar quoted here is frozen there, and that
# file's selftest() asserts each one against THIS FILE's own text.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# HF.py's additive operator (PATCH_HIER, `_apply_hier`, layerwise branch) is
#
#     d  = beta_t - beta_{t-1};   dm = mean(d);
#     beta_t := beta_{t-1} + dm + r * (d - dm)
#
# so r scales the DEVIATION component of the meta-update and leaves the shared
# component at full rate.  r = 1 is the arithmetic identity (plain layerwise).
#
# THE OBJECTION.  "r is not a hierarchy; it is a meta-stepsize dial in disguise."
# MEASURED from results/all_runs.csv, not quoted: all 637 hierarchical runs in
# the corpus sit at meta_stepsize = 1e-3.  ONE value.  The confound has never
# been swept.  GUARD 2 below re-derives that on the live CSV and aborts if it
# has stopped being true.
#
# -----------------------------------------------------------------------------
# THE ms GRID IS **NOT THE ONE THE BRIEF NAMED**, AND THIS IS WHY
# -----------------------------------------------------------------------------
# The brief proposed ms in {1e-5, 3e-5, 1e-4, 3e-4, 1e-3}.  results/all_runs.csv
# ALREADY carries the r = 1 (HIER=none) ms ladder at exactly this cell -- the
# `ms-lay-*` runs, five seeds each -- and GUARD 2b re-derives it:
#
#       ms      plateau5     state
#     1e-5       19.570      DEAD  (beta cannot climb out of alpha0 in 100 ep)
#     3e-5       22.766      DEAD
#     1e-4       46.362      crippled
#     3e-4       90.228      live
#     1e-3       90.893      live, and the LARGEST value on the ladder (n=17)
#     > 1e-3     NO DATA
#
# The ladder is MONOTONE INCREASING to the top of the briefed grid.  At
# alpha0 = 1e-6, log-beta has ~11.5 nats to climb, the SHARED component does the
# climbing, and a smaller GLOBAL ms slows the climb as well as the deviations.
# On the briefed grid every arm's ms-argmax would land on the upper EDGE,
# "optimise ms within each r" would be a no-op, and the batch would return
# REFUTED by construction while spending 36 of 60 jobs at 19-46% accuracy.
#
# A test that can only return one answer is not a test.  So: the two provably
# dead rungs are dropped, the three live briefed rungs are kept, and three rungs
# are added ABOVE the observed argmax so the max over ms is BRACKETED, not
# truncated.  Bracketing is then ENFORCED as gate G2, not assumed.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  72 jobs, ONE submission, so BATCH (the unit of replication,
# F(62,85)=5.47) is held constant and every contrast is WITHIN this batch.
#   r  in {0, 0.06, 0.2, 1}                        endpoints r=0 and r=1
#   ms in {1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2}
#   seed in {0, 1, 2}
#   ResNet18 / CIFAR-10 / layerwise / SGDm+Lion / alpha0 1e-6 / 100 ep /
#   batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / PROBE=5.
#
# -----------------------------------------------------------------------------
# PRE-REGISTERED, and it can come out either way.  plateau5 = mean of the LAST 5
# test epochs is PRIMARY; the CSV `plateau` (last-20) column is BANNED and
# `best_test` is not a plateau.
#
#   OPT(r) = max over the 6 ms rungs of the 3-seed mean plateau5
#   PEAK   = max(OPT(0.06), OPT(0.2)) - max(OPT(0), OPT(1))
#   SPAN   = max_r OPT(r) - min_r OPT(r)
#
#   REFUTED_BAR=0.50   PEAK >= 0.50 pp -> r is MORE than a step-size dial.
#   CONFIRMED_BAR=0.30 SPAN <= 0.30 pp -> r IS a step-size dial.
#   Anything strictly between is UNRESOLVED and is reported as UNRESOLVED.
#
# WHERE THE BARS COME FROM.  SIGMA_W=0.1438 pp is the pooled WITHIN-CELL SD of
# plateau5 re-derived from the 57 archived additive runs at this exact
# configuration (10 r-cells, df 47).  SE of a 3-seed mean = 0.0830 pp.
#   0.30 pp is the 95th percentile of the range of four 3-seed means under a
#     flat null at that sigma (3.63 * 0.0830 = 0.301), so CONFIRMED carries a
#     <= 5% false-positive rate at this n.
#   0.50 pp is 4.26 SE-of-difference AND exactly half the interior lift the
#     archived ms=1e-3 data already shows (93.285 at r=0.06 against 92.288 at
#     r=0 and 90.971 at r=1, i.e. +0.997 over the better endpoint).
#
# -----------------------------------------------------------------------------
# THE GATES, IN ORDER.  Full text in analysis/cA1_g4_score.py.
#   G0  PROVENANCE, absolutely gating.  Read from each run's OWN ARGS/ENV line.
#   G1  THE NO-OP CONTROL.  r=1 additive must reproduce plain layerwise
#       HIER=none at ms=1e-3 (archived grand mean 90.906 over 6 batches),
#       G1_BAND=0.45 pp.  A miss means the tree moved: VOID.
#   G2  THE BRACKET GATE.  An ms-argmax on a grid EDGE truncates that OPT(r).
#       At an ENDPOINT r it inflates PEAK and kills a REFUTED; at an INTERIOR r
#       it deflates PEAK and kills only a CONFIRMED.
#   G3  THE MANIFOLD GATE.  The ms=1e-3 slice of THIS batch must itself show
#       PEAK >= REFUTED_BAR, else a flat answer cannot be told from an absent
#       effect and the verdict is UNRESOLVED-NOMANIFOLD.
#   G4  THE BOX METER.  Rail occupancy from each run's own probe.jsonl.
#       DESCRIPTIVE ONLY -- it cannot change G5.
#   G5  THE PRIMARY.  PEAK and SPAN against the two registered bars.
#
# MAY NOT CLAIM: anything about ResNet50, CIFAR-100, alpha0=1e-3, a granularity
# other than layerwise, or an operator other than `additive`.  The r grid has
# four rungs and cannot locate an optimum.
#
# USAGE   bash bin/cA1_g4_ms_ladder.sh            # dry run, prints all 72 lines
#         bash bin/cA1_g4_ms_ladder.sh --submit
# =============================================================================
set -u
SUBMIT=0; [ "${1:-}" = "--submit" ] && SUBMIT=1
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

USER_NAME="${USER:-$(whoami)}"
WS=${METAOPT_WS:-/cluster/account_a/metaopt}
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/g4m
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}

# ---- venv activation preamble.  Guards that import torch fail without it;
# ---- five consecutive submission aborts were caused by omitting this.
module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

# shellcheck disable=SC1090
. "$(dirname "$0")/_lib_guards.sh"

PREFIX=g4m
NET=ResNet18; DSET=CIFAR10; GRAN=layerwise
ALPHA0=1e-6; EPOCHS=100; BATCH=100; GAMMA=1
CLIP="-15:-2.3026"; AUG=1; PROBE=5
M_LAYERWISE=62
RS="0 0.06 0.2 1"
MSS="1e-4 3e-4 1e-3 3e-3 1e-2 3e-2"
SEEDS="0 1 2"
NJOBS=72
SIGMA_W=0.1438; CONFIRMED_BAR=0.30; REFUTED_BAR=0.50; G1_BAND=0.45
SCORER="$REPO/analysis/cA1_g4_score.py"
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
# WALL measured, not guessed: the 69 archived additive/R18/layerwise/alpha0=1e-6
# 100-epoch runs read p50 31 min, p90 39, p99 78, max 78.  78 x 1.5 = 117 min.
# Over-provisioning gets you priority-starved -- a 10 h wall sat unschedulable
# for a week while a 7 h one ran.
WALL=02:00:00

echo "=============================================================="
echo " g4m -- THE r x META-STEPSIZE LADDER.  All 637 hierarchical runs"
echo " in the corpus sit at ONE meta-stepsize (1e-3), so the 'r is just a"
echo " step-size dial' objection has never been swept.  This sweeps it."
echo " r {$RS} x ms {$MSS} x seeds {$SEEDS} = $NJOBS jobs"
echo " $NET/$DSET/$GRAN  alpha0 $ALPHA0  box $CLIP  AUGMENT=$AUG  PROBE=$PROBE"
echo " bars: REFUTED PEAK>=$REFUTED_BAR   CONFIRMED SPAN<=$CONFIRMED_BAR   else UNRESOLVED"
echo "=============================================================="

CSVP="$REPO/results/all_runs.csv"

# ---- GUARD 0 -- the corpus is where we think it is -------------------------
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV at $CSVP"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- GUARD 1 -- STANDING RULE 21: the scorer exists and is committed --------
guard_scorer_registered "$SCORER" || guard_fail "guard 1: no registered scorer"
python3 "$SCORER" --selftest >/tmp/g4m_selftest.$$ 2>&1
if tail -1 /tmp/g4m_selftest.$$ | grep -q PASS; then
  echo "guard 1: scorer selftest $(tail -1 /tmp/g4m_selftest.$$)"
else
  echo "--- scorer selftest output ---"; cat /tmp/g4m_selftest.$$
  guard_fail "guard 1: the registered scorer's selftest does not pass"
fi
rm -f /tmp/g4m_selftest.$$
if git -C "$REPO" status --porcelain -- analysis/cA1_g4_score.py 2>/dev/null | grep -q .; then
  guard_fail "guard 1b: analysis/cA1_g4_score.py is not committed -- STANDING RULE 21"
else
  echo "guard 1b: the scorer is committed (git reports it clean)"
fi

# ---- GUARD 2 -- THE PREMISE, re-derived on the live CSV ---------------------
python3 - "$CSVP" <<'PYEOF' || guard_fail "guard 2: premise"
import csv, sys, collections
c = collections.Counter()
for r in csv.DictReader(open(sys.argv[1])):
    if (r["hier"] or "").strip():
        c[r["meta_stepsize"]] += 1
print("guard 2: %d hierarchical runs, by meta_stepsize: %s" % (sum(c.values()), dict(c)))
assert sum(c.values()) > 0, "no hierarchical runs in the corpus at all"
assert len(c) == 1, "the confound has already been swept; re-check whether g4m is still needed"
assert list(c) == ["1e-3"], "the single value is not 1e-3; the design must be re-derived"
print("guard 2: the meta-stepsize confound is REAL -- ONE value across all 637 runs")
PYEOF

# ---- GUARD 2b -- THE GRID CHANGE, justified by measurement ------------------
python3 - "$CSVP" <<'PYEOF' || guard_fail "guard 2b: dead-rung measurement"
import csv, re, sys, statistics, collections
tag = {"1e5": "1e-5", "3e5": "3e-5", "1e4": "1e-4", "3e4": "3e-4"}
by = collections.defaultdict(list)
plain = []
for r in csv.DictReader(open(sys.argv[1])):
    m = re.match(r"^ms-lay-(\w+)-s\d+$", r["run"])
    if m and m.group(1) in tag and r["plateau5"]:
        by[tag[m.group(1)]].append(float(r["plateau5"]))
    if ((r["hier"] or "") == "" and r["network"] == "ResNet18"
            and r["dataset"] == "CIFAR10" and r["granularity"] == "layerwise"
            and r["alpha0"] == "1e-6" and r["meta_stepsize"] == "1e-3"
            and r["base"] == "SGDm" and r["meta"] == "Lion"
            and r["beta_clip"] == "-15:-2.3026" and r["augment"] == "1"
            and r["epochs_done"] == "100" and r["collapsed"] == "0" and r["plateau5"]):
        plain.append(float(r["plateau5"]))
for k in ("1e-5", "3e-5", "1e-4", "3e-4"):
    print("guard 2b: archived r=1 ms=%-5s plateau5 %7.3f  (n=%d)"
          % (k, statistics.mean(by[k]), len(by[k])))
print("guard 2b: archived r=1 ms=1e-3 plateau5 %7.3f  (n=%d)"
      % (statistics.mean(plain), len(plain)))
assert statistics.mean(by["1e-5"]) < 30 and statistics.mean(by["3e-5"]) < 30, \
    "the two dropped rungs are NOT dead; the grid change is unjustified"
assert (statistics.mean(by["1e-5"]) < statistics.mean(by["3e-5"])
        < statistics.mean(by["1e-4"]) < statistics.mean(by["3e-4"])
        < statistics.mean(plain)), \
    "the archived ms ladder is not monotone to 1e-3; re-derive the grid"
print("guard 2b: the ladder is MONOTONE to the top of the briefed grid, so the")
print("guard 2b: grid MUST extend above 1e-3 or the max over ms is truncated.")
PYEOF

# ---- GUARD 3 -- m MEASURED on the LIVE tree, not declared -------------------
python3 - "$CIF" "$NET" "$ALPHA0" "$GRAN" "$M_LAYERWISE" <<'PYEOF' || guard_fail "guard 3: beta allocation"
import sys, os
cif, net, a0, gran, want = sys.argv[1:6]
sys.path.insert(0, cif); sys.path.insert(0, os.path.join(cif, "Optimizers"))
try:
    import torch
    from build_network import build_network
    from HF import HF
except Exception as ex:
    sys.exit("GUARD FAIL: cannot import the LIVE tree (%s) -- run on the cluster "
             "inside the venv." % ex)
class NW:
    def add_scalar(self, *a, **k): pass
B = {"alg": "SGD", "weight_decay": 0.1}
M = {"alg": "Lion", "meta_stepsize": 1e-3, "momentum_param": 0.99,
     "Lion_beta2": 0.9, "weight_decay": 0}
torch.manual_seed(0)
opt = HF(build_network(net, "cpu"), stepsize_groups=gran, alpha0=float(a0),
         args_base=dict(B), args_meta=dict(M), gamma=1, writer=NW())
m = sum(int(b.numel()) for b in opt.beta)
assert m == int(want), "m(%s) is %d, registered %s" % (gran, m, want)
print("guard 3: %s on %s ALLOCATED beta m=%d (registered %s)" % (gran, net, m, want))
PYEOF

# ---- GUARD 3b -- the additive operator exists and r=1 is the identity -------
grep -q "PATCH_HIER" "$CIF/Optimizers/HF.py" 2>/dev/null \
  && grep -q "_hier_ratio" "$CIF/Optimizers/HF.py" 2>/dev/null \
  && echo "guard 3b: PATCH_HIER and _hier_ratio present in the live HF.py" \
  || guard_fail "guard 3b: the additive operator is missing -- this batch cannot run here"
python3 - "$CIF" <<'PYEOF' || guard_fail "guard 3b: r=1 identity"
import sys, re
src = open(sys.argv[1] + "/Optimizers/HF.py").read()
i = src.index("def _apply_hier")
body = src[i:i + 1200]
need = "self._beta_prev + dm + self._hier_ratio * (d - dm)"
assert need in body, "the layerwise additive branch is not the registered arithmetic"
print("guard 3b: layerwise additive branch reads  beta_prev + dm + r*(d - dm)")
print("guard 3b: therefore r=1 is the ARITHMETIC IDENTITY -- G1 is a real control")
PYEOF

# ---- GUARD 4 -- no name collision ------------------------------------------
if grep -q "^$PREFIX-" "$CSVP"; then guard_fail "guard 4: $PREFIX-* already in the CSV"
else echo "guard 4: no $PREFIX-* name collision in the corpus"; fi
if ls "$WS/runs"/${PREFIX}-*.out >/dev/null 2>&1; then
  guard_fail "guard 4b: $WS/runs already holds $PREFIX-*.out"
else echo "guard 4b: no $PREFIX-*.out under $WS/runs"; fi

# ---- GUARD 5 -- room and queue ---------------------------------------------
FREE=$(df -BG "$WS" 2>/dev/null | tail -1 | awk '{print $4}' | tr -d 'G')
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: ${FREE:-?} GB free; pending=$PEND, adding $NJOBS"
[ "${PEND:-0}" -le 40 ] || guard_fail "guard 5: $PEND pending, over cap"

# ---- compose, check EVERY line, then (maybe) submit ------------------------
DESIGN=(alg-base=SGDm alg-meta=Lion dataset=$DSET NN-name=$NET \
        stepsize-groups=$GRAN alpha0=$ALPHA0 num-epochs=$EPOCHS \
        batch-size=$BATCH gamma=$GAMMA)

mkdir -p "$SAVE"
N=0; REJ=0; NCHK=0
for S in $SEEDS; do
 for R in $RS; do
  RTAG=$(echo "$R" | tr -d '.')
  for MST in $MSS; do
   MTAG=$(echo "$MST" | tr -d '.-')
   RN="$PREFIX-r${RTAG}-m${MTAG}-s${S}"
   CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=14G --time=$WALL \
--export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,HIER=additive,ETA_RATIO=$R,SCHED=none,PROBE=$PROBE,PROBE_DIR=$SAVE/probe_${RTAG}_${MTAG}_${PREFIX}_s${S} \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 \
--Lion-beta2-meta 0.9 --weight-decay-meta 0 --dataset $DSET --NN-name $NET \
--batch-size $BATCH --max-time 999:00:00 --gamma $GAMMA --meta-stepsize $MST \
--alpha0 $ALPHA0 --num-epochs $EPOCHS --stepsize-groups $GRAN --seed $S \
--save-directory $SAVE --run-name $RN"

   # STANDING RULE 20, BEFORE sbatch: no repeated flag anywhere on the line,
   # and the effective value of every declared design flag is what we declared.
   guard_presubmit "$CMD" "${DESIGN[@]}" meta-stepsize="$MST" seed="$S" \
      >/tmp/g4m_pre.$$ 2>&1 || { echo "--- $RN ---"; cat /tmp/g4m_pre.$$; \
      guard_fail "guard 6: $RN failed the pre-submission ARGS check"; }
   NCHK=$((NCHK+1))
   rm -f /tmp/g4m_pre.$$

   if [ "$SUBMIT" = 1 ] && [ "$FAIL" = 0 ]; then
     eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
   else
     echo "$CMD"
     N=$((N+1))
   fi
  done
 done
done

echo "guard 6: $NCHK/$NJOBS composed command lines passed the RULE 20 pre-check"
[ "$NCHK" = "$NJOBS" ] || guard_fail "guard 6: only $NCHK of $NJOBS lines checked"

if [ "$FAIL" = 1 ]; then
  echo "!!! SUBMIT ABORTED -- guards failed."
  [ "$SUBMIT" = 1 ] && echo "!!! If any job was already accepted, scancel --name by prefix $PREFIX-."
  exit 2
fi

echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'dry run, nothing submitted' )); $REJ rejected ----"

if [ "$SUBMIT" = 1 ]; then
  grep -q "^$PREFIX-" "$REPO/bin/PROTECTED.txt" 2>/dev/null || \
    echo "$PREFIX-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- $PREFIX- added to bin/PROTECTED.txt; no trim may cancel it ----"
  # STANDING RULE 20, AFTER launch: read the FIRST RUN'S OWN ARGS line and abort
  # the batch on any mismatch.  This is the check that would have caught ml2/sm3.
  # NOTE the directory.  jobs/run_cifar.sh carries
  #   #SBATCH --output=/cluster/account_a/metaopt/runs/%x-%j.out
  # so the .out files land in $WS/runs, NOT in --save-directory ($SAVE, which
  # holds only the probe dirs).  Watching $SAVE here returns a FALSE
  # "UNVERIFIED" -- corrected after the g4m launch, where the hand audit at the
  # right path returned PASS on all 21 started runs.  The composed sbatch lines
  # are untouched by this fix.
  guard_postlaunch "$WS/runs" "$PREFIX-" "${DESIGN[@]}" || {
    echo "!!! THE LAUNCHED BATCH IS VOID AS DESIGNED.  Cancel it now."
    exit 2; }
  echo "---- score with:  python3 analysis/cA1_g4_score.py --runs $WS/runs --probes $SAVE"
fi
