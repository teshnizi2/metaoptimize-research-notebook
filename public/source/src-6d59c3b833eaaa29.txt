#!/bin/bash
# =============================================================================
# cG1_twolevel_ladder.sh -- `tl1`, THE TWO-LEVEL SHRINKAGE LADDER.
#
# GAP G1 (docs/MAIN-IDEA-GAPS): the hierarchical proposal AS LITERALLY WRITTEN
# has never been measured.  This batch measures it.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.
# Nothing here touches /cluster/account_a and nothing is submitted to cluster_a.
#
# REGISTERED SCORER: analysis/cG1_tl_score.py, committed BEFORE this batch is
# submitted (STANDING RULE 21).  Every bar quoted here is frozen there, and that
# file's selftest() asserts each one against THIS FILE'S own text.
#
# CODE CHANGE: patches/patch_twolevel.py, applied to the live tree.  It is a
# GUARDED patch -- every line it adds is reached only when HIER == 'twolevel'.
# tests/tl_equivalence.py proves the disabled path is BITWISE unchanged across
# HIER-unset and all four pre-existing modes; GUARD 3 below re-proves the
# operator's defining identity against the LIVE tree at submission time.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# The proposal: learn a scalar and layerwise step sizes JOINTLY and shrink the
# layerwise ones toward the scalar with a tunable strength.  What the tree had
# was a per-step PROJECTION,   beta <- beta - lam*(beta - mean(beta)),   whose
# "strength" lam is a TIME CONSTANT: the surviving deviation after T steps is
# (1-lam)^T, a half-life of ln2/-ln(1-lam) steps.  A 100-epoch CIFAR-10 run at
# batch_size 100 is 500*100 = 50,000 steps, so
#
#     lam = 1e-3  ->  693 steps  ->     72 half-lives per run
#     lam = 1e-1  ->   6.6 steps  ->  7,600 half-lives per run
#
# and EVERY rung of the only archived layerwise ladder (batch `h2`, 22 runs) is
# at lam >= 1e-3.  All seven rungs measured FULL POOLING.  The three genuinely
# partial rungs that exist (1e-5, 3e-5, 1e-4) are batch `lp`, which is 6 runs
# and NOTHING ELSE -- no plain arm, no scalar arm, no control of any kind.
# GUARD 2 re-derives all of that from results/all_runs.csv and aborts if it has
# stopped being true.
#
# -----------------------------------------------------------------------------
# THE OPERATOR.  State it exactly; a referee must be able to check it.
# -----------------------------------------------------------------------------
#     beta_g(t) = s(t) + d_g(t)                                            (1)
#     s     <-  META( s,   sum_g z_g,  eta       )                         (2)
#     d_g   <-  META( d_g, z_g,        rho * eta )                         (3)
#
# z_g = HtT_gradft[g] = dJ/d beta_g is the meta-gradient HF already computes and
# already feeds to meta_update; sum_g z_g is its chain rule through (1).  META
# is the run's own --alg-meta rule (here Lion), applied verbatim, with ONE
# INDEPENDENT trace/momentum state per level.  rho is the env var ETA_RATIO --
# a name that already means exactly this, so the dial is already on every run's
# own ENV line and already in results/all_runs.csv with no change to
# jobs/run_cifar.sh or analysis/aggregate.py.
#
#     SHRINKAGE STRENGTH = 1 - rho.
#     rho = 0  ->  d_g is driven by a step size of exactly 0, so d_g == 0 for
#                  all t: the arm IS the scalar arm, with the shared level still
#                  learned at the full rate eta.  A checkable identity, checked
#                  offline (tests/tl_equivalence.py), at submission (GUARD 3b)
#                  and in-batch against a scalar control (scorer gate G1).
#     rho = 1  ->  the deviations learn as fast as the shared level.
#
# WHY rho IS A STRENGTH AND lam IS NOT.  `shrink` multiplies an existing
# quantity down; nothing opposes the decay, so its only fixed point is d = 0 and
# lam merely selects how fast the run gets there.  In (1)-(3), d_g is a STATE
# with its own driven dynamics and no decay term (the meta weight decay is 0 in
# every run in this corpus), so its scale is set by rho and is stationary in T.
# MEASURED, not argued, on this cluster before this batch was written
# (tests/tl_equivalence.py, 3,000 steps, ResNet-free MLP, box -15:-2.3026):
#
#     std(beta) at t=500 ... t=3000        ratio to plain      ratio t3000/t500
#     twolevel rho=0.03   0.00141 flat          0.0290              1.0000
#     twolevel rho=0.1    0.00477 flat          0.0981              1.0000
#     twolevel rho=0.3    0.01430 flat          0.2942              1.0000
#     twolevel rho=1      0.04861 flat          0.9998              0.9998
#     shrink   lam=1e-4   0.04645 -> 0.03605                        0.7761
#                                        (predicted (1-lam)^2500 =  0.7788)
#     shrink   lam=1e-3   0.03028 -> 0.00248                        0.0819
#                                        (predicted (1-lam)^2500 =  0.0820)
#     shrink   lam>=1e-2  already at machine zero by t=500
#
# std(beta|rho) = rho * std(beta|plain) to within 3%, and it does not move in t.
# That is a dimensionless strength dial.  The lam ladder decays at exactly the
# geometric rate that makes it a half-life.  The scorer re-tests the same thing
# in-batch at t ~ 50,000 as gate G2.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  42 jobs, ONE submission, so BATCH -- the unit of replication in
# this corpus (F(62,85)=5.47, p 6.9e-13; SEED is null, F 1.21, p 0.213) -- is
# held constant and EVERY contrast is WITHIN this batch.
#
#   twolevel arms  rho in {0, 0.03, 0.1, 0.3, 1}   HIER=twolevel, layerwise
#   plain control  HIER unset, layerwise      the free-layerwise reference
#   scal control   HIER unset, scalar         the arm rho=0 must reproduce
#   x alpha0 in {1e-6, 1e-3}  x  seed in {0, 1, 2}                 = 42 jobs
#   ResNet18 / CIFAR-10 / SGDm+Lion / meta-stepsize 1e-3 / 100 ep / batch 100 /
#   gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / PROBE=500.
#
# WHY BOTH alpha0, AND WHY THEY ARE NOT POOLED.  The archived additive interior
# lift exists at alpha0=1e-6 (+0.997 pp over the better endpoint) and does not
# at alpha0=1e-3.  Pooling them would hide exactly the thing this batch is for.
# They are CO-PRIMARY and each gets its own verdict line.
#
# WHY THE rho GRID IS {0, 0.03, 0.1, 0.3, 1}.  rho=0 and rho=1 are the two
# endpoints of the parameterisation and rho=0 is an exact identity, so the grid
# is anchored at both ends by construction rather than by measurement.  The
# three interior rungs are log-spaced across the decade and a half between them.
# Five rungs CANNOT locate an optimum and this batch does not claim to; they can
# only say whether an interior arm beats the endpoints by the registered margin.
#
# -----------------------------------------------------------------------------
# PRE-REGISTERED, AND IT CAN COME OUT EITHER WAY.  plateau5 = mean of the LAST 5
# test epochs is PRIMARY; the CSV `plateau` (last-20) column is BANNED and
# `best_test` is not a plateau.
#
#   M(arm)    = mean plateau5 over that arm's seeds, within alpha0
#   ENDPOINTS = {rho=0, rho=1, plain}      INTERIOR = {rho=0.03, 0.1, 0.3}
#   PEAK      = max_INTERIOR M  -  max_ENDPOINTS M
#
#   SUPPORT_BAR=0.50   PEAK >= 0.50 pp -> partial shrinkage toward the scalar
#                      genuinely helps.
#   NULL_BAR=0.25      PEAK <= 0.25 pp -> no interior optimum at the archived
#                      effect size.
#   Anything strictly between is UNRESOLVED and is reported as UNRESOLVED.
#
# WHERE THE BARS COME FROM.  SIGMA_W=0.1735 pp is the pooled WITHIN-CELL SD of
# plateau5 re-derived from the archived runs at this exact configuration -- the
# 44 additive-layerwise (batch x alpha0 x r) cells and the 25 plain
# layerwise/scalar (batch x alpha0 x granularity) cells, df 153.  SE of a
# 3-seed mean = 0.1002 pp; SE of a difference = 0.1417 pp.
#   Under the flat null "all six arms share one true mean", a 10^6-draw
#   simulation puts PEAK's 99th percentile at 0.248 pp and P(PEAK >= 0.50) at
#   0.00000.  So NULL_BAR=0.25 IS that 99th percentile (a flat ladder returns
#   NULL 99% of the time) and SUPPORT_BAR=0.50 is a <1e-5 false-positive
#   threshold that is also exactly HALF the archived interior lift of +0.997 pp.
#   Power against a true +0.997 pp lift is 0.999; against +0.50 pp it is 0.25,
#   so a half-strength effect will mostly read UNRESOLVED.  That limit is
#   registered here so it cannot be discovered afterwards and spun.
#
# -----------------------------------------------------------------------------
# THE GATES, IN ORDER.  Full text in analysis/cG1_tl_score.py.
#   G0  PROVENANCE, absolutely gating.  Read from each run's OWN ARGS/ENV line;
#       the arm is identified by (stepsize-groups, HIER, ETA_RATIO), never by
#       the run's name.
#   G1  THE IDENTITY GATE.  |M(rho=0) - M(scal)| <= 0.45 pp at each alpha0.
#       A miss means the operator does not reduce to the scalar arm: VOID.
#   G2  THE MECHANISM GATE.  std(beta) at the last probe record (~50,000 steps):
#       0 exactly at rho=0, strictly increasing over rho, > 0 at rho=0.03.
#       A violation is VOID; absent probes are UNVERIFIED, carried as a suffix.
#   G3  THE TREE GATE.  The in-batch plain and scalar controls within 0.60 pp of
#       their archived grand means.  A miss means the tree moved: VOID.
#   G4  THE RAIL METER.  Box occupancy from each run's own probe.jsonl.
#       DESCRIPTIVE ONLY -- it cannot change G5.
#   G5  THE PRIMARY.  PEAK against the two bars, per alpha0.
#
# MAY NOT CLAIM: anything about ResNet-50/34/10, CIFAR-100, Tiny-ImageNet, a
# granularity other than layerwise, a meta-stepsize other than 1e-3, an
# optimizer pair other than SGDm+Lion, a box other than -15:-2.3026, an epoch
# budget other than 100, or the shrink/additive/zpool/zmpool operators.
#
# USAGE   bash bin/cG1_twolevel_ladder.sh            # dry run, prints all 42 lines

import sys, os
cif = sys.argv[1]
sys.path.insert(0, cif); sys.path.insert(0, os.path.join(cif, "Optimizers"))
import torch
src = open(os.path.join(cif, "Optimizers", "HF.py")).read()

assert src.count("self._hier == 'twolevel'") == 1, "twolevel is not dispatched once"

from HF import HF
class NW:
    def add_scalar(self, *a, **k): pass
B = {"alg": "SGDm", "weight_decay": 0.1, "momentum_param": 0.99}
M = {"alg": "Lion", "meta_stepsize": 1e-3, "momentum_param": 0.99,
     "Lion_beta2": 0.9, "weight_decay": 0}
def mknet(seed=0):
    torch.manual_seed(seed)
    return torch.nn.Sequential(torch.nn.Linear(32, 24), torch.nn.Tanh(),
                               torch.nn.Linear(24, 8))
g = torch.Generator().manual_seed(1234)
x = torch.randn(64, 32, generator=g); y = torch.randint(0, 8, (64,), generator=g)
lf = torch.nn.CrossEntropyLoss()
def go(env, gran, steps=200):
    keep = {k: os.environ.get(k) for k in ("HIER", "ETA_RATIO", "BETA_CLIP")}
    try:
        for k in keep: os.environ.pop(k, None)
        os.environ.update(env)
        net = mknet()
        o = HF(net, stepsize_groups=gran, alpha0=1e-6, args_base=dict(B),
               args_meta=dict(M), gamma=1, writer=NW())
        for _ in range(steps):
            o.step(net, lf(net(x), y))
        return [b.detach().clone() for b in o.beta], \
               [p.detach().clone() for p in net.parameters()]
    finally:
        for k, v in keep.items():
            os.environ.pop(k, None)
            if v is not None: os.environ[k] = v
CLIP = "-15:-2.3026"
bs, ws = go({"BETA_CLIP": CLIP}, "scalar")
bt, wt = go({"HIER": "twolevel", "ETA_RATIO": "0", "BETA_CLIP": CLIP}, "layerwise")
flat = bt[0].reshape(-1)
spread = float((flat - flat[0]).abs().max())
gap = float((flat - bs[0].reshape(-1)[0]).abs().max())
wgap = max(float((a.double() - b.double()).abs().max()) for a, b in zip(ws, wt))
print("guard 3b: rho=0, 200 steps:  max_g|beta_g - beta_1| = %.3e" % spread)
print("guard 3b:                    max_g|beta_g - beta_scalar| = %.3e" % gap)
print("guard 3b:                    max|w - w_scalar| = %.3e" % wgap)
assert spread == 0.0, "rho=0 left non-zero deviations"
assert gap < 1e-4 and wgap < 1e-4, "rho=0 does not track the scalar arm"
# and rho must scale the deviations rather than decay them
sp = {}
for r in ("0.1", "0.3", "1"):
    b, _ = go({"HIER": "twolevel", "ETA_RATIO": r, "BETA_CLIP": CLIP}, "layerwise")
    sp[r] = float(b[0].reshape(-1).std())

print("guard 3b: std(beta) rho=0.1 %.6f  rho=0.3 %.6f  rho=1 %.6f  plain %.6f"

print("guard 3b: dimensionless STRENGTH, not a time constant.  IDENTITY PROVED.")

 for S in $SEEDS; do
  # the seven arms: five twolevel rungs, then the two controls
  for ARM in $RHOS plain scal; do
   case "$ARM" in
     plain) GRAN=layerwise; HENV="" ; TAG=plain ;;
     scal)  GRAN=scalar;    HENV="" ; TAG=scal  ;;
     *)     GRAN=layerwise; HENV=",HIER=twolevel,ETA_RATIO=$ARM"
            TAG="r$(echo "$ARM" | tr -d '.')" ;;
   esac
   RN="$PREFIX-$ATAG-$TAG-s${S}"
   PDIR="$SAVE/probe_${PREFIX}_${ATAG}_${TAG}_s${S}"
   CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=14G --time=$WALL \
--export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,SCHED=none,PROBE=$PROBE,PROBE_DIR=$PDIR$HENV \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 \
--Lion-beta2-meta 0.9 --weight-decay-meta 0 --dataset $DSET --NN-name $NET \
--batch-size $BATCH --max-time 999:00:00 --gamma $GAMMA --meta-stepsize $MS \
--alpha0 $A0 --num-epochs $EPOCHS --stepsize-groups $GRAN --seed $S \
--save-directory $SAVE --run-name $RN"

   # STANDING RULE 20, BEFORE sbatch: no repeated flag anywhere on the line, and
   # the effective value of every declared design flag is what we declared.
   guard_presubmit "$CMD" "${DESIGN[@]}" alpha0="$A0" seed="$S" \
      stepsize-groups="$GRAN" run-name="$RN" \
      >/tmp/tl1_pre.$$ 2>&1 || { echo "--- $RN ---"; cat /tmp/tl1_pre.$$; \
      guard_fail "guard 6: $RN failed the pre-submission ARGS check"; }
   NCHK=$((NCHK+1))
   rm -f /tmp/tl1_pre.$$

   if [ "$SUBMIT" = 1 ] && [ "$FAIL" = 0 ]; then
     eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
   else
     echo "$CMD"
     N=$((N+1))
   fi
  done
 done
done

echo "guard 6: $NCHK/$NJOBS composed command lines passed the RULE 20 pre-check"
[ "$NCHK" = "$NJOBS" ] || guard_fail "guard 6: only $NCHK of $NJOBS lines checked"

if [ "$FAIL" = 1 ]; then
  echo "!!! SUBMIT ABORTED -- guards failed."
  [ "$SUBMIT" = 1 ] && echo "!!! If any job was already accepted, scancel --name by prefix $PREFIX-."
  exit 2
fi

echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo 'ACCEPTED BY SLURM' || echo 'dry run, nothing submitted' )); $REJ rejected ----"

if [ "$SUBMIT" = 1 ]; then
  grep -q "^$PREFIX-" "$REPO/bin/PROTECTED.txt" 2>/dev/null || \
    echo "$PREFIX-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- $PREFIX- added to bin/PROTECTED.txt; no trim may cancel it ----"
  # STANDING RULE 20, AFTER launch: read the FIRST RUN'S OWN ARGS line and abort
  # the batch on any mismatch.  NOTE the directory: jobs/run_cifar.sh carries
  #   #SBATCH --output=/cluster/account_b/metaopt/runs/%x-%j.out
  # so the .out files land in $WS/runs, NOT in --save-directory ($SAVE, which
  # holds only the probe dirs).  Watching $SAVE returns a FALSE "UNVERIFIED".
  guard_postlaunch "$WS/runs" "$PREFIX-" "${DESIGN[@]}" || {
    echo "!!! THE LAUNCHED BATCH IS VOID AS DESIGNED.  Cancel it now."
    exit 2; }
  echo "---- score with:  python3 analysis/cG1_tl_score.py --runs $WS/runs --probes $SAVE"
fi
