def avg(v):
    return sum(v) / float(len(v))


def sdev(v):
    if len(v) < 2:
        return 0.0
    m = avg(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def median(v):
    w = sorted(v)
    n = len(w)
    return w[n // 2] if n % 2 else 0.5 * (w[n // 2 - 1] + w[n // 2])

def is_num(t):
    if t == "nan":
        return True
    if not t or any(c not in "0123456789." for c in t):
        return False
    return t.count(".") <= 1 and t != "."


def parse_epoch(line):
    """`Epoch N, Train Accuracy: X %, Test Accuracy: Y %` anywhere in the line -> (N, X, Y) or None."""
    i = line.find("Epoch")
    while i >= 0:
        rest = line[i + len("Epoch"):]
        parts = rest.split(",")
        if len(parts) >= 3:
            n = parts[0].strip()
            tr = parts[1].strip()
            te = parts[2].strip()
            if n.isdigit() and parts[0][:1] in (" ", "\t") and tr.startswith("Train Accuracy:") and \
                    te.startswith("Test Accuracy:"):
                x = tr[len("Train Accuracy:"):].strip()
                if x.endswith("%"):
                    x = x[:-1].strip()
                y = te[len("Test Accuracy:"):].strip().split()
                y = y[0] if y else ""
                if y.endswith("%"):
                    y = y[:-1]
                if is_num(x) and is_num(y):
                    return int(n), float(x), float(y)
        i = line.find("Epoch", i + 1)
    return None


def kind_on(line):
    """`<UPPER_NAME>: on` at the start of a line (a patch witness)."""
    c = line.find(":")
    if c < 2:
        return False
    head = line[:c]
    if not head[0].isupper() or any(ch not in "ABCDEFGHIJKLMNOPQRSTUVWXYZ_" for ch in head):
        return False
    tail = line[c + 1:]
    if not tail.startswith(" on"):
        return False
    nxt = tail[3:4]
    return nxt == "" or not (nxt.isalnum() or nxt == "_")


def read_out(path):
    r = dict(args=None, env=None, gpu=None, node=None, done=False, ep={}, kinds=0, pt=False)
    for ln in open(path, errors="replace").read().splitlines():
        if ln.startswith("ARGS:") and r["args"] is None:
            r["args"] = ln
        elif ln.startswith("ENV:") and r["env"] is None:
            r["env"] = ln
        elif ln.startswith("NODE=") and r["node"] is None:
            r["node"] = ln.split("|")[0].strip()
        elif ln.startswith("NVIDIA") and r["gpu"] is None:
            r["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            r["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            r["pt"] = True
        else:
            e = parse_epoch(ln)
            if e:
                r["ep"][e[0]] = (e[1], e[2])
            if kind_on(ln):
                r["kinds"] += 1
    r["complete"] = r["done"] and all(k in r["ep"] for k in range(N_EPOCHS))
    return r


def split_args(line):
    """`ARGS: --f v --f v ...` -> list of pairs, or None.  No value in this batch contains a quote or a space."""
    toks = line[len("ARGS:"):].split()
    if len(toks) % 2:
        return None
    pairs = []
    for i in range(0, len(toks), 2):
        if not toks[i].startswith("--") or toks[i + 1].startswith("--"):
            return None
        pairs.append((toks[i][2:], toks[i + 1]))
    return pairs


def read_probe(path, wd):
    """-> dict of the structural facts plus the per-record realised shrink series."""
    steps, nb, mins, maxs, lo, hi, shrink, bad = [], set(), [], [], [], [], [], 0
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            d = json.loads(ln)
        except ValueError:
            bad += 1
            continue
        if not isinstance(d, dict):
            bad += 1
            continue
        steps.append(d.get("step"))
        nb.add(d.get("n_beta"))
        mins.append(d.get("beta_true_min"))
        maxs.append(d.get("beta_true_max"))
        lo.append((d.get("step"), d.get("n_at_lo"), d.get("n_beta")))
        hi.append(d.get("n_at_hi"))
        b = d.get("beta")
        if isinstance(b, list) and b:
            e = sorted(b)
            n = len(e)
            med = math.exp(e[n // 2]) if n % 2 else 0.5 * (math.exp(e[n // 2 - 1]) + math.exp(e[n // 2]))
            bx = d.get("beta_true_max")
            shrink.append((d.get("step"), med * wd, len(b), math.exp(bx) * wd))
        else:
            shrink.append((d.get("step"), None, 0, None))
    return dict(steps=steps, nb=nb, mins=mins, maxs=maxs, lo=lo, hi=hi, shrink=shrink, bad=bad)


def csv_split(line):
    out, cur, q, i = [], [], False, 0
    while i < len(line):
        c = line[i]
        if q:
            if c == '"':
                if i + 1 < len(line) and line[i + 1] == '"':
                    cur.append('"')
                    i += 1
                else:
                    q = False
            else:
                cur.append(c)
        elif c == '"':
            q = True
        elif c == ",":
            out.append("".join(cur))
            cur = []
        else:
            cur.append(c)
        i += 1
    out.append("".join(cur))
    return out


def corpus_count(csvp, tsvp):
    if not (csvp and tsvp and os.path.exists(csvp) and os.path.exists(tsvp)):
        return None
    ex = set()
    head = None
    for ln in open(tsvp):
        ln = ln.rstrip("\n")
        if not ln.strip() or ln.startswith("#"):
            continue
        f = ln.split("\t")
        if head is None:
            head = f
            continue
        d = dict(zip(head, f))
        ex.add((d["run"], d["job_id"]))
    L = open(csvp).read().splitlines()
    h = csv_split(L[0])
    ir, ij = h.index("run"), h.index("job_id")
    n = 0
    for ln in L[1:]:
        if not ln:
            continue
        f = csv_split(ln)
        if f[ir].startswith(BATCH + "-"):
            continue
        if (f[ir], f[ij]) in ex:
            continue
        n += 1
    return n

def state_of(d, se, div, healthy, bfree):
    if div:
        return "DIVERGED"
    if not healthy:
        return "UNHEALTHY"
    if not bfree:
        return "BOXBOUND"
    if d <= -BAR_NULL and d < -RES * se:
        return "REVERSES"
    if d <= BAR_NULL and d + RES * se < POOL:
        return "VANISHES"
    if d >= BAR_SURV and d > RES * se:
        return "SURVIVES"
    return "UNDECIDED"


def primary_word(s1, s4):
    if s1 in ("DIVERGED", "UNHEALTHY", "BOXBOUND"):
        return "ANCHOR-UNREADABLE"
    if s1 in ("VANISHES", "REVERSES"):
        return "ANCHOR-NOT-REPRODUCED"
    if s4 in ("DIVERGED", "UNHEALTHY", "BOXBOUND"):
        return "UNRESOLVED-" + s4 + "-W4"
    return "AUDIT-" + s4


def scalar_word(k, kdiv, cmp_ok, tch, tnd, se):
    if kdiv:
        return "SCALAR-DIVERGED"
    if k < HEALTH:
        return "SCALAR-COLLAPSED"
    if not cmp_ok:
        return "SCALAR-UNREADABLE"
    tmin = min(tch, tnd)
    if tch >= BAR_SURV and tnd >= BAR_SURV and tch > RES * se and tnd > RES * se:
        return "SCALAR-BEATS-BEST"
    if tmin - RES * se >= -BAR_SURV:
        return "SCALAR-TIES-BEST"
    if tmin <= -BAR_SURV and tmin < -RES * se:
        return "SCALAR-BELOW-BEST"
    return "SCALAR-UNRESOLVED"

        j, p, r = comp[0]
        why = []
        pr = split_args(r["args"] or "")
        if pr is None or pr != want_pairs(a, s):
            why.append("ARGS")
        pdir = SAVE_DIR + "/probe_" + rname(a, s)
        if r["env"] != ENV_T.replace("@@", pdir):
            why.append("ENV")
        if r["kinds"] or r["pt"]:
            why.append("KIND")
        if why:
            harness.append("%s %s" % (rname(a, s), "+".join(why)))
        info[(a, s)] = (j, r)
    check(not incomplete, "40 / 40 runs have exactly one complete .out (100 epoch lines + RUN_DONE)",
          ",".join(incomplete[:4]))
    check(not harness, "every complete run's ARGS pairs == the rebuilt 20-flag line, ENV == the registered line, "
          "no `<KIND>: on` / PROBE_TENSOR line", ";".join(harness[:4]))
    cov = {}
    for (a, s), (j, r) in sorted(info.items()):
        d = dict(split_args(r["args"]) or [])
        k = d.get("weight-decay-base")
        cov.setdefault(k, []).append(a)
    for r_ in RID:
        arms = cov.get(TOK[r_], [])
        say("      rung %s  --weight-decay-base %-5s %2d runs  (%s)" % (
            r_, TOK[r_], len(arms), " + ".join("%d %s" % (sum(1 for x in arms if GRN[x] == g), SPEC[g]) for g in GID)))
    check(sorted((k, len(v)) for k, v in cov.items()) == [("0.1", 12), ("1e-2", 12), ("5e-4", 16)],
          "the axis coverage is exactly the registered 0.1 x12 / 1e-2 x12 / 5e-4 x16")
    check(all(RUNG[a] == [r_ for r_ in RID if TOK[r_] == k][0] for k, v in cov.items() for a in v),
          "every run carries ITS OWN rung's token")
    gpus = {}
    for (a, s), (j, r) in info.items():
        gpus[r["gpu"]] = gpus.get(r["gpu"], 0) + 1

    # ---- [3] probe records -------------------------------------------------------------------------------------
    say("\n[3] PROBE STRUCTURE AND BOX OCCUPANCY, FROM THE RAW RECORDS")
    occ = {}
    shr = {}
    pbad = []
    win0 = 95 * STEPS_PER_EP
    for (a, s) in RUNS:
        pp = os.path.join(sub, "probe_" + rname(a, s), "probe.jsonl")
        if not os.path.exists(pp):
            pbad.append(rname(a, s) + " none")
            continue
        P = read_probe(pp, WDV[RUNG[a]])
        T = len(P["steps"])
        if P["bad"] or T != N_REC or P["steps"] != list(range(0, N_REC * PROBE_EVERY, PROBE_EVERY)) \
                or P["nb"] != set([MM[GRN[a]]]) or not (max(P["maxs"]) - min(P["mins"]) > 1e-9):
            pbad.append(rname(a, s))
            continue
        if any(n != BETA_LEN[GRN[a]] for _st, _v, n, _x in P["shrink"]):
            pbad.append(rname(a, s) + " beta-length")
            continue
        rl = sum(1 for x in P["mins"] if x <= BOX_LO + BOXEPS) / float(T)
        rh = sum(1 for x in P["maxs"] if x >= BOX_HI - BOXEPS) / float(T)
        wi = [k for k in range(T) if P["steps"][k] >= win0]
        tot = float(sum(P["lo"][k][2] for k in wi))
        have = all(P["lo"][k][1] is not None and P["hi"][k] is not None for k in wi)
        occ[(a, s)] = dict(rl=rl, rh=rh, bf=(rl < BOXFREE and rh < BOXFREE),
                           cl=(sum(P["lo"][k][1] for k in wi) / tot) if have else None,
                           chh=(sum(P["hi"][k] for k in wi) / tot) if have else None)
        shr[(a, s)] = [(st, v, x) for st, v, _n, x in P["shrink"]]
    check(not pbad, "40 / 40 probe files: 10,000 records at steps 0,5,..,49995, n_beta == the arm's m on every "
          "record, the record's beta vector of the harness's length (scalar 1, layerwise 62; chunk777 / nodewise: 62 "
          "PER-TENSOR MEANS of log step size, HF.py _probe), beta moved", ";".join(pbad[:4]))
    say("  box occupancy per arm (max over seeds): rec_lo / rec_hi, the published 5 % gate")
    for a in ARMS:
        if all((a, s) in occ for s in SEEDS_AT[RUNG[a]]):
            o = [occ[(a, s)] for s in SEEDS_AT[RUNG[a]]]
            say("    %-6s rec_lo %.4f  rec_hi %.4f  %s" % (a, max(x["rl"] for x in o), max(x["rh"] for x in o),
                                                          "free" if all(x["bf"] for x in o) else "BOUND"))

    ok_all = (not incomplete) and (not harness) and (not pbad)
    if not ok_all:
        say("\nSTOP: a structural check failed; nothing is read.")
        say("\nSUMMARY  PASS %d  FAIL %d" % (NPASS[0], NFAIL[0]))
        return 1

    # ---- [4] levels and the decision ---------------------------------------------------------------------------
    say("\n[4] LEVELS, SIGMA, CONTRASTS AND TOKENS, RECOMPUTED (plateau5 = mean TEST, epochs 95-99, raw .out)")
    lev = {}
    trn = {}
    for (a, s), (j, r) in info.items():
        lev[(a, s)] = avg([r["ep"][e][1] for e in range(95, 100)])
        trn[(a, s)] = avg([r["ep"][e][0] for e in range(95, 100)])
    A = {}
    ss = 0.0
    df = 0
    for a in ARMS:
        xs = [lev[(a, s)] for s in SEEDS_AT[RUNG[a]]]
        m = avg(xs)
        ss += sum((x - m) ** 2 for x in xs)
        df += len(xs) - 1
        A[a] = dict(level=m, train=avg([trn[(a, s)] for s in SEEDS_AT[RUNG[a]]]), rng=max(xs) - min(xs), n=len(xs),
                    bf=all(occ[(a, s)]["bf"] for s in SEEDS_AT[RUNG[a]]), sd=sdev(xs), xs=xs)
    sin = math.sqrt(ss / df)
    sig, which = (SIGMA_FLOOR, "SIGMA_PRIOR_frozen") if SIGMA_FLOOR >= sin else (sin, "SIGMA_INBATCH")
    for a in ARMS:
        say("  %-6s %s  n %d  plateau5 %.4f  sd %.3f  range %.3f  seeds %s" % (
            a, TOK[RUNG[a]], A[a]["n"], A[a]["level"], A[a]["sd"], A[a]["rng"],
            " ".join("%.2f" % x for x in A[a]["xs"])))
    say("  sigma_inbatch %.6f (df %d)  sigma_used %.6f (%s)" % (sin, df, sig, which))
    st = {}
    K = {}
    for r_ in RID:
        c, n = A["ch" + r_], A["nd" + r_]
        d = c["level"] - n["level"]
        se = sig * math.sqrt(1.0 / c["n"] + 1.0 / n["n"])
        st[r_] = state_of(d, se, c["rng"] > DIVG or n["rng"] > DIVG,
                          c["level"] >= HEALTH and n["level"] >= HEALTH, c["bf"] and n["bf"])
        K["D" + r_] = (d, se)
        K["DT" + r_] = c["train"] - n["train"]
        k, L = A["k01" + r_], A["kL" + r_]
        for g in ("ch", "nd"):
            x = A[g + r_]
            K["T" + g + r_] = (k["level"] - x["level"], sig * math.sqrt(1.0 / k["n"] + 1.0 / x["n"]))
        K["G" + r_] = (L["level"] - k["level"], sig * math.sqrt(1.0 / k["n"] + 1.0 / L["n"]))
    prim = primary_word(st["W1"], st["W4"])
    c4, n4, k4 = A["chW4"], A["ndW4"], A["k01W4"]
    cmp_ok = c4["level"] >= HEALTH and n4["level"] >= HEALTH and c4["rng"] <= DIVG and n4["rng"] <= DIVG
    scal = scalar_word(k4["level"], k4["rng"] > DIVG, cmp_ok, K["TchW4"][0], K["TndW4"][0], K["TchW4"][1])
    K["DDW4"] = (K["DW4"][0] - K["DW1"][0], math.sqrt(K["DW1"][1] ** 2 + K["DW4"][1] ** 2))
    K["DDW2"] = (K["DW2"][0] - K["DW1"][0], math.sqrt(K["DW1"][1] ** 2 + K["DW2"][1] ** 2))
    for r_ in RID:
        d, se = K["D" + r_]
        say("  D_%s %+.4f  SE %.6f  [%+.4f, %+.4f]  state %s" % (r_, d, se, d - RES * se, d + RES * se, st[r_]))
    say("  PRIMARY %s   SCALAR %s" % (prim, scal))

    stamps = list(BOUND_STAMPS)
    stamps.append("SIGMA-PRIOR-FROZEN" if which == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH")
    stamps.append("ANCHOR-MATCHES-POOL" if abs(K["DW1"][0] - POOL) <= MATCHW else "ANCHOR-DIFFERS-FROM-POOL")
    for g in GID:
        if abs(A[g + "W1"]["level"] - LANDED[g]) > MATCHW:
            stamps.append("W1-LEVEL-DIFFERS-" + g)
    if prim in ("AUDIT-SURVIVES", "AUDIT-REVERSES"):
        stamps.append("TRAIN-AGREES" if (K["DTW4"] > 0) == (K["DW4"][0] > 0) else "TRAIN-DISAGREES")
    g4, gse = K["GW4"]
    if g4 <= -BAR_SURV and g4 < -RES * gse:
        stamps.append("LAYERWISE-BELOW-SCALAR-W4")
    elif g4 >= BAR_SURV and g4 > RES * gse:
        stamps.append("LAYERWISE-ABOVE-SCALAR-W4")
    if not A["k01W4"]["bf"]:
        stamps.append("SCALAR-W4-BOX-BOUND")
    for a in ARMS:
        if A[a]["rng"] > DIVG:
            stamps.append("DIVERGED-" + a)
    final = "FINAL %s | %s | %s | %s" % (prim, scal, "+".join("%s-%s" % (r_, st[r_]) for r_ in RID), " ".join(stamps))

    # ---- [5] rebuild the scorer's stdout and compare -----------------------------------------------------------
    say("\n[5] THE SCORER'S STDOUT, REBUILT LINE BY LINE AND COMPARED WITH EACH COMMITTED LOG")
    ncorp = corpus_count(csvp, tsvp)
    R = []
    R.append(("=", "=" * 110))
    R.append(("=", TITLE))
    R.append(("=", "scorer sha256 %s   design sha256 %s" % (SCORER_SHA, DESIGN_SHA)))
    R.append(("prefix", "runsdir "))
    R.append(("=", "=" * 110))
    R.append(("=", "COMPLETION  40 / 40 registered runs complete (100 epoch lines + RUN_DONE)"))
    R.append(("gprov", "  MODE submit  SCORER %s  DESIGN %s  HF %s" % (SCORER_SHA[:16], DESIGN_SHA[:16], HF_SHA[:16])))
    R.append(("=", "G-ARGS / G-ENV / G-KIND / G-STRUCT  40 of 40 complete runs pass (20 flags each, the registered ENV "
                   "line, no patch witness, 10,000 probe records with n_beta == the arm's m)"))
    R.append(("=", "G-HW (DISCLOSURE ONLY, never a gate): %s" % ", ".join(
        "%s x%d" % (k, v) for k, v in sorted(gpus.items(), key=str))))
    R.append(("=", "corpus: %s  (DISCLOSURE ONLY -- enters no bar, sigma, level, state, contrast or stamp)" % (
        "not readable" if ncorp is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (ncorp, BATCH))))
    R.append(("=", ""))
    R.append(("=", "ARMS  plateau5 = mean TEST over epochs 95-99 from each run's OWN .out; box occupancy from its OWN "
                   "probe.jsonl"))
    R.append(("=", "  %-6s %-4s %-6s %-9s %2s %9s %7s %7s %9s  %-8s %8s %8s %11s %11s" % (
        "arm", "rung", "wd", "grain", "n", "plateau5", "sd", "range", "train5", "box", "rec_lo", "rec_hi",
        "coordlo_win", "coordhi_win")))
    for a in ARMS:
        o = [occ[(a, s)] for s in SEEDS_AT[RUNG[a]]]
        cl = [x["cl"] for x in o]
        chh = [x["chh"] for x in o]
        R.append(("=", "  %-6s %-4s %-6s %-9s %2d %9.4f %7.3f %7.3f %9.4f  %-8s %8.4f %8.4f %11s %11s" % (
            a, RUNG[a], TOK[RUNG[a]], SPEC[GRN[a]], A[a]["n"], A[a]["level"], A[a]["sd"], A[a]["rng"], A[a]["train"],
            "free" if A[a]["bf"] else "BOUND", max(x["rl"] for x in o), max(x["rh"] for x in o),
            "na" if None in cl else "%.6f" % max(cl), "na" if None in chh else "%.6f" % max(chh))))
    R.append(("=", ""))
    R.append(("=", "SIGMA  O2: max(frozen floor, in-batch) -- nothing from the corpus"))
    R.append(("=", "  SIGMA_PRIOR_frozen   %.6f  (df %d, filtered, literal)" % (SIGMA_FLOOR, SIGMA_FLOOR_DF)))
    R.append(("=", "  SIGMA_INBATCH        %.6f  (df %d, pooled within the 12 arms)" % (sin, df)))
    R.append(("=", "  SIGMA_USED           %.6f  (= %s)" % (sig, which)))
    R.append(("=", ""))
    R.append(("=", "CONTRASTS (every one WITHIN a rung, IN BATCH; +/-2 SE is a HALF-WIDTH, not a bound on any effect)"))
    for r_ in RID:
        d, se = K["D" + r_]
        R.append(("=", "  D_%s  = ch%s - nd%s = %+.4f pp = %+.2f SE (SE %.6f)  [%+.4f, %+.4f]  TRAIN %+.4f   state %s" % (
            r_, r_, r_, d, d / se, se, d - RES * se, d + RES * se, K["DT" + r_], st[r_])))
    for kk in ("DDW2", "DDW4"):
        v, se = K[kk]
        R.append(("=", "  DD_%s = D_%s - D_W1 = %+.4f pp = %+.2f SE (SE %.6f)   DESCRIPTIVE" % (kk[2:], kk[2:], v, v / se, se)))
    for r_ in RID:
        for g in ("ch", "nd"):
            v, se = K["T" + g + r_]
            R.append(("=", "  T%s_%s = k01%s - %s%s = %+.4f pp = %+.2f SE%s" % (
                g, r_, r_, g, r_, v, v / se, "" if r_ == "W4" else "   DESCRIPTIVE")))
    for r_ in RID:
        v, se = K["G" + r_]
        R.append(("=", "  G_%s  = kL%s - k01%s = %+.4f pp = %+.2f SE   DESCRIPTIVE (layerwise vs scalar in this cell)" % (
            r_, r_, r_, v, v / se)))
    R.append(("=", ""))
    R.append(("=", "LADDERS (LEVELS, not effects; this cell only)"))
    for g in GID:
        R.append(("=", "  %-9s %s" % (SPEC[g], " | ".join("%s %.4f" % (TOK[r_], A[g + r_]["level"]) for r_ in RID))))
    R.append(("=", ""))
    R.append(("=", "LICENCE (primary)  %s" % LIC[prim]))
    R.append(("=", "LICENCE (scalar)   %s" % LIC[scal]))
    R.append(("=", ""))
    R.append(("=", final))
    for lg in logs:
        got = open(lg).read().split("\n")
        if got and got[-1] == "":
            got = got[:-1]
        mism = []
        nhost = 0
        for k in range(max(len(got), len(R))):
            if k >= len(got) or k >= len(R):
                mism.append(k + 1)
                continue
            how, w = R[k]
            g_ = got[k]
            if how == "=":
                if g_ != w:
                    mism.append(k + 1)
            elif how == "prefix":
                nhost += 1
                if not g_.startswith(w) or len(g_) <= len(w):
                    mism.append(k + 1)
            elif how == "gprov":
                nhost += 1
                if not (g_.startswith("G-PROV  ") and g_.endswith(w) and "PROVENANCE.txt" in g_
                        and g_[len("G-PROV  "):len(g_) - len(w)].endswith("/cgw1/PROVENANCE.txt")):
                    mism.append(k + 1)
        check(not mism, "log %s: %d lines, %d rebuilt byte for byte, %d host-dependent matched by form" % (
            os.path.basename(lg), len(got), len(R) - nhost, nhost),
              ("first mismatch at line %d" % mism[0]) if mism else "")
        if mism:
            k = mism[0] - 1
            say("      rebuilt: %r" % (R[k][1] if k < len(R) else None))
            say("      log    : %r" % (got[k] if k < len(got) else None))
    say("  the FINAL line, rebuilt token by token:")
    say("    " + final)

    # ---- [6] realised shrink -----------------------------------------------------------------------------------
    say("\n[6] THE REALISED PER-STEP SHRINK a*wd (DESCRIPTIVE).  caw2's registered definition (CORRECTIONS 290): per")
    say("    record, the median over the arm's step sizes of exp(beta) times the run's wd.  EXACT for scalar (m 1) and")
    say("    layerwise (m 62), whose records carry every step size.  For chunk777 / nodewise the records carry only the")
    say("    62 per-tensor MEANS of log step size, so their 'med' is a PROXY (median over tensors of the per-tensor")
    say("    geometric-mean step, times wd) -- labelled, not the same quantity.  'max' = exp(beta_true_max) * wd, the")
    say("    EXACT per-record largest single-coordinate shrink (every grain).  peak = max over the 10,000 records; win =")
    say("    median over the plateau window (step >= 47,500).  Arm value = the MEDIAN over its seeds.  Reference: a")
    say("    standard SGD CIFAR recipe's per-step lr*wd = 0.1 x 5e-4 = %.1e at its peak lr." % STD_RECIPE_SHRINK)
    say("  %-6s %-5s %-5s %12s %12s %12s %12s %11s %11s" % ("arm", "wd", "med", "med_peak", "med_win", "max_peak",
                                                         "max_win", "win/std", "maxwin/std"))
    rung_win = {}
    for a in ARMS:
        pk, wn, xp, xw = [], [], [], []
        for s in SEEDS_AT[RUNG[a]]:
            R6 = shr[(a, s)]
            pk.append(max(v for _st, v, _x in R6))
            wn.append(median([v for st_, v, _x in R6 if st_ >= win0]))
            xp.append(max(x for _st, _v, x in R6))
            xw.append(median([x for st_, _v, x in R6 if st_ >= win0]))
        a_pk, a_wn, a_xp, a_xw = median(pk), median(wn), median(xp), median(xw)
        rung_win.setdefault(RUNG[a], []).append((GRN[a], a_wn, a_xw))
        say("  %-6s %-5s %-5s %12.4e %12.4e %12.4e %12.4e %11.4f %11.4f" % (
            a, TOK[RUNG[a]], "exact" if GRN[a] in ("k01", "kL") else "proxy", a_pk, a_wn, a_xp, a_xw,
            a_wn / STD_RECIPE_SHRINK, a_xw / STD_RECIPE_SHRINK))
    for r_ in RID:
        v = dict((g, (w, x)) for g, w, x in rung_win[r_])
        ex = [v[g][0] for g in ("k01", "kL")]
        say("  rung %s (wd %-4s): exact plateau-window median shrink (scalar, layerwise) %.4e, %.4e = 1/%.0f, 1/%.0f "
            "of the standard 5e-5; per-record LARGEST coordinate, window median, highest grain %.4e = %.3fx standard"
            % (r_, TOK[r_], ex[0], ex[1], STD_RECIPE_SHRINK / ex[0], STD_RECIPE_SHRINK / ex[1],
               max(v[g][1] for g in GID), max(v[g][1] for g in GID) / STD_RECIPE_SHRINK))
    say("  NOTE: the standard recipe's decay sits INSIDE a 0.9 momentum buffer (L2), so its effective shrink is up to")
    say("        10x its lr*wd; the harness's decay sits OUTSIDE the buffer (291.2).  The ratios above are per-step lr*wd")
    say("        only, which is the comparison most favourable to 'W4 is decayed'.")

    # ---- [7] bounds and sensitivity ----------------------------------------------------------------------------
    say("\n[7] THE FLOOR / BOUND AUDIT (164.6) AND A LEAVE-ONE-SEED-OUT SENSITIVITY OF D_W4 (DESCRIPTIVE, NOT A GATE)")
    for a in ARMS:
        if RUNG[a] == "W4":
            say("  %-6s level %.4f   above HEALTH_MIN 85 by %+.4f   below 100 by %.4f" % (
                a, A[a]["level"], A[a]["level"] - HEALTH, 100.0 - A[a]["level"]))
    near = [a for a in ARMS if A[a]["level"] - HEALTH < 5.0 or 100.0 - A[a]["level"] < 5.0]
    say("  NOTE (not a gate): arms within 5 pp of HEALTH_MIN 85 or of 100: %s.  HEALTH_MIN is a run-FAILURE bar, not"
        % (", ".join(near) if near else "none"))
    say("       a floor: chance is 10 %, so no contrast here is floor-bounded; 291.5 PREDICTED every W4 level >= +5.3 pp")
    say("       above 85, and the realised W4 partition levels sit closer to it than predicted (a miss of the prior).")
    check(all(A[a]["level"] - 10.0 >= 50.0 and 100.0 - A[a]["level"] >= 5.0 for a in ARMS),
          "no arm is near chance (10 %) or within 5 pp of 100, so no contrast is floor- or ceiling-bounded")
    say("  every VANISHES / TIES reading is a BOUND ('below the bar'), never a measured zero (291 bound 4)")
    loo = []
    for drop in SEEDS_AT["W4"]:
        c = avg([lev[("chW4", s)] for s in SEEDS_AT["W4"] if s != drop])
        n = avg([lev[("ndW4", s)] for s in SEEDS_AT["W4"] if s != drop])
        loo.append(c - n)
        say("  drop seed %d: D_W4 (3 v 3) %+.4f" % (drop, c - n))
    say("  leave-one-seed-out D_W4 range %+.4f .. %+.4f" % (min(loo), max(loo)))
    for r_ in RID:
        say("  rung %s per-seed paired ch - nd (same seed): %s" % (r_, " ".join(
            "s%d %+.2f" % (s, lev[("ch" + r_, s)] - lev[("nd" + r_, s)]) for s in SEEDS_AT[r_])))

    say("\nSUMMARY  PASS %d  FAIL %d" % (NPASS[0], NFAIL[0]))
    return 0 if NFAIL[0] == 0 else 1
