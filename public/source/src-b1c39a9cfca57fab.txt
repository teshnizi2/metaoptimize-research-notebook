#!/bin/bash
# =============================================================================
# cPL2_plainnet_head.sh -- `cpl2`: TURN cpl1's SUBTRACTION INTO A MEASUREMENT.
#   ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS {75,76,77},
#   PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.
#   CORRECTIONS 222.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account; every guard and scancel line is filtered on the
# `cpl2-` prefix alone.
#
# REGISTERED SCORER: analysis/cPL2_plainnet_head_score.py, committed BEFORE this
# batch is submitted (RULE 21).  Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cPL2_plainnet_head_score.py <runsdir>
#
# THE NETWORK RUNS FROM cpl1's ISOLATED HARNESS TREE, NOT THE LIVE ONE.
#   $WS/harness_cpl1/cifar10 (built by bin/cPL1_stage_harness.sh, CORRECTIONS 215,
#   VERIFIED unedited by guard 4a, which also re-runs tests/test_plainnet.py) and
#   $WS/jobs/run_cifar_cpl1.sh, which differs from jobs/run_cifar.sh in EXACTLY its
#   `cd` line.  Nothing new is staged: cpl2 reuses the tree whose build_network.py
#   sha (POST_SHA e65e6773...a18b) cpl1 ran on.
#
# THE ARMS.  INDICES ARE PlainNet's (53 tensors), NOT ResNet's; guard 4 RE-DERIVES
# every string on the LIVE PlainNet model by NAME and NUMEL and writes the manifest.
#   k01   scalar                                                      m=1
#   kL    layerwise                                                   m=53
#   ISO   sets:1-43,45-49,51-53/layer4.0.bn2.weight,layer4.1.bn2.weight
#         [51,2]  isolates 44, 50   cpl1's ISO, byte-identical: the in-batch POSITIVE CONTROL
#   HEAD  sets:1-49,51-53/layer4.1.bn2.weight
#         [52,1]  isolates 50       THE ARM: the tensor cpl1 credited BY SUBTRACTION, alone
#   TWIN  sets:1-46,48-53/layer4.1.bn1.weight
#         [52,1]  isolates 47       its exact twin (same block, BN scale, numel 512), alone
#
# COST.  cpl1 measured 9.9503 GPU-h for 15 PlainNet runs (0.663 GPU-h each; 29:31
# on an L4).  15 jobs ~= 10.0 GPU-h expected.  HARD BOUND = WALL x 15 = 45 GPU-h.
# WALL=03:00:00 as cpl1 (1.89x the worst parent run) fits gpu-short's 4:00:00.
#
# RULE 20.  Every value is ONE shell token of [A-Za-z0-9_.,:/-].  BETA_CLIP and
# PROBE ride the runner's ENV line; PROBE_TENSOR rides the patch's own line.
# analysis/argsline_guard.py IS NOT EDITED.
#
# USAGE   bash bin/cPL2_plainnet_head.sh               # dry run: every guard, every
#         bash bin/cPL2_plainnet_head.sh --dry-run     #   composed line printed; submits nothing
#         bash bin/cPL2_plainnet_head.sh --submit      # the ONLY submitting form
# ENV     CPL2_REGISTERED_COMMIT=<sha>  on a host with no git repo, --submit requires
#                                       the operator to declare the commit whose tree
#                                       was staged (guard 1b); written to PROVENANCE.txt.
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cpl1/cifar10
RUNNER=$WS/jobs/run_cifar_cpl1.sh
SAVE=$WS/runs/cpl2

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=PlainNet18_c100
REFNET=ResNet18_c100
SEEDS="75 76 77"
MST=1e-3
A0=1e-6
NTENS=53
TOTPAR=11046308
G_K01='scalar'
G_KL='layerwise'
G_ISO='sets:1-43,45-49,51-53/layer4.0.bn2.weight,layer4.1.bn2.weight'
G_HEAD='sets:1-49,51-53/layer4.1.bn2.weight'
G_TWIN='sets:1-46,48-53/layer4.1.bn1.weight'
ARMS="k01 kL ISO HEAD TWIN"
NJOBS=15
WALL=03:00:00
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cPL2_plainnet_head_score.py"
SELF="$REPO/bin/cPL2_plainnet_head.sh"
TESTBW="$REPO/tests/test_probe_tensor_blockwise_cpl2.py"
TESTB="$REPO/tests/test_probe_tensor_blockwise.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_plainnet.py"
PATCHF="$REPO/patches/patch_plainnet.py"
STAGEF="$REPO/bin/cPL1_stage_harness.sh"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"

echo "=============================================================="
echo " cpl2 -- turn cpl1's subtraction into a measurement: layer4.1.bn2.weight ALONE"
echo " $DSET / $NET (ResNet18_c100 with every residual addition removed; PlainNet indices)"
echo " AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01  $G_K01"
echo "   kL   $G_KL"
echo "   ISO  $G_ISO"
echo "   HEAD $G_HEAD"
echo "   TWIN $G_TWIN"
echo " 5 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY: DELTA_HEAD = plateau5(HEAD) - plateau5(TWIN); KEY SECONDARY D_HEAD = HEAD - k01; ISO = positive control"
echo " DOCUMENTED INVOCATION: python3 analysis/cPL2_plainnet_head_score.py <runsdir>"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$SELF" "$TESTBW" "$TESTB" "$TESTT" "$TESTF" "$PATCHF" "$STAGEF" "$LIBF" "$AGF" "$CSVP" \
         "$RUNNER" "$HFH" "$HFPRE" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cPL2_plainnet_head_score.py bin/cPL2_plainnet_head.sh tests/test_probe_tensor_blockwise_cpl2.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF bin/_lib_guards.sh analysis/argsline_guard.py 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the three registration files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
  DEL=$(git -C "$REPO" diff HEAD --numstat -- analysis/ 2>/dev/null | awk '$2+0>0{print}')
  [ -z "$DEL" ] && echo "guard 1b2: no uncommitted deletions under analysis/ (RULE 16)" \
               || guard_fail "guard 1b2: analysis/ has uncommitted deletions: $DEL"
else
  if [ -n "${CPL2_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CPL2_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CPL2_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CPL2_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING its O2 proof and the invocation ----
SELFT=/tmp/cpl2_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
grep -qE "^  PASS O2 SUBTRACT +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cpl2's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cPL2_plainnet_head_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 75, 76 or 77" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" \
  || guard_fail "guard 1c4: the scorer's fresh-seed check did not pass"
grep -q "^  PASS ISO is BYTE-IDENTICAL to cpl1's ISO" "$SELFT" \
  && echo "guard 1c5: ISO is byte-identical to cpl1's ISO (the positive control is the rescue being decomposed)" \
  || guard_fail "guard 1c5: ISO not proved byte-identical to cpl1's ISO"
grep -q "^  PASS neither singleton spec (HEAD, TWIN) has EVER run" "$SELFT" \
  && echo "guard 1c6: neither singleton has ever run in the corpus (50 was never isolated alone)" \
  || guard_fail "guard 1c6: the never-run check on HEAD/TWIN did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cpl2-' "$CSVP" && guard_fail "guard 2: cpl2-* already in the CSV" || echo "guard 2: no cpl2-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
gn = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], gn))
raise SystemExit(1 if hit or gn not in ([], ["cpl1"]) else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cpl2-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cpl2-*.out on disk" || guard_fail "guard 2b: $NOUT cpl2-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (75|76|77)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 75/76/77" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 75/76/77"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cpl2-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cpl2-* jobs" || guard_fail "guard 2e: sacct shows $NS cpl2-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cpl2-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cpl2-* in squeue" || guard_fail "guard 2f: $NQ cpl2-* already queued"
fi

# ---- guard 3/4a: cpl1's isolated tree is still consistent and sees CIFAR-100 ------
bash "$STAGEF" >/tmp/cpl2_stage.$$ 2>&1 \
  && { echo "guard 4a: bin/cPL1_stage_harness.sh (UNEDITED) VERIFIES the cpl1 tree cpl2 runs from (V-e re-runs tests/test_plainnet.py):"; grep -E 'V-|VERIFIED' /tmp/cpl2_stage.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: the cpl1 tree is NOT consistent"; tail -20 /tmp/cpl2_stage.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b" ] \
  && echo "guard 3b: the tree's build_network.py is the one cpl1 ran on (POST_SHA e65e6773...a18b)" \
  || guard_fail "guard 3b: the tree's build_network.py is not cpl1's POST_SHA"
[ "$(sha "$HFH")" = "$(sha "$LIVE/Optimizers/HF.py")" ] && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: the tree's HF.py and HF.py.pre_probe_tensor are byte-identical to the live ones ($(sha "$HFH" | cut -c1-16)...)" \
  || guard_fail "guard 3c: the tree's HF.py or its .pre_probe_tensor differs from the live one"

# ---- guard 4c: the PlainNet patch suite on the tree --------------------------------
if python3 "$TESTF" --pre "$HARN/build_network.py.pre_plainnet" --post "$BNH" \
     --hf "$HFH" --csv "$CSVP" >/tmp/cpl2_pltest.$$ 2>&1; then
  echo "guard 4c: tests/test_plainnet.py PASSES on the cpl1 tree ($(grep -c '^  PASS' /tmp/cpl2_pltest.$$) PASS, $(grep -c '^  FAIL' /tmp/cpl2_pltest.$$) FAIL)"
else
  guard_fail "guard 4c: tests/test_plainnet.py FAILED on the cpl1 tree"; grep '^  FAIL' /tmp/cpl2_pltest.$$ | head -20
fi

# ---- guard 4b: the optimiser carries its patches, proved inert -----------------------
grep -q 'PATCH_NAMESETS' "$HFH" && echo "guard 4b: HF.py carries PATCH_NAMESETS" || guard_fail "guard 4b: HF.py lacks PATCH_NAMESETS"
NPT=$(grep -c 'PATCH_PROBE_TENSOR' "$HFH" 2>/dev/null); NPT=${NPT:-0}
[ "$NPT" = 3 ] && echo "guard 4b: HF.py carries PATCH_PROBE_TENSOR (3 markers)" \
               || guard_fail "guard 4b: HF.py has $NPT PATCH_PROBE_TENSOR markers, the patch inserts exactly 3"
python3 "$TESTT" --pre "$HFPRE" --post "$HFH" --structural-only >/tmp/cpl2_struct.$$ 2>&1 \
   && echo "guard 4b2: structural inertness R0 PASSES on $HFH" \
   || { guard_fail "guard 4b2: structural inertness R0 FAILED"; tail -20 /tmp/cpl2_struct.$$; }

# ---- guard 4e: BLOCKWISE inertness ON PlainNet18_c100, for cpl2's THREE specs, on THIS HF and THIS build_network
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH")
BLOG="$SAVE/inertness_blockwise_plainnet_cpl2.log"
if [ -f "$BLOG" ] && grep -q '^ALL PASS$' "$BLOG" && grep -q "^POST_SHA256 $HFSHA" "$BLOG" \
   && grep -q "^BN_SHA256 $BNSHA" "$BLOG"; then
  echo "guard 4e: the BLOCKWISE inertness test ALL PASS on THIS HF sha and THIS build_network sha, on $NET"
  for G in "$G_ISO" "$G_HEAD" "$G_TWIN"; do
    grep -qF -- "$G" "$BLOG" || guard_fail "guard 4e: $BLOG does not mention the spec $G"
  done
  grep -q "live $NET: $NTENS parameter tensors" "$BLOG" || guard_fail "guard 4e: $BLOG not produced on a $NTENS-tensor $NET"
  grep -q "^SCORER_SPECS_AGREE" "$BLOG" || guard_fail "guard 4e: $BLOG lacks the successor test's scorer cross-check"
  echo "guard 4e: $(grep -c '^  PASS' "$BLOG") PASS / $(grep -c '^  FAIL' "$BLOG") FAIL in $BLOG"
else
  guard_fail "guard 4e: no ALL PASS blockwise-on-PlainNet inertness log for HF $HFSHA / build_network $BNSHA at $BLOG -- run:
    mkdir -p $SAVE
    { echo POST_SHA256 $HFSHA; echo BN_SHA256 $BNSHA; python3 $TESTBW --pre $HFPRE --post $HFH \\
        --cifar-dir $HARN --work /tmp/cpl2_bw; } 2>&1 | tee $BLOG"
fi

# ---- guard 4c': the literals here, in the scorer and in the stage script agree ------
python3 - "$SC" "$STAGEF" "$G_K01" "$G_KL" "$G_ISO" "$G_HEAD" "$G_TWIN" "$NET" "$NTENS" "$TOTPAR" "$SEEDS" <<'PYEOF' \
  || guard_fail "guard 4c': launcher / scorer / stage-script literals disagree"
import sys, re, ast
sc, st = open(sys.argv[1]).read(), open(sys.argv[2]).read()
bad = []
spec = ast.literal_eval(re.search(r"^SPEC = (\{.*?^\})", sc, re.S | re.M).group(1))
mine = dict(zip(("k01", "kL", "ISO", "HEAD", "TWIN"), sys.argv[3:8]))
bad += [k for k in set(mine) | set(spec) if mine.get(k) != spec.get(k)]
def lit(name):
    mm = re.search(r"^%s = (.+?)(\s*#.*)?$" % name, sc, re.M)
    return ast.literal_eval(mm.group(1)) if mm else None
if lit("NET") != sys.argv[8]: bad.append("NET")
if lit("NTENS") != int(sys.argv[9]): bad.append("NTENS")
if lit("TOTPAR") != int(sys.argv[10]): bad.append("TOTPAR")
if tuple(lit("SEEDS") or ()) != tuple(int(x) for x in sys.argv[11].split()): bad.append("SEEDS")
for name in ("PRE_SHA", "POST_SHA"):
    b = re.search(r"^%s=([0-9a-f]{64})$" % name, st, re.M)
    if not b or lit(name) != b.group(1): bad.append(name)
if bad:
    print("!!! guard 4c': disagreement on %s" % sorted(bad)); raise SystemExit(1)
print("guard 4c': all 5 spec strings, NET, NTENS, TOTPAR, SEEDS, PRE_SHA and POST_SHA identical in launcher / scorer / stage script")
PYEOF

# ---- guard 4: THE LIVE-MODEL MANIFEST, built from the cpl1 tree, per arm with NAMES ----
mkdir -p "$SAVE"
MTMP=/tmp/cpl2_manifest.$$
python3 - "$HARN" "$MTMP" "$NTENS" "$TOTPAR" "$NET" "$REFNET" "$SC" "$G_ISO" "$G_HEAD" "$G_TWIN" <<'PYEOF' \
  || guard_fail "guard 4: live partition / structure / source check FAILED"
import sys, os, inspect, re, operator, importlib.util
harn, out, ntens, totpar, net_name, ref_name, scp = sys.argv[1], sys.argv[2], int(sys.argv[3]), int(sys.argv[4]), sys.argv[5], sys.argv[6], sys.argv[7]
specs = dict(zip(("ISO", "HEAD", "TWIN"), sys.argv[8:11]))
spm = importlib.util.spec_from_file_location("cpl2sc", scp); S = importlib.util.module_from_spec(spm); spm.loader.exec_module(S)
sys.path.insert(0, harn); os.chdir(harn)
import torch, torch.nn as nn, torch.fx
from build_network import build_network
from Optimizers.HF import HF
net = build_network(net_name, "cpu"); par = build_network(ref_name, "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]; numel = [int(p.numel()) for _n, p in net.named_parameters()]
pnames = [n for n, _ in par.named_parameters()]
own = {}
for mn, m in net.named_modules():
    for pn, _p in m.named_parameters(recurse=False):
        own[(mn + "." if mn else "") + pn] = type(m).__name__
T, tot = len(nps), sum(numel); bad = 0
print("guard 4: live %s from the cpl1 tree -- %d tensors, %d parameters (parent %s: %d)" % (net_name, T, tot, ref_name, len(pnames)))
if T != ntens or tot != totpar:
    print("!!! guard 4: live model is %d / %d, the batch assumes %d / %d" % (T, tot, ntens, totpar)); raise SystemExit(1)
removed = [n for n in pnames if n not in names]
psh = dict((n, tuple(p.shape)) for n, p in par.named_parameters())
if removed != list(S.SHORTCUT_9) or [(n, tuple(s)) for n, s in nps] != [(n, psh[n]) for n in pnames if n not in S.SHORTCUT_9]:
    print("!!! guard 4: PlainNet is not the parent minus exactly the 9 shortcut tensors"); bad += 1
else:
    print("guard 4: PlainNet's ordered (name, shape) list == %s's minus EXACTLY the 9 shortcut tensors" % ref_name)
if any("shortcut" in n for n, _ in net.named_modules()):
    print("!!! guard 4: a module is named shortcut"); bad += 1
def adds(m):
    g = torch.fx.symbolic_trace(m).graph
    return sum(1 for nd in g.nodes if (nd.op == "call_function" and nd.target in (operator.add, operator.iadd, torch.add))
               or (nd.op == "call_method" and nd.target in ("add", "add_", "__add__", "__iadd__")))
na, nr = adds(net), adds(par)
print("guard 4: torch.fx add ops -- %s %d, %s %d" % (net_name, na, ref_name, nr))
if na != 0 or nr != 8: bad += 1
nbn = sum(isinstance(m, nn.BatchNorm2d) for m in net.modules()); ngn = sum(isinstance(m, nn.GroupNorm) for m in net.modules())
print("guard 4: normalisers BatchNorm2d %d  GroupNorm %d" % (nbn, ngn))
if (nbn, ngn) != (S.N_BATCHNORM, 0): bad += 1
lines = ["NETWORK %s" % net_name, "NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot,
         "NORM BatchNorm2d %d GroupNorm %d" % (nbn, ngn), "PARENT %s %d" % (ref_name, len(pnames)),
         "PARENT_REMOVED %s" % " ".join(removed),
         "CLIP_C %s" % S.CLIP, "EPOCHS %d" % S.EPOCHS, "META_STEPS %d" % (S.EPOCHS * (50000 // S.BATCH))]
lines += ["TENSOR %d %s %d %s" % (i + 1, names[i], numel[i], own[names[i]]) for i in range(T)]
w512 = [(i + 1, names[i]) for i in range(T) if numel[i] == 512 and names[i].endswith(".weight") and len(nps[i][1]) == 1]
print("guard 4: the 512-wide BN scales on the LIVE model (1-based, PlainNet indices): %s" % w512)
lines.append("W512 %s" % ",".join("%d:%s" % x for x in w512))
