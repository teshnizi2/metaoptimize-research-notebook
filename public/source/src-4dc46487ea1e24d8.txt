set -u
# CANCEL: sw-cos-* (21) is a 7-point AdamW+cosine LR sweep on ResNet18/CIFAR-10.
# It is a strict DUPLICATE of fxcos-*, which is already complete at n=3 and
# brackets the peak with a flat top (3e-4 -> 94.062, 1e-3 -> 94.093) and a
# collapsing high side (constant-LR 3e-3 -> 85.962).  sw adds only 1e-5, 3e-5
# and 1e-2, all far outside the plateau.  No open question is attached to it.
IDS=$(squeue -h -u account_a -t PD -o "%i|%j" | grep "|sw-cos-" | cut -d"|" -f1 | tr "\n" " ")
echo "cancelling sw-cos: $(echo $IDS | wc -w) jobs"
[ -n "$IDS" ] && scancel $IDS
# PROMOTE: sc50-*-s1 is the second seed of the ResNet50 rung (see B2).
for j in $(squeue -h -u account_a -t PD -o "%i|%j" | grep "|sc50-" | cut -d"|" -f1); do
  scontrol update JobId=$j Nice=50 2>/dev/null
done
# PROMOTE: fc100-cos-* is the CIFAR-100 non-meta baseline -- axis 4 on the
# second dataset, currently zero coverage.  40000 -> 200.
for j in $(squeue -h -u account_a -t PD -o "%i|%j" | grep "|fc100-cos-" | cut -d"|" -f1); do
  scontrol update JobId=$j Nice=200 2>/dev/null
done
# PROMOTE: fxcos-3e-3 (turns the R18 sweep over on the high side) and
# fxcos-{1e-3,3e-4}-s3/s4 (takes the headline baseline cell to n=5).  -> 250.
for j in $(squeue -h -u account_a -t PD -o "%i|%j" | grep "|fxcos-" | cut -d"|" -f1); do
  scontrol update JobId=$j Nice=250 2>/dev/null
done
echo "AFTER_TRIM=$(squeue -h -u account_a | wc -l)"
