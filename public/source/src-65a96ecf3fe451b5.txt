def _betacf(a, b, x):
    MAXIT, EPS, FPMIN = 300, 3e-16, 1e-300
    qab, qap, qam = a + b, a + 1.0, a - 1.0
    c = 1.0
    d = 1.0 - qab * x / qap
    if abs(d) < FPMIN:
        d = FPMIN
    d = 1.0 / d
    h = d
    for m in range(1, MAXIT + 1):
        m2 = 2 * m
        aa = m * (b - m) * x / ((qam + m2) * (a + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        h *= d * c
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        de = d * c
        h *= de
        if abs(de - 1.0) < EPS:
            break
    return h


def betainc(a, b, x):
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    lbeta = (math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b)
             + a * math.log(x) + b * math.log1p(-x))
    if x < (a + 1.0) / (a + b + 2.0):
        return math.exp(lbeta) * _betacf(a, b, x) / a
    return 1.0 - math.exp(lbeta) * _betacf(b, a, 1.0 - x) / b


def t_sf(t, df):
    """P(T > t) for Student-t on df degrees of freedom."""
    if df <= 0:
        return float("nan")
    x = df / (df + t * t)
    p = 0.5 * betainc(0.5 * df, 0.5, x)
    return p if t > 0 else 1.0 - p


def t_two(t, df):
    return 2.0 * t_sf(abs(t), df)


def binom_sf(k, n, p=0.5):
    """P(X >= k) for Binomial(n, p) -- the SIGN TEST, exact."""
    tot = 0.0
    for i in range(k, n + 1):
        tot += math.comb(n, i) * (p ** i) * ((1 - p) ** (n - i))
    return tot


def welch(a, b):
    """(diff, se, df) for mean(a) - mean(b), Welch-UNPAIRED.

    Unpaired on purpose: the campaign's two-way ANOVA finds SEED statistically
    NULL (F(60,85)=1.21, p=0.213) while BATCH is overwhelming (F(62,85)=5.47,
    p=6.9e-13).  Per-seed agreement is reported as a SIGN TEST, never as a
    pairing argument."""
    if len(a) < 2 or len(b) < 2:
        return (st.mean(a) - st.mean(b) if a and b else float("nan"),
                float("nan"), float("nan"))
    va, vb = st.variance(a), st.variance(b)
    na, nb = len(a), len(b)
    se = math.sqrt(va / na + vb / nb)
    if se == 0:
        return st.mean(a) - st.mean(b), 0.0, float("nan")
    df = (va / na + vb / nb) ** 2 / ((va / na) ** 2 / (na - 1) + (vb / nb) ** 2 / (nb - 1))
    return st.mean(a) - st.mean(b), se, df


def ols(xs, ys):
    """(slope, intercept, se_slope, r2) -- plain least squares."""
    n = len(xs)
    mx, my = sum(xs) / n, sum(ys) / n
    sxx = sum((x - mx) ** 2 for x in xs)
    sxy = sum((x - mx) * (y - my) for x, y in zip(xs, ys))
    b = sxy / sxx
    a = my - b * mx
    resid = [y - (a + b * x) for x, y in zip(xs, ys)]
    if n > 2:
        s2 = sum(r * r for r in resid) / (n - 2)
        seb = math.sqrt(s2 / sxx)
    else:
        seb = float("nan")
    syy = sum((y - my) ** 2 for y in ys)
    r2 = 1.0 - sum(r * r for r in resid) / syy if syy > 0 else float("nan")
    return b, a, seb, r2

EP_RE = re.compile(r"Epoch\s+(\d+).*?Test Accuracy:\s*([0-9.]+)", re.I)
ARGS_RE = re.compile(r"^ARGS:\s*(.*)$")
ENV_RE = re.compile(r"^ENV:\s*(.*)$")


def series_from_out(path):
    """[(epoch, test_acc)] sorted by epoch, deduplicated on the LAST occurrence."""
    seen = {}
    with open(path, "r", errors="replace") as fh:
        for line in fh:
            m = EP_RE.search(line)
            if m:
                seen[int(m.group(1))] = float(m.group(2))
    return sorted(seen.items())


def window_mean(series, upto_n, w=WINDOW):
    """Mean test accuracy over the last `w` recorded epochs among the first
    `upto_n`.  At upto_n == len(series) and w == 5 this IS plateau5, exactly --
    the selftest asserts that identity."""
    if upto_n < w or upto_n > len(series):
        return None
    seg = series[:upto_n][-w:]
    return sum(v for _, v in seg) / float(w)


def header_pins(path):
    """The ARGS: and ENV: header lines of a run, as dicts.  This is the ONLY
    instrument that can prove what a completed run actually ran -- the CSV is a
    derived artefact and an unidentified schedule has already made one batch of
    this corpus unreadable (CLOSEOUT 5.1)."""
    args, env = {}, {}
    with open(path, "r", errors="replace") as fh:
        for _ in range(12):
            line = fh.readline()
            if not line:
                break
            m = ARGS_RE.match(line.strip())
            if m:
                toks = m.group(1).split()
                i = 0
                while i < len(toks):
                    if toks[i].startswith("--"):
                        k = toks[i][2:]
                        v = toks[i + 1] if i + 1 < len(toks) and not toks[i + 1].startswith("--") else ""
                        args[k] = v
                        i += 2 if v else 1
                    else:
                        i += 1
            m = ENV_RE.match(line.strip())
            if m:
                for tok in m.group(1).split():
                    if "=" in tok:
                        k, v = tok.split("=", 1)
                        env[k] = v
    return args, env

# THE RAIL READER.  CORRECTED 2026-08-30, CORRECTIONS 117.1.
#
# The per-coordinate rail counts live in  n_at_lo / n_at_hi  over  n_beta.
# r["beta"] is a PER-TENSOR summary -- 62 entries on ResNet18 against
# n_beta = 14,420 -- and reading rails off it is BLIND to real clipping and,
# where it does fire, gets the arm ranking BACKWARDS.  Both failures were
# measured on data already on disk, in the plateau5 window:
#
#   fa1 (ms=3e-4, 100 ep, box -25:-2.3026).  The 62-list reads 0.000000 on
#   ALL 24 runs.  The coordinate counts read nodewise s1/s4/s5 rec_hi = 1.0000
#   -- a clipped coordinate in EVERY record -- while all 18 merged-tail runs
#   are exactly 0.  The ceiling binds, and it binds only on the arm the paper
#   is about.
#
#   ar1 (ms=3e-4, 100 ep, box -15:-2.3026).  The 62-list says chunk777 is
#   7x MORE floor-bound than nodewise (0.2305 vs 0.0325).  The coordinate
#   counts say nodewise is 75x more bound than chunk777 (0.1103 vs 0.00145).
#   The instrument inverted the sign of the arm asymmetry.
#
# The 62-list is retained as a THIRD, LABELLED column so the historical
# statistic stays visible, and is never used for a gate.
# ---------------------------------------------------------------------------
def rail_fracs(rec, lo, hi):
    """(coord_lo, coord_hi, rec_lo, rec_hi, tensor_lo, tensor_hi) or None.

    coord_*  : fraction of the run's beta COORDINATES at a rail (n_at_*/n_beta)
    rec_*    : 1 if ANY coordinate is at that rail in this record, else 0
    tensor_* : the historical 62-entry per-tensor statistic; DIAGNOSTIC ONLY
    """
    nb = rec.get("n_beta")
    nlo = rec.get("n_at_lo")
    nhi = rec.get("n_at_hi")
    if not isinstance(nb, (int, float)) or not nb or nlo is None or nhi is None:
        return None
    nb = float(nb)
    clo, chi = float(nlo) / nb, float(nhi) / nb
    rlo = 1.0 if float(nlo) > 0 else 0.0
    rhi = 1.0 if float(nhi) > 0 else 0.0
    b = rec.get("beta")
    if isinstance(b, list) and b:
        n = float(len(b))
        tlo = sum(1 for x in b if x <= lo + 1e-9) / n
        thi = sum(1 for x in b if x >= hi - 1e-9) / n
    else:
        tlo = thi = float("nan")
    return clo, chi, rlo, rhi, tlo, thi


def read_occupancy(probe_dir, lo, hi):
    """Occupancy at BOTH denominators, over the last quarter of the run.

    coord_lo/coord_hi -- the MEAN FRACTION OF beta COORDINATES at a rail.
                         THIS IS THE GATE.  Read from n_at_lo / n_at_hi over
                         n_beta (CORRECTIONS 117.1); the 62-entry per-tensor
                         summary that this scorer previously used is blind to
                         real clipping and inverts the arm ranking on ar1.
    rec_lo/rec_hi     -- the FRACTION OF RECORDS in which AT LEAST ONE
                         coordinate sits at a rail.  Printed beside the first,
                         never instead of it: the two disagree by up to three
                         orders of magnitude on the same run.
    Returns (coord_lo, coord_hi, rec_lo, rec_hi, n_records)."""
    p = os.path.join(probe_dir, "probe.jsonl")
    if not os.path.exists(p):
        return (None,) * 4 + (0,)
    recs = []
    with open(p, "r", errors="replace") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            try:
                recs.append(json.loads(line))
            except Exception:
                continue
    if not recs:
        return (None,) * 4 + (0,)
    tail = recs[int(len(recs) * 0.75):]
    c_lo, c_hi, r_lo, r_hi = [], [], [], []
    for r in tail:
        f = rail_fracs(r, lo, hi)
        if f is None:
            continue
        c_lo.append(f[0]); c_hi.append(f[1]); r_lo.append(f[2]); r_hi.append(f[3])
    if not c_lo:
        return (None,) * 4 + (len(recs),)
    n = float(len(c_lo))
    return (sum(c_lo) / n, sum(c_hi) / n, sum(r_lo) / n, sum(r_hi) / n, len(recs))

def find_out(runs_dirs, fam, short, seed):
    """The .out for one run.  .out files live in the ROOT of runs/, NOT in
    runs/<family>/ -- a layout that has already once turned a real result into a
    would-be VOID (CORRECTIONS 116, 'A PROCESS NOTE')."""
    stem = "%s-%s-s%d" % (fam, short, seed)
    hits = []
    for d in runs_dirs:
        if not os.path.isdir(d):
            continue
        for fn in os.listdir(d):
            if fn.startswith(stem + "-") and fn.endswith(".out"):
                hits.append(os.path.join(d, fn))
            elif fn == stem + ".out":
                hits.append(os.path.join(d, fn))
    return sorted(hits)[-1] if hits else None


def find_probe(probe_dirs, fam, short, seed):
    """probe dirs live in <root>/<family>/probe_<short>_<family>_s<seed>."""
    name = "probe_%s_%s_s%d" % (short, fam, seed)
    for d in probe_dirs:
        p = os.path.join(d, fam, name)
        if os.path.isdir(p):
            return p
        p = os.path.join(d, name)
        if os.path.isdir(p):
            return p
    return None


def load_cell(fam, spec, runs_dirs, probe_dirs):
    """{short: {seed: rec}} with plateau5, the full series, and occupancy."""
    lo, hi = (float(x) for x in spec["box"].split(":"))
    out = {}
    for short in spec["arms"]:
        out[short] = {}
        for s in spec["seeds"]:
            f = find_out(runs_dirs, fam, short, s)
            if f is None:
                continue
            ser = series_from_out(f)
            pd = find_probe(probe_dirs, fam, short, s)
            clo, chi, rlo, rhi, nrec = read_occupancy(pd, lo, hi) if pd else ((None,) * 4 + (0,))
            boxfree = (clo is not None and clo < BOX_GATE and chi < BOX_GATE)
            out[short][s] = dict(
                path=f, series=ser, n_ep=len(ser),
                plateau5=window_mean(ser, len(ser)) if len(ser) >= WINDOW else None,
                coord_lo=clo, coord_hi=chi, rec_lo=rlo, rec_hi=rhi,
                n_records=nrec, probe=pd, boxfree=boxfree,
                complete=(len(ser) >= spec["epochs"]))
    return out


def arm_values(cell, short, boxfree_only=True):
    d = cell.get(short, {})
    return {s: r["plateau5"] for s, r in d.items()
            if r["plateau5"] is not None and r["complete"]
            and (r["boxfree"] or not boxfree_only)}


def contrast(cell, a, b, boxfree_only=True):
    """(diff, se, df, n_a, n_b, k, n_pairs) for arm a minus arm b."""
    A, B = arm_values(cell, a, boxfree_only), arm_values(cell, b, boxfree_only)
    if len(A) < 2 or len(B) < 2:
        return (float("nan"),) * 3 + (len(A), len(B), 0, 0)
    d, se, df = welch(list(A.values()), list(B.values()))
    pair = [s for s in A if s in B]
    k = sum(1 for s in pair if A[s] > B[s])
    return d, se, df, len(A), len(B), k, len(pair)


def traj_contrast(cell, a, b, frac, budget, boxfree_only=True):
    """(diff, se, df, n) for arm a minus arm b at a budget FRACTION."""
    n_ep = max(WINDOW, int(round(frac * budget)))
    A, B = {}, {}
    for short, dst in ((a, A), (b, B)):
        for s, r in cell.get(short, {}).items():
            if boxfree_only and not r["boxfree"]:
                continue
            v = window_mean(r["series"], min(n_ep, r["n_ep"]))
            if v is not None and r["n_ep"] >= n_ep:
                dst[s] = v
    if len(A) < 2 or len(B) < 2:
        return float("nan"), float("nan"), float("nan"), 0, n_ep
    d, se, df = welch(list(A.values()), list(B.values()))
    return d, se, df, min(len(A), len(B)), n_ep

            for i in range(1, len(xs)):
                if xs[i - 1][1] < 0 <= xs[i][1]:
                    e0, d0 = xs[i - 1]
                    e1, d1 = xs[i]
                    cross = e0 + (e1 - e0) * (0 - d0) / (d1 - d0)
            if cross is not None:
                P("      crossing epoch (linear interpolation on the grid): ~%.0f of %d"
                  " = %.0f%% of budget" % (cross, spec["epochs"],
                                           100.0 * cross / spec["epochs"]))
        elif allpos:
            v = "NO REVERSAL"
            P("      VERDICT: NO REVERSAL -- D >= 0 at every grid point.")
        else:
            v = "UNRESOLVED"
            P("      VERDICT: UNRESOLVED -- %s, so neither branch of the registered"
              % ("D dips below zero but never at |t| >= %.1f at or below %.2f of budget"
                 % (SIG_T, REVERSAL_FRAC) if dips
                 else "D never resolvably dips, but a grid point is unreadable"))
            P("      rule fires.  This cell supports NEITHER a reversal claim NOR a")
            P("      no-reversal claim.")
        verdicts[fam] = v
    P("")
    P("    TRAJECTORY VERDICTS: %s" % ", ".join("%s %s" % (f, v) for f, v in verdicts.items()))
    verdicts = {f: v for f, v in verdicts.items() if v != "VOID"}
    if verdicts and len(set(verdicts.values())) > 1:
        P("    **THE REVERSAL IS NOT UNIVERSAL.**  It is present in some cells and")
        P("    absent in others, so it is a property OF A CELL, not of the effect.")
        P("    A horizon caveat must be written per cell, with this table beside it.")
    elif verdicts and set(verdicts.values()) == {"REVERSAL PRESENT"}:
        P("    THE REVERSAL IS PRESENT IN EVERY CELL SCORED.  Every headline in the")
        P("    paper must be written 'at a 100-epoch budget' until a longer horizon")
        P("    is measured.")

    # =====================================================================
    # DESCRIPTIVE
    # =====================================================================
    P("")
    P("=" * 78)
    P("DESCRIPTIVE ONLY -- NO VERDICT IS ISSUED FROM ANY OF THIS.")
    P("=" * 78)
    P("    per-arm plateau5 means (box-free seeds only), and the m each arm allocates:")
    for fam, spec in CELLS.items():
        parts = []
        for short, gran in spec["arms"].items():
            v = arm_values(data[fam], short)
            m = REG_M.get((spec["net"], gran), "?")
            parts.append("%s(%s, m=%s) %s" % (short, gran, m,
                                              "%.3f" % st.mean(v.values()) if v else "n/a"))
        P("      %-5s %s" % (fam, "; ".join(parts)))
    if csvp and os.path.exists(csvp):
        P("")
        P("    CSV CROSS-CHECK (second instrument; plateau5 from the .out is PRIMARY):")
        rows = list(csv.DictReader(open(csvp)))
        for fam in CELLS:
            n = sum(1 for r in rows if r["run"].startswith(fam + "-"))
            P("      %-5s %d rows in results/all_runs.csv" % (fam, n))
    P("")
    P("NOT COMPUTED, BY REGISTRATION: any field statistic.  Direction C was dropped")
    P("on its own pre-registered adoption gate (C2 t -11.14, C3 t -23.26).")
    P("NOT CLAIMED: a RULE-11 tuned result on any cell; any pooling across")
    P("architectures or datasets; any new effect.  an0 CORRECTS PUBLISHED NUMBERS.")
    return dict(f1=f1, shares=shares, traj=verdicts, ladder=lad)


def ladder_fits(runs_dirs, probe_dirs, P):
    """The ck1/cx2 chunk ladder, fitted every way the record fits it.
    DESCRIPTIVE: ck1 and cx2 are TWO batches, so any slope spanning them carries
    the 0.20-0.29 pp cross-batch floor that neither pairing nor Welch removes."""
    pts = []
    for fam, arms in LADDER.items():
        for short, gran in arms.items():
            vals = []
            for s in LADDER_SEEDS:
                f = find_out(runs_dirs, fam, short, s)
                if not f:
                    continue
                ser = series_from_out(f)
                v = window_mean(ser, len(ser))
                if v is not None and len(ser) >= 100:
                    vals.append(v)
            if vals:
                pts.append((fam, gran, REG_M[("ResNet18", gran)],
                            sum(vals) / len(vals), len(vals)))
    if len(pts) < 3:
        P("      (ladder unavailable at these paths -- slope not re-fitted)")
        return {}
    pts.sort(key=lambda p: p[2])
    P("")
    P("      the ck1/cx2 chunk ladder, re-derived from the .out (plateau5):")
    for fam, gran, m, v, n in pts:
        P("        %-5s %-11s m=%9d  plateau5 %.3f (n=%d)" % (fam, gran, m, v, n))
    ck = [p for p in pts if p[0] == "ck1"]
    fits = {}
    if len(ck) >= 3:
        b, a, seb, r2 = ols([math.log10(p[2]) for p in ck], [p[3] for p in ck])
        fits["ck1 5-rung OLS"] = (b, seb, r2)
        P("        ck1-only OLS (%d rungs, ONE batch): %+0.4f +- %.4f pp/decade, R2 %.3f"
          % (len(ck), b, seb, r2))
    if len(pts) >= 4:
        b, a, seb, r2 = ols([math.log10(p[2]) for p in pts], [p[3] for p in pts])
        fits["ck1+cx2 pooled OLS"] = (b, seb, r2)
        P("        ck1+cx2 pooled OLS (%d rungs, TWO batches): %+0.4f +- %.4f pp/decade,"
          " R2 %.3f" % (len(pts), b, seb, r2))
        P("        ^ CROSS-BATCH.  sd_batch ~ 0.21 pp is inside this fit and neither")
        P("          pairing nor Welch removes it.  DESCRIPTIVE.")
    # the local secant the record's -0.407 comes from
    by_m = {p[1]: p for p in pts}
    if "chunk1024" in by_m and "chunk128" in by_m:
        p1, p2 = by_m["chunk1024"], by_m["chunk128"]
        sec = (p1[3] - p2[3]) / (math.log10(p1[2]) - math.log10(p2[2]))
        fits["local secant chunk1024<->chunk128"] = (sec, float("nan"), float("nan"))
        P("        local secant bracketing m=14,420 (chunk1024 <-> chunk128): %+0.4f"
          " pp/decade" % sec)
    P("        THE LADDER RUNGS SPLIT 1-D TENSORS.  Every K here is below the")
    P("        largest 1-D tensor (%d), so these rungs do NOT hold the tail intact"
      % MAX_1D["ResNet18"])
    P("        the way chunk777/chunk2325/chunk835/chunk2500 do.  That is a second,")
    P("        structural reason the ladder slope is the wrong instrument for a")
    P("        count correction between two TAIL-PRESERVING partitions.")
    return fits

def selftest():
    ok = fail = 0

    def chk(cond, msg):
        nonlocal ok, fail
        if cond:
            ok += 1
        else:
            fail += 1
            print("FAIL: %s" % msg)

    here = os.path.dirname(os.path.abspath(__file__))
    repo = os.path.dirname(here)
    bs_path = os.path.join(repo, BATCH_SCRIPT)
    chk(os.path.exists(bs_path), "batch script %s must exist" % BATCH_SCRIPT)
    bs = open(bs_path).read() if os.path.exists(bs_path) else ""

    # --- the registration cannot drift from the batch script -----------------
    chk("TAG=an0" in bs, "batch script must pin TAG=an0")
    chk("NJOBS=0" in bs, "an0 must emit ZERO Slurm jobs")
    chk("GPU_HOURS=0" in bs, "an0 must cost ZERO GPU-hours")
    chk("BOX_GATE=%s" % BOX_GATE in bs, "the box gate must match the batch script")
    chk("BOX_ASYM=%s" % BOX_ASYM in bs, "the asymmetry bar must match")
    chk("T06_RATIO=%s" % T06_RATIO in bs, "the T0.6 gate must match")
    chk("SHARE_SPAN_BAR=%s" % SHARE_SPAN_BAR in bs, "the share-span bar must match")
    chk("SLOPE_SPREAD_BAR=%s" % SLOPE_SPREAD_BAR in bs, "the slope-spread bar must match")
    chk("REVERSAL_FRAC=%s" % REVERSAL_FRAC in bs, "the reversal fraction must match")
    chk("IDENT_TOL=%s" % IDENT_TOL in bs, "the identity tolerance must match")
    for fam in CELLS:
        chk(fam in bs, "batch script must name cell %s" % fam)
    for fam in LADDER:
        chk(fam in bs, "batch script must name ladder family %s" % fam)

    # --- the batch script must carry every guard the build requires ----------
    for needle, why in (
            ("guard 1b", "the ENV/ARGS pin, read off the .out headers on disk"),
            ("guard 2g", "the CSV name-collision guard"),
            ("guard 4", "THE BETA-ALLOCATION GUARD -- m MEASURED, never asserted"),
            ("guard 4e", "the torch-free second instrument"),
            ("guard 5", "the disk guard"),
            ("guard 6", "the queue-depth guard"),
            ("guard 7", "the throughput-headroom guard"),
            ("guard 8", "the box-reachability guard"),
            ("guard 9", "the ZERO-GPU guard"),
            ("--rule13", "RULE 13: guards tested in BOTH directions"),
            ("RULE 19 IS NOT SATISFIED", "the RULE-19 disclosure must be explicit"),
            ("BINDING", "guard 7 must name its binding quantity")):
        chk(needle in bs, "batch script must contain %r (%s)" % (needle, why))

    # --- registered m: internal consistency ----------------------------------
    # every COUNT-MATCHED comparator must use K strictly greater than the largest
    # 1-D tensor, or it splits the tensors whose merging is the prescription.
    for fam, spec in CELLS.items():
        for role in ("fine", "coarse"):
            sh = spec[role]
            if not sh:
                continue
            gran = spec["arms"][sh]
            chk((spec["net"], gran) in REG_M, "m must be registered for %s/%s"
                % (spec["net"], gran))
            k = int(gran.replace("chunk", ""))
            chk(k > MAX_1D[spec["net"]],
                "%s's %s comparator K=%d must exceed the largest 1-D tensor (%d) or "
                "it SPLITS the tail" % (fam, role, k, MAX_1D[spec["net"]]))
    # and the ladder rungs must NOT -- they are descriptive precisely because
    # they split it.  Both directions of the same structural fact.
    for gran in LADDER["ck1"].values():
        k = int(gran.replace("chunk", ""))
        chk(k <= MAX_1D["ResNet18"] or k == 1024,
            "ck1 rung %s is expected to split the tail" % gran)
    chk(int("1024") > MAX_1D["ResNet18"], "chunk1024 does NOT split the tail -- the "
        "ladder is mixed, which is itself the disclosure")

    # --- count matching, from the registered m -------------------------------
    for fam, spec in CELLS.items():
        if spec["fine"]:
            mf = REG_M[(spec["net"], spec["arms"][spec["fine"]])]
            mn = REG_M[(spec["net"], spec["arms"][spec["node"]])]
            dec = abs(math.log10(mf / float(mn)))
            chk(dec < 1e-3, "%s's D leg must be count-matched to <1e-3 decades "
                            "(got %.2e)" % (fam, dec))
        if spec["coarse"]:
            mc = REG_M[(spec["net"], spec["arms"][spec["coarse"]])]
            m1 = REG_M[(spec["net"], spec["arms"][spec["n1d"]])]
            dec = abs(math.log10(mc / float(m1)))
            chk(dec < 1e-3, "%s's G leg must be count-matched to <1e-3 decades "
                            "(got %.2e)" % (fam, dec))

    # --- THE IDENTITY.  Algebra, not a fit.  Exact on synthetic arms. --------
    node, fine, coarse, n1d = 91.0, 91.7, 92.9, 92.8
    D, G, U, T = fine - node, coarse - n1d, coarse - fine, n1d - node
    chk(abs(T - (D + (U - G))) < IDENT_TOL,
        "the SECOND decomposition T = D + C with C = U - G must close identically "
        "-- it is the same algebra with the count charged on the other side")
    chk(abs(((D - G) + U) - (D + (U - G))) < IDENT_TOL,
        "and the two decompositions must agree on T while disagreeing on which "
        "term the count sits in -- that disagreement is the whole point")
    chk(abs(T - ((D - G) + U)) < IDENT_TOL,
        "T = (D-G) + U must close EXACTLY (residual %.3e)" % abs(T - ((D - G) + U)))
    # and it must close for ANY four numbers -- the identity has no content about
    # the data, which is exactly why F2 is safe to run post-hoc.
    for quad in ((0, 0, 0, 0), (1, -2, 3.5, -4.25), (-10, 10, -10, 10)):
        n_, f_, c_, o_ = quad
        chk(abs((o_ - n_) - (((f_ - n_) - (c_ - o_)) + (c_ - f_))) < IDENT_TOL,
            "the identity must close for %s" % (quad,))

    # --- THE F3 GRID.  Its endpoint must be plateau5 BY CONSTRUCTION. --------
    ser = [(i, 80.0 + 0.1 * i) for i in range(100)]
    chk(abs(window_mean(ser, len(ser)) - sum(v for _, v in ser[-5:]) / 5.0) < 1e-12,
        "the last grid point must be plateau5 exactly")
    chk(window_mean(ser, 40) == sum(v for _, v in ser[35:40]) / 5.0,
        "a grid point must be the 5-epoch window ENDING at that epoch")
    chk(window_mean(ser, 3) is None, "a window shorter than 5 epochs must return None")
    chk(window_mean(ser, 200) is None, "a window past the end must return None")
    chk(FRACS[-1] == 1.0, "the grid must end at the full budget")
    chk(all(FRACS[i] < FRACS[i + 1] for i in range(len(FRACS) - 1)),
        "the grid must be strictly increasing")

    # --- distribution tails, against published critical values ---------------
    chk(abs(t_sf(6.3138, 1) - 0.05) < 1e-4, "t_{.95,1} = 6.3138")
    chk(abs(t_sf(1.7459, 16) - 0.05) < 1e-4, "t_{.95,16} = 1.7459")
    chk(abs(t_sf(2.2281, 10) - 0.025) < 1e-4, "t_{.975,10} = 2.2281")
    chk(abs(t_two(0.0, 9) - 1.0) < 1e-9, "two-sided t at 0 is 1")
    chk(abs(binom_sf(9, 9) - 1.0 / 512) < 1e-12, "sign test 9/9 -> 1/512")
    chk(abs(binom_sf(5, 9) - 0.5) < 1e-12, "sign test 5/9 -> 0.5")

    # --- welch and ols -------------------------------------------------------
    d, se, df = welch([1.0, 2.0, 3.0], [0.0, 1.0, 2.0])
    chk(abs(d - 1.0) < 1e-12, "welch mean difference")
    chk(abs(se - math.sqrt(2.0 / 3)) < 1e-12, "welch se")
    b, a, seb, r2 = ols([0.0, 1.0, 2.0, 3.0], [1.0, 3.0, 5.0, 7.0])
    chk(abs(b - 2.0) < 1e-12 and abs(a - 1.0) < 1e-12, "ols on an exact line")
    chk(abs(r2 - 1.0) < 1e-12, "ols R2 on an exact line is 1")

    # --- THE DECISION RULES, TESTED IN BOTH DIRECTIONS (RULE 13) -------------
    # F1's rule fires when the argmax MOVES between scales, and must NOT fire
    # when it does not.  Both constructed here from arithmetic alone.
    cellsA = {"a": dict(d=1.640, err=29.689), "b": dict(d=0.727, err=8.110)}
    relA = {k: 100 * v["d"] / v["err"] for k, v in cellsA.items()}
    chk(max(cellsA, key=lambda k: cellsA[k]["d"]) != max(relA, key=lambda k: relA[k]),
        "F1 must FIRE when a big-pp/big-error cell loses on the relative scale")
    cellsB = {"a": dict(d=1.640, err=8.200), "b": dict(d=0.727, err=8.110)}
    relB = {k: 100 * v["d"] / v["err"] for k, v in cellsB.items()}
    chk(max(cellsB, key=lambda k: cellsB[k]["d"]) == max(relB, key=lambda k: relB[k]),
        "F1 must NOT fire when the argmax is the same on both scales")
    chk(29.689 / 8.110 > T06_RATIO,
        "the CIFAR-100 vs CIFAR-10 error budgets must TRIP the T0.6 gate")
    chk(8.664 / 8.110 < T06_RATIO,
        "two CIFAR-10 cells must NOT trip it -- the gate must be a real cut")
    # F2's rule
    chk(max([12.0, 3.0, -4.0, 35.0]) - min([12.0, 3.0, -4.0, 35.0]) > SHARE_SPAN_BAR,
        "F2 must FIRE on a 39-pp span of count share")
    chk(max([12.0, 10.0, 14.0]) - min([12.0, 10.0, 14.0]) < SHARE_SPAN_BAR,
        "F2 must NOT fire on a 4-pp span")
    chk(max(SLOPE_RECORD.values()) - min(SLOPE_RECORD.values()) > SLOPE_SPREAD_BAR,
        "the record's four slope values must TRIP the spread bar")
    chk(0.51 - 0.49 < SLOPE_SPREAD_BAR,
        "two nearly equal values must NOT trip it")
    # F3's rule
    chk(0.40 <= REVERSAL_FRAC and 1.00 > REVERSAL_FRAC,
        "the reversal window must sit inside the budget, below the endpoint")
    chk(SIG_T == 2.0, "a sign is called at |t| >= 2, everywhere")

    # --- NO FIELD STATISTIC, over this file's OWN identifiers ---------------
    me = open(os.path.abspath(__file__)).read()
    import ast as _ast
    idents = set()
    for node_ in _ast.walk(_ast.parse(me)):
        if isinstance(node_, _ast.Name):
            idents.add(node_.id)
        elif isinstance(node_, _ast.Attribute):
            idents.add(node_.attr)
        elif isinstance(node_, (_ast.FunctionDef, _ast.ClassDef)):
            idents.add(node_.name)
        elif isinstance(node_, _ast.arg):
            idents.add(node_.arg)
    low = {i.lower() for i in idents}
    for banned in ("rho", "neff", "n_eff", "negcount", "neg_count"):
        chk(not any(banned in i for i in low),
            "this scorer must compute no field statistic (identifier %r)" % banned)
    reads = [m for m in _ast.walk(_ast.parse(me))
             if isinstance(m, _ast.Constant) and isinstance(m.value, str)
             and "neg_" + "counts" in m.value]
    chk(not reads, "this scorer must not name a neg_" + "counts artefact in any path")

    # --- HONESTY CLAUSES, asserted against this file's own text -------------
    for phrase, why in (
            ("RULE 19 IS NOT SATISFIED", "the RULE-19 disclosure must be in the file"),
            ("POST-HOC", "every verdict must be labelled post-hoc"),
            ("CORRECTION", "an0 produces corrections, not tests"),
            ("SIGN TEST", "per-seed agreement is a sign test"),
            ("RULE 11 is open", "the RULE-11 gap must be printed"),
            ("DESCRIPTIVE ONLY", "the descriptive section must be labelled"),
            ("BOTH denominators", "occupancy must be printed at both denominators"),
            ("GUARD-CONFOUNDED", "an asymmetric guard must remove a contrast"),
            ("cross-batch", "the cross-batch floor must be named where it applies"),
            ("mean-of-last-20", "the CSV plateau column must be disowned"),
            ("NEITHER a reversal claim NOR a no-reversal claim",
             "an unresolved cell must be refused in BOTH directions"),
            ("VERDICT: VOID", "a cell the box gate emptied must get NO trajectory "
                              "verdict, not a substantive one")):
        chk(phrase in me, "%s (%r)" % (why, phrase))

    # --- structural constants ------------------------------------------------
    chk(len(CELLS) == 8, "an0 reads 8 accuracy cells")
    chk(sum(len(c["seeds"]) * len(c["arms"]) for c in CELLS.values()) == 113,
        "an0 reads 113 accuracy runs (got %d)"
        % sum(len(c["seeds"]) * len(c["arms"]) for c in CELLS.values()))
    chk(MIN_SEEDS == 3, "a cell needs at least 3 box-free seeds per arm")
    chk(WINDOW == 5, "plateau5's window is 5 epochs")

    print("selftest: %d/%d PASS" % (ok, ok + fail) if fail == 0
          else "selftest: %d PASS, %d FAIL" % (ok, fail))
    return 0 if fail == 0 else 1


def main():
    here = os.path.dirname(os.path.abspath(__file__))

                         "(repeatable)")

        return selftest()
    runs = a.runs or [os.path.join(repo, "..", "runs"),

    probes = a.probes or list(runs)
    present = [d for d in runs if os.path.isdir(d)]
    if not present:
        print("no run directory found at %s -- point --runs at the raw .out root."
              % ", ".join(runs))
        return 0
    score(runs, probes, a.csv)
    return 0
