#!/bin/bash
# =============================================================================
# cA3_lsm1_meta_r_curve.sh -- `lsm1`, THE META-OPTIMISER r-CURVE.
# IS THE SURVIVING INTERIOR PEAK LION-SPECIFIC?
#
# NO CODE CHANGE.  Nothing in the live tree is touched.  This batch varies two
# knobs that already exist: the additive operator's ETA_RATIO and train.py's
# --alg-meta.
#
# REGISTERED SCORER: analysis/cA3_lsm1_score.py, committed BEFORE this batch is
# submitted (STANDING RULE 21).  Every bar quoted here is frozen there, and that
# file's selftest() asserts each one against THIS FILE's own text.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# HF.py's additive operator (PATCH_HIER, `_apply_hier`, layerwise branch) is
#
#     d  = beta_t - beta_{t-1};   dm = mean(d);
#     beta_t := beta_{t-1} + dm + r * (d - dm)
#
# r = 1 is the arithmetic identity (plain layerwise).  The archived r-curve at
# this cell has an INTERIOR OPTIMUM near r = 0.06, and g4m has just shown it is
# NOT a meta-stepsize dial: optimising ms within each r leaves +0.821 pp.
#
# THE ACCOUNT g4m DID NOT KILL, AND THE ONE THIS BATCH TESTS.  The audit's
# mechanism was LION-specific.  Read from the live tree, not from prose --
# HF.py's two meta updates at weight_decay_meta = 0 are
#
#     Lion_meta_update:  beta[i] -= ms * sign( b2*mom + (1-b2)*HtT_gradft )
#     Adam_meta_update:  beta[i] -= ms * mom / sqrt( mu*trace + eps )
#
# Under LION every coordinate of d is EXACTLY +-ms, so the shared component is
# dm = ms*(2p-1) with p the fraction of groups stepping up: pooling multiplies
# the shared motion by |2p-1|, measured 8-10x smaller than ms at layerwise,
# while the deviations stay of order ms.  On that account the interior optimum
# sits where the two magnitudes balance, r* ~ |2p-1| ~ 0.10-0.125 against an
# archived argmax of 0.06-0.07 -- the right order -- and the whole shape is an
# artefact of the sign quantisation.  UNDER ADAM THERE IS NO SIGN QUANTISATION,
# the identity dm = ms*(2p-1) does not exist, and the account predicts THE PEAK
# VANISHES.  GUARD 3c re-derives both update lines from the live HF.py.
#
# THE INFERENCE IS ASYMMETRIC AND THAT IS DECLARED IN ADVANCE.  Lion -> Adam
# changes the normaliser, the momentum form and the per-step magnitude too, so:
#   * PEAK SURVIVES -> the sign account is REFUTED outright (it predicted the
#     opposite; no auxiliary hypothesis rescues a prediction that ran backwards).
#   * PEAK VANISHES -> CONSISTENT WITH the sign account and NOT proof of it.
# The registered wording for the vanishing branch says "consistent with".
#
# -----------------------------------------------------------------------------
# THE DESIGN.  24 jobs, ONE submission, so BATCH (the unit of replication,
# F(62,85)=5.47) is held constant and every contrast is WITHIN this batch.
#   meta = Adam :  r in {0, 0.03, 0.06, 0.2, 1}  x seeds {0,1,2}   15 jobs
#   meta = Lion :  r in {0,       0.06,      1}  x seeds {0,1,2}    9 jobs
#   ResNet18 / CIFAR-10 / layerwise / SGDm base / alpha0 1e-6 / ms 1e-3 /
#   100 ep / batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / PROBE=5.
#
# WHY THE ADAM ARM IS SAMPLED MORE FINELY.  The registered prediction is that
# the ADAM peak vanishes.  A sparse Adam interior would satisfy that prediction
# by construction -- the exact defect g4m's grid change existed to avoid.  So
# Adam gets THREE interior rungs and Lion ONE (0.06, the archived argmax).  The
# asymmetry can only make an Adam peak EASIER to find, i.e. it works against the
# registered prediction.  The headline interaction is nevertheless reported on
# the MATCHED sub-grid {0, 0.06, 1} so no part of it rests on the asymmetry.
#
# WHY THE LION ARM IS RUN AT ALL, given 57 archived Lion runs exist: BATCH is
# the unit of replication.  The 9 Lion jobs buy a WITHIN-BATCH manifold gate and
# a within-batch interaction; they are the control that makes Adam readable.
#
# WHY ms IS HELD AT 1e-3.  The comparator is the LION curve at the SAME ms, and
# GUARD 2b re-derives that Adam is ALIVE and RESPONSIVE there: plain layerwise
# Adam 90.532 (n=9, 3 batches) against Lion's 90.906 (n=17, 6 batches), and the
# archived `adg-A` Adam additive cell at r=0.03 reads 91.471, i.e. +0.939 pp
# above plain Adam.  The Adam r-curve is known to MOVE before this is submitted;
# a flat result cannot be dismissed as a dead arm.  GUARD 2 re-derives what is
# missing: `adg` ran r = 0.03 and 0.1 and NEITHER ENDPOINT, so the archive
# cannot form PEAK at all.  That gap is what this batch fills.
#
# THE momentum-param-meta ASYMMETRY IS DECLARED, NOT HIDDEN.  Lion runs
# --momentum-param-meta 0.99 and Adam 0.9, the canonical setting of each family
# in this corpus and the value each arm's archived comparator was measured at.
# In the live tree Lion's momentum is an EMA and Adam's a bare accumulator, so
# the constants are not interchangeable.  A declared confound with the meta axis.
#
# -----------------------------------------------------------------------------
# PRE-REGISTERED, and it can come out either way.  plateau5 = mean of the LAST 5
# test epochs is PRIMARY; the CSV `plateau` (last-20) column is BANNED and
# `best_test` is not a plateau.
#
#   c(meta,r)   = 3-seed mean plateau5
#   PEAK_A      = max_{r in 0.03,0.06,0.2} c(Adam,r) - max_{r in 0,1} c(Adam,r)
#   PEAKm(meta) = c(meta,0.06) - max(c(meta,0), c(meta,1))     matched sub-grid
#   DELTA       = PEAKm(Lion) - PEAKm(Adam)                    interaction
#
#   SURVIVE_BAR=0.50  PEAK_A >= 0.50 pp -> NOT the Lion sign mechanism.
#   VANISH_BAR=0.20   PEAK_A <= 0.20 pp -> consistent with the sign mechanism.
#   Anything strictly between is UNRESOLVED and is reported as UNRESOLVED.
#
# PEAK_A IS A LOWER BOUND on the true interior lift: both endpoints are sampled
# exactly, so the subtrahend is not truncated, while the interior maximum is
# taken over sampled rungs only.  A SURVIVES therefore can never be a grid
# artefact; only a VANISHES can, which is why GUARD/gate G2 constrains only that.
#
# WHERE THE BARS COME FROM.  SIGMA_W=0.1438 pp is the pooled WITHIN-CELL SD of
# plateau5 re-derived from the 57 archived Lion additive runs at this exact
# configuration (10 r-cells, df 47).  SE of a 3-seed mean = 0.0830 pp.
#   0.50 pp is 4.26 SE-of-difference AND exactly half the archived Lion interior
#     lift at this cell and this ms (93.286 at r=0.06 against 92.288 at r=0 and
#     90.971 at r=1, i.e. +0.997 over the better endpoint).  It is the SAME
#     number g4m registered as its REFUTED_BAR, deliberately.
#   0.20 pp is at the simulated 95th percentile (0.176) of PEAK_A under a flat
#     null at that sigma, so VANISHES catches >= 95% of genuinely flat curves;
#     against a TRUE peak of archived Lion size the chance of seeing PEAK_A
#     <= 0.20 is below 1e-6, so the bar cannot manufacture a vanishing.
#
# -----------------------------------------------------------------------------
# THE GATES, IN ORDER.  Full text in analysis/cA3_lsm1_score.py.
#   G0  PROVENANCE, absolutely gating, from each run's OWN ARGS/ENV line.  THE
#       META AXIS IS CHECKED AT FLAG LEVEL: `ml2` and `sm3` were composed with a
#       duplicated --alg-meta, ran to completion and were reported before anyone
#       read a run's own ARGS line -- argparse keeps the LAST occurrence, so 24
#       runs labelled Adam/RMSProp in the corpus are meta=Lion to this day.
#       This batch IS a meta-optimiser contrast, so that bug would silently turn
#       it into a null.  --alg-meta must appear exactly once; an Adam run must
#       carry --normalizer-param-meta 0.999 and --momentum-param-meta 0.9 and
#       must NOT carry --Lion-beta2-meta; a Lion run the mirror image.
#   G1  THE NO-OP CONTROL, ONE PER META.  c(meta,1) must reproduce the archived
#       plain-layerwise HIER=none grand mean OF THE SAME META: Lion 90.906
#       (6 batches), Adam 90.532 (3 batches).  Bands re-derived as 3 sigma on the
#       combined between-batch + 3-seed + prior-mean noise: G1_BAND_LION=0.4263,
#       G1_BAND_ADAM=0.7518.  A miss on either: VOID.  G1 is NOT the
#       meta-identity check -- the two priors are only 0.374 pp apart.  G0 is.
#   G2  THE TRUNCATION GATE, and it constrains ONLY a vanishing.  The unsampled
#       interval is (0, 0.03); if the Adam argmax sits on r=0.03 while PEAK_A is
#       inside the vanishing band, a taller peak may hide there ->
#       UNRESOLVED-TRUNCATED.  It can never overturn a SURVIVES.
#   G3  THE MANIFOLD GATE.  The LION arm of THIS batch must itself show
#       PEAKm(Lion) >= SURVIVE_BAR, else a flat Adam curve cannot be told from an
#       absent effect and the verdict is UNRESOLVED-NOMANIFOLD.
#   G4  THE NOISE GATE.  This batch's own pooled within-cell SD must be
#       <= NOISE_MULT=2.0 x SIGMA_W, else the bars do not hold: UNRESOLVED-NOISY.
#   G5  CROSS-BATCH REPRODUCIBILITY.  c(Adam,0.03) against archived `adg-A`
#       (91.471).  DESCRIPTIVE ONLY -- every G7 contrast is within-batch.
#   G6  THE BOX METER.  Rail occupancy from each run's own probe.jsonl.
#       DESCRIPTIVE ONLY -- it cannot change G7.
#   G7  THE PRIMARY.  PEAK_A against the two registered bars, with DELTA, SPAN,
#       ARGMAX and POOL reported alongside as declared secondaries.
#
# MAY NOT CLAIM: anything about RMSProp, any meta-stepsize other than 1e-3,
# alpha0=1e-3, ResNet34/50, CIFAR-100, a granularity other than layerwise, or an
# operator other than `additive`.  A vanishing is "consistent with" the sign
# account, never "confirms" it.
#
# USAGE   bash bin/cA3_lsm1_meta_r_curve.sh            # dry run, prints all 24
#         bash bin/cA3_lsm1_meta_r_curve.sh --submit
# =============================================================================

# ---- GUARD 2b -- THE FEASIBILITY PREMISE, measured not assumed --------------
python3 - "$CSVP" <<'PYEOF' || guard_fail "guard 2b: feasibility"
import csv, sys, collections, statistics
rows=list(csv.DictReader(open(sys.argv[1])))
def cell(meta,hier):
    return [r for r in rows if (r["hier"] or "")==hier and r["network"]=="ResNet18"
            and r["dataset"]=="CIFAR10" and r["granularity"]=="layerwise"
            and r["alpha0"]=="1e-6" and r["base"]=="SGDm" and r["meta"]==meta
            and r["meta_stepsize"]=="1e-3" and r["beta_clip"]=="-15:-2.3026"
            and r["augment"]=="1" and r["batch_size"]=="100" and r["gamma"]=="1"
            and r["epochs_requested"]=="100" and r["collapsed"]=="0"
            and r["complete"]=="1" and r["superseded"]=="0" and r["plateau5"]]
def grand(meta):
    by=collections.defaultdict(list)
    for r in cell(meta,""): by[r["run"].split("-")[0].split("_")[0]].append(float(r["plateau5"]))
    m=[statistics.mean(v) for v in by.values()]
    return statistics.mean(m), len(m), sum(len(v) for v in by.values())
gl,bl,nl = grand("Lion"); ga,ba,na = grand("Adam")
print("guard 2b: plain layerwise  Lion %.3f (n=%d, %d batches)   Adam %.3f (n=%d, %d batches)"
      % (gl,nl,bl,ga,na,ba))
best=max(statistics.mean([float(r["plateau5"]) for r in cell("Adam","additive")
                          if "%g"%float(r["eta_ratio"])==k])
         for k in {"%g"%float(r["eta_ratio"]) for r in cell("Adam","additive")})
print("guard 2b: best archived Adam ADDITIVE cell %.3f  -> the operator moves Adam by %+.3f pp"
      % (best, best-ga))

print("guard 2b: cannot be dismissed as a dead arm.")
PYEOF

# ---- GUARD 3 -- m MEASURED on the LIVE tree, not declared -------------------
python3 - "$CIF" "$NET" "$ALPHA0" "$GRAN" "$M_LAYERWISE" "$MST" <<'PYEOF' || guard_fail "guard 3: beta allocation"
import sys, os
cif, net, a0, gran, want, mst = sys.argv[1:7]
sys.path.insert(0, cif); sys.path.insert(0, os.path.join(cif, "Optimizers"))
try:
    import torch
    from build_network import build_network
    from HF import HF
except Exception as ex:
    sys.exit("GUARD FAIL: cannot import the LIVE tree (%s) -- run on the cluster "
             "inside the venv." % ex)
class NW:
    def add_scalar(self, *a, **k): pass
B = {"alg": "SGD", "weight_decay": 0.1}
# BOTH meta arms are constructed here, so a missing required attribute is caught
# before submission rather than 30 minutes into 24 jobs.
METAS = {"Lion": {"alg": "Lion", "meta_stepsize": float(mst), "momentum_param": 0.99,
                  "Lion_beta2": 0.9, "weight_decay": 0},
         "Adam": {"alg": "Adam", "meta_stepsize": float(mst), "normalizer_param": 0.999,
                  "momentum_param": 0.9, "weight_decay": 0}}
for name, M in METAS.items():
    torch.manual_seed(0)
    opt = HF(build_network(net, "cpu"), stepsize_groups=gran, alpha0=float(a0),
             args_base=dict(B), args_meta=dict(M), gamma=1, writer=NW())
    m = sum(int(b.numel()) for b in opt.beta)
    assert m == int(want), "m(%s) is %d, registered %s" % (gran, m, want)
    print("guard 3: meta=%-5s %s on %s ALLOCATED beta m=%d (registered %s)"
          % (name, gran, net, m, want))
PYEOF

import sys

i = src.index("def _apply_hier")
body = src[i:i + 1200]

# ---- GUARD 3c -- THE MECHANISM PREMISE, read from the LIVE meta updates -----

import sys, re
src = open(sys.argv[1] + "/Optimizers/HF.py").read()
def body(name, n=700):
    i = src.index("def %s" % name)
    return src[i:i+n]
lion = body("Lion_meta_update")

print("guard 3c: LIVE Lion_meta_update  ->  beta -= ms * sign(...)   EXACTLY +-ms")
print("guard 3c: LIVE Adam_meta_update  ->  beta -= ms * mom/sqrt(...)  CONTINUOUS")

print("guard 3c: THAT is the discriminating premise, and it is now MEASURED.")

--export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,HIER=additive,ETA_RATIO=$R,SCHED=none,PROBE=$PROBE,PROBE_DIR=$SAVE/probe_${MTAG}_${RTAG}_${PREFIX}_s${S} \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 $MFLAGS --dataset $DSET --NN-name $NET \
--batch-size $BATCH --max-time 999:00:00 --gamma $GAMMA --meta-stepsize $MST \
--alpha0 $ALPHA0 --num-epochs $EPOCHS --stepsize-groups $GRAN --seed $S \
--save-directory $SAVE --run-name $RN"

   # STANDING RULE 20, BEFORE sbatch: no repeated flag anywhere on the line,
   # and the effective value of every declared design flag is what we declared.
   # --alg-meta is passed as a per-arm expectation because it is THE axis: the
   # ml2/sm3 bug was a duplicated --alg-meta on exactly this flag.
   guard_presubmit "$CMD" "${DESIGN[@]}" alg-meta="$ME" seed="$S" \
      >/tmp/lsm1_pre.$$ 2>&1 || { echo "--- $RN ---"; cat /tmp/lsm1_pre.$$; \
      guard_fail "guard 6: $RN failed the pre-submission ARGS check"; }
   NCHK=$((NCHK+1))
   rm -f /tmp/lsm1_pre.$$

   if [ "$SUBMIT" = 1 ] && [ "$FAIL" = 0 ]; then
     eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
   else
     echo "$CMD"
     N=$((N+1))
   fi
  done
 done
done

echo "guard 6: $NCHK/$NJOBS composed command lines passed the RULE 20 pre-check"
[ "$NCHK" = "$NJOBS" ] || guard_fail "guard 6: only $NCHK of $NJOBS lines checked"

if [ "$FAIL" = 1 ]; then
  echo "!!! SUBMIT ABORTED -- guards failed."
  [ "$SUBMIT" = 1 ] && echo "!!! If any job was already accepted, scancel --name by prefix $PREFIX-."
  exit 2
fi

  grep -q "^$PREFIX-" "$REPO/bin/PROTECTED.txt" 2>/dev/null || \
    echo "$PREFIX-" >> "$REPO/bin/PROTECTED.txt"
  echo "---- $PREFIX- added to bin/PROTECTED.txt; no trim may cancel it ----"
  # STANDING RULE 20, AFTER launch: read the FIRST RUN'S OWN ARGS line and abort
  # the batch on any mismatch.  This is the check that would have caught ml2/sm3.
  # NOTE the directory.  jobs/run_cifar.sh carries
  #   #SBATCH --output=/cluster/account_b/metaopt/runs/%x-%j.out
  # so the .out files land in $WS/runs, NOT in --save-directory ($SAVE, which
  # holds only the probe dirs).  Watching $SAVE returns a FALSE "UNVERIFIED".
  guard_postlaunch "$WS/runs" "$PREFIX-" "${DESIGN[@]}" || {
    echo "!!! THE LAUNCHED BATCH IS VOID AS DESIGNED.  Cancel it now."
    exit 2; }
  echo "---- the post-launch guard checks only the design flags common to BOTH"
  echo "---- arms; --alg-meta is per-arm, so audit it across the whole batch with:"
  echo "----   python3 analysis/argsline_guard.py $WS/runs --name $PREFIX-"
  echo "---- score with:  python3 analysis/cA3_lsm1_score.py --runs $WS/runs --probes $SAVE"
fi
