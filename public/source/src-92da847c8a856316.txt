def mean(v):
    return sum(v) / float(len(v))


def sd(v):
    if len(v) < 2:
        return 0.0
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def median(v):
    s = sorted(v)
    n = len(s)
    return s[n // 2] if n % 2 else 0.5 * (s[n // 2 - 1] + s[n // 2])


def file_sha(path):

def gammainc_lower_reg(a, x):
    """P(a, x), the regularised lower incomplete gamma function (series for x < a + 1, else continued fraction)."""
    if x <= 0:
        return 0.0
    gl = math.lgamma(a)
    if x < a + 1.0:
        term = s = 1.0 / a
        ap = a
        for _ in range(100000):
            ap += 1.0
            term *= x / ap
            s += term
            if abs(term) < abs(s) * 1e-15:
                break
        return s * math.exp(-x + a * math.log(x) - gl)
    b = x + 1.0 - a
    c = 1.0 / 1e-300
    d = 1.0 / b
    h = d
    for i in range(1, 100000):
        an = -i * (i - a)
        b += 2.0
        d = an * d + b
        d = 1e-300 if abs(d) < 1e-300 else d
        c = b + an / c
        c = 1e-300 if abs(c) < 1e-300 else c
        d = 1.0 / d
        de = d * c
        h *= de
        if abs(de - 1.0) < 1e-15:
            break
    return 1.0 - math.exp(-x + a * math.log(x) - gl) * h


def chi2_quantile(p, df):
    lo, hi = 0.0, max(10.0, 10.0 * df)
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if gammainc_lower_reg(df / 2.0, mid / 2.0) < p:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def sigma_gate(sig_in, df):
    """-> (ok, threshold).  ok is False iff the in-batch sigma rejects SIGMA_FROZEN at the 0.5 % level."""
    if df <= 0:
        return True, float("inf")
    thr = SIGMA_FROZEN * math.sqrt(chi2_quantile(SIGMA_GATE_P, df) / df)
    return sig_in <= thr, thr


def select(cands, key):
    """cands = [(config, dict)] in CONFIG_IDS order, already non-diverged; argmax of dict[key]; exact tie -> the earlier."""
    best = None
    for c, x in cands:
        if best is None or x[key] > best[1][key]:
            best = (c, x)
    return best[0] if best else None


def reading_of(k_ok, k_level, p_ok, p_levels_ok, t_ch, t_nd, se, b):
    if not k_ok:
        return "SCALAR-DIVERGED"
    if k_level < HEALTH_MIN:
        return "SCALAR-COLLAPSED"
    if not (p_ok and p_levels_ok):
        return "UNREADABLE"
    lo_ch, lo_nd = t_ch - b, t_nd - b
    if lo_ch >= SURV and lo_nd >= SURV and lo_ch > RES * se and lo_nd > RES * se:
        return "SCALAR-BEATS-BEST"
    tmin = min(t_ch, t_nd)
    if tmin + b <= -SURV and tmin + b < -RES * se:
        return "PARTITION-ABOVE-SCALAR"
    if tmin - b - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tmin + b + RES * se <= SURV:
        return "PARTITION-TIES-SCALAR"
    return "UNRESOLVED"


def cfg_state(k_level, k_div, cmp_ok, t_ch, t_nd, se):
    """The same ladder at ONE config, no selection so no bias term."""
    if k_div:
        return "DIVERGED"
    if k_level < HEALTH_MIN:
        return "COLLAPSED"
    if not cmp_ok:
        return "UNREADABLE"
    r = reading_of(True, k_level, True, True, t_ch, t_nd, se, 0.0)
    return {"SCALAR-BEATS-BEST": "BEATS", "PARTITION-ABOVE-SCALAR": "BELOW", "SCALAR-TIES-BEST": "TIES",
            "PARTITION-TIES-SCALAR": "LEADBOUNDED", "UNRESOLVED": "UNRESOLVED"}[r]


def branch_of(tuned, oracle, sigma_ok):
    t = tuned[len("TUNED-"):]
    o = oracle[len("ORACLE-"):]
    if t in GATE_READINGS:
        return "BRANCH-UNREADABLE"
    if not sigma_ok:
        return "BRANCH-SIGMA-INCONSISTENT"
    if t == "PARTITION-ABOVE-SCALAR":
        return "HEADLINE-REFUTED-BY-RETUNE"
    if t == "SCALAR-BEATS-BEST":
        return "NOT-A-TUNING-ARTEFACT" if o == "SCALAR-BEATS-BEST" else "SELECTION-DEPENDENT"
    if t == "SCALAR-TIES-BEST":
        return "WEAKENED-TO-TIE"
    if t == "PARTITION-TIES-SCALAR":
        return "TIE-SCALAR-LEAD-BOUNDED"
    return "RETUNE-UNDECIDED"


def bias_bound(sigma):
    return EMAX5 * sigma / math.sqrt(N_SEED)


def located(g, cfg, ceil_bulk):
    """One grain's LOC state from its TUNED selection (313.5)."""
    if cfg is None:
        return "NO-SELECTION"
    if ceil_bulk:
        return "NOT-LOCATED-CEILING"
    if cfg == D.CENTRE:
        return "LOCATED"
    if cfg == "A2":
        return "LOCATED-A0-XB" if D.CRT1_A2_MINUS_M2[g] > D.CRT1_2SE else "NOT-LOCATED-A0LO"
    return "NOT-LOCATED-" + D.EDGE_OF[cfg]


def _selected_reading(A, sel, sigma, b):
    k = sel["k01"]
    p_ok = sel["ch"] is not None and sel["nd"] is not None
    if k is None:
        return "SCALAR-DIVERGED", None, None, None
    kx = A[D.arm_of("k01", k)]
    if not p_ok:
        return reading_of(True, kx["level"], False, False, 0, 0, 1, b), None, None, None
    cx, nx = A[D.arm_of("ch", sel["ch"])], A[D.arm_of("nd", sel["nd"])]
    t_ch, t_nd = kx["level"] - cx["level"], kx["level"] - nx["level"]
    se = sigma * math.sqrt(1.0 / kx["n"] + 1.0 / cx["n"])
    r = reading_of(True, kx["level"], True, cx["level"] >= HEALTH_MIN and nx["level"] >= HEALTH_MIN, t_ch, t_nd, se, b)
    return r, t_ch, t_nd, se


def decide(A, sig_in, df_in, sigma=SIGMA_FROZEN):
    """A[arm] = dict(level, train, range, n, ceil_bulk).  sigma is SIGMA_FROZEN in score(); the argument exists only for
    the DESCRIPTIVE sensitivity scan and the selftest.  -> (branch, tuned, oracle, cfg_states, K)."""
    b = bias_bound(sigma)
    ok, thr = sigma_gate(sig_in, df_in)
    K = {"B": b, "SIGMA_OK": ok, "SIGMA_THR": thr}
    sel_t, sel_o = {}, {}
    for g in D.GRAIN_IDS:
        cands = [(c, A[D.arm_of(g, c)]) for c in D.CONFIG_IDS if A[D.arm_of(g, c)]["range"] <= DIVERGED]
        sel_t[g] = select(cands, "train")
        sel_o[g] = select(cands, "level")
    K["SEL_TUNED"], K["SEL_ORACLE"] = sel_t, sel_o
    rt, tch, tnd, se = _selected_reading(A, sel_t, sigma, b)
    ro, och, ond, ose = _selected_reading(A, sel_o, sigma, b)
    K["T_TUNED"] = (tch, tnd, se)
    K["T_ORACLE"] = (och, ond, ose)
    K["LOC"] = dict((g, located(g, sel_t[g], sel_t[g] is not None and A[D.arm_of(g, sel_t[g])]["ceil_bulk"]))
                    for g in D.GRAIN_IDS)
    K["LOCWORD"] = "LOC-ALL" if all(v in ("LOCATED", "LOCATED-A0-XB") for v in K["LOC"].values()) else "LOC-PARTIAL"
    tuned, oracle = "TUNED-" + rt, "ORACLE-" + ro
    cfg = {}
    for c in D.CONFIG_IDS:
        k, ch, nd = (A[D.arm_of(g, c)] for g in ("k01", "ch", "nd"))
        t_ch, t_nd = k["level"] - ch["level"], k["level"] - nd["level"]
        se_c = sigma * math.sqrt(1.0 / k["n"] + 1.0 / ch["n"])
        cmp_ok = (ch["level"] >= HEALTH_MIN and nd["level"] >= HEALTH_MIN and ch["range"] <= DIVERGED
                  and nd["range"] <= DIVERGED)
        cfg[c] = cfg_state(k["level"], k["range"] > DIVERGED, cmp_ok, t_ch, t_nd, se_c)
        K["T_" + c] = (t_ch, t_nd, se_c)
        K["D_" + c] = (ch["level"] - nd["level"], se_c)
    return branch_of(tuned, oracle, ok), tuned, oracle, cfg, K


def sigma_change_points(A, sig_in, df_in, lo=0.05, hi=0.60, step=0.0005):
    """DESCRIPTIVE: the sigma values (had sigma been a different literal) at which the TUNED reading would change."""
    out, prev = [], None
    n = int(round((hi - lo) / step))
    for i in range(n + 1):
        s = lo + i * step
        r = decide(A, 0.0, df_in, sigma=s)[1]
        if prev is not None and r != prev[1]:
            out.append((s, prev[1], r))
        prev = (s, r)
    return out

def parse_args_line(line):
    toks = shlex.split(line[len("ARGS:"):])
    pairs, seen, rep, i = [], set(), [], 0
    while i < len(toks):
        t = toks[i]
        if not t.startswith("--"):
            return None, ["stray token %r" % t]
        if i + 1 >= len(toks) or toks[i + 1].startswith("--"):
            return None, ["flag %s has no value" % t]
        f = t[2:]
        if f in seen:
            rep.append(f)
        seen.add(f)
        pairs.append((f, toks[i + 1]))
        i += 2
    return pairs, rep


def read_out(path):
    """A REPEATED epoch, ARGS or ENV line is recorded and becomes a harness failure (crt1's rule)."""
    L = open(path, errors="replace").read().splitlines()
    info = dict(path=path, args=None, env=None, gpu=None, done=False, eps={}, kinds_on=[], node=None, pt=False,
                dup_ep=[], n_args=0, n_env=0)
    for ln in L:
        if ln.startswith("ARGS:"):
            info["n_args"] += 1
            if info["args"] is None:
                info["args"] = ln
        elif ln.startswith("ENV:"):
            info["n_env"] += 1
            if info["env"] is None:
                info["env"] = ln
        elif ln.startswith("NODE=") and info["node"] is None:
            info["node"] = ln.split("|")[0].strip()
        elif ln.startswith("NVIDIA") and info["gpu"] is None:
            info["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            info["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            info["pt"] = True
        else:
            m = EPTR_RE.search(ln)
            if m:
                e = int(m.group(1))
                if e in info["eps"]:
                    info["dup_ep"].append(e)
                else:
                    info["eps"][e] = (float(m.group(2)), float(m.group(3)))
            k = KIND_ON_RE.match(ln)
            if k:
                info["kinds_on"].append(ln.strip())
    info["complete"] = info["done"] and all(e in info["eps"] for e in range(D.EPOCHS))
    return info


def find_outs(runsdir, prefix=PREFIX):
    out = []
    for base in (runsdir, os.path.join(runsdir, prefix)):
        out += glob.glob(os.path.join(base, "%s-*.out" % prefix))
    return sorted(set(os.path.realpath(p) for p in out))


def probe_path(runsdir, arm, seed):
    for base in (os.path.join(runsdir, PREFIX), runsdir):
        p = os.path.join(base, D.probe_dir_name(arm, seed), "probe.jsonl")
        if os.path.exists(p):
            return p
    return None


def read_probe(path):
    recs = []
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            r = json.loads(ln)
        except ValueError:
            recs.append(None)
            continue
        recs.append({k: r.get(k) for k in ("step", "n_beta", "beta_true_min", "beta_true_max", "n_at_lo", "n_at_hi", "beta")}
                    if isinstance(r, dict) else None)
    return recs


def probe_audit(recs, m_want):
    """-> (ok, why, occupancy).  G-STRUCT: 10,000 records at steps 0,5,..; n_beta == the arm's m on EVERY record; a beta
    list and an n_at_hi count on every record."""
    if recs is None:
        return False, "no probe.jsonl", None
    if len(recs) != D.N_RECORDS:
        return False, "%d records, want %d" % (len(recs), D.N_RECORDS), None
    if any(r is None for r in recs):
        return False, "an unreadable record", None
    steps = [r["step"] for r in recs]
    if steps != list(range(0, D.N_RECORDS * D.PROBE, D.PROBE)):
        return False, "record steps are not 0,5,...,%d" % ((D.N_RECORDS - 1) * D.PROBE), None
    nb = set(r["n_beta"] for r in recs)
    if nb != {m_want}:
        return False, "n_beta %s != the arm's registered m %d (the grain is not what ARGS says)" % (sorted(nb)[:3], m_want), None
    mins = [r["beta_true_min"] for r in recs]
    maxs = [r["beta_true_max"] for r in recs]
    if not (max(maxs) - min(mins) > 1e-9):
        return False, "beta never moved", None
    if any(not isinstance(r.get("beta"), list) or not r["beta"] for r in recs):
        return False, "a record without a beta list", None
    if any(not isinstance(r.get("n_at_hi"), int) for r in recs):
        return False, "a record without an n_at_hi count", None
    T = len(recs)
    rec_lo = sum(1 for x in mins if x <= D.LO + BOX_EPS) / float(T)
    rec_hi = sum(1 for x in maxs if x >= D.HI - BOX_EPS) / float(T)
    win = [r for r in recs if r["step"] >= WIN_STEP]
    occ = dict(rec_lo=rec_lo, rec_hi=rec_hi, boxfree=(rec_lo < BOXFREE_MAX and rec_hi < BOXFREE_MAX),
               ceil_frac=median([r["n_at_hi"] / float(m_want) for r in win]),
               beta_min_win=min(r["beta_true_min"] for r in win), beta_max_win=max(r["beta_true_max"] for r in win))
    occ.update(shrink_of(recs, D.WD_VALUE))
    return True, "", occ


def rec_shrink(r, wd):
    """caw2's registered definition (290): per record, the median over the record's step sizes of exp(beta) * wd;
    APPROXIMATE for the partitions (62 per-tensor MEAN log step sizes, 296.4)."""
    return median([math.exp(x) for x in r["beta"]]) * wd


def shrink_of(recs, wd):
    allv = [rec_shrink(r, wd) for r in recs]
    win = [(rec_shrink(r, wd), median(r["beta"])) for r in recs if r["step"] >= WIN_STEP]
    return dict(shrink_plateau=median([w[0] for w in win]), shrink_peak=max(allv),
                beta_plateau=median([w[1] for w in win]), beta_peak=max(r["beta_true_max"] for r in recs))


def read_corpus_disclosure(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    sys.path.insert(0, HERE)
    try:
        import corpus_exclusions as ce
    finally:
        sys.path.pop(0)
    rows = [r for r in csv.DictReader(open(csvpath)) if not r.get("run", "").startswith(PREFIX + "-")]
    return len(ce.filter_rows(rows))

        if os.path.exists(p):
            return p
    return None

def score(runsdir, csvpath=DEFAULT_CSV, out=print):
    P = out
    P("=" * 110)
    P("crt2 -- AFTER TUNING EACH GRAIN UNTIL ITS OPTIMUM IS LOCATED, DOES SCALAR TIE, BEAT OR FALL BELOW THE BETTER PARTITION?"
      "  (CORRECTIONS 313)")
    P("scorer sha256 %s   design sha256 %s" % (file_sha(os.path.abspath(__file__)), file_sha(os.path.join(HERE, "crt2_design.py"))))
    P("runsdir %s" % runsdir)
    P("=" * 110)
    harness = []

    def hfail(msg):
        harness.append(msg)
        P("  !!! HARNESS %s" % msg)

    found = {}
    for p in find_outs(runsdir):
        m = OUT_RE.match(os.path.basename(p))
        if not m:
            hfail("unparseable file name %s" % os.path.basename(p))
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        if arm not in D.ARMS or seed not in D.SEEDS[arm]:
            hfail("a %s .out that is not a registered run: %s" % (PREFIX, os.path.basename(p)))
            continue
        found.setdefault((arm, seed), []).append((jid, p))
    runs, missing, ignored = {}, [], []
    for (arm, seed) in D.RUNS:
        cands = [(jid, p, read_out(p)) for jid, p in sorted(found.get((arm, seed), []))]
        comp = [c for c in cands if c[2]["complete"]]
        ignored += [os.path.basename(c[1]) for c in cands if not c[2]["complete"]]
        if not comp:
            missing.append(D.run_name(arm, seed))
        elif len(comp) > 1:
            hfail("%s has %d COMPLETE .out files (%s) -- ambiguous" % (D.run_name(arm, seed), len(comp),
                  ", ".join(os.path.basename(c[1]) for c in comp)))
        else:
            runs[(arm, seed)] = comp[0]
    P("COMPLETION  %d / %d registered runs complete (100 epoch lines + RUN_DONE)" % (len(runs), D.NJOBS))
    if ignored:
        P("  ignored (incomplete, a complete twin or none): %s" % ", ".join(sorted(ignored)))
    if missing:
        P("  MISSING / INCOMPLETE: %s" % ", ".join(missing))

    prov = resolve_prov(runsdir)
    if prov is None:
        hfail("G-PROV: no PROVENANCE.txt under %s/%s" % (runsdir, PREFIX))
    else:
        kv = dict(ln.split(" ", 1) for ln in open(prov).read().splitlines() if " " in ln)
        want = {"SCORER_SHA256": file_sha(os.path.abspath(__file__)),
                "DESIGN_SHA256": file_sha(os.path.join(HERE, "crt2_design.py"))}
        for k, v in sorted(want.items()):
            if kv.get(k, "").strip() != v:
                hfail("G-PROV: %s in %s is %r, the file that is running is %s" % (k, prov, kv.get(k), v[:16]))
        P("G-PROV  %s  MODE %s  SCORER %s  DESIGN %s  HF %s" % (prov, kv.get("MODE", "?"), kv.get("SCORER_SHA256", "?")[:16],
          kv.get("DESIGN_SHA256", "?")[:16], kv.get("HF_SHA256", "?")[:16]))
        if kv.get("HF_SHA256", D.HF_LIVE_SHA).strip() != D.HF_LIVE_SHA:
            hfail("G-PROV: HF_SHA256 is not the registered live HF %s" % D.HF_LIVE_SHA[:16])

    per, gpus, save_dirs = {}, {}, set()
    for (arm, seed), (jid, path, info) in sorted(runs.items()):
        rn = D.run_name(arm, seed)
        if info["n_args"] != 1 or info["n_env"] != 1 or info["dup_ep"]:
            hfail("G-ONCE %s: %d ARGS lines, %d ENV lines, repeated epoch lines %s (each must appear exactly once)"
                  % (rn, info["n_args"], info["n_env"], info["dup_ep"][:5]))
            continue
        pairs, rep = parse_args_line(info["args"])
        if pairs is None or rep:
            hfail("G-ARGS %s: %s" % (rn, rep))
            continue
        sd_ = dict(pairs).get("save-directory", "")
        save_dirs.add(sd_)
        want = D.expected_args(arm, seed, sd_)
        if pairs != want:
            diff = [(f, v, dict(want).get(f)) for f, v in pairs if dict(want).get(f) != v]
            miss = [f for f, _v in want if f not in dict(pairs)]
            hfail("G-ARGS %s: %d flags differ from the registered line %s missing %s" % (rn, len(diff), diff[:4], miss[:4]))
            continue
        want_env = D.ENV_TEMPLATE.replace("<P>", os.path.join(sd_, D.probe_dir_name(arm, seed)))
        if info["env"] != want_env:
            hfail("G-ENV %s: ENV line is not the registered one (%r)" % (rn, (info["env"] or "")[:160]))
            continue
        if info["kinds_on"] or info["pt"]:
            hfail("G-KIND %s: prints a patch witness %s" % (rn, (info["kinds_on"] or ["PROBE_TENSOR"])[:2]))
            continue
        pp = probe_path(runsdir, arm, seed)
        ok, why, occ = probe_audit(read_probe(pp) if pp else None, D.M_OF_ARM[arm])
        if not ok:
            hfail("G-STRUCT %s: %s" % (rn, why))
            continue
        eps = info["eps"]
        per[(arm, seed)] = dict(level=mean([eps[e][1] for e in range(PL_LO, PL_HI + 1)]),
                                train=mean([eps[e][0] for e in range(PL_LO, PL_HI + 1)]), occ=occ, jid=jid)
        gpus[info["gpu"]] = gpus.get(info["gpu"], 0) + 1
    if len(save_dirs) > 1:
        hfail("G-ARGS: the runs name %d different save directories %s" % (len(save_dirs), sorted(save_dirs)))
    P("G-ONCE / G-ARGS / G-ENV / G-KIND / G-STRUCT  %d of %d complete runs pass (one ARGS / ENV line and each epoch once, "
      "20 flags, the registered ENV line, no patch witness, 10,000 probe records with n_beta == the arm's m)" % (len(per), len(runs)))
    P("G-HW (DISCLOSURE ONLY, never a gate): %s" % ", ".join("%s x%d" % (k, v) for k, v in sorted(gpus.items(), key=str)))
    n_corp = read_corpus_disclosure(csvpath)
    P("corpus: %s  (DISCLOSURE ONLY -- enters no bar, sigma, level, selection, reading or stamp)"
      % ("not readable" if n_corp is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (n_corp, PREFIX)))
    if harness:
        P("\nFINAL HARNESS-FAIL | %d harness failure(s); nothing is read" % len(harness))
        return 3
    if missing:
        P("\nFINAL INCOMPLETE | %d of %d runs missing or incomplete; nothing is read" % (len(missing), D.NJOBS))
        return 2

    A = {}
    ss_all, df_all, ss_ok, df_ok = 0.0, 0, 0.0, 0
    P("\nARMS  plateau5 = mean TEST, train5 = mean TRAIN (the SELECTOR), epochs %d-%d, from each run's OWN .out" % (PL_LO, PL_HI))
    P("  %-6s %-3s %-5s %-5s %-9s %2s %9s %7s %7s %9s  %-5s %8s %9s %11s %9s %9s" % (
        "arm", "cfg", "ms", "a0", "grain", "n", "plateau5", "sd", "range", "train5", "box", "rec_hi", "ceil_frac",
        "shrink_pl", "beta_pl", "beta_pk"))
    for a in D.ARMS:
        xs = [per[(a, s)]["level"] for s in D.SEEDS[a]]
        tr = [per[(a, s)]["train"] for s in D.SEEDS[a]]
        occs = [per[(a, s)]["occ"] for s in D.SEEDS[a]]
        m_ = mean(xs)
        ss_ = sum((x - m_) ** 2 for x in xs)
        ss_all += ss_
        df_all += len(xs) - 1
        rng_ = max(xs) - min(xs)
        if rng_ <= DIVERGED:
            ss_ok += ss_
            df_ok += len(xs) - 1
        cf = median([o["ceil_frac"] for o in occs])
        A[a] = dict(level=m_, train=mean(tr), range=rng_, n=len(xs), sd=sd(xs),
                    boxfree=all(o["boxfree"] for o in occs), ceil_frac=cf, ceil_bulk=cf >= CEIL_BULK,
                    shrink_pl=median([o["shrink_plateau"] for o in occs]),
                    beta_pl=median([o["beta_plateau"] for o in occs]), beta_pk=max(o["beta_peak"] for o in occs))
        P("  %-6s %-3s %-5s %-5s %-9s %2d %9.4f %7.3f %7.3f %9.4f  %-5s %8.4f %9.2e %11.4e %9.4f %9.4f" % (
            a, D.CONFIG_OF[a], D.MS[a], D.A0[a], D.SPEC[a], len(xs), m_, A[a]["sd"], rng_, A[a]["train"],
            "free" if A[a]["boxfree"] else "BOUND", max(o["rec_hi"] for o in occs), cf,
            A[a]["shrink_pl"], A[a]["beta_pl"], A[a]["beta_pk"]))
    P("  ceil_frac = window-median fraction of GROUPS at the ceiling (median over seeds); CEIL-BULK at >= %.2f.  box = crt1's"
      " rec-based stamp (any coordinate at an edge on >= 5 %% of records).  shrink = realised a*wd (caw2's definition);"
      " partitions APPROXIMATE.  DESCRIPTIVE.  A standard SGD recipe's steady-state shrink is ~%.0e (296.4)."
      % (CEIL_BULK, D.STANDARD_RECIPE_SHRINK))
    sig_all = math.sqrt(ss_all / df_all) if df_all else 0.0
    sig_ok = math.sqrt(ss_ok / df_ok) if df_ok else 0.0
    branch, tuned, oracle, cfg, K = decide(A, sig_ok, df_ok)

        SIGMA_FROZEN, SIGMA_FROZEN_DF, b, SIGMA_FROZEN * math.sqrt(2.0 / N_SEED), 2 * SIGMA_FROZEN * math.sqrt(2.0 / N_SEED)))
    P("  SIGMA_INBATCH         %.6f  (df %d, pooled within the non-diverged arms)   all arms %.6f (df %d)"
      % (sig_ok, df_ok, sig_all, df_all))
    P("  GATE  frozen rejected iff in-batch > SIGMA_FROZEN x sqrt(chi2_%.3f,%d / %d) = %.6f  -> %s" % (
        SIGMA_GATE_P, df_ok, df_ok, K["SIGMA_THR"], "PASS (frozen not rejected)" if K["SIGMA_OK"] else "REJECTED"))
    P("  DISCLOSURE: corpus floor %.6f (crt1's, df 190) and noise-floor demonstration %.6f (df 274) are read by nothing."
      % (SIGMA_CORPUS_FLOOR, NOISE_FLOOR_DEMO))

    P("\nSELECTION  (per grain over its non-diverged configs; exact ties to the earlier of %s)" % ",".join(D.CONFIG_IDS))
    P("  selection-bias bound B = EMAX5 x SIGMA_FROZEN / sqrt(3) = %.4f x %.6f / %.4f = %.6f pp (EXPECTED bias, max over 5)"

        P("  %-9s TRAIN-selected %-3s (train5 %s)   TEST-selected %-3s (plateau5 %s)   LOC %s" % (
            D.SPEC_OF_GRAIN[g], K["SEL_TUNED"][g] or "-", " ".join("%s %.4f" % (c, A[D.arm_of(g, c)]["train"]) for c in D.CONFIG_IDS),
            K["SEL_ORACLE"][g] or "-", " ".join("%s %.4f" % (c, A[D.arm_of(g, c)]["level"]) for c in D.CONFIG_IDS), K["LOC"][g]))
    P("\nCONTRASTS (IN BATCH; +/-2 SE is a HALF-WIDTH).  Registered edges: BEATS / PARTITION-ABOVE beyond +/-%.4f; a tie "
      "reading needs |min T| >= %.4f" % (EDGE_BEATS, EDGE_TIE))
    for lab, key in (("TUNED ", "T_TUNED"), ("ORACLE", "T_ORACLE")):
        t_ch, t_nd, se = K[key]
        if t_ch is None:
            P("  %s  unreadable" % lab)
            continue
        tm = min(t_ch, t_nd)
        P("  %s  T_ch = %+.4f pp (T-B %+.4f, T+B %+.4f)   T_nd = %+.4f pp (T-B %+.4f, T+B %+.4f)   SE %.6f   interval of "
          "min T [%+.4f, %+.4f]" % (lab, t_ch, t_ch - b, t_ch + b, t_nd, t_nd - b, t_nd + b, se, tm - b - RES * se, tm + b + RES * se))
    for c in D.CONFIG_IDS:
        t_ch, t_nd, se = K["T_" + c]
        d, _se = K["D_" + c]
        P("  %s (ms %s, a0 %s)  T_ch %+.4f (%+.2f SE)  T_nd %+.4f (%+.2f SE)  -> %-11s   D = ch - nd %+.4f (%+.2f SE) DESCRIPTIVE"
          % (c, D.MS_TOKEN[c], D.A0_TOKEN[c], t_ch, t_ch / se, t_nd, t_nd / se, cfg[c], d, d / se))
    cps = sigma_change_points(A, sig_ok, df_ok)
    P("  SENSITIVITY (DESCRIPTIVE, licenses nothing): had the frozen sigma been another literal in [0.05, 0.60], the TUNED "
      "reading would change at %s" % ("; ".join("%.4f (%s -> %s)" % x for x in cps) if cps else "no point"))

    P("\nREPLICATE (DESCRIPTIVE, CROSS-BATCH, NOT POOLED): A2 is crt1's selected config with fresh seeds")
    se_x = SIGMA_FROZEN * math.sqrt(1.0 / N_SEED + 1.0 / D.CRT1_N)
    for g in D.GRAIN_IDS:
        x = A[D.arm_of(g, "A2")]["level"]
        P("  %-9s crt2 A2 %.4f   crt1 A2 %.4f   diff %+.4f (%+.2f x cross-batch SE %.4f)" % (
            D.SPEC_OF_GRAIN[g], x, D.CRT1_A2[g], x - D.CRT1_A2[g], (x - D.CRT1_A2[g]) / se_x, se_x))
    a2c, a2n, _s = K["T_A2"]
    P("  T at A2: ch %+.4f (crt1 %+.4f), nd %+.4f (crt1 %+.4f)" % (a2c, D.CRT1_T_TUNED["ch"], a2n, D.CRT1_T_TUNED["nd"]))

    stamps = list(BOUNDS)
    stamps.insert(4, "SELBIAS-B=%.4f" % b)
    stamps.append("SIGMA-INBATCH=%.4f" % sig_ok)
    diffs = [g for g in D.GRAIN_IDS if abs(A[D.arm_of(g, "A2")]["level"] - D.CRT1_A2[g]) > MATCH]
    stamps += ["REPLICATE-MATCHES-CRT1"] if not diffs else ["REPLICATE-DIFFERS-%s" % g for g in diffs]
    for g in D.GRAIN_IDS:
        st_, so_ = K["SEL_TUNED"][g], K["SEL_ORACLE"][g]
        if st_ in D.GRID_EDGE:
            stamps.append("GRID-EDGE-%s-%s" % (g, st_))
        if st_ is not None and so_ is not None:
            stamps.append("SEL-AGREE-%s" % g if st_ == so_ else "SEL-DISAGREE-%s" % g)
        if st_ is not None and not A[D.arm_of(g, st_)]["boxfree"]:
            stamps.append("SELECTED-BOX-BOUND-%s" % g)
        if st_ is not None and A[D.arm_of(g, st_)]["ceil_bulk"]:
            stamps.append("SELECTED-CEIL-BULK-%s" % g)
    if oracle == "ORACLE-PARTITION-ABOVE-SCALAR":
        stamps.append("ORACLE-PARTITION-ABOVE")
    for a in D.ARMS:
        if not A[a]["boxfree"]:
            stamps.append("BOX-BOUND-%s" % a)
        if A[a]["ceil_bulk"]:
            stamps.append("CEIL-BULK-%s" % a)
        if A[a]["range"] > DIVERGED:
            stamps.append("DIVERGED-%s" % a)
    cfgword = "+".join("%s-%s" % (c, cfg[c]) for c in D.CONFIG_IDS)
    selword = "SEL-TRAIN:%s SEL-TEST:%s" % (",".join("%s=%s" % (g, K["SEL_TUNED"][g] or "-") for g in D.GRAIN_IDS),
                                            ",".join("%s=%s" % (g, K["SEL_ORACLE"][g] or "-") for g in D.GRAIN_IDS))
    locword = "%s:%s" % (K["LOCWORD"], ",".join("%s=%s" % (g, K["LOC"][g]) for g in D.GRAIN_IDS))
    P("\nLICENCE  %s%s" % (LICENCE[branch], LOC_CLAUSE[K["LOCWORD"]]))
    P("\nFINAL %s | %s | %s | %s | %s | %s | %s" % (branch, locword, tuned, oracle, cfgword, selword, " ".join(stamps)))
    return 0

RESULTS = {"PASS": 0, "FAIL": 0, "SKIP": 0}


def chk(cond, label):
    RESULTS["PASS" if cond else "FAIL"] += 1
    print("  %s %s" % ("PASS" if cond else "FAIL", label))
    return cond


def skip(label):
    RESULTS["SKIP"] += 1
    print("  SKIP %s" % label)


def _write_out(d, arm, seed, jid, level, train, save_dir, args_over=None, env_over=None, extra_line=None, done=True,
               n_ep=D.EPOCHS, gpu="NVIDIA L4, 23034 MiB", dup_epoch=None):
    rn = D.run_name(arm, seed)
    pairs = D.expected_args(arm, seed, save_dir)
    if args_over:
        pairs = args_over(pairs)
    args = "ARGS: " + " ".join("--%s %s" % kv for kv in pairs)
    env = D.ENV_TEMPLATE.replace("<P>", os.path.join(save_dir, D.probe_dir_name(arm, seed)))
    if env_over:
        env = env_over(env)
    L = ["NODE=[HOST_2f2f29d9e1] | JOB=%s %d | AUGMENT=1" % (rn, jid), args, env, gpu, ""]
    if extra_line:
        L.append(extra_line)
    for e in range(n_ep):
        te = level if e >= PL_LO - 5 else level * (1 - math.exp(-(e + 1) / 12.0))
        tr = train if e >= PL_LO - 5 else train * (1 - math.exp(-(e + 1) / 12.0))
        L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.99, tr), te))
        if dup_epoch is not None and e == dup_epoch:
            L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.99, tr), 10.0))
    L.append("%d  minutes" % 55)
    if done:
        L.append("RUN_DONE")
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(L) + "\n")


def _write_probe(pdir, m, lo=-8.0, hi=-5.0, n=D.N_RECORDS, nb=None, hi_frac=0.0, no_beta=False, n_hi=0, no_nhi=False):
    os.makedirs(pdir, exist_ok=True)
    with open(os.path.join(pdir, "probe.jsonl"), "w") as fh:
        for i in range(n):
            bmax = D.HI if (i >= n * (1 - hi_frac) or n_hi) else hi + 1e-4 * (i % 7)
            beta = "" if no_beta else ', "beta": [%.6f%s]' % (lo + 1.0, "" if m == 1 else ", %.6f" % (lo + 2.0))
            nh = "" if no_nhi else ', "n_at_hi": %d' % (n_hi if n_hi else (1 if bmax >= D.HI else 0))
            fh.write('{"step": %d, "beta_true_min": %.6f, "beta_true_max": %.6f, "n_beta": %d, "n_at_lo": 0%s%s}\n'
                     % (i * D.PROBE, lo - 1e-4 * (i % 5), bmax, nb if nb is not None else m, nh, beta))


JIT = (0.0, 0.03, -0.03)       # zero-mean over the 3 seeds


def _build(root, template, levels, trains=None, save_dir="/SAVE/runs/crt2", over=None, drop=(), extra=None,
           probe_over=None, prov_over=None, jit=None):
    os.makedirs(os.path.join(root, PREFIX), exist_ok=True)
    jid = 7200000
    for (arm, seed) in D.RUNS:
        jid += 1
        if (arm, seed) in drop:
            continue
        i = D.SEEDS[arm].index(seed)
        lv = levels[arm] + (JIT[i] if jit is None else jit(arm, i))
        tr = (trains or {}).get(arm, levels[arm] + 8.0)
        kw = dict((over or {}).get((arm, seed), {}))
        _write_out(os.path.join(root, PREFIX), arm, seed, jid, lv, tr, save_dir, **kw)
        dst = os.path.join(root, PREFIX, D.probe_dir_name(arm, seed))
        if probe_over and (arm, seed) in probe_over:
            _write_probe(dst, D.M_OF_ARM[arm], **probe_over[(arm, seed)])
        else:
            os.symlink(os.path.join(template, D.probe_dir_name(arm, seed)), dst)
    for f in (extra or []):
        f(os.path.join(root, PREFIX))
    kv = {"MODE": "submit", "SCORER_SHA256": file_sha(os.path.abspath(__file__)),
          "DESIGN_SHA256": file_sha(os.path.join(HERE, "crt2_design.py")), "HF_SHA256": D.HF_LIVE_SHA}
    kv.update(prov_over or {})
    open(os.path.join(root, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % x for x in kv.items()))
    return root

def _emax(k, steps=200000, lo=-10.0, hi=10.0):
    h = (hi - lo) / steps
    s = 0.0
    for i in range(steps):
        x = lo + (i + 0.5) * h
        phi = math.exp(-0.5 * x * x) / math.sqrt(2 * math.pi)
        Phi = 0.5 * (1 + math.erf(x / math.sqrt(2)))
        s += x * k * phi * Phi ** (k - 1)
    return s * h

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir required")
    raise SystemExit(score(a.runsdir, a.csv))
