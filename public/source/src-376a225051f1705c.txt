#!/bin/bash
# =============================================================================
# cVT7_grouphold.sh -- `cvt7`: DOES "THE CARRIER NEEDS A SMALL STEP SIZE" TRANSFER TO ResNet18_c100?
#   ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS {99,100,101}, ResNet18_c100 / CIFAR-100,
#   PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.  CORRECTIONS 243.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here touches the other account;
# every guard and scancel line is filtered on `cvt7-`.
#
# REGISTERED SCORER: analysis/cVT7_grouphold_score.py, committed BEFORE this batch is submitted (RULE 21).
# Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT7_grouphold_score.py <runsdir>
#
# NEW HARNESS CODE, ISOLATED AND OPT-IN.  The batch runs from $WS/harness_cvt7/cifar10 = the LIVE ResNet harness (the
# tree ciso1 / cdep1 / ciso2 ran from, unchanged) + PATCH_VOTEWEIGHT + PATCH_BETAHOLD + PATCH_GROUPHOLD in HF.py
# (patches/patch_grouphold.py), built and verified by bin/cVT7_stage_harness.sh, and the runner
# $WS/jobs/run_cifar_cvt7.sh (run_cifar.sh with the cd line changed).  All three switches off == bitwise the LIVE
# tree; guard 4f requires the inertness / bite log (tests/test_grouphold.py + tests/test_grouphold_realrun.py,
# bin/cVT7_realrun_inertness.sbatch) on THIS HF.py and THIS scorer.
#
# THE ARMS.  ResNet indices (62 tensors); guard 4 re-derives every one on the LIVE model by NAME, constructs each
# arm's HF with its GROUP_HOLD, checks the witnesses, and writes the manifest.
#   k01       scalar     GROUP_HOLD unset
#   ISO       ISO's      GROUP_HOLD unset     (sets:1-49,51-52,54-58,60-62/<the three carriers>, ciso1's string)
#   HOLDLOW   ISO's      GROUP_HOLD=layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight:floor
#   HOLDHIGH  ISO's      GROUP_HOLD=layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight:tri:8609
#   HOLDISO   ISO's      GROUP_HOLD=layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight:tri:5153
#
# RULE 20.  Every ARGS value is ONE shell token; every held arm's ARGS line is ISO's except the run name.
# GROUP_HOLD rides sbatch's --export list (value class [A-Za-z0-9_.:+], no comma) and CANNOT ride the runner's ENV
# line: the patch prints `GROUP_HOLD: off` / `GROUP_HOLD: on ...` at construction on EVERY run (with `BETA_HOLD: off`
# and `VOTE_W: off`), and those lines are what the separate ENV audit and the scorer's G-GHOLD gate read.
# analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  ciso1 / cdep1: median ~2,600 s per 100-epoch ResNet18_c100 run off [HOST_3c1b003035], ~5,500 s on [HOST_3c1b003035] (sacct);
# mean ~0.86 GPU-h.  15 jobs ~= 12.9 GPU-h expected (24 if every run lands on [HOST_3c1b003035]);
# HARD BOUND = WALL x 15 = 45 GPU-h (WALL 03:00:00 = x2 the slowest ciso1 / cdep1 run, 5,701 s).
#
# USAGE   bash bin/cVT7_grouphold.sh               # dry run: every guard, every composed line printed
#         bash bin/cVT7_grouphold.sh --dry-run     #   submits nothing
#         bash bin/cVT7_grouphold.sh --submit      # the ONLY submitting form
# ENV     CVT7_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# the control arms must NOT inherit a GROUP_HOLD (or a BETA_HOLD / VOTE_W) through --export=ALL
unset VOTE_W BETA_HOLD GROUP_HOLD
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cvt7/cifar10
RUNNER=$WS/jobs/run_cifar_cvt7.sh
SAVE=$WS/runs/cvt7

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="99 100 101"
MST=1e-3
A0=1e-6
NTENS=62
TOTPAR=11220132
G_K01='scalar'
G_ISO='sets:1-49,51-52,54-58,60-62/layer4.0.bn2.weight,layer4.0.shortcut.1.weight,layer4.1.bn2.weight'
CARR='layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight'
H_HOLDLOW="$CARR:floor"
H_HOLDHIGH="$CARR:tri:8609"
H_HOLDISO="$CARR:tri:5153"
ARMS="k01 ISO HOLDLOW HOLDHIGH HOLDISO"
NJOBS=15
WALL=03:00:00
LIVE_BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_VW_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_PRE_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
HF_POST_SHA=2396f2be8102724c120ea5271d2d84acf3835ffb9a5c7947fc282121cf2a588f
RUNNER_SHA=805267e4cbcc64172d79824d553a2577ef7cdd55fb8f5e7731c0ea3251a04c56
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT7_grouphold_score.py"
SELF="$REPO/bin/cVT7_grouphold.sh"
DERIV="$REPO/analysis/cvt7_registration_derivations.py"
DERIV4="$REPO/analysis/cvt4_registration_derivations.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cVT7_realrun_inertness.sbatch"
TESTG="$REPO/tests/test_grouphold.py"
TESTGR="$REPO/tests/test_grouphold_realrun.py"
STAGE7="$REPO/bin/cVT7_stage_harness.sh"
PATCHG="$REPO/patches/patch_grouphold.py"
PATCHB="$REPO/patches/patch_betahold.py"
PATCHV="$REPO/patches/patch_voteweight.py"
TESTB="$REPO/tests/test_betahold.py"
TESTV="$REPO/tests/test_voteweight.py"
TESTR="$REPO/tests/test_voteweight_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFGH="$HARN/Optimizers/HF.py.pre_grouphold"
HFBH="$HARN/Optimizers/HF.py.pre_betahold"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"
ILOG="$SAVE/inertness_grouphold.log"

echo "=============================================================="
echo " cvt7 -- does 'the carrier needs a small step size' transfer to ResNet18_c100?  (PATCH_GROUPHOLD)"
echo " $DSET / $NET (ResNet indices)   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01       $G_K01      GROUP_HOLD unset"
echo "   ISO       $G_ISO      GROUP_HOLD unset"
echo "   HOLDLOW   ISO's      GROUP_HOLD=$H_HOLDLOW"
echo "   HOLDHIGH  ISO's      GROUP_HOLD=$H_HOLDHIGH"
echo "   HOLDISO   ISO's      GROUP_HOLD=$H_HOLDISO"
echo " 5 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (the LIVE tree + PATCH_VOTEWEIGHT + PATCH_BETAHOLD + PATCH_GROUPHOLD)   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " CO-PRIMARY: P_HIGH = ISO - HOLDHIGH; P_LOW = HOLDLOW - ISO"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT7_grouphold_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${GROUP_HOLD+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD and GROUP_HOLD are UNSET in the launcher's environment (control arms cannot inherit them)" \
  || guard_fail "guard 0b: VOTE_W, BETA_HOLD or GROUP_HOLD is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$SELF" "$DERIV" "$DERIV4" "$EXCL" "$EXCLT" "$RRSB" "$TESTG" "$TESTGR" "$STAGE7" "$PATCHG" "$PATCHB" "$PATCHV" \
         "$TESTB" "$TESTV" "$TESTR" "$TESTT" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$HFGH" "$HFBH" "$HFVW" "$HFPRE" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cVT7_grouphold_score.py bin/cVT7_grouphold.sh analysis/cvt7_registration_derivations.py bin/cVT7_realrun_inertness.sbatch tests/test_grouphold.py tests/test_grouphold_realrun.py patches/patch_grouphold.py bin/cVT7_stage_harness.sh"
REUSEF="analysis/cvt4_registration_derivations.py analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv patches/patch_betahold.py patches/patch_voteweight.py tests/test_betahold.py tests/test_voteweight.py tests/test_voteweight_realrun.py tests/test_probe_tensor.py bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the eight registration files and eleven reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT7_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT7_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT7_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT7_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING O2, RULE P, the real-record null and the invocation ----
SELFT=/tmp/cvt7_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
grep -qE "^  PASS O2 TRANSFERS +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt7's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT7_grouphold_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 99, 100 or 101" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cIS1's" "$SELFT" && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cDP1's" "$SELFT" \
  && grep -q "^  PASS ISO's spec is BYTE-IDENTICAL to cIS2's" "$SELFT" \
  && echo "guard 1c5: ISO is byte-identical to ciso1's, cdep1's and ciso2's ISO" || guard_fail "guard 1c5: ISO not proved byte-identical"
grep -q "^  PASS B RULE P re-run on THIS host's landed records == P_HIGH 8609, P_ISO 5153" "$SELFT" \
  && echo "guard 1c6: RULE P re-derived P_HIGH 8609 and P_ISO 5153 from the landed records ON THIS HOST" \
  || guard_fail "guard 1c6: RULE P was not re-derived on this host"
grep -q "^  PASS B G-BITE's Lion half PASSES on all 18 landed" "$SELFT" && grep -q "^  PASS B THE NULL ON REAL RECORDS" "$SELFT" \
  && grep -q "^  PASS B non-vacuity" "$SELFT" \
  && echo "guard 1c7: G-BITE passes 18 landed controls and FAILS all 27 real-record nulls on this host; non-vacuity re-derived" \
  || guard_fail "guard 1c7: the real-record G-BITE checks did not pass"
grep -q "^  PASS SIGMA_R18ALL: the FROZEN literal 0.69444284693959901 IS the FILTERED value" "$SELFT" \
  && echo "guard 1c8: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value on the registration corpus" \
  || guard_fail "guard 1c8: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E BROKEN PATCH: every held arm UNHELD" "$SELFT" && grep -q "^  PASS E HOLD ONLY AT PROBE STEPS" "$SELFT" \
  && grep -q "^  PASS E WRONG P" "$SELFT" && grep -q "^  PASS E COMPLEMENT TOUCHED" "$SELFT" \
  && echo "guard 1c9: the broken-patch null, a probe-step-only hold, a wrong P and a touched complement reach PATCH-NOT-VERIFIED" \
  || guard_fail "guard 1c9: G-BITE break cases did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

# ---- guard 1d: the derivation reproduces RULE P, the triangle property and the floor on THIS host ----------
DOUT=/tmp/cvt7_deriv.$$
python3 "$DERIV" "$WS/runs" "$CSVP" >"$DOUT" 2>&1 \
  && grep -q "^  REGISTERED: P_HIGH 8609 (HOLDHIGH), P_ISO 5153 (HOLDISO)." "$DOUT" \
  && grep -q "^  MAX-RATE-TRIANGLE PROPERTY at this cell (both sources <= 0.15 log units): HOLDS" "$DOUT" \
  && grep -q "SIGMA_R18ALL  FILTERED 0.694442846939599 (df 209" "$DOUT" \
  && echo "guard 1d: analysis/cvt7_registration_derivations.py reproduces RULE P, the triangle property and the filtered floor on this host (output sha $(sha "$DOUT"))" \
  || { guard_fail "guard 1d: the derivation did not reproduce RULE P / the triangle / the floor"; tail -5 "$DOUT"; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt7-' "$CSVP" && guard_fail "guard 2: cvt7-* already in the CSV" || echo "guard 2: no cvt7-* row in the corpus"
python3 - "$CSVP" "$SEEDS" <<'PYEOF' || guard_fail "guard 2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
raise SystemExit(1 if hit else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt7-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt7-*.out on disk" || guard_fail "guard 2b: $NOUT cvt7-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (99|100|101)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 99/100/101" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 99/100/101"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt7-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt7-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt7-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt7-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt7-* in squeue" || guard_fail "guard 2f: $NQ cvt7-* already queued"
fi

# ---- guard 3/4a: the cvt7 tree consistent (the LIVE base unchanged) and sees CIFAR-100 -----------
bash "$STAGE7" >/tmp/cvt7_stage7.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT7_stage_harness.sh (verify mode) VERIFIES cvt7's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt7_stage7.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt7's tree is NOT consistent"; tail -20 /tmp/cvt7_stage7.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "$LIVE_BN_SHA" ] && [ "$(sha "$LIVE/build_network.py")" = "$LIVE_BN_SHA" ] \
  && echo "guard 3b: build_network.py == the LIVE one c7998883...77d (no PlainNet patch)" \
  || guard_fail "guard 3b: build_network.py is not the LIVE one (or the LIVE one moved)"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFGH")" = "$HF_PRE_SHA" ] && [ "$(sha "$HFBH")" = "$HF_VW_SHA" ] \
  && [ "$(sha "$HFVW")" = "$HF_LIVE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA 2396f2be...; .pre_grouphold == cvt4's 84b345ad...; .pre_betahold == cvt1's 3f2b98e1...; .pre_voteweight == the LIVE HF.py 4732b74a...; .pre_probe_tensor == live" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or the live HF.py moved)"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == 805267e4... (run_cifar.sh with the cd line changed)" \
  || guard_fail "guard 3e: $RUNNER is not the pinned cvt7 runner"
grep -q PATCH_GROUPHOLD "$LIVE/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE HF.py carries PATCH_GROUPHOLD" \
  || echo "guard 3d: the LIVE HF.py carries no PATCH_GROUPHOLD"

# ---- guard 4b/4f: the optimiser's patches, proved inert; the new patch proved inert and biting ----
python3 "$TESTT" --pre "$HFPRE" --post "$HFVW" --structural-only >/tmp/cvt7_struct.$$ 2>&1 \
   && echo "guard 4b: PATCH_PROBE_TENSOR structural inertness R0 PASSES on HF.py.pre_voteweight (= live)" \
   || { guard_fail "guard 4b: PATCH_PROBE_TENSOR structural R0 FAILED"; tail -8 /tmp/cvt7_struct.$$; }
python3 "$TESTG" --pre "$HFGH" --post "$HFH" --structural-only >/tmp/cvt7_gh0.$$ 2>&1 \
   && echo "guard 4b2: PATCH_GROUPHOLD structural G0 PASSES (deleting the four regions == cvt4's HF.py, byte for byte)" \
   || { guard_fail "guard 4b2: PATCH_GROUPHOLD structural G0 FAILED"; tail -8 /tmp/cvt7_gh0.$$; }
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH"); SCSHA=$(sha "$SC")
if [ -f "$ILOG" ] && grep -q "^TEST_GROUPHOLD_RC 0$" "$ILOG" && grep -q "^REALRUN_RC 0$" "$ILOG" \
   && [ "$(grep -c '^ALL PASS$' "$ILOG")" = 2 ] && [ "$(grep -c '^  FAIL' "$ILOG")" = 0 ] \
   && grep -q "^TREE_POST $HARN HF $HFSHA build_network $BNSHA " "$ILOG" \
   && grep -q "^TREE_PRE $LIVE HF $HF_LIVE_SHA build_network $LIVE_BN_SHA " "$ILOG" && grep -q "^SCORER_SHA256 $SCSHA$" "$ILOG" \
   && grep -q "^DRIVER_SHA256 $(sha "$TESTGR")$" "$ILOG" \
   && grep -q "^$(sha "$TESTG")  tests/test_grouphold.py$" "$ILOG" && grep -q "^$(sha "$PATCHG")  patches/patch_grouphold.py$" "$ILOG" \
   && grep -q "PASS RR0 live k01 vs itself: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 ISO GROUP_HOLD unset: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS RR1 k01 GROUP_HOLD empty: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && grep -q "PASS this child == tests/test_voteweight_realrun.py's registered child" "$ILOG" \
   && grep -q "PASS RR2 HOLDHIGH  beta\[1\] after EVERY one" "$ILOG" && grep -q "PASS RR3 HOLDLOW's final parameters differ" "$ILOG" \
   && [ "$(grep -c "PASS RR2 HOLD[A-Z]* *the scorer's own bite_check on the 30 real records" "$ILOG")" = 3 ] \
   && grep -q "PASS G4 HOLDHIGH  the WHOLE momentum_meta vector" "$ILOG"; then
  echo "guard 4f: the inertness / bite log (tests/test_grouphold.py + tests/test_grouphold_realrun.py) ALL PASS on THIS HF, THIS scorer: $(grep -c '^  PASS' "$ILOG") PASS / $(grep -c '^  FAIL' "$ILOG") FAIL (sha $(sha "$ILOG" | cut -c1-8))"
else
  guard_fail "guard 4f: no ALL PASS inertness log for HF $HFSHA and scorer $SCSHA at $ILOG -- sbatch $RRSB, then copy its .out there"
fi
