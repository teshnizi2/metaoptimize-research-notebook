#!/bin/bash
# =============================================================================
# cGW1_auditwd.sh -- `cgw1`: DOES THE COUNT-MATCHED PARTITION AUDIT SURVIVE AT STANDARD WEIGHT DECAY?
#   ONE SUBMISSION, 40 JOBS (12 arms; seeds {152,153,154} at W1 and W2, {152..155} at W4), 100 EPOCHS,
#   ResNet18 / CIFAR-10 -- the audit's core SGDm cell VERBATIM except `--weight-decay-base` -- PROBE=5,
#   --constraint=L4 ON EVERY JOB.  CORRECTIONS 291 (gate G3 of docs/ICML-PLAN.md).
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Every guard and scancel line is filtered on `cgw1-`.
#
# REGISTERED SCORER: analysis/cGW1_auditwd_score.py (+ analysis/cgw1_design.py), committed and pushed BEFORE this
# batch is submitted (RULE 21).   DOCUMENTED INVOCATION:  python3 analysis/cGW1_auditwd_score.py <runsdir>
#
# THE ARMS (no patch, no hold variable, no mask on any arm; the grain and the decay are the only things that vary)
#   rung W1  wd 0.1   chW1 chunk777 | ndW1 nodewise | k01W1 scalar | kLW1 layerwise   the audit's own value, IN BATCH
#   rung W2  wd 1e-2  chW2          | ndW2          | k01W2        | kLW2             one decade down
#   rung W4  wd 5e-4  chW4          | ndW4          | k01W4        | kLW4             standard CIFAR; THE PRIMARY RUNG
#
# RULE 20.  Every ARGS value is ONE shell token.  `weight-decay-base` and `stepsize-groups` VARY BY ARM and are
# checked per composed line against the arm's OWN pair.  A non-standard weight decay is an ARGS-VALUE deviation: the
# 28 runs at wd != 0.1 go to results/CORPUS-EXCLUSIONS.tsv at landing as ARGS_WD_BASE rows (kind exists since 263).
# corpus_exclusions.py is NOT edited.
#
# COST.  Expected 29.3 GPU-h (per-grain L4 minutes at this cell from the corpus: chunk777 52, nodewise 44, scalar 42,
# layerwise 34, +1 each).  HARD BOUND = WALL x 40 = 80 GPU-h.
#
# USAGE   bash bin/cGW1_auditwd.sh [--dry-run]   |   bash bin/cGW1_auditwd.sh --submit   (the ONLY submitting form)
# ENV     on a host with no git repo (the cluster_b stage copy), --submit requires BOTH
#           CGW1_REGISTERED_COMMIT=<40-hex sha of the PUSHED registration commit>
#           CGW1_SCORER_SHA256=<sha256 of analysis/cGW1_auditwd_score.py AT THAT COMMIT, computed from origin>
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
unset VOTE_W BETA_HOLD COMP_HOLD WINDOW_HOLD GROUP_HOLD REST_HOLD DECAY_MASK PROBE_TENSOR
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
B=cgw1
HARN=$WS/harness_cgw1/cifar10
RUNNER=$WS/jobs/run_cifar_cgw1.sh
SAVE=$WS/runs/cgw1
CWD1_HF=$WS/harness_cwd1/cifar10/Optimizers/HF.py
CWD1_HF_SHA=94aedc33046afb12782ffb7f2eb729bafc7fcba5f42227fec373fa67b6f58ebf
BITE5=$WS/runs/cwd5/bite_decaymask_cwd5.log

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true
export PYTHONDONTWRITEBYTECODE=1

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=5
DSET=CIFAR10
NET=ResNet18
MST=1e-4
A0=1e-3
NTENS=62
TOTPAR=11173962
WD_W1=0.1
WD_W2=1e-2
WD_W4=5e-4
SEEDS_W1="152 153 154"
SEEDS_W2="152 153 154"
SEEDS_W4="152 153 154 155"
GRAINS="ch nd k01 kL"
NJOBS=40
WALL=02:00:00
GPU_CONSTRAINT=L4
HF_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
TRAIN_SHA=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
RUNNER_SRC_SHA=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL" "gpu-l4-24g")

SC="$REPO/analysis/cGW1_auditwd_score.py"
DES="$REPO/analysis/cgw1_design.py"
LIVEC="$REPO/analysis/cgw1_live_check.py"
SELF="$REPO/bin/cGW1_auditwd.sh"
STAGE="$REPO/bin/cGW1_stage_harness.sh"
R20="$REPO/bin/cGW1_rule20.sh"
R20E="$REPO/analysis/cgw1_rule20_envaudit.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"

echo "=============================================================="
echo " cgw1 -- DOES THE COUNT-MATCHED PARTITION AUDIT SURVIVE AT STANDARD WEIGHT DECAY?  (CORRECTIONS 291; G3)"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  clamp $CLIP_C   ms=$MST alpha0=$A0  $EPOCHS epochs"
echo "   rung W1  wd $WD_W1   seeds $SEEDS_W1       ch chunk777 | nd nodewise | k01 scalar | kL layerwise   (anchor)"
echo "   rung W2  wd $WD_W2  seeds $SEEDS_W2       the same four grains"
echo "   rung W4  wd $WD_W4  seeds $SEEDS_W4   the same four grains   (PRIMARY)"
echo " 12 arms, $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN   runner $RUNNER   partitions $PARTS   constraint $GPU_CONSTRAINT   walltime $WALL"
echo " DOCUMENTED INVOCATION: python3 analysis/cGW1_auditwd_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${GROUP_HOLD+x}" ] && [ -z "${REST_HOLD+x}" ] && [ -z "${DECAY_MASK+x}" ] && [ -z "${PROBE_TENSOR+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD, GROUP_HOLD, REST_HOLD, DECAY_MASK and PROBE_TENSOR are UNSET in the launcher's environment" \
  || guard_fail "guard 0b: a hold / mask / probe-tensor variable is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" || guard_fail "guard 0: no CSV at $CSVP"

for f in "$SC" "$DES" "$LIVEC" "$SELF" "$STAGE" "$R20" "$R20E" "$EXCL" "$EXCLT" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$CWD1_HF" "$BITE5"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

NEWF="analysis/cGW1_auditwd_score.py analysis/cgw1_design.py analysis/cgw1_live_check.py analysis/cgw1_rule20_envaudit.py bin/cGW1_auditwd.sh bin/cGW1_stage_harness.sh bin/cGW1_rule20.sh"
REUSEF="analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  PUSHED=$(git -C "$REPO" branch -r --contains HEAD 2>/dev/null | grep -c 'origin/master')
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ] && [ "${PUSHED:-0}" -ge 1 ]; then
    echo "guard 1b: the registration files are TRACKED, COMMITTED and PUSHED (HEAD $(git -C "$REPO" rev-parse --short HEAD) is on origin/master) (RULE 21)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: not committed and pushed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}; on origin/master: ${PUSHED:-0}"
  else
    echo "guard 1b: NOTE (dry run) -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}; on origin/master: ${PUSHED:-0}; --submit would refuse"
  fi
else
  if [ -n "${CGW1_REGISTERED_COMMIT:-}" ] && [ -n "${CGW1_SCORER_SHA256:-}" ]; then
    if printf '%s' "$CGW1_REGISTERED_COMMIT" | grep -qE '^[0-9a-f]{40}$' && [ "$CGW1_SCORER_SHA256" = "$(sha "$SC")" ]; then
      echo "guard 1b: no git repo on this host; the operator declares commit $CGW1_REGISTERED_COMMIT (40 hex) and the staged"
      echo "          scorer's sha256 EQUALS the declared sha of that commit's scorer ($(echo "$CGW1_SCORER_SHA256" | cut -c1-16))"
    else
      guard_fail "guard 1b: CGW1_REGISTERED_COMMIT is not 40 hex, or the staged scorer $(sha "$SC" | cut -c1-16) != CGW1_SCORER_SHA256"
    fi
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CGW1_REGISTERED_COMMIT / CGW1_SCORER_SHA256 are unset"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CGW1_REGISTERED_COMMIT and CGW1_SCORER_SHA256"
  fi
fi

SELFT=/tmp/cgw1_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|Error' "$SELFT" | head -20; tail -5 "$SELFT"; }
[ "$(grep -c '^  SKIP' "$SELFT")" = 0 ] && echo "guard 1c1: no SKIP (the corpus and the landed gn1 probe records were found on this host)" || guard_fail "guard 1c1: the selftest SKIPPED a section"
grep -q "^  PASS B SIGMA_PRIOR: the FROZEN literal 0.1789141365540324 IS the FILTERED value" "$SELFT" \
  && echo "guard 1c2: the frozen floor IS the corpus_exclusions.filter_rows value on THIS host's corpus" \
  || guard_fail "guard 1c2: the frozen floor was not shown to be the filtered value"
grep -q "^  PASS E O2: FINAL byte-identical" "$SELFT" && grep -q "^  PASS E the DOCUMENTED one-argument invocation" "$SELFT" \
  && echo "guard 1c3: O2 invariance ran against the real corpus + invented rows; the documented invocation reached FINAL" \
  || guard_fail "guard 1c3: the O2 invariance / documented-invocation checks did not pass"
grep -q "^  PASS F landed gn1 bn-ch s0" "$SELFT" && grep -q "^  PASS G every predicted W4 level" "$SELFT" \
  && echo "guard 1c4: the probe audit passed on REAL landed records of this cell; the 164.6 floor gate passed" \
  || guard_fail "guard 1c4: real-record probe audit or floor gate did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"

grep -q '^cgw1-' "$CSVP" && guard_fail "guard 2: cgw1-* already in the CSV" || echo "guard 2: no cgw1-* row in the corpus"
NOUT=$(ls "$WS/runs"/cgw1-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cgw1-*.out on disk" || guard_fail "guard 2b: $NOUT cgw1-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (15[2-9])( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 152..159 (the block is fresh)" \
                      || guard_fail "guard 2d: $NSEED .out files already ran a seed in 152..159"
python3 - "$CSVP" <<'PYEOF' || guard_fail "guard 2d2: a corpus row already carries a seed in 152..159"
import csv, sys
n = sum(1 for r in csv.DictReader(open(sys.argv[1])) if r.get("seed", "").strip() in [str(s) for s in range(152, 160)])
print("guard 2d2: %d corpus rows carry a seed in 152..159" % n)
raise SystemExit(0 if n == 0 else 1)
PYEOF
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cgw1-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cgw1-* jobs" || guard_fail "guard 2e: sacct shows $NS cgw1-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cgw1-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cgw1-* in squeue" || guard_fail "guard 2f: $NQ cgw1-* already queued"
fi

# ---- guard 3: the isolated tree ----------------------------------------------------------------------------------
bash "$STAGE" >/tmp/cgw1_stage.$$ 2>&1 \
  && { echo "guard 3: cGW1_stage_harness.sh (verify mode) VERIFIES the cgw1 tree:"; grep -E '^V-' /tmp/cgw1_stage.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 3: the cgw1 tree is NOT consistent"; tail -20 /tmp/cgw1_stage.$$; }
[ "$(sha "$HARN/Optimizers/HF.py")" = "$HF_SHA" ] && [ "$(sha "$HARN/train.py")" = "$TRAIN_SHA" ] && [ "$(sha "$HARN/build_network.py")" = "$BN_SHA" ] \
  && echo "guard 3b: tree HF.py / train.py / build_network.py == the pinned shas" || guard_fail "guard 3b: a pinned tree file moved"
grep -q -- '--weight-decay-base' "$HARN/train.py" \
  && echo "guard 3f: train.py declares --weight-decay-base (the axis is an EXISTING registered CLI flag; no new code)" \
  || guard_fail "guard 3f: train.py has no --weight-decay-base flag"

# ---- guard 4f: the premise, inherited by code identity from cwd5's landed bite log --------------------------------
if [ "$(sha "$CWD1_HF")" = "$CWD1_HF_SHA" ] && [ -f "$BITE5" ] && grep -q "^REALRUN_RC 0$" "$BITE5" \
   && [ "$(grep -c '^  FAIL' "$BITE5")" = 0 ] && grep -q "^TREE_POST $WS/harness_cwd1/cifar10 HF $CWD1_HF_SHA " "$BITE5" \
   && [ "$(grep -c "PASS RW2 W. wd .*: w', h', m' BITWISE the coupled formula" "$BITE5")" = 4 ] \
   && grep -q "PASS RW2 W2 wd 1e-2 : w', h', m' BITWISE" "$BITE5" && grep -q "PASS RW2 W4 wd 5e-4 : w', h', m' BITWISE" "$BITE5" \
   && grep -q "PASS RW2 W1 wd 0.1  : w', h', m' BITWISE" "$BITE5"; then
  echo "guard 4f: cwd5's landed bite log (sha $(sha "$BITE5" | cut -c1-8)) on harness_cwd1 HF $CWD1_HF_SHA: $(grep -c '^  PASS' "$BITE5") PASS / 0 FAIL,"
  echo "          RW2 BITWISE at 0.1, 1e-2 and 5e-4 (this ladder's three tokens); guard 4e below ties that HF's"
  echo "          SGDm_base_update to this tree's by source text"
else
  guard_fail "guard 4f: cwd5's bite log / harness_cwd1 HF do not vouch for the ladder's premise"
fi

# ---- guard 4c': the launcher's literals == the design module's == the stage script's -------------------------------
python3 - "$DES" "$STAGE" "$WD_W1" "$WD_W2" "$WD_W4" "$SEEDS_W1" "$SEEDS_W2" "$SEEDS_W4" "$GRAINS" "$NET" "$NTENS" "$TOTPAR" \
          "$HF_SHA" "$TRAIN_SHA" "$BN_SHA" "$RUNNER_SRC_SHA" "$GPU_CONSTRAINT" "$NJOBS" "$MST" "$A0" "$CLIP_C" "$WALL" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage literals disagree"
import sys, re, importlib.util
spm = importlib.util.spec_from_file_location("cgw1_design", sys.argv[1]); Dm = importlib.util.module_from_spec(spm); spm.loader.exec_module(Dm)
(stage, w1, w2, w4, s1, s2, s4, grains, net, ntens, totpar, hf, tr, bn, rsrc, gpu, nj, mst, a0, clip, wall) = sys.argv[2:23]
bad = []
if [Dm.WD_TOKEN[r] for r in Dm.RUNG_IDS] != [w1, w2, w4]: bad.append("WD_LADDER")
if [tuple(int(x) for x in s.split()) for s in (s1, s2, s4)] != [Dm.SEEDS_OF_RUNG[r] for r in Dm.RUNG_IDS]: bad.append("SEEDS")
if tuple(grains.split()) != Dm.GRAIN_IDS: bad.append("GRAINS")
if (net, int(ntens), int(totpar)) != (Dm.NET, Dm.NTENS, Dm.TOTPAR): bad.append("NET/NTENS/TOTPAR")
if (hf, tr, bn, rsrc) != (Dm.HF_LIVE_SHA, Dm.TRAIN_SHA, Dm.BN_SHA, Dm.RUNNER_SRC_SHA): bad.append("SHAS")
if gpu != Dm.GPU_CONSTRAINT or int(nj) != Dm.NJOBS or wall != Dm.WALL: bad.append("GPU/NJOBS/WALL")
if (mst, a0, clip) != (Dm.MST, Dm.A0, Dm.CLIP): bad.append("MST/A0/CLIP")
st = open(stage).read()
for name, v in (("HF_SHA", Dm.HF_LIVE_SHA), ("TRAIN_SHA", Dm.TRAIN_SHA), ("BN_SHA", Dm.BN_SHA), ("BUILDOPT_SHA", Dm.BUILDOPT_SHA),
                ("LOADDATA_SHA", Dm.LOADDATA_SHA), ("TINDATA_SHA", Dm.TINDATA_SHA), ("RUNNER_SRC_SHA", Dm.RUNNER_SRC_SHA)):
    m = re.search(r"^%s=([0-9a-f]{64})$" % name, st, re.M)
    if not m or m.group(1) != v: bad.append("stage " + name)
if any(Dm.SEEDS[g + r] != Dm.SEEDS_OF_RUNG[r] for r in Dm.RUNG_IDS for g in Dm.GRAIN_IDS): bad.append("GRAIN-SEEDS")
if Dm.N_EXCLUSION_ROWS != 28: bad.append("EXCLUSION-COUNT")
if bad:
    print("!!! guard 4c': disagreement on %s" % sorted(bad)); raise SystemExit(1)
print("guard 4c': the three weight-decay tokens, the seeds per rung, the four grains, NET/NTENS/TOTPAR, ms/alpha0/clip,")
print("           the tree / runner shas (launcher, design AND stage script), the GPU constraint, NJOBS and WALL agree;")
print("           every grain at a rung runs that rung's seeds; 28 of 40 runs deviate from wd 0.1 and owe an exclusion row")
PYEOF

# ---- guard 4: THE LIVE-MODEL MANIFEST (CPU construction only; no forward pass, no GPU) ------------------------------
mkdir -p "$SAVE"
MTMP=/tmp/cgw1_manifest.$$
python3 "$LIVEC" "$HARN" "$MTMP" "$CWD1_HF" 2>&1 | sed 's/^/  /'
[ -s "$MTMP" ] || guard_fail "guard 4: the live-model check FAILED (spec / m / wd / base_update / manifest / SGDm source)"
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"
if [ -s "$MTMP" ]; then
  if [ -f "$MANIFEST" ] && ! cmp -s "$MTMP" "$MANIFEST"; then
    guard_fail "guard 4b3: an existing $MANIFEST differs from the live one -- refusing to overwrite"
  else
    cp "$MTMP" "$MANIFEST"
    echo "guard 4b3: live manifest ($(wc -c < "$MANIFEST") bytes, sha $(sha "$MANIFEST" | cut -c1-16)) -> $MANIFEST"
  fi
fi

PROV="$SAVE/PROVENANCE.txt"; [ "$SUBMIT" = 1 ] || PROV="$SAVE/PROVENANCE.dryrun.txt"
{ echo "MODE $([ "$SUBMIT" = 1 ] && echo submit || echo dry-run)"
  echo "SCORER_SHA256 $(sha "$SC")"
  echo "DESIGN_SHA256 $(sha "$DES")"
  echo "LIVECHECK_SHA256 $(sha "$LIVEC")"
  echo "LAUNCHER_SHA256 $(sha "$SELF")"
  echo "STAGE_SCRIPT_SHA256 $(sha "$STAGE")"
  echo "HF_SHA256 $(sha "$HARN/Optimizers/HF.py")"
  echo "TRAIN_SHA256 $(sha "$HARN/train.py")"
  echo "BUILD_NETWORK_SHA256 $(sha "$HARN/build_network.py")"
  echo "BUILD_OPTIMIZER_SHA256 $(sha "$HARN/Optimizers/build_optimizer.py")"
  echo "LOAD_DATA_SHA256 $(sha "$HARN/load_data.py")"
  echo "RUNNER_SHA256 $(sha "$RUNNER")"
  echo "STAGE_MANIFEST_SHA256 $(sha "$WS/harness_cgw1/STAGE-MANIFEST.txt")"
  echo "CWD1_HF_SHA256 $(sha "$CWD1_HF")"
  echo "CWD5_BITE_LOG_SHA256 $(sha "$BITE5")"
  echo "REGISTERED_COMMIT ${CGW1_REGISTERED_COMMIT:-$(git -C "$REPO" rev-parse HEAD 2>/dev/null || echo unknown)}"
  echo "HARNESS_DIR $HARN"
  echo "CORPUS_SHA256 $(sha "$CSVP")"
  echo "CORPUS_EXCLUSIONS_SHA256 $(sha "$EXCLT")"
  echo "UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "WD_LADDER W1=$WD_W1 W2=$WD_W2 W4=$WD_W4"
  echo "SEEDS W1=$SEEDS_W1 W2=$SEEDS_W2 W4=$SEEDS_W4"
  echo "GRAINS ch=chunk777 nd=nodewise k01=scalar kL=layerwise"
  echo "EPOCHS $EPOCHS PROBE $PROBE"
  echo "GPU_CONSTRAINT $GPU_CONSTRAINT PARTITIONS $PARTS WALL $WALL"
  echo "VOTE_W BETA_HOLD GROUP_HOLD REST_HOLD DECAY_MASK PROBE_TENSOR unset on every arm"; } > "$PROV"
echo "guard 4g: provenance written to $PROV"

guard_parts_for_wall "$PARTS" "$WALL" || guard_fail "guard 5a: PARTS/WALL disagree"
case ",$PARTS," in *,gpu-mig-40g,*|*,gpu-a100-80g,*|*,gpu-2080ti-11g,*) guard_fail "guard 5b: a partition without L4 nodes is in $PARTS" ;; *) echo "guard 5b: partitions $PARTS all carry L4 nodes; --constraint=$GPU_CONSTRAINT" ;; esac
FREE=$(df -BG "$WS" 2>/dev/null | tail -1 | awk '{print $4}' | tr -d 'G')
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: ${FREE:-?} GB free on \$WS (40 probe dirs at PROBE=5 need ~3 GB: gn1's landed ch/nd dirs are 67 / 70 MB); $PEND jobs pending, adding $NJOBS"
[ "${FREE:-0}" -ge 20 ] || guard_fail "guard 5: under 20 GB free"
[ "${PEND:-0}" -le 60 ] || guard_fail "guard 5: $PEND pending, over cap"

RUN_CIF=$(awk '$1=="cd"{print $2; exit}' "$RUNNER" 2>/dev/null)
[ "$(cd "$HARN" 2>/dev/null && pwd -P)" = "$(cd "$RUN_CIF" 2>/dev/null && pwd -P)" ] \
  && echo "guard 9: the runner cd's into $RUN_CIF -- the SAME tree guards 3/4 tested" \
  || guard_fail "guard 9: guards tested $HARN but the runner cd's into $RUN_CIF"

if [ "$FAIL" = 1 ]; then
  if [ "$SUBMIT" = 1 ]; then echo "!!! SUBMIT ABORTED -- nothing submitted."; exit 2; fi
  echo; echo "!!! ONE OR MORE GUARDS FAILED.  DRY RUN: the composition below still prints, but --submit WOULD ABORT."; echo
fi

DESIGN=(optimizer=HF alg-base=SGDm momentum-param-base=0.99 alg-meta=Lion momentum-param-meta=0.99 Lion-beta2-meta=0.9 \

  for S in $SS; do

      case $GR in ch) G=chunk777 ;; nd) G=nodewise ;; k01) G=scalar ;; kL) G=layerwise ;; *) echo "!!! unknown grain $GR"; exit 2 ;; esac

      RN="cgw1-$ARM-s$S"

$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \

raise SystemExit(0 if "ARGS: " + tail == Dm.args_line(arm, seed, save) else 1)
PYEOF
