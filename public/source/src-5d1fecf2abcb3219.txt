def mean(v):

def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

def u_of(n, P):
    if n <= 0:
        return 0
    return min(n - 1, P) - max(0, n - 1 - P)


def v_of(n, P, b0=B0):
    return f32(min(HI, max(LO, b0 + MS * u_of(n, P))))


def lion_natural(beta_pre, mom_pre, z):
    a, b = B2 * mom_pre, (1 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(LO, min(HI, beta_pre - MS * s)), tie


def in_window(arm, n):
    if arm not in WINDOW:
        return True
    n0, n1 = WINDOW[arm]
    return n >= n0 and (n1 is None or n < n1)


def hold_value(arm, n):
    """50's applied schedule (group 1), window included, as float32."""
    m, P = MODE[arm]
    if m == "floor" or not in_window(arm, n):
        return f32(LO)
    return v_of(n, P)


_KNOTS = {}


def load_knots(path=HEADPATH_FILE, want_sha=HEADPATH_SHA):
    if path in _KNOTS:
        return _KNOTS[path]
    out = None
    if os.path.exists(path):
        body = open(path, "rb").read()
        if hashlib.sha256(body).hexdigest() == want_sha:
            d = json.loads(body.decode("utf-8"))
            out = ([int(x) for x in d["n"]], [float(x) for x in d["v"]])
    _KNOTS[path] = out
    return out


def rec_value(n, kn, kv, b0=B0):
    """PATCH_COMPHOLD's rec: schedule as float32 (242.3(2))."""
    if n < kn[0]:
        x = b0
    elif n >= kn[-1]:
        x = kv[-1]
    else:
        lo_, hi_ = 0, len(kn) - 1
        while hi_ - lo_ > 1:
            mid = (lo_ + hi_) // 2
            if kn[mid] <= n:
                lo_ = mid
            else:
                hi_ = mid
        k = lo_
        x = kv[k] + (kv[k + 1] - kv[k]) * (n - kn[k]) / float(kn[k + 1] - kn[k])
    return f32(min(HI, max(LO, x)))


def hold_c_value(n):
    K = load_knots()
    if K is None:
        return None
    return rec_value(n, K[0], K[1])


def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]

def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names=None):
    names = names or NAMES
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "bh": [ln for ln in lines if ln.startswith("BETA_HOLD")],
           "ch": [ln for ln in lines if ln.startswith("COMP_HOLD")],
           "wh": [ln for ln in lines if ln.startswith("WINDOW_HOLD")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s" % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"

def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0 and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm

def load_probe(runsdir, arm, seed, prefix=PREFIX):
    for base in (os.path.join(runsdir, prefix), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (prefix, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            return recs
    return None


BH_KEYS = ("bh_n", "bh_active", "bh_nat", "bh_held")
CH_KEYS = ("ch_n", "ch_active", "ch_nat", "ch_held")
WH_KEYS = ("wh_n", "wh_active", "wh_nat", "wh_held")


def _groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    return [float(x) for x in v[0]]


def _audit(r, pfx, n, g, beta, want_nat, prev, tie=False):
    """one patch's own record on one group.  -> (ok, active_flag, count); count None if unreadable."""
    try:
        cn, ca, cnat, cheld = (int(r[pfx + "_n"]), int(r[pfx + "_active"]), float(r[pfx + "_nat"]),
                               float(r[pfx + "_held"]))
    except (KeyError, TypeError, ValueError):
        return False, False, None
    ok = (cn == n and abs(cheld - beta[g]) <= BH_TOL and 0 <= ca <= cn and ca >= prev
          and (tie or abs(cnat - want_nat) <= BH_TOL))
    return ok, abs(cnat - cheld) > BH_TOL, ca


def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail).  Pure function of the records."""
    ng = len(arm_groups(arm))
    held = arm in HELD
    win = arm in WINDOWED
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0,
         "bad_sched": 0, "bad_sched_pre": 0, "bad_csched": 0, "bad_csched_pre": 0, "bad_bh": 0, "bad_ch": 0,
         "bad_wh": 0, "keys_on_control": 0, "active_records": 0, "cactive_records": 0, "wactive_records": 0,
         "bh_active_last": None, "ch_active_last": None, "wh_active_last": None,
         "worst_lion": 0.0, "worst_sched": 0.0, "worst_csched": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    if held and hold_c_value(2) is None:
        d["why"] = "replay file %s unavailable or sha != HEADPATH_SHA" % HEADPATH_FILE
        return False, d
    prev_b = prev_c = prev_w = -1
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) \
                or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre = _groupvec(r, "beta_pre", ng)
            mom = _groupvec(r, "mom_pre", ng)
            z = _groupvec(r, "z_agg", ng)
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        s = r["step"]
        if s != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        n = s + 2
        has_bh = any(key in r for key in BH_KEYS)
        has_ch = any(key in r for key in CH_KEYS)
        has_wh = any(key in r for key in WH_KEYS)
        if not held:
            for g in range(ng):
                nat, tie = lion_natural(bpre[g], mom[g], z[g])
                if tie:
                    d["ties"] += 1
                    continue
                e = abs(nat - beta[g])
                d["worst_lion"] = max(d["worst_lion"], e)
                d["bad_lion"] += e > BH_TOL
            d["keys_on_control"] += has_bh or has_ch or has_wh
            continue
        if not win:
            d["keys_on_control"] += has_wh
        es, ep = abs(beta[1] - hold_value(arm, n)), abs(bpre[1] - hold_value(arm, n - 1))
        d["worst_sched"] = max(d["worst_sched"], es, ep)
        d["bad_sched"] += es > BH_TOL
        d["bad_sched_pre"] += ep > BH_TOL
        nat1, tie1 = lion_natural(bpre[1], mom[1], z[1])
        ok, act, ca = _audit(r, "bh", n, 1, beta, nat1, prev_b, tie1)
        d["bad_bh"] += not ok
        if ca is not None:
            d["active_records"] += act
            prev_b = max(prev_b, ca)
            d["bh_active_last"] = ca
        cs, cp = abs(beta[0] - hold_c_value(n)), abs(bpre[0] - hold_c_value(n - 1))
        d["worst_csched"] = max(d["worst_csched"], cs, cp)
        d["bad_csched"] += cs > BH_TOL
        d["bad_csched_pre"] += cp > BH_TOL
        nat0, tie0 = lion_natural(bpre[0], mom[0], z[0])
        ok, act, ca = _audit(r, "ch", n, 0, beta, nat0, prev_c, tie0)
        d["bad_ch"] += not ok
        if ca is not None:
            d["cactive_records"] += act
            prev_c = max(prev_c, ca)
            d["ch_active_last"] = ca
        if win:
            ok, act, ca = _audit(r, "wh", n, 1, beta, v_of(n, MODE[arm][1]), prev_w)
            d["bad_wh"] += not ok
            if ca is not None:
                d["wactive_records"] += act
                prev_w = max(prev_w, ca)
                d["wh_active_last"] = ca
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_step"] == 0 and d["bad_lion"] == 0 and d["keys_on_control"] == 0)
    if held:
        ok = (ok and d["bad_sched"] == 0 and d["bad_sched_pre"] == 0 and d["bad_bh"] == 0
              and d["bad_csched"] == 0 and d["bad_csched_pre"] == 0 and d["bad_ch"] == 0
              and (d["ch_active_last"] is None or d["ch_active_last"] >= d["cactive_records"]))
        if not win:   # under a window bh_active counts tri-vs-Lion differences, not held-vs-Lion (patch_windowhold.py)
            ok = ok and (d["bh_active_last"] is None or d["bh_active_last"] >= d["active_records"])
    if win:
        ok = (ok and d["bad_wh"] == 0 and d["wh_active_last"] is not None
              and d["wh_active_last"] >= d["wactive_records"])
    return ok, d

def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord \
                and isinstance(r.get("step"), int):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    pk = max(vals, key=lambda sv: sv[1])
    win = [math.exp(v + 15.0) for s, v in vals if 15 <= s / float(STEPS_PER_EPOCH) < 30]
    win0 = [math.exp(v + 15.0) for s, v in vals if 5 <= s / float(STEPS_PER_EPOCH) < 18.858]
    win1 = [math.exp(v + 15.0) for s, v in vals if 18.858 <= s / float(STEPS_PER_EPOCH) < 41]
    return {"peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH), "r1530": mean(win), "rE": mean(win0),
            "rL": mean(win1)}

# THE BRANCH MAP -- a pure function of the in-batch means, one flag and LATE's onset level
# =============================================================================
def state(L, M):
    if L >= M[CTL] - MATCH_BAR:
        return "AT-CTL"
    if L <= M["k01"] + NULL_BAR:
        return "AT-K01"
    return "BETWEEN"


def stallbar(arm, M, onset_late):
    if arm == "LATE":
        return max(M["k01"] + NULL_BAR, onset_late + MATCH_BAR)
    return M["k01"] + NULL_BAR


def wstate(arm, M, onset_late):
    L = M[arm]
    if L >= M[CTL] - MATCH_BAR:
        return "RESCUED"
    if L <= stallbar(arm, M, onset_late):
        return "STALLED"
    return "PARTIAL"


def decide(M, diverged=False, onset_late=ONSET_PRIOR):
    """M: arm -> mean plateau5; onset_late: LATE's arm-mean TEST at Epoch line E1_LINE."""
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M[CTL] - M["k01"] < GAP_MIN:
        return "CONTROL-FAILED"
    if state(M["HIGHHEADPATH"], M) != "AT-K01":
        return "REPLICATE-FAILED"
    sm, sr = state(M["MIDDOSE"], M), state(M["RESDOSE"], M)
    if M["RESDOSE"] - M["MIDDOSE"] > MATCH_BAR:
        dose = "DOSE-NONMONOTONE"
    elif sm == "AT-K01" and sr == "AT-K01":
        dose = "DOSE-ALL-STALL"
    elif sm == "AT-CTL" and sr == "AT-K01":
        dose = "DOSE-THRESHOLD"
    elif sm == "AT-CTL" and sr == "AT-CTL":
        dose = "DOSE-ONLY-HIGH-STALLS"
    else:
        dose = "DOSE-GRADED"
    if stallbar("LATE", M, onset_late) >= M[CTL] - MATCH_BAR:
        window = "WINDOW-ONSET-UNRESOLVED"
    else:
        we, wl = wstate("EARLY", M, onset_late), wstate("LATE", M, onset_late)
        if we == "STALLED" and wl == "RESCUED":
            window = "EARLY-WINDOW"
        elif we == "RESCUED" and wl == "STALLED":
            window = "LATE-WINDOW"
        elif we == "STALLED" and wl == "STALLED":
            window = "BOTH-WINDOWS"
        elif we == "RESCUED" and wl == "RESCUED":
            window = "NEITHER-WINDOW"
        else:
            window = "WINDOW-GRADED"
    return "%s | %s" % (dose, window)

def license_lines(branch):
    out = []
    for tok in branch.split(" | "):
        out += LICENSE[tok]
    return out

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    print("=" * 78)
    print(" cvt9 -- PLAINNET TENSOR 50: WHEN (EARLY / LATE) AND HOW MUCH (MIDDOSE / RESDOSE)?  (PATCH_WINDOWHOLD)")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-12s %-38s BETA_HOLD %-30s COMP_HOLD %-18s WINDOW_HOLD %s" % (a, SPEC[a], BETAHOLD[a] or "(unset)",
              COMPHOLD[a] or "(unset)", WINDOWHOLD[a] or "(unset)"))
    print(" CO-PRIMARY: P_MID, P_RES, P_EARLY, P_LATE (= LOWHEADPATH - arm), P_WIN = LATE - EARLY.")
    print(" plateau5 = mean TEST over epochs %d..%d of each run's own .out.  BARS ARE FROZEN LITERALS (O2)." % (EPOCHS - 5, EPOCHS - 1))
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = read_runs(runsdir)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    every run's OWN ARGS line says what its file name registers")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s),
                    "momentum-param-base": "0.99", "weight-decay-base": "0.1", "momentum-param-meta": "0.99",
                    "Lion-beta2-meta": "0.9", "weight-decay-meta": "0", "gamma": "1"}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad]) if (dup or bad) else "")

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()), "G-ENV one distinct ENV line, one per run",
         "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
    hchk(envs == [ENV_EXPECTED], "G-ENV the ENV line is cvt1's / cvt4's / cvt6's, byte for byte (PROBE_DIR stripped)")
    badpt = [(a, s) for (a, s), r in sorted(runs.items())
             if len(r["pt"]) != 1 or not PT_RE.match(r["pt"][0]) or PT_RE.match(r["pt"][0]).group(1) != PT_TYPE[a]]
    hchk(not badpt, "G-ENV every run printed ONE `PROBE_TENSOR: on every=%d type=<its arm's> tensors=%d`" % (PROBE, NTENS),
         "bad: %s" % badpt[:5] if badpt else "%d runs" % len(runs))

    print("\nG-VOTEW / G-HOLDW / G-CHOLDW / G-WHOLDW   one line of each kind per run == the arm's registered witness")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            hchk(r["vw"] == [WITNESS_VW], "G-VOTEW %s-s%d" % (a, s), "" if r["vw"] == [WITNESS_VW] else "got %r" % (r["vw"][:2],))
            hchk(r["bh"] == [WITNESS_BH[a]], "G-HOLDW %s-s%d" % (a, s),
                 "" if r["bh"] == [WITNESS_BH[a]] else "got %r" % ([x[:160] for x in r["bh"][:2]],))
            hchk(r["ch"] == [WITNESS_CH[a]], "G-CHOLDW %s-s%d" % (a, s),
                 "" if r["ch"] == [WITNESS_CH[a]] else "got %r" % ([x[:200] for x in r["ch"][:2]],))
            hchk(r["wh"] == [WITNESS_WH[a]], "G-WHOLDW %s-s%d" % (a, s),
                 "" if r["wh"] == [WITNESS_WH[a]] else "got %r" % ([x[:160] for x in r["wh"][:2]],))

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == the frozen manifest, byte for byte")
    mp, how = resolve_manifest(runsdir, manifest_path)
    if not mp:
        hchk(False, "G-STRUCT manifest present", how)
    else:
        txt = open(mp).read()
        print("  manifest: %s (%s)" % (mp, how))
        hchk(txt == synthetic_manifest_text(),
             "G-STRUCT byte-identical to synthetic_manifest_text() (53 tensors, HEAD [52,1], the three hold tables)",
             "%d vs %d bytes" % (len(txt), len(synthetic_manifest_text())))

    print("\nG-PROV    PROVENANCE.txt")
    pp, phow = resolve(runsdir, "", "PROVENANCE.txt")
    prov = read_kv(pp)
    if prov is None:
        hchk(False, "G-PROV PROVENANCE.txt present", phow)
    else:
        print("  provenance: %s (%s)" % (pp, phow))
        hchk(prov.get("MODE") == "submit", "G-PROV MODE submit", str(prov.get("MODE")))
        hchk(prov.get("BUILD_NETWORK_SHA256") == POST_SHA, "G-PROV build_network.py sha256 == POST_SHA %s..." % POST_SHA[:12],
             str(prov.get("BUILD_NETWORK_SHA256"))[:16])
        hchk(prov.get("HF_SHA256") == HF_POST_SHA, "G-PROV HF.py sha256 == HF_POST_SHA %s... (cvt9's PATCH_WINDOWHOLD tree)"
             % HF_POST_SHA[:12], str(prov.get("HF_SHA256"))[:16])
        hchk(prov.get("RUNNER_SHA256") == RUNNER_SHA, "G-PROV RUNNER_SHA256 == cvt9's runner %s..." % RUNNER_SHA[:12],
             str(prov.get("RUNNER_SHA256"))[:16])
        hchk(prov.get("REPLAY_SHA256") == HEADPATH_SHA, "G-PROV REPLAY_SHA256 == HEADPATH_SHA %s..." % HEADPATH_SHA[:12],
             str(prov.get("REPLAY_SHA256"))[:16])
        me = file_sha(os.path.abspath(__file__))
        hchk(prov.get("SCORER_SHA256") == me, "G-PROV SCORER_SHA256 == this scorer's sha256 %s..." % me[:12],
             str(prov.get("SCORER_SHA256"))[:16])

    print("\nG-REPLAY  the replay file this scorer reads")
    hchk(os.path.exists(HEADPATH_FILE) and file_sha(HEADPATH_FILE) == HEADPATH_SHA and load_knots() is not None
         and len(load_knots()[0]) == HEADPATH_KNOTS,
         "G-REPLAY %s sha256 == HEADPATH_SHA, %d knots" % (os.path.relpath(HEADPATH_FILE, REPO), HEADPATH_KNOTS))

    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH; tail slope = TEST OLS over epochs 80-99 (descriptive)")
    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)
    arm_t = dict((a, [runs[(a, s)]["train5"] for s in SEEDS]) for a in ARMS)
    for a in ARMS:
        v, t = arm_v[a], arm_t[a]
        sl = [tail_slope(runs[(a, s)]["eps"]) for s in SEEDS]
        print("  %-12s TEST %.4f  sd %s  range %.4f   TRAIN %.4f   tail slope %s pp/ep"
              % (a, mean(v), fmt(sd(v), 4), max(v) - min(v), mean(t), "/".join(fmt(x, 3) for x in sl)))
        print("               seeds: %s" % ", ".join("s%d %.4f (train %.4f)" % (s, x, y) for s, x, y in zip(SEEDS, v, t)))
    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    hi = max(M.values())
    print("\nG-FLOOR / G-CEIL")
    hchk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE), "max %.4f" % hi)
    hchk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)

    if harness:
        final = "FINAL: HARNESS-UNSOUND | %d harness gate(s) failed: %s" % (len(harness), "; ".join(h.split()[0] for h in harness))
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)
        for h in harness:
            print("   FAILED  %s" % h)
        for ln in LICENSE["HARNESS-UNSOUND"]:
            print("  " + ln)
        return 1, final

    print("\nG-BITE    the registered holds reached the step sizes at every record, and touched nothing else")
    print("  (a) 500 records; (b) k01's beta == Lion recomputed (tol %.0e); (c) every held arm: group 1 == 50's (windowed)" % BH_TOL)
    print("  schedule and group 0 == HEADPATH, in beta (step+2) AND beta_pre (step+1); (d) bh_* / ch_* / wh_* audit;")
    print("  (e) no hold key on k01, no wh_* key on a non-windowed arm.")
    bite_fail = []
    nonfin = {}
    for s in SEEDS:
        for a in ARMS:
            recs = load_probe(runsdir, a, s)
            ok, d = bite_check(a, recs)
            nonfin[(a, s)] = d["nonfinite"]
            print("  %-4s G-BITE %-12s s%d  records %s  bad %d  nonfinite %d  lion-mismatch %d (ties %d, worst %.1e)  "
                  "sched50 %d/%d (worst %.1e)  schedC %d/%d (worst %.1e)  bh-audit %d  ch-audit %d  wh-audit %d  "
                  "keys-on-control %d  active %d/%d/%d  bh_active %s  ch_active %s  wh_active %s%s"
                  % ("PASS" if ok else "FAIL", a, s, "absent" if recs is None else len(recs), d["bad_rec"], d["nonfinite"],
                     d["bad_lion"], d["ties"], d["worst_lion"], d["bad_sched"], d["bad_sched_pre"], d["worst_sched"],
                     d["bad_csched"], d["bad_csched_pre"], d["worst_csched"], d["bad_bh"], d["bad_ch"], d["bad_wh"],
                     d["keys_on_control"], d["active_records"], d["cactive_records"], d["wactive_records"],
                     d["bh_active_last"], d["ch_active_last"], d["wh_active_last"], ("  (%s)" % d["why"]) if "why" in d else ""))
            if not ok:
                bite_fail.append("%s-s%d" % (a, s))
    if bite_fail:
        final = "FINAL: PATCH-NOT-VERIFIED | G-BITE failed on %s" % ",".join(bite_fail)
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)
        for ln in LICENSE["PATCH-NOT-VERIFIED"]:
            print("  " + ln)
        return 1, final

    print("\nSIGMA     O2: max(frozen floor, in-batch) -- nothing from the corpus")
    ss = sum(sum((x - M[a]) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    which, sigma_used, cands = compose_sigma(sigma_in)
    for nm, v in cands:
        print("  %-20s %.6f" % (nm, v))
    se = sigma_used * math.sqrt(2.0 / 3.0)
    print("  SIGMA_USED           %.6f  (= %s)" % (sigma_used, which))
    print("  SE_ARM_DIFF          %.6f  = SIGMA_USED * sqrt(2/3)  (df_inbatch %d)" % (se, df))

    onset = mean([runs[("LATE", s)]["eps"][E1_LINE][1] for s in SEEDS])
    print("\nONSET     LATE's own arm-mean TEST at Epoch line %d (the last line wholly before update %d): %.4f  (seeds %s)"
          % (E1_LINE, N_E, onset, " / ".join("%.3f" % runs[("LATE", s)]["eps"][E1_LINE][1] for s in SEEDS)))

    print("\nCONTRASTS (every one WITHIN batch)")
    CP = {}
    for nm, (x, y) in COPRIMARY:
        CP[nm] = (M[x] - M[y], T[x] - T[y])
        print("  PRIMARY  %-7s = %-11s - %-11s = %+.4f pp = %+.2f SE   (TRAIN %+.4f)   seeds %s"
              % (nm, x, y, CP[nm][0], CP[nm][0] / se, CP[nm][1],
                 " / ".join("%+.3f" % (runs[(x, s)]["plateau5"] - runs[(y, s)]["plateau5"]) for s in SEEDS)))
    D_CTL = M[CTL] - M["k01"]
    R_HIGH = M[CTL] - M["HIGHHEADPATH"]
    print("  KEY      D_CTL   = LOWHEADPATH - k01          = %+.4f pp = %+.2f SE   (TRAIN %+.4f)  [positive control]"
          % (D_CTL, D_CTL / se, T[CTL] - T["k01"]))
    print("  KEY      R_HIGH  = LOWHEADPATH - HIGHHEADPATH = %+.4f pp = %+.2f SE   (TRAIN %+.4f)  [cvt6's P_50, replicated]"
          % (R_HIGH, R_HIGH / se, T[CTL] - T["HIGHHEADPATH"]))
    ADD = CP["P_EARLY"][0] + CP["P_LATE"][0] - R_HIGH
    print("  DESCR    ADD     = P_EARLY + P_LATE - R_HIGH  = %+.4f pp  [0 if the two halves' losses add up to the whole]" % ADD)
    for a in HELD:
        print("  KEY      D_%-12s = %-12s - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)   kept fraction of R_HIGH %.3f"
              % (a, a, M[a] - M["k01"], (M[a] - M["k01"]) / se, T[a] - T["k01"],
                 (M[a] - M["HIGHHEADPATH"]) / R_HIGH if R_HIGH else float("nan")))

    print("  state bars: AT-CTL / RESCUED iff L >= %.4f ; AT-K01 / STALLED(EARLY) iff L <= %.4f ; STALLED(LATE) iff L <= %.4f"
          % (M[CTL] - MATCH_BAR, M["k01"] + NULL_BAR, stallbar("LATE", M, onset)))

    print("\nG-DIVERGE (a branch condition, not a harness gate)")
    div = [a for a in ARMS if max(arm_v[a]) - min(arm_v[a]) > DIVERGED_BAR]
    print("  %s every arm's seed range <= %.1f pp%s" % ("PASS" if not div else "FAIL", DIVERGED_BAR,

    branch = decide(M, bool(div), onset)
    stamps = ["HARNESS-CLEAN", "PATCH-BITES", "HOLD-FROM-INIT", "COMPLEMENT-ON-HEADPATH", "FORCED-ARMS-OPEN-LOOP",

              "SIGMA-PRIOR-FROZEN" if which == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH"]
    if div:
        stamps.append("DIVERGED-%s" % "-".join(div))
    stamps.append("POSITIVE-CONTROL-REPRODUCES" if D_CTL >= GAP_MIN else "POSITIVE-CONTROL-FAILED")
    if D_CTL >= GAP_MIN:
        for a in ("HIGHHEADPATH",) + DOSE_ARMS:
            stamps.append("%s-%s" % (a, state(M[a], M)))
        for a in WINDOWED:
            stamps.append("%s-%s" % (a, wstate(a, M, onset)))
    stamps.append("LATE-ONSET:%.2f" % onset)
    for a in HELD:
        if M[a] - M[CTL] > MATCH_BAR:
            stamps.append("%s-ABOVE-CTL" % a)
        if M[a] - M["k01"] < -NULL_BAR:
            stamps.append("%s-BELOW-K01" % a)
        if any(nonfin[(a, s)] for s in SEEDS):
            stamps.append("NONFINITE-%s" % a)
    if any(abs(M[a] - M["k01"]) <= NULL_BAR for a in HELD):
        stamps.append("FLOOR-READINGS-ARE-BOUNDS")
    agree = all((t > 0) == (p > 0) for p, t in CP.values() if abs(p) >= RESCUE_BAR)
    stamps.append("TRAIN-AGREES" if agree else "TRAIN-DISAGREES")
    final = "FINAL: %s | %s" % (branch, " | ".join(stamps))
    print("\n" + "=" * 78)
    print(final)
    print("=" * 78)

    print("\nBETWEEN-BATCH, NON-GATING (frozen literals; NO other batch enters any cvt9 bar or contrast):")
    for b, dct in R_BETWEEN.items():
        print("  %s: %s" % (b, "  ".join("%s %.4f" % kv for kv in dct.items())))
    print("  cvt9 {105,106,107}: %s" % "  ".join("%s %.4f" % (a, M[a]) for a in ARMS))

    try:
        descriptive_probe_readout(runsdir, runs)
    except Exception as ex:
        print("  (descriptive probe readout skipped: %s)" % type(ex).__name__)

    print("\nWHAT THIS DECIDES -- as registered, before any run existed:")
    for ln in license_lines(branch):
        print("  " + ln)
    print("\nWHAT IT DOES NOT LICENSE, UNCONDITIONALLY:")
    print("  * Anything about ResNet18_c100, VGG or any other tensor: one network, one cell, one held tensor (idx 50).")
    print("  * A window finer than the one cut at the peak (update 9429); which sub-window of a half acts.")
    print("  * A dose curve beyond the four triangle rungs (5041 between batch, 7235, 8609, 9428); the family moves peak,")
    print("    fall onset and duration together.")
    print("  * That the forced (open-loop) complement stands for a free one; necessity of anything; point agreement at k01.")
    print("  * Anything beyond 100 epochs.")
    return 0, final

_E_ONSET_CTL = 12.3       # LATE follows the control's curve to line E1_LINE, reaching about this value there


def _e_curve(level, arm, onset=None):
    """an invented TEST curve: exactly `level` over the last 5 epochs.  LATE follows a control-like curve to line
    E1_LINE (value `onset`) and then moves to its level."""
    out = []
    for e in range(EPOCHS):
        if e >= EPOCHS - 5:
            out.append(level)
        elif arm == "LATE":
            on = _E_ONSET_CTL if onset is None else onset
            if e <= E1_LINE:
                out.append(on * (e + 1) / float(E1_LINE + 1))
            else:
                out.append(level + (on - level) * math.exp(-(e - E1_LINE) / 6.0))
        else:
            out.append(level * (1 - math.exp(-(e + 1) / 12.0)))
    return out


def _e_write(d, arm, s, jid, level, n_ep=EPOCHS, run_done=True, env=_E_ENV, spec=None, bh=None, ch=None, wh=None, vw=None,
             onset=None):
    rn = "%s-%s-s%d" % (PREFIX, arm, s)
    args = ("--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 "
            "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 "
            "--weight-decay-meta 0 --dataset CIFAR100 --NN-name %s --batch-size 100 "
            "--max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-6 "
            "--num-epochs 100 --stepsize-groups %s --seed %d "
            "--save-directory /cluster/account_b/metaopt/runs/cvt9 --run-name %s" % (NET, spec or SPEC[arm], s, rn))
    lines = ["NODE=[HOST_45cba27b74] | JOB=%s %d | AUGMENT=1" % (rn, jid), "ARGS: " + args, env, "NVIDIA L4, 23034 MiB",
             WITNESS_VW if vw is None else vw, WITNESS_BH[arm] if bh is None else bh,
             WITNESS_CH[arm] if ch is None else ch, WITNESS_WH[arm] if wh is None else wh,
             "Files already downloaded and verified", "Files already downloaded and verified",
             "PROBE_TENSOR: on every=100 type=%s tensors=%d meta_alg=Lion momentum_param=0.99 Lion_beta2=0.9 dir=/x"
             % (PT_TYPE[arm], NTENS)]
    lines = [ln for ln in lines if ln != ""]
    curve = _e_curve(level, arm, onset)
    for e in range(n_ep):
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.4f %%" % (e, min(99.9, curve[e] + 15.0), curve[e]))
    if run_done:
        lines += ["41  minutes", "RUN_DONE"]
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(lines) + "\n")


def _e_probe_records(arm, seed=0, n=N_RECORDS, as_arm=None, unheld=False, cunheld=False, unwindowed=False,
                     pre_natural=False, wpre_natural=False, bh_forge=False, wh_forge=False, wh_off_by_one=False,
                     strip_wh=False, add_bh=False, add_wh=False, touch=False):
    """synthetic PROBE_TENSOR records consistent with Lion and the arm's holds.  as_arm: 50 on ANOTHER arm's schedule;
    unheld / cunheld: that group follows Lion; unwindowed: a windowed arm's 50 carries the WHOLE tri (the null)."""
    import random
    rnd = random.Random(3000 + seed + 11 * ARMS.index(arm))
    ng = len(arm_groups(arm))
    harm = as_arm or arm
    out = []
    active = cactive = wactive = 0
    for k in range(n):
        s = 100 * k
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        z = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [f32(max(LO, min(HI, -13.8 + 0.3 * math.sin(k / 30.0) - 0.004 * k))) for _ in range(ng)]
        beta = [f32(lion_natural(bpre[g], mom[g], z[g])[0]) for g in range(ng)]
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99}
        if arm in HELD:
            if not unheld:
                if unwindowed:
                    sch = lambda nn: v_of(nn, MODE[harm][1])
                else:
                    sch = lambda nn: hold_value(harm, nn)
                bpre[1] = sch(s + 1) if not (pre_natural or wpre_natural) else f32(bpre[1] + 0.25)
                beta[1] = sch(s + 2)
            nat1 = f32(lion_natural(bpre[1], mom[1], z[1])[0])
            if abs(nat1 - beta[1]) > BH_TOL:
                active += 1
            rec.update({"bh_n": s + 2, "bh_active": active, "bh_nat": (nat1 + 0.01) if bh_forge else nat1, "bh_held": beta[1]})
            if not cunheld:
                bpre[0] = hold_c_value(s + 1)
                beta[0] = hold_c_value(s + 2)
            nat0 = f32(lion_natural(bpre[0], mom[0], z[0])[0])
            if abs(nat0 - beta[0]) > BH_TOL:
                cactive += 1
            rec.update({"ch_n": s + 2, "ch_active": cactive, "ch_nat": nat0, "ch_held": beta[0]})
            if (harm in WINDOWED and not strip_wh) or add_wh:
                wn = v_of(s + 2, MODE[harm][1]) if harm in MODE and MODE[harm][0] == "tri" else beta[1]
                if abs(wn - beta[1]) > BH_TOL:
                    wactive += 1
                rec.update({"wh_n": s + 2 + (1 if wh_off_by_one else 0), "wh_active": wactive,
                            "wh_nat": (wn + 0.01) if wh_forge else wn, "wh_held": beta[1]})
        if touch:
            beta[0] = f32(beta[0] + 0.002)
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[-1], "bh_held": beta[-1]})
        rec.update({"beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [z],
                    "z_tensor": [0.0] * NTENS, "m_tensor": [0.0] * NTENS})
        out.append(rec)
    return out

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None, env_break=None,
             spec_break=None, bh_break=None, ch_break=None, wh_break=None, vw_break=None, manifest=None, prov=None,
             probe_break=None, probe_drop=None, onset=None):
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(manifest if manifest is not None else synthetic_manifest_text())
    _e_prov(d, **(prov or {}))
    jid = 9900000
    pb = probe_break or {}
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = "scalar" if spec_break == (a, s) else None
            bh = bh_break[1] if bh_break and bh_break[0] == (a, s) else None
            ch = ch_break[1] if ch_break and ch_break[0] == (a, s) else None
            wh = wh_break[1] if wh_break and wh_break[0] == (a, s) else None
            vw = vw_break[1] if vw_break and vw_break[0] == (a, s) else None
            _e_write(d, a, s, jid, L, env=env, spec=spec, bh=bh, ch=ch, wh=wh, vw=vw, onset=onset)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt9/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
    sys.exit(main())
