# ---- the derivation's disclosure numbers, on the landed records -------------------------------------------------------
def cumulative_dose(recs, wd=SRC_WD, lo=0, hi=None):
    """sum over records with lo <= step < hi of PROBE * wd * median(exp(beta)) -- the cumulative log-shrink, approx."""
    tot = 0.0
    for r in recs:
        if r["step"] < lo or (hi is not None and r["step"] >= hi):
            continue
        a = sorted(math.exp(b) for b in r["beta"])
        tot += PROBE * wd * a[len(a) // 2]
    return tot


def derive_lambda(root):
    """-> (csh1's per-seed dict, Q_M token, cumulative disclosure per seed) or None if the records are not under root."""
    got = H.derive_gammas(root)
    if got is None:
        return None
    per, dv = got
    cum = {}
    for s in SRC_SEEDS:
        recs = [json.loads(ln) for ln in open(H.src_probe_path(root, s)) if ln.strip()]
        cum[s] = (cumulative_dose(recs, hi=H.ONSET_STEP), cumulative_dose(recs))
    return per, dv["Q_M"], cum
