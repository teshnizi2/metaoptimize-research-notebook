set -e
cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
# The alpha0 CONTROL for the cycle-6 beta-drift mechanism (cycle 7 §1).
#
# p2-* measured the shared beta's drift at alpha0=1e-6, i.e. while every arm is
# still climbing out of a 7-log-unit hole (beta0 = -13.8). The pooled arms all
# converged to beta in [-6.0, -6.5], and ln(1e-3) = -6.91 -- so alpha0=1e-3
# starts essentially AT the destination. If the 20-epoch ordering
# (scal 70.7 > blk6 68.5 > lay 53.1 > node 33.0 > w 14.8) is a startup artefact,
# it collapses here. Identical to betadrift.sh in every other respect.
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-3 --gamma 1 --NN-name ResNet18 --dataset CIFAR10 --num-epochs 20 --batch-size 100 --max-time 999:00:00"
C="BETA_CLIP=-15:-2.3026"
sub(){ sbatch --parsable --job-name=$1 --partition=gpu-l4-24g,gpu-short --gres=gpu:l4:1 --time=00:45:00 \
  --export="ALL,AUGMENT=1,$C,$3,PROBE=100,PROBE_DIR=$R/bdrift3/$1" \
  jobs/run_cifar.sh $B --stepsize-groups $2 --seed 0 \
  --save-directory $R/bdrift3 --run-name $1 >/dev/null; }
sub p3-scal      scalar          "HIER="
sub p3-blk6-L1p0 resnet18_blocks "HIER=shrink,LAM=1.0"
sub p3-lay-L1p0  layerwise       "HIER=shrink,LAM=1.0"
sub p3-node-L1p0 nodewise        "HIER=shrink,LAM=1.0"
sub p3-w-L1p0    weightwise      "HIER=shrink,LAM=1.0"
sub p3-lay-plain layerwise       "HIER="
sleep 3
echo "P3_SUBMITTED=$(squeue -h -u account_a -o '%j' | grep -c '^p3-')"
echo "TOTAL_QUEUE=$(squeue -h -u account_a | wc -l)"
