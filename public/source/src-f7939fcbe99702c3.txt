#!/usr/bin/env python3
"""
c98_release.py -- assemble `release/`, the deposit-ready artefact for the DOI.

    python3 analysis/c98_release.py            # build into <repo>/release
    python3 analysis/c98_release.py --outdir X # elsewhere

The build is deterministic and idempotent: it copies from the working tree,
regenerates the four figures, runs the reproduction audit, writes README.md,
ENVIRONMENT.md, Makefile, CITATION.cff and MANIFEST.md5, and refuses to finish if
the audit reports a failed check.  The README's headline numbers are GENERATED by
the audit, not typed, so the artefact cannot ship a number the CSV disagrees with.
"""
import argparse, hashlib, io, os, re, shutil, subprocess, sys, tarfile, textwrap, time

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.normpath(os.path.join(HERE, ".."))
BACKUP = os.path.normpath(os.path.join(ROOT, ".."))
sys.path.insert(0, HERE)

SCORERS = ["c76_mm1_score.py", "c77_pp1_score.py", "c78_bn1_score.py",
           "c81_cc1_score.py", "c82_fa1_score.py", "c83_gc1_score.py",
           "c83_gen_score.py", "c84_gn1_score.py", "c87_rl3_score.py",
           "c87_hz3_score.py", "c88_scorers.py", "c97_bm2_score.py",
           "c97_rp1_score.py", "c97_sm4_score.py",
           # c99 imports c87 from its own directory; both land in code/scorers/,
           # so the deposit can run the scorer that withdrew §4.8's flatness
           # sentence.  Its H1 box gate still needs the excluded probe records.
           "c99_hz3q_score.py"]
TOOLS   = ["aggregate.py", "args_repair.py", "argsline_guard.py",
           "c98_figures.py", "c98_reproduce.py", "c98_release.py",
           # c98_reproduce's [15b] imports this; without it `make reproduce`
           # crashes on the deposit with ModuleNotFoundError.  Caught in cycle
           # 106 by running `make reproduce` on the built deposit, which is the
           # only place that import is exercised outside the source tree.
           "c99_qcalibration.py"]
DOCS    = ["CORRECTIONS.md", "ARGS-AUDIT.md", "DATASETS.md", "OPERATIONS.md",
           "MASTER-TABLE.md"]

def sh(cmd, cwd=ROOT):
    try:
        return subprocess.run(cmd, shell=True, cwd=cwd, capture_output=True,
                              text=True, timeout=600)
    except Exception as e:
        class R: returncode, stdout, stderr = 1, "", str(e)
        return R()

def md5(path, chunk=1 << 20):
    h = hashlib.md5()
    with open(path, "rb") as fh:
        for b in iter(lambda: fh.read(chunk), b""): h.update(b)
    return h.hexdigest()

def copy(src, dst):
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    shutil.copy2(src, dst)

# Editor and mid-flight backup droppings in bin/, patches/ and tests/ are NOT part
# of the deposit.  Without this the file count -- which §8 of the paper prints and
# `make verify` checks -- moves whenever somebody leaves a `.bak_*` beside a
# submission script.  Caught in cycle 102, when one such file made the build ship
# 138 files against a paper that says 137.
_JUNK = re.compile(r"(?:\.pyc|\.pyo|\.orig|\.rej|\.swp|~)$|"
                   r"(?:^\.|\.bak|\.backup|\.tmp|\.old)", re.I)
def skipfile(name):
    return bool(_JUNK.search(name))

HEADLINE_KEYS = [
    "rows in results/all_runs.csv", "admissible rows", "GPU-hours",
    "cells with D > 0", "live pool, 11 byte-identical cells",
    "within-SGDm pool", "within-SGDm Q (k=8)",
    "A = permnode", "95% CI low", "95% CI high",
    "best ResNet-18/C10 MetaOptimize arm", "tuned SGD+cosine baseline",
    "deficit",
]

def headline_block(audit):
    """The lines of the audit that carry the paper's headline numbers.

    Generated from the audit output, never typed, so the README cannot ship a
    number this deposit's own code disagrees with.  A key that no longer
    matches is skipped rather than faked.
    """
    out = []
    for key in HEADLINE_KEYS:
        for ln in audit.splitlines():
            if ln.strip().startswith(key):
                out.append("    " + ln.rstrip()); break
    return "\n".join(out).lstrip()

_CENSUS_PATS = (
    ("distinct", r"=\s*QUANTITIES\s+\d+\s+\((\d+) distinct\)"),
    ("sites",    r"chk\(\) assertion sites executed in this run\s+(\d+)"),
    ("covered",  r"distinct quantity-numerals this run asserts\s+(\d+)"),
    ("pct",      r"coverage of distinct quantity-numerals\s+([\d.]+)%"),
)

def census_sentence(audit):
    """The audit's OWN coverage census, cut out of its output at build time.

    Never typed, for the same reason the headline block is never typed: the
    deposit must not be able to overstate what its audit checks.  Returns a
    sentence, or a pointer to the command if the census could not run (it needs
    the manuscript, which the deposit does not ship).
    """
    got = {}
    for name, pat in _CENSUS_PATS:
        m = re.search(pat, audit)
        if not m: return ("Run `code/c98_reproduce.py --census` against the "
                          "manuscript to measure the exact coverage.")
        got[name] = m.group(1)
    return textwrap.fill(
        "Measured at build time against the manuscript, it executed %(sites)s "
        "claim-carrying assertions covering %(covered)s of the %(distinct)s distinct "
        "quantity-numerals the manuscript prints, which is %(pct)s%% of them. The "
        "audit asserts that triple as well as measuring it, so the coverage claim "
        "cannot drift unnoticed; the census block it comes from is at the foot of "
        "`REPRODUCTION-AUDIT.txt`." % got, width=78)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--outdir", default=os.path.join(ROOT, "release"))
    ap.add_argument("--skip-logs", action="store_true",
                    help="do not build logs/raw_out.tar.gz (much faster)")
    a = ap.parse_args()
    OUT = os.path.abspath(a.outdir)
    if os.path.isdir(OUT): shutil.rmtree(OUT)
    os.makedirs(OUT)

    commit = (sh("git rev-parse HEAD").stdout or "").strip() or "unknown"
    short  = commit[:7]
    dirty  = bool((sh("git status --porcelain").stdout or "").strip())

    # ---- 1. regenerate the figures, then run the audit -----------------------
    print("[1/6] regenerating figures")
    r = sh("%s analysis/c98_figures.py --all" % sys.executable)
    if r.returncode: sys.exit("figure build failed:\n" + r.stderr)
    print("[2/6] running the reproduction audit")
    # --allow-stale-census: a mismatch between §3.4's printed coverage sentence and
    # the code's measured coverage is a manuscript defect, not an artefact defect --
    # the deposit ships no manuscript, and the README's own census sentence is
    # generated from the MEASURED census below, never from §3.4.  Every other failed
    # check still refuses the build.  The stale lines stay in REPRODUCTION-AUDIT.txt.
    r = sh("%s analysis/c98_reproduce.py --allow-stale-census" % sys.executable)
    audit = r.stdout
    if r.returncode:
        sys.exit("REFUSING TO BUILD: the reproduction audit failed.\n" + audit)
    if "COVERAGE CHECK(S) STALE" in audit:
        print("    ** WARNING: the paper's section 3.4 coverage sentence is behind the")
        print("    ** code.  The deposit is unaffected and was built; the MANUSCRIPT")
        print("    ** must be fixed and the deposit rebuilt before the DOI is minted.")

    # ---- 2. copy the payload -------------------------------------------------
    print("[3/6] copying data, code, scripts, patches, figures, docs")
    copy(os.path.join(ROOT, "results", "all_runs.csv"),
         os.path.join(OUT, "data", "all_runs.csv"))
    for f in SCORERS:
        p = os.path.join(ROOT, "analysis", f)
        if os.path.exists(p): copy(p, os.path.join(OUT, "code", "scorers", f))
    for f in TOOLS:
        p = os.path.join(ROOT, "analysis", f)
        if os.path.exists(p): copy(p, os.path.join(OUT, "code", f))
    for d, sub in (("bin", "scripts"), ("patches", "code/patches"),
                   ("tests", "code/tests")):
        src = os.path.join(ROOT, d)
        for f in sorted(os.listdir(src)):
            p = os.path.join(src, f)
            if os.path.isfile(p) and not skipfile(f):
                copy(p, os.path.join(OUT, sub.replace("/", os.sep), f))
    for f in sorted(os.listdir(os.path.join(ROOT, "paper", "figures"))):
        copy(os.path.join(ROOT, "paper", "figures", f),
             os.path.join(OUT, "figures", f))
    for f in DOCS:
        p = os.path.join(ROOT, "docs", f)
        if os.path.exists(p): copy(p, os.path.join(OUT, "docs", f))

    # ---- 3. the raw per-epoch logs, as one archive ---------------------------
    n_out, log_bytes = 0, 0
    if not a.skip_logs:
        print("[4/6] archiving the raw per-epoch .out logs")
        tgz = os.path.join(OUT, "logs", "raw_out.tar.gz")
        os.makedirs(os.path.dirname(tgz), exist_ok=True)
        with tarfile.open(tgz, "w:gz") as tf:
            for d in ("runs", "runs_cluster_b"):
                base = os.path.join(BACKUP, d)
                if not os.path.isdir(base): continue
                for dirpath, _, files in os.walk(base):
                    for f in sorted(files):
                        if not f.endswith(".out"): continue
                        p = os.path.join(dirpath, f)
                        tf.add(p, arcname=os.path.join(d, os.path.relpath(p, base)))
                        n_out += 1; log_bytes += os.path.getsize(p)
    else:
        print("[4/6] SKIPPED raw log archive (--skip-logs)")

    # ---- 4. the generated text files ----------------------------------------
    print("[5/6] writing README, ENVIRONMENT, Makefile, CITATION")
    stamp = time.strftime("%Y-%m-%d", time.gmtime())
    open(os.path.join(OUT, "REPRODUCTION-AUDIT.txt"), "w").write(audit)

    open(os.path.join(OUT, "Makefile"), "w").write(MAKEFILE)
    open(os.path.join(OUT, "code", "verify_manifest.py"), "w").write(VERIFIER)
    open(os.path.join(OUT, "ENVIRONMENT.md"), "w").write(ENVIRONMENT)
    open(os.path.join(OUT, "CITATION.cff"), "w").write(CITATION % dict(date=stamp))
    open(os.path.join(OUT, "README.md"), "w").write(README % dict(
        stamp=stamp, commit=commit, short=short,
        dirty=("  **The working tree was DIRTY at build time; re-build from a "
               "clean checkout before minting the DOI.**" if dirty else ""),
        n_out=n_out, log_mb=log_bytes / 1e6,
        audit_head=headline_block(audit),
        census=census_sentence(audit),
    ))

    # ---- 5. the manifest -----------------------------------------------------
    print("[6/6] writing MANIFEST.md5")
    lines, total, nfiles = [], 0, 0
    for dirpath, dirs, files in os.walk(OUT):
        dirs.sort()
        for f in sorted(files):
            if f == "MANIFEST.md5": continue
            p = os.path.join(dirpath, f)
            rel = os.path.relpath(p, OUT)
            lines.append("%s  %10d  %s" % (md5(p), os.path.getsize(p), rel))
            total += os.path.getsize(p); nfiles += 1
    with open(os.path.join(OUT, "MANIFEST.md5"), "w") as fh:
        fh.write("# md5  bytes  path      -- %d files, %.1f MB, built %s from %s\n"
                 % (nfiles, total / 1e6, stamp, short))
        fh.write("\n".join(lines) + "\n")
    print("\nrelease/ built: %d files, %.1f MB" % (nfiles, total / 1e6))
    print("verify with:  cd %s && make verify" % os.path.relpath(OUT, os.getcwd()))

# ------------------------------------------------------------------ templates
VERIFIER = '''#!/usr/bin/env python3
"""Check every file in this artefact against MANIFEST.md5.  Exit 0 iff all match."""
import hashlib, os, sys
HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.normpath(os.path.join(HERE, ".."))
bad = n = 0
for line in open(os.path.join(ROOT, "MANIFEST.md5")):
    if line.startswith("#") or not line.strip():
        continue
    h, _size, rel = line.split(None, 2)
    rel = rel.rstrip("\\n")
    p = os.path.join(ROOT, rel)
    n += 1
    if not os.path.exists(p):
        print("MISSING  %s" % rel); bad += 1; continue
    d = hashlib.md5(open(p, "rb").read()).hexdigest()
    if d != h:
        print("CHANGED  %s" % rel); bad += 1
print("%d files checked, %d bad" % (n, bad))
sys.exit(1 if bad else 0)
'''

## The rules the code enforces, so that a number cannot drift

* **`plateau5`** — the mean test accuracy over the last 5 epochs of the requested
  budget — is the only accuracy metric read. The CSV also carries `plateau`, the
  20-epoch analogue; it is **banned** as a primary and the code never reads it.
  Two headlines were withdrawn during writing for quoting it.
* **Admissibility** is `window_ok == 1 AND complete == 1 AND a readable plateau5`,
  applied before any arm mean is formed.
* **Every contrast is within one batch** — one contiguous submission — so any
  offset shared by the two arms cancels identically.
* **Rows sharing a `dup_group` are averaged within the group first**, and n counts
  distinct groups. This is what makes the `ml2` cell 3 v 3 (se 0.195) rather than
  6 v 6 (se 0.142): its two halves resolve to the same command line including the
  seed. See `docs/ARGS-AUDIT.md`.
* **CIFAR-10 and CIFAR-100 effects are never averaged or plotted on one
  percentage-point axis.** They sit on ≈7–8 pp and ≈29–30 pp error budgets.
  Figure 1 carries the rule as a hard axis break and gives the commensurable
  quantity, D / headroom, beside it.

## What is *not* here

* **The per-group β probe traces** (`probe*.jsonl`, ≈42 GB). They are what the
  box-occupancy checks read. They are excluded from this deposit for size and are
  available from the authors on request; the occupancy summaries they produce are
  in the run table (`beta_clip`) and in the scorers' output.
* **The CIFAR-10 / CIFAR-100 datasets**, which are standard public downloads and
  are fetched by `torchvision` at run time.
