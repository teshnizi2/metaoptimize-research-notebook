# - plateau5(scalar) at the standard CIFAR-100 cell is +45.75 .. +46.99 pp on
# ResNet18_c100 (9 batches, median +46.548667) and +30.74 .. +31.27 on
# VGG11_bn_c100 (3 batches, median +30.828).  The corpus's ONLY non-BatchNorm
# network is `ResNet18_gn` (17 rows: CIFAR-10, chunk777 / nodewise only; no
# scalar or layerwise arm ever ran on it).  cgn1 asks the cheapest question
# that separates the gap from batch statistics: does the gap EXIST on
# ResNet18_c100 with every BatchNorm2d(C) replaced by GroupNorm(32, C)?
#
# THE NETWORK.  `ResNet18_gn_c100`, added by patches/patch_resnet_gn_c100.py
# (ONE inserted dispatch region; tests/test_resnet_gn_c100.py).  Proved on the
# built models: its ordered (name, shape) list is ELEMENTWISE EQUAL to
# ResNet18_c100's -- 62 tensors, 11,220,132 parameters, the (conv, norm.weight,
# norm.bias) triple at every 1-based index 1..60 (class == index mod 3, every
# 1-D tensor there OWNED by a GroupNorm), then linear.weight / linear.bias --
# and its initial parameters under a fixed seed are BITWISE EQUAL to
# ResNet18_c100's.  So HF.init_meta receives the IDENTICAL partition object and
# the IDENTICAL starting point; the 512-wide scales sit at the same 47/50/53/
# 56/59 with the same names.  What differs is the normaliser: 20 GroupNorm vs
# 20 BatchNorm2d, and BN's 60 running-stat BUFFERS (never parameters).
#
# =============================================================================
# THE ACCOUNTS AND THEIR PREDICTIONS, FROZEN
# =============================================================================
# (A) NOT-BATCH-STATISTICS.  The gap is a property of sharing ONE step size
#     across this tensor list (62 tensors, the same scale/shift tensors in the
#     same positions) under a per-feature affine normaliser; batch statistics
#     are not needed.  PREDICTS k01 in [18, 32] (the BN scalar arm is 22.95),
#     kL in [55, 70], D in [+25, +50].
# (B) BATCH-STATISTICS.  The gap needs BatchNorm's batch statistics (or the
#     per-channel scale invariance BN gives and GN(32) only partly keeps);
#     GroupNorm removes it.  PREDICTS k01 ~ kL ~ L with L in [50, 68], |D| <= 2.
# (C) ATTENUATED.  GN keeps part of the mechanism (per-GROUP invariance, 16
#     channels to a group in layer4).  PREDICTS k01 in (32, 55), kL in [55, 70],
#     D in (+2, +25).
#
# =============================================================================
# THE FLOOR GATE (164.6), EVERY ARM UNDER EVERY ACCOUNT
# =============================================================================
#   floor references (frozen, corpus, cgn1 excluded): chance 1.00 pp; the BN
#   scalar floor band at this cell [21.694, 24.174] (38 rows, mean 22.948632).
#   (A) k01 18-32: IT IS THE IN-BATCH FLOOR ANCHOR BY CONSTRUCTION -- the same
#       disclosed role k01 plays in cvg1 / cdep1 / ciso1; D is the gap BETWEEN
#       the anchor and a free arm, not a censored contrast.  kL 55-70: >= 30.83
#       pp above the band top, >= 30 pp below the ceiling.
#   (B) both arms 50-68: >= 25.83 pp above the band top.  NO arm near the floor.
#   (C) k01 > 32 (>= 7.83 above the band top); kL 55-70 as in (A).
#   The primary D is bounded in NEITHER direction under any account: kL is free
#   (>= 30 pp from both floor and ceiling in every prediction).  The ONE route to
#   a floor-bounded D is kL itself collapsing -- which H-TRAINS (kL >= 40.0)
#   catches as HARNESS-UNSOUND before any branch is read.  So GAP-ABSENT can
#   never be issued with k01 below 38 pp: a null can never be two dead arms.
#
# =============================================================================
# THE NOISE FLOOR, RE-DERIVED AT REGISTRATION, cgn1 EXCLUDED, THEN FROZEN (O2)
# =============================================================================
#   SIGMA_NARROW_PIN 0.5862321062215915  ResNet18_c100/CIFAR100 standard cell,
#                                        df 77, 11 cells, 88 members
#   SIGMA_GN_PIN     0.2731176486424861  ResNet18_gn/CIFAR10 (gn1: ms 1e-4,
#                                        alpha0 1e-3), df 14, 2 cells, 16 members

#   SIGMA_PRIOR := max(the two pins) = 0.5862321062215915   (a LITERAL)
#   At score time:  SIGMA_USED = max(SIGMA_PRIOR, SIGMA_INBATCH)
#   SIGMA_INBATCH is the pooled within-arm SD of cgn1's OWN 6 runs (df 4).

#   invented rows only.
#   SE_ARM_DIFF = SIGMA_USED * sqrt(2/3);  at the prior, 0.478657 pp.
#
# =============================================================================
# THE BARS, FROZEN
# =============================================================================
#   D = plateau5(kL) - plateau5(k01), in-batch 3-seed means, pp scale (the scale
#   is PRE-REGISTERED here -- the 118(b) lesson; a budget-relative D_rel is
#   printed as a SECONDARY, descriptive statistic and moves no branch).
#   GAP_BAR   10.0 pp  (= 20.89 SE at the prior; 21.5 % of the BN in-batch
#                       median +46.548667, 32.4 % of VGG's +30.828)
#   NULL_BAR   2.0 pp  (= 4.18 SE at the prior; 4.3 % of the BN gap)
#   K_SE       3.0     the gap and reversal bars are max(literal, 3 * SE_USED),
#                      so only cgn1's OWN noise can widen them
#   DIVERGED_BAR 5.0 pp within-arm seed range
#   KL_MIN    40.0 pp  H-TRAINS: the GN net must train under layerwise
#   CEIL_MAX  90.0 pp  H-CEIL: no CIFAR-100 arm this high at this cell
#   FULLSIZE_FRAC 0.5  a REPLICATES D >= 0.5 x 46.548667 = 23.274 is stamped
#                      FULL-SIZE, below it ATTENUATED (stamp only)
#
# =============================================================================
# THE BRANCH MAP, FROZEN.  FIRST MATCH WINS.
# =============================================================================
#   INCOMPLETE          any of the 6 runs missing, short of 100 epoch lines, or
#                       lacking RUN_DONE.  No branch; nothing quotable.
#   HARNESS-UNSOUND     CHECKED FIRST among complete batches.  Any of:
#                       H-ARGS   every run's own ARGS line: NN-name
#                                ResNet18_gn_c100, dataset CIFAR100, the arm's
#                                stepsize-groups, its seed, ms 1e-3, alpha0
#                                1e-6, 100 epochs, batch 100, SGDm / Lion / HF,
#                                run-name; no flag repeated (RULE 20)
#                       H-ENV    one ENV line (PROBE_DIR stripped) with
#                                AUGMENT=1, BETA_CLIP=-15:-2.3026, PROBE=100,
#                                HIER=none, SCHED=none; each run prints
#                                `PROBE_TENSOR: on ... type=<its spec> tensors=62`
#                       H-STRUCT the batch's PARTITION-MANIFEST.txt: NETWORK
#                                ResNet18_gn_c100, 62 tensors, 11,220,132
#                                params, GroupNorm 20 / BatchNorm2d 0, every
#                                1-D tensor at 1..60 owned by GroupNorm, the
#                                512-wide scales at 47/50/53/56/59 by name
#                       H-PROV   PROVENANCE.txt: MODE submit and the cgn1
#                                tree's build_network.py sha256 == POST_SHA
#                       H-TRAINS mean(kL) >= KL_MIN
#                       H-CEIL   max arm mean <= CEIL_MAX
#   UNRESOLVED-DIVERGED a within-arm seed range > DIVERGED_BAR (report, do not
#                       average a bimodal arm)
#   GAP-REPLICATES      D >= max(GAP_BAR, K_SE * SE_USED)
#   GAP-REVERSED        D <= -max(NULL_BAR, K_SE * SE_USED)
#   GAP-ABSENT          |D| <= NULL_BAR  AND  SE_USED <= NULL_BAR / 2
#   UNRESOLVED-NOISY    |D| <= NULL_BAR but SE_USED > NULL_BAR / 2 (no power)
#   GAP-PARTIAL         D > NULL_BAR (and below the gap bar)
#   UNRESOLVED-SMALL-NEGATIVE   anything left (-rev bar < D < -NULL_BAR)
#
# =============================================================================
# WHAT EACH BRANCH LICENSES -- AND WHAT IT DOES NOT
# =============================================================================
# GroupNorm(32, C) for BatchNorm2d(C) is ONE code change, but it is not one
# mechanistic change: it removes batch statistics AND replaces per-channel with
# per-group normalisation (2/4/8/16 channels per group at 64/128/256/512), AND
# GN networks train differently (a different accuracy regime -- gn1's GN cell
# sat ~2.9 pp below its BN twin on CIFAR-10).  Hence:
#   GAP-REPLICATES  LICENSES "the scalar->layerwise gap does not need batch
#       statistics": it exists on a network that has none, with the same tensor
#       list and the same initial point.  Accounts (B) is refuted in its
#       batch-statistics form.  DOES NOT license "the carriers are the same
#       tensors on GN" (no isolation arm here -- that is the follow-on, which
#       nominates from THIS batch's PROBE_TENSOR records at no extra GPU cost),
#       nor anything about other normalisers (LayerNorm, no-norm).
#   GAP-ABSENT  LICENSES "at this cell, replacing BN with GN(32) removes the
#       gap".  DOES NOT license "batch statistics cause the gap": the swap also
#       changes per-channel invariance and the training regime, and no argmax is
#       located on GN (RULE 11 -- mistuning can suppress a gap; stamped
#       MISTUNING-NOT-EXCLUDED).  It promotes (B) to the leading candidate and
#       registers the next step: GN with num_groups == C (per-channel, no batch
#       statistics) to split batch statistics from per-channel invariance.
#   GAP-PARTIAL  present but attenuated: (A)/(B)/(C) not separated; mistuning
#       not excluded; the follow-on is the same GN(C) arm plus a tuning bracket.
#   HARNESS-UNSOUND  says nothing about the gap.  It reports on the harness (or
#       on GN failing to train at this cell under layerwise) and stops.
#

def fmt(x, nd=4):
    return "n/a" if x is None else ("%.*f" % (nd, x))


def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def compose_sigma(sigma_in):
    """O2.  The ONLY inputs are the frozen literal and cgn1's own in-batch sigma."""
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, val = max(cands, key=lambda kv: kv[1])
    return which, val, cands

# the model's parameter list, composed from ResNet18's definition (no torch).
# The launcher's guard 4d proves it BYTE-IDENTICAL to the manifest written from
# the live model, so the synthetic manifest cannot drift from reality.
# =============================================================================
def resnet18_c100_tensors():
    out = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "GroupNorm"),
           ("bn1.bias", 64, "GroupNorm")]
    cin = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for bi in range(2):
            stride2 = (li > 1 and bi == 0)
            p = "layer%d.%d." % (li, bi)
            a = cin if bi == 0 else w
            out += [(p + "conv1.weight", w * a * 9, "Conv2d"), (p + "bn1.weight", w, "GroupNorm"),
                    (p + "bn1.bias", w, "GroupNorm"), (p + "conv2.weight", w * w * 9, "Conv2d"),
                    (p + "bn2.weight", w, "GroupNorm"), (p + "bn2.bias", w, "GroupNorm")]
            if stride2:
                out += [(p + "shortcut.0.weight", w * a, "Conv2d"),
                        (p + "shortcut.1.weight", w, "GroupNorm"),
                        (p + "shortcut.1.bias", w, "GroupNorm")]
        cin = w
    out += [("linear.weight", 100 * 512, "Linear"), ("linear.bias", 100, "Linear")]
    return out

             "META_STEPS %d" % (EPOCHS * (50000 // BATCH))]
    for i, (n, q, o) in enumerate(t, 1):
        lines.append("TENSOR %d %s %d %s" % (i, n, q, o))
    lines.append("W512 %s" % ",".join("%d:%s" % x for x in zip(W512_IDX, W512_NAMES)))
    lines.append("ARMSPEC k01 SPEC scalar TYPE scalar M 1")

def read_corpus(csvpath):
    if not csvpath or not os.path.exists(csvpath):
        return None
    return [r for r in csv.DictReader(open(csvpath))
            if not r.get("run", "").startswith(PREFIX + "-")]


def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def read_runs(runsdir):
    """-> {(arm, seed): rec} from cgn1-<arm>-s<seed>-<jobid>.out; the LATEST job id
    (compared as an INTEGER) wins when a run was resubmitted."""
    got = {}
    if not runsdir or not os.path.isdir(runsdir):
        return got
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        txt = open(os.path.join(runsdir, fn), errors="replace").read()
        lines = txt.splitlines()
        eps = sorted((int(a), float(b), float(c)) for a, b, c in EPTR_RE.findall(txt))
        rec = {"file": fn, "job": jid, "arm": arm, "seed": seed,
               "epochs": [e[0] for e in eps],
               "traceback": "Traceback (most recent call last)" in txt,
               "done": any(ln.strip() == "RUN_DONE" for ln in lines),
               "args": [ln for ln in lines if ln.startswith("ARGS:")],
               "env": [ln for ln in lines if ln.startswith("ENV:")],
               "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
               "plateau5": None, "train5": None}
        tail = [e for e in eps if e[0] >= EPOCHS - 5]
        if len(tail) == 5 and [e[0] for e in tail] == list(range(EPOCHS - 5, EPOCHS)):
            rec["plateau5"] = mean([e[2] for e in tail])
            rec["train5"] = mean([e[1] for e in tail])
        prev = got.get((arm, seed))
        if prev is None or jid > prev["job"]:
            got[(arm, seed)] = rec
    return got


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for cand, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                      (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(cand):
            return cand, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    d = {"TENSOR": []}
    if not path or not os.path.exists(path):
        return None
    for ln in open(path):
        p = ln.split()
        if not p:
            continue
        if p[0] == "TENSOR" and len(p) >= 4:
            d["TENSOR"].append((int(p[1]), p[2], int(p[3]), p[4] if len(p) > 4 else None))
        else:
            d[p[0]] = " ".join(p[1:])
    return d


def is_complete(r):
    return (r is not None and r["epochs"] == list(range(EPOCHS)) and r["done"]
            and r["plateau5"] is not None)

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 clean branch, 1 harness-unsound, 2 incomplete."""
    print("=" * 78)
    print(" cgn1 -- IS THE SCALAR->LAYERWISE GAP A BATCHNORM PHENOMENON?")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    print(" arms: k01 = %s   kL = %s" % (SPEC["k01"], SPEC["kL"]))
    print(" PRIMARY: plateau5 = mean TEST accuracy over epochs %d..%d from the runs' own"
          % (EPOCHS - 5, EPOCHS - 1))
    print("          .out files.  TRAIN printed beside TEST everywhere.")
    print("=" * 78)
    rows = read_corpus(csvpath)
    print("corpus: %s  (DESCRIPTIVE ONLY -- no bar, sigma, branch or stamp reads it)"
          % ("not readable" if rows is None else "%d rows, %s- excluded" % (len(rows), PREFIX)))

    runs = read_runs(runsdir)
    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    # ---- completeness ---------------------------------------------------------
    print("\nCOMPLETE  6 runs, epochs 0..%d, RUN_DONE" % (EPOCHS - 1))
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = is_complete(r)
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                                           "absent" if r is None else
                                           "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s"
                                           % (len(r["epochs"]), r["done"], r["traceback"],
                                              fmt(r["plateau5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/6 | missing %s" % (6 - len(missing), ",".join(missing))
        print("\n" + final)
        print("No branch is taken and no number here is quotable.")
        return 2, final

    # ---- HARNESS GATES, FIRST ---------------------------------------------------
    print("\nH-ARGS    every run's OWN ARGS line")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "H-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s)}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "H-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad])
                 if (dup or bad) else "NN-name %s, %s, seed %d" % (NET, SPEC[a], s))
    print("\nH-ENV     the environment rode the ENV line on every run")
    envs = sorted({re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]})
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),
         "H-ENV one ENV line per run, one distinct across runs", "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
    for tok in ("AUGMENT=%s" % AUG, "BETA_CLIP=%s" % CLIP, "PROBE=%d" % PROBE, "HIER=none", "SCHED=none"):
        hchk(len(envs) == 1 and (" " + tok + " ") in (envs[0] + " "), "H-ENV carries %s" % tok)
    ptbad = [k for k, r in runs.items()
             if len(r["pt"]) != 1 or not PT_RE.match(r["pt"][0])
             or PT_RE.match(r["pt"][0]).group(1) != SPEC[k[0]]]
    hchk(not ptbad, "H-ENV every run printed `PROBE_TENSOR: on every=%d type=<its spec> tensors=%d`"
         % (PROBE, NTENS), "bad: %s" % ptbad if ptbad else "6/6")

    print("\nH-STRUCT  the batch's own PARTITION-MANIFEST.txt")
    mp, how = resolve_manifest(runsdir, manifest_path)
    man = read_kv(mp)
    if man is None:
        hchk(False, "H-STRUCT manifest present", how)
    else:
        print("  manifest: %s (%s)" % (mp, how))
        t = man["TENSOR"]
        hchk(man.get("NETWORK") == NET, "H-STRUCT NETWORK %s" % NET, str(man.get("NETWORK")))
        hchk(len(t) == NTENS and [x[0] for x in t] == list(range(1, NTENS + 1)),
             "H-STRUCT %d tensors, 1-based" % NTENS, str(len(t)))
        hchk(sum(x[2] for x in t) == TOTPAR, "H-STRUCT %d parameters" % TOTPAR, str(sum(x[2] for x in t)))
        hchk(man.get("NORM") == "GroupNorm %d BatchNorm2d 0" % N_GROUPNORM,
             "H-STRUCT NORM GroupNorm %d BatchNorm2d 0" % N_GROUPNORM, str(man.get("NORM")))
        own_bad = [x[0] for x in t if x[0] <= 60 and x[0] % 3 != 1 and x[3] != "GroupNorm"]
        hchk(not own_bad, "H-STRUCT every scale/shift at 1..60 is owned by GroupNorm",
             "violations %s" % own_bad[:6])

        print("        seeds: %s" % ", ".join("s%d %.4f (train %.4f)" % (s, x, y)
                                              for s, x, y in zip(SEEDS, v, t)))

    hchk(kL >= KL_MIN, "H-TRAINS mean(kL) >= %.1f pp (the GN net trains under layerwise)" % KL_MIN,
         "%.4f" % kL)
    hchk(max(kL, k01) <= CEIL_MAX, "H-CEIL max arm mean <= %.1f pp" % CEIL_MAX,
         "%.4f" % max(kL, k01))

    if harness:
        final = "FINAL: HARNESS-UNSOUND | %d harness gate(s) failed: %s" % (
            len(harness), "; ".join(h.split()[0] for h in harness))

    # ---- sigma (O2) --------------------------------------------------------------
    ss = sum(sum((x - mean(arm_v[a])) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    which, sigma_used, _c = compose_sigma(sigma_in)

    gap_bar = max(GAP_BAR, K_SE * se)
    rev_bar = max(NULL_BAR, K_SE * se)

    stamps.append("K01-%s-BN-FLOOR-BAND" % ("BELOW" if k01 < BN_SCALAR_BAND[0] else
                                            "IN" if k01 <= BN_SCALAR_BAND[1] else "ABOVE"))

    pr = [os.path.isdir(os.path.join(runsdir, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s)))
          and bool(os.listdir(os.path.join(runsdir, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s))))
          for s in SEEDS for a in ARMS]
    stamps.append("PROBE-RECORDS-PRESENT" if all(pr) else "PROBE-RECORDS-MISSING-%d" % pr.count(False))
    final = "FINAL: %s | %s" % (branch, " | ".join(stamps))
    print("\n" + "=" * 78)
    print(final)
    print("=" * 78)
    print("\nWHAT THIS LICENSES -- as registered, before any run existed:")
    lic = {
        "GAP-REPLICATES": ["The gap does NOT need batch statistics: it exists on a network with",
                           "none, the same tensor list and the same initial point.  Account (B) in",
                           "its batch-statistics form is refuted.  NOT licensed: that GN's carriers",
                           "are ResNet's 50/53/59 (the follow-on isolation nominates from THIS",
                           "batch's PROBE_TENSOR records), or anything about other normalisers."],

                       "statistics cause the gap' -- the swap also changes per-channel",
                       "invariance and the training regime, and no argmax is located on GN",

                       "GroupNorm(num_groups == C) to split batch statistics from per-channel",

def _mk_out(d, arm, seed, jid, level, train, n_ep=EPOCHS, net=NET, done=True, env=None, spec=None,
            tensors=NTENS):
    rn = "%s-%s-s%d" % (PREFIX, arm, seed)
    spec = spec or SPEC[arm]
    pdir = os.path.join(d, PREFIX, "probe_" + rn)
    L = ["NODE=[HOST_45cba27b74] | JOB=%s %d | AUGMENT=1" % (rn, jid),
         "ARGS: --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 "
         "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 "
         "--dataset CIFAR100 --NN-name %s --batch-size 100 --max-time 999:00:00 --gamma 1 "
         "--meta-stepsize 1e-3 --alpha0 1e-6 --num-epochs 100 --stepsize-groups %s --seed %d "
         "--save-directory %s --run-name %s" % (net, spec, seed, os.path.join(d, PREFIX), rn),
         env or (ENV_TMPL % pdir), "NVIDIA L4, 23034 MiB",
         "PROBE_TENSOR: on every=100 type=%s tensors=%d meta_alg=Lion momentum_param=0.99 "
         "Lion_beta2=0.9 dir=%s" % (spec, tensors, pdir)]
    jit = (0.02, -0.01, 0.03, -0.03, -0.01)          # sums to 0: plateau5 == level exactly
    for e in range(n_ep):
        if e >= EPOCHS - 5:
            te, tr = level + jit[e - (EPOCHS - 5)], train + jit[e - (EPOCHS - 5)]
        else:
            te, tr = level * (e + 1) / EPOCHS, train * (e + 1) / EPOCHS
        L.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, tr, te))
    if done:
        L += ["44  minutes", "RUN_DONE"]
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(L) + "\n")
    os.makedirs(pdir, exist_ok=True)
    open(os.path.join(pdir, "rec_000.npz"), "w").write("x")


def _mk_batch(d, lv, manifest=None, prov_sha=POST_SHA, net=NET, env_bad=False, drop=None,
              trunc=None, dupjob=None):
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(
        manifest if manifest is not None else synthetic_manifest_text())
    open(os.path.join(d, PREFIX, "PROVENANCE.txt"), "w").write(
        "MODE submit\nBUILD_NETWORK_SHA256 %s\n" % prov_sha)
    jid = 10000
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            if drop == (a, s):
                continue
            env = None
            if env_bad and (a, s) == (ARMS[0], SEEDS[0]):
                env = ENV_TMPL.replace("AUGMENT=1", "AUGMENT=0") % "/x"
            n_ep = 97 if trunc == (a, s) else EPOCHS
            _mk_out(d, a, s, jid, lv[a][i], min(99.9, lv[a][i] + 20.0), n_ep=n_ep, net=net,
                    done=(trunc != (a, s)), env=env)
            if dupjob == (a, s):
                _mk_out(d, a, s, 999, 1.0, 1.0)          # an OLDER job (999 < 10000 as integers)

def _run_score(d, csvpath):
    saved = GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()


def pooled_sigma(rows, keep):
    cells = {}
    for r in rows:
        if keep(r) and r.get("superseded") == "0" and r.get("collapsed") == "0" \
                and r.get("complete") == "1" and r.get("plateau5", "").strip():
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss = df = nc = nm = 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm


def std_cell(r, net):
    return (r.get("network") == net and r.get("dataset") == DSET
            and r.get("granularity") in ("scalar", "layerwise")
            and r.get("epochs_requested") == str(EPOCHS) and r.get("augment") == AUG
            and r.get("beta_clip") == CLIP and r.get("meta_stepsize") == MST
            and r.get("alpha0") == A0 and r.get("batch_size") == str(BATCH))


def inbatch_gaps(rows, net):
    by = {}
    for r in rows:
        if std_cell(r, net) and r.get("hier") in ("none", "") and r.get("superseded") == "0" \
                and r.get("collapsed") == "0" and r.get("complete") == "1" and r.get("plateau5", "").strip():
            by.setdefault(r["run"].split("-")[0], {}).setdefault(r["granularity"], []).append(float(r["plateau5"]))
    out = {}
    for b, v in by.items():
        if len(v.get("scalar", [])) >= 3 and len(v.get("layerwise", [])) >= 3:
            s, l = mean(v["scalar"]), mean(v["layerwise"])
            out[b] = (s, l, l - s, (l - s) / (100 - s))
    return out


def median(v):
    v = sorted(v)
    n = len(v)
    return None if not n else (v[n // 2] if n % 2 else (v[n // 2 - 1] + v[n // 2]) / 2)

        sN = pooled_sigma(rows, lambda r: std_cell(r, REF_NET))

        band = [float(r["plateau5"]) for r in rows if std_cell(r, REF_NET) and r["granularity"] == "scalar"

        n_new = sum(r.get("network") == NET for r in rows)

    for acc in ("A", "B", "C"):
        for arm in ARMS:

                  % (acc, arm, lo, hi, lo - BN_SCALAR_BAND[1], hi - BN_SCALAR_BAND[1],
                     100 - hi, 100 - lo, status))
            if not anchor:
                chk(lo - BN_SCALAR_BAND[1] >= 7.0 and 100 - hi >= 30.0,
                    "(%s) %s predicted >= 7 pp above the floor band and >= 30 pp below 100" % (acc, arm))
    chk(all(PRED[a]["kL"][0] - BN_SCALAR_BAND[1] >= 25.0 for a in PRED),
        "kL (the free arm of D) is >= 25 pp above the floor band under EVERY account -> D bounded in neither direction")

    print("\nE. THE REAL score() ON SYNTHETIC RUNS (invented numbers; nothing about the real batch)")
    live_man = os.path.join(runsdir, PREFIX, "PARTITION-MANIFEST.txt") if runsdir else ""

    else:
        note("E0 no live manifest under %s/%s -- the launcher's guard 4d makes the byte comparison" % (runsdir, PREFIX))
    L = {

        "REV": {"k01": (64.0, 64.3, 63.8), "kL": (50.0, 50.5, 49.6)},

        "NORM BatchNorm2d 20 BatchNorm2d 0", "NORM GroupNorm 0 BatchNorm2d 20")
