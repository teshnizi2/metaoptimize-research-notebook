#!/bin/bash
# =============================================================================
# cIS2_isolate_horizon.sh -- `ciso2`: DOES THE H-ISOLATE RESCUE SURVIVE THE
#   59-GROUP'S PINNING?  ONE SUBMISSION, 12 JOBS (4 arms x 3 seeds), 250
#   EPOCHS, SEEDS {31,32,33}, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account, nothing is submitted to `cluster_a`, and every guard
# and scancel line below is filtered on the `ciso2-` prefix alone.
#
# REGISTERED SCORER: analysis/cIS2_ciso2_horizon_score.py, git-committed BEFORE
# this batch is submitted (STANDING RULE 21; the proof is the commit time
# against the earliest sacct Submit, stated in CORRECTIONS with the margin in
# seconds).  The question, the horizon derivation, every bar in pp AND in SE
# from a noise floor re-derived at registration with ciso2's own rows excluded
# from every reader, the branch map, the UNRESOLVED-NOT-PINNED branch and the
# point predictions are frozen there.  --selftest re-derives all of it and, on
# this host, ciso1's own plateau5 and beta premises from the raw records.
#
# THE DOCUMENTED SCORER INVOCATION -- no second argument to forget:
#     python3 analysis/cIS2_ciso2_horizon_score.py <runsdir>
#
# -----------------------------------------------------------------------------
# THE ARMS.  m = 2 group ORDER: the big group is beta[0], the isolated group
# beta[1].  1-BASED indices, verified BY NAME on the live model by guard 4,
# which prints the full manifest per arm.  The four spec strings are BYTE
# IDENTICAL to ciso1's, so this is the SAME intervention at a longer horizon.
#   k01   scalar                                     floor anchor, in batch
#   k62   layerwise                                  ceiling anchor, in batch
#   ISO   sets:<59 others>/50,53,59   [59,3]         THE ARM
#   ONE   sets:<61 others>/50         [61,1]         THE TIMING ARM
#
# CTRL IS DELIBERATELY NOT CARRIED.  At 100 epochs ciso1 measured CTRL
# indistinguishable from k01 (-0.0633 pp = -0.08 SE), reaching the floor at the
# SAME record (184) with seed-pair beta differences of 0.006-0.108 nats.  A
# second floor arm buys nothing this batch asks; the identity-vs-depth question
# it half-answers belongs to the DEPTH arm (CORRECTIONS 187.4).  The three
# saved jobs pay for the longer horizon.  STATED AS A TRADE.
#
# WHY ONE IS CARRIED.  ONE's 61-group pins 43-77 epochs LATER than ISO's
# 59-group and ONE scores 5.4613 pp LOWER at 100 epochs.  If pinning destroys
# the rescue, the later-pinning arm must end HIGHER -- a CROSSOVER.  If the
# rescue survives, ORDER-HELD.  That is a within-batch, falsifiable timing
# contrast, and it is the only reason a fourth arm is here.
#
# -----------------------------------------------------------------------------
# THE HORIZON, DERIVED FROM THE MEASURED DESCENT (re-derived by an independent
# parser from ciso1's own 1,500 probe records before anything was committed):
#   ISO beta[0] terminal -14.3847 / -14.9450 / -14.7329, slope (OLS last 50
#     records) -0.02645 / -0.03580 / -0.02925 per record  ==> pins at epoch
#     104.7 / 100.3 / 101.8.  The descent is ACCELERATING, so that is an upper
#     bound.  ISO's ISOLATED beta[1] is already pinned from epoch 22.4-23.6, so
#     when the 59-group pins EVERY tensor is on exp(-15) = 3.06e-07.
#   ONE beta[0] terminal -11.8497 / -11.9077 / -11.7617, slope -0.01323 /
#     -0.01025 / -0.00789 ==> pins at epoch 147.6 / 160.3 / 182.1 (worst case).
#   ciso1's floored arms pin at epoch 36.8 and their trailing-5 test accuracy is
#     inside +-0.5 pp of terminal from epoch 37 and FLAT for 63.2 epochs after.
#     63.2 epochs is the only post-pin window this campaign has measured.
#   ==> E = 250.  ONE gets 250 - 182.1 = 67.9 post-pin epochs (>= 63.2); ISO
#       gets 145.3.  E = 240 gives ONE only 57.9 and is rejected.  E = 772
#       (the cts3 horizon) costs 3.1x and decides nothing extra, because the
#       accuracy consequence of pinning here is measured to be IMMEDIATE.
#
# THE 100-EPOCH CONTROL IS IN BATCH, IN THE SAME RUNS, AT ZERO COST, and guard
# 4h PROVES the precondition on the LIVE train.py instead of asserting it:
# args.num_epochs occurs in exactly TWO places (the argparse declaration and
# `for epoch in range(args.num_epochs)`), there is no schedule / total-step
# count / warmup keyed to it, SCHED=none, and --max-time 999:00:00 keeps
# train.py's own time break from firing.  So epochs 95-99 of a 250-epoch run
# ARE a 100-epoch run of an identical configuration -- and a WITHIN-RUN paired
# control, which twelve separate 100-epoch jobs could not buy at any price.
# (cts3 / cpk2 / cpk3 precedent, CORRECTIONS 152 / 158 / 161.)
#
# WHY FRESH SEEDS 31/32/33.  The corpus carries NO row at any network with seed
# 31, 32 or 33 (guard 2c proves it).  The free 100-epoch readout is therefore an
# INDEPENDENT seed triple, not a re-run of ciso1's 21/22/23 -- so it also tests
# whether ciso1's headline replicates.  That comparison is BETWEEN BATCH and the
# scorer stamps it NON-GATING: BATCH is the unit of replication (F(62,85)=5.47),
# and no ciso1 arm enters any ciso2 contrast.
#
# -----------------------------------------------------------------------------
# WALLTIME AND PARTITIONS -- MEASURED ON THIS BATCH'S OWN PREDECESSOR
# Per-epoch cost of THIS cell from ciso1's 15 runs (100 epochs each):
#   [HOST_89e2c8be67] / 880-887 / 863 (the 7-day pool)   37-48 min = 22.2-28.8 s/epoch
#   [HOST_3c1b003035] (2080ti, reached only via gpu-short) 81-94 min = 48.6-56.4 s/epoch
#   corpus-worst on the 7-day pool (CORRECTIONS 152's table)      36.6 s/epoch
#   corpus-worst on the 2080ti nodes                              70.8 s/epoch
# 250 epochs on a 2080ti at 70.8 s/epoch is 4.92 h against gpu-short's 4:00:00
# MaxTime, so gpu-short is excluded BY ARITHMETIC: WALL=05:00:00 makes
# bin/_lib_guards.sh's own composer drop it, and NO PARTS_NO_SHORT_REASON
# override is used.  WALL is 1.97x the worst rate ever observed on the three
# 7-day partitions and 2.82x the expected rate.
# COST: 12 jobs x 250 epochs.  expected 25.5 s/ep -> 21.3 GPU-h;
#       ciso1-worst 28.8 -> 24.0 GPU-h; corpus-worst 36.6 -> 30.5 GPU-h.
# QUEUE DEPTH IS CHECKED BEFORE COMMITTING by guard 5b, not assumed.
#
# RULE 20.  Every value below is ONE shell token of [A-Za-z0-9_.,:/-].
# analysis/argsline_guard.py IS NOT EDITED.  BETA_CLIP, PROBE and PROBE_TENSOR
# are ENVIRONMENT variables; the first two are audited from the runner's ENV
# line, PROBE_TENSOR from the `PROBE_TENSOR: on` line the patch prints.
#
# USAGE   bash bin/cIS2_isolate_horizon.sh            # dry run, submits nothing
#         bash bin/cIS2_isolate_horizon.sh --submit
# =============================================================================

clip_c, epochs, bsz = sys.argv[5], int(sys.argv[6]), int(sys.argv[7])

sys.path.insert(0, cif); os.chdir(cif)
import torch
from build_network import build_network
from Optimizers.HF import HF
net = build_network("ResNet18_c100", "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]
numel = [int(p.numel()) for _, p in net.named_parameters()]
T, tot = len(nps), sum(numel)
bad = 0
print("guard 4: live ResNet18_c100 -- %d parameter tensors, %d parameters" % (T, tot))
if T != ntens or tot != totpar:
    print("!!! guard 4: live model is %d / %d, the batch assumes %d / %d" % (T, tot, ntens, totpar)); raise SystemExit(1)
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot, "CLIP_C %s" % clip_c, "EPOCHS %d" % epochs,

    lines.append("TENSOR %d %s %d" % (i + 1, names[i], numel[i]))

                                         ("k62", "layerwise", "layerwise", (T,))):
    o = HF.__new__(HF); o.num_layers = T; o._device = torch.device("cpu")
    HF.init_meta(o, spec, nps, 1e-6)
    shapes = [tuple(b.shape) for b in o.beta]
    print("guard 4: %-4s %-10s init_meta: stepsize_type %s  beta shapes %s  rednorm %s"
          % (arm, spec, o.stepsize_type, shapes, getattr(o, "_rednorm", None)))
