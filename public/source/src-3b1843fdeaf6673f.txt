#   AND (ii) BASE WEIGHT DECAY 0?
#   ONE SUBMISSION, 27 JOBS (9 arms x 3 seeds), 100 EPOCHS, SEEDS {108,109,110} (111 of the pre-assigned block is NOT
#   used), ResNet18_c100 / CIFAR-100 at ciso1's cell, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM, the LIVE shared harness

#     python3 analysis/cMO1_momwd_score.py <runsdir>
# --csv and --manifest are OPTIONAL and DEFAULTED.
#
# FROZEN BARS (O2).  Every bar, sigma and tolerance score() uses is a LITERAL here; score() reads the corpus ONLY for a
# row-count disclosure.  The only data-dependent term that can enter a bar is SIGMA_INBATCH through compose_sigma()
# (ONE argument).  --selftest proves the FINAL line invariant across four corpora.  The noise floor was read through
# analysis/corpus_exclusions.filter_rows (OPERATIONS 36).
#
# THE ARMS (ISO = ciso1's spec, byte-identical; kL = `layerwise`, 62 groups)
#   arm     stepsize-groups  --momentum-param-base  --weight-decay-base
#   k01     scalar           0.99                   0.1        ANCHOR (the mechanism cell, in batch)
#   kL      layerwise        0.99                   0.1        ANCHOR reference
#   ISO     ISO              0.99                   0.1        ANCHOR positive control
#   M9k01   scalar           0.9                    0.1        momentum factor
#   M9kL    layerwise        0.9                    0.1
#   M9ISO   ISO              0.9                    0.1
#   W0k01   scalar           0.99                   0          weight-decay factor
#   W0kL    layerwise        0.99                   0
#   W0ISO   ISO              0.99                   0
#   Everything else as ciso1 / cdep1 / cvt7 / cvt8: Lion meta (0.99, 0.9, wd 0), meta step 1e-3, alpha0 1e-6, gamma 1,
#   BETA_CLIP -15:-2.3026, AUGMENT 1, batch 100.  The base SGDm here is the EMA form m <- mu m + (1-mu) g (live HF.py
#   SGDm_base_update), so 0.9 vs 0.99 changes the averaging window, NOT the effective step scale (no 1/(1-mu)).
#   Weight decay 0 removes WD from EVERY tensor (BN gammas included) AND from the meta trace h <- gamma(1-wd a)h - delta:
#   W0 cannot separate those two routes.
#
# THE FLAG WITNESS.  No .out line prints the base momentum or WD.  train.py writes vars(args) AFTER argparse into the
# run's own TensorBoard events file (`|momentum_param_base|0.9|`); G-TBARGS reads those bytes for every run.  A flag that
# never reached args (the BROKEN-FLAG null, which predicts every X arm == its anchor arm, i.e. the ROBUST branch
# exactly) FAILS G-TBARGS before any level is read.  args -> HF is plain code (Optimizers/build_optimizer.py
# args_base comprehension; HF.SGDm_base_update), proved before launch by tests/test_cmo1_flags_bite.py on the LIVE files.
#
# CO-PRIMARY (in batch; bounded in neither direction), X in {M9, W0}:
#   G_X  = XkL - Xk01      the scalar-vs-layerwise gap at X
#   R_X  = XISO - Xk01     the carrier-isolation rescue at X
#   L_X  = Xk01 - k01      how far the factor lifts the scalar arm
# KEY: G_A = kL - k01; R_A = ISO - k01; IG_X = G_X - G_A; IR_X = R_X - R_A; DL_X = XkL - kL; DI_X = XISO - ISO.
#
# BRANCH MAP (decide(); first match wins):
#   UNRESOLVED-DIVERGED   an ANCHOR arm's seed range > 5 pp
#   ANCHOR-FAILED         k01 > 30, or kL - k01 < 20, or ISO - k01 < 20 (the mechanism cell did not reproduce in batch)
#   otherwise TWO halves "M9:<C>/<I>+W0:<C>/<I>" (a half whose own arm diverged reads X:DIVERGED):
#     C  XkL < 45                                  -> REF-UNHEALTHY
#        XkL - Xk01 <= 5                           -> COLLAPSE-IS-CONFIG     (k01 recovers to within 5 pp of layerwise)
#        Xk01 <= 0.5 XkL and XkL - Xk01 >= 20      -> COLLAPSE-PERSISTS      (the census's R50 plus the 20 pp gap bar)
#        otherwise                                 -> COLLAPSE-ATTENUATED
#     I  (C is REF-UNHEALTHY or COLLAPSE-IS-CONFIG, or XkL - Xk01 <= 10) -> ISO-UNREADABLE
#        XISO >= XkL - 5                           -> ISO-RESCUES
#        XISO <= Xk01 + 5                          -> ISO-FAILS
#        otherwise                                 -> ISO-PARTIAL
# =============================================================================

def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

def r18_tensors():
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if stride != 1 or cin != w:
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


NAMES = [n for n, _q, _o in r18_tensors()]


def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out

    return "\n".join(lines) + "\n"

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "sw": [ln for ln in lines if ln.startswith(SWITCH_PREFIXES)],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest job id among COMPLETE runs wins."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s" % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def tb_values(runsdir, arm, s):
    """The run's own TensorBoard events file(s): key -> sorted set of values written by train.py's vars(args) table."""
    d = os.path.join(runsdir, PREFIX, "Tensorboard_outputs", run_name(arm, s))
    files = sorted(glob.glob(os.path.join(d, "events.out.tfevents.*")))
    if not files:
        return None, 0
    out = {}
    for f in files:
        data = open(f, "rb").read()
        for k in expected_tb(arm, s):
            for v in re.findall(rb"\|" + re.escape(k.encode()) + rb"\|([^|\n]*)\|", data):
                out.setdefault(k, set()).add(v.decode("utf-8", "replace"))
    return out, len(files)


def tb_check(runsdir, arm, s):
    got, nf = tb_values(runsdir, arm, s)
    if got is None:
        return False, "no events file"
    want = expected_tb(arm, s)
    bad = ["%s=%s (want %s)" % (k, sorted(got.get(k, [])), v) for k, v in want.items() if got.get(k) != {v}]
    return not bad, ("%d events file(s); " % nf) + ("; ".join(bad) if bad else "every registered key once-valued and right")


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r):
    return (r.get("epochs_requested") == str(EPOCHS) and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0 and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
    return (math.sqrt(ss / df) if df else None), df, nc

def collapse_word(k, l):
    if l < REF_MIN:
        return "REF-UNHEALTHY"
    if l - k <= MATCH_BAR:
        return "COLLAPSE-IS-CONFIG"
    if k <= R50 * l and l - k >= GAP_MIN:
        return "COLLAPSE-PERSISTS"
    return "COLLAPSE-ATTENUATED"


def iso_word(k, l, i, cw):
    if cw in ("REF-UNHEALTHY", "COLLAPSE-IS-CONFIG") or l - k <= RESCUE_BAR:
        return "ISO-UNREADABLE"
    if i >= l - MATCH_BAR:
        return "ISO-RESCUES"
    if i <= k + MATCH_BAR:
        return "ISO-FAILS"
    return "ISO-PARTIAL"


def half_word(M, X, diverged):
    if any((X + b) in diverged for b in BASEARMS):
        return "%s:DIVERGED" % X
    k, l, i = M[X + "k01"], M[X + "kL"], M[X + "ISO"]
    cw = collapse_word(k, l)
    return "%s:%s/%s" % (X, cw, iso_word(k, l, i, cw))


def decide(M, diverged=()):
    """M: arm -> mean plateau5; diverged: the arms whose seed range exceeds DIVERGED_BAR."""
    if any(a in diverged for a in ANCHORS):
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX or M["kL"] - M["k01"] < GAP_MIN or M["ISO"] - M["k01"] < GAP_MIN:
        return "ANCHOR-FAILED"
    return "+".join(half_word(M, X, diverged) for X in FACTORS)

def licence_lines(branch):
    if branch in LICENSE:
        return LICENSE[branch]

    if len(halves) != 2:
        return None
    out = []
    for X, h in zip(FACTORS, halves):

            return None
        body = h[len(X) + 1:]
        sub = {"X": X, "F": FACTOR_TEXT[X], "V": VALUE_TEXT[X], "R": ROUTE_TEXT[X]}
        if body == "DIVERGED":
            out += [ln.format(**sub) for ln in LICENSE_C["DIVERGED"]]
            continue
        if "/" not in body:
            return None
        c, i = body.split("/", 1)

            return None
        out += [ln.format(**sub) for ln in LICENSE_C[c]] + [ln.format(**sub) for ln in LICENSE_I[i]]

            "(ResNet18_c100, CIFAR-100, Lion meta, meta step 1e-3, alpha0 1e-6, clip, 100 epochs).  Readings at k01 are LOCATIONS."]
    return out

            for i in ("ISO-RESCUES", "ISO-FAILS", "ISO-PARTIAL", "ISO-UNREADABLE"):

        for c in ("COLLAPSE-IS-CONFIG", "REF-UNHEALTHY"):

        halves[X] = hs
    for a in halves["M9"]:

            out.add("%s+%s" % (a, b))

    qual = [(M[x] - M[y], T[x] - T[y]) for _n, x, y in C]

def descriptive_probe_readout(runsdir):
    """NON-GATING: on each scalar arm, each carrier's share of sum |z_tensor| over the 500 records, and the top 5 tensors."""
    print("\nDESCRIPTIVE (NON-GATING, after FINAL): share of sum_t |z_tensor| per tensor on the scalar arms, mean over seeds")
    for a in ("k01", "M9k01", "W0k01"):
        tot = None
        n = 0
        for s in SEEDS:
            p = os.path.join(runsdir, PREFIX, "probe_%s" % run_name(a, s), "probe.jsonl")
            if not os.path.exists(p):
                continue
            acc = [0.0] * NTENS
            for ln in open(p):
                try:
                    z = json.loads(ln).get("z_tensor")
                except ValueError:
                    continue
                if z and len(z) == NTENS:
                    for j, x in enumerate(z):
                        if x is not None and math.isfinite(x):
                            acc[j] += abs(x)
            S = sum(acc)
            if S > 0:
                sh = [x / S for x in acc]
                tot = sh if tot is None else [u + v for u, v in zip(tot, sh)]
                n += 1
        if not n:
            print("  %-6s no probe records" % a)
            continue
        tot = [x / n for x in tot]
        top = sorted(range(NTENS), key=lambda j: -tot[j])[:5]
        print("  %-6s (%d seeds) carriers %s   top5 %s" % (a, n, " ".join("%d:%.3f" % (i, tot[i - 1]) for i in CARRIER_IDX),
                                                          " ".join("%d:%s:%.3f" % (j + 1, NAMES[j], tot[j]) for j in top)))

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

            chk(got == want, "account %-42s -> %s" % (acc, want), "" if got == want else "got %s" % got)
        lo = dict((a, pl[0]) for a, pl in lv.items())
        hi_ = dict((a, pl[1]) for a, pl in lv.items())
        note("  corners: low %s | high %s" % (decide(lo, ()), decide(hi_, ())))

    print("\nE. THE REAL score() ON SYNTHETIC HARNESS-FORMAT RUNS")
    tmp = tempfile.mkdtemp(prefix="cmo1_selftest_")
    variants = _e_csv_variants(csvpath, tmp)
    results = {}
    cases = [(acc, dict(levels=_levels_of(acc)), EXPECTED_BRANCH[acc], 0 if EXPECTED_BRANCH[acc] != "FLAGS-NOT-VERIFIED" else 1)
             for acc in ACCOUNTS if not acc.startswith("BROKEN-FLAG")]
    robust = _levels_of("ROBUST (collapse and rescue need neither)")
    cases += [
        ("BROKEN-FLAG null: M9k01-s109 witness 0.99", dict(levels=robust, tb_break=(("M9k01", 109), {"momentum_param_base": "0.99"})),
         "FLAGS-NOT-VERIFIED", 1),
        ("BROKEN-FLAG null: W0ISO-s110 witness 0.1", dict(levels=robust, tb_break=(("W0ISO", 110), {"weight_decay_base": "0.1"})),

        ("events file missing on kL-s108", dict(levels=robust, tb_drop=("kL", 108)), "FLAGS-NOT-VERIFIED", 1),

         "FLAGS-NOT-VERIFIED", 1),
        ("a DROPPED run", dict(levels=robust, drop=("W0kL", 109)), "INCOMPLETE", 2),
        ("a shadow partial file does not replace the complete run", dict(levels=robust, shadow=("M9ISO", 108)),
         EXPECTED_BRANCH["ROBUST (collapse and rescue need neither)"], 0),

        ("ARGS break: repeated --weight-decay-base on W0k01-s109", dict(levels=robust, args_break=(("W0k01", 109), expected_args("W0k01", 109)
                                                                                                    + " --weight-decay-base 0.1")), "HARNESS-UNSOUND", 1),
        ("a switch line (DECAY_MASK) on ISO-s108", dict(levels=robust, extra_break=(("ISO", 108), ["DECAY_MASK: on x"])), "HARNESS-UNSOUND", 1),
        ("manifest break", dict(levels=robust, manifest=synthetic_manifest_text().replace("layerwise TYPE", "scalar TYPE", 1)), "HARNESS-UNSOUND", 1),
        ("provenance: a patched HF", dict(levels=robust, prov={"hf": "0" * 64}), "HARNESS-UNSOUND", 1),
        ("provenance: dry-run mode", dict(levels=robust, prov={"mode": "dry-run"}), "HARNESS-UNSOUND", 1),
        ("an anchor arm bimodal", dict(levels=robust, over={("k01", 110): 60.0}), "UNRESOLVED-DIVERGED", 0),
        ("a W0 arm bimodal", dict(levels=robust, over={("W0k01", 110): 60.0}),
         "M9:COLLAPSE-PERSISTS/ISO-RESCUES+W0:DIVERGED", 0),
        ("M9 small gap (attenuated, ISO unreadable)", dict(levels=dict(robust, M9k01=62.0, M9kL=70.0, M9ISO=70.0)),
         "M9:COLLAPSE-ATTENUATED/ISO-UNREADABLE+W0:COLLAPSE-PERSISTS/ISO-RESCUES", 0),
        ("every arm under the floor", dict(levels=dict((a, 5.0) for a in ARMS)), "HARNESS-UNSOUND", 1),
    ]
    for label, kw, want, wrc in cases:
        d = os.path.join(tmp, re.sub(r"[^A-Za-z0-9]+", "_", label)[:60])
        os.makedirs(d)
        _e_build(d, with_probe=(label.startswith("ROBUST")), **kw)
        rc, fin, out = _e_score(d, variants.get("real", variants["minimal-invented"]))
        results[label] = (rc, fin, out, d)
        got = fin.split(" |")[0][len("FINAL: "):] if fin and fin.startswith("FINAL: ") else fin
        chk(rc == wrc and got == want, "E %-58s -> %s (rc %d)" % (label[:58], want, wrc), "" if (rc == wrc and got == want) else "got rc %s %s" % (rc, fin))
    r0 = results["ROBUST (collapse and rescue need neither)"]

def main():
    ap = argparse.ArgumentParser(description="cmo1 registered scorer.  Documented invocation: "
                                             "python3 analysis/cMO1_momwd_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cmo1/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--derive-floor", action="store_true", help="print the filtered / naive ResNet18_c100 floor on --csv and exit")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.derive_floor:
        (sg, dfg, ncg), (sn, dfn, _n), nr, nn = derive_floor(a.csv)
        print("SIGMA_R18ALL FILTERED %r df %d cells %d rows %d | NAIVE %r df %d rows %d | csv sha256 %s"
              % (sg, dfg, ncg, nr, sn, dfn, nn, file_sha(a.csv)))
        return 0
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
