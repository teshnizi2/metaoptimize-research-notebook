#!/bin/bash
# =============================================================================
# Cycle 53 -- THE BUDGET REPLICATION, BOX-FREE BY CONSTRUCTION.  cluster_b.  12 jobs.
# Tag `br6-*`.
#
# OPEN QUESTION IT ANSWERS, and it is now the campaign's biggest one.
#
#   `bl5-*` was registered as the LAST untested threat to the headline.  It did
#   not merely fail to clear the threat -- it found one.  On the two rungs that
#   were box-free on every seed and needed NO salvage of any kind (FINDINGS 53.4):
#
#       rung   20 ep (cl5 cU)   40 ep (bl5)    delta     bar     verdict
#       lay          0.7262        0.6912     -0.035    0.10   STATIONARY
#       node         0.4056        0.2774     -0.128    0.10   **DRIFTING**
#
#   and on the weightwise rung -- the one CORRECTIONS 59's headline is quoted at --
#   the fall is far larger, in a window that is box-free on every seed:
#
#       window            epochs      N_eff/m (w)   vs 20-ep control
#       bl5 0.25-0.5       10-20           0.5191          +0.013     box-free
#       bl5 0.5-0.759   20.0-30.4          0.2634          -0.243     box-free
#
#   IF THAT REPLICATES, "at the adapted operating point per-weight meta-gradients
#   carry 16%-55% of the independent information their count implies" IS A
#   20-EPOCH SENTENCE, and the 4-family range needs a budget column exactly as it
#   nearly needed a box one.  That is a correction to the campaign's headline and
#   it must not rest on a post-hoc window.
#
# WHY IT CANNOT REST ON `bl5`, stated plainly.
#   bl5's B0.3 FAILED on the weightwise rung: seed 0 hit the HI guard at record
#   3037 (epoch 30.4) and spent 96.3% of its last quarter there.  The registration
#   says an arm that fails B0.3 has an UNINTERPRETABLE B1, and **that verdict
#   stands; it is not overturned here.**  Everything quoted above is either a
#   clean rung (node, lay -- 3/3 seeds box-free, nothing post-hoc) or explicitly
#   labelled POST-HOC.  This batch buys the weightwise number pre-registered.
#
# WHY THE CEILING GOES TO +2.0 AND WHY THAT IS NOT ANOTHER EXTRAPOLATION.
#   At HI=0.0 the binding seed's top coordinate reached EXACTLY 0.000.  +2.0 adds
#   2.0 log units above the measured pin.  This is a MEASURED margin over an
#   OBSERVED bind, not a velocity extrapolation -- which matters, because the
#   velocity extrapolation is what just failed:
#
#       seed   ep 0-10   10-20   20-30   30-40      (v/vmax, top coordinate)
#       s0       0.845   0.123   0.379   0.037   <-- RE-ACCELERATED 3.1x
#       s1       0.835   0.064   0.002  -0.049
#       s2       0.819   0.116  -0.041   0.097
#
#   Cycle 52 projected 22.8 more epochs of headroom from the WORST SEED's Q4
#   velocity, having already corrected the mean-trajectory figure of 56.6 for
#   being too generous.  The measured bind was 10.4 epochs later -- the worst-seed
#   projection was still optimistic by 2.2x, because the deceleration is NOT
#   MONOTONE and is seed-dependent.  See CORRECTIONS 72.
#   **C0.3 is therefore still scored FIRST, and +2.0 earns a gate, not trust.**
#
# THE DESIGN.  Byte-identical to `bl5-*` except BETA_CLIP and the seed count:
#   R18/CIFAR-10, SGDm/Lion, ms=1e-3, alpha0=1e-3, AUGMENT=1, 40 epochs,
#   rungs w/node/lay, **BETA_CLIP=-30:2.0**, **seeds 0-3 (n=4, not 3)**.
#   n=4 because the weightwise post-hoc number rests on n=2 box-free seeds, and
#   n=2 is below this campaign's own bar for anything that goes in a headline.
#   The 20-epoch rung (`cl5-*-cU-*`, n=3) and the HI=0.0 40-epoch rung (`bl5-*`,
#   n=3) are both already on disk, so 12 jobs buy a THREE-point comparison.
#
# PRE-REGISTERED PREDICTIONS -- score against these, not against what lands.
#
#   (C0) VALIDITY GATE.
#        1. beta must MOVE: beta_true_max != beta_true_min on the last record, 12/12.
#        2. n_records == 4000.
#        3. n_beta byte-matches: w 11,173,962 / node 14,420 / lay 62.
#
#   (C0.3) **THE BOX-FREE GATE, SCORED FIRST**, and scored PER SEED, not pooled.
#        THE POOLED ROW IS WHAT HID bl5's PROBLEM: bl5's weightwise arm read
#        "8.03% of records at HI", which is one seed at 24.07% and two at 0.00%.
#        A mean over a binary event is not a diagnostic.  Report, per seed:
#        first_hi, its epoch, %rec, %Q4 and PATCH_CLIPCOUNT's %coord at BOTH
#        guards (STANDING RULES 5, 6).
#        BAR: **every seed** < 5% of records at either guard.
#        IF ANY SEED BINDS, that seed's C1 is UNINTERPRETABLE, and the pair
#        (HI=0.0 -> epoch 30.4, HI=+2.0 -> epoch X) is a two-point measurement of
#        how the bind budget moves with the ceiling -- report it as the result.
#
#   (C0.4) **THE STABILITY GATE, SCORED SECOND, and it is NEW.**
#        X3 (cycle 51) found the operating point INTERIOR at alpha_max=1.0 and
#        concluded the ceiling was not load-bearing.  That was measured at 20
#        epochs.  At 40 epochs one seed in three wants to go ABOVE alpha=1, so
#        X3's conclusion is now known to be budget-scoped, and letting alpha
#        reach e^2 = 7.39 is a real stability question rather than a formality.
#        BAR: plateau within 2.0 pp of the matching `bl5-*` rung (w 86.367 /
#        node 89.971 / lay 88.828), and `collapsed` false, on every seed.
#        IF IT FAILS: the ceiling IS load-bearing at 40 epochs, C1 is
#        uninterpretable, and **that is itself the result** -- it narrows X3 to
#        20 epochs and says the box is doing real work at longer budgets.
#        Scored BEFORE C1 because an unstable run's correlation statistics are
#        not a measurement of anything.
#
#   (C1) THE PRIMARY TEST.  N_eff/m, weightwise, VARIANCE instrument, STEADY half
#        (records 2000-4000).  Control: **0.5064** (cl5-w-cU, 20 epochs, steady
#        half, re-derived this tick).
#        PREDICTION, with a DIRECTION AND A MAGNITUDE, both registered:
#          **DRIFTING DOWN, and by more than the bar: N_eff/m < 0.40.**
#          I expect ~0.26, replicating the box-free window 0.5-0.759.
#        REFUTATION: N_eff/m within +-0.10 of 0.5064, i.e. inside [0.41, 0.61].
#          --> the fall seen in bl5 was the box after all, the post-hoc window
#              misled, and the headline is budget-robust.  Write that.
#        THIS PREDICTION IS NOT INDEPENDENT AND IS LABELLED AS SUCH: it was made
#        AFTER seeing bl5's post-hoc box-free window.  A confirmation is a
#        REPLICATION, not a discovery, and must be written as a replication.
#        What makes the batch worth running is that the refutation branch is
#        live: the post-hoc number rests on n=2 box-free seeds and a window
#        chosen after seeing which seed bound.
#
#   (C2) THE GRANULARITY ASYMMETRY -- the genuinely independent prediction, and
#        the one to weight most heavily, because nothing post-hoc informed it.
#        bl5's clean rungs said layerwise is budget-STATIONARY (-0.035) and
#        nodewise is budget-DRIFTING (-0.128), on n=3 with no salvage.
#        PREDICTION: that asymmetry replicates at n=4 under a different ceiling --
#          lay within +-0.10 of 0.7262, node BELOW 0.31.
#        REFUTATION: either rung on the wrong side.
#          --> N_eff/m's budget-robustness is not granularity-dependent and
#              FINDINGS 53.4's asymmetry was a 3-seed accident.
#
#   (C2.5) THE ORDERING.  bl5 refuted B2.5 (`node < w < lay` did NOT survive to
#        40 epochs) and the refutation SURVIVED restriction to the box-free
#        window, so it is not a contamination artefact.
#        PREDICTION: the 40-epoch ordering is **w < node < lay**, replicating.
#        REFUTATION: `node < w < lay` returns --> the flip WAS the box, B2.5's
#        refutation falls, and CORRECTIONS 70 is not budget-scoped after all.
#
#   (C3) ACCURACY, reported regardless and NOT a test.  Against bl5 at the same
#        40 epochs; a raised ceiling should not change plateau if the operating
#        point is interior for the seeds that never bound.
#
#   RESOLUTION, per CORRECTIONS 33, computed BEFORE submission.  rho_min falls as
#   1/sqrt(T); T = 2000 steady records at n=4 seeds, better than bl5's n=3:
#       weightwise  ~5.2e-09   RESOLVES comfortably
#       nodewise    ~4.3e-06   RESOLVES
#       layerwise   ~9.5e-04   MARGINAL at m=62, as always.  A layerwise null is
#                              a BOUND, never an absence -- and C2 asks layerwise
#                              to STAY PUT, which a bound cannot establish.  Say
#                              "not shown to move", never "shown not to move".
#
# NOT INCLUDED, deliberately:
#   * 80 epochs.  The obvious next rung, and it is NOT funded here, because
#     +2.0 is an untested ceiling and funding a longer budget on it before C0.3
#     scores would repeat the exact error CORRECTIONS 72 was written about.
#     If C0.3 and C0.4 both pass cleanly, 80 epochs is cheap and gets registered
#     then, with a ceiling chosen from THIS batch's measured pin.
#   * R10 / R34 / CIFAR-100.  The budget threat must be characterised in ONE
#     family before it is spent across four; R18 is where the ladder lives.
#   * A raised FLOOR.  Measured non-binding at -30 in every arm of this ladder.
#
# WHY cluster_b.  Account assignment is BY COMPARATOR (CORRECTIONS 46(6)): both
# controls (`cl5-*-cU-*` at 20 epochs and `bl5-*` at 40) are on cluster_b.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. C0, C0.3, C0.4 above, run not assumed, C0.3 FIRST and PER SEED
#   2. python3 analysis/c53_score.py --selftest          (14/14)
#   3. python3 analysis/c53_budget_window.py --selftest  (6/6)
#   4. python3 analysis/neff_instrument.py --selftest    (23/23)
#   5. python3 analysis/probe5_window.py --selftest      (41/41)
#   6. Window scan on every arm.  A 4000-record run has DIFFERENT absolute
#      windows from a 2000-record one; say which you mean, every time.
#
# USAGE
#   bash c53_budget_replication.sh            # dry run: prints every sbatch line
#   bash c53_budget_replication.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

WS=/cluster/account_b/metaopt                          # cluster_b
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/br6
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_b
PEND_CAP=40
NJOBS=12
BOX=-30:2.0                                        # +2.0 above the MEASURED pin
EPOCHS=40
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c53 BUDGET REPLICATION, BOX-FREE BY CONSTRUCTION (cluster_b) ==="
echo "    BETA_CLIP=$BOX  epochs=$EPOCHS  seeds 0-3  controls = cl5-*-cU-* (20ep) and bl5-* (40ep)"

for P in PATCH_PROBE5 PATCH_PROBE5_FIX PATCH_CLIPCOUNT; do
  grep -q "$P" "$HF" 2>/dev/null || { echo "GUARD FAIL: $P not present in $HF"; exit 1; }
done
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX + PATCH_CLIPCOUNT present  OK"

python3 -c "import py_compile; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 2: HF.py compiles  OK"

# GUARD 3 -- "byte-matched except two fields" is a CLAIM and must be DIFFED
# (the uc5 lesson, CORRECTIONS 71).  The control is bl5, and the ONLY fields that
# may differ are BETA_CLIP and the seed list.
BL5=$(dirname "$0")/c52_budget_ladder.sh
if [ -r "$BL5" ]; then
  for f in "ResNet18" "CIFAR10" "SGDm" "Lion" "--meta-stepsize 1e-3" "--alpha0 1e-3"; do
    grep -q -- "$f" "$BL5" || { echo "GUARD FAIL: '$f' absent from the bl5 control"; exit 1; }
  done
  grep -q -- 'EPOCHS=40' "$BL5" || {
    echo "GUARD FAIL: bl5 is not a 40-epoch batch -- this is not a matched replication"; exit 1; }
  grep -q -- 'BOX=-30:0.0' "$BL5" || {
    echo "GUARD FAIL: bl5's box is not -30:0.0 -- the ceiling contrast is not what it claims"; exit 1; }
  echo "guard 3: bl5 control is R18/CIFAR10/SGDm/Lion/ms=1e-3 at 40 epochs, BOX=-30:0.0  OK"
else
  echo "guard 3: c52_budget_ladder.sh not readable here -- SKIPPED (say so in FINDINGS)"
fi

# GUARD 4 -- the ceiling must actually be ABOVE the measured pin, or this batch
# is a rerun of bl5 under a different name.
python3 - <<'PYEOF' || exit 1
hi = 2.0
pinned_at = 0.0     # bl5 probe_w_e40_s0, beta_true_max reached exactly 0.000
assert hi > pinned_at, "the new ceiling is not above the measured pin"
print(f"guard 4: ceiling {hi:+.1f} is {hi - pinned_at:.1f} log units above the MEASURED pin {pinned_at:+.1f}  OK")
PYEOF

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
FAIL=0
for G in layerwise nodewise weightwise; do
  case $G in
    layerwise)  TAG=lay  ;;
    nodewise)   TAG=node ;;
    weightwise) TAG=w    ;;
  esac
  for S in 0 1 2 3; do
    RN="br6-${TAG}-c2-s${S}"
    PD="$SAVE/probe_${TAG}_c2_s${S}"
    # --time=03:50:00: gpu-short caps below 5h and rejects the WHOLE batch with
    # "Requested time limit is invalid" (CORRECTIONS 71).  40 epochs at R18 took
    # 16-32 min in bl5, so this is a 7x margin.
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
      --export="ALL,AUGMENT=1,BETA_CLIP=$BOX,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
      "$RUNNER"
      --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
      --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
      --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3
      --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    # COUNT WHAT SLURM ACCEPTED, NOT WHAT WE TRIED (CORRECTIONS 71).
    if [ "$SUBMIT" = "1" ]; then
      if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
    else
      N=$((N+1)); printf '%s\n' "${CMD[*]}"
    fi
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
echo "reduce with, IN THIS ORDER:"
echo "  1. C0.3 PER SEED   -- analysis/c53_budget_window.py pattern, on ../probes_br6"
echo "  2. C0.4 STABILITY  -- plateau vs bl5 from results/all_runs.csv, per seed"
echo "  3. C1  neff_instrument.py ../probes_br6            # THE PRIMARY TEST"
echo "  4. C2  the lay/node asymmetry, same table"
echo "  5. C2.5 the ordering; probe5_window.py ../probes_br6 for the window scan"
