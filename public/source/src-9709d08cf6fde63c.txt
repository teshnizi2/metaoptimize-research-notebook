# COST: cwd5's sacct mean 0.6855 GPU-h per run of this cell on L4 -> 18 jobs ~= 12.3 GPU-h expected.
#   HARD BOUND = WALL 3 h x 18 = 54 GPU-h.
#
# USAGE   bash bin/cRD1_route.sh [--dry-run]   |   bash bin/cRD1_route.sh --submit   (the ONLY submitting form)
# ENV     CRD1_REGISTERED_COMMIT=<40-hex sha>   on a host with no git repo, --submit requires it (271.6 F1: full sha)
# =============================================================================
set -u
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;

[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }

USER_NAME="${USER:-$(whoami)}"

# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

echo " $DSET / $NET   SGDm(0.99)+Lion   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C   ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"

  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done

  && [ "$(sha "$HDES")" = 7925e0edf31baa827c4b446739a7b0f9192855295eaa720956a056a3f8dad394 ] && [ "$(sha "$PATCHF")" = "$PATCH_SHA" ] \
  && echo "guard 1b3: the REUSED cwd_design.py / cwd_common.py / csh1_design.py / patch_decayroute.py and argsline_guard.py are their registered bytes (RULE 16)" \
  || guard_fail "guard 1b3: a reused registration file moved"

SELFT=/tmp/crd1_selftest.$$

  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION\|Traceback' "$SELFT" | head -20; tail -5 "$SELFT"; }

  && grep -q "^  PASS B the PREMISE: the landed OFF cell cwd5 W1" "$SELFT" && grep -q "^  PASS B the DOSE word" "$SELFT" \
  && echo "guard 1c5: LAMBDA RE-DERIVED from the landed cwd5 k01W1 records; the OFF premise (cwd5 W1 COLLAPSE) and the dose words re-read" \
  || guard_fail "guard 1c5: the derivation / calibration on landed records did not pass"

  || guard_fail "guard 1c6: the frozen floor was not shown to be the filtered value on the current corpus"
[ "$(grep -c "^  PASS A [STA][RI][SL]'s flags == the landed cwd5-k[0L][1W]*-s146 ARGS line" "$SELFT")" = 6 ] \

                      || guard_fail "guard 2d: $NSEED .out files already ran a seed of 196..198"

  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^crd1-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 crd1-* in squeue" || guard_fail "guard 2f: $NQ crd1-* already queued"

  && { echo "guard 4a1: cDR1_stage_harness.sh (verify mode) VERIFIES cdr1's tree, which crd1 uses UNCHANGED:"; grep -E 'V-a|V-h|V-d|V-c|VERIFIED' /tmp/crd1_stagew.$$ | sed 's/^/          /'; } \

[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" || guard_fail "guard 3: CIFAR-100 not visible"

  && echo "guard 3c: HF.py == $HF_SHA (PATCH_DECAYROUTE), HF.py.pre_decayroute == $HF_PRE_SHA (the live HF.py)" || guard_fail "guard 3c: the tree's HF.py is not cdr1's"

# ---- guard 4h: the harness-null for this batch's wiring, on the login node's CPU (no GPU) ----------------------------

   && [ "$(grep -c '^  PASS R2 weights BITWISE' "$GLOG")" = 6 ] && [ "$(grep -c '^  PASS R2 trace BITWISE' "$GLOG")" = 6 ] \
   && [ "$(grep -c '^  PASS R3 NON-VACUITY' "$GLOG")" = 6 ] && [ "$(grep -c '^  PASS R1 exactly one DECAY_ROUTE line' "$GLOG")" = 6 ] \
   && [ "$(grep -c '^  PASS R4 ' "$GLOG")" = 2 ]; then
  echo "guard 4h: crd1_route_cpu_check ALL PASS ($(grep -c '^  PASS' "$GLOG") PASS / 0 FAIL): every arm's OWN ARGS + DECAY_ROUTE value"
  echo "          print its registered witness and reach the patched update BITWISE (weights 6/6, trace 6/6), OFF's formula"
  echo "          FAILS on every arm (6/6), and the two wrong wd combinations are LOUD (2/2); CPU only"
  mkdir -p "$SAVE"; cp "$GLOG" "$SAVE/route_cpu_check$([ "$SUBMIT" = 1 ] || echo .dryrun).log"

python3 - "$DES" "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$ARMS" "$BN_SHA" "$HF_SHA" "$RUNNER_SHA" "$STAGEW" "$GPU_CONSTRAINT" \
          "$NJOBS" "$WALL" "$MST" "$A0" "$DSET" "$EPOCHS" "$BATCH" "$CLIP_C" "$PROBE" "$LAMBDA_TOKEN" "$TRAIN_SHA" "$HF_PRE_SHA" "$PATCH_SHA" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"

# ---- guard 4: THE LIVE MODEL AND THE LIVE OPTIMISER, per arm, from cdr1's tree, DECAY_ROUTE set per arm (CPU construction only) --------------

python3 - "$HARN" "$MTMP" "$DES" <<'PYEOF' || guard_fail "guard 4: live model / optimiser / witness / manifest check FAILED"

        own[(mn + "." if mn else "") + pn] = type(m).__name__
bad = 0
