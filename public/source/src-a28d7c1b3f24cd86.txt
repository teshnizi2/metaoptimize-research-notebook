#!/bin/bash
# =============================================================================
# c78_degenerate_tail.sh -- `bn1`, IS THE MATCHED-COUNT PARTITION GAP THE
# DEGENERATE SIZE-1 TAIL?
# Runs on cluster_a ONLY (needs PATCH_CHUNKWISE and PATCH_NODEBN).
# 9 jobs: {nodewise1d, chunk2325, nodewise} x seeds 0-2, ms=1e-4, 100 ep, probe ON.
#
# WHY THIS BATCH EXISTS
#   cycle 78's `pp1` decomposed the matched-count partition gap and put ALL of it
#   on one leg (CORRECTIONS 107.2/107.4, FINDINGS 77.1-77.3):
#
#     A  nodewise -> permnode  ALIGNMENT       -0.009 pp  t = -0.06   -> NULL
#     B  permnode -> chunk777  SIZE DISTRIBUTION +0.590 pp  t = +5.53
#     D  nodewise -> chunk777  both            +0.581 pp  t = +4.11   (mm1: +0.485)
#
#   So grouping BY ARCHITECTURE is worth nothing, and the group-size DISTRIBUTION
#   is worth everything.  This batch asks WHAT ABOUT THE DISTRIBUTION.
#
#   MEASURED on the built ResNet18 (FINDINGS 77.6), not inferred:
#
#     nodewise size    1 :  9,610 groups = 66.64% of groups, covering 0.09% of weights
#                  4,608 :  1,536 groups = 10.65% of groups, covering 63.34% of weights
#
#   Every ONE-DIMENSIONAL tensor -- 40 BatchNorm scales and shifts plus linear.bias,
#   41 tensors holding 9,610 scalars -- has numel//shape[0] == 1, so nodewise gives
#   each of those scalars its OWN step size.  **On ResNet18, "one step size per output
#   channel" is two-thirds WEIGHTWISE meta-learning on the normalisation parameters,
#   by group count.**  chunk777 gives each of those tensors ONE group, because every
#   channel count on this net (64..512) is below 777.
#
#   That is the concrete difference the B leg is made of.  `nodewise1d` (PATCH_NODEBN,
#   cycle 78) removes it and NOTHING else: nodewise on every tensor with ndim >= 2 --
#   so the per-output-channel partition on every conv and linear WEIGHT MATRIX is left
#   exactly in place -- and ONE group per 1-D tensor.
#
#     m(nodewise)   = 14,420
#     m(nodewise1d) =  4,851
#     m(chunk2325)  =  4,851   EXACTLY.  Found by search over K, verified from the
#                              ALLOCATED beta by guard 4.  A PERFECT match, not mm1's
#                              one-group-apart.
#
# THE THREE ARMS, ONE BATCH, ms=1e-4, seeds 0-2:
#     n1d   nodewise1d  m= 4,851   channels on ndim>=2, one group per 1-D tensor
#     c23   chunk2325   m= 4,851   uniform flat chunks, architecture-blind
#     node  nodewise    m=14,420   the anchor: bitwise mm1's and pp1's nodewise arm
#
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
#
#   T0    VALIDITY.  n_records == 10000; beta moved; epochs_done == requested == 100.
#   T0.2  n_beta EXACT on EVERY record: n1d 4851, c23 4851, node 14420.  Guard 4
#         MEASURES all three from the ALLOCATED beta on the real built network before
#         anything is submitted, and ALSO asserts m(c23) == m(n1d) exactly.
#   T0.3  THE INSTRUMENT FIRED.  neg_counts.json, n_tot == n_beta, npy shape from its
#         HEADER == (n_tot,).
#   T0.4  BOX-FREE at BOTH guards, rec_-based 5% gate, PRIMARY and UNCHANGED.  All 9
#         pp1 arms and all 6 mm1 arms read 0.0000 on all four columns in this SAME
#         box, so a bind on n1d or c23 would be NEW and would be reported as such.
#
#   T1    **THE PRIMARY.  FIVE-WAY, SYMMETRIC, WITHIN-BATCH, MATCHED COUNT EXACTLY.**
#         G = plateau5(chunk2325) - plateau5(nodewise1d).  Both arms m = 4,851.
#         The bands are mm1's M1 bands and pp1's P2 bands, deliberately, so all three
#         legs of this programme are read on ONE scale.
#           [-0.15, +0.15] -> **THE DEGENERATE TAIL WAS THE CARRIER.**  The +0.53 pp
#                matched-count gap COLLAPSES once nodewise's 9,610 size-1 groups are
#                merged away, while the per-output-channel partition on every weight
#                MATRIX is left in place.  This is a PRESCRIPTIVE result for the
#                Adam-mini / Adalayer / SGG line: do not give each normalisation
#                scalar its own step size; the rest of the architectural partition is
#                fine.  **This is the outcome the batch is built to be able to see,
#                and it is the NULL band -- so it cannot be reached by finding an
#                effect, only by failing to.**
#           (+0.15, +0.30] -> **UNDECIDED**, registered in advance.
#           G >  +0.30      -> **THE GAP SURVIVES THE TAIL'S REMOVAL.**  Uniform still
#                beats architecture-aligned at matched count with the degenerate
#                groups gone, so size HETEROGENEITY as such -- not the degenerate tail
#                -- is the carrier.  More general than the NULL branch, and it means
#                the fix is NOT simply "merge the 1-D tensors".
#           [-0.30, -0.15) -> **UNDECIDED**, registered in advance.
#           G <= -0.30      -> **REVERSAL: ALIGNMENT PAYS ONCE THE TAIL IS GONE.**
#                nodewise1d BEATS uniform chunks at matched count.  **THIS IS A LIVE
#                POSSIBILITY AND IT IS REGISTERED HERE, NOT ADDED AFTERWARDS:** pp1's
#                P2 nulled alignment WITH the degenerate tail present, and that tail is
#                two thirds of nodewise's groups, so a real alignment benefit on the
#                weight matrices could have been swamped by it.  If this branch fires,
#                pp1's P2 must be re-read as "alignment is null IN THE PRESENCE OF the
#                tail", which is a weaker statement than 107.2 currently makes.
#         SYMMETRIC because there is no prior favouring either sign over the other.
#
#   T2    **THE SECONDARY.  DIRECTIONAL, REGISTERED -- AND CONFOUNDED BY DESIGN.**
#         H = plateau5(nodewise1d) - plateau5(nodewise), within-batch.
#         PREDICTS H > 0: removing 9,569 degenerate groups helps.
#           H >  +0.30 -> CONFIRMS | (-0.15, +0.30] -> UNDECIDED | H <= -0.15 -> REFUTES
#         **THE CONFOUND IS STATED BEFORE THE DATA EXISTS AND IS NOT REPAIRABLE INSIDE
#         THIS CONTRAST:** n1d and node differ in COUNT (4,851 vs 14,420) as well as in
#         the tail, and ck1's K3 measured that within ONE partition family COARSER IS
#         BETTER over this range (~0.51 pp/decade).  A count effect alone therefore
#         predicts H > 0.  **So a CONFIRM here is NOT evidence about the tail
#         specifically, and must never be reported as if it were.**  T1 is the
#         confound-free contrast.  T2 is registered because a REFUTE would be
#         informative (it would mean the tail helps enough to beat the count effect),
#         and because leaving it unregistered would invite reading it post-hoc.
#
#   T2b   THE COUNT-CORRECTED REMAINDER.  **POST-HOC, DESCRIPTIVE, NO VERDICT.**
#         The chunk family now spans SEVEN rungs at a FIXED partition (ck1 + cx2 +
#         mm1 + this batch's c23), so the count effect between m=14,420 and m=4,851
#         can be read off that curve and subtracted from H, exactly as
#         analysis/c76_partition_vs_count.py did for ck1.  It carries no direction and
#         is a reading, not a result -- and it inherits the +-0.25 pp cross-batch
#         offset of CORRECTIONS 106.3 wherever the curve crosses batches.
#
#   T3    THE BATCH-OFFSET METER.  nodewise@1e-4 here vs pp1's 92.012 and mm1's
#         92.044, bar +-0.50.  **THIS CANNOT VOID T1 OR T2** -- both are within-batch
#         differences and are immune to any offset common to their arms.  T3 measures
#         the offset for the RECORD.  106.3 found the cross-batch offset has no stable
#         sign (+0.083 against ck1's -0.255), so this is a third draw from it.
#
#   T4    THE FIELD, THREE WAYS.  **DESCRIPTIVE, NO DIRECTION.**
#         a_raw, a_deb, pbar, dev_bias, bias_share, N_eff/m for all three arms.
#         **The "53.1%" sentence is NOT reproduced and must NOT be written.**
#
#   T5    **THE DISSOCIATION RE-TEST.  REGISTERED, AND ITS POST-HOC ORIGIN DECLARED.**
#         FINDINGS 77.5 found, POST-HOC and outside every registered gate, that across
#         nodewise -> permnode the field moved hugely while accuracy did not move at
#         all: N_eff/m 0.0547 +-0.0010 -> 0.0414 +-0.0003 (t ~ 13) against a plateau5
#         difference of -0.009 pp (t = -0.06).  That is ONE contrast at n=3.  A
#         prediction DERIVED from a post-hoc finding and tested on NEW data is a
#         legitimate registration, which is why it is written here rather than scored
#         off pp1's own numbers.
#         This batch supplies an INDEPENDENT matched-count pair (c23 vs n1d, both
#         m=4,851).  Let t_F = |N_eff/m(c23) - N_eff/m(n1d)| / se.
#           T1 is NULL and t_F >= 3  -> **DISSOCIATION REPLICATED** on a second and
#                independent contrast.  The sign-agreement field is not a sufficient
#                statistic for what a partition does to optimisation, and the brief's
#                direction-C default is then refuted on its own terms rather than
#                merely blocked.
#           T1 is NULL and t_F <  3  -> the field TRACKS accuracy here; 77.5's
#                dissociation does not generalise and must be reported as a one-off.
#           T1 is NOT NULL           -> **NOT APPLICABLE.**  The test is defined only
#                on an accuracy tie, and this branch is registered so that a non-NULL
#                T1 cannot be used to fish for a dissociation afterwards.
#         The threshold is on t, not on an absolute field difference, because the sem
#         is measured per batch and an absolute bar fixed from pp1's sems would import
#         pp1's noise level into this batch's verdict.
#
# WHAT THIS BATCH DELIBERATELY DOES NOT DO
#   * It does NOT sweep ms.  ms=1e-4 is held because that is the ms mm1's M1, ck1's K2
#     and pp1's P1/P2 were all measured at, so T1 sits on the same scale as them.
#     **THE RISK IS INHERITED AND RESTATED: nodewise is read at 1e-4, not at its own
#     argmax of 3e-4, so every verdict here is a statement at ms=1e-4.**  nodewise1d's
#     own argmax has never been measured at all, which is a stronger version of the
#     same caveat and applies to T1 as well as T2.
#   * It does NOT test whether LAYER boundaries matter.  Every arm here is a
#     within-tensor partition (verified: chunkwise iterates layers and zero-pads the
#     ragged tail; nodewise1d is per-tensor by construction).  CORRECTIONS 107.7 item
#     (2) is still open and is not touched by this batch.
#   * It does NOT separate "the tail is degenerate (size 1)" from "the tail is on the
#     NORMALISATION parameters".  Those coincide exactly on ResNet18 -- every 1-D
#     tensor is a BN scale/shift or the output bias -- so this batch cannot tell a
#     size story from a parameter-role story.  Stated, not hidden; separating them
#     needs a net where they come apart, not another arm here.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 tests/test_nodebn.py           (CPU, N0 asserts determinism first)
#   2. python3 tests/test_permnode.py         (CPU, unchanged by PATCH_NODEBN)
#   3. python3 tests/test_chunkwise.py        (CPU, unchanged by PATCH_NODEBN)
#   4. python3 analysis/c55_neff_noise.py --selftest   (51/51 -- the selftest does not
#      enumerate BOXES, so this count does NOT move when a family is registered)
#   5. `bn1` registered in c55_neff_noise.BOXES BEFORE scoring, from THIS CLIP=.
#   6. plateau5 is PRIMARY.  best_test inflates ~0.37 pp.
#
# USAGE
#   bash c78_degenerate_tail.sh            # dry run
#   bash c78_degenerate_tail.sh --submit   # submits, ONLY if the guards pass
# =============================================================================

sys.path.insert(0, ".")
from build_network import build_network
from Optimizers.HF import HF
K, want_chunk, want_node, want_n1d = (int(x) for x in sys.argv[1:5])

BASE = {"alg": "SGDm", "weight_decay": 0.1, "momentum_param": 0.99}
META = {"alg": "Lion", "meta_stepsize": 1e-4, "momentum_param": 0.99,
        "Lion_beta2": 0.9, "weight_decay": 0}

class NullWriter:
    def add_scalar(self, *a, **k):
        pass

def build(gran):
    torch.manual_seed(0)
    net = build_network("ResNet18", "cpu")
    return HF(net, stepsize_groups=gran, alpha0=1e-3, args_base=dict(BASE),
              args_meta=dict(META), gamma=1, writer=NullWriter())

def m_of(opt):
    return int(sum(int(b.numel()) for b in opt.beta))

net = build_network("ResNet18", "cpu")
shapes = [tuple(p.shape) for p in net.parameters()]
numels = [int(p.numel()) for p in net.parameters()]

o_n1d = build("nodewise1d")
