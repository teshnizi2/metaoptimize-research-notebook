# =============================================================================
# THE QUESTION, STATED BEFORE ANY NUMBER IT PRODUCES IS READ
# =============================================================================
#   ctd1 measured, under TRUE scalar dynamics at the standard CIFAR-100 cell,
#   that the sign the single beta follows is carried on 83 % of records -- and
#   on 100 % of the pinned records -- by exactly three 512-parameter BN scales
#   (tensors 50, 53, 59; 1,536 of 11,220,132 parameters) whose L_i = 0.9 m_i +
#   0.1 z_i sum is POSITIVE (beta DOWN on the live source) against the other 59
#   tensors, whose own sum is NEGATIVE (beta UP) on 944 of 944 pinned records.
#   That is an OBSERVATION.  H-ISOLATE is the INTERVENTION: if those three
#   tensors are given their OWN step size, the remaining 59 should stop being
#   dragged to the -15 floor, and the run should escape the scalar collapse.
#
#   ARMS (x seeds 21, 22, 23; ONE submission; PROBE=100, PROBE_TENSOR=1 on all):
#     k01   scalar                             m=1  FLOOR anchor, in batch
#     k62   layerwise                          m=1x62 CEILING anchor, in batch
#     ISO   sets:<59 others>/<50,53,59>        m=2, sizes [59,3]   THE ARM
#     CTRL  sets:<59 others>/<20,23,29>        m=2, sizes [59,3]   THE CONTROL
#     ONE   sets:<61 others>/<50>              m=2, sizes [61,1]   DOSE arm
#
#   WHY THIS CONTROL, AND WHY IT IS NOT NUMEL-MATCHED.  The brief asked for
#   "three NON-carrier 512-param BN scales from layer1/layer2".  On the live
#   ResNet18_c100 that object does not exist: 512-wide BN scales live only in
#   layer4 (layer1 64, layer2 128, layer3 256), and layer4 has exactly TWO
#   non-carrier BN scales (47 layer4.0.bn1.weight, 56 layer4.1.bn1.weight).
#   CTRL is therefore the STRUCTURAL HOMOLOGUE of the carrier set two stages
#   earlier: block-0 bn2 scale, block-0 shortcut BN scale, block-1 bn2 scale of
#   layer2 (tensors 20, 23, 29; 128 params each).  Matched on class (BN scale),
#   count (3), structural role, group sizes ([59,3]) and on measured
#   NON-involvement (carrier share on ctd1's DISAGREE records 0.0000 / 0.0000 /
#   0.0008, versus 0.0089 / 0.0218 for the layer3 homologues, which is why
#   layer3 was not used).  NOT matched on numel (384 vs 1,536) -- disclosed, and
#   architecturally unavoidable.
#
#   WHY A FIFTH ARM.  On ctd1's 944 pinned scalar records the three carriers
#   dominate the other 59 together on 0.9799, but tensor 50 ALONE dominates on
#   0.0000 and the pair {53, 59} dominates the other 59 on 0.0000 (their
#   |L| are 0.387 / 0.257 / 0.436 of the remainder's sum|L_i|; the trio's is
#   1.080).  A STATIC reading of those terms says removing ANY one carrier
#   breaks the domination, so ONE (isolate 50 alone) should ALSO rescue.  The
#   corpus's prefix ladder says the opposite about ADDING tensors to a coarse
#   group (k50 30.34, k53 22.08: tensor 50 alone halves the rescue, 50+53
#   completes the collapse).  ONE is the cheapest test of whether the domination
#   is a static property of the terms or re-forms under its own dynamics.  3
#   more jobs, ~2.6 GPU-h.
#
# =============================================================================
# THE NOISE FLOOR, RE-DERIVED AT REGISTRATION, ciso1's OWN ROWS EXCLUDED FROM
# EVERY READER (the 165.4 pattern -- invariant under this batch's own ingest)
# =============================================================================
#   cR1's estimator: pooled within-(batch x granularity) SD of plateau5 over
#   m=2 arms in this exact cell at 100 epochs, from results/all_runs.csv at
#   2,740 rows, superseded rows dropped, collapsed rows dropped:
#     SIGMA_NARROW  (^\[\d+,\d+\]$ matcher)             0.925518  df 60  30 cells
#     SIGMA_WIDE    (matcher extended to sets: m=2)     0.889041  df 78  39 cells
#     SIGMA_W = max                                     0.925518
#   SE_ARM_DIFF = SIGMA_W * sqrt(2/3) = 0.755682   (a 3-seed arm vs a 3-seed arm)
#
# =============================================================================
# THE BARS, FROZEN, IN pp AND IN SE
# =============================================================================
#   D_ISO  = plateau5(ISO)  - plateau5(k01)      in-batch, both 3-seed means
#   D_CTRL = plateau5(CTRL) - plateau5(k01)
#   D_ONE  = plateau5(ONE)  - plateau5(k01)
#   SUPPORT_BAR = +25.0 pp = 33.08 SE.  RATIONALE: the largest rescue any m=2
#       partition of this cell has ever produced is k49 - k01 = 55.4688 -
#       22.8361 = +32.63 pp (n 21 / n 29).  +25 is 77 % of it; an escape from
#       the floor that recovers three quarters of the best two-group partition
#       is "a large margin" by any reading, and nothing between the floor and
#       there could be called the mechanism working as claimed.
#   REFUTE_BAR  = +5.0 pp = 6.62 SE.  RATIONALE: the WORST non-floor m=2
#       configuration on the corpus ladder is k50 (tensor 50 coarse, 53 and 59
#       fine) at +7.50 over the floor; +5 is below it.  If giving the three
#       carriers their own step size buys less than the ladder's own worst
#       partial state, isolation did not act on the collapse.
#   Between the two: UNRESOLVED-PARTIAL.
#   READ_BAR = 2 SE = 1.511363 pp: |D_CTRL| <= READ_BAR is CTRL-NULL.
#   DEAD_BAR = 5.00 pp: a within-arm seed RANGE above it in k01 or ISO is

#   Anchor bands (provenance): k01 inside 22.8361 +/- 4 x 0.5568, k62 inside
#       69.5345 +/- 4 x 0.4916 (each re-derived at score time, ciso1 excluded).

#   group 0 = the big group, group 1 = the isolated group):

#   UNRESOLVED-MANIFEST-MISMATCH   no manifest; live model != the frozen 62

# =============================================================================
# the `sets:` grammar, re-implemented HERE so the registered index sets are a
# property of this file and not of the harness (G1 compares both)
# =============================================================================
def spec_to_groups(spec, names=TENSOR_NAMES):
    """`sets:<g>/<g>` -> list of 1-based index lists in model order, exactly the
    harness's rule (range, then int, then name; total, loud)."""
    if spec in ("scalar",):
        return [list(range(1, len(names) + 1))]
    if spec in ("layerwise",):
        return [[i] for i in range(1, len(names) + 1)]
    assert spec.startswith("sets:"), spec
    T = len(names)
    owner = {}
    out = []
    for gi, g in enumerate(spec[5:].split("/")):
        idx = []
        for item in g.split(","):
            m = re.match(r"^(\d+)-(\d+)$", item)
            if m:
                lo, hi = int(m.group(1)), int(m.group(2))
                assert 1 <= lo <= hi <= T, item
                got = list(range(lo, hi + 1))
            elif re.match(r"^\d+$", item):
                got = [int(item)]
            else:
                assert item in names, item
                got = [names.index(item) + 1]
            for i in got:
                assert i not in owner, (i, owner.get(i), gi)
                owner[i] = gi
            idx.extend(got)
        out.append(sorted(idx))
    assert len(owner) == T, (len(owner), T)
    return out


def tail5(d, budget=EPOCHS):
    v = [d[e] for e in range(budget - 5, budget) if e in d]
    return (sum(v) / 5.0) if len(v) == 5 else None


def carrying_set(X, s_agg):
    order = sorted(range(len(X)), key=lambda i: -abs(X[i]))
    tot_abs = sum(abs(x) for x in X)
    part, rem = 0.0, tot_abs
    for k, i in enumerate(order, 1):
        part += X[i]
        rem -= abs(X[i])
        if sgn(part) == s_agg and abs(part) > rem:
            return order[:k]
    return order

def _csv_rows(path, exclude_own=True):
    out = []
    if not os.path.isfile(path):
        return out
    with open(path) as fh:
        for r in csv.DictReader(fh):
            if (r.get("superseded") or "0") == "1":
                continue
            if exclude_own and str(r.get("run") or "").startswith(PREFIX):
                continue
            out.append(r)
    return out


def _batch_of(run):
    return re.split(r"[-_]", str(run))[0]


def _in_cell(r, epochs=EPOCHS):
    return (r["network"] == NET and r["dataset"] == DSET
            and r["base"] == "SGDm" and r["meta"] == "Lion"
            and r["meta_stepsize"] == "1e-3" and r["alpha0"] == "1e-6"
            and r["gamma"] == "1" and r["augment"] == "1"
            and r["beta_clip"] == CLIP_C and not (r.get("hier") or "").strip()
            and r["batch_size"] == str(BATCH_SIZE)
            and r["epochs_done"] == str(epochs) and r["collapsed"] == "0")


M2_NARROW = re.compile(r"^\[\d+,\d+\]$")


def _is_m2(g, wide):
    g = str(g or "")
    if M2_NARROW.match(g):
        return True
    if wide and g.startswith("sets:"):
        return g.count("/") == 1
    return False


def noise_floor_from_csv(path, wide):
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not _in_cell(r) or not _is_m2(r["granularity"], wide):
            continue
        try:
            cells[(_batch_of(r["run"]), r["granularity"])].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        nm += len(v)
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc, nm


def cell_level(path, grans):
    v, b = [], set()
    for r in _csv_rows(path):
        if _in_cell(r) and str(r["granularity"]) in grans:
            try:
                v.append(float(r["plateau5"]))
                b.add((_batch_of(r["run"]), r["granularity"]))
            except (TypeError, ValueError):
                pass
    if not v:
        return None
    return {"n": len(v), "mean": statistics.mean(v),
            "sd": statistics.stdev(v) if len(v) > 1 else float("nan"), "cells": len(b)}


def band(path, gran):
    L = cell_level(path, (gran,))
    if L is None or L["n"] < 2:
        return None
    return {"n": L["n"], "mean": L["mean"], "sd": L["sd"],
            "lo": L["mean"] - BAND_K * L["sd"], "hi": L["mean"] + BAND_K * L["sd"]}


def own_rows(path):
    return sum(1 for r in _csv_rows(path, exclude_own=False) if str(r.get("run") or "").startswith(PREFIX))


def resolve_manifest(runsdir, explicit=None):
    cands = []
    if explicit:
        cands.append((explicit, "--manifest"))
    cands.append((os.path.join(runsdir, BATCH, "PARTITION-MANIFEST.txt"), "<runsdir>/ciso1/PARTITION-MANIFEST.txt"))
    cands.append((os.path.join(runsdir, BATCH + "-PARTITION-MANIFEST.txt"), "<runsdir>/ciso1-PARTITION-MANIFEST.txt"))
    for p, how in cands:
        if os.path.isfile(p):
            return p, how
    return None, "none of: " + ", ".join(h for _, h in cands)


def read_manifest(path):
    man = {"tensors": [], "arms": {}, "meta": {}}
    for ln in open(path):
        p = ln.split()
        if not p:
            continue
        if p[0] == "TENSOR" and len(p) >= 4:
            man["tensors"].append((int(p[1]), p[2], int(p[3])))
        elif p[0] == "ARMSPEC" and len(p) >= 4:
            # ARMSPEC <arm> SPEC <spec> M <m> SIZES a,b PARAMS x,y GROUP0 <names...> GROUP1 <names...>
            arm = p[1]
            d = {"spec": p[3], "m": int(p[5]), "sizes": tuple(int(x) for x in p[7].split(",")),
                 "params": tuple(int(x) for x in p[9].split(","))}
            i0 = p.index("GROUP0")
            i1 = p.index("GROUP1")
            d["group0"] = p[i0 + 1:i1]
            d["group1"] = p[i1 + 1:]
            man["arms"][arm] = d
        elif p[0] == "ISOLATED" and len(p) >= 3:
            man["meta"].setdefault("isolated", {})[p[1]] = tuple(int(x) for x in p[2].split(","))
        else:
            man["meta"][p[0]] = " ".join(p[1:])
    return man


def load_outs(runsdir):
    recs = {}
    pats = [os.path.join(runsdir, PREFIX + "*.out"), os.path.join(runsdir, BATCH, PREFIX + "*.out")]
    for f in sorted(set(sum([glob.glob(p) for p in pats], []))):
        m = OUT_RE.match(os.path.basename(f))
        if not m:
            continue
        arm, seed, job = m.group(1), int(m.group(2)), int(m.group(3))
        txt = open(f, errors="replace").read()
        lines = txt.splitlines()
        te, tr = {}, {}
        for a, b, c in EPTR_RE.findall(txt):
            te[int(a)] = float(c)
            tr[int(a)] = float(b)
        recs.setdefault((arm, seed), []).append({
            "file": f, "job": job,
            "args": [l for l in lines if l.startswith("ARGS:")],
            "env": [l for l in lines if l.startswith("ENV:")],
            "pt": [l for l in lines if l.startswith("PROBE_TENSOR:")],
            "done": "RUN_DONE" in lines, "traceback": "Traceback" in txt,
            "test": te, "train": tr, "n_ep": len(te), "p5": tail5(te), "t5": tail5(tr)})
    return recs


def load_probes(runsdir):
    root = os.path.join(runsdir, BATCH)
    runs = {}
    for d in sorted(glob.glob(os.path.join(root, "probe_*"))):
        m = PROBE_DIR_RE.match(os.path.basename(d))
        pj = os.path.join(d, "probe.jsonl")
        if not m or not os.path.isfile(pj):
            continue
        recs, bad = [], 0
        for ln in open(pj):
            ln = ln.strip()
            if not ln:
                continue
            try:
                recs.append(json.loads(ln))
            except Exception:
                bad += 1
        hdr = None
        hp = os.path.join(d, "probe_tensor.json")
        if os.path.isfile(hp):
            try:
                hdr = json.load(open(hp))
            except Exception:
                hdr = None
        runs[(m.group(1), int(m.group(2)))] = {"dir": d, "recs": recs, "bad": bad, "hdr": hdr}
    return runs


# =============================================================================
# the decisions, as pure functions (selftested)
# =============================================================================
def occupancy(recs, k):
    n = sum(1 for r in recs if "beta" in r and len(r["beta"]) > k)
    at = sum(1 for r in recs if "beta" in r and len(r["beta"]) > k and r["beta"][k] <= BETA_LO + PIN_EPS)
    term = recs[-1]["beta"][k] if recs and "beta" in recs[-1] and len(recs[-1]["beta"]) > k else None
    return (at / n if n else None), term, n


def free_state(occ, term):
    """-> 'FREE' | 'PINS' | 'UNKNOWN' by the registered rule."""
    if occ is None or term is None:
        return "UNKNOWN"
    if occ <= FREE_OCC_MAX and term > BETA_LO + FREE_TERMINAL_MARGIN:
        return "FREE"
    if occ >= PIN_OCC_MIN:
        return "PINS"
    return "NEITHER"


def level_token(d):
    if d is None:
        return "UNKNOWN"
    if d >= SUPPORT_BAR:
        return "RESCUES"
    if d <= REFUTE_BAR:
        return "FLOOR"
    return "PARTIAL"


def ctrl_token(d):
    if d is None:
        return "UNKNOWN"
    if d >= SUPPORT_BAR:
        return "CTRL-RESCUES"
    if abs(d) <= READ_BAR:
        return "CTRL-NULL"
    return "CTRL-PARTIAL"


def decide(d_iso, d_ctrl, iso59, stamps):
    """The registered branch on the three measured quantities.  Gates (manifest,
    provenance, divergence) are applied by the caller before this is reached."""
    stamps.append("SPECIFIC:" + ctrl_token(d_ctrl))
    if d_ctrl is not None and d_ctrl >= SUPPORT_BAR:
        return "NOT-SPECIFIC"
    if d_iso is None:
        return "UNRESOLVED-PARTIAL"
    if d_iso >= SUPPORT_BAR:
        if iso59 == "FREE":
            return "CAUSAL-SUPPORTED"
        stamps.append("ISO-RESCUES-BUT-59-" + ("PINS" if iso59 == "PINS" else "NOT-FREE"))
        return "UNRESOLVED-PARTIAL"
    if d_iso <= REFUTE_BAR:
        return "CAUSAL-REFUTED"
    return "UNRESOLVED-PARTIAL"


# =============================================================================
# the premise, re-derived from a scalar arm's PROBE_TENSOR records
# (used on ctd1 in --selftest and on the in-batch k01 in score)
# =============================================================================
def premise_from_scalar_records(recs_by_seed):
    car = [TENSOR_NAMES.index(n) for n in CARRIERS]
    n_det = n_dis = 0
    dis_car = collections.Counter()
    pin_n = pin_three = pin_one = pin_pair = pin_rest_up = 0
    for recs in recs_by_seed:
        for r in recs:
            try:
                z, m = r["z_tensor"], r["m_tensor"]
                zh, mh = float(r["z_agg"][0][0]), float(r["mom_pre"][0][0])
            except (KeyError, IndexError, TypeError):
                continue
            b2 = float(r.get("pt_b2", B2))
            L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
            s = sgn(b2 * mh + (1 - b2) * zh)
            if s == 0 or sgn(sum(L)) != s:
                continue
            pos = sum(1 for x in L if x > 0)
            neg = sum(1 for x in L if x < 0)
            smaj = sgn(pos - neg)
            if smaj == 0:
                continue
            n_det += 1
            if s != smaj:
                n_dis += 1
                for i in carrying_set(L, s):
                    dis_car[i] += 1
            if r.get("n_at_lo") == 1:
                pin_n += 1
                rest = [L[i] for i in range(len(L)) if i not in car]
                arest = sum(abs(x) for x in rest)
                l50, l53, l59 = L[car[0]], L[car[1]], L[car[2]]
                pin_three += sgn(l50 + l53 + l59) == s and abs(l50 + l53 + l59) > arest
                pin_one += sgn(l50) == s and abs(l50) > arest + abs(l53) + abs(l59)
                pin_pair += sgn(l53 + l59) == s and abs(l53 + l59) > arest
                pin_rest_up += sum(rest) < 0
    shares = {TENSOR_NAMES[i]: (dis_car[i] / n_dis if n_dis else None) for i in car}
    top = [(TENSOR_NAMES[i], c / n_dis) for i, c in dis_car.most_common(6)] if n_dis else []
    return {"n_det": n_det, "n_dis": n_dis, "R_T": (n_dis / n_det if n_det else None),
            "shares": shares, "top": top, "pin_n": pin_n,
            "three": (pin_three / pin_n if pin_n else None),
            "one": (pin_one / pin_n if pin_n else None),
            "pair": (pin_pair / pin_n if pin_n else None),
            "rest_up": (pin_rest_up / pin_n if pin_n else None)}


def premise_ok(P):
    return (P["n_dis"] > 0 and all(v is not None and v >= PREMISE_SHARE for v in P["shares"].values()))


# =============================================================================
# score
# =============================================================================
def score(runsdir, csv_path, manifest=None):
    print("=" * 78)
    print("cIS1 -- H-ISOLATE on `ciso1`.  runsdir %s" % runsdir)
    print("=" * 78)
    stamps, unresolved = [], []
    outs = load_outs(runsdir)
    probes = load_probes(runsdir)
    names = list(TENSOR_NAMES)

    # ---- G1: the manifest ------------------------------------------------------------
    print("\n[G1] MANIFEST -- the live model and the partitions, as written at submit time")
    mp, how = resolve_manifest(runsdir, manifest)
    if mp is None:
        chk(False, "manifest found (%s)" % how)
        unresolved.append("UNRESOLVED-MANIFEST-MISMATCH")
    else:
        print("  manifest: %s (%s)" % (mp, how))
        man = read_manifest(mp)
        chk([(i, n, k) for i, n, k in man["tensors"]] == [(i + 1, names[i], TENSOR_NUMEL[i]) for i in range(N_TENSORS)],
            "live model == the frozen 62 (index, name, numel)")
        for arm in M2_ARMS:
            d = man["arms"].get(arm)
            reg = spec_to_groups(SPEC[arm])
            ok = d is not None and d["spec"] == SPEC[arm] and d["m"] == 2 and d["sizes"] == SIZES[arm] \
                and d["params"] == PARAMS[arm] \
                and [names.index(n) + 1 for n in d["group0"]] == reg[0] \
                and [names.index(n) + 1 for n in d["group1"]] == reg[1] \
                and tuple(reg[1]) == ISOLATED[arm]
            chk(ok, "%s: spec, m=2, sizes %s, params %s, both groups EXACTLY as registered; isolated %s"
                % (arm, SIZES[arm], PARAMS[arm], ISOLATED[arm]))
        iso = man["arms"].get("ISO"); ctl = man["arms"].get("CTRL"); one = man["arms"].get("ONE")
        if iso and ctl:
            sd = set(iso["group1"]) ^ set(ctl["group1"])
            chk(sd == set(CARRIERS) | set(CONTROLS) and len(iso["group1"]) == len(ctl["group1"]) == 3
                and set(iso["group0"]) ^ set(ctl["group0"]) == sd,
                "ISO ^ CTRL == exactly the six names, three swapped at identical group sizes", str(sorted(sd)))
            n_iso = [TENSOR_NUMEL[names.index(n)] for n in iso["group1"]]
            n_ctl = [TENSOR_NUMEL[names.index(n)] for n in ctl["group1"]]
            print("       isolated numels ISO %s  CTRL %s  (numel NOT matched -- disclosed in the header)" % (n_iso, n_ctl))
        if iso and one:
            chk(set(iso["group1"]) - set(one["group1"]) == set(CARRIERS[1:]) and one["group1"] == [CARRIERS[0]],
                "ONE isolates exactly layer4.0.bn2.weight; ISO adds exactly 53 and 59")
        if GATE_FAIL:
            unresolved.append("UNRESOLVED-MANIFEST-MISMATCH")
    g1_fail = list(GATE_FAIL)

    # ---- G0: provenance ---------------------------------------------------------------
    print("\n[G0] PROVENANCE -- every run's OWN lines")
    chk(len(outs) == 15, "15 (arm, seed) runs with a .out", "%d" % len(outs))
    nrec_ok = True
    for arm in ARMS:
        for seed in SEEDS:
            tag = "%s-s%d" % (arm, seed)
            oo = outs.get((arm, seed))
            if not oo:
                chk(False, "%s: .out present" % tag)
                continue
            o = oo[-1]
            if len(oo) > 1:
                print("  NOTE %s: %d .out files, the latest job (%d) is audited" % (tag, len(oo), o["job"]))
            am = ARGS_RE[arm].match(o["args"][-1]) if o["args"] else None
            chk(am is not None and int(am.group(1)) == seed and int(am.group(2)) == seed,
                "%s: ARGS line is the registered line for %s" % (tag, arm))
            env = re.sub(r" PROBE_DIR=\S+", "", o["env"][-1]).replace("ENV: ", "") if o["env"] else ""
            chk(env == ENV_FIXED, "%s: ENV line (modulo PROBE_DIR) is the registered line" % tag)
            pm = PT_RE.match(o["pt"][-1]) if o["pt"] else None
            chk(pm is not None and pm.group(1) == PT_TYPE[arm], "%s: PROBE_TENSOR line present, type=%s" % (tag, PT_TYPE[arm]))
            P = probes.get((arm, seed))
            h = P["hdr"] if P else None
            chk(h is not None and h.get("PROBE_TENSOR") == 1 and h.get("every") == PROBE_EVERY
                and h.get("stepsize_type") == PT_TYPE[arm] and h.get("n_tensors") == N_TENSORS
                and tuple(h.get("param_numels", ())) == TENSOR_NUMEL and h.get("meta_alg") == "Lion",
                "%s: probe_tensor.json is the registered header" % tag)
            n = len(P["recs"]) if P else 0
            if n != N_RECORDS:
                nrec_ok = False
            chk(P is not None and n == N_RECORDS and P["bad"] == 0
                and [r.get("step") for r in P["recs"]] == list(range(0, META_STEPS, PROBE_EVERY)),
                "%s: %d probe records at steps 0..49900" % (tag, n))
    if len(GATE_FAIL) > len(g1_fail):
        unresolved.append("UNRESOLVED-PROVENANCE")
    g0_fail = list(GATE_FAIL)

    # ---- A: decomposition on every record of every arm --------------------------------
    print("\n[A] DECOMPOSITION -- group sums of z_tensor / m_tensor vs the harness's own z / momentum; applied sign")
    n_rec = n_bad_z = n_bad_m = n_sign = n_sign_bad = n_clamped = 0
    worst_z = worst_m = 0.0
    for (arm, seed), P in sorted(probes.items()):
        groups = [[i - 1 for i in g] for g in spec_to_groups(SPEC[arm])]
        for r in P["recs"]:
            try:
                z, m = r["z_tensor"], r["m_tensor"]
                za, ma = r["z_agg"][0], r["mom_pre"][0]
                b2 = float(r.get("pt_b2", B2))
                bpre, bpost = r["beta_pre"][0], r["beta"]
            except (KeyError, IndexError, TypeError):
                continue
            if len(za) != len(groups) or len(ma) != len(groups) or len(bpre) != len(groups) or len(bpost) != len(groups):
                n_bad_z += 1
                n_rec += 1
                continue
            n_rec += 1
            gz = [sum(z[i] for i in g) for g in groups]
            gm = [sum(m[i] for i in g) for g in groups]
            sz = sum(abs(v) for v in z) + 1e-30
            sm = sum(abs(v) for v in m) + 1e-30
            dz = max(abs(p - q) for p, q in zip(gz, za)) / sz
            dm = max(abs(p - q) for p, q in zip(gm, ma)) / sm
            worst_z, worst_m = max(worst_z, dz), max(worst_m, dm)
            n_bad_z += dz > DECOMP_TOL_Z
            n_bad_m += dm > DECOMP_TOL_M
            for k in range(len(groups)):
                Lh = b2 * ma[k] + (1 - b2) * za[k]
                if not (bpost[k] > BETA_LO + 1e-9 and bpost[k] < BETA_HI - 1e-9):
                    n_clamped += 1
                    continue
                n_sign += 1
                n_sign_bad += abs(-(bpost[k] - bpre[k]) / MS - sgn(Lh)) > 1e-3
    ok_a = chk(n_rec > 0 and n_bad_z / max(n_rec, 1) <= DECOMP_FAIL_MAX,
               "group sums of z_tensor == harness z within %.0e on >= 99%% of records" % DECOMP_TOL_Z,
               "%d/%d fail, worst rel %.2e" % (n_bad_z, n_rec, worst_z))
    ok_a &= chk(n_rec > 0 and n_bad_m / max(n_rec, 1) <= DECOMP_FAIL_MAX,
                "group sums of m_tensor == harness momentum_meta within %.0e on >= 99%% of records" % DECOMP_TOL_M,
                "%d/%d fail, worst rel %.2e" % (n_bad_m, n_rec, worst_m))
    ok_a &= chk(n_sign > 0 and n_sign_bad / max(n_sign, 1) <= SIGN_FAIL_MAX,
                "-(beta - beta_pre)/ms == sign(b2*mom_pre + (1-b2)*z_agg) on >= 99%% of unclamped coordinates",
                "%d/%d fail, %d clamped" % (n_sign_bad, n_sign, n_clamped))
    if not ok_a and "UNRESOLVED-PROVENANCE" not in unresolved:
        unresolved.append("UNRESOLVED-PROVENANCE")

    # ---- R: completion, bands, divergence ------------------------------------------------
    print("\n[R] COMPLETION, ANCHOR BANDS, WITHIN-ARM DIVERGENCE")
    lev, levt = {}, {}
    for arm in ARMS:
        vs, ts = [], []
        for seed in SEEDS:
            oo = outs.get((arm, seed))
            if not oo:
                continue
            o = oo[-1]
            ok = chk(o["done"] and not o["traceback"] and o["n_ep"] >= EPOCHS and o["p5"] is not None,
                     "%s-s%d: RUN_DONE, no traceback, %d epoch lines, plateau5 %s (train %s)"
                     % (arm, seed, o["n_ep"], fmt(o["p5"], 3), fmt(o["t5"], 3)))
            if not ok:
                unresolved.append("UNRESOLVED-DIVERGED")
            if o["p5"] is not None:
                vs.append(o["p5"])
            if o["t5"] is not None:
                ts.append(o["t5"])
        if len(vs) == 3:
            lev[arm] = statistics.mean(vs)
            levt[arm] = statistics.mean(ts) if len(ts) == 3 else None
            rng = max(vs) - min(vs)
            print("  %-4s plateau5 %.4f (seeds %s)  range %.3f   train %s"
                  % (arm, lev[arm], ", ".join("%.3f" % v for v in vs), rng, fmt(levt[arm], 3)))
            if arm in ("k01", "ISO") and rng > DEAD_BAR:
                chk(False, "%s: seed range %.3f <= DEAD_BAR %.2f" % (arm, rng, DEAD_BAR))
                unresolved.append("UNRESOLVED-DIVERGED")
            elif rng > DEAD_BAR:
                stamps.append("%s-SEEDS-DIVERGE" % arm)
        else:
            print("  %-4s INCOMPLETE (%d of 3 plateau5)" % (arm, len(vs)))
    bands = {"k01": band(csv_path, "scalar"), "k62": band(csv_path, "layerwise")}
    for arm, gran in (("k01", "scalar"), ("k62", "layerwise")):
        b = bands[arm]
        if b is None:
            skip("%s: corpus band unavailable" % arm)
            continue
        print("  corpus %-9s n %2d  %.4f +/- %.4f  band [%.3f, %.3f]" % (gran, b["n"], b["mean"], b["sd"], b["lo"], b["hi"]))
        if arm in lev:
            inside = b["lo"] <= lev[arm] <= b["hi"]
            if not chk(inside, "%s in-batch level %.4f inside the corpus band" % (arm, lev[arm])):
                unresolved.append("UNRESOLVED-DIVERGED")
    if own_rows(csv_path) == 0:
        stamps.append("FLOOR-NOT-INGESTED")

    # ---- B: the beta states ------------------------------------------------------------------
    print("\n[B] BETA STATES on the m=2 arms (coordinate 0 = the big group, 1 = the isolated group)")
    state = {}
    for arm in M2_ARMS:
        for k, lab in ((0, "big"), (1, "isolated")):
            occs, terms = [], []
            for seed in SEEDS:
                P = probes.get((arm, seed))
                if not P or not P["recs"]:
                    continue
                occ, term, n = occupancy(P["recs"], k)
                if occ is not None:
                    occs.append(occ)
                if term is not None:
                    terms.append(term)
                print("  %-4s s%d  beta[%d] (%s) floor occupancy %s  terminal %s  -> %s"
                      % (arm, seed, k, lab, fmt(occ), fmt(term, 3), free_state(occ, term)))
            if occs and terms:
                st = free_state(statistics.mean(occs), statistics.mean(terms))
                state[(arm, k)] = st
                print("  %-4s     beta[%d] pooled occupancy %.4f  mean terminal %.3f  -> %s"
                      % (arm, k, statistics.mean(occs), statistics.mean(terms), st))
    iso59 = state.get(("ISO", 0), "UNKNOWN")
    stamps.append("ISO-59-" + iso59)
    stamps.append("ISO-3-" + state.get(("ISO", 1), "UNKNOWN"))
    stamps.append("CTRL-59-" + state.get(("CTRL", 0), "UNKNOWN"))
    stamps.append("ONE-61-" + state.get(("ONE", 0), "UNKNOWN"))

    # ---- P: the premise, replicated in batch on k01 -------------------------------------------
    print("\n[P] PREMISE -- the in-batch scalar anchor's own carrying set (cTD2's statistic, PRIMARY L)")
    P = premise_from_scalar_records([probes[("k01", s)]["recs"] for s in SEEDS if ("k01", s) in probes])
    print("  determinate %d  DISAGREE %d  R_T %s  pinned %d" % (P["n_det"], P["n_dis"], fmt(P["R_T"]), P["pin_n"]))
    for nm, v in P["shares"].items():
        print("    %-30s carrier share on DISAGREE %s" % (nm, fmt(v)))
    print("    top carriers: %s" % ", ".join("%s %.3f" % t for t in P["top"]))
    print("    pinned: trio dominates %s  tensor 50 alone %s  pair {53,59} %s  59-rest votes UP %s"
          % (fmt(P["three"]), fmt(P["one"]), fmt(P["pair"]), fmt(P["rest_up"])))
    stamps.append("PREMISE-REPLICATES" if premise_ok(P) else "PREMISE-NOT-REPLICATED")

    # ---- D: the contrasts ------------------------------------------------------------------------
    print("\n[D] CONTRASTS against the in-batch floor anchor (SE_ARM_DIFF %.6f pp)" % SE_ARM_DIFF)
    d_iso = lev["ISO"] - lev["k01"] if "ISO" in lev and "k01" in lev else None
    d_ctrl = lev["CTRL"] - lev["k01"] if "CTRL" in lev and "k01" in lev else None
    d_one = lev["ONE"] - lev["k01"] if "ONE" in lev and "k01" in lev else None
    d_ceil = lev["k62"] - lev["k01"] if "k62" in lev and "k01" in lev else None
    for lab, d in (("D_ISO  = ISO  - k01", d_iso), ("D_CTRL = CTRL - k01", d_ctrl),
                   ("D_ONE  = ONE  - k01", d_one), ("D_CEIL = k62  - k01", d_ceil)):
        print("  %s : %s" % (lab, se_str(d)))
    if all(a in levt and levt[a] is not None for a in ("ISO", "k01", "CTRL", "ONE")):
        print("  TRAIN: ISO-k01 %+.3f  CTRL-k01 %+.3f  ONE-k01 %+.3f"
              % (levt["ISO"] - levt["k01"], levt["CTRL"] - levt["k01"], levt["ONE"] - levt["k01"]))
    print("  bars: SUPPORT %.1f pp (%.2f SE)  REFUTE %.1f pp (%.2f SE)  READ %.6f pp (2 SE)"
          % (SUPPORT_BAR, SUPPORT_BAR / SE_ARM_DIFF, REFUTE_BAR, REFUTE_BAR / SE_ARM_DIFF, READ_BAR))
    if "ISO" in lev:
        print("  DESCRIPTIVE: ISO %.4f vs point prediction %.1f [%.0f, %.0f]; vs corpus k49 %.4f (%+.3f); vs k62 in batch %s"
              % (lev["ISO"], ISO_POINT, ISO_RANGE[0], ISO_RANGE[1], L_K49, lev["ISO"] - L_K49,
                 ("%+.3f" % (lev["ISO"] - lev["k62"])) if "k62" in lev else "n/a"))
    stamps.append("ONE-" + level_token(d_one))

    # ---- FINAL ------------------------------------------------------------------------------------
    for u in ("UNRESOLVED-MANIFEST-MISMATCH", "UNRESOLVED-PROVENANCE", "UNRESOLVED-DIVERGED"):
        if u in unresolved:
            return finish(u, stamps)
    return finish(decide(d_iso, d_ctrl, iso59, stamps), stamps)


def finish(branch, stamps):
    print("\n" + "=" * 78)
    print("FINAL: %s | %s" % (branch, " | ".join(stamps) if stamps else "no stamps"))
    print("=" * 78)
    print("PRIMARY: plateau5 @ 100 epochs from the runs' own .out files, in-batch contrasts against k01,")
    print("bars frozen in this file (SE_ARM_DIFF %.6f from a corpus reader that excludes ciso1-* rows)." % SE_ARM_DIFF)
    return branch

def selftest(csv_path, runsdir):
    print("=" * 78)
    print("cIS1 --selftest: the grammar, the frozen partitions, the noise floor, the anchors, the branch")
    print("=" * 78)
    names = TENSOR_NAMES
    chk(len(names) == N_TENSORS and len(TENSOR_NUMEL) == N_TENSORS and sum(TENSOR_NUMEL) == N_PARAMS,
        "62 names, 62 numels, 11,220,132 parameters")
    chk(all(n in names for n in CARRIERS + CONTROLS), "carriers and controls are live parameter names")
    chk(tuple(names.index(n) + 1 for n in CARRIERS) == (50, 53, 59), "carrier 1-based indices are (50, 53, 59)")
    chk(tuple(names.index(n) + 1 for n in CONTROLS) == (20, 23, 29), "control 1-based indices are (20, 23, 29)")
    chk(all(TENSOR_NUMEL[names.index(n)] == 512 for n in CARRIERS), "every carrier has 512 parameters")
    chk(all(TENSOR_NUMEL[names.index(n)] == 128 for n in CONTROLS), "every control has 128 parameters (numel NOT matched, disclosed)")
    non_car_512 = [n for n in names if n.endswith(".weight") and ("bn" in n or "shortcut.1" in n)
                   and TENSOR_NUMEL[names.index(n)] == 512 and n not in CARRIERS]
    chk(len(non_car_512) == 2, "exactly TWO non-carrier 512-param BN scales exist on the model (a numel-matched triple is impossible)",
        str(non_car_512))
    for arm in M2_ARMS:
        g = spec_to_groups(SPEC[arm])
        chk(len(g) == 2 and tuple(len(x) for x in g) == SIZES[arm] and tuple(g[1]) == ISOLATED[arm],
            "%s parses to m=2 sizes %s, isolated %s" % (arm, SIZES[arm], ISOLATED[arm]))
        pc = tuple(sum(TENSOR_NUMEL[i - 1] for i in x) for x in g)
        chk(pc == PARAMS[arm], "%s params %s" % (arm, PARAMS[arm]), str(pc))
        chk(sorted(g[0] + g[1]) == list(range(1, 63)), "%s covers all 62 exactly once" % arm)
        tok = re.match(r"^[A-Za-z0-9_.,:/-]+$", SPEC[arm]) is not None and not SPEC[arm].startswith("--")
        chk(tok, "%s spec is ONE shell token of [A-Za-z0-9_.,:/-] (RULE 20, argsline_guard.py unedited)" % arm)
    gi, gc, go = spec_to_groups(SPEC["ISO"]), spec_to_groups(SPEC["CTRL"]), spec_to_groups(SPEC["ONE"])
    chk(set(gi[1]) ^ set(gc[1]) == {50, 53, 59, 20, 23, 29} and set(gi[0]) ^ set(gc[0]) == {50, 53, 59, 20, 23, 29},
        "ISO ^ CTRL is exactly the six indices, three swapped at identical sizes")
    chk(set(gi[1]) - set(go[1]) == {53, 59} and go[1] == [50], "ONE isolates 50 only; ISO adds 53 and 59")
    # the grammar is loud
    for bad in ("sets:1-50/50-62", "sets:1-49/51-62", "sets:1-49/50-62/x"):
        try:
            spec_to_groups(bad)
            chk(False, "grammar rejects %r" % bad)
        except AssertionError:
            chk(True, "grammar rejects %r" % bad)
    # arithmetic
    chk(abs(SIGMA_W * math.sqrt(2 / 3) - SE_ARM_DIFF) < 1e-6, "SE_ARM_DIFF = SIGMA_W*sqrt(2/3)", "%.6f" % (SIGMA_W * math.sqrt(2 / 3)))
    chk(abs(2 * SE_ARM_DIFF - READ_BAR) < 1e-6, "READ_BAR = 2 SE")
    chk(SIGMA_W == max(SIGMA_NARROW, SIGMA_WIDE), "SIGMA_W is the max of the two estimators")
    print("       bars in SE: SUPPORT %.2f  REFUTE %.2f  READ 2.00  DEAD %.2f"
          % (SUPPORT_BAR / SE_ARM_DIFF, REFUTE_BAR / SE_ARM_DIFF, DEAD_BAR / SE_ARM_DIFF))
    chk(abs((L_K49 - L_FLOOR) - 32.6327) < 1e-3, "k49 - floor = +32.63 pp (SUPPORT_BAR is 77 %% of it: %.3f)"
        % (SUPPORT_BAR / (L_K49 - L_FLOOR)))
    chk(REFUTE_BAR < L_K50 - L_FLOOR, "REFUTE_BAR sits below the ladder's worst partial state k50 - floor = %+.3f" % (L_K50 - L_FLOOR))
    chk(abs(math.log(A0) - (-13.815510557964274)) < 1e-9 and BETA_LO + FREE_TERMINAL_MARGIN == -13.0,
        "beta0 = ln(1e-6) = -13.8155; FREE terminal bar -13.0")
    # regexes
    a = ("ARGS: --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 "
         "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 "
         "--dataset CIFAR100 --NN-name ResNet18_c100 --batch-size 100 --max-time 999:00:00 --gamma 1 "
         "--meta-stepsize 1e-3 --alpha0 1e-6 --num-epochs 100 --stepsize-groups %s --seed 21 "
         "--save-directory /cluster/account_b/metaopt/runs/ciso1 --run-name ciso1-%s-s21")
    for arm in ARMS:
        chk(ARGS_RE[arm].match(a % (SPEC[arm], arm)) is not None, "ARGS regex matches the registered %s line" % arm)
    chk(ARGS_RE["ISO"].match(a % (SPEC["CTRL"], "ISO")) is None, "ARGS regex rejects a swapped spec")
    chk(ARGS_RE["k01"].match((a % ("scalar", "k01")).replace("1e-3", "1e-4")) is None, "ARGS regex rejects a wrong ms")
    e = ("ENV: AUGMENT=1 BETA_CLIP=-15:-2.3026 HIER=none LAM=na ETA_RATIO=na COS_TOTAL=default COS_WARMUP=default "
         "SCHED=none SCHED_TOTAL=none SCHED_WARMUP=none SCHED_MIN=none PROBE=100 "
         "PROBE_DIR=/cluster/account_b/metaopt/runs/ciso1/probe_ciso1-ISO-s21 EB_RHO=na EB_LOG=0")
    chk(re.sub(r" PROBE_DIR=\S+", "", e).replace("ENV: ", "") == ENV_FIXED, "ENV line modulo PROBE_DIR is the registered one")
    chk(PT_RE.match("PROBE_TENSOR: on every=100 type=blockwise tensors=62 meta_alg=Lion momentum_param=0.99 "
                    "Lion_beta2=0.9 dir=/x/probe_ciso1-ISO-s21") is not None, "PROBE_TENSOR regex matches a blockwise ON line")
    chk(EPTR_RE.search("Epoch 99, Train Accuracy: 23.65 %, Test Accuracy: 23.68 %") is not None, "epoch-line regex")
    chk(OUT_RE.match("ciso1-CTRL-s22-4925000.out") is not None and OUT_RE.match("ctd1-sc-s18-4924919.out") is None, "out-file regex")
    # decisions
    st = []
    chk(decide(30.0, 0.5, "FREE", st) == "CAUSAL-SUPPORTED" and "SPECIFIC:CTRL-NULL" in st, "decide: support + null control + free 59")
    st = []
    chk(decide(30.0, 0.5, "PINS", st) == "UNRESOLVED-PARTIAL" and "ISO-RESCUES-BUT-59-PINS" in st, "decide: rescue with pinned 59 is PARTIAL")
    st = []
    chk(decide(30.0, 28.0, "FREE", st) == "NOT-SPECIFIC", "decide: control rescues -> NOT-SPECIFIC before the ISO read")
    st = []
    chk(decide(2.0, 0.0, "PINS", st) == "CAUSAL-REFUTED", "decide: D_ISO <= 5 -> REFUTED")
    st = []
    chk(decide(12.0, 3.0, "FREE", st) == "UNRESOLVED-PARTIAL" and "SPECIFIC:CTRL-PARTIAL" in st, "decide: middle -> PARTIAL, control partial")
    chk(free_state(0.0, -8.0) == "FREE" and free_state(0.63, -15.0) == "PINS" and free_state(0.3, -15.0) == "NEITHER"
        and free_state(0.05, -14.5) == "NEITHER", "free_state rule")
    chk(level_token(26.0) == "RESCUES" and level_token(4.0) == "FLOOR" and level_token(10.0) == "PARTIAL", "level_token rule")
    recs = [{"beta": [-15.0, -15.0]}] * 6 + [{"beta": [-10.0, -15.0]}] * 4
    occ, term, n = occupancy(recs, 0)
    chk(abs(occ - 0.6) < 1e-12 and term == -10.0 and n == 10, "occupancy on synthetic records")
    chk(carrying_set([5.0, -1.0, -1.0, -1.0, 0.5], 1) == [0], "carrying set: one dominant term")
    # the manifest round trip
    ln = ("ARMSPEC ISO SPEC %s M 2 SIZES 59,3 PARAMS 11218596,1536 GROUP0 %s GROUP1 %s"
          % (SPEC["ISO"], " ".join(names[i - 1] for i in gi[0]), " ".join(names[i - 1] for i in gi[1])))
    import tempfile
    tf = tempfile.NamedTemporaryFile("w", suffix=".txt", delete=False)
    tf.write("NUM_PARAM_TENSORS 62\n" + "\n".join("TENSOR %d %s %d" % (i + 1, names[i], TENSOR_NUMEL[i]) for i in range(62))
             + "\n" + ln + "\nISOLATED ISO 50,53,59\n")
    tf.close()
    man = read_manifest(tf.name)
    os.unlink(tf.name)
    chk(man["arms"]["ISO"]["sizes"] == (59, 3) and len(man["arms"]["ISO"]["group0"]) == 59
        and man["arms"]["ISO"]["group1"] == list(CARRIERS) and man["meta"]["isolated"]["ISO"] == (50, 53, 59),
        "manifest reader round-trips an ARMSPEC line")

    # ---- the corpus: EVERY reader excludes ciso1-* -----------------------------------------
    print("\n  THE CORPUS (%s), ciso1-* EXCLUDED FROM EVERY READER" % csv_path)
    if not os.path.isfile(csv_path):
        skip("corpus not reachable; the frozen constants stand as registered")
    else:
        own = own_rows(csv_path)
        print("       ciso1-* rows present in the file (excluded everywhere): %d" % own)
        sn, dfn, cn, mn = noise_floor_from_csv(csv_path, wide=False)
        sw, dfw, cw, mw = noise_floor_from_csv(csv_path, wide=True)
        print("       SIGMA_NARROW %.6f df %d cells %d members %d" % (sn, dfn, cn, mn))
        print("       SIGMA_WIDE   %.6f df %d cells %d members %d" % (sw, dfw, cw, mw))
        chk(abs(sn - SIGMA_NARROW) < 1e-6, "SIGMA_NARROW re-derived == frozen 0.925518", "%.6f" % sn)
        chk(abs(sw - SIGMA_WIDE) < 1e-6, "SIGMA_WIDE re-derived == frozen 0.889041", "%.6f" % sw)
        for lab, grans, want, wn in (("floor (scalar)", ("scalar",), L_FLOOR, L_FLOOR_N),
                                     ("ceiling (layerwise)", ("layerwise",), L_CEIL, L_CEIL_N),
                                     ("k49 pooled", ("[49,13]", "sets:1-49/50-62"), L_K49, L_K49_N),
                                     ("k50", ("[50,12]",), L_K50, L_K50_N), ("k53", ("[53,9]",), L_K53, L_K53_N)):
            L = cell_level(csv_path, grans)
            chk(L is not None and L["n"] == wn and abs(L["mean"] - want) < 5e-5,
                "%s re-derived: n %s mean %s == frozen %.4f (n %d)" % (lab, L and L["n"], L and fmt(L["mean"]), want, wn))
        b = band(csv_path, "scalar")
        chk(b is not None and abs(b["sd"] - L_FLOOR_SD) < 5e-5, "scalar SD %.4f" % L_FLOOR_SD, fmt(b and b["sd"]))
        b = band(csv_path, "layerwise")
        chk(b is not None and abs(b["sd"] - L_CEIL_SD) < 5e-5, "layerwise SD %.4f" % L_CEIL_SD, fmt(b and b["sd"]))
        rows = _csv_rows(csv_path)
        chk(not any(r["network"] == NET and r["seed"] in ("21", "22", "23") for r in rows),
            "no ResNet18_c100 row (ciso1 excluded) carries seed 21, 22 or 23")
        chk(not any(r["granularity"] in (SPEC["ISO"], SPEC["CTRL"], SPEC["ONE"]) for r in rows),
            "none of the three m=2 specs has ever been run (ciso1 excluded)")

    # ---- the premise on ctd1's records, when reachable ----------------------------------------
    print("\n  THE PREMISE, on ctd1's scalar PROBE_TENSOR records (SKIP when unreachable)")
    cands = []
    if runsdir:
        cands.append(os.path.join(runsdir, "ctd1"))
    ws = os.environ.get("METAOPT_WS")
    if ws:
        cands.append(os.path.join(ws, "runs", "ctd1"))
    root = next((c for c in cands if os.path.isdir(c)), None)
    if root is None:
        skip("ctd1 probe root not reachable (pass --runsdir or set METAOPT_WS)")
    else:
        by = []
        for s in (18, 19, 20):
            pj = os.path.join(root, "probe_ctd1-sc-s%d" % s, "probe.jsonl")
            if os.path.isfile(pj):
                by.append([json.loads(l) for l in open(pj) if l.strip()])
        if len(by) < 3:
            skip("fewer than 3 ctd1 scalar probe files under %s" % root)
        else:
            P = premise_from_scalar_records(by)
            print("       determinate %d DISAGREE %d R_T %s pinned %d" % (P["n_det"], P["n_dis"], fmt(P["R_T"]), P["pin_n"]))
            chk(P["n_det"] == 1499 and P["n_dis"] == 1239, "ctd1: 1,499 determinate, 1,239 DISAGREE (cTD2's count)")
            chk(all(v >= 0.999 for v in P["shares"].values()), "ctd1: carriers 50/53/59 on >= 0.999 of DISAGREE records",
                str({k: fmt(v, 3) for k, v in P["shares"].items()}))
            chk(P["pin_n"] == 944 and P["rest_up"] == 1.0, "ctd1: 944 pinned records, the 59-rest votes UP on all of them")
            chk(P["three"] is not None and P["three"] >= 0.97, "ctd1 pinned: the trio dominates the rest on >= 0.97", fmt(P["three"]))
            chk(P["one"] == 0.0 and P["pair"] == 0.0, "ctd1 pinned: tensor 50 alone 0.0000, pair {53,59} 0.0000 (the ONE-arm premise)")
            chk(premise_ok(P), "ctd1 passes the in-batch PREMISE rule (>= 0.90 each)")

    print("\n%s" % ("SELFTEST ALL PASS" if not GATE_FAIL else "SELFTEST FAILURES: %r" % GATE_FAIL))
    return 0 if not GATE_FAIL else 1


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default=None)
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default=None)
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default=None, help="selftest: where ctd1/ lives")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if a.runsdir is None:
        ap.error("runsdir is required (or --selftest)")
    score(a.runsdir, a.csv, a.manifest)


if __name__ == "__main__":
    main()
