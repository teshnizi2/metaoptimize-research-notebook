def avg(v):
    return sum(v) / float(len(v))


def sdev(v):
    if len(v) < 2:
        return 0.0
    m = avg(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def med(v):
    s = sorted(v)
    n = len(s)
    return s[n // 2] if n % 2 else 0.5 * (s[n // 2 - 1] + s[n // 2])

def is_num(t):
    if t == "nan":
        return True
    if not t or any(c not in "0123456789." for c in t):
        return False
    return t.count(".") <= 1 and t != "."


def parse_epoch(line):
    """`Epoch N, Train Accuracy: X %, Test Accuracy: Y %` anywhere in the line -> (N, X, Y) or None."""
    i = line.find("Epoch")
    while i >= 0:
        rest = line[i + len("Epoch"):]
        parts = rest.split(",")
        if len(parts) >= 3 and rest[:1] in (" ", "\t"):
            n = parts[0].strip()
            tr = parts[1].strip()
            te = parts[2].strip()
            if n.isdigit() and tr.startswith("Train Accuracy:") and te.startswith("Test Accuracy:"):
                x = tr[len("Train Accuracy:"):].strip()
                if x.endswith("%"):
                    x = x[:-1].strip()
                y = te[len("Test Accuracy:"):].strip().split()
                y = y[0] if y else ""
                if y.endswith("%"):
                    y = y[:-1]
                if is_num(x) and is_num(y):
                    return int(n), float(x), float(y)
        i = line.find("Epoch", i + 1)
    return None


def kind_on(line):
    c = line.find(":")
    if c < 2:
        return False
    head = line[:c]
    if not head[0].isupper() or any(ch not in "ABCDEFGHIJKLMNOPQRSTUVWXYZ_" for ch in head):
        return False
    tail = line[c + 1:]
    if not tail.startswith(" on"):
        return False
    nxt = tail[3:4]
    return nxt == "" or not (nxt.isalnum() or nxt == "_")


def read_out(path):
    r = dict(args=None, env=None, gpu=None, done=False, ep={}, dup=0, kinds=0, pt=False, nargs=0, nenv=0, minutes=None)
    for ln in open(path, errors="replace").read().splitlines():
        if ln.startswith("ARGS:"):
            r["nargs"] += 1
            if r["args"] is None:
                r["args"] = ln
        elif ln.startswith("ENV:"):
            r["nenv"] += 1
            if r["env"] is None:
                r["env"] = ln
        elif ln.startswith("NVIDIA") and r["gpu"] is None:
            r["gpu"] = ln.split(",")[0].strip()
        elif ln.strip() == "RUN_DONE":
            r["done"] = True
        elif ln.startswith("PROBE_TENSOR:"):
            r["pt"] = True
        elif ln.startswith("NODE="):
            pass
        else:
            e = parse_epoch(ln)
            if e:
                if e[0] in r["ep"]:
                    r["dup"] += 1
                else:
                    r["ep"][e[0]] = (e[1], e[2])
            if kind_on(ln):
                r["kinds"] += 1
            w = ln.split()
            if len(w) == 2 and w[1] == "minutes" and w[0].isdigit():
                r["minutes"] = int(w[0])
    r["complete"] = r["done"] and all(k in r["ep"] for k in range(N_EPOCHS))
    return r


def split_args(line):
    toks = line[len("ARGS:"):].split()
    if len(toks) % 2:
        return None
    pairs = []
    for i in range(0, len(toks), 2):
        if not toks[i].startswith("--") or toks[i + 1].startswith("--"):
            return None
        pairs.append((toks[i][2:], toks[i + 1]))
    return pairs


def read_probe(path):
    recs, bad = [], 0
    for ln in open(path, errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            d = json.loads(ln)
        except ValueError:
            bad += 1
            continue
        if not isinstance(d, dict):
            bad += 1
            continue
        recs.append(d)
    return recs, bad


def rec_med_shrink(b):
    """the median over the record's step sizes of exp(beta) * wd (caw2's definition, 290)."""
    e = sorted(math.exp(x) for x in b)
    n = len(e)
    return (e[n // 2] if n % 2 else 0.5 * (e[n // 2 - 1] + e[n // 2])) * WD


def csv_split(line):
    out, cur, q, i = [], [], False, 0
    while i < len(line):
        c = line[i]
        if q:
            if c == '"':
                if i + 1 < len(line) and line[i + 1] == '"':
                    cur.append('"')
                    i += 1
                else:
                    q = False
            else:
                cur.append(c)
        elif c == '"':
            q = True
        elif c == ",":
            out.append("".join(cur))
            cur = []
        else:
            cur.append(c)
        i += 1
    out.append("".join(cur))
    return out


def corpus_count(csvp, tsvp):
    if not (csvp and tsvp and os.path.exists(csvp) and os.path.exists(tsvp)):
        return None
    ex, head = set(), None
    for ln in open(tsvp):
        ln = ln.rstrip("\n")
        if not ln.strip() or ln.startswith("#"):
            continue
        f = ln.split("\t")
        if head is None:
            head = f
            continue
        d = dict(zip(head, f))
        ex.add((d.get("run"), d.get("job_id")))
    L = open(csvp).read().splitlines()
    h = csv_split(L[0])
    ir, ij = h.index("run"), h.index("job_id")
    n = 0
    for ln in L[1:]:
        if not ln:
            continue
        f = csv_split(ln)
        if f[ir].startswith(BATCH + "-"):
            continue
        if (f[ir], f[ij]) in ex:
            continue
        n += 1
    return n


def sha_file(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        while True:
            b = fh.read(1 << 22)
            if not b:
                break
            h.update(b)
    return h.hexdigest()


def emax_quad(k, n=200000, lo=-10.0, hi=10.0):
    """E[max of k iid N(0,1)] = integral of x * k * phi(x) * Phi(x)^(k-1), composite Simpson."""
    hstep = (hi - lo) / n
    tot = 0.0
    for i in range(n + 1):
        x = lo + i * hstep
        f = x * k * math.exp(-0.5 * x * x) / math.sqrt(2 * math.pi) * (0.5 * (1 + math.erf(x / math.sqrt(2)))) ** (k - 1)
        w = 1 if i in (0, n) else (4 if i % 2 else 2)
        tot += w * f
    return tot * hstep / 3.0


def chi2_cdf_simpson(x, df, n=20000):
    """P(chi2_df <= x) by composite Simpson on the chi-square density -- NOT the scorer's series / continued fraction."""
    if x <= 0:
        return 0.0
    k = df / 2.0
    lg = math.lgamma(k) + k * math.log(2.0)
    hstep = x / n
    tot = 0.0
    for i in range(n + 1):
        t = i * hstep
        f = 0.0 if t <= 0 else math.exp((k - 1.0) * math.log(t) - 0.5 * t - lg)
        w = 1 if i in (0, n) else (4 if i % 2 else 2)
        tot += w * f
    return tot * hstep / 3.0


def chi2_quantile_indep(p, df):
    lo, hi = 0.0, max(10.0, 10.0 * df)
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        if chi2_cdf_simpson(mid, df) < p:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def pick(cands, key):
    best = None
    for c, x in cands:
        if best is None or x[key] > best[1][key]:
            best = (c, x)
    return best[0] if best else None


def read_word(k_ok, k_lvl, p_ok, p_lvl_ok, t_ch, t_nd, se, b):
    if not k_ok:
        return "SCALAR-DIVERGED"
    if k_lvl < HEALTH:
        return "SCALAR-COLLAPSED"
    if not (p_ok and p_lvl_ok):
        return "UNREADABLE"
    lo_ch, lo_nd = t_ch - b, t_nd - b
    if lo_ch >= SURV and lo_nd >= SURV and lo_ch > RES * se and lo_nd > RES * se:
        return "SCALAR-BEATS-BEST"
    tm = min(t_ch, t_nd)
    if tm + b <= -SURV and tm + b < -RES * se:
        return "PARTITION-ABOVE-SCALAR"
    if tm - b - RES * se >= -SURV:
        return "SCALAR-TIES-BEST"
    if tm + b + RES * se <= SURV:
        return "PARTITION-TIES-SCALAR"
    return "UNRESOLVED"

def cfg_word(k_lvl, k_div, ok, t_ch, t_nd, se):
    if k_div:
        return "DIVERGED"
    if k_lvl < HEALTH:
        return "COLLAPSED"
    if not ok:
        return "UNREADABLE"
    return CFG_MAP[read_word(True, k_lvl, True, True, t_ch, t_nd, se, 0.0)]


def branch_word(t, o, sigma_ok):
    if t in GATES:
        return "BRANCH-UNREADABLE"
    if not sigma_ok:
        return "BRANCH-SIGMA-INCONSISTENT"
    if t == "PARTITION-ABOVE-SCALAR":
        return "HEADLINE-REFUTED-BY-RETUNE"
    if t == "SCALAR-BEATS-BEST":
        return "NOT-A-TUNING-ARTEFACT" if o == "SCALAR-BEATS-BEST" else "SELECTION-DEPENDENT"
    if t == "SCALAR-TIES-BEST":
        return "WEAKENED-TO-TIE"
    if t == "PARTITION-TIES-SCALAR":
        return "TIE-SCALAR-LEAD-BOUNDED"
    return "RETUNE-UNDECIDED"


def loc_word(g, cfg, ceil_bulk):
    if cfg is None:
        return "NO-SELECTION"
    if ceil_bulk:
        return "NOT-LOCATED-CEILING"
    if cfg == CENTRE:
        return "LOCATED"
    if cfg == "A2":
        return "LOCATED-A0-XB" if CRT1_A2_MINUS_M2[g] > CRT1_2SE else "NOT-LOCATED-A0LO"
    return "NOT-LOCATED-" + EDGE_OF[cfg]


def selected(A, sel, sig, b):
    k = sel["k01"]
    p_ok = sel["ch"] is not None and sel["nd"] is not None
    if k is None:
        return "SCALAR-DIVERGED", None, None, None
    kx = A["k01" + k]
    if not p_ok:
        return read_word(True, kx["level"], False, False, 0, 0, 1, b), None, None, None
    cx, nx = A["ch" + sel["ch"]], A["nd" + sel["nd"]]
    t_ch, t_nd = kx["level"] - cx["level"], kx["level"] - nx["level"]
    se = sig * math.sqrt(1.0 / kx["n"] + 1.0 / cx["n"])
    r = read_word(True, kx["level"], True, cx["level"] >= HEALTH and nx["level"] >= HEALTH, t_ch, t_nd, se, b)
    return r, t_ch, t_nd, se


def decide(A, sigma, sigma_ok=True):
    b = EMAX5_TYPED * sigma / math.sqrt(N_SEED)
    st, so = {}, {}
    for g in G_IDS:
        cands = [(c, A[g + c]) for c in CFG_IDS if A[g + c]["range"] <= DIVG]
        st[g] = pick(cands, "train")
        so[g] = pick(cands, "level")
    rt = selected(A, st, sigma, b)
    ro = selected(A, so, sigma, b)
    loc = dict((g, loc_word(g, st[g], st[g] is not None and A[g + st[g]]["ceil_bulk"])) for g in G_IDS)
    locword = "LOC-ALL" if all(v in ("LOCATED", "LOCATED-A0-XB") for v in loc.values()) else "LOC-PARTIAL"
    cw, T, Dd = {}, {}, {}
    for c in CFG_IDS:
        k, ch, nd = A["k01" + c], A["ch" + c], A["nd" + c]
        t_ch, t_nd = k["level"] - ch["level"], k["level"] - nd["level"]
        se = sigma * math.sqrt(1.0 / k["n"] + 1.0 / ch["n"])
        ok = ch["level"] >= HEALTH and nd["level"] >= HEALTH and ch["range"] <= DIVG and nd["range"] <= DIVG
        cw[c] = cfg_word(k["level"], k["range"] > DIVG, ok, t_ch, t_nd, se)
        T[c] = (t_ch, t_nd, se)
        Dd[c] = (ch["level"] - nd["level"], se)
    branch = branch_word(rt[0], ro[0], sigma_ok)
    return b, st, so, rt, ro, cw, T, Dd, loc, locword, branch


def sigma_change_points(A, lo=0.05, hi=0.60, step=0.0005):
    out, prev = [], None
    n = int(round((hi - lo) / step))
    for i in range(n + 1):
        s = lo + i * step
        r = "TUNED-" + decide(A, s)[3][0]
        if prev is not None and r != prev[1]:
            out.append((s, prev[1], r))
        prev = (s, r)
    return out

def crt1_levels(runsdir):
    """RE-DERIVE crt1's A2 levels, its A2 - M2 contrasts and its in-batch pooled sd from crt1's OWN 36 raw .out files."""
    sub = os.path.join(runsdir, "crt1")
    lv, ss, df = {}, 0.0, 0
    for c, _ms, _a0t, _a0 in CRT1_CFGS:
        for g in G_IDS:
            xs = []
            for s in CRT1_SEEDS:
                f = None
                for base in (runsdir, sub):
                    if not os.path.isdir(base):
                        continue
                    for fn in sorted(os.listdir(base)):
                        if fn.startswith("crt1-%s%s-s%d-" % (g, c, s)) and fn.endswith(".out"):
                            f = os.path.join(base, fn)
                if f is None:
                    return None
                r = read_out(f)
                if not r["complete"]:
                    return None
                xs.append(avg([r["ep"][e][1] for e in range(PL_LO, PL_HI + 1)]))
            m_ = avg(xs)
            lv[g + c] = m_
            ss += sum((x - m_) ** 2 for x in xs)
            df += len(xs) - 1
    return lv, math.sqrt(ss / df), df
