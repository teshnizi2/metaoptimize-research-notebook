set -e
cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
CLIP="BETA_CLIP=-15:-2.3026"
# SGDm+Lion: the cell where granularity has the largest effect at ResNet-18
# (scalar 88.07 vs layerwise 91.31), so a reversal at depth is maximally visible.
B="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0 --meta-stepsize 1e-3 --alpha0 1e-6 --gamma 1 --dataset CIFAR10 --num-epochs 100 --batch-size 100 --max-time 999:00:00"

sub () { # 1=name 2=nn 3=granularity 4=seed 5=walltime
  sbatch --parsable --job-name=$1-s$4 --partition=gpu-l4-24g --gres=gpu:l4:1 --time=$5 \
    --export="ALL,AUGMENT=1,$CLIP" \
    jobs/run_cifar.sh $B --NN-name $2 --stepsize-groups $3 --seed $4 \
    --save-directory $R/scale --run-name $1_s$4 >/dev/null
}

for s in 0 1; do
  sub sc-r34-scal  ResNet34  scalar    $s 06:00:00
  sub sc-r34-layer ResNet34  layerwise $s 06:00:00
  sub sc-r50-scal  ResNet50  scalar    $s 10:00:00
  sub sc-r50-layer ResNet50  layerwise $s 10:00:00
  sub sc-r101-scal  ResNet101 scalar    $s 20:00:00
  sub sc-r101-layer ResNet101 layerwise $s 20:00:00
done
sleep 3
echo "LADDER_SUBMITTED=$(squeue -h -u account_a -o '%j' | grep -c '^sc-')"
