#     ms=3e-3  travel 150.0 nats   floor from epoch 15.4, ceiling from epoch  5.9

#       80 ep, R18/CIFAR-10, flat nodewise: floor -30 reads 92.112 (n=2), -60
#       reads 92.006 (n=6), -90 reads 92.146 (n=2); ceiling +2.0 reads 92.112 and

#   The campaign paid for this evidence at cycles 54/71/73 and never used it.

# -----------------------------------------------------------------------------

#   * rl3 CAN decide a FULL INVERSION or a FULL COLLAPSE of D at matched optima.

#   * rl3 CANNOT distinguish D = +0.15 from D = 0, and CANNOT resolve a partial

#   * rl3 CAN return FLAT-PROFILE -- no arm's ms profile varies by more than

# -----------------------------------------------------------------------------
# THE ARMS, AND WHY m IS HELD FIXED ACROSS ALL THREE STEPSIZES (RULE 10)
# -----------------------------------------------------------------------------
#   nodewise    m = 14,420   the architecture-aligned partition (sum of shape[0])

            continue

        if p is not None and p > 80:

    return out


def welch(a, b):
    ma, mb = st.mean(a), st.mean(b)
    va = st.variance(a) if len(a) > 1 else 0.0
    vb = st.variance(b) if len(b) > 1 else 0.0
    se = math.sqrt(va / len(a) + vb / len(b))
    return ma - mb, se

  <<'PYEOF' || guard_fail "guard 8: box reachability / saturation"
import csv, math, sys, statistics as st
from collections import defaultdict
clip, a0, mstl, ep, spe, csvp, net, dset, bar = sys.argv[1:10]
lo, hi = (float(x) for x in clip.split(":"))
b0 = math.log(float(a0))
T = int(ep) * int(spe)
bar = float(bar)
rungs = mstl.split()
d_lo, d_hi = b0 - lo, hi - b0

# 8a -- THE ALGEBRAIC TABLE, every rung, both rails.  |dbeta| = ms exactly when
# --weight-decay-meta 0 (GUARD 1b-3 asserts that flag), so beta is confined to
# [b0 - ms*T, b0 + ms*T] for ANY velocity profile.
print("guard 8a: beta_0 = ln(%s) = %.6f in box (%.4f, %.4f); T = %d updates; the "
      "FLOOR is %.4f nats away and the CEILING %.4f" % (a0, b0, lo, hi, T, d_lo, d_hi))
free = []
for ms in rungs:
    tr = float(ms) * T
    lo_ok, hi_ok = tr >= d_lo, tr >= d_hi
    print("guard 8a:   ms=%-6s travel %7.2f nats  FLOOR %-28s CEILING %s"
          % (ms, tr,
             ("reachable from epoch %.1f" % (d_lo / float(ms) / int(spe))) if lo_ok
             else "UNREACHABLE (provably free)",
             ("reachable from epoch %.1f" % (d_hi / float(ms) / int(spe))) if hi_ok
             else "UNREACHABLE (provably free)"))
    if not (lo_ok or hi_ok):
        free.append(ms)
    else:
        sys.exit("GUARD FAIL: ms=%s is NOT provably box-free in (%.4f, %.4f) -- "
                 "travel %.2f nats against rails %.2f and %.2f away.  CORRECTIONS "
                 "117.3: a statistic from a run at a guard is UNINTERPRETABLE, and "
                 "an unbracketed argmax read off a clipped rung answers nothing "
                 "about RULE 11.  Drop the rung or widen the box."
                 % (ms, lo, hi, tr, d_lo, d_hi))
print("guard 8a: **ALL %d RUNGS ARE PROVABLY BOX-FREE ON BOTH RAILS.**  The binding "
      "rung is ms=%s, which clears the ceiling by %.4f nats and the floor by %.4f."
      % (len(free), rungs[-1], d_hi - float(rungs[-1]) * T,
         d_lo - float(rungs[-1]) * T))

# 8a2 -- **THE FREE LIMIT, AND RULE 13 ON IT.**  The rungs this batch DROPPED must
# be refused by the same arithmetic, or the drop was arbitrary.  ms_max_free is
# the identity (hi - beta_0)/T; above it no run at this budget can be free, and
# the ceiling that WOULD make ms=1e-3 free implies a step size of exp(beta_0+50).
ms_max_free = d_hi / T
print("guard 8a2: the largest box-free meta-stepsize at this budget is "
      "ms_max_free = (%.4f - %.6f)/%d = %.6e" % (hi, b0, T, ms_max_free))
if float(rungs[-1]) >= ms_max_free:
    sys.exit("GUARD FAIL: the top rung %s is at or above ms_max_free %.6e"
             % (rungs[-1], ms_max_free))
for bad in ("1e-3", "3e-3"):
    tb = float(bad) * T
    if tb <= d_hi and tb <= d_lo:
        sys.exit("GUARD FAIL (RULE 13): the same arithmetic calls ms=%s box-free in "
                 "this box.  Then dropping it was unnecessary." % bad)
    need = b0 + tb
    print("guard 8a2 (RULE 13): ms=%-4s travels %.0f nats -> a provably free ceiling "
          "would have to sit at %+.1f, i.e. a step size of %.1e.  REFUSED, and the "
          "upper bracket for RULE 11 is therefore an ARITHMETIC bound rather than a "
          "purchasable measurement." % (bad, tb, need, math.exp(need)))

# 8b -- STRICT WIDENING.  REFUSAL.  The anchors' box is -15:-2.3026, where at
# ms=1e-3 the floor binds from epoch 16.2 and the ceiling from epoch 9.2.  A box
# that is not strictly wider on BOTH rails buys nothing and this batch refuses it.
A_LO, A_HI = -15.0, -2.3026
if not (lo < A_LO and hi > A_HI):
    sys.exit("GUARD FAIL: CLIP=(%.4f, %.4f) is NOT strictly wider than the anchors' "
             "(%.4f, %.4f) on BOTH rails.  At ms=%s the anchors' box binds the floor "
             "from epoch %.1f and the ceiling from epoch %.1f, so every existing "
             "row at that rung in the corpus is guard-uninterpretable.  Running "
             "this ladder in that box would buy 24 more uninterpretable rows."
             % (lo, hi, A_LO, A_HI, rungs[-1], (b0 - A_LO) / float(rungs[-1]) / int(spe),
                (A_HI - b0) / float(rungs[-1]) / int(spe)))
print("guard 8b: CLIP (%.1f, %.1f) is STRICTLY WIDER than the anchors' (%.1f, %.4f) "
      "on both rails (floor +%.1f nats, ceiling +%.4f nats of slack)"
      % (lo, hi, A_LO, A_HI, A_LO - lo, hi - A_HI))

# 8c -- **EMPIRICAL SATURATION, BOTH AXES.**  In the BUILD this was a REFUSAL,
# because nothing was provably free at ms=1e-3 and an empirical argument was all
# there was.  Every emitted rung is now provably free on both rails (8a), so the
# algebraic proof SUPERSEDES this test and 8c is demoted to SUPPORTING EVIDENCE:
# it still runs and still prints, and a large move is still reported, but it may
# no longer refuse a batch whose box is proven free.  Registered here, in the
# guard, so the demotion is visible rather than silent.  If a future edit
# re-introduces a rung that is NOT provably free, 8a refuses it outright and this
# test never gets the chance to be the last line of defence again.
SATURATION_IS_A_REFUSAL = False
R = list(csv.DictReader(open(csvp)))
def f(x):
    try: return float(x)
    except Exception: return None
cells = defaultdict(list)
for r in R:
    if r["network"] != net or r["dataset"] != dset: continue
    if r["granularity"] != "nodewise" or r["meta_stepsize"] != rungs[1]: continue
    if r["alpha0"] != a0 or r["hier"] not in ("", "none"): continue
    if r["collapsed"] != "0" or r["superseded"] not in ("0", ""): continue
    if r["epochs_done"] != r["epochs_requested"]: continue
    p = f(r["plateau5"])
    if p is None: continue
    try: blo, bhi = (float(x) for x in r["beta_clip"].split(":"))
    except Exception: continue
    cells[(r["epochs_done"], blo, bhi)].append(p)
budgets = sorted({k[0] for k in cells if k[1] == lo and k[2] == hi
                  and len(cells[k]) >= 2}, key=int)
if not budgets:
    msg = ("guard 8c: the box %s has no cell with >=2 complete nodewise runs at "
           "ms=%s in the corpus, so its saturation cannot be tested empirically."
           % (clip, rungs[-1]))
    if SATURATION_IS_A_REFUSAL:
        sys.exit("GUARD FAIL: " + msg)
    print(msg + "  NOT A REFUSAL: guard 8a has already proved BOTH RAILS "
          "UNREACHABLE at BOTH rungs (binding slack %.4f nats), and an algebraic "
          "proof does not need empirical support.  What IS lost is the ability to "
          "check the box against measured accuracy, and the scorer's per-seed "
          "occupancy gate -- which now reads n_at_lo/n_at_hi over n_beta rather "
          "than the per-tensor summary (CORRECTIONS 117.1) -- is the check that "
          "replaces it.  It expects 0.000000 and treats anything else as a defect."
          % min(d_hi - float(m) * T for m in rungs))
    budgets = []
bud = budgets[-1] if budgets else None
if bud is not None:
    ref = st.mean(cells[(bud, lo, hi)])
    print("guard 8c: reference cell -- ms=%s, %s ep, box %s: %.3f (n=%d)"
          % (rungs[1], bud, clip, ref, len(cells[(bud, lo, hi)])))
    floor_ax = ceil_ax = 0
    worst = 0.0
    for (b, blo, bhi), v in sorted(cells.items()):
        if b != bud or len(v) < 2: continue
        if (blo, bhi) == (lo, hi): continue
        if blo > lo or bhi < hi: continue          # not strictly wider -- skip
        d = st.mean(v) - ref
        axis = "FLOOR" if blo < lo else ("CEILING" if bhi > hi else "BOTH")
        if blo < lo: floor_ax += 1
        if bhi > hi: ceil_ax += 1
        worst = max(worst, abs(d))
        print("guard 8c:   wider box (%.0f, %.1f) [%s axis] = %.3f (n=%d)  delta %+0.3f"
              % (blo, bhi, axis, st.mean(v), len(v), d))
        if abs(d) > bar:
            m = ("the strictly wider box (%.0f, %.1f) moves plateau5 by %+0.3f pp, "
                 "above the registered BOX_SAT_BAR of %.2f" % (blo, bhi, d, bar))
            if SATURATION_IS_A_REFUSAL:
                sys.exit("GUARD FAIL: " + m)
            print("guard 8c: **REPORTED, NOT A REFUSAL** -- " + m + ".  Guard 8a "
                  "has proved both rails unreachable at every emitted rung, so this "
                  "is a statement about the box's accuracy effect, not about "
                  "clipping.  It must appear beside every LEVEL this batch reports.")
    if (floor_ax == 0 or ceil_ax == 0) and SATURATION_IS_A_REFUSAL:
        sys.exit("GUARD FAIL: the saturation test is VACUOUS -- %d wider box(es) on the "
                 "FLOOR axis and %d on the CEILING axis.  RULE 13 in the guard's own "
                 "structure: a box justified by deepening the floor alone says nothing "
                 "about the ceiling, and at ms=%s the ceiling binds FIRST (epoch %.1f vs "
                 "%.1f)." % (floor_ax, ceil_ax, rungs[1],
                             d_hi / float(rungs[1]) / int(spe),
                             d_lo / float(rungs[1]) / int(spe)))
    print("guard 8c: **EMPIRICAL SATURATION PASSES -- %d wider boxes on the floor axis "
          "and %d on the ceiling axis, worst move %.3f pp against a %.2f pp bar.**  The "
          "campaign paid for this evidence at cycles 54/71/73 and never used it."
          % (floor_ax, ceil_ax, worst, bar))
print("guard 8d: **THE BOX IS PROVED, NOT ARGUED.**  Both rails are algebraically "
      "unreachable at both emitted rungs, with %.4f nats of slack at the binding "
      "one.  The empirical saturation evidence above is SUPPORTING and is measured "
      "at %s epochs against this batch's %s.  Every number rl3 produces is still "
      "read behind the scorer's per-seed occupancy gate -- which now reads "
      "n_at_lo/n_at_hi over n_beta and NOT the 62-entry per-tensor beta summary "
      "(CORRECTIONS 117.1: the old statistic reads 0.000000 on all 24 fa1 runs "
      "while three of them are ceiling-bound in every record, and INVERTS the arm "
      "ranking on ar1) -- and that gate now expects exactly 0.000000."
      % (min(d_hi - float(m) * T for m in rungs), bud if bud else "no", ep))
PYEOF
