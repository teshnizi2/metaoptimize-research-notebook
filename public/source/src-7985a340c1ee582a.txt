#!/bin/bash
# =============================================================================
# Cycle 55 -- THE SPAN / BUDGET DISSOCIATION.  Is the argmin band a band in
# ADAPTATION EXTENT, or does it have a budget scope of its own?
# cluster_b.  9 jobs.  Tag `sp8-*`.
#
# WRITTEN WHILE cluster_a WAS DOWN (FINDINGS 55.0) AND NOT SUBMITTED.  Both login
# nodes were unreachable for the whole of cycle 55; this batch is registered now
# so the next tick submits rather than designs.  Its own guards re-check every
# precondition at submit time.
#
# WHAT CYCLE 55 ESTABLISHED, WITH ZERO JOBS.
#   (a) Adaptation lifts N_eff/m MONOTONICALLY IN BLOCK SIZE -- 4 of 4
#       network-matched frozen->free pairs give lift(w) 3.84-12.03x >
#       lift(node) 1.26-4.73x > lift(lay) 0.72-1.19x ~ 1 (FINDINGS 55.5).  With
#       beta frozen the minimum is WEIGHTWISE in 4 of 4 families.  So the
#       "nodewise minimum" is not a property of the nodewise partition: it is
#       where a large weightwise lift has carried weightwise PAST nodewise.
#   (b) THE LIFT IS TRANSIENT (55.6).  Inside `br6` -- box-free 4/4, n=4 -- the
#       weightwise curve in absolute epoch windows runs
#           0-10ep 0.0092 -> 10-20ep 0.5073 -> 20-30ep 0.2551 -> 30-40ep 0.1201
#       against nodewise 0.0217 -> 0.4074 -> 0.3270 -> 0.2768, so the argmin
#       goes  w -> node -> w  with the crossover BETWEEN EPOCHS 20 AND 30.
#       `bl5`, an independent batch at a different ceiling, reproduces the curve
#       to three decimals.
#   (c) CORRECTIONS 74 (N2) IS CORRECTED (55.7).  Eight cells sit at ms=1e-3 and
#       split 4-4 on the argmin -- the four FROZEN arms (span 0.00 -> `w`)
#       against the four FREE arms (span 12.18-15.87 -> `node`).  A variable held
#       constant across a 4-4 split cannot govern it, so "the flip tracks ms" is
#       WITHDRAWN.  On N2's OWN statistic (max-min of beta on the LAST record,
#       weightwise arm; `c53_score.beta_span`) the 20-epoch argmin separates
#       PERFECTLY, 11 of 11:
#
#           span   0.00  0.00  0.00  0.00  1.96  3.81  8.27 | 12.18 13.56 15.09 15.87
#           argmin   w     w     w     w     w     w     w  |  node  node  node  node
#
#       -- boundary between 8.27 and 12.18.  N2 refuted its own GUESSED "~5 log
#       units" threshold (c52_nodemin_onset.sh:93), not its variable.
#
# THE ONE THING (a)-(c) CANNOT SEPARATE, AND WHY THIS BATCH EXISTS.
#   Span is NOT sufficient: `br6` has the LARGEST span in the inventory (23.85)
#   and gives `w`.  So the argmin is `node` on an INTERVAL of span with `w` on
#   both sides.  But every cell above `node`'s interval is also a 40-EPOCH cell,
#   and every cell below it is a 20-epoch cell.  **Span and budget are perfectly
#   confounded at the upper edge of the band.**  The upper edge might be a span
#   edge (adaptation extent has passed the band) or a budget edge (something
#   about long horizons that span does not encode).
#
#   THIS BATCH BREAKS THAT CONFOUND, and it is the only design that does: run a
#   40-EPOCH arm whose span lands back INSIDE the band.
#
# WHY ms=5e-4 AT 40 EPOCHS, ARITHMETIC SHOWN.
#   N2-statistic spans measured at 20 epochs (FINDINGS 55.7):
#       ms 1e-4 -> 1.96   2e-4 -> 3.81   5e-4 -> 8.27   1e-3 -> 13.56
#   log-log slope over the decade 1e-4 -> 1e-3 is log10(13.56/1.96) = 0.84, so
#   span ~ ms^0.84.  The budget factor is read off the ONE pair we have at
#   matched ms: ms=1e-3 gives 13.56 at 20 ep and 23.85 at 40 ep (`br6`), a
#   factor 1.759.  Therefore
#       span(5e-4, 40 ep)  ~  8.27 * 1.759  =  **14.5**
#   which lands inside the OBSERVED `node` range [12.18, 15.87].
#
#   AND THE 20-EPOCH PARTNER ALREADY EXISTS AND IS DECIDED.  `ns5-*-m5p4-*` is
#   byte-identical except the budget: ms=5e-4, 20 ep, span 8.27, argmin **`w` at
#   3.80 SE**.  This batch is a PURE BUDGET DOUBLING of a decided `w` cell into
#   the middle of the `node` band.
#
# THE TWO BRANCHES ARE OPPOSITE, WHICH IS THE POINT.
#   argmin = `node`  -> SPAN GOVERNS, BUDGET DOES NOT.  A 40-epoch run recovers
#       the nodewise minimum purely by having the right adaptation extent.  The
#       band is a band in span, CORRECTIONS 70's remaining scopes collapse into
#       one measurable scalar, and `bo7`'s AdamW result becomes a statement about
#       that optimizer's effect on SPAN rather than a fourth independent scope.
#   argmin = `w`     -> SPAN DOES NOT GOVERN AT 40 EPOCHS.  FINDINGS 55.7's band
#       acquires a budget scope of its own, and the campaign's own cycle-55
#       conclusion is narrowed by its own pre-registered test -- the cycle-48
#       pattern (CORRECTIONS 47), which is the outcome this design most needs to
#       be able to produce.
#   NO DIRECTION IS REGISTERED.  55.6's transient predicts `w` (the lift has
#   decayed by epoch 30 regardless of ms) and 55.7's band predicts `node`.  THE
#   CAMPAIGN'S OWN TWO CYCLE-55 RESULTS DISAGREE ON THIS CELL, which is exactly
#   why it is worth 9 jobs and exactly why inventing a direction now would be the
#   uc5 failure mode (CORRECTIONS 68).
#
# PRE-REGISTERED PREDICTIONS -- score against these, not against what lands.
#
#   (S0) VALIDITY.  n_records == 4000 (40 ep x 100 rec/ep); n_beta byte-matches
#        the SGDm ladder (w 11,173,962 / node 14,420 / lay 62 -- the partition is
#        a property of the NETWORK, not of ms or of the budget); beta MOVED, 9/9.
#        A mismatch VOIDS the batch.  The n_beta match is ALSO the cross-account
#        condition: `ns5` is on cluster_a and this is on cluster_b, and the comparison is
#        via N_eff/m from a shared selftested instrument on probe files.  If the
#        byte-match fails, that assumption is wrong and the batch is VOID
#        (the condition ns5, bo6 and bo7 all carried, CORRECTIONS 63/71).
#
#   (S0.3) THE BOX-FREE GATE, SCORED FIRST, PER SEED (STANDING RULE 8).  < 5% of
#        records at EITHER guard on EVERY seed, from beta_true_min/max AND
#        PATCH_CLIPCOUNT's per-coordinate counts.  A bound arm's S1/S2 are
#        UNINTERPRETABLE, and under CORRECTIONS 79 a bound rung ANYWHERE in the
#        ranked set makes the ARGMIN uninterpretable even when it is not the
#        argmin.  Report first_hi and its epoch if any arm binds.
#        THE CEILING IS NOT EXTRAPOLATED.  BOX=-30:2.0 is the ceiling `br6`
#        measured box-free 4/4 at the SAME 40-epoch budget and a STRICTLY LARGER
#        ms (1e-3 vs 5e-4).  That is a measured envelope, not a headroom
#        projection -- STANDING RULE 7's corollary forbids the latter, and this
#        batch does not use one.  The gate still runs.
#
#   (S0.4) THE TRAINS-AT-ALL GATE, SCORED SECOND.  plateau > 40.0 and `collapsed`
#        false, per seed.  bl5/br6 at 40 ep gave healthy plateaus at ms=1e-3;
#        half the meta-stepsize is not a stability risk.  A failure voids the arm.
#
#   (S0.5) **THE SPAN-REALISATION GATE, AND IT IS THE ONE THAT MAKES THIS BATCH
#        HONEST.**  The design's whole claim is that this cell lands INSIDE the
#        observed `node` band.  Measure the N2 statistic -- `c53_score.beta_span`
#        on the WEIGHTWISE arm, max-min on the LAST record, seeds averaged -- and
#        require
#              11.0  <=  span  <=  18.0
#        (the observed `node` range is [12.18, 15.87]; the bar is widened by ~1
#        log unit either side because the 14.5 estimate rests on a ONE-PAIR budget
#        factor and a slope fitted on four points).
#        **IF THE REALISED SPAN FALLS OUTSIDE [11, 18] THE DISSOCIATION WAS NOT
#        ACHIEVED AND S1 IS NOT SCORED.**  Report the span, say the cell missed
#        the band, and re-aim ms from the newly measured (ms, budget) -> span
#        surface.  Do NOT report an argmin as though it had answered the question:
#        a `w` at span 22 is consistent with span governing and says nothing.
#        This gate can only VOID the batch, never decide it.
#
#   (S1) THE PRIMARY TEST -- THE ARGMIN, scored ONLY if S0-S0.5 all pass.
#        argmin of N_eff/m over {lay, node, w}, VARIANCE instrument, STEADY half,
#        box-free arms only, priced in ITS OWN SE (`c55_neff_noise.py --argmin`),
#        NOT against the retired flat +-0.02 bar (STANDING RULE 9).
#        DECIDED at gap >= 2 SE; below that say UNDECIDED and price the extra
#        seeds with `c55_neff_noise.py --price GAP SD` rather than guessing.
#        Controls, both already measured and both DECIDED:
#            ns5 m5p4  20 ep  span  8.27  argmin **w**     3.80 SE
#            br6 c2    40 ep  span 23.85  argmin **w**    13.07 SE
#            cl5 cU    20 ep  span 13.56  argmin **node**  2.81 SE
#        -> `node` here means span governs; `w` means it does not.
#
#   (S2) THE TRANSIENT, RE-MEASURED IN ABSOLUTE EPOCH WINDOWS.
#        `c55_neff_noise.py --timecourse` on this root.  55.6's weightwise curve
#        at ms=1e-3 peaks in 10-20ep and falls 4.2x by 30-40ep.  At half the ms
#        the lift should arrive LATER.  **NO DIRECTION IS REGISTERED for where the
#        peak lands** -- there is no mechanism that predicts the timing, and the
#        curve is reported as data.  This is the diagnostic that will say WHY S1
#        went the way it did, whichever way that is.
#
#   (S3) ACCURACY, reported regardless and NOT a test.  ms and alpha0 are the
#        ns5 ladder's, untuned for a 40-epoch budget; no plateau here may be
#        quoted against any other batch's.
#
#   RESOLUTION, per CORRECTIONS 33, computed BEFORE submission.  rho_min falls
#   with T, and T DOUBLES here (4000 vs ns5's 2000), so every rung resolves at
#   least as well as it does in ns5 -- where all three read resolved=True.
#   Layerwise stays the marginal one at m=62; see S1's box-free clause.
#
# NOT INCLUDED, deliberately:
#   * A second ms.  S0.5 is the contingency if 5e-4 misses the band; guessing a
#     second ms now spends jobs on a contingency that has not happened.
#   * A ceiling ladder.  br6 measured -30:2.0 box-free at this budget and a
#     LARGER ms; a second ceiling would buy an invariance check this batch does
#     not need (bd7 already carries D0.5 for that question).
#   * A fourth seed.  n=3 against a control gap of 0.10 is ~6 SE at the measured
#     40-epoch sds (node 0.0196, w 0.0035); the UNDECIDED branch says what to do.
#   * Other families.  R18/CIFAR-10 is where every control on this axis lives.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. S0, S0.3, S0.4, S0.5 above, run not assumed, IN THAT ORDER and PER SEED
#   2. python3 analysis/c55_neff_noise.py --selftest    (51/51)
#   3. python3 analysis/c53_score.py       --selftest   (14/14)
#   4. python3 analysis/neff_instrument.py --selftest   (23/23)
#   5. python3 analysis/probe5_window.py   --selftest   (41/41)
#   6. Window scan on every arm.  4000 records: absolute windows differ from a
#      2000-record run's.  Say which you mean, every time.
#
# USAGE
#   bash c55_span_dissociation.sh            # dry run: prints every sbatch line
#   bash c55_span_dissociation.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

WS=/cluster/account_b/metaopt                          # cluster_b -- br6's workspace
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/sp8
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_b
PEND_CAP=40
NJOBS=9
BOX=-30:2.0                    # the ceiling br6 MEASURED box-free at 40 ep, ms=1e-3
EPOCHS=40
MST=5e-4                       # ns5's m5p4 rung, at DOUBLE its budget
WALL=03:50:00
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c55 SPAN/BUDGET DISSOCIATION -- does the argmin band travel with span? (cluster_b) ==="
echo "    ms=$MST  epochs=$EPOCHS  BETA_CLIP=$BOX  seeds 0-2"
echo "    partners: ns5-*-m5p4-* (20ep, span 8.27, argmin w) and br6-* (40ep, span 23.85, argmin w)"
echo "    predicted span ~14.5, inside the observed node band [12.18, 15.87]; S0.5 gates it"

for P in PATCH_PROBE5 PATCH_PROBE5_FIX PATCH_CLIPCOUNT; do
  grep -q "$P" "$HF" 2>/dev/null || { echo "GUARD FAIL: $P not present in $HF"; exit 1; }
done
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX + PATCH_CLIPCOUNT present  OK"

python3 -c "import py_compile; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 2: HF.py compiles  OK"

# GUARD 3 -- "a pure budget doubling of ns5's m5p4 rung" is a CLAIM and is DIFFED
# against ns5 itself, never asserted (the uc5 lesson, CORRECTIONS 71).  The ONLY
# fields that may differ from ns5 are the budget, the ceiling and the seed list.
NS5=$(dirname "$0")/c52_nodemin_onset.sh
if [ -r "$NS5" ]; then
  for f in "ResNet18" "CIFAR10" "SGDm" "Lion" "--alpha0 1e-3" \
           "--momentum-param-base 0.99" "--batch-size 100" "5e-4"; do
    grep -q -- "$f" "$NS5" || { echo "GUARD FAIL: '$f' absent from the ns5 control"; exit 1; }
  done
  # c52_nodemin_onset.sh writes the budget INLINE (`--num-epochs 20`, line 219) rather
  # than as an EPOCHS= variable, which br6's script happens to use.  Grepping only the
  # variable spelling made this guard a FALSE NEGATIVE (cycle 71: the ASSERTION was
  # wrong, not the code).  Accept either spelling; the INTENT -- ns5 must be a 20-epoch
  # batch, so that sp8 at 40 is a doubling -- is unchanged and is independently true:
  # all 15 ns5 rows in results/all_runs.csv have epochs_requested == epochs_done == 20.
  grep -qE -- '(EPOCHS=20|--num-epochs 20)' "$NS5" || {
    echo "GUARD FAIL: ns5 is not a 20-epoch batch -- this is not a budget doubling"; exit 1; }
  echo "guard 3: ns5 control is R18/CIFAR10/SGDm/Lion, ms=5e-4 present, 20 epochs  OK"
else
  echo "GUARD FAIL: c52_nodemin_onset.sh not readable -- cannot diff the claim"; exit 1
fi

# GUARD 3b -- the CEILING claim.  BOX must be the one br6 measured box-free at
# THIS budget, not a number chosen here.  Diffed against br6's own script.
BR6=$(dirname "$0")/c53_budget_replication.sh
if [ -r "$BR6" ]; then
  grep -q -- 'BOX=-30:2.0' "$BR6" || {
    echo "GUARD FAIL: br6's box is not -30:2.0 -- the ceiling claim is not what it says"; exit 1; }
  grep -q -- 'EPOCHS=40' "$BR6" || {
    echo "GUARD FAIL: br6 is not a 40-epoch batch -- it is not the ceiling envelope"; exit 1; }
  echo "guard 3b: br6 measured BOX=-30:2.0 box-free at 40 epochs and a LARGER ms  OK"
else
  echo "GUARD FAIL: c53_budget_replication.sh not readable -- cannot diff the ceiling"; exit 1
fi

# GUARD 4 -- the design arithmetic, re-run rather than trusted to the comment.
python3 - <<'PYEOF' || exit 1
import math
# N2-statistic spans at 20 epochs (FINDINGS 55.7), and the ONE matched-ms budget pair.
span20 = {1e-4: 1.96, 2e-4: 3.81, 5e-4: 8.27, 1e-3: 13.56}
budget_factor = 23.85 / 13.56           # br6 40ep / cl5 20ep, both ms=1e-3
slope = math.log10(span20[1e-3] / span20[1e-4])      # over one decade of ms
pred = span20[5e-4] * budget_factor
band = (12.18, 15.87)                   # the OBSERVED node-argmin span range
gate = (11.0, 18.0)                     # S0.5
assert 0.7 < slope < 1.0, f"the ms->span slope moved: {slope:.2f}"
assert band[0] <= pred <= band[1], f"predicted span {pred:.1f} is not inside the node band"
assert gate[0] < band[0] and gate[1] > band[1], "the S0.5 gate does not contain the band"
print(f"guard 4: span ~ ms^{slope:.2f}; budget factor {budget_factor:.3f}; "
      f"predicted span {pred:.1f} inside the node band [{band[0]}, {band[1]}]; "
      f"S0.5 gate [{gate[0]}, {gate[1]}]  OK")
PYEOF

# GUARD 5 -- queue depth.  9 jobs is well inside the ~20-job / 0.5%-RawUsage rule.
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

# GUARD 6 -- THE WALL-CLOCK REQUEST MUST FIT THE TIGHTEST PARTITION IN $PARTS.
# CORRECTIONS 71's failure mode, silent-until-total: sbatch rejects against the
# SMALLEST MaxTime among the requested partitions, so one over-long --time loses
# the WHOLE batch.  Read from `sinfo` LIVE, never from a comment (FINDINGS 54.6).
python3 - "$WALL" "$PARTS" <<'PYEOF' || exit 1
import subprocess, sys
def secs(t):
    d, _, rest = t.partition("-")
    if not rest: rest, d = d, "0"
    p = [int(x) for x in rest.split(":")]
    while len(p) < 3: p.insert(0, 0)
    return int(d) * 86400 + p[0] * 3600 + p[1] * 60 + p[2]
want, parts = sys.argv[1], sys.argv[2].split(",")
out = subprocess.run(["sinfo", "-h", "-o", "%P %l"], capture_output=True, text=True).stdout
lim = {}
for ln in out.split("\n"):
    f = ln.split()
    if len(f) == 2 and f[1] not in ("infinite", "n/a"):
        lim[f[0].rstrip("*")] = secs(f[1])
tight = [(p, lim[p]) for p in parts if p in lim]
if not tight:
    sys.exit("GUARD FAIL: sinfo returned no limit for any requested partition")
name, cap = min(tight, key=lambda kv: kv[1])
if secs(want) > cap:
    sys.exit(f"GUARD FAIL: --time={want} exceeds {name}'s MaxTime "
             f"({cap // 3600}:{cap % 3600 // 60:02d}) -- sbatch would reject the WHOLE batch")
print(f"guard 6: --time={want} fits the tightest partition {name} "
      f"({cap // 3600}:{cap % 3600 // 60:02d})  OK")
PYEOF

mkdir -p "$SAVE"
N=0
FAIL=0
for G in layerwise nodewise weightwise; do
  case $G in
    layerwise)  TAG=lay  ;;
    nodewise)   TAG=node ;;
    weightwise) TAG=w    ;;
  esac
  for S in 0 1 2; do
    RN="sp8-${TAG}-m5p4e40-s${S}"
    PD="$SAVE/probe_${TAG}_m5p4e40_s${S}"
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time="$WALL"
      --export="ALL,AUGMENT=1,BETA_CLIP=$BOX,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
      "$RUNNER"
      --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
      --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
      --max-time 999:00:00 --gamma 1 --meta-stepsize "$MST" --alpha0 1e-3
      --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    # COUNT WHAT SLURM ACCEPTED, NOT WHAT WE TRIED (CORRECTIONS 71).
    if [ "$SUBMIT" = "1" ]; then
      if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
    else
      N=$((N+1)); printf '%s\n' "${CMD[*]}"
    fi
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
echo "reduce with, IN THIS ORDER:"
echo "  1. S0    n_beta byte-match vs the SGDm ladder + n_records==4000 + beta moved, 9/9"
echo "  2. S0.3  PER SEED box-free -- analysis/c52_boxfree.py on ../probes_sp8"
echo "  3. S0.4  plateau > 40 per seed, from results/all_runs.csv"
echo "  4. S0.5  THE SPAN GATE -- c53_score.beta_span on the WEIGHTWISE arm; 11 <= span <= 18"
echo "           OUTSIDE THAT RANGE, STOP.  Report the span, re-aim ms, do NOT score S1."
echo "  5. S1    python3 analysis/c55_neff_noise.py ../probes_sp8 --argmin   # THE PRIMARY TEST"
echo "  6. S2    python3 analysis/c55_neff_noise.py ../probes_sp8 --timecourse"
