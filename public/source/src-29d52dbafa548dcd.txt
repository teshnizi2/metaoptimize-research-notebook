"""cwd4 -- mean |L| RE-DERIVED IN BATCH from cwd4's OWN k01 pinned probe records.

NOT A REGISTERED SCORER.  It reads no bar, level, contrast, state, branch or stamp,
nothing in any verdict path imports it, and it writes nothing but its own report.
Added by the CORRECTIONS 283 landing to discharge 280.10's obligation that the |L|
figures a landing quotes be RE-DERIVED rather than inherited from SCORE-cdep1's [P]
table.  Its output is DESCRIPTIVE: it moves no bar, level, contrast, state, branch
word or stamp of cwd4, and it does NOT separate the magnitude rival -- it measures it.

CORRECTIONS 280.10 / the cwd4 landing's bound (f): the |L| ordering must be
re-derived, not inherited from SCORE-cdep1's [P] table.  This does it on cwd4's
own scalar anchor, at cwd4's own cell and seeds {143,144,145}.

Stdlib only.  The DEFINITION is re-typed from cDP1_cdep1_depth_score.py's
premise_from_scalar_records (read, not imported):
  L_i = b2*m_i + (1-b2)*z_i     b2 = record's own pt_b2
  determinate: s = sgn(b2*z_agg[0][0]... ) != 0 and sgn(sum L) == s and sgn(#pos-#neg) != 0
  pinned: n_at_lo == 1
  mean |L_i| = sum over pinned records of |L_i| / (number of pinned records)
"""
import json, os, collections
BASE = os.environ.get("CWD4_PROBE_ROOT", "../runs_cluster_b/cwd4")
SEEDS = ("s143", "s144", "s145")
B2_DEFAULT = 0.99
NAMES = {}
for ln in open(os.path.join(BASE, "PARTITION-MANIFEST.txt")):
    p = ln.split()
    if p and p[0] == "TENSOR":
        NAMES[int(p[1])] = p[2]

def sgn(x):
    return (x > 0) - (x < 0)

absL = collections.defaultdict(float); absN = 0; n_det = 0; n_pin = 0
for s in SEEDS:
    for ln in open(os.path.join(BASE, "probe_cwd4-k01-%s" % s, "probe.jsonl")):
        r = json.loads(ln)
        try:
            z, m = r["z_tensor"], r["m_tensor"]
            zh, mh = float(r["z_agg"][0][0]), float(r["mom_pre"][0][0])
        except (KeyError, IndexError, TypeError):
            continue
        b2 = float(r.get("pt_b2", B2_DEFAULT))
        L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
        sg = sgn(b2 * mh + (1 - b2) * zh)
        if sg == 0 or sgn(sum(L)) != sg:
            continue
        pos = sum(1 for x in L if x > 0); neg = sum(1 for x in L if x < 0)
        if sgn(pos - neg) == 0:
            continue
        n_det += 1
        if r.get("n_at_lo") == 1:
            n_pin += 1; absN += 1
            for i, x in enumerate(L):
                absL[i] += abs(x)

mean = {i + 1: absL[i] / absN for i in range(len(NAMES))}
order = sorted(mean.items(), key=lambda kv: -kv[1])
rank = {i: r for r, (i, _) in enumerate(order, 1)}
print("cwd4 k01, seeds %s: determinate %d, PINNED %d records" % (",".join(SEEDS), n_det, n_pin))
print("mean |L| on pinned records, IN BATCH (cwd4's own k01):")
for i in (50, 53, 59, 47, 56, 48):
    print("  idx %-3d %-32s %.4e   rank %2d" % (i, NAMES[i], mean[i], rank[i]))
print()
print("  carriers ranked:", [i for i, _ in order[:5]])
print("  SUM carriers {50,53,59} %.4e ; SUM {47,56} %.4e ; ratio %.1f"
      % (sum(mean[i] for i in (50, 53, 59)), sum(mean[i] for i in (47, 56)),
         sum(mean[i] for i in (50, 53, 59)) / sum(mean[i] for i in (47, 56))))
print("  smallest carrier / largest admissible control = %.4e / %.4e = %.1fx"
      % (min(mean[i] for i in (50, 53, 59)), max(mean[i] for i in (47, 56)),
         min(mean[i] for i in (50, 53, 59)) / max(mean[i] for i in (47, 56))))
print()
print("ORDER AMONG THE THREE SINGLES by in-batch mean |L|: %s"
      % " > ".join("ONE%d" % i for i in sorted((50, 53, 59), key=lambda i: -mean[i])))
print("cwd4's plateau5 order (scorer's ONE-ORDER stamp): ONE59 > ONE50 > ONE53")
print("INHERITED cdep1 [P] figures for comparison: 59 3.2984e-01, 50 2.9155e-01, 53 1.9526e-01,"
      " 56 1.5166e-03, 47 1.1298e-03")

# ---------------------------------------------------------------------------
# [2] THE REMOVED-DECAY DOSE AT RECORD 0, re-derived in batch.
# At probe record 0 every BatchNorm scale is still at its initialisation of 1.0
# (dm_absmin == 1.0 on every masked arm), so the removed weight-decay term
# dm_wdterm is a pure function of k.  This closes a "dose of removed decay"
# reading of 278's T19 with an IN-BATCH number instead of an inherited table.
# It does NOT bear on the magnitude rival, which concerns the meta-trace term L
# and not the weight-decay term, and must not be written as if it did.
ARMS = ("k01", "CARWD0", "TWOWD0", "CTL2WD0", "ONE50", "ONE53", "ONE59")
print()
print("[2] removed-decay dose at PROBE RECORD 0 (every BN scale still 1.0):")
by_k = collections.defaultdict(set)
for s in SEEDS:
    for a in ARMS:
        with open(os.path.join(BASE, "probe_cwd4-%s-%s" % (a, s), "probe.jsonl")) as fh:
            r0 = json.loads(fh.readline())
        k, w, am = r0.get("dm_masked"), r0.get("dm_wdterm"), r0.get("dm_absmin")
        print("  %-4s %-8s k=%-5s dm_wdterm=%-24r dm_absmin=%r" % (s, a, k, w, am))
        if k is not None:
            by_k[k].add(repr(w))
print()
for k in sorted(by_k):
    assert len(by_k[k]) == 1, (k, by_k[k])
    print("  k=%d -> ONE distinct dm_wdterm across all arms and all three seeds: %s"
          % (k, sorted(by_k[k])[0]))
print("  TWOWD0 and CTL2WD0 are both k=2 and therefore remove NUMERICALLY IDENTICAL")
print("  weight decay at initialisation, and still differ by P_2SPEC = +44.2167 pp.")
