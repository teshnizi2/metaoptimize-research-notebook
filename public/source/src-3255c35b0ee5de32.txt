# -----------------------------------------------------------------------------

# defaults are then pinned by build_optimizer.py's sha256 (guard 3c).
#
# -----------------------------------------------------------------------------
# WALLTIME AND PARTITIONS -- MEASURED, NOT GUESSED (results/all_runs.csv,

#   SGD+cosine   bl-sgd (20) + cdn1-lr (18): median 29 min, worst 67 ([HOST_3c1b003035], 2080ti)

#   corpus-worst 70.8 s/epoch on the 2080ti nodes -> 100 epochs = 1.97 h.
# WALL=03:00:00 is 1.52x the corpus-worst and 1.94x the worst observed run,

#       21 x 67 + 6 x 93 = 32.8 GPU-h; corpus-worst 27 x 1.97 = 53.1 GPU-h;
#       HARD BOUND = WALL x 27 = 81 GPU-h.
#
# USAGE   bash bin/cAU1_unaug_denominator.sh              # DRY RUN, submits NOTHING
#         bash bin/cAU1_unaug_denominator.sh --dry-run    # identical, explicit
#         bash bin/cAU1_unaug_denominator.sh --envaudit   # AFTER launch: ENV audit
#         bash bin/cAU1_unaug_denominator.sh --submit     # the ONLY submitting form

import sys, importlib.util

PYEOF

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ----------

PYEOF

                         || guard_fail "guard 3c: build_optimizer.py sha256 $GOT, pinned $BOPT_SHA"

  && grep -q "SGD_MOM', 0.9" "$BO" && grep -q "COS_TOTAL" "$BO" && grep -q "COS_WARMUP" "$BO" \

[ -d "$CIF/data/cifar-10-batches-py" ] && echo "guard 3e: cifar-10-batches-py staged under $CIF/data" \

  echo "BUILD_OPTIMIZER_SHA256 $BOPT_SHA"; echo "LOAD_DATA_SHA256 $LOAD_SHA";
  echo "SEEDS $SEEDS"; echo "LADDER $LRS"; echo "EPOCHS $EPOCHS"; echo "AUGMENT $AUG";
  echo "M6 $M6_A0 $M6_CLIP"; echo "M6T $M6T_A0 $M6T_CLIP"; echo "WALL $WALL"; echo "PARTS $PARTS";

# ---- compose, guard each line under RULE 20, prove it against the scorer -------
LINES=/tmp/cau1_lines.$$; : > "$LINES"
N=0; REJ=0; BAD=0
compose_sgd() {  # $1 lr $2 tag $3 seed
  local RN="cau1-lr$2-s$3"
  CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=14G --time=$WALL \
--export=ALL,AUGMENT=$AUG,COS_TOTAL=$COS_T,COS_WARMUP=$COS_W,HIER=none,SCHED=none,PROBE=0 \
$RUNNER --optimizer SGD --dataset $DSET --NN-name $NET --batch-size $BATCH \
--max-time 999:00:00 --num-epochs $EPOCHS --alpha0 $1 --seed $3 \
--save-directory $SAVE --run-name $RN"
  DESIGN=(optimizer=SGD dataset=$DSET NN-name=$NET batch-size=$BATCH num-epochs=$EPOCHS \
          alpha0=$1 seed=$3 run-name=$RN)
}
compose_meta() {  # $1 arm $2 alpha0 $3 clip $4 seed
  local RN="cau1-$1-s$4"
  PDIR="$SAVE/probe_$RN"
  CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=14G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$3,HIER=none,SCHED=none,PROBE=$PROBE_M,PROBE_DIR=$PDIR \
$RUNNER --optimizer HF --alg-base AdamW --normalizer-param-base 0.999 --momentum-param-base 0.9 \
--weight-decay-base 0.1 --alg-meta Adam --normalizer-param-meta 0.999 --momentum-param-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET \
--batch-size $BATCH --max-time 999:00:00 --gamma $GAMMA --meta-stepsize $MST --alpha0 $2 \
--num-epochs $EPOCHS --stepsize-groups resnet18_blocks --seed $4 --save-directory $SAVE --run-name $RN"
  DESIGN=(optimizer=HF alg-base=AdamW alg-meta=Adam meta-stepsize=$MST gamma=$GAMMA \
          stepsize-groups=resnet18_blocks weight-decay-meta=0 dataset=$DSET NN-name=$NET \
          batch-size=$BATCH num-epochs=$EPOCHS alpha0=$2 seed=$4 run-name=$RN)
}
emit() {
  if ! guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1; then
    echo "!!! RULE 20 PRE-CHECK FAILED:"; guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1))
  fi
  printf '%s\n' "$CMD" >> "$LINES"
