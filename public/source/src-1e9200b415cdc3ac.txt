#!/bin/bash
# =============================================================================
# cP1_second_cliff_split.sh -- `scl1`, THE SINGLE-TENSOR RESOLUTION OF THE
#                              **SECOND** CLIFF, k = 52 -> k = 55, ONE BATCH.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.
# Nothing here touches the other account's tree and nothing is submitted to
# `cluster_a`.  Every guard and every scancel line below is filtered on the
# `scl1-` prefix alone, so no job this script did not submit can be affected.
#
# REGISTERED SCORER: analysis/cP1_scl1_score.py, git-committed BEFORE this
# batch is submitted (STANDING RULE 21).  Every bar quoted here is frozen
# there; that file's --selftest re-derives the noise floor, cpk1's own second
# cliff and cts1's own GAP from results/all_runs.csv, and asserts each
# constant below against THIS FILE'S own text.
#
# NO CODE PATCH.  Optimizers/HF.py::polish_the_stepsize_groups already parses
# an explicit int list given as the STRING "[a,b]" into CONTIGUOUS groups in
# named_parameters() order.  GUARD 4 below re-proves that on the LIVE model at
# submission time and writes PARTITION-MANIFEST.txt (sizes, per-group PARAMS
# and the full FIRSTGROUP / LASTGROUP tensor NAME lists), which the scorer's
# G1 and G1b both REQUIRE.  The shared tree is not modified by this batch.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
# cpk1 (39 jobs, CORRECTIONS 146) swept the cut position k at m = 2 and found a
# single-peaked capture curve with argmax k* = 49.  cts1 (18 jobs, CORRECTIONS
# 147) cut the FIRST collapse k = 49 -> 52 at single-tensor resolution and
# found ONE 512-parameter tensor carrying it.
#
# cpk1's curve has a SECOND collapse that appears in no doc, no scorer and no
# launcher, flagged only in passing at CORRECTIONS 147.5e.  Re-derived here
# from cpk1's OWN plateau5 means in results/all_runs.csv:
#
#     k = 49   55.4473 pp        k = 52 -> k = 55   -14.9700 pp
#     k = 52   38.3420 pp        k = 49 -> k = 52   -17.1053 pp
#     k = 55   23.3720 pp        the second cliff is 87.52 % of the first
#
# (Over cpk1's own in-batch GAP of 46.9900 pp those are -0.318578 and
# -0.364021 CAPTURE, which is where CORRECTIONS 147.5e's -0.3185 / -0.3641
# came from.  Nothing was inherited: both were recomputed from the corpus.)
#
# This batch is the direct test of whether "one tensor carries the step" is a
# PATTERN or a ONE-SITE ACCIDENT -- exactly the class-level claim CORRECTIONS
# 147.6 had to WITHDRAW.
#
# -----------------------------------------------------------------------------
# THE LIVE-MODEL MANIFEST.  MEASURED, NOT INHERITED.
# -----------------------------------------------------------------------------
# The cts1 briefing was off by one on the index convention, so no index here is
# taken on trust.  Measured on ResNet18_c100 (62 tensors, 11,220,132 params):
#
#     idx  tensor                          params
#      52  layer4.0.shortcut.0.weight      131072   (already COARSE at k = 52)
#      53  layer4.0.shortcut.1.weight         512   moves at k=52 -> k=53
#      54  layer4.0.shortcut.1.bias           512   moves at k=53 -> k=54
#      55  layer4.1.conv1.weight          2359296   moves at k=54 -> k=55
#
# CONFIRMED: this region does contain another 512-parameter normalisation
# SCALE beside a 2,359,296-parameter conv, exactly as CORRECTIONS 147.5e said.
# CONVENTION, verified: with 1-based k, the tensor at 1-based index k+1 is the
# one that moves at the step k -> k+1.  Guard 4d re-proves all of it at submit
# time from the live model, and fails if any step moves more or fewer than one.
#
# The three steps move 2,360,320 parameters = 49.4519 % of the fine group's
# 4,772,964 at k = 52.  DISCLOSED ASYMMETRY WITH cts1, whose three steps moved
# 2.6931 %.  The two cliffs are homologous in tensor CLASS (scale, shift,
# conv) but NOT in moved MASS.
#
# -----------------------------------------------------------------------------
# THE DESIGN.  4 arms x 3 seeds = 12 jobs, ONE submission.  NO ANCHORS.
# -----------------------------------------------------------------------------
#   THE FOUR CUTS, [k, 62-k]:  k = 52  53  54  55
#
#   BOTH CLIFF EDGES (k = 52 and k = 55) ARE RE-RUN HERE.  cpk1's and cts1's
#   arms are NOT spliced in: batch, not seed, is the unit of replication
#   (F(62,85) = 5.47, p 6.9e-13 for batch; F(11,1976) = 0.152 for seed), so the
#   whole cliff sits inside ONE submission.
#
#   WHY NO scalar / layerwise ANCHORS -- i.e. WHY THE PRIMARY IS pp, NOT
#   CAPTURE.  The primary is (a) which of three single-tensor steps carries the
#   majority and (b) a matched 512-vs-512 control.  Both are differences and
#   ratios of pp differences measured WITHIN this batch, and dividing all four
#   arms by one common GAP is a positive rescaling that cannot change the
#   ordering of the drops, the majority test, the separation test, the even
#   test or the matched control.  CORRECTIONS 147.6 had to qualify cts1's
#   CAPTURE SHAREs precisely because their denominator is a non-monotone path
#   sum; cts2 registered its own primary in pp for the same reason.  The one
#   thing anchors would buy -- magnitude comparability with cpk1's CAPTURE
#   curve -- is bought instead by the R1 bar, which is derived in pp from
#   cpk1's own plateau5 means.  Six fewer jobs, ~4.2 GPU-hours.
#   THE PRICE, DISCLOSED: this batch cannot state the second cliff's size in
#   CAPTURE units from its own data, and a batch-level LEVEL SHIFT relative to
#   cpk1 is neither detectable nor removable here.  R1 guards that, is checked
#   FIRST, and its bar is a generous 50 %.
#
#   CIFAR-100 / ResNet18_c100 / SGDm base / Lion meta / ms 1e-3 / alpha0 1e-6 /
#   100 ep / batch 100 / gamma 1 / BETA_CLIP -15:-2.3026 / AUGMENT=1 / HIER
#   unset / PROBE=0 / seeds 0,1,2 / walltime 02:30:00.  Matched cell-for-cell,
#   READ OUT OF bin/cL1_cliff_tensor_split.sh, not reconstructed.
#
#   DROP1 = M(k52) - M(k53), DROP2 = M(k53) - M(k54), DROP3 = M(k54) - M(k55),
#   TOTAL = M(k52) - M(k55).  plateau5 PRIMARY; train5 reported at every arm.
#   A POSITIVE drop means accuracy is LOST as that tensor leaves the FINE group.
#
# -----------------------------------------------------------------------------
# THE TWO ACCOUNTS ON RECORD BEFORE THE DATA, frozen in the scorer
# -----------------------------------------------------------------------------
#   B1 STRUCTURAL HOMOLOGY -- THIS AGENT'S POINT PREDICTION.  The majority is
#      DROP1, `layer4.0.shortcut.1.weight`: a 512-parameter normalisation SCALE
#      whose own convolution is ALREADY coarse, leaving the fine group --
#      structurally the same step as cts1's winner `layer4.0.bn2.weight`.
#        1. cts1's DROP3 is the corpus's only clean single-tensor datum on a
#           CONV moving fine -> coarse and it GAINED +7.707 pp.  A DROP3
#           majority here must reverse that sign at a conv 18x larger.
#        2. Parameter mass cannot be the currency: HF.py:189 forms an
#           UNNORMALISED group sum and HF.py:881 applies torch.sign(), so
#           |dbeta| == meta_stepsize exactly and parameter count enters the
#           meta-update nowhere (CORRECTIONS 147.5b).  The 2.36M conv has no
#           mass-based route to dominance and the 512-param scale is not
#           handicapped.
#        3. DROP2 moves the matched BN SHIFT and is predicted ~ 0, as cts1's
#           own DROP2 was (-0.233 pp, t -0.19).
#   B2 BULK CONV -- THE RIVAL TO BEAT.  The majority is DROP3,
#      `layer4.1.conv1.weight` (2,359,296 params, 49.4 % of the fine group at
#      k = 52, and the first tensor of a whole new residual block).  Under B2
#      the second cliff is a BLOCK-BOUNDARY / BULK effect with nothing to do
#      with cts1's mechanism.
#   NO THIRD ACCOUNT MAY BE INTRODUCED AFTER THE DATA.
#
#   B1 IS PREDICTING AGAINST THE CORPUS BASE RATE AND SAYS SO IN ADVANCE.  The
#   class-level form of B1 was WITHDRAWN at CORRECTIONS 147.6 on cpk1's own
#   data: the "scale first in the fine group while its conv is already coarse"
#   condition holds at cpk1's k = 31/49/52/55 with CAPTUREs
#   +0.2190/+0.6965/+0.3324/+0.0139 -- nearly the whole range of the sweep --
#   and 8 of cpk1's 10 norm-scale-out steps RAISE capture.  The base rate for
#   "moving a scale out is costly" is 2 in 10.  That is the point of the batch:
#   this is the site where the withdrawn claim gets a fair, matched test.
#
# -----------------------------------------------------------------------------
# PRE-REGISTERED PREDICTIONS, frozen in the scorer
# -----------------------------------------------------------------------------
#   Q1 (B1) the largest drop is DROP1 (k=52->53, layer4.0.shortcut.1.weight).
#   Q2 (B2) the largest drop is DROP3 (k=54->55, layer4.1.conv1.weight).
#   Q3 the cliff REPRODUCES: TOTAL >= 7.4850 pp.
#   Q4 the split is NOT even.
#   Q5 MATCHED CONTROL |DROP1 - DROP2| > 2.3476 pp.  The two tensors hold
#      EXACTLY 512 parameters each, are adjacent in named_parameters(), belong
#      to the SAME BatchNorm2d and move at consecutive steps.  DISCLOSED
#      (CORRECTIONS 147.5d): the control is NESTED, not matched.
#   Q6 TRAIN and TEST agree on which step is largest.
#   Q7 DROP3 <= 0 -- the direct cross-cliff replication of cts1's only
#      conv-move datum (its DROP3 = -7.707 pp).
#
# -----------------------------------------------------------------------------
# PRE-REGISTERED REFUTATION BRANCHES, frozen, CHECKED IN THIS ORDER
# -----------------------------------------------------------------------------
#   R1 CLIFF-NOT-REPRODUCED   TOTAL < 7.4850 pp, i.e. less than HALF of cpk1's
#                             own 14.9700 pp.  THE refutation that VOIDS the
#                             question, CHECKED FIRST.  If the cliff is not
#                             here there is nothing to decompose, NO TENSOR MAY
#                             BE NAMED, the deliverable is UNRESOLVED, and
#                             cpk1's or cts1's arms may NOT be spliced in to
#                             rescue it.  A negative TOTAL lands here too.
#                             7.4850 pp = 10.06 SE of a 3-seed arm difference
#                             at the re-derived noise floor.
#   R2 SPLIT-EVEN             all three drops within 2.3476 pp of TOTAL/3.
#                             Then it is NOT a single-tensor effect, no tensor
#                             may be named, B1 and B2 both fail, and cts1's
#                             result is a LOCAL ACCIDENT -- CORRECTIONS 147.6's
#                             withdrawal was RIGHT.  2.3476 pp = 3.58 SE on
#                             DROP1/DROP3, 2.99 SE on DROP2.
#   R3 NO-MAJORITY            the leader does not exceed TOTAL/2, or does not
#                             beat the runner-up by 2.3476 pp.  Naming requires
#                             the CONJUNCTION of majority AND separation.
#                             2.3476 pp = 1.82 SE adjacent, 2.23 SE end-to-end.
#
#   THESE ARE cts1'S OWN BARS.  2.3476 pp is cts1's registered 0.05 CAPTURE
#   expressed in pp at cts1's OWN measured in-batch GAP of 46.9513 pp.  A
#   PATTERN-vs-ACCIDENT comparison in which the two cliffs were judged against
#   different yardsticks would be worthless.  They are weaker in SE units here
#   (3.58 / 2.99 / 1.82 against cts1's 4.5 / 3.7 / 2.3) because this scorer
#   takes the CONSERVATIVE, larger noise floor -- the m = 2-only pool,
#   sigma_w 0.9111 over df 50 -- rather than cts1's all-granularity pool,
#   sigma_w 0.7245.  Every arm here is an m = 2 cut and there are no anchors.
#
# -----------------------------------------------------------------------------
# WHAT WOULD SHOW THE TWO CLIFFS SHARE A MECHANISM, AND WHAT WOULD NOT
# -----------------------------------------------------------------------------
#   SHARED-MECHANISM  <=>  the cliff reproduces AND the split is a single-
#                          tensor majority AND that tensor is DROP1's, the
#                          normalisation SCALE.  Two cliffs, two sites, two
#                          submissions, same tensor class in the same
#                          structural relation: "one tensor carries the step"
#                          is a PATTERN and CORRECTIONS 147.6's withdrawal was
#                          TOO CAUTIOUS for this sub-class.
#   NOT-SHARED        <=>  the cliff reproduces AND (R2 fires, or R3 fires, or
#                          the majority is DROP2 or DROP3).  cts1's result is
#                          a ONE-SITE accident and the withdrawal STANDS.
#   UNRESOLVED        <=>  R1 fires.  No tensor named, no mechanism claim.
#
# MAY NOT CLAIM: anything about the group COUNT m (every arm is m = 2 and there
# are no anchors, so CAPTURE is not defined in this batch at all), about
# resnet18_blocks, CIFAR-10, Tiny-ImageNet, ImageNet-489, ResNet-50, alpha0
# 1e-3, any meta-stepsize but 1e-3, any optimiser pair but SGDm+Lion, or any
# hierarchical/shrinkage operator.  A tensor named here is named at ONE SITE in
# ONE architecture in ONE named_parameters() order.  Normalised entropy stays
# DISQUALIFIED (cbl1 R5).  BETA_CLIP binds in this cell, so nothing here
# separates stepsize allocation from clamp turnover at THIS cut position --
# cts2 did that at k = 49/50 only.  plateau5 is a 100-epoch snapshot and
# CORRECTIONS 147.5f showed these arms still rising at epoch 99, so the
# ASYMPTOTIC existence of either cliff remains untested.
#
# WALLTIME, MEASURED not guessed.  cpk1 ran this exact cell on this account and
# measured 42.0 min mean per run over 39 jobs, 0 TIMEOUT and 0 FAILED; cts1 ran
# 18/18 clean at the same request.  02:30:00 is 3.6x that mean and stays inside
# gpu-short's 4 h ceiling, so no partition in the list is excluded.
# Total cost of this batch: 12 jobs x ~42 min ~= 8.4 GPU-hours.
#
# USAGE   bash bin/cP1_second_cliff_split.sh            # dry run, submits nothing
#         bash bin/cP1_second_cliff_split.sh --submit
# =============================================================================
set -u
SUBMIT=0; [ "${1:-}" = "--submit" ] && SUBMIT=1
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

USER_NAME="${USER:-$(whoami)}"
# NOTE: bin/_lib_guards.sh re-assigns WS at source time from METAOPT_WS with a
# default that points at the OTHER account's tree, which would clobber this
# value and point guard_postlaunch at the wrong directory (it did exactly that
# to tn1 and to hb1).  Exporting METAOPT_WS here is the documented work-around;
# the library is NOT patched while jobs are queued.
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/scl1
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP="-15:-2.3026"; MST=1e-3; ALPHA0=1e-6; EPOCHS=100; BATCH=100
GAMMA=1; AUG=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="0 1 2"
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
WALL=02:30:00
NTENS=62

# ---- the registered grid and the registered specs, written out LITERALLY ----
# Both forms are here on purpose: K_LIST is the grid the scorer cross-checks,
# and the G_K* strings are the exact argv the jobs receive.  Guard 4c below
# proves they agree, so neither can drift from the other or from the scorer.
K_LIST="52 53 54 55"
G_K52='[52,10]'
G_K53='[53,9]'
G_K54='[54,8]'
G_K55='[55,7]'
ARMS="k52 k53 k54 k55"

echo "=============================================================="
echo " scl1 -- CIFAR-100 SINGLE-TENSOR resolution of the SECOND cliff"
echo " the k = 52 -> k = 55 collapse (87.52 % of the first cliff)"
echo " $DSET / $NET  box $CLIP  ms $MST  alpha0 $ALPHA0  AUGMENT=$AUG"
echo " cut positions k = $K_LIST   NO anchors: the primary is in pp"
echo " 4 arms x 3 seeds = 12 jobs, one submission"
echo " scorer: analysis/cP1_scl1_score.py (registered before any run exists)"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
CSVP="$REPO/results/all_runs.csv"
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- guard 1: the registered scorer exists, is committed, and selftests ------
SC="$REPO/analysis/cP1_scl1_score.py"
[ -f "$SC" ] || guard_fail "guard 1a: scorer missing"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  DIRTY=$(git -C "$REPO" status --porcelain -- analysis/cP1_scl1_score.py bin/cP1_second_cliff_split.sh 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: scorer AND this script are committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires the scorer committed BEFORE any run exists: $DIRTY"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 was verified"
  echo "          on the host that committed the scorer.  Not a pass, not a fail."
fi
python3 "$SC" --selftest >/tmp/scl1_selftest.$$ 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES" \
  || { guard_fail "guard 1c: scorer selftest FAILED"; tail -25 /tmp/scl1_selftest.$$; }

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ---------
grep -q '^scl1-' "$CSVP" && guard_fail "guard 2: scl1-* already in the CSV" \
                         || echo "guard 2: no scl1-* name collision in the corpus"
if [ -d "$WS/runs" ]; then
  NOUT=$(ls "$WS/runs"/scl1-*.out 2>/dev/null | wc -l | tr -d ' ')
  [ "${NOUT:-0}" = 0 ] && echo "guard 2b: no scl1-*.out on disk" \
                       || guard_fail "guard 2b: $NOUT scl1-*.out files already exist"
fi

# ---- guard 3: the CIFAR-100 data is actually staged on THIS account ---------
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3: cifar-100-python present under $CIF/data" \
  || guard_fail "guard 3: CIFAR-100 is not staged at $CIF/data/cifar-100-python"

# ---- guard 4: EVERY PARTITION PROVEN ON THE LIVE MODEL, AND WRITTEN DOWN ----
# No assumption about tensor ordering, tensor count or parameter count:
# named_parameters() is read from the model this batch will actually train,
# polish_the_stepsize_groups is called on it, the GROUP MEMBERSHIP of every arm
# is printed in full, and the manifest records sizes AND per-group parameter
# counts AND the full tensor NAME lists so the scorer's G1 and G1b can check
# all three.  Guard 4d additionally prints, from the live model, WHICH TENSOR
# moves at each of the three registered single-tensor steps -- the whole point
# of the batch -- and fails if any step moves more or fewer than one tensor.
mkdir -p "$SAVE"
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"

# guard 4c: the literal spec strings and the grid must agree, BEFORE the model
# is ever built.  A typo in one of the four G_K* strings dies here.
SPECS=""
for K in $K_LIST; do
  eval "LIT=\${G_K$K:-}"
  WANT="[$K,$((NTENS-K))]"
  if [ "$LIT" != "$WANT" ]; then
    guard_fail "guard 4c: G_K$K is '$LIT', but the grid says '$WANT'"
  fi
  SPECS="$SPECS $LIT"
done
NK=$(echo $K_LIST | wc -w | tr -d ' ')
NA=$(echo $ARMS | wc -w | tr -d ' ')
[ "$NA" = "$NK" ] || guard_fail "guard 4c: $NA arms for $NK cut positions and no anchors"
# the four cuts must be CONSECUTIVE, or the steps are not single-tensor
PREV=""
for K in $K_LIST; do
  if [ -n "$PREV" ] && [ "$K" != "$((PREV+1))" ]; then
    guard_fail "guard 4c: the grid is not consecutive ($PREV -> $K); the steps would not be single-tensor"
  fi
  PREV=$K
done
echo "guard 4c: $NK literal specs agree with the grid, and the grid is consecutive; $NA arms in total"
# shellcheck disable=SC2086
python3 - "$CIF" "$MANIFEST" "$NTENS" $SPECS <<'PYEOF' \
  || guard_fail "guard 4: live partition check FAILED"
import sys, os
cif, out, ntens = sys.argv[1], sys.argv[2], int(sys.argv[3])
specs = sys.argv[4:]
sys.path.insert(0, cif)
os.chdir(cif)
from build_network import build_network
from Optimizers.HF import HF
net = build_network("ResNet18_c100", "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in net.named_parameters()]
numel = [int(p.numel()) for _, p in net.named_parameters()]
tot = sum(numel)
T = len(nps)
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot]
print("guard 4: live ResNet18_c100 has %d parameter tensors, %d parameters"
      % (T, tot))
if T != ntens:
    print("!!! guard 4: the live model has %d tensors, the batch assumes %d"
          % (T, ntens))
    raise SystemExit(1)

groups, cur = [], None
for i, (n, _) in enumerate(nps, 1):
    key = n.split(".")[0]
    if key.startswith("layer"):
        key = ".".join(n.split(".")[:2])
    if key != cur:
        groups.append([key, i, i])
        cur = key
    else:
        groups[-1][2] = i
print("guard 4: named_parameters() structure, measured on the live model:")
for name, lo, hi in groups:
    print("         %-12s tensors %2d..%-2d  params %9d"
          % (name, lo, hi, sum(numel[lo - 1:hi])))
print("guard 4: the SECOND cliff's region, tensor by tensor (1-based index):")
for i in range(50, 56):
    print("         %2d  %-32s %9d" % (i + 1, names[i], numel[i]))

bad = 0
seen = {}
for spec in specs:
    g = HF.polish_the_stepsize_groups(None, spec, nps)
    sizes = [len(x) for x in g]
    if sum(sizes) != T:
        print("!!! guard 4: %s does not cover the model (%d of %d)"
              % (spec, sum(sizes), T))
        bad += 1
        continue
    if len(g) != 2:
        print("!!! guard 4: %s composed to m=%d, but every cut arm must be m=2"
              % (spec, len(g)))
        bad += 1
        continue
    want = [int(x) for x in spec[1:-1].split(",")]
    if sizes != want:
        print("!!! guard 4: %s composed to sizes %s on the live model" % (spec, sizes))
        bad += 1
        continue
    k = sizes[0]
    if list(g[0]) != names[:k] or list(g[-1]) != names[k:]:
        print("!!! guard 4: %s does not split named_parameters() at %d BY NAME"
              % (spec, k))
        bad += 1
        continue
    pc, i = [], 0
    for s in sizes:
        pc.append(sum(numel[i:i + s]))
        i += s
    lines.append("SPEC %s M %d SIZES %s PARAMS %s FIRSTGROUP %s LASTGROUP %s"
                 % (spec, len(g), ",".join(str(s) for s in sizes),
                    ",".join(str(s) for s in pc), " ".join(g[0]), " ".join(g[-1])))
    print("guard 4: %-9s m=2  sizes %-8s  params %9d / %-9d  share %.6f"
          % (spec, str(sizes), pc[0], pc[1], pc[0] / float(tot)))
    print("         coarse group (%d tensors) ends at   %s" % (k, g[0][-1]))
    print("         fine   group (%d tensors) starts at %s" % (sizes[1], g[1][0]))
    print("         FULL coarse = %s" % " ".join(g[0]))
    print("         FULL fine   = %s" % " ".join(g[1]))
    seen[k] = (set(g[-1]), pc[0])

# guard 4d: THE POINT OF THE BATCH.  Which tensor moves at each step, measured.
ks = sorted(seen)
print("guard 4d: the single-tensor steps, MEASURED on the live model:")
moved_total = 0
for a, b in zip(ks, ks[1:]):
    moved = [n for n in names if n in seen[a][0] and n not in seen[b][0]]
    dp = seen[b][1] - seen[a][1]
    moved_total += dp
    flag = "" if len(moved) == 1 else "   <<< NOT A SINGLE TENSOR"
    print("         k=%d -> k=%d moves %-30s %8d params%s"
          % (a, b, " ".join(moved), dp, flag))
    if len(moved) != 1:
        bad += 1
    if sum(numel[names.index(n)] for n in moved) != dp:
        print("!!! guard 4d: parameter bookkeeping does not close at k=%d->%d" % (a, b))
        bad += 1
if ks:
    print("guard 4d: the %d steps move %d parameters in total, %.4f %% of the "
          "fine group at k=%d"
          % (len(ks) - 1, moved_total,
             100.0 * moved_total / float(sum(numel[ks[0]:])), ks[0]))

# This batch has NO named-granularity arm, so no spec may collide with one.
# init_meta routes exact string matches away from the blockwise path; the guard
# proves the routing list on the LIVE source rather than assuming it.
import inspect, re as _re
src = inspect.getsource(HF.init_meta)
mm = _re.search(r"stepsize_groups\s+in\s+\[([^\]]*)\]", src)
if not mm:
    print("!!! guard 4: could not find init_meta's exact-match routing list")
    bad += 1
else:
    named = [x.strip().strip("'\"") for x in mm.group(1).split(",")]
    print("guard 4: init_meta routes these names away from blockwise: %s" % named)
    for spec in specs:
        if spec in named:
            print("!!! guard 4: cut spec %s collides with a named granularity" % spec)
            bad += 1
    print("guard 4: no cut spec collides with a named granularity; every arm "
          "takes the blockwise path")

if bad:
    raise SystemExit(1)
open(out, "w").write("\n".join(lines) + "\n")
print("guard 4: %d cut specs verified; PARTITION-MANIFEST.txt written to %s"
      % (len(specs), out))
PYEOF
[ -s "$MANIFEST" ] && echo "guard 4b: manifest is non-empty" || guard_fail "guard 4b: no manifest"

# ---- guard 5: capacity ------------------------------------------------------

[ "${PEND:-0}" -le 40 ] || guard_fail "guard 5: $PEND pending, over cap"

        num-epochs=$EPOCHS batch-size=$BATCH gamma=$GAMMA dataset=$DSET NN-name=$NET)

N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    case $ARM in
      k52) G=$G_K52 ;;
      k53) G=$G_K53 ;;
      k54) G=$G_K54 ;;
      k55) G=$G_K55 ;;
      *)   echo "!!! unknown arm $ARM"; exit 2 ;;
    esac
    # NOTE the single quotes around $G below: "[53,9]" is a bash bracket GLOB.
    # Unquoted, inside the eval, it would expand against the CWD if a file named
    # 5 or 3 or , ever existed there.  Quoted, argv reaches train.py literally
    # and the runner's `echo "ARGS: $@"` prints it unchanged for RULE 20.
    CMD="sbatch --job-name=scl1-$ARM-s$S --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP,HIER=none,SCHED=none,PROBE=0 \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
--gamma $GAMMA --meta-stepsize $MST --alpha0 $ALPHA0 --num-epochs $EPOCHS \
--stepsize-groups '$G' --seed $S --save-directory $SAVE --run-name scl1-$ARM-s$S"
    if guard_presubmit "$CMD" "${DESIGN[@]}" >/dev/null 2>&1; then :; else
      echo "!!! RULE 20 PRE-CHECK FAILED for scl1-$ARM-s$S"
      guard_presubmit "$CMD" "${DESIGN[@]}"; BAD=$((BAD+1))
    fi
    if [ "$SUBMIT" = 1 ]; then eval "$CMD" >/dev/null 2>&1 && N=$((N+1)) || REJ=$((REJ+1))
    else echo "$CMD"; echo; N=$((N+1)); fi
  done
done

[ "$BAD" = 0 ] || { echo "!!! composed lines are malformed"; exit 2; }
