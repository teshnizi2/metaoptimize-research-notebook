def sd(v):
    if len(v) < 2:
        return None

def u_of(n, P):
    if n <= 0:
        return 0
    return min(n - 1, P) - max(0, n - 1 - P)


def v_of(n, P, b0=B0):
    return f32(min(HI, max(LO, b0 + MS * u_of(n, P))))


def lion_natural(beta_pre, mom_pre, z):
    a, b = B2 * mom_pre, (1 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(LO, min(HI, beta_pre - MS * s)), tie


def hold_value(arm, n):
    """the carrier group's schedule (group 1)."""
    m, P = MODE[arm]
    if m == "floor":
        return LO
    return v_of(n, P)


_KNOTS = {}


def load_knots(path=ISOPATH_FILE, want_sha=ISOPATH_SHA):
    if path in _KNOTS:
        return _KNOTS[path]
    out = None
    if os.path.exists(path):
        body = open(path, "rb").read()
        if hashlib.sha256(body).hexdigest() == want_sha:
            d = json.loads(body.decode("utf-8"))
            out = ([int(x) for x in d["n"]], [float(x) for x in d["v"]])
    _KNOTS[path] = out
    return out


def rec_value(n, kn, kv, b0=B0):
    """PATCH_RESTHOLD's rec: schedule as float32 -- the SAME arithmetic, in the same order, as _rh_value."""
    if n < kn[0]:
        x = b0
    elif n >= kn[-1]:
        x = kv[-1]
    else:
        lo_, hi_ = 0, len(kn) - 1
        while hi_ - lo_ > 1:
            mid = (lo_ + hi_) // 2
            if kn[mid] <= n:
                lo_ = mid
            else:
                hi_ = mid
        k = lo_
        x = kv[k] + (kv[k + 1] - kv[k]) * (n - kn[k]) / float(kn[k + 1] - kn[k])
    return f32(min(HI, max(LO, x)))


def hold_r_value(arm, n):
    """the complement's schedule (group 0) of a forced arm; None if the replay file is unavailable."""
    m, x = RMODE[arm]
    if m == "tri":
        return v_of(n, x)
    K = load_knots()
    if K is None:
        return None
    return rec_value(n, K[0], K[1])


def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

def r18_tensors():
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if stride != 1 or cin != w:
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]

def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names=None):
    names = names or NAMES
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

        lines.append("TENSOR %d %s %d %s" % (i, n, q, o))
    lines.append("CARRIERS %s" % ",".join("%d:%s" % x for x in zip(CARRIER_IDX, CARRIERS)))
    lines.append("REPLAY B0 %r MS %r P_HIGH %d P_BIG %d" % (B0, MS, P_HIGH, P_BIG))
    lines.append("RESTREPLAY ISOPATH %s SHA256 %s KNOTS %d" % (ISOPATH_ID, ISOPATH_SHA, ISOPATH_KNOTS))
    for a in ARMS:
        g = arm_groups(a, names)
        if SPEC[a] == "scalar":
            lines.append("ARMSPEC %s SPEC scalar TYPE scalar M 1" % a)
        else:
            lines.append("ARMSPEC %s SPEC %s TYPE blockwise M %d SIZES %s PARAMS %s GROUP1 %s"
                         % (a, SPEC[a], len(g), ",".join(str(len(x)) for x in g),
                            ",".join(str(sum(numel[i - 1] for i in x)) for x in g),
                            " ".join(names[i - 1] for i in g[1])))
        lines.append("GROUPHOLD %s %s" % (a, GROUPHOLD[a] if GROUPHOLD[a] else "off"))
        lines.append("RESTHOLD %s %s" % (a, RESTHOLD[a] if RESTHOLD[a] else "off"))
        lines.append("WITNESS %s %s" % (a, WITNESS_GH[a]))
        lines.append("RWITNESS %s %s" % (a, WITNESS_RH[a]))
    return "\n".join(lines) + "\n"

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "bh": [ln for ln in lines if ln.startswith("BETA_HOLD")],
           "gh": [ln for ln in lines if ln.startswith("GROUP_HOLD")],
           "rh": [ln for ln in lines if ln.startswith("REST_HOLD")],
           "ch": [ln for ln in lines if ln.startswith("COMP_HOLD")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest job id among COMPLETE runs wins."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s" % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r):
    return (r.get("epochs_requested") == str(EPOCHS) and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0 and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
    return (math.sqrt(ss / df) if df else None), df, nc

def load_probe(runsdir, arm, seed, prefix=PREFIX, max_step=None):
    """-> list of records, or None if absent.  Unparseable lines are None entries.  max_step (selftest only) keeps
    records with step < max_step."""
    for base in (os.path.join(runsdir, prefix), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (prefix, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            if max_step is not None:
                recs = [r for r in recs if isinstance(r, dict) and isinstance(r.get("step"), int) and r["step"] < max_step]
            return recs
    return None


GH_KEYS = ("gh_n", "gh_active", "gh_nat", "gh_held")
RH_KEYS = ("rh_n", "rh_active", "rh_nat", "rh_held")
FOREIGN_KEYS = ("bh_n", "bh_active", "bh_nat", "bh_held", "ch_n", "ch_active", "ch_nat", "ch_held")


def _groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    return [float(x) for x in v[0]]


def _audit(r, pfx, n, g, beta, bpre, mom, z, prev):
    """one patch's own record on one group.  -> (ok, active_flag, count); count None if unreadable."""
    try:
        cn, ca, cnat, cheld = (int(r[pfx + "_n"]), int(r[pfx + "_active"]), float(r[pfx + "_nat"]), float(r[pfx + "_held"]))
    except (KeyError, TypeError, ValueError):
        return False, False, None
    nat, tie = lion_natural(bpre[g], mom[g], z[g])
    ok = (cn == n and abs(cheld - beta[g]) <= BH_TOL and 0 <= ca <= cn and ca >= prev and (tie or abs(cnat - nat) <= BH_TOL))
    return ok, abs(cnat - cheld) > BH_TOL, ca


def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail).  Pure function of the records."""
    ng = len(arm_groups(arm))
    held = arm in HELD
    forced = arm in FORCED
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0,
         "bad_sched": 0, "bad_sched_pre": 0, "bad_rsched": 0, "bad_rsched_pre": 0, "bad_gh": 0, "bad_rh": 0,
         "keys_wrong": 0, "active_records": 0, "ractive_records": 0, "gh_active_last": None, "rh_active_last": None,
         "worst_lion": 0.0, "worst_sched": 0.0, "worst_rsched": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    if forced and hold_r_value(arm, 2) is None:
        d["why"] = "replay file %s unavailable or sha != ISOPATH_SHA" % ISOPATH_FILE
        return False, d
    prev_g, prev_r = -1, -1
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) or len(r["beta"]) != ng:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre = _groupvec(r, "beta_pre", ng)
            mom = _groupvec(r, "mom_pre", ng)
            z = _groupvec(r, "z_agg", ng)
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z):
            d["nonfinite"] += 1
            continue
        s = r["step"]
        if s != 100 * k:
            d["bad_step"] += 1
        d["n"] += 1
        n = s + 2
        unheld = range(ng) if not held else (() if forced else (0,))
        for g in unheld:
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst_lion"] = max(d["worst_lion"], e)
            d["bad_lion"] += e > BH_TOL
        has_gh = any(key in r for key in GH_KEYS)
        has_rh = any(key in r for key in RH_KEYS)
        d["keys_wrong"] += any(key in r for key in FOREIGN_KEYS)
        if not held:
            d["keys_wrong"] += has_gh or has_rh
            continue
        if not forced:
            d["keys_wrong"] += has_rh
        es, ep = abs(beta[1] - hold_value(arm, n)), abs(bpre[1] - hold_value(arm, n - 1))
        d["worst_sched"] = max(d["worst_sched"], es, ep)
        d["bad_sched"] += es > BH_TOL
        d["bad_sched_pre"] += ep > BH_TOL
        ok, act, ca = _audit(r, "gh", n, 1, beta, bpre, mom, z, prev_g)
        d["bad_gh"] += not ok
        if ca is not None:
            d["active_records"] += act
            prev_g = max(prev_g, ca)
            d["gh_active_last"] = ca
        if forced:
            cs, cp = abs(beta[0] - hold_r_value(arm, n)), abs(bpre[0] - hold_r_value(arm, n - 1))
            d["worst_rsched"] = max(d["worst_rsched"], cs, cp)
            d["bad_rsched"] += cs > BH_TOL
            d["bad_rsched_pre"] += cp > BH_TOL
            ok, act, ca = _audit(r, "rh", n, 0, beta, bpre, mom, z, prev_r)
            d["bad_rh"] += not ok
            if ca is not None:
                d["ractive_records"] += act
                prev_r = max(prev_r, ca)
                d["rh_active_last"] = ca
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_step"] == 0 and d["bad_lion"] == 0 and d["keys_wrong"] == 0)
    if held:
        ok = (ok and d["bad_sched"] == 0 and d["bad_sched_pre"] == 0 and d["bad_gh"] == 0
              and (d["gh_active_last"] is None or d["gh_active_last"] >= d["active_records"]))
    if forced:
        ok = (ok and d["bad_rsched"] == 0 and d["bad_rsched_pre"] == 0 and d["bad_rh"] == 0
              and (d["rh_active_last"] is None or d["rh_active_last"] >= d["ractive_records"]))
    return ok, d

def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord and isinstance(r.get("step"), int):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    pin = None
    kpk = max(range(len(vals)), key=lambda i: vals[i][1])
    for k in range(kpk, len(vals)):
        if all(math.exp(v + 15.0) <= 2.0 for _s, v in vals[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    win = [math.exp(v + 15.0) for s, v in vals if 15 <= s / float(STEPS_PER_EPOCH) < 30]
    win2 = [math.exp(v + 15.0) for s, v in vals if 30 <= s / float(STEPS_PER_EPOCH) < 60]
    return {"last": vals[-1][1], "peak": vals[kpk][1], "peak_epoch": vals[kpk][0] / float(STEPS_PER_EPOCH),
            "pin_epoch": pin, "r1530": mean(win), "r3060": mean(win2)}

# THE BRANCH MAP -- a pure function of the in-batch means and one flag

def state(L, M):
    if L >= M["ISO"] - MATCH_BAR:
        return "AT-ISO"
    if L <= M["k01"] + NULL_BAR:
        return "AT-K01"
    return "BETWEEN"


def dose_word(M):
    if M["HOLDBIG"] <= M["k01"] + NULL_BAR:
        return "DOSE-FULL"
    if abs(M["HOLDBIG"] - M["HOLDHIGH"]) <= MATCH_BAR:
        return "NO-DOSE"
    if M["HOLDBIG"] > M["HOLDHIGH"] + MATCH_BAR:
        return "DOSE-REVERSED"
    return "DOSE-PARTIAL"


def route_word(L, ref, M, pfx):
    if ref > M["ISO"] - REP_MARGIN:
        return pfx + "-UNREADABLE"
    if L >= M["ISO"] - MATCH_BAR:
        return pfx + "-VIA"
    if L <= ref + MATCH_BAR:
        return pfx + "-DIRECT"
    return pfx + "-PARTIAL"


def decide(M, diverged=False):
    """M: arm -> mean plateau5."""
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M["ISO"] - M["k01"] < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED"
    st = state(M["LOWISOPATH"], M)
    if st == "AT-K01":
        return "COMP-HOLD-CONTROL-FAILED"
    if st == "BETWEEN":
        return "COMP-HOLD-CONTROL-PARTIAL"
    if not (M["k01"] + REP_MARGIN <= M["HOLDHIGH"] <= M["ISO"] - REP_MARGIN):
        return "REPLICATE-FAILED"
    return "%s+%s+%s" % (dose_word(M), route_word(M["HIGHISOPATH"], M["HOLDHIGH"], M, "ROUTE"),
                         route_word(M["BIGISOPATH"], M["HOLDBIG"], M, "BIGROUTE"))

def licence_lines(branch):
    if branch in LICENSE:
        return LICENSE[branch]
    parts = branch.split("+")
    if len(parts) != 3 or parts[0] not in LICENSE_DOSE or parts[1] not in LICENSE_ROUTE or parts[2] not in LICENSE_BIG:
        return None
    out = LICENSE_DOSE[parts[0]] + LICENSE_ROUTE[parts[1]] + LICENSE_BIG[parts[2]]
    if parts[0] == "DOSE-FULL" and parts[2] == "BIGROUTE-DIRECT":
        out = out + ["TOGETHER: at PlainNet's dose the ResNet carrier group stalls the run with the complement on ISO's own path --",
                     "the transplant of cvt6's HIGHHEADPATH-AT-K01.  EFFECT ON 240's PlainNet sentence: it TRANSFERS to",
                     "ResNet18_c100 at this cell AT PlainNet's DOSE (three carriers held together; open-loop complement)."]
    out = out + ["EFFECT ON " + READING + ": as the three words say, at this cell only (ResNet18_c100, ciso1's cell, 100",
                 "epochs, the three carriers held together, the complement's path forced OPEN-LOOP).  Readings at k01 are LOCATIONS."]
    return out


def all_branches():
    out = set(LICENSE)
    for d in LICENSE_DOSE:
        for r in LICENSE_ROUTE:
            for b in LICENSE_BIG:
                out.add("%s+%s+%s" % (d, r, b))
    return out

def _e_write(d, arm, s, jid, level, n_ep=EPOCHS, run_done=True, env=_E_ENV, spec=None, gh=None, rh=None, vw=None,
             bh=None, ch=None):
    rn = "%s-%s-s%d" % (PREFIX, arm, s)
    args = ("--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 "
            "--alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 "
            "--weight-decay-meta 0 --dataset CIFAR100 --NN-name %s --batch-size 100 "
            "--max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-6 "
            "--num-epochs 100 --stepsize-groups %s --seed %d "
            "--save-directory /cluster/account_b/metaopt/runs/cvt8 --run-name %s" % (NET, spec or SPEC[arm], s, rn))
    lines = ["NODE=[HOST_45cba27b74] | JOB=%s %d | AUGMENT=1" % (rn, jid), "ARGS: " + args, env, "NVIDIA L4, 23034 MiB",
             WITNESS_VW if vw is None else vw, WITNESS_BH if bh is None else bh,
             WITNESS_GH[arm] if gh is None else gh, WITNESS_RH[arm] if rh is None else rh, ch or "",
             "Files already downloaded and verified", "Files already downloaded and verified",
             "PROBE_TENSOR: on every=100 type=%s tensors=%d meta_alg=Lion momentum_param=0.99 Lion_beta2=0.9 dir=/x"
             % (PT_TYPE[arm], NTENS)]
    lines = [ln for ln in lines if ln != ""]
    for e in range(n_ep):
        te = level if e >= EPOCHS - 5 else level * (1 - math.exp(-(e + 1) / 12.0))
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(99.9, te + 15.0), te))
    if run_done:
        lines += ["43  minutes", "RUN_DONE"]
    open(os.path.join(d, "%s-%d.out" % (rn, jid)), "w").write("\n".join(lines) + "\n")


def _e_probe_records(arm, seed=0, n=N_RECORDS, as_arm=None, as_rarm=None, unheld=False, runheld=False,
                     touch_complement=False, pre_natural=False, rpre_natural=False, gh_forge=False, rh_forge=False,
                     gh_off_by_one=False, rh_off_by_one=False, strip_gh=False, strip_rh=False, add_gh=False,
                     add_rh=False, add_bh=False, add_ch=False, rh_as_tri=None):
    """synthetic PROBE_TENSOR records consistent with the harness's Lion step and the arm's holds.
    as_arm / as_rarm: the carrier / complement hold of ANOTHER arm.  unheld / runheld: that group follows Lion (the nulls).
    rh_as_tri: the complement follows tri:<P> instead of ISOPATH (wrong replay)."""
    import random
    rnd = random.Random(8000 + seed + 7 * ARMS.index(arm))
    ng = len(arm_groups(arm))
    harm = as_arm or arm
    rarm = as_rarm or arm
    out = []
    active = ractive = 0
    for k in range(n):
        s = 100 * k
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        z = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [f32(max(LO, min(HI, -13.8 + 0.3 * math.sin(k / 30.0) - 0.004 * k))) for _ in range(ng)]
        beta = [f32(lion_natural(bpre[g], mom[g], z[g])[0]) for g in range(ng)]
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99}
        if ng == 2 and arm in HELD and not unheld:
            bpre[1] = hold_value(harm, s + 1) if not pre_natural else f32(bpre[1] + 0.25)
            beta[1] = hold_value(harm, s + 2)
            nat1 = f32(lion_natural(bpre[1], mom[1], z[1])[0])
            if abs(nat1 - beta[1]) > BH_TOL:
                active += 1
            rec.update({"gh_n": s + 2 + (1 if gh_off_by_one else 0), "gh_active": active,
                        "gh_nat": (nat1 + 0.01) if gh_forge else nat1, "gh_held": beta[1]})
            if strip_gh:
                for key in GH_KEYS:
                    rec.pop(key)
        if ng == 2 and rarm in FORCED and not runheld:
            hv = (lambda m: v_of(m, rh_as_tri)) if rh_as_tri else (lambda m: hold_r_value(rarm, m))
            bpre[0] = hv(s + 1) if not rpre_natural else f32(bpre[0] + 0.25)
            beta[0] = hv(s + 2)
            nat0 = f32(lion_natural(bpre[0], mom[0], z[0])[0])
            if abs(nat0 - beta[0]) > BH_TOL:
                ractive += 1
            rec.update({"rh_n": s + 2 + (1 if rh_off_by_one else 0), "rh_active": ractive,
                        "rh_nat": (nat0 + 0.01) if rh_forge else nat0, "rh_held": beta[0]})
            if strip_rh:
                for key in RH_KEYS:
                    rec.pop(key)
        if touch_complement:
            beta[0] = f32(beta[0] + 0.002)
        if add_gh:
            rec.update({"gh_n": s + 2, "gh_active": 0, "gh_nat": beta[-1], "gh_held": beta[-1]})
        if add_rh:
            rec.update({"rh_n": s + 2, "rh_active": 0, "rh_nat": beta[0], "rh_held": beta[0]})
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[-1], "bh_held": beta[-1]})
        if add_ch:
            rec.update({"ch_n": s + 2, "ch_active": 0, "ch_nat": beta[0], "ch_held": beta[0]})
        rec.update({"beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [z],
                    "z_tensor": [0.0] * NTENS, "m_tensor": [0.0] * NTENS})
        out.append(rec)
    return out


def _e_probe_template(root):
    for s in SEEDS:
        for a in ARMS:
            p = os.path.join(root, "probe_%s-%s-s%d" % (PREFIX, a, s))
            os.makedirs(p, exist_ok=True)
            with open(os.path.join(p, "probe.jsonl"), "w") as f:
                for r in _e_probe_records(a, seed=s):
                    f.write(json.dumps(r) + "\n")

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None, env_break=None,
             spec_break=None, gh_break=None, rh_break=None, vw_break=None, bh_break=None, ch_break=None, manifest=None,
             prov=None, probe_break=None, probe_drop=None):
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(manifest if manifest is not None else synthetic_manifest_text())
    _e_prov(d, **(prov or {}))
    jid = 9900000
    pb = probe_break or {}

    def pick(br, a, s):
        return br[1] if br and br[0] == (a, s) else None
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = "scalar" if spec_break == (a, s) else None
            _e_write(d, a, s, jid, L, env=env, spec=spec, gh=pick(gh_break, a, s), rh=pick(rh_break, a, s),
                     vw=pick(vw_break, a, s), bh=pick(bh_break, a, s), ch=pick(ch_break, a, s))
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()


def _bar_lines(out):
    return [ln for ln in out.splitlines() if ln.startswith("  SIGMA_USED") or ln.startswith("  SE_ARM_DIFF")
            or ln.startswith("  bars:") or ln.startswith("  PRIMARY")]


def _spec_from_scorer(fname, key):
    p = os.path.join(HERE, fname)
    if not os.path.exists(p):
        return None
    import ast
    txt = open(p).read()
    m = re.search(r"^SPEC = (\{.*?^\})", txt, re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)


def _load_module(name, fname):
    import importlib.util
    p = os.path.join(HERE, fname)
    if not os.path.exists(p):
        return None
    spec = importlib.util.spec_from_file_location(name, p)
    m = importlib.util.module_from_spec(spec)
    sys.path.insert(0, HERE)
    try:
        spec.loader.exec_module(m)
    finally:
        sys.path.pop(0)
    return m

        ("probe: a LOWISOPATH probe dir missing", {}, {"probe_drop": ("LOWISOPATH", 103)}, "PATCH-NOT-VERIFIED"),
        ("probe: an ISO file short (400 records)", {}, {"probe_break": {("ISO", 104): _e_probe_records("ISO", n=400, seed=104)}}, "PATCH-NOT-VERIFIED"),
        ("a DROPPED run", {}, {"drop": ("HOLDBIG", 103)}, "INCOMPLETE"),
        ("a TRUNCATED higher-jid resubmission", {}, {"shadow": ("BIGISOPATH", 102)}, B0R),
    ]
    results = {}
    for lab, lv, kw, want in cases:
        dd = os.path.join(tmp, re.sub(r"\W+", "_", lab))
        os.makedirs(dd)
        _e_build(dd, template, levels=lv, **kw)
        rc, fin, out = _e_score(dd, csvpath)
        results[lab] = (rc, fin, out, dd)
        chk(fin.startswith("FINAL: %s |" % want), "E %-60s -> %s" % (lab, want), fin[:100])
    rc, fin, out, dpath = results["DOSE-FULL x DIRECT x BIG-DIRECT"]
    chk(rc == 0 and all(x in fin for x in ("HARNESS-CLEAN", "PATCH-BITES", "HOLDBIG-AT-K01", "HOLDHIGH-BETWEEN", "LOWISOPATH-AT-ISO",
                                          "FLOOR-READINGS-ARE-BOUNDS", "ONE-NETWORK-RESNET", "BIG-IS-PLAINNET-DOSE",
                                          "ISOPATH-REPLAY-FILE", "FORCED-ARMS-OPEN-LOOP", "TRAIN-AGREES")),
        "E the base case exits 0 with the state stamps and the unconditional stamps", fin[:160])
    chk("HIGHISOPATH-BELOW-HOLDHIGH" in results["HIGHISOPATH below HOLDHIGH is DIRECT, stamped"][1],
        "E HIGHISOPATH more than 5 pp below HOLDHIGH is stamped, and reads ROUTE-DIRECT")
    chk("UNBRANCHED-DOSE-FULL" in results["COMP-HOLD-CONTROL-FAILED"][1], "E an unbranched batch stamps its dose word")
    chk(results["BROKEN REST_HOLD: every forced complement UNHELD at DIRECT's levels"][0] == 1,
        "E the BROKEN-REST_HOLD null at DIRECT's levels exits 1, reaching NO branch")
    chk(results["a DROPPED run"][0] == 2, "E a DROPPED run exits 2 (INCOMPLETE)")
    chk(results["harness: AUGMENT=0 on one run"][0] == 1, "E a harness failure exits 1")
    chk(results["a TRUNCATED higher-jid resubmission"][1] == results["DOSE-FULL x DIRECT x BIG-DIRECT"][1],
        "E a truncated higher-jid resubmission does not shadow a complete run")
    chk("G-BITE BIGISOPATH   s102 records 500  bad 0  nonfinite 0  lion 0" in out and "keys-wrong 0" in out,
        "E G-BITE reads 500 records per run and audits every half")
    chk("arm-mean TEST (TRAIN) at epoch lines" in out and "TOGETHER: at PlainNet's dose" in out,
        "E the descriptive readout and the composed licence print (after FINAL)")

    print("\n   O2 INVARIANCE: the SAME synthetic runs against four corpora")
    variants = _e_csv_variants(csvpath if csvpath and os.path.exists(csvpath) else None, tmp)
    for case in ("DOSE-FULL x DIRECT x BIG-DIRECT", "NO-DOSE x VIA x BIG-VIA", "DOSE-PARTIAL x DIRECT x BIG-DIRECT"):
        ref_fin, ref_bars = None, None
        dd = results[case][3]
        for lab in ("real", "real+invented", "minimal-invented", "no-corpus"):
            if lab not in variants:
                skip("O2 %s corpus not available on this host" % lab)
                continue
            rc, fin, out = _e_score(dd, variants[lab])
            bars = _bar_lines(out)
            if ref_fin is None:
                ref_fin, ref_bars = fin, bars
            chk(fin == ref_fin and bars == ref_bars and len(bars) == 6,
                "O2 %-36s %-17s FINAL and SIGMA/SE/bars/PRIMARY lines are IDENTICAL" % (case, lab), fin[:44])
    if "real+invented" in variants:
        a_ = pooled_sigma(read_corpus(variants["real"]), net_is(NET))[0]
        b_ = pooled_sigma(read_corpus(variants["real+invented"]), net_is(NET))[0]
        chk(a_ is not None and b_ is not None and b_ > 5 * a_,
            "O2 non-vacuity: invented rows at cvt8's own cells MOVE the FILTERED ResNet sigma (%.4f -> %.4f); FINAL did not move"
            % (a_ or 0, b_ or 0))

    print("\n   THE DOCUMENTED ONE-ARGUMENT INVOCATION, as a subprocess")
    me = os.path.abspath(__file__)
    try:
        p = subprocess.run([sys.executable, me, dpath], stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                           universal_newlines=True, timeout=900)
        fl = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 0 and fl and fl[-1] == results["DOSE-FULL x DIRECT x BIG-DIRECT"][1],
            "`python3 %s <runsdir>` exits 0 with the in-process FINAL line" % os.path.basename(me), "rc %d" % p.returncode)
        p = subprocess.run([sys.executable, me, results["a DROPPED run"][3]], stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                           universal_newlines=True, timeout=900)
        fl = [ln for ln in p.stdout.splitlines() if ln.startswith("FINAL:")]
        chk(p.returncode == 2 and fl and fl[-1].startswith("FINAL: INCOMPLETE"), "... and on a dropped run exits 2 with FINAL: INCOMPLETE",
            "rc %d" % p.returncode)
    except Exception as e:
        chk(False, "subprocess invocation", "%s: %s" % (type(e).__name__, e))

    print("\nF. THE BRANCH MAP IS TOTAL -- 20,000 random in-batch means, every branch registered")
    import random
    rnd = random.Random(20260917)
    seen = set()
    for _ in range(20000):
        m = dict((a, rnd.uniform(1.0, 80.0)) for a in ARMS)
        if rnd.random() < 0.7:
            m["k01"], m["ISO"] = rnd.uniform(15, 28), rnd.uniform(62, 78)
            m["LOWISOPATH"] = m["ISO"] - rnd.uniform(-3, 4)
            m["HOLDHIGH"] = rnd.uniform(m["k01"] + 10, m["ISO"] - 10)
        seen.add(decide(m, rnd.random() < 0.02))
    chk(seen <= all_branches() and all(licence_lines(b) for b in seen), "every branch decide() returns has a registered licence text",
        "%d distinct" % len(seen))
    reached = set(r[1].split(" |")[0][len("FINAL: "):] for r in results.values())
    note("branches reached by section E or the sweep: %d of %d registered (the rest are word combinations of reached words)"
         % (len(reached | seen), len(all_branches())))
    words = set(w for b in (reached | seen) for w in b.split("+"))
    chk(set(LICENSE_DOSE) | set(LICENSE_ROUTE) | set(LICENSE_BIG) - {"ROUTE-UNREADABLE"} <= words | {"ROUTE-UNREADABLE"}
        and set(LICENSE) - {"INCOMPLETE", "HARNESS-UNSOUND", "PATCH-NOT-VERIFIED"} <= reached | seen,
        "every registered word and every single-word branch is reached (ROUTE-UNREADABLE is unreachable past the replicate gate)")
    chk(all("247" in " ".join(licence_lines(b)) for b in all_branches() if b not in ("INCOMPLETE", "SCALAR-NOT-COLLAPSED", "POSITIVE-CONTROL-FAILED")),
        "every substantive licence states its effect on 247's bounds")

    print("\nG. RULE 21 PREMISE: no cvt8 run may exist yet")
    runs, _n = read_runs(runsdir) if runsdir else ({}, [])
    chk(not runs, "no cvt8-*.out under %s" % runsdir, "%d found" % len(runs))
    if csvpath and os.path.exists(csvpath):
        chk(not any(r.get("run", "").startswith(PREFIX + "-") for r in csv.DictReader(open(csvpath))), "no cvt8- row in the CSV")
    shutil.rmtree(tmp, ignore_errors=True)

    print("\n%s" % ("ALL PASS" if not GATE_FAIL else "FAILURES: %d" % len(GATE_FAIL)))
    for f in GATE_FAIL:
        print("   FAIL", f)
    return 1 if GATE_FAIL else 0


def main():
    ap = argparse.ArgumentParser(description="cvt8 registered scorer.  Documented invocation: "
                                             "python3 analysis/cVT8_doseroute_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt8/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
    sys.exit(main())
