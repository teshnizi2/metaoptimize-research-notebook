#!/bin/bash
# =============================================================================
# Cycle 51 -- THE CEILING LADDER, a.k.a. THE FIRST BOX-FREE MEASUREMENT OF THE
# HEADLINE.  Runs on cluster_b.  18 jobs.  Job tag `cl5-*`.
#
# OPEN QUESTION IT ANSWERS.
#   The campaign's headline sentence (CORRECTIONS 59) is:
#     "At the adapted equilibrium, per-weight meta-gradients carry between 16% and 55%
#      of the independent information their count implies -- N_eff/m = 0.163 / 0.481 /
#      0.552 / 0.305."
#   EVERY number in it was measured inside a BETA_CLIP box, and until this tick nobody
#   had checked whether the box sets the number.  Cycle 50 tried, with `wc5-*`, and
#   moved the wrong wall.
#
# WHY wc5 MOVED THE WRONG WALL, and this is the whole reason this batch exists.
#   `c50_wideclip_ladder.sh` chose to move only the LOW bound, on this premise:
#     "The HIGH bound is at 0.00% in EVERY arm of the whole 36-job batch ... it
#      provably never binds and moving it would vary a second field."
#   That 0.00% was read from the PER-TENSOR `beta[]` list -- one MEAN per tensor.
#   The clamp is applied PER COORDINATE.  Read at the resolution the clamp actually
#   operates at (`beta_true_max`, already in the same records), FINDINGS 51.1:
#
#       ms=1e-2, weightwise    LO binds 91.9% of records (first at 162)
#                              HI binds 95.4% of records (first at  92)   <-- dominant
#       ms=1e-3, weightwise    LO binds  8.2% of records (first at 1836)
#                              HI binds  7.4% of records (first at 1523)
#
#   The high guard binds MORE, and EARLIER, than the low guard wc5 moved.  So W2
#   could not pass (layerwise record-binding fell only 88.0% -> 71.7% against its 5%
#   bar) and W1 came back UNINTERPRETABLE by its own fallback rule.
#   `analysis/c51_wideclip.py` (selftest 26/26) prints all three denominators.
#
# WHAT wc5 DID ESTABLISH, and this batch is built on it:
#   * LO = -30 NEVER binds at ms=1e-3 (0.0% of records, all three rungs).  So the
#     floor is already out of the way and only the ceiling is left to vary.
#   * The clip is ACCURACY-NEUTRAL: +0.027 pp mean over 6 matched cells at n=3.
#   * The headline is FLOOR-ROBUST: N_eff/m (weightwise, variance, steady half)
#     0.481 at LO=-15 -> 0.519 at LO=-30, i.e. +0.038 for a 2.2x deeper box.
#
# THE DESIGN.  LO = -30 throughout (measured non-binding).  HI takes three values;
# the middle one is ALREADY ON DISK at n=3, byte-matched, so 18 jobs buy a 3-point
# ladder:
#       HI = -4.6052  (alpha_max 0.01)   NEW, 9 jobs   -- ceiling pressed DOWN
#       HI = -2.3026  (alpha_max 0.10)   wc5-*-m3-*    -- the existing control
#       HI =  0.0000  (alpha_max 1.00)   NEW, 9 jobs   -- predicted NEVER to bind
#   ms = 1e-3 only.  That is where the headline lives.  ms=1e-2 is already labelled
#   boundary-dominated (CORRECTIONS 62) and is not quoted in any claim, so spending
#   jobs to un-confound a rung nobody quotes would be the wrong purchase.
#   Rungs w / node / lay: the profile b is fit across exactly these three.
#
# THE PREDICTION THAT MAKES HI=0.0 THE POINT OF THE BATCH, computed BEFORE submission.
#   At ms=1e-3 the TOP coordinate climbs at a measured 0.00337 log-units/record
#   (it travels from beta0 = -6.908 to -2.3026 by record 1367, FINDINGS 51.2).
#   Extrapolating that measured rate:
#       ceiling -4.6052  ->  reached at record ~683  ->  ~66% of records bound
#       ceiling -2.3026  ->  reached at record 1367  ->  ~32% of records bound
#                            (measured: 8.9%, so the linear extrapolation is an
#                             UPPER bound -- the climb decelerates.  Say so.)
#       ceiling  0.0000  ->  reached at record ~2050 > 2000  ->  NEVER binds
#   So the HI=0.0 arm, with LO=-30 also non-binding, is the FIRST COMPLETELY
#   UNCLIPPED measurement of N_eff/m this campaign has ever taken.
#
# PRE-REGISTERED PREDICTIONS -- score the verdict against these, not against what lands.
#
#   (X0) VALIDITY GATE.
#        1. beta must MOVE: beta_true_max != beta_true_min on the last record, 18/18.
#        2. n_records == 2000 and n_tot matches the wc5 partner EXACTLY per rung:
#           w 11,173,962 / node 14,420 / lay 62.  A mismatch voids the contrast.
#        3. THE NULL.  At LAYERWISE the HI guard binds on 0.0% of records at ms=1e-3
#           in BOTH wc5 and ml5 (measured).  So moving the ceiling CANNOT matter
#           there.  Both new layerwise arms must reproduce the wc5 layerwise control:
#           plateau within +-0.6 pp (2 pooled seed sd, pooled sd = 0.30 pp measured
#           across the three wc5 m3 rungs) AND rho_w(k=1, steady) within 2x.
#           If layerwise moves, the intervention changed something other than the
#           ceiling and NOTHING in this batch is quotable.
#
#   (X0.3) THE DOSE CHECK -- **SCORE THIS BEFORE X1**.  This is the W2 lesson applied:
#        a contrast is only interpretable if the intervention demonstrably bit.
#        Using the per-COORDINATE denominator (the one the clamp operates at):
#           HI=-4.6052, weightwise: records with a coordinate at HI must be > 40%
#                                   (control 8.9%; linear extrapolation says ~66%)
#           HI= 0.0000, weightwise: records with a coordinate at HI must be < 5%
#                                   (i.e. the arm is genuinely unclipped above)
#        If NEITHER moves, the ladder has no dose and X1/X2 are UNINTERPRETABLE --
#        say that, do not score them anyway.  If only one moves, score only that leg.
#        PATCH_CLIPCOUNT (applied to cluster_b's HF.py this tick, CORRECTIONS 63) makes
#        these runs the first to also write the per-COORDINATE *count* n_at_lo /
#        n_at_hi / n_beta, so the fourth denominator -- per (record, coordinate) --
#        becomes available for the first time.  Report it alongside, never instead.
#
#   (X1) THE PRIMARY TEST -- IS THE HEADLINE A BOX ARTEFACT?
#        N_eff/m, weightwise, VARIANCE instrument, STEADY half (the only quotable
#        combination, CORRECTIONS 55/59).  Control = 0.519 (wc5 m3, LO=-30).
#        PASS: both new arms within +-0.10 absolute, i.e. in [0.42, 0.62].
#          --> The headline is not a boundary artefact.  CORRECTIONS 59 stands as
#              written and the N_eff/m sentence needs no box caveat.
#        REFUTATION: either arm outside [0.42, 0.62] AND monotone in the ceiling.
#          --> The number is partly set by the clip.  Every published N_eff/m must
#              then be re-quoted WITH ITS BOX, and the 4-family headline (0.163 /
#              0.481 / 0.552 / 0.305) needs a box-sensitivity column before it can
#              be written down.  Say so plainly; do not retreat to "confounded".
#        WHY +-0.10.  It is 2.6x the measured FLOOR sensitivity (0.038 for a 2.2x
#        deeper box).  A ceiling move of comparable log-extent moving the number by
#        more than 2.6x what the floor move did is a real dependence, not noise.
#
#   (X2) THE PROFILE.  b(steady, k^-b across w/node/lay).  Control b = 0.061.
#        PASS if all three ceilings agree within 0.05 -- the same bar wc5's own W0.3
#        used, and the bar the floor move satisfied (0.081 -> 0.061).
#        REFUTATION: monotone in the ceiling with span > 0.10 across the ladder.
#        SECONDARY to X1: b is a 3-point fit and N_eff/m is not a fit at all.  If b
#        and N_eff/m disagree, report both and claim neither -- cycle 50's C2 found
#        exactly that failure and the pre-registration there bundled them.
#
#   (X3) STABILITY, registered as a REAL OUTCOME and not a hazard footnote.
#        At HI=0.0 a coordinate may reach alpha=1 against an SGDm base with
#        momentum 0.99.  If `collapsed=1`, or plateau falls more than 5 pp below the
#        wc5 control, in >=2 of 3 seeds at any rung: record it as **"the ceiling is
#        load-bearing for stability at that granularity"** -- a positive result about
#        fine-grained meta-optimization, not a wasted arm -- and score X1/X2 on
#        {HI=-4.6052, control} only, reporting b as a SLOPE BETWEEN TWO POINTS
#        rather than as an exponent.  Say which happened.
#
#   RESOLUTION, per CORRECTIONS 33's standing rule, computed BEFORE submission.
#   rho_min is set by T and m and does not depend on the ceiling, so wc5's measured
#   floors carry over unchanged:
#       rung        rho_min (T=1000 steady, 3 seeds)   verdict at ms=1e-3
#       weightwise  ~8.5e-09                           RESOLVES (wc5 measured 8.2e-08)
#       nodewise    ~7.0e-06                           RESOLVES
#       layerwise   ~1.5e-03                           MARGINAL -- wc5's own m3 rung
#                                                      resolved, ml5's lay_m2_s1 did
#                                                      not.  A layerwise null is a
#                                                      BOUND, never an absence.
#
# NOT INCLUDED, deliberately:
#   * ms=1e-2.  Boundary-dominated at BOTH walls (CORRECTIONS 62); it is not a point
#     on the dial and no claim quotes it.  Un-confounding it needs both walls moved
#     at once, which is a different batch and a lower priority than the headline.
#   * blk6.  Excluded from every steady-half fit in ml5 (below its own rho_min in
#     9 of 9 arms).  It would contribute a confound and no number.
#   * A symmetric both-walls arm.  It would vary two fields at once, which is exactly
#     the mistake this batch is correcting.  LO=-30 is held FIXED and measured
#     non-binding; only HI varies.
#
# WHY cluster_b.  Account assignment is BY COMPARATOR (CORRECTIONS 46(6)).  The control
# arm (`wc5-*-m3-*`) and the whole `ml5` ladder are on cluster_b.  PATCH_CLIPCOUNT was
# applied to cluster_b's HF.py only, so cluster_a's copy is unchanged and its fz3/ff5 family
# ladders remain byte-comparable to their own history.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. X0.1-X0.3 above, run not assumed, X0.3 FIRST
#   2. python3 analysis/c51_wideclip.py --selftest   (26/26)  -- all three denominators
#   3. python3 analysis/probe5_window.py --selftest && analysis/neff_instrument.py --selftest
#   4. Window scan on every arm.  Never quote a profile number without its window.
#
# USAGE
#   bash c51_ceiling_ladder.sh            # dry run: prints every sbatch line
#   bash c51_ceiling_ladder.sh --submit   # submits, ONLY if the guards below pass
# =============================================================================
set -u

WS=/cluster/account_b/metaopt                          # cluster_b
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/cl5
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_b
PEND_CAP=40
NJOBS=18
BETA_LO=-30                                        # measured non-binding at ms=1e-3
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c51 CEILING LADDER + PROBE5 (cluster_b) ==="
echo "    LO fixed at $BETA_LO;  HI in {-4.6052, 0.0};  control HI=-2.3026 is wc5-*-m3-*"

if ! grep -q PATCH_PROBE5 "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5 not present in $HF"; exit 1
fi
if ! grep -q PATCH_PROBE5_FIX "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_PROBE5_FIX not present -- unfixed patch crashes at record 500"; exit 1
fi
if ! grep -q PATCH_CLIPCOUNT "$HF" 2>/dev/null; then
  echo "GUARD FAIL: PATCH_CLIPCOUNT not present -- X0.3's per-coordinate denominator"; exit 1
fi
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX + PATCH_CLIPCOUNT present  OK"

python3 -c "import py_compile,sys; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 2: HF.py compiles  OK"

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 3: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
for G in layerwise nodewise weightwise; do
  case $G in
    layerwise)  TAG=lay  ;;
    nodewise)   TAG=node ;;
    weightwise) TAG=w    ;;
  esac
  # cD = ceiling DOWN (alpha_max 0.01);  cU = ceiling UP (alpha_max 1.0, predicted free)
  for HI in -4.6052:cD 0.0:cU; do
    HIV=${HI%:*}; HIT=${HI#*:}
    for S in 0 1 2; do
      RN="cl5-${TAG}-${HIT}-s${S}"
      PD="$SAVE/probe_${TAG}_${HIT}_s${S}"
      CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
        --export="ALL,AUGMENT=1,BETA_CLIP=${BETA_LO}:${HIV},PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
        "$RUNNER"
        --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
        --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
        --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
        --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3
        --num-epochs 20 --stepsize-groups "$G" --seed "$S"
        --save-directory "$SAVE" --run-name "$RN")
      N=$((N+1))
      if [ "$SUBMIT" = "1" ]; then "${CMD[@]}" >/dev/null; else printf '%s\n' "${CMD[*]}"; fi
    done
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo SUBMITTED || echo 'dry run, nothing submitted' )) ----"
echo "reduce with, IN THIS ORDER:"
echo "  1. python3 analysis/c51_wideclip.py               # X0.3 FIRST -- did the ceiling bite?"
echo "  2. python3 analysis/probe5_window.py ../probes_cl5  # X0 null (layerwise), then X2"
echo "  3. python3 analysis/neff_instrument.py ../probes_cl5 # X1, N_eff/m -- THE PRIMARY TEST"
