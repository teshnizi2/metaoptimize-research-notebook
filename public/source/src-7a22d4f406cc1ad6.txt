#!/bin/bash
# =============================================================================
# cGN3_gn_horizon.sh -- `cgn3`: IS THE GROUPNORM RESCUE A RESCUE OR A DELAYED
#   COLLAPSE?  The cvh1 / ciso2 analogue on ResNet18_gn_c100 at cgn2's cell.
#   ONE SUBMISSION, 12 JOBS (4 arms x 3 seeds), 430 EPOCHS, SEEDS {81,82,83},
#   PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.  CORRECTIONS 228.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches `cluster_a`; every guard and scancel line is filtered on `cgn3-` alone.
#
# REGISTERED SCORER: analysis/cGN3_gn_horizon_score.py (+ its synthetic-run
# generator analysis/cGN3_smoke_gen.py), committed BEFORE this batch is
# submitted (RULE 21).  The horizon derivation, the pin gate (cvh1's step-size
# MAGNITUDE form, after the primary), every bar as a FROZEN LITERAL, the branch
# map and the floor-gate table are in the scorer's header.
#     DOCUMENTED INVOCATION:  python3 analysis/cGN3_gn_horizon_score.py <runsdir>
#
# THE NETWORK RUNS FROM cgn1's ISOLATED HARNESS TREE, NOT THE LIVE ONE -- the
# tree cgn1 and cgn2 ran on: $WS/harness_cgn1/cifar10 (bin/cGN1_stage_harness.sh,
# CORRECTIONS 212, VERIFIED unedited by guard 4a; build_network.py POST_SHA
# 9f6e4ec9...c892) and $WS/jobs/run_cifar_cgn1.sh, which differs from
# jobs/run_cifar.sh in EXACTLY its `cd` line (guard 9b).  NO PATCH: nothing is
# added to either tree; guard 3e pins the live harness's hashes and guard 10
# re-reads them after composition.
#
# THE ARMS -- spec strings BYTE-IDENTICAL to cgn2's (guard 4c'' checks cGN2's
# registered SPEC table AND cgn2's own 12 ARGS lines for these four arms).
#   k01  scalar                                                          m=1
#   kL   layerwise                                                       m=62
#   ISO  sets:1-49,51-52,54-58,60-62/layer4.0.bn2.weight,layer4.0.shortcut.1.weight,layer4.1.bn2.weight
#        [59,3]  isolates 50, 53, 59
#   ONE  sets:1-49,51-62/layer4.0.bn2.weight
#        [61,1]  isolates 50
#
# THE HORIZON, DERIVED FROM cgn2's MEASURED DESCENT (the 190 method; scorer header):
#   ISO complement @100: -8.9405 / -8.8005 / -8.9025, tail slopes -0.023..-0.029
#   nats/epoch.  Tail-linear extrapolation (last 50 / 100 records): exact clamp
#   <= 365.1491, r <= 2 <= 335.5036.  E = ceil(365.1491 + W_POST 64.2) = 430.
#   The descent is NOT linear (window slopes slowed then accelerated); long-
#   window linear gives r<=2 at 387-460 and sqrt/log never: UNRESOLVED-NOT-PINNED
#   is REGISTERED AS INFORMATIVE and never voids the primary.
#
# THE 100-EPOCH CONTROL IS IN THE SAME RUNS.  Guard 4h proves on BOTH the tree's
# and the live train.py that num_epochs is only the argparse default and the
# loop bound (no scheduler / total_steps / warmup / environ); guard 4h2 proves
# on the tree's HF.py that SCHED=none's `_sched_factor()` is exactly 1.0 at every
# counter up to 1e7; guard 4h3 refuses if a schedule variable is set in this
# shell (--export=ALL would carry it).
#
# WALLTIME, PARTITIONS, COST -- MEASURED ON THIS NETWORK'S OWN RUNS
#   cgn2 (15 runs, 100 ep, the same L4 / A100-MIG node set as the 7-day
#   partitions): 40:28-48:39 per 100 epochs; worst 29.19 s/epoch.
#   430 epochs at the worst rate = 3.49 h.  WALL=07:00:00 = 58.6 s/epoch =
#   2.01x the worst measured GN rate, NO checkpointing needed.  7 h > gpu-short's
#   4:00:00, so slurm_parts_for_wall DROPS gpu-short (and the 2080 Ti nodes) by
#   arithmetic; no PARTS_NO_SHORT_REASON override.
#   COST: 11.1997 GPU-h / 1,500 run-epochs (cgn2 sacct) x 12 x 430 = 38.53 GPU-h
#   EXPECTED; HARD BOUND = WALL x 12 = 84 GPU-h.
#
# RULE 20.  Every value is ONE shell token of [A-Za-z0-9_.,:/-].  BETA_CLIP and
# PROBE ride the runner's ENV line; PROBE_TENSOR rides the patch's own line.
# analysis/argsline_guard.py IS NOT EDITED.
#
# USAGE   bash bin/cGN3_gn_horizon.sh               # dry run: every guard, every
#         bash bin/cGN3_gn_horizon.sh --dry-run     #   composed line printed; submits nothing
#         bash bin/cGN3_gn_horizon.sh --submit      # the ONLY submitting form
# ENV     CGN3_REGISTERED_COMMIT=<sha>  on a host with no git repo, --submit requires
#                                       the operator to declare the commit whose tree
#                                       was staged (guard 1b); written to PROVENANCE.txt.
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cgn1/cifar10
RUNNER=$WS/jobs/run_cifar_cgn1.sh
LIVERUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cgn3

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=430; CTRL_EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=ResNet18_gn_c100
REFNET=ResNet18_c100
SEEDS="81 82 83"
MST=1e-3
A0=1e-6
NTENS=62
TOTPAR=11220132
G_K01='scalar'
G_KL='layerwise'
G_ISO='sets:1-49,51-52,54-58,60-62/layer4.0.bn2.weight,layer4.0.shortcut.1.weight,layer4.1.bn2.weight'
G_ONE='sets:1-49,51-62/layer4.0.bn2.weight'
ARMS="k01 kL ISO ONE"
NJOBS=12
WALL=07:00:00
# the live harness, pinned (guard 3e before, guard 10 after)
PIN_LIVE_BN=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
PIN_LIVE_HF=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
PIN_LIVE_TRAIN=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
PIN_LIVE_LOAD=b52b58a35dcecd80ce6d8f8033b09c20dad9f4827b808f4ea2a828ce88138c46
PIN_LIVE_RUNNER=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
PIN_TREE_RUNNER=c98e1a568da8c815d7fb55757f81542390ccee4cecf96b2b4f0f8dbfd315c53b
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cGN3_gn_horizon_score.py"
GEN="$REPO/analysis/cGN3_smoke_gen.py"
SC2="$REPO/analysis/cGN2_gn_isolation_score.py"
SELF="$REPO/bin/cGN3_gn_horizon.sh"
TESTBW="$REPO/tests/test_probe_tensor_blockwise_cgn2.py"
TESTB="$REPO/tests/test_probe_tensor_blockwise.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_resnet_gn_c100.py"
PATCHF="$REPO/patches/patch_resnet_gn_c100.py"
STAGEF="$REPO/bin/cGN1_stage_harness.sh"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFPRE="$LIVE/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"

echo "=============================================================="
echo " cgn3 -- is the GroupNorm rescue a rescue or a delayed collapse?"
echo " $DSET / $NET (GroupNorm(32,C) for every BatchNorm2d(C)), cgn2's cell"
echo " AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs (in-run control at $CTRL_EPOCHS); seeds $SEEDS"
echo "   k01  $G_K01"
echo "   kL   $G_KL"
echo "   ISO  $G_ISO"
echo "   ONE  $G_ONE"
echo " 4 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY: RHO = D_ISO@$EPOCHS / D_ISO@$CTRL_EPOCHS, both IN BATCH, IN RUN; pin gate AFTER it, as a stamp"
echo " DOCUMENTED INVOCATION: python3 analysis/cGN3_gn_horizon_score.py <runsdir>"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$GEN" "$SC2" "$SELF" "$TESTBW" "$TESTB" "$TESTT" "$TESTF" "$PATCHF" "$STAGEF" "$LIBF" "$AGF" "$CSVP" \
         "$RUNNER" "$LIVERUNNER" "$HFH" "$HFPRE" "$BNH" "$HARN/train.py" "$LIVE/train.py"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cGN3_gn_horizon_score.py analysis/cGN3_smoke_gen.py bin/cGN3_gn_horizon.sh"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF bin/_lib_guards.sh analysis/argsline_guard.py analysis/cGN2_gn_isolation_score.py 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the three registration files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
  DEL=$(git -C "$REPO" diff HEAD --numstat -- analysis/ 2>/dev/null | awk '$2+0>0{print}')
  [ -z "$DEL" ] && echo "guard 1b2: no uncommitted deletions under analysis/ (RULE 16)" \
               || guard_fail "guard 1b2: analysis/ has uncommitted deletions: $DEL"
else
  if [ -n "${CGN3_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CGN3_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CGN3_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CGN3_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING its premise, calibration and O2 sections ----
SELFT=/tmp/cgn3_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep -E '^  (FAIL|SKIP)|EXCEPTION|Traceback' "$SELFT" | head -20; tail -5 "$SELFT"; }
for PAT in "^  PASS cgn2 PIN_BOUND re-derives" \
           "^  PASS cgn2 PIN_BOUND_R2 re-derives" \
           "^  PASS DECOMPOSITION passes on cgn2's REAL blockwise records" \
           "^  PASS NEGATIVE CONTROL: the SWAPPED partition" \
           "^  PASS cvh1 ISO complement re-derives as PINNED" \
           "^  PASS ciso2-ISO median r re-derives" \
           "^  PASS SYNTHETIC survive_freeze -> RESCUE-SURVIVES" \
           "^  PASS SYNTHETIC survive_free -> RESCUE-SURVIVES" \
           "^  PASS SYNTHETIC collapse -> RESCUE-COLLAPSES" \
           "^  PASS SYNTHETIC dropped -> INCOMPLETE" \
           "^  PASS the stress corpus is NOT VACUOUS" \
           "^  PASS O2 FINAL identical across 5 corpora on near_bar" \
           "^  PASS ZERO non-cgn3 rows carry seed 81, 82 or 83" \
           "^  PASS ISO is BYTE-IDENTICAL to cgn2's registered SPEC" \
           "^  PASS ONE is BYTE-IDENTICAL to cgn2's registered SPEC"; do
  grep -q "$PAT" "$SELFT" || guard_fail "guard 1c2: the selftest line /$PAT/ did not PASS on this host"
done
[ "$FAIL" = 0 ] && echo "guard 1c2: premise (cgn2), calibration (cvh1, ciso2), synthetic scoring, O2 and spec-identity sections RAN and PASSED"
echo "guard 1c: scorer sha256 $(sha "$SC")"
echo "guard 1c: generator sha256 $(sha "$GEN")"

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cgn3-' "$CSVP" && guard_fail "guard 2: cgn3-* already in the CSV" || echo "guard 2: no cgn3-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
gn = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], gn))
raise SystemExit(1 if hit or gn not in ([], ["cgn1"], ["cgn1", "cgn2"]) else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cgn3-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cgn3-*.out on disk" || guard_fail "guard 2b: $NOUT cgn3-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (81|82|83)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 81/82/83" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 81/82/83"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%80 2>/dev/null | awk '{print $1}' | grep -c '^cgn3-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cgn3-* jobs" || guard_fail "guard 2e: sacct shows $NS cgn3-* jobs already"
  NSS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%80 2>/dev/null | awk '{print $1}' | grep -c -E -- '-s8[123]$')
  [ "${NSS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 job names ending -s81/-s82/-s83" || guard_fail "guard 2e: sacct shows $NSS jobs on seeds 81-83"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cgn3-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cgn3-* in squeue" || guard_fail "guard 2f: $NQ cgn3-* already queued"
fi

# ---- guard 3/4a: cgn1's isolated tree is still consistent and sees CIFAR-100 ------
bash "$STAGEF" >/tmp/cgn3_stage.$$ 2>&1 \
  && { echo "guard 4a: bin/cGN1_stage_harness.sh (UNEDITED) VERIFIES the cgn1 tree cgn3 runs from:"; grep -E 'V-|VERIFIED' /tmp/cgn3_stage.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: the cgn1 tree is NOT consistent"; tail -20 /tmp/cgn3_stage.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "9f6e4ec97157311bb3b664509258144235319db88962b9d8c3a4484b1837c892" ] \
  && echo "guard 3b: the tree's build_network.py is the one cgn1 and cgn2 ran on (POST_SHA 9f6e4ec9...c892)" \
  || guard_fail "guard 3b: the tree's build_network.py is not cgn1's POST_SHA"
[ "$(sha "$HFH")" = "$(sha "$LIVE/Optimizers/HF.py")" ] \
  && echo "guard 3c: the tree's HF.py is byte-identical to the live HF.py ($(sha "$HFH" | cut -c1-16)...)" \
  || guard_fail "guard 3c: the tree's HF.py differs from the live HF.py"
[ "$(sha "$HARN/train.py")" = "$(sha "$LIVE/train.py")" ] && [ "$(sha "$HARN/load_data.py")" = "$(sha "$LIVE/load_data.py")" ] \
  && echo "guard 3d: the tree's train.py and load_data.py are byte-identical to the live ones" \
  || guard_fail "guard 3d: the tree's train.py / load_data.py differ from the live ones"
# guard 3e: the LIVE harness is the pinned one, BEFORE anything else happens
LIVEHASH_BEFORE="$(sha "$LIVE/build_network.py") $(sha "$LIVE/Optimizers/HF.py") $(sha "$LIVE/train.py") $(sha "$LIVE/load_data.py") $(sha "$LIVERUNNER") $(sha "$RUNNER")"
[ "$LIVEHASH_BEFORE" = "$PIN_LIVE_BN $PIN_LIVE_HF $PIN_LIVE_TRAIN $PIN_LIVE_LOAD $PIN_LIVE_RUNNER $PIN_TREE_RUNNER" ] \
  && echo "guard 3e: the live harness (build_network c7998883, HF 4732b74a, train 3fea309e, load_data b52b58a3, run_cifar a0d0a1b9) and run_cifar_cgn1 (c98e1a56) are the PINNED files" \
  || guard_fail "guard 3e: the live harness hashes are not the pinned ones: $LIVEHASH_BEFORE"

# ---- guard 4c: the GN patch suite on the tree --------------------------------------
if python3 "$TESTF" --pre "$HARN/build_network.py.pre_gn_c100" --post "$BNH" \
     --hf "$HFH" --csv "$CSVP" >/tmp/cgn3_gntest.$$ 2>&1; then
  echo "guard 4c: tests/test_resnet_gn_c100.py PASSES on the cgn1 tree ($(grep -c '^  PASS' /tmp/cgn3_gntest.$$) PASS, $(grep -c '^  FAIL' /tmp/cgn3_gntest.$$) FAIL)"
else
  guard_fail "guard 4c: tests/test_resnet_gn_c100.py FAILED on the cgn1 tree"; grep '^  FAIL' /tmp/cgn3_gntest.$$ | head -20
fi

# ---- guard 4b: the optimiser carries its patches, proved inert -----------------------
grep -q 'PATCH_NAMESETS' "$HFH" && echo "guard 4b: HF.py carries PATCH_NAMESETS" || guard_fail "guard 4b: HF.py lacks PATCH_NAMESETS"
NPT=$(grep -c 'PATCH_PROBE_TENSOR' "$HFH" 2>/dev/null); NPT=${NPT:-0}
[ "$NPT" = 3 ] && echo "guard 4b: HF.py carries PATCH_PROBE_TENSOR (3 markers)" \
               || guard_fail "guard 4b: HF.py has $NPT PATCH_PROBE_TENSOR markers, the patch inserts exactly 3"
python3 "$TESTT" --pre "$HFPRE" --post "$HFH" --structural-only >/tmp/cgn3_struct.$$ 2>&1 \
   && echo "guard 4b2: structural inertness R0 PASSES on $HFH" \
   || { guard_fail "guard 4b2: structural inertness R0 FAILED"; tail -20 /tmp/cgn3_struct.$$; }

# ---- guard 4e: BLOCKWISE inertness ON ResNet18_gn_c100, on THIS HF and THIS build_network
# cgn3's ISO and ONE specs are byte-identical to cgn2's, so cgn2's log qualifies IF its
# shas are this tree's; a cgn3-own log is accepted first.
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH")
B_OK=0
for BLOG in "$SAVE/inertness_blockwise_gn.log" "$WS/runs/cgn2/inertness_blockwise_gn.log"; do
  if [ -f "$BLOG" ] && grep -q '^ALL PASS$' "$BLOG" && grep -q "^POST_SHA256 $HFSHA" "$BLOG" \
     && grep -q "^BN_SHA256 $BNSHA" "$BLOG" && grep -qF -- "$G_ISO" "$BLOG" && grep -qF -- "$G_ONE" "$BLOG" \
     && grep -q "live $NET: $NTENS parameter tensors" "$BLOG" && grep -q "^SCORER_SPECS_AGREE" "$BLOG"; then
    echo "guard 4e: BLOCKWISE inertness ALL PASS on THIS HF sha and THIS build_network sha, on $NET, for ISO and ONE ($BLOG; $(grep -c '^  PASS' "$BLOG") PASS / $(grep -c '^  FAIL' "$BLOG") FAIL)"
    B_OK=1; break
  fi
done
[ "$B_OK" = 1 ] || guard_fail "guard 4e: no ALL PASS blockwise-on-GN inertness log for HF $HFSHA / build_network $BNSHA -- run:
    mkdir -p $SAVE
    { echo POST_SHA256 $HFSHA; echo BN_SHA256 $BNSHA; python3 $TESTBW --pre $HFPRE --post $HFH \\
        --cifar-dir $HARN --work /tmp/cgn3_bw; } 2>&1 | tee $SAVE/inertness_blockwise_gn.log"

# ---- guard 4c': the literals here, in the scorer, the generator and the stage script agree ----
# ---- guard 4c'': == cGN2's registered SPEC table AND cgn2's OWN ARGS lines ---------------------
python3 - "$SC" "$GEN" "$SC2" "$STAGEF" "$WS/runs" "$G_K01" "$G_KL" "$G_ISO" "$G_ONE" "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$EPOCHS" <<'PYEOF' \
  || guard_fail "guard 4c'/4c'': launcher / scorer / generator / cgn2 literals disagree"
import sys, re, ast, glob, os
sc, gen, sc2, st = (open(p).read() for p in sys.argv[1:5])
runs = sys.argv[5]
mine = dict(zip(("k01", "kL", "ISO", "ONE"), sys.argv[6:10]))
def spec_of(txt):
    m = re.search(r"^SPEC = (\{.*?^\})", txt, re.S | re.M)
    return ast.literal_eval(m.group(1)) if m else None
def lit(txt, name):
    mm = re.search(r"^%s = (.+?)(\s*#.*)?$" % name, txt, re.M)
    return ast.literal_eval(mm.group(1)) if mm else None
bad = []
if spec_of(sc) != mine: bad.append("scorer SPEC")
if spec_of(gen) != mine: bad.append("generator SPEC")
c2 = spec_of(sc2) or {}
if any(c2.get(k) != v for k, v in mine.items()): bad.append("cGN2 SPEC")
if lit(sc, "EPOCHS") != int(sys.argv[14]) or lit(gen, "EPOCHS") != int(sys.argv[14]): bad.append("EPOCHS")
