def sha_bytes(b):

def sha(path):
    with open(path, "rb") as fh:

def avg(v):
    return sum(v) / len(v)


def sdev(v):
    m = avg(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def median(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return float("nan")
    return v[n // 2] if n % 2 else 0.5 * (v[n // 2 - 1] + v[n // 2])


def sgn(x):
    return (x > 0) - (x < 0)

def epoch_line(ln):
    """'Epoch E, Train Accuracy: X %, Test Accuracy: Y %' -> (E, X, Y) or None."""
    if not ln.startswith("Epoch "):
        return None
    parts = ln.split(",")
    if len(parts) != 3:
        return None
    try:
        e = int(parts[0][len("Epoch "):].strip())
        tr = parts[1].strip()
        te = parts[2].strip()
        if not (tr.startswith("Train Accuracy:") and tr.endswith("%") and te.startswith("Test Accuracy:") and te.endswith("%")):
            return None
        x = float(tr[len("Train Accuracy:"):-1].strip())
        y = float(te[len("Test Accuracy:"):-1].strip())
    except ValueError:
        return None
    return e, x, y


def parse_out(path):
    lines = open(path, errors="replace").read().splitlines()
    eps, loose = {}, 0
    for ln in lines:
        t = epoch_line(ln)
        if t is not None:
            eps[t[0]] = (t[1], t[2])
        elif ln.startswith("Epoch"):
            loose += 1
    r = {"eps": eps, "loose": loose, "lines": lines,
         "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
         "tb": any("Traceback (most recent call last)" in ln for ln in lines),
         "args": [ln for ln in lines if ln.startswith("ARGS:")],
         "env": [ln for ln in lines if ln.startswith("ENV:")],
         "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")]}
    for k in [o.split(":")[0] for o in OFF] + list(HOLDS):
        r[k] = [ln for ln in lines if ln.startswith(k)]
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    ok = all(t is not None and math.isfinite(t[0]) and math.isfinite(t[1]) for t in tail)
    r["p5"] = avg([t[1] for t in tail]) if ok else None
    r["t5"] = avg([t[0] for t in tail]) if ok else None
    r["complete"] = sorted(eps) == list(range(EPOCHS)) and r["run_done"] and not r["tb"] and ok
    gpu = None
    for ln in lines:
        s = ln.strip()
        if s.endswith(" MiB") and ", " in s and not s.startswith(("ARGS:", "ENV:", "Epoch")):
            gpu = s.split(",")[0].strip()
            break
    r["gpu"] = gpu
    return r


def args_tokens(line):
    """-> ordered list of (flag, value) and a dict flag -> [values]."""
    seq, cur = [], None
    for t in line[len("ARGS:"):].split():
        if t.startswith("--"):
            seq.append([t[2:], None])
        elif seq:
            seq[-1][1] = t if seq[-1][1] is None else seq[-1][1] + " " + t
    d = {}
    for f, v in seq:
        d.setdefault(f, []).append(v)
    return [tuple(x) for x in seq], d

def vote(r):
    try:
        b2 = float(r["pt_b2"])
        float(r["pt_mp"])
        z, m = r["z_tensor"], r["m_tensor"]
        mom, za = float(r["mom_pre"][0][0]), float(r["z_agg"][0][0])
    except (KeyError, TypeError, IndexError, ValueError):
        return None
    if len(z) != NTENS or len(m) != NTENS or not all(fin(x) for x in z) or not all(fin(x) for x in m):
        return None
    L = [b2 * a + (1.0 - b2) * b for a, b in zip(m, z)]
    app = b2 * mom + (1.0 - b2) * za
    if not (fin(app) and all(fin(x) for x in L)):
        return None
    return L, app


def decomp(recs):
    n = bad = bads = nonf = 0
    for r in recs:
        v = vote(r) if isinstance(r, dict) else None
        if v is None:
            nonf += 1
            continue
        L, app = v
        s, sc = sum(L), sum(abs(x) for x in L)
        n += 1
        bad += not (abs(s - app) <= DECOMP_TOL * sc or abs(s - app) <= 1e-30)
        bads += sgn(s) != sgn(app) and abs(app) > DECOMP_TOL * sc
    return n, bad, bads, nonf


def a_of(r):
    b = r.get("beta") if isinstance(r, dict) else None
    if not isinstance(b, list) or not b or not all(fin(x) for x in b):
        return None
    return median([math.exp(x) for x in b])


def trace(recs, gamma):
    q, qo, dz, late, turn = [], [], [], [], None
    for r in recs:
        a = a_of(r)
        if a is None or not isinstance(r.get("step"), int) or isinstance(r.get("step"), bool):
            continue
        v = 1.0 - gamma * (1.0 - WD * a)
        q.append(v)
        dz.append(WD * a)
        if ONSET_LO <= r["step"] < ONSET_HI:
            qo.append(v)
        if r["step"] >= 40000:
            late.append(min(r["beta"]) <= FLOOR_BETA)
        if turn is None and "z_agg" in r:
            vv = vote(r)
            if vv is not None and sgn(vv[1]) == 1:
                turn = r["step"]
    nan = float("nan")
    return {"q_onset": median(qo), "q_peak": max(q) if q else nan, "q_min": min(q) if q else nan,
            "dose_peak": max(dz) if dz else nan, "turn": turn, "floor_share": (sum(late) / len(late)) if late else nan,
            "n_onset": len(qo)}


def state(Lv, kv):
    L, k = avg(Lv), avg(kv)
    if L < REF_MIN or max(Lv) - min(Lv) > DIVERGED_BAR:
        return "UNREADABLE"
    n = sum(1 for x in kv if x <= R50 * L)
    if 1 <= n <= len(kv) - 1:
        return "SPLIT"
    if k <= R50 * L:
        return "COLLAPSE"
    if L - k < GAP_BAR:
        return "NOGAP"
    return "PARTIAL"

def tail_ols(eps):
    xs = [e for e in range(80, 100) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = avg(xs), avg(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def floor_sigma(rows, net):
    cells = {}
    for r in rows:
        if not (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
                and (r.get("plateau5") or "").strip()):
            continue
        if not (r.get("epochs_requested") == "100" and r.get("augment") == "1" and r.get("beta_clip") == "-15:-2.3026"
                and r.get("meta_stepsize") == "1e-3" and r.get("alpha0") == "1e-6" and r.get("batch_size") == "100"
                and r.get("network") == net and r.get("dataset") == "CIFAR100"):
            continue
        cells.setdefault(tuple(r.get(c, "") for c in CK), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) >= 2:
            m = avg(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc

def compare(R, log):
    got = open(log, errors="replace").read().splitlines()
    while got and got[-1] == "":
        got.pop()
    bad = []
    n = max(len(R), len(got))
    for i in range(n):
        if i >= len(R) or i >= len(got):
            bad.append((i + 1, R[i][1] if i < len(R) else "<none>", got[i] if i < len(got) else "<none>"))
            continue
        kind, want = R[i]
        g = got[i]
        if kind == "=":
            ok = g == want
        else:
            head, tail = want.split("|", 1)
            mid, how = tail.split(" (", 1)
            ok = g.startswith(head) and g.endswith(mid + " (" + how)
            if ok:
                p = g[len(head):len(g) - len(" (" + how)]
                ok = p.startswith("/") and p.endswith(mid)   # host paths may contain spaces (the Mac's do)
        if not ok:
            bad.append((i + 1, want, g))
    return len(got), bad

    chk(all(runs[k]["job"] == JOBS[k] for k in runs), "every .out's job id == 301.11's (5080645 + 6*seed_idx + arm_idx)")
    for a in ARMS:
        toks = sorted(set(args_tokens(runs[(a, s)]["args"][0])[1]["gamma"][0] for s in SEEDS))
        wds = sorted(set(args_tokens(runs[(a, s)]["args"][0])[1]["weight-decay-base"][0] for s in SEEDS))
        grs = sorted(set(args_tokens(runs[(a, s)]["args"][0])[1]["stepsize-groups"][0] for s in SEEDS))
        chk(toks == [TAB[a][2]] and wds == [WD_TOKEN] and grs == [TAB[a][1]],
            "%s: --gamma %s, --weight-decay-base %s, --stepsize-groups %s on all 3 ARGS lines" % (a, toks, wds, grs))
    chk(all(len(args_tokens(runs[k]["args"][0])[0]) == 20 for k in runs), "every ARGS line carries exactly 20 flags, in design order")
    chk(F.get("probe_dir_ok", False), "every ENV line's PROBE_DIR is the run's own probe directory")
    chk(all(runs[k]["loose"] == 0 for k in runs), "no malformed Epoch line in any .out")
    chk(all(runs[k]["tb"] is False for k in runs), "no traceback in any .out")
    mt = F["manifest"].decode("utf-8", "replace").splitlines()
    marms = [ln for ln in mt if ln.startswith("ARM ")]
    chk(marms == ["ARM %s CELL %s GRAIN %s GAMMA %s" % (x, TAB[x][0], TAB[x][1], TAB[x][2]) for x in ARMS],
        "manifest ARM lines == the design table (cell, grain, gamma)")
    margs = dict((ln.split()[1], ln.split(" ", 2)[2]) for ln in mt if ln.startswith("ARGS "))
    chk(all(margs.get(x) == " ".join("--%s %s" % (f, (v if f not in ("seed", "save-directory") else
                                                       ("0" if f == "seed" else "<save>")))
                                     for f, v in expected_args(x, 0)) for x in ARMS),
        "manifest ARGS lines == this file's rebuilt flags (seed 0, save <save>)")
    prov = F.get("prov", {})
    chk(prov.get("REGISTERED_COMMIT") == REGISTERED_COMMIT, "PROVENANCE REGISTERED_COMMIT == 5ac1dd79...")
    chk(prov.get("CORPUS_SHA256") == REG_CORPUS_SHA and prov.get("CORPUS_EXCLUSIONS_SHA256") == REG_EXCL_SHA,
        "PROVENANCE corpus / exclusion shas == registered")
    chk(prov.get("GAMMAS") == "G1=1 GM=0.999685 GP=0.99941 WD_BASE 5e-4", "PROVENANCE GAMMAS line == 301.4's tokens")

    if F["final"] in ("INCOMPLETE", "HARNESS-UNSOUND"):
        P("")
        P("rebuild stopped at %s" % F["final"])
    else:
        V, M, se = F["V"], F["M"], F["se"]
        P("")
        P("[3] PROBE RECORDS, raw (this file's own reader)")
        chk(all(len(F["recs"][k]) == N_RECORDS for k in F["recs"]), "500 records on every run")
        chk(all([r["step"] for r in F["recs"][k]] == list(range(0, 50000, 100)) for k in F["recs"]),
            "record steps are exactly 0, 100, ..., 49,900 on every run")
        for a in ARMS:
            ns = [p["n_onset"] for p in F["per_seed"][a]]
            chk(ns == [12, 12, 12], "%s: 12 onset-window records (7,500 <= step < 8,700) per seed" % a, str(ns))
        # the realised trace factor is gamma*(1 - wd*a): a gamma-1 arm's q equals its dose; a gamma<1 arm's q >= 1-gamma
        for a in ARMS:
            g = float(TAB[a][2])
            qmins = [p["q_min"] for p in F["per_seed"][a]]
            chk(all(q >= (1 - g) - 1e-15 for q in qmins), "%s: every record's q >= 1 - gamma = %.6g (the arm's own decay only "
                "shortens the trace)" % (a, 1 - g), "min q %s" % "/".join("%.4e" % q for q in qmins))

        P("")
        P("[4] THE NOISE FLOOR, re-derived from the stage corpus by this file's own filter and pooling (selftest C's claim)")
        s_f, df_f, nc_f = floor_sigma(F["kept_rows"], NET)
        P("  filtered: sigma %.12f  df %d  cells %d   (frozen %.12f, df %d; demo value 0.636566)"
          % (s_f, df_f, nc_f, SIGMA_PRIOR, SIGMA_PRIOR_DF))
        chk(abs(s_f - SIGMA_PRIOR) < 1e-12 and df_f == SIGMA_PRIOR_DF, "frozen SIGMA_PRIOR re-derived to 1e-12 at df 274")
        s_n, df_n, _ = floor_sigma(F["rows_all"], NET)
        P("  unfiltered (the OPERATIONS 36 trap, disclosure): sigma %.6f  df %d" % (s_n, df_n))
        P("  in-batch pooled sigma %.6f (df 12) %s the frozen floor -> SIGMA_USED %.6f, SE %.6f"
          % (F["sig_in"], "<" if F["sig_in"] < SIGMA_PRIOR else ">=", F["su"], se))

        P("")
        P("[5] BOUNDS AND FLOOR / CEILING AUDIT (164.6) -- what bounds the verdict, printed before any reading")
        for c in CELLS:
            k, L = SCAL[c], LAYR[c]
            P("  cell %s: layerwise %.4f (seed range %.4f, bar 5.0; REF_MIN 55.0 -> margin %+.4f); scalar %.4f; COLLAPSE bar "
              "0.5 x L = %.4f (scalar margin above it %+.4f); NOGAP bar G < 10: G %+.4f (margin to the bar %+.4f)"
              % (c, M[L], max(V[L]) - min(V[L]), M[L] - REF_MIN, M[k], R50 * M[L], M[k] - R50 * M[L], F["G"][c],
                 GAP_BAR - F["G"][c]))
        worst = min(min(V[a]) for a in ARMS)
        best = max(max(V[a]) for a in ARMS)
        P("  every seed of every arm lies in [%.4f, %.4f]: %.2f pp above the 15 pp floor gate, %.2f pp below the 90 pp ceiling"
          % (worst, best, worst - FLOOR_MIN, CEIL_MAX - best))
        P("  NOGAP is a BOUND: it says |L - k| < 10 pp in that cell (and L - k may be negative), never 'the grains are equal'.")
        P("  late beta-floor share (scalar arms; >= 0.5 would stamp BETA-FLOOR): %s"
          % ", ".join("%s %s" % (a, f4(F["tr"][a]["floor_share"], 3)) for a in ("G1S", "GMS", "GPS")))
        P("  RHO_D (decay dose peak / the collapsing source arm's): %s -- the dose is ABSENT by construction and measured so"
          % ", ".join("%s %.4f" % (c, F["RHO_D"][c]) for c in CELLS))

        P("")
        P("[6] LEAVE-ONE-SEED-OUT (DESCRIPTIVE, not a gate): the cell state with each seed pair dropped")
        for c in CELLS:
            k, L = SCAL[c], LAYR[c]
            outs2 = []
            for i, s in enumerate(SEEDS):
                Lv = [x for j, x in enumerate(V[L]) if j != i]
                kv = [x for j, x in enumerate(V[k]) if j != i]
                outs2.append("drop s%d: G %+.4f -> %s" % (s, avg(Lv) - avg(kv), state(Lv, kv)))
            P("  cell %s: %s" % (c, "; ".join(outs2)))
        P("  per-seed within-seed gaps L - k: %s" % "; ".join(
            "%s %s" % (c, " / ".join("%+.4f" % (runs[(LAYR[c], s)]["p5"] - runs[(SCAL[c], s)]["p5"]) for s in SEEDS))
            for c in CELLS))

        P("")
        P("[7] THE EXCLUSION ROWS OWED (results/CORPUS-EXCLUSIONS.tsv full row form; PROPOSED, not applied here)")
        P("  run\tjob_id\tbatch\tarm\tlooks_like\tintervention\twitness\tregistered_at\treason")
        for s in SEEDS:
            for a in ARMS:
                r = runs[(a, s)]
                wd = args_tokens(r["args"][0])[1]["weight-decay-base"][0]
                P("  %s-%s-s%d\t%d\t%s\t%s\t%s (granularity %s)\t--weight-decay-base %s\tARGS_WD_BASE: weight-decay-base=%s\t"
                  "CORRECTIONS 301\tbase-optimiser CLI flag not carried by any CSV column (ARGS-value kind, CORRECTIONS 263)"
                  % (PREFIX, a, s, r["job"], PREFIX, a, a, TAB[a][1], wd, wd))
        chk(all(args_tokens(runs[k]["args"][0])[1]["weight-decay-base"] == [WD_TOKEN] for k in runs),
            "18 rows, each witnessed by its own ARGS line (weight-decay-base=5e-4, once)")
        chk(all(args_tokens(runs[k]["args"][0])[1]["momentum-param-base"] == ["0.99"] for k in runs),
            "no run deviates on a second ARGS kind (momentum-param-base 0.99 = the standard)")

        P("")
        P("[8] THE REBUILT FINAL")
        P("  " + F["final"])

    P("")
    P("[9] LINE-BY-LINE AGREEMENT with each scorer stdout (%d lines rebuilt)" % len(R))
    ok_all = True
    for i, lg in enumerate(opt.against):
        n, bad = compare(R, lg)
        ok_all = ok_all and not bad
        chk(not bad, "scorer log #%d: %d lines, %d rebuilt, %d differ (2 host-path lines matched by suffix)"
            % (i + 1, n, len(R), len(bad)))
        for ln, w, g in bad[:20]:
            P("      line %d\n        rebuilt: %s\n        scorer : %s" % (ln, w, g))
    if not opt.against:
        P("  (no --against log given)")
    P("")
    P("ATTACK: %d PASS / %d FAIL" % (ATT["PASS"], ATT["FAIL"]))
    sys.stdout.write("\n".join(OUT) + "\n")
    return 0 if ATT["FAIL"] == 0 else 1
