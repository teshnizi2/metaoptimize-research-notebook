#!/bin/bash
# =============================================================================
# Cycle 53 -- DOES THE MEASUREMENT SURVIVE A DIFFERENT BASE OPTIMIZER?
# cluster_a.  6 jobs.  Tag `bo6-*`.
#
# OPEN QUESTION IT ANSWERS, and it is the first question a referee asks about
# external validity.
#
#   CORRECTIONS 59's headline -- "at the adapted operating point per-weight
#   meta-gradients carry 16%-55% of the independent information their count
#   implies" -- and CORRECTIONS 70's nodewise minimum are BOTH measured under
#   exactly one base optimizer: **SGDm**.  The CSV, counted this tick:
#
#       base   meta   granularity        runs
#       SGDm   Lion   weightwise          185
#       SGDm   Lion   nodewise            125
#       SGDm   Lion   layerwise           643
#       AdamW  Lion   weightwise            3   <-- pre-PROBE5, no probe files
#       AdamW  Lion   nodewise              0   <-- **nothing at all**
#
#   Every N_eff/m the campaign has ever quoted is SGDm/Lion.  And the CLAIM the
#   nodewise minimum is being pointed at is about the ADAM FAMILY:
#
#       "the Adam-mini / Adalayer / SGG line places its blocks at nodewise-or-
#        coarser on the premise that within-block averaging recovers independent
#        information, and at the operating point our headline is quoted at, that
#        is the worst available partition."  (CORRECTIONS 70)
#
#   Adam-mini and Adalayer partition an **Adam** second-moment.  If the retained
#   fraction is a property of SGDm's update rather than of the network, that
#   sentence cannot be written at all.  Nobody has checked.  6 jobs check it.
#
# WHAT THIS BATCH IS NOT.  It is NOT a performance comparison and no plateau from
# it may be quoted as "AdamW-base beats/loses to SGDm-base": the meta-stepsize
# and alpha0 are carried over from the SGDm ladder unchanged and are therefore
# UNTUNED for AdamW.  It is a measurement of a CORRELATION STATISTIC under a
# different base update, and the arms only need to train and be box-free for that
# statistic to mean what it means.  V0.4 below is exactly that gate and nothing
# more.
#
# THE DESIGN.  Byte-identical to `cl5-*-cU-*` except the base optimizer:
#   R18/CIFAR-10, **AdamW** base, Lion meta, ms=1e-3, alpha0=1e-3, AUGMENT=1,
#   BETA_CLIP=-30:0.0, 20 epochs, rungs w/node/lay, seeds 0-1.
#   AdamW's base hyper-parameters are the repo defaults its own branch requires
#   (`required_attributes_base = ['normalizer_param','momentum_param','weight_decay']`):
#   --normalizer-param-base 0.999, --momentum-param-base 0.9, --weight-decay-base 0.1.
#   NOTE the momentum differs from the SGDm arm's 0.99 ON PURPOSE: 0.9 is AdamW's
#   beta1 and 0.99 is SGDm's heavy-ball coefficient.  They are different
#   parameters that share a flag name.  Copying 0.99 across would be matching the
#   STRING and not the ALGORITHM.
#
# PRE-REGISTERED PREDICTIONS -- score against these, not against what lands.
#
#   (V0) VALIDITY GATE.  n_records == 2000; n_beta byte-matches the SGDm ladder
#        (w 11,173,962 / node 14,420 / lay 62 -- the partition is a property of
#        the NETWORK, so a mismatch means the network changed); beta MOVED, 6/6.
#
#   (V0.3) THE BOX-FREE GATE, SCORED FIRST, and PER SEED (CORRECTIONS 72).
#        < 5% of records at either guard on EVERY seed, from beta_true_min/max
#        AND PATCH_CLIPCOUNT's per-coordinate counts.
#        THIS ONE IS NOT EXPECTED TO PASS TRIVIALLY and that is why it is here:
#        AdamW normalises its update, so alpha means something different and the
#        beta trajectory has no reason to match the SGDm arm's.  If an arm binds,
#        report first_hi and its epoch; that arm's V1 is UNINTERPRETABLE, and the
#        batch is re-run at a raised ceiling rather than reported.
#
#   (V0.4) THE TRAINS-AT-ALL GATE, SCORED SECOND.  plateau > 40.0 and `collapsed`
#        false on every seed and rung.  An untuned ms on a normalised base update
#        can simply diverge; a correlation statistic computed on a diverged run is
#        not a measurement of anything.  A FAILURE here does NOT refute V1 -- it
#        voids the arm and says the ladder needs its own ms, which would be the
#        next batch rather than a result.
#
#   (V1) THE PRIMARY TEST -- DOES THE HEADLINE RANGE SURVIVE?
#        N_eff/m, weightwise, VARIANCE instrument, STEADY half.
#        Reference: the published BOX-FREE 4-family range **0.157 - 0.547**
#        (R10 0.1571 / C100 0.3124 / R18 0.5045 / R34 0.5474, CORRECTIONS 68).
#        PASS: the AdamW-base R18 value lands INSIDE [0.157, 0.547].
#          --> the retained fraction is a property of the network and the task,
#              not of SGDm, and CORRECTIONS 59 may be written without a base
#              optimizer caveat.
#        REFUTATION: it lands OUTSIDE that range.
#          --> the 16%-55% range is SGDm-specific and every quotation of it needs
#              a base-optimizer column.  That is a real narrowing of the headline
#              and it must be written as one.
#        NO DIRECTION IS REGISTERED, deliberately.  I have no mechanism that
#        predicts one, and inventing a direction I cannot justify in order to
#        collect a confirmation is the failure mode CORRECTIONS 68 recorded when
#        uc5's "unpinning should move N_eff/m UP" story passed its bar while its
#        stated mechanism did not move the number at all.
#
#   (V2) THE SHAPE -- THE ONE THAT DECIDES WHETHER CORRECTIONS 70 CAN CITE
#        ADAM-MINI AT ALL.  argmin of N_eff/m over {lay, node, w} at ms=1e-3.
#        SGDm control (cl5 cU, n=3): lay 0.7258 / node 0.4016 / w 0.5045,
#        argmin **node**, gap 0.103 -- decided.
#        PREDICTION: **argmin stays `node`** under AdamW base.
#          --> CORRECTIONS 70 transfers to the family the Adam-mini / Adalayer /
#              SGG line actually lives in, and the sentence about where those
#              methods place their blocks becomes defensible.
#        REFUTATION: argmin is `w` or `lay` with a gap > 0.02.
#          --> the nodewise minimum is an SGDm phenomenon.  CORRECTIONS 70 keeps
#              its ms scope AND gains a base-optimizer scope, and the Adam-mini
#              sentence must be deleted rather than hedged.
#        UNDECIDED BRANCH: gap <= 0.02 between the two smallest -> say UNDECIDED
#        and buy a third seed, exactly as ns5/ns6 did (CORRECTIONS 73).
#
#   (V3) ACCURACY, reported regardless and NOT a test.  See "WHAT THIS BATCH IS
#        NOT" above: ms and alpha0 are untuned for AdamW, so no plateau from this
#        batch may be compared to an SGDm plateau as an optimizer comparison.
#
#   RESOLUTION, per CORRECTIONS 33, computed BEFORE submission.  rho_min depends
#   on T and m, not on the base update, so the cl5 floors carry over:
#       weightwise  ~8.5e-09   RESOLVES
#       nodewise    ~7.0e-06   RESOLVES
#       layerwise   ~1.5e-03   MARGINAL at m=62.  V2 is an argmin over three
#                              rungs, so an unresolved layerwise rung makes the
#                              argmin UNDECIDABLE for that arm rather than
#                              defaulting it to one of the other two.  Say which.
#
# NOT INCLUDED, deliberately:
#   * A meta-optimizer sweep.  One axis at a time; the meta is held at Lion so a
#     V1/V2 failure is attributable.  Adam-meta is the obvious next batch and is
#     NOT funded until this one scores.
#   * Other families.  R18 is where every control on this axis already lives.
#   * An ms sweep for AdamW.  If V0.4 fails, that becomes the next batch with its
#     own registration; guessing a second ms now would spend 6 more jobs on a
#     contingency that has not happened.
#   * A third seed.  n=2 buys the argmin (V2's control gap is 0.103) and the
#     UNDECIDED branch above says what to do if the gap comes back small.
#
# WHY cluster_a.  cluster_b is carrying `br6-*` (12 jobs) and cluster_a is otherwise idle.
# This batch's comparator (`cl5-*-cU-*`) is on cluster_b, so CORRECTIONS 46(6)'s
# by-comparator rule is BROKEN DELIBERATELY here and said so: the comparison is
# via N_eff/m computed by a shared selftested instrument on probe files, not via
# raw accuracy, and both accounts' HF.py are byte-identical after PATCH_CLIPCOUNT
# (CORRECTIONS 63).  **If V0's n_beta byte-match fails, that assumption is wrong
# and the batch is VOID** -- the same condition ns5 carried for the same reason.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. V0, V0.3, V0.4 above, run not assumed, V0.3 FIRST and PER SEED
#   2. python3 analysis/c53_score.py --selftest        (14/14)
#   3. python3 analysis/neff_instrument.py --selftest  (23/23)
#   4. python3 analysis/probe5_window.py --selftest    (41/41)
#   5. Window scan on every arm.
#
# USAGE
#   bash c53_base_optimizer.sh            # dry run
#   bash c53_base_optimizer.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

WS=/cluster/account_a/metaopt                   # cluster_a
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/bo6
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_a
PEND_CAP=40
NJOBS=6
BOX=-30:0.0
EPOCHS=20
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c53 BASE-OPTIMIZER GENERALITY: AdamW base, Lion meta (cluster_a) ==="
echo "    BETA_CLIP=$BOX  epochs=$EPOCHS  seeds 0-1  control = cl5-*-cU-* (SGDm base)"

for P in PATCH_PROBE5 PATCH_PROBE5_FIX PATCH_CLIPCOUNT; do
  grep -q "$P" "$HF" 2>/dev/null || { echo "GUARD FAIL: $P not present in $HF"; exit 1; }
done
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX + PATCH_CLIPCOUNT present  OK"

python3 -c "import py_compile; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 2: HF.py compiles  OK"

# GUARD 3 -- AdamW's base branch must EXIST and must require exactly the three
# attributes this batch passes.  The uc5 lesson (CORRECTIONS 71) generalised: an
# unchecked assertion about the code in a batch header is how 6 jobs die at step 0.
grep -q "elif self.args_base\['alg'\] == 'AdamW'" "$HF" || {
  echo "GUARD FAIL: no AdamW base branch in HF.py"; exit 1; }
grep -q "required_attributes_base = \['normalizer_param', 'momentum_param', 'weight_decay'\]" "$HF" || {
  echo "GUARD FAIL: AdamW's required_attributes_base is not the triple this batch passes"; exit 1; }
echo "guard 3: AdamW base branch present and requires normalizer/momentum/weight_decay  OK"

# GUARD 4 -- the granularity rungs must be the PATCHED ones, or 'nodewise' and
# 'weightwise' silently fall through to blockwise and this measures nothing.
grep -q "PATCH_GRANULARITY" "$HF" || {
  echo "GUARD FAIL: PATCH_GRANULARITY absent -- nodewise/weightwise would fall through"; exit 1; }
echo "guard 4: PATCH_GRANULARITY present  OK"

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
FAIL=0
for G in layerwise nodewise weightwise; do
  case $G in
    layerwise)  TAG=lay  ;;
    nodewise)   TAG=node ;;
    weightwise) TAG=w    ;;
  esac
  for S in 0 1; do
    RN="bo6-${TAG}-adw-s${S}"
    PD="$SAVE/probe_${TAG}_adw_s${S}"
    # --momentum-param-base 0.9 is AdamW's beta1, NOT a copy of the SGDm arm's
    # 0.99 heavy-ball coefficient.  Different parameters, shared flag name.
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
      --export="ALL,AUGMENT=1,BETA_CLIP=$BOX,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
      "$RUNNER"
      --optimizer HF --alg-base AdamW --normalizer-param-base 0.999
      --momentum-param-base 0.9 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
      --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
      --max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-3
      --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    if [ "$SUBMIT" = "1" ]; then
      if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
    else
      N=$((N+1)); printf '%s\n' "${CMD[*]}"
    fi
  done
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
echo "reduce with, IN THIS ORDER:"
echo "  1. V0.3 box-free gate, PER SEED, on ../probes_bo6"
echo "  2. V0.4 trains-at-all: plateau > 40 from results/all_runs.csv, per seed"
echo "  3. V1  neff_instrument.py ../probes_bo6   # vs the 0.157-0.547 box-free range"
echo "  4. V2  the argmin -- the one that decides the Adam-mini sentence"
