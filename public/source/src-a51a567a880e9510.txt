#!/bin/bash
# =============================================================================
# cVK1_vgg_cut_ladder.sh -- `cvk1`: THE CUT-POSITION LADDER [k, 26-k] ON
#   VGG11_bn_c100 / CIFAR-100.  ONE SUBMISSION, 27 JOBS (9 arms x 3 seeds),
#   100 EPOCHS, SEEDS {55,56,57}, PROBE=0.
#
# REGISTERED, NOT SUBMITTED.  Default is a DRY RUN; `--submit` is the ONLY
# submitting form, and it is the operator's call.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the shared `cluster_a` account; every guard and scancel line is filtered
# on the `cvk1-` prefix alone.
#
# REGISTERED SCORER: analysis/cVK1_vggcut_score.py, committed BEFORE any cvk1
# run exists (RULE 21).  The prediction, the accounts, the floor table, the
# primary (the ARGMAX LOCATION), every bar in pp and SE and the branch map are
# frozen there.  DOCUMENTED INVOCATION, one argument:
#     python3 analysis/cVK1_vggcut_score.py <runsdir>
#
# -----------------------------------------------------------------------------
# THE QUESTION.  On ResNet18_c100 the ladder [k, 62-k] peaks at k* = 49 --
# the last cut before the first nominated carrier (tensor 50).  VGG11_bn_c100
# has ONE nominated carrier, bn8.weight at 23.  If that is not a coincidence,
# VGG's ladder [k, 26-k] peaks at k* = 22.  On ResNet, 49 is ALSO the
# parameter-balance argmin (cpk1's winning rival); on VGG the balance point is
# 18, so this batch separates the two for the first time.
#
# THE ARMS (all m = 2 sweep arms compose through the LIVE
# HF.polish_the_stepsize_groups; guard 4 proves it and writes the manifest):
#   k01  scalar       floor anchor, IN BATCH
#   kL   layerwise    ceiling anchor, IN BATCH
#   k13  [13,13]      left edge (sparse)                   junction conv5|bn5
#   k16  [16,10]      A2's peak (balance + parity)         junction conv6|bn6
#   k19  [19,7]       the previous junction                conv7|bn7
#   k20  [20,6]       after bn7.weight (non-carrier scale)
#   k21  [21,5]       after bn7.bias; A4's peak (20.55)
#   k22  [22,4]       THE PREDICTION: last cut before bn8  conv8|bn8
#   k23  [23,3]       right flank: bn8 in the big group
# 19..23 is consecutive: the class sequence of cpk3's ResNet window 46..50.
#
# WHY NOT CHEAPER.  Drop k23 and 22 is the grid's edge, where PEAK and a
# monotone rise are the same measurement.  Drop k13 and 16 is an edge.  Drop
# k21 and a peak "at 22" cannot be told from one at 21.  k20 is the one arm not
# strictly needed for the argmax; it costs 3 jobs and is the only VGG
# measurement of a NON-carrier BN scale's step inside the class-matched window.
#
# WALLTIME -- MEASURED on this cell's predecessor (cvg1, 6 runs, 100 epochs):
#   L4 19-22 min, A100 28 min, 2080 Ti 61-63 min (worst observed);
#   corpus-worst 2080 Ti rate 70.8 s/epoch (CORRECTIONS 152) = 1.97 h.
# WALL=03:00:00 is 1.52x the corpus-worst and 2.86x cvg1's worst, and fits
# gpu-short's 4:00:00 MaxTime, so slurm_parts_for_wall ADDS gpu-short.
# COST: 27 jobs.  ~35 min/job mixed pool -> 15.8 GPU-h; cvg1-worst 63 min ->
#       28.4 GPU-h; corpus-worst -> 53.1 GPU-h; HARD BOUND 27 x 3 h = 81 GPU-h.
#
# RULE 20.  "[k,26-k]" is a bash bracket GLOB, so it is SINGLE-QUOTED inside
# the eval'd command (the cK1 precedent) and argv reaches train.py literally.
# analysis/argsline_guard.py IS NOT EDITED.  BETA_CLIP and PROBE are ENV
# variables audited from the runner's ENV line.
#
# USAGE   bash bin/cVK1_vgg_cut_ladder.sh              # DRY RUN, submits NOTHING
#         bash bin/cVK1_vgg_cut_ladder.sh --dry-run    # identical, explicit
#         bash bin/cVK1_vgg_cut_ladder.sh --submit     # the only submitting form
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)  SUBMIT=1 ;;
  ""|--dry-run) SUBMIT=0 ;;
  *) echo "usage: $0 [--dry-run|--submit]"; exit 2 ;;
esac
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/cvk1
CIF=${CIFAR10_DIR:-$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10}
HFLIVE="$CIF/Optimizers/HF.py"
BNLIVE="$CIF/build_network.py"

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=0
DSET=CIFAR100
NET=VGG11_bn_c100
SEEDS="55 56 57"
MST=1e-3
A0=1e-6
NTENS=26
TOTPAR=9274532
K_LIST="13 16 19 20 21 22 23"
# ---- the registered specs, LITERAL.  guard 4c' proves they equal the scorer's.
G_K01='scalar'
G_KL='layerwise'
G_K13='[13,13]'
G_K16='[16,10]'
G_K19='[19,7]'
G_K20='[20,6]'
G_K21='[21,5]'
G_K22='[22,4]'
G_K23='[23,3]'
ARMS="k01 kL k13 k16 k19 k20 k21 k22 k23"
NJOBS=27
WALL=03:00:00
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

echo "=============================================================="
echo " cvk1 -- THE CUT-POSITION LADDER ON VGG: IS k* THE LAST CUT BEFORE THE CARRIER?"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"
for A in $ARMS; do eval "echo \"   $A  \${G_$(echo $A | tr a-z A-Z)}\""; done
echo " 9 arms x 3 seeds = $NJOBS jobs, ONE submission"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY: the ARGMAX LOCATION of plateau5 over the 7 cuts; predicted k* = 22"
echo " DOCUMENTED INVOCATION: python3 analysis/cVK1_vggcut_score.py <runsdir>"
echo "=============================================================="

# ---- guard 0: the corpus ----------------------------------------------------
CSVP="$REPO/results/all_runs.csv"
[ -f "$CSVP" ] || guard_fail "guard 0: no CSV at $CSVP"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines)"

# ---- guard 1: the registered scorer exists, is committed, and selftests -----
SC="$REPO/analysis/cVK1_vggcut_score.py"
PATCHV="$REPO/patches/patch_vggbn.py"
TESTV="$REPO/tests/test_vggbn.py"
TESTT="$REPO/tests/test_probe_tensor.py"
for f in "$SC" "$PATCHV" "$TESTV" "$TESTT"; do
  [ -f "$f" ] || guard_fail "guard 1a: missing $f"
done
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  DIRTY=$(git -C "$REPO" status --porcelain -- \
      analysis/cVK1_vggcut_score.py bin/cVK1_vgg_cut_ladder.sh 2>/dev/null)
  [ -z "$DIRTY" ] && echo "guard 1b: the scorer AND this script are committed (RULE 21)" \
                  || guard_fail "guard 1b: uncommitted -- RULE 21 requires them committed BEFORE any run exists: $DIRTY"
  DEL=$(git -C "$REPO" diff HEAD --numstat -- analysis/ 2>/dev/null | awk '$2+0>0{print}')
  [ -z "$DEL" ] && echo "guard 1b2: no uncommitted DELETIONS under analysis/ (RULE 16)" \
               || guard_fail "guard 1b2: analysis/ has uncommitted deletions: $DEL"
else
  echo "guard 1b: NOTE -- no git repo at $REPO on this host; RULE 21 is verified on"
  echo "          the committing host by wall clock against the earliest sacct Submit."
fi
ST=/tmp/cvk1_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$ST" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$ST") checks, $(grep -c '^  FAIL' "$ST") failures, $(grep -c '^  SKIP' "$ST") skipped)" \
  || { guard_fail "guard 1c: scorer selftest FAILED"; grep '^  FAIL' "$ST" | head -20; }
# guard 1c2: the three data-bearing premise sections must have RUN here.
grep -q "^  PASS k\* = 49 IS the last cut before the first carrier" "$ST" \
  && grep -q "^  PASS THE CONFOUND: on ResNet" "$ST" \
  && echo "guard 1c2: the RESNET PREMISE section RAN on this host's raw cpk .out files" \
  || guard_fail "guard 1c2: the ResNet premise section did not run (cpk*.out / cdep1 manifest not under $WS/runs)"
grep -q "^  PASS PREFIX-SIGN:" "$ST" \
  && echo "guard 1c2: the PREFIX-SIGN section RAN on this host's cvg1 probe records" \
  || guard_fail "guard 1c2: the prefix-sign section did not run"
grep -q "^  PASS PRED\[A1\] re-derives from cpk1's raw curve" "$ST" \
  && echo "guard 1c2: the floor-table models RE-DERIVE from cpk1's raw curve" \
  || guard_fail "guard 1c2: the floor-table models were not re-derived"
grep -q "^  PASS ZERO rows ANYWHERE carry seed 55, 56, 57" "$ST" \
  && echo "guard 1c3: the scorer's OWN fresh-seed check ran and passed" \
  || guard_fail "guard 1c3: the scorer's fresh-seed check did not pass"
echo "guard 1c: scorer sha256 $( (sha256sum "$SC" 2>/dev/null || shasum -a 256 "$SC") | awk '{print $1}')"

# ---- guard 1d: PATCH_VGGBN is present and PROVED ADDITIVE on the LIVE tree ---
grep -q 'PATCH_VGGBN' "$BNLIVE" && echo "guard 1d: live build_network.py carries PATCH_VGGBN" \
                                || guard_fail "guard 1d: live build_network.py lacks PATCH_VGGBN"
grep -q 'PATCH_VGGBN' "$HFLIVE" && guard_fail "guard 1d2: PATCH_VGGBN appears in HF.py" \
                                || echo "guard 1d2: PATCH_VGGBN appears NOWHERE in HF.py"
BNPRE="$BNLIVE.pre_vggbn"
if [ -f "$BNPRE" ]; then
  if python3 "$TESTV" --pre "$BNPRE" --post "$BNLIVE" --hf "$HFLIVE" --csv "$CSVP" >/tmp/cvk1_vggbn.$$ 2>&1; then
    echo "guard 1d3: tests/test_vggbn.py PASSES on the LIVE tree ($(grep -c 'PASS' /tmp/cvk1_vggbn.$$) PASS lines)"
  else
    guard_fail "guard 1d3: tests/test_vggbn.py FAILED on the LIVE tree"; tail -30 /tmp/cvk1_vggbn.$$
  fi
else
  guard_fail "guard 1d3a: no pre-patch backup at $BNPRE"
fi

# ---- guard 2: RULE 21 premise -- no run of this batch may exist yet ---------
grep -q '^cvk1-' "$CSVP" && guard_fail "guard 2: cvk1-* already in the CSV" \
                         || echo "guard 2: no cvk1-* name collision in the corpus"
NOUT=$(ls "$WS/runs"/cvk1-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvk1-*.out on disk" \
                     || guard_fail "guard 2b: $NOUT cvk1-*.out files already exist"
NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvk1-')
[ "${NQ:-0}" = 0 ] && echo "guard 2b2: no cvk1-* job in the queue" \
                   || guard_fail "guard 2b2: $NQ cvk1-* jobs already queued"
NDUP=$(python3 - "$CSVP" <<'PYEOF'
import sys, csv
print(sum(1 for r in csv.DictReader(open(sys.argv[1])) if str(r.get("seed")) in ("55", "56", "57")))
PYEOF
)
[ "${NDUP:-1}" = 0 ] && echo "guard 2c: ZERO corpus rows carry seed 55, 56 or 57" \
                     || guard_fail "guard 2c: $NDUP corpus rows already carry seed 55/56/57"
# guard 2d: seeds against every UNLANDED .out too (another track's batch in flight)
NSEED=$(grep -l -E -- '--seed (55|56|57)( |$)' "$WS/runs"/*.out 2>/dev/null | grep -v '/cvk1-' | wc -l | tr -d ' ')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out files under \$WS/runs carry --seed 55/56/57" \
                      || guard_fail "guard 2d: $NSEED non-cvk1 .out files already use seed 55/56/57"

# ---- guard 3: the CIFAR-100 data is staged on THIS account ------------------
[ -d "$CIF/data/cifar-100-python" ] && echo "guard 3: cifar-100-python present under $CIF/data" \
  || guard_fail "guard 3: CIFAR-100 is not staged at $CIF/data/cifar-100-python"

# ---- guard 4a: PROBE_TENSOR is unset in this batch; its patch is structurally inert
mkdir -p "$SAVE"
[ -f "$HFLIVE" ] || guard_fail "guard 4: no live HF.py at $HFLIVE"
HFSHA=$( (sha256sum "$HFLIVE" 2>/dev/null || shasum -a 256 "$HFLIVE") | awk '{print $1}')
PRE="$HFLIVE.pre_probe_tensor"
if [ -f "$PRE" ]; then
  python3 "$TESTT" --pre "$PRE" --post "$HFLIVE" --structural-only >/tmp/cvk1_struct.$$ 2>&1 \
     && echo "guard 4a: PATCH_PROBE_TENSOR structural inertness R0 PASSES on the live file" \
     || { guard_fail "guard 4a: structural inertness R0 FAILED"; tail -20 /tmp/cvk1_struct.$$; }
else
  guard_fail "guard 4a: no pre-patch backup at $PRE"
fi

# ---- guard 4c: the literal specs and the grid agree -------------------------
SPECS=""
for K in $K_LIST; do
  eval "LIT=\${G_K$K:-}"
  WANT="[$K,$((NTENS-K))]"
  [ "$LIT" = "$WANT" ] || guard_fail "guard 4c: G_K$K is '$LIT', the grid says '$WANT'"
  SPECS="$SPECS $LIT"
done
echo "guard 4c: 7 literal specs agree with the grid"

# ---- guard 4c': the launcher's specs and the scorer's SPEC table agree ------
python3 - "$SC" "$G_K01" "$G_KL" "$G_K13" "$G_K16" "$G_K19" "$G_K20" "$G_K21" "$G_K22" "$G_K23" <<'PYEOF' \
  || guard_fail "guard 4c': the launcher's specs and the scorer's SPEC table disagree"
import sys, re, ast
sc = open(sys.argv[1]).read()
mine = dict(zip(("k01", "kL", "k13", "k16", "k19", "k20", "k21", "k22", "k23"), sys.argv[2:11]))
m = re.search(r"^SPEC = (\{.*?^\})", sc, re.S | re.M)
if not m:
    print("!!! guard 4c': SPEC not found in the scorer"); raise SystemExit(1)
theirs = ast.literal_eval(m.group(1))
bad = [k for k in set(mine) | set(theirs) if mine.get(k) != theirs.get(k)]
for k in bad:
    print("!!! guard 4c': %s launcher=%r scorer=%r" % (k, mine.get(k), theirs.get(k)))
if bad:
    raise SystemExit(1)
print("guard 4c': all 9 spec strings are byte-identical in the launcher and the scorer")
PYEOF

# ---- guard 4: THE LIVE-MODEL MANIFEST, every partition composed and printed --
MANIFEST="$SAVE/PARTITION-MANIFEST.txt"
MANIFEST2="$WS/runs/cvk1-PARTITION-MANIFEST.txt"
# shellcheck disable=SC2086
python3 - "$CIF" "$MANIFEST" "$NTENS" "$TOTPAR" "$CLIP_C" "$EPOCHS" "$BATCH" "$NET" $SPECS <<'PYEOF' \
  || guard_fail "guard 4: live partition check FAILED"
import sys, os
cif, out, ntens, totpar = sys.argv[1], sys.argv[2], int(sys.argv[3]), int(sys.argv[4])
clip_c, epochs, bsz, net_name = sys.argv[5], int(sys.argv[6]), int(sys.argv[7]), sys.argv[8]
specs = sys.argv[9:]
sys.path.insert(0, cif); os.chdir(cif)
import torch
from build_network import build_network
from Optimizers.HF import HF
net = build_network(net_name, "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]
numel = [int(p.numel()) for _, p in net.named_parameters()]
T, tot = len(nps), sum(numel)
bad = 0
print("guard 4: live %s -- %d parameter tensors, %d parameters" % (net_name, T, tot))
if T != ntens or tot != totpar:
    print("!!! guard 4: the batch assumes %d / %d" % (ntens, totpar)); raise SystemExit(1)
if names[22] != "bn8.weight" or numel[22] != 512:
    print("!!! guard 4: tensor 23 is %s (%d), not bn8.weight (512)" % (names[22], numel[22])); bad += 1
else:
    print("guard 4: tensor 23 is bn8.weight, numel 512 -- the nominated carrier; k* = 22 is the cut before it")
lines = ["NUM_PARAM_TENSORS %d" % T, "TOTAL_PARAMS %d" % tot, "NETWORK %s" % net_name,
         "CLIP_C %s" % clip_c, "EPOCHS %d" % epochs, "META_STEPS %d" % (epochs * (50000 // bsz))]
for i in range(T):
    lines.append("TENSOR %d %s %d" % (i + 1, names[i], numel[i]))
for arm, spec, want_type, want_shape in (("k01", "scalar", "scalar", ()),
                                         ("kL", "layerwise", "layerwise", (T,))):
    o = HF.__new__(HF); o.num_layers = T; o._device = torch.device("cpu")
    HF.init_meta(o, spec, nps, 1e-6)
    shapes = [tuple(b.shape) for b in o.beta]
    print("guard 4: %-4s %-10s init_meta: stepsize_type %s  beta shapes %s  rednorm %s"
          % (arm, spec, o.stepsize_type, shapes, getattr(o, "_rednorm", None)))
    if o.stepsize_type != want_type or shapes != [want_shape] or getattr(o, "_rednorm", False):
        print("!!! guard 4: %s does not route to %s" % (arm, want_type)); bad += 1
    lines.append("ARMSPEC %s SPEC %s M %d SIZES %s PARAMS %s"
                 % (arm, spec, 1 if arm == "k01" else T,
                    str(T) if arm == "k01" else ",".join(["1"] * T),
                    str(tot) if arm == "k01" else ",".join(str(x) for x in numel)))
for spec in specs:
    k = int(spec[1:-1].split(",")[0])
    g = HF.polish_the_stepsize_groups(None, spec, nps)
    sizes = [len(x) for x in g]
    pc = [sum(numel[names.index(n)] for n in x) for x in g]
    ok = (len(g) == 2 and sizes == [k, T - k] and list(g[0]) == names[:k]
          and list(g[1]) == names[k:])
    print("guard 4: k%-2d %-8s m=%d sizes %s params %s  first-group tail %s | second-group head %s"
          % (k, spec, len(g), sizes, pc, g[0][-1] if g[0] else "-", g[1][0] if len(g) > 1 else "-"))
    if not ok:
        print("!!! guard 4: %s is not the contiguous prefix cut [%d,%d]" % (spec, k, T - k)); bad += 1
    o = HF.__new__(HF); o.num_layers = T; o._device = torch.device("cpu")
    HF.init_meta(o, spec, nps, 1e-6)
    shapes = [tuple(b.shape) for b in o.beta]
    if o.stepsize_type != "blockwise" or shapes != [(2,)] or getattr(o, "_rednorm", False) \
       or [len(x) for x in o.param_groups_indices] != sizes:
        print("!!! guard 4: %s does not route to a 2-coordinate blockwise beta without rednorm" % spec); bad += 1
    lines.append("ARMSPEC k%d SPEC %s M %d SIZES %s PARAMS %s GROUP0 %s GROUP1 %s"
                 % (k, spec, len(g), ",".join(str(s) for s in sizes),
                    ",".join(str(s) for s in pc), " ".join(g[0]), " ".join(g[1])))
if bad:
    raise SystemExit(1)
open(out, "w").write("\n".join(lines) + "\n")
print("guard 4: 7 cuts + 2 anchors verified on the LIVE model; manifest -> %s" % out)
PYEOF
if [ -s "$MANIFEST" ]; then
  cp "$MANIFEST" "$MANIFEST2"
  echo "guard 4b: manifest written ($(wc -c < "$MANIFEST") bytes) to $MANIFEST and $MANIFEST2"
else
  guard_fail "guard 4b: no manifest"
fi

# ---- guard 4g: provenance ----------------------------------------------------
PROV="$SAVE/PROVENANCE.txt"
BNSHA=$( (sha256sum "$BNLIVE" 2>/dev/null || shasum -a 256 "$BNLIVE") | awk '{print $1}')
{ echo "HF_SHA256 $HFSHA"; echo "BUILD_NETWORK_SHA256 $BNSHA";
  echo "MODE $( [ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN )";
  echo "STAMP_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)";
  [ "$SUBMIT" = 1 ] && echo "SUBMIT_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)";
  echo "ARMS $ARMS"; echo "SEEDS $SEEDS"; echo "EPOCHS $EPOCHS"; echo "PROBE $PROBE"; } > "$PROV"
echo "guard 4g: live HF.py sha256 $HFSHA; build_network.py sha256 $BNSHA; provenance -> $PROV"

# ---- guard 5: capacity, walltime and partitions -----------------------------
FREE=$(df -BG "$WS" 2>/dev/null | tail -1 | awk '{print $4}' | tr -d 'G')
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: ${FREE:-?} GB free on \$WS; $PEND jobs pending, adding $NJOBS"
[ "${PEND:-0}" -le 40 ] || guard_fail "guard 5: $PEND pending, over cap"
for P in $(echo "$PARTS" | tr ',' ' '); do
  TL=$(sinfo -h -p "$P" -o '%l' 2>/dev/null | head -1)
  R=$(squeue -h -p "$P" -t RUNNING 2>/dev/null | wc -l | tr -d ' ')
  D=$(squeue -h -p "$P" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
  echo "guard 5b: $P timelimit $TL   running $R   pending $D"
done
guard_parts_for_wall "$PARTS" "$WALL" || guard_fail "guard 5c: PARTS and WALL disagree"

# ---- guard 8: the scorer's DEFAULT resolution finds the manifest ------------
python3 - "$SC" "$WS/runs" "$MANIFEST" <<'PYEOF' || guard_fail "guard 8: the scorer's DEFAULT manifest resolution does not find the manifest"
import sys, os, importlib.util
spec = importlib.util.spec_from_file_location("cvk1", sys.argv[1])
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
p, how = m.resolve_manifest(sys.argv[2], None)
print("guard 8: with NO --manifest, the scorer resolves the manifest to %s (%s)" % (p, how))
raise SystemExit(0 if p and os.path.realpath(p) == os.path.realpath(sys.argv[3]) else 1)
PYEOF

if [ "$FAIL" = 1 ]; then
  if [ "$SUBMIT" = 1 ]; then
    echo "!!! SUBMIT ABORTED -- nothing submitted."; exit 2
  fi
  echo
  echo "!!! ONE OR MORE GUARDS FAILED.  DRY RUN: the composition below still prints,"
  echo "!!! but --submit WOULD ABORT HERE."
  echo
fi

# ---- compose, guard each line under RULE 20, then (only with --submit) submit
DESIGN=(alg-base=SGDm alg-meta=Lion num-epochs=$EPOCHS batch-size=$BATCH \
        gamma=$GAMMA dataset=$DSET NN-name=$NET weight-decay-meta=0 \
        meta-stepsize=$MST alpha0=$A0)
N=0; REJ=0; BAD=0
for S in $SEEDS; do
  for ARM in $ARMS; do
    eval "G=\${G_$(echo $ARM | tr a-z A-Z)}"
    RN="cvk1-$ARM-s$S"
    CMD="sbatch --job-name=$RN --partition=$PARTS --gres=gpu:1 --cpus-per-task=6 \
--mem=16G --time=$WALL --export=ALL,AUGMENT=$AUG,BETA_CLIP=$CLIP_C,HIER=none,SCHED=none,PROBE=$PROBE \
$RUNNER --optimizer HF --alg-base SGDm --momentum-param-base 0.99 \
--weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 \
--weight-decay-meta 0 --dataset $DSET --NN-name $NET --batch-size $BATCH --max-time 999:00:00 \
