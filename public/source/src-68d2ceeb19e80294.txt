#!/bin/bash
# =============================================================================
# c84_normaliser_transfer.sh -- `gn1`, NORMALISER TRANSFER AT FIXED SINGLETON
# STRUCTURE.  Runs on cluster_a ONLY (needs PATCH_CHUNKWISE **and** the new
# PATCH_RESNET_GN).  24 jobs: {ResNet18, ResNet18_gn} x {nodewise, chunk777} x
# seeds 0-5, ms=1e-4, 100 ep, BETA_CLIP=-25:-2.3026, probe ON.
#
# THE PATCH THIS BATCH NEEDS IS TO `build_network.py`, **NOT** TO `HF.py`.
# `patches/patch_resnet_gn.py` + `tests/test_resnet18_gn.py`.  GUARD 1d refuses
# to run if it is absent from the LIVE tree; GUARD 1e re-runs the suite on CPU.
#
# -----------------------------------------------------------------------------
# THE QUESTION
# -----------------------------------------------------------------------------
#   IS THE PARTITION GAP CARRIED BY THE DEGENERATE SIZE-1 TAIL **PER SE**, OR BY
#   **BATCHNORM** SPECIFICALLY?
#
#   The campaign's tail mechanism, four readings deep, is
#       D = plateau5(chunk777) - plateau5(nodewise)
#         = +0.485 (mm1) / +0.581 (pp1) / +0.727 (cc1) at ms=1e-4, IV pool +0.581
#           (se 0.094, re-derived from the CSV by GUARD 2c at submit time)
#   and it collapses once the tail is merged on BOTH sides
#       G = chunk2325 - nodewise1d = +0.295 (UNDECIDED) / +0.011 / -0.001.
#   nodewise's m = 14,420 groups contain 9,610 SIZE-1 groups (66.6436%) covering
#   0.086004% of the weights, and every one of them sits on the 41 one-dimensional
#   tensors -- all 40 BatchNorm scales and shifts, plus `linear.bias`.
#
#   **THAT SENTENCE HAS TWO READINGS AND NOTHING THE CAMPAIGN OWNS SEPARATES
#   THEM.**  Every D was measured on a net whose only 1-D tensors are BatchNorm
#   parameters, so "the carrier is the degenerate tail" and "the carrier is
#   BatchNorm" are one measurement seen twice.  `gn1` separates them by changing
#   the NORMALISER while holding the PARTITION byte-identical.
#
# -----------------------------------------------------------------------------
# WHY THE TEST IS UNUSUALLY TIGHT -- VERIFIED, NOT ASSUMED
# -----------------------------------------------------------------------------
#   `nn.GroupNorm(G, C)` has `weight` and `bias` of shape (C,), EXACTLY as
#   `nn.BatchNorm2d(C)` does.  BN's `running_mean`/`running_var` are BUFFERS: they
#   never enter `named_parameters()`, so they never received a step size under any
#   granularity and their absence changes nothing.  The GN ResNet18 census is
#   IDENTICAL TO THE BN ROW IN EVERY COLUMN -- 11,173,962 params, 62 tensors, 41
#   one-D tensors, nodewise m 14,420, 9,610 singletons, nodewise1d m 4,851, the
#   same count-matched K=777 -> 14,421.  GUARD 4 MEASURES all of that from the
#   ALLOCATED beta on both really-built networks before anything is submitted.
#
#   And `train.py` / `Optimizers/HF.py` contain **NO MODULE-TYPE LOGIC WHATSOEVER**
#   -- no `BatchNorm`, no `GroupNorm`, no `named_modules`, no isinstance-on-modules.
#   `HF.init_meta` is driven entirely by `[(name, p.data.size()) for n,p in
#   net.named_parameters()]`.  GUARD 1c asserts that on the live file rather than
#   quoting it.  So the partition is not merely SIMILAR across the two nets --
#   **it is provably the same object** -- and NO OPTIMISER PATCH IS REQUIRED.
#   Exactly one thing in the whole experiment changes.
#
#   G = 32 is buildable everywhere: every ResNet18 channel count is 64/128/256/512,
#   each divisible by 32, including the three downsample shortcuts.  **LayerNorm is
#   REJECTED**: on conv activations it is GroupNorm(1, C), the confound below in its
#   most extreme form with no compensating advantage.
#
# -----------------------------------------------------------------------------
# THE FOUR ARMS, ONE BATCH.  m MEASURED BY GUARD 4, NOT ASSERTED.
# -----------------------------------------------------------------------------
#     bn-node   ResNet18      nodewise   m = 14,420      <- positive control
#     bn-ch     ResNet18      chunk777   m = 14,421      <- positive control
#     gn-node   ResNet18_gn   nodewise   m = 14,420
#     gn-ch     ResNet18_gn   chunk777   m = 14,421
#   Count mismatch +1 group in 14,420 = 0.007%, identical to every previous D.
#
#   **ALL FOUR IN ONE BATCH** so D_BN and D_GN are WITHIN-batch.  That is not
#   convenience: STANDING RULE 14 (CORRECTIONS 113) established that seeds do NOT
#   pair across batches on this cluster (356 same-config same-seed pairs, median
#   |diff| 0.171 vs across-seed sd 0.159), and the cross-batch offset has three
#   draws with no stable sign.  Within one batch any offset common to the arms
#   cancels exactly.
#
# -----------------------------------------------------------------------------
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
# Transcribed from docs/REGISTER-ideas-ABC.md section 4; the scorer
# `analysis/c84_gn1_score.py` carries every threshold verbatim and its selftest
# asserts them against THIS FILE'S OWN TEXT.
# -----------------------------------------------------------------------------
#
#   T0    **TRAINS-AT-ALL AND BOX GATE.**  Per run, before any difference is formed.
#         T0.1  epochs_done == epochs_requested == 100.
#         T0.2  collapsed == 0, using the corpus detector verbatim
#               (aggregate.py:155: int(tests[-1] <= 11.0 and max(tests) > 20.0)).
#         T0.3  final_test > 60.0 AND plateau5 >= 85.0.
#         T0.4  NO LATE BLOW-UP: max(test) - plateau5 <= 2.0 pp.  This is the
#               `bd7-w-c6` phenotype (best 88.82, final 10.0) caught before it
#               reaches the arithmetic.
#         T0.5  BOX OCCUPANCY IN BOTH DIRECTIONS (RULE 13), **MEASURED ON THE gn
#               ARMS AND NEVER INHERITED FROM THE bn ANCHOR** -- the GN net's beta
#               trajectory is not the BN net's.  rec_lo <= 0.05 AND rec_hi <= 0.05
#               on every arm.
#         FIRING: an arm failing T0.1-T0.4 on >= 2 of its 6 seeds is VOID; an arm
#         failing T0.5 is UNINTERPRETABLE (RULE 5) and drops.  **>= 2 of the 4 arms
#         void or dropped => THE WHOLE BATCH IS VOID.**
#         Expected GN cost is 0.5-1.5 pp of absolute accuracy vs BN, leaving the
#         plateau near 91-92 against the T0.3 floor of 85 -- ample margin, but that
#         is an EXPECTATION and T0 is what decides.
#
#   T0.6  **THE COMMENSURABILITY GATE.  ADDED AT CYCLE-84 REVIEW; THE T0.3 FLOOR OF
#         85 IS FAR TOO LOOSE TO PROTECT THIS BATCH'S ARITHMETIC ON ITS OWN.**  The
#         whole batch turns on comparing two differences measured in PERCENTAGE
#         POINTS on two networks that need not sit at the same accuracy.  BN
#         plateaus near 92.2, an error budget of 7.8 pp.  A GN arm at 88.0 would
#         pass T0.1-T0.5 cleanly -- 100 epochs, collapsed=0, final > 60, plateau5
#         >= 85, no late blow-up -- while sitting on a 12.0 pp budget, a 1.6x change
#         in the scale the effect lives on.  If the partition gap is even partly
#         MULTIPLICATIVE in the error budget, D_GN is distorted by ~60%, which is
#         about 7x se(D_GN) and straddles BOTH the +0.30 TRANSFERS line and the
#         [-0.15,+0.15] null band: a T2 firing could then be a pure accuracy-regime
#         artefact rather than a statement about carriers.
#           GATE: |mean plateau5 over the 2 GN arms - mean over the 2 BN arms|
#                 <= 2.0 pp.  Evaluated AFTER T0.1-T0.5 and BEFORE T2.
#           IF IT FIRES: the batch is reported as "GroupNorm sits in a DIFFERENT
#                 ACCURACY REGIME; D_GN and D_BN are not commensurable in pp" and
#                 **NO transfer verdict is issued at all.**  It is not a null.
#           AND, REPORTED ALWAYS, PASS OR FAIL: the error-budget-normalised
#                 statistic D / (100 - plateau5) per net, printed beside the pp
#                 numbers, so a reader can see for themselves whether a verdict is a
#                 scale artefact.  It is DESCRIPTIVE and gates nothing -- registering
#                 two primaries would just move the argmax problem.
#
#   T1    **THE POSITIVE-CONTROL GATE.**  If D_BN < +0.30 OR t(D_BN) < 2, **THE
#         BATCH IS VOID AS A TEST OF GROUPNORM.**  You cannot ask whether an effect
#         transfers in a batch where your own control does not show it.  D_BN is
#         then recorded as a failed sixth replication of D -- a serious finding in
#         its own right -- but NOTHING ABOUT GROUPNORM MAY BE CONCLUDED.
#
#   T2    **THE MECHANISM REFUTATION.**  With T0 and T1 passed:
#           PRIMARY     D_GN = plateau5(gn-ch) - plateau5(gn-node), Welch 8v8.
#           CO-PRIMARY  D_BN = plateau5(bn-ch) - plateau5(bn-node), Welch 4v4,
#                       same batch.  It is a GATE, so it carries the smaller n.
#           SECONDARY   dD  = D_GN - D_BN.
#         THREE BRIGHT LINES, NO REINTERPRETABLE MIDDLE.  The failure this avoids is
#         bn1's, which landed at +0.295 -- 0.005 pp under its own threshold -- and
#         had to be closed UNDECIDED.
#           TRANSFERS        D_GN >= +0.30 AND t(D_GN) >= 2.  The line is ~half the
#                pooled anchor +0.581; attenuation is expected and is NOT failure,
#                because GroupNorm is a different network at a different level.
#           DOES NOT TRANSFER  D_GN in [-0.15, +0.15] AND THE EQUIVALENCE IS
#                POWERED: se(D_GN) <= 0.113.  That threshold is FIXED HERE IN
#                ADVANCE at 1.3x the planning se of 0.0870 (n=8) and is NOT to be
#                recomputed post hoc.
#           UNDECIDED        anything else, including D_GN in (0.15, 0.30), and
#                including ANY NULL WHOSE se EXCEEDS 0.113.  **An underpowered null
#                is UNDECIDED and MAY NOT be reported as "no gap".**
#           SECONDARY on dD (se 0.151): |dD| <= 0.30 => the two normalisers agree
#                within resolution; dD <= -0.30 with t <= -2 => RESOLVED ATTENUATION
#                under GroupNorm, reported as attenuation and NOT as collapse unless
#                D_GN independently lands in the null band.
#
#         THE TAIL MECHANISM AS CURRENTLY WRITTEN IS **REFUTED** if, in the same
#         batch, D_GN in [-0.15,+0.15] with se(D_GN) <= 0.113 WHILE D_BN >= +0.30
#         with t >= 2.  That is a within-batch dissociation at byte-identical
#         partition structure -- same 9,610 size-1 groups, same 41 one-D tensors,
#         same chunk777 count-match -- so the carrier would not be the degenerate
#         size-1 tail per se.
#         **CONSEQUENCES, WRITTEN NOW SO THEY CANNOT BE SOFTENED LATER:** the paper's
#         prescriptive sentence narrows from "do not give normalisation scalars their
#         own step sizes" to "do not give **BATCHNORM** scalars their own step
#         sizes", with NO LICENCE TO GENERALISE to normalisation layers at large.
#
#         **T2's AMBIGUITY, REGISTERED IN ADVANCE.**  A T2 firing DOES NOT IDENTIFY
#         THE REPLACEMENT CARRIER.  `nn.GroupNorm(32, C)` takes num_groups=32, so it
#         forms **32 GROUPS OF C/32 CHANNELS** -- 2, 4, 8 and 16 channels per group
#         at ResNet18's widths 64/128/256/512.  (An earlier draft of this file said
#         "blocks of 32 channels".  THAT WAS BACKWARDS and it was the sole basis of
#         this paragraph, so it is corrected here, BEFORE any run, at cycle-84
#         review.)  Per-channel scale invariance -- which BatchNorm does provide,
#         making gamma_c the sole controller of a channel's effective magnitude --
#         therefore does NOT hold; only the PARTITION role of a singleton is
#         preserved, not its FUNCTIONAL role.
#         **AND THE DEPARTURE IS GRADED BY DEPTH, WHICH THE ORIGINAL WORDING HID.**
#         At layer1 (C=64) GN(32,64) puts TWO channels per group and is very nearly
#         per-channel; at layer4 (C=512) it puts SIXTEEN.  So the loss of scale
#         invariance is mild where the net is wide-shallow and strongest where it is
#         deep.  A null therefore CANNOT be attributed uniformly to "loss of
#         per-channel scale invariance", and the registered GN(num_groups == C)
#         disambiguator is a SMALLER perturbation from GN(32,C) than the original
#         text implied, which makes it a cheaper follow-up than registered.
#         The test is ASYMMETRIC: **a POSITIVE IS CLEAN** (the gap survives a
#         normaliser change at byte-identical singleton structure; no alternative
#         reading), **A NULL IS AMBIGUOUS BETWEEN THREE READINGS** and this batch
#         separates NONE of them:
#           (i)   the carrier is batch statistics;
#           (ii)  the carrier is per-channel scale invariance (graded by depth, as
#                 above);
#           (iii) **THE TUNING READING (RULE 11).**  This batch runs a DIFFERENT
#                 NETWORK at a cell -- alpha0=1e-3, ms=1e-4 -- inherited wholesale
#                 from BatchNorm, and no arm's argmax is located on either net.  A
#                 null at a MISTUNED cell is observationally identical to "the
#                 carrier is BatchNorm-specific".  This is the same structural
#                 defect that (correctly) killed the BN-free arm, one level up.
#         **CONSEQUENCE, BINDING: a null from gn1 ALONE may NOT be written as a
#         mechanism narrowing.**  It is reported as "DOES NOT TRANSFER AT THIS CELL,
#         three unseparated readings", and the prescriptive sentence may not be
#         narrowed until `gn2` has run.
#         **`gn2`, THE CONDITIONAL FOLLOW-UP, IS SPECIFIED NOW SO IT CANNOT BE
#         DESIGNED AFTER SEEING THE ANSWER.  It runs ONLY if T2 fires:**
#           gn2a  THE TUNING BRACKET, which discharges reading (iii): all four arms
#                 re-run at alpha0=3e-4 (beta0 = -8.1117), n=4, 16 jobs.  **THAT
#                 CELL IS PROVABLY FREE IN BOTH DIRECTIONS** by the same Lion
#                 identity GUARD 8 applies here -- reach 5.0 nats gives
#                 [-13.112, -3.112], so the -15 floor AND the -2.3026 ceiling are
#                 BOTH unreachable -- which is why alpha0, not ms, is the bracket
#                 axis: at ms=3e-4 there is NO free box for the singleton arm (fa1's
#                 nodewise arm reads rec_hi 0.0000/0.1672/0.0023/0.0137/0.2192/
#                 0.0600, i.e. 3 of 6 seeds OVER the 5% T0.5 threshold, so the
#                 bracket arm would be dropped as uninterpretable before it was
#                 read).  BOTH nets are re-run so a null at gn2a cannot be confused
#                 with "the a0=3e-4 cell kills the gap for everyone".
#           gn2b  THE CARRIER DISAMBIGUATOR, which separates (i) from (ii):
#                 GroupNorm(num_groups == C), restoring per-channel scale invariance
#                 exactly and leaving batch-vs-sample statistics as the sole
#                 difference.  It is a weaker network, so it is not the primary;
#                 running it first would trade a clean test for a poor one.
#
#   T3    **ANTI-NARRATION.**
#         T3.1  A VOID GN CELL IS NOT EVIDENCE OF ANYTHING.  If T0 voids the GN arms
#               the batch reports "GroupNorm did not train to a usable plateau at
#               this cell" and stops.  It may NOT be reported as an absent gap, and
#               it may NOT be salvaged by adding seeds after seeing which side it
#               fell on (the move CORRECTIONS 108.6(a) already refused for bn1).
#         T3.2  **NO REBRANDING.**  This batch may NOT be written up as a
#               no-BatchNorm test, a norm-free test, or a falsification of the
#               singleton mechanism by removal.  **IT REMOVES ZERO SINGLETONS.**
#               Its headline is: does the gap survive a change of normaliser at
#               FIXED singleton structure?
#         T3.3  NO CROSS-BOX POOLING.  D_GN and D_BN come from THIS batch at
#               ms=1e-4, -15:-2.3026.  CORRECTIONS 113 established that the box
#               changed the OPTIMISER, not merely the instrument (nodewise1d moved
#               -0.177, t -2.22, RESOLVED).  Every prose sentence about a D carries
#               its cell.
#         T3.4  NO ARGMAX HARVESTING.  The four-arm table is reported whole; the
#               maximum cell is not "the effect".
#
#   MANDATORY: all four arm means reported separately with sd, n and BOTH occupancy
#   channels.  The absolute GN level is itself the headroom check, and T0.6 turns it
#   from a remark into a gate.
#
# -----------------------------------------------------------------------------
# WHY ms = 1e-4 AND WHY THIS BOX -- AND WHY -25 IS NOT A BOX CHANGE HERE
# -----------------------------------------------------------------------------
#   `Lion_meta_update` with --weight-decay-meta 0 gives |dbeta| = ms EXACTLY per
#   coordinate per step, so |beta_T - ln(alpha0)| <= ms*T for ANY velocity profile.
#   That is a PROOF, not an extrapolation.  At ms=1e-4, T = 100 ep x 500 steps:
#       ms*T = 5.0, beta reachable in [-11.9078, -1.9078] from beta0 = -6.907755.
#   **The floor at -25 is therefore PROVABLY DEAD, and -25 and -15 are IDENTICAL IN
#   EFFECT at this cell.**  GUARD 8 recomputes that table at submit time.  So this
#   is NOT a box change relative to the mm1/pp1/bn1/cc1 anchors at ms=1e-4, and no
#   cross-box caveat attaches on that axis; -25:-2.3026 is used for consistency with
#   the current standard.  The ceiling at -2.3026 IS reachable and is deliberately
#   NOT released: the corpus's only released-ceiling runs (bd7-w-c6-s0/s1, box
#   -30:6.0) are 2/2 collapsed=1.
#
#   Why not ms=3e-4, where fa1 already has 6 seeds: at 3e-4 there is NO box that is
#   free for the singleton arm.  Closing the floor at -15 pins ~12% of nodewise's
#   coordinates (ar1: rec_lo 0.4521-0.4596 on 12/12); releasing it to -25 frees the
#   floor but nodewise -- and ONLY nodewise, the arm that loses the contrast --
#   touches the CEILING on 3 of 6 seeds above R4's 5% threshold.  At ms=1e-4 both
#   guards read EXACTLY 0.0000 on 33/33 probe arms with a 1.0-nat margin.
#
# -----------------------------------------------------------------------------
# NOT AVAILABLE AT THIS CELL -- RECORDED SO IT IS NOT REACHED FOR LATER
# -----------------------------------------------------------------------------
#   The clip-severity corroboration channel (`coord_lo`, the fraction of
#   (record, coordinate) cells at the floor) discriminates 24-88x on BN at ar1's
#   cell: nodewise 0.03430/0.03425/0.03406 against chunk777 0.00039-0.00040,
#   chunk2325 0.00092-0.00093, nodewise1d 0.00138-0.00141.  **At the registered cell
#   NO coordinate reaches either guard on any arm, so coord_lo is identically 0 and
#   this channel yields nothing here.**  It may NOT be recovered by moving the batch
#   to a binding box: that would trade an interpretable accuracy contrast for an
#   uninterpretable one.  rec_lo/rec_hi are carried as the T0.5 gate only.
#
# -----------------------------------------------------------------------------
# WHAT THIS BATCH EXPLICITLY DOES NOT CLAIM
# -----------------------------------------------------------------------------
#   * It is NOT a no-BatchNorm test.  It removes ZERO singletons (T3.2).
#   * It does not identify the carrier if the null fires (T2's ambiguity).
#   * It does not touch the ARCHITECTURE or the DATASET axis of the generalisation
#     debt.  It discharges the NORMALISATION axis only.
#   * It is 4 (BN) / 8 (GN) seeds at ONE cell on ONE architecture and ONE dataset.
#   * It says nothing about GroupNorm as a training recipe; the absolute GN level is
#     a headroom check, not a comparison.
#
# -----------------------------------------------------------------------------
# **SEQUENCING: THIS BATCH IS NOT FIRST IN THE QUEUE, AND GUARD 9 ENFORCES THAT.**
# -----------------------------------------------------------------------------
#   gn1 REFINES a mechanism on ResNet18/CIFAR-10.  The campaign's own rank-1
#   question -- does the tail prescription GENERALISE at all -- has been rank 1 for
#   four consecutive cycles (CORRECTIONS 110.7 item 1, 111.7 item 1) and has THREE
#   fully-built, guard-complete, pre-registered batches sitting on disk with zero
#   rows.  GUARD 9 DISCOVERS THEM BY GLOB (bin/c83_gen_*.sh) rather than by name,
#   because they were restructured mid-cycle -- the two ResNet34 scripts were merged
#   into bin/c83_gen_r34_merged.sh (`g3m`, 36 jobs, D and G in ONE batch at ONE ms) --
#   and a guard that hard-codes filenames goes quietly green the moment one is renamed.
#   NONE of them needs a patch.
#   AND THE CORPUS IS UNAMBIGUOUS ABOUT THE HOLE: over all 1,891 rows, nodewise1d,
#   chunk777 and chunk2325 appear on ResNet18/CIFAR-10 AND NOWHERE ELSE (15/18/15
#   runs); ResNet34, ResNet10 and ResNet18_c100 carry `nodewise` ONLY.  There is not
#   one tail-relevant run on any second architecture or dataset.
#   **IF D DOES NOT SURVIVE ON A SECOND ARCHITECTURE, gn1 IS A REFINEMENT OF
#   SOMETHING THAT DOES NOT EXIST.**  A normaliser-transfer result would be
#   retrospectively worthless, and it costs more than either generalisation batch.
#   GUARD 9 therefore refuses --submit until the operator exports GN1_GEN_OK naming
#   the generalisation batch that has been SCORED and the D leg that survived.  It
#   is an interlock of the same kind as GN1_MIN_PER_EPOCH: it cannot check the
#   science for you, it can only stop the batch jumping the queue silently.  It also
#   fails CLOSED on an empty glob: if no generalisation batch is found at all, that is
#   reported as 0 built and --submit is still refused while the tail contrasts sit on
#   one (network, dataset) cell.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 tests/test_resnet18_gn.py       (CPU, G0 asserts determinism first)
#   2. python3 tests/test_chunkwise.py         (CPU)
#   3. python3 analysis/c84_gn1_score.py --selftest
#   4. `gn1` registered in c55_neff_noise.BOXES BEFORE scoring, from THIS CLIP=.
#   5. plateau5 is PRIMARY.  The CSV `plateau` column is mean-of-last-20 and is NOT
#      comparable across budgets.
#
# USAGE
#   bash bin/c84_normaliser_transfer.sh            # DRY RUN (default)
#   bash bin/c84_normaliser_transfer.sh --submit   # submits, ONLY if guards pass
# =============================================================================
set -u

SUBMIT=0
while [ "$#" -gt 0 ]; do
  case "$1" in
    --submit) SUBMIT=1 ;;
    *) echo "unknown argument $1" >&2; exit 64 ;;
  esac
  shift
done

GUARD_FAILS=0
guard_fail() {
  GUARD_FAILS=$((GUARD_FAILS + 1))
  echo "!!! GUARD FAIL: $1"
  if [ "$SUBMIT" = "1" ]; then
    echo "!!! SUBMIT ABORTED -- nothing was submitted."
    exit 1
  fi
  echo "    (dry run: continuing so the emitter can be validated off-cluster."
  echo "     THIS BATCH MUST NOT BE SUBMITTED UNTIL THIS GUARD PASSES ON cluster_a.)"
}

USER_NAME="${USER:-$(whoami)}"
# PORTABLE (cycle 85): default is cluster_a; set METAOPT_WS to run on cluster_b, whose
# HF.py was brought to byte-identical parity on 30 Aug (md5 d3202635c3fc) with all
# three equivalence suites green, and which carries PATCH_RESNET_GN.
WS=${METAOPT_WS:-/cluster/account_a/metaopt}
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/gn1
LIVE_TREE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
# GN_TREE exists ONLY so the guards can be validated against a SCRATCH copy before
# the operator edits the live tree.  GUARD -1 REFUSES --submit whenever it is not the
# live tree, so a validation override can never become a submission.
TREE="${GN_TREE:-$LIVE_TREE}"
HF_PY=$TREE/Optimizers/HF.py
BUILD_NET=$TREE/build_network.py

WALL=03:50:00                  # gpu-short caps at 4:00:00; GUARD 6b checks it against
                               # sinfo and GUARD 7 against the CSV's own R18 rates
                               # inflated by the GN slowdown factor.
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g

TAG=gn1
NJOBS=24                       # 4 arms, ONE batch (STANDING RULE 14).  **SEEDS ARE
                               # NOT EQUAL ACROSS ARMS**, by design: the BN control
                               # only has to CLEAR a gate (t >= 2 against +0.581), which
                               # n=4 does at t = 4.7, while the GN arms carry the
                               # PRIMARY and want every seed available.  Rebalancing
                               # 6/6/6/6 -> 4/4/8/8 keeps the batch at 24 jobs and moves
                               # se(D_GN) from 0.100 to 0.0870.
SEEDS_BN="0 1 2 3"             # n=4, bn-node + bn-ch  -> se(D_BN) 0.1230, t 4.72
SEEDS_GN="0 1 2 3 4 5 6 7"     # n=8, gn-node + gn-ch  -> se(D_GN) 0.0870
SEEDS="$SEEDS_BN $SEEDS_GN"    # union, for the guards that only need the distinct set
CLIP=-15:-2.3026               # **MOVED FROM -25 AT CYCLE 84 REVIEW.**  (-25,ms=1e-4)
                               # exists in ZERO of the 1,891 corpus rows; every ms=1e-4
                               # run sits at -15, and EVERY frozen threshold in this
                               # file (the +0.30 T1 gate, the +0.5805 anchor, sigma
                               # 0.1740) is computed from -15 data.  GUARD 8 proves the
                               # two floors are IDENTICAL IN EFFECT here, so the move is
                               # free -- and it makes the cell byte-identical to the
                               # anchors instead of a gratuitous cross-box import.
MST=1e-4                       # the ONLY cell where both guards read EXACTLY 0.0000
ALPHA0=1e-3                    # beta_0 = ln(alpha0) = -6.907755
GAMMA=1
EPOCHS=100
BATCH=100
DSET=CIFAR10
STEPS_PER_EPOCH=500            # 50,000 CIFAR-10 train images / batch 100
PROBE_EVERY=5                  # -> 50,000/5 = 10,000 records, the corpus denominator

             "frozen se(D_GN) <= 0.113 threshold was computed from %.3f and may not be "

# three built batches waiting.  This guard cannot judge the science -- it exists so
# that jumping the queue has to be a DECISION someone typed, not a default.
python3 - "$REPO" "$CSVP" "${GN1_GEN_OK:-}" "$SUBMIT" \
  <<'PY9EOF' || guard_fail "guard 9: sequencing"
import csv, glob, os, sys
repo, csvp, token, submit = sys.argv[1], sys.argv[2], sys.argv[3].strip(), sys.argv[4]
rd = list(csv.DictReader(open(csvp)))
TAIL = ("nodewise1d", "chunk777", "chunk2325")
where = {}
for r in rd:
    if r["granularity"] in TAIL:
        where[(r["network"], r["dataset"])] = where.get((r["network"], r["dataset"]), 0) + 1
print("guard 9: tail-relevant granularities %s appear on: %s"
      % (list(TAIL), sorted("%s/%s x%d" % (k[0], k[1], v) for k, v in where.items())))
# **DISCOVERED BY GLOB, NOT HARD-CODED.**  The generalisation batches were
# restructured mid-cycle (the two ResNet34 scripts were merged into one), and a guard
# that names files by hand goes quietly green the moment someone renames one.  Any
# bin/c83_gen_*.sh is a generalisation batch by construction of the naming scheme.
built = sorted(os.path.basename(f)
               for f in glob.glob(os.path.join(repo, "bin", "c83_gen_*.sh")))
print("guard 9: %d built-and-unrun generalisation batches on disk: %s"
      % (len(built), built))
if len(where) == 1 and not token and submit == "1":
    sys.exit("GUARD FAIL: the tail contrasts exist on EXACTLY ONE (network, dataset) "
             "cell, %s, and %d generalisation batches are built and unrun.  gn1 is a "
             "REFINEMENT: if D does not survive on a second architecture it refines "
             "nothing.  Run one of %s FIRST, score it, then export "
             "GN1_GEN_OK='<tag>: D=<value> (t=<value>) on <net>' to record WHY gn1 may "
             "now jump the queue."
             % (list(where)[0], len(built), built))
if token:
    print("guard 9: GN1_GEN_OK = %r -- the operator has recorded a generalisation "
          "result; gn1 proceeds on that basis." % token)
elif submit != "1":
    print("guard 9: GN1_GEN_OK unset -- REFUSED at --submit, allowed in a dry run so "
          "the emitter can be validated.")
PY9EOF

# ---------------------------------------------------------------------------
if [ "$SUBMIT" = "1" ]; then
  mkdir -p "$SAVE" || { echo "GUARD FAIL: cannot create $SAVE"; exit 1; }
else
  mkdir -p "$SAVE" 2>/dev/null || true
fi
N=0
FAIL=0
# --- EMITTER BEGIN ---
# GUARD 1b checks EVERY registered flag against this region and nothing outside it,
# so the base/meta optimiser pins live INSIDE it, not above it.
SGDM=(--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0)
# The arm table carries its OWN seed list per arm (field 4) so the 4/4/8/8 split is
# declared in ONE place and GUARD 1b-4 reads it from here.
for ARM in "ResNet18:nodewise:bn-node:BN" "ResNet18:chunk${CHUNK_K}:bn-ch:BN" \
           "ResNet18_gn:nodewise:gn-node:GN" "ResNet18_gn:chunk${CHUNK_K}:gn-ch:GN"; do
  NET="${ARM%%:*}"; REST="${ARM#*:}"; GRAN="${REST%%:*}"
  REST2="${REST#*:}"; SHORT="${REST2%%:*}"; WHICH="${REST2##*:}"
  if [ "$WHICH" = "BN" ]; then ARM_SEEDS="$SEEDS_BN"; else ARM_SEEDS="$SEEDS_GN"; fi
  for S in $ARM_SEEDS; do
    RN="${TAG}-${SHORT}-s${S}"
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --gres=gpu:1 --cpus-per-task=6
      --mem=14G --time="$WALL" --nice=0
      --export="ALL,AUGMENT=1,BETA_CLIP=${CLIP},HIER=none,SCHED=none,PROBE=${PROBE_EVERY},PROBE5=1,PROBE5_WRITE_EVERY=${WRITE_EVERY},PROBE_DIR=$SAVE/probe_${SHORT}_${TAG}_s${S}"
      "$RUNNER" "${SGDM[@]}"
      --dataset "$DSET" --NN-name "$NET" --batch-size "$BATCH"
      --max-time 999:00:00 --gamma "$GAMMA" --meta-stepsize "$MST" --alpha0 "$ALPHA0"
      --num-epochs "$EPOCHS" --stepsize-groups "$GRAN" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    # COUNT WHAT SLURM ACCEPTED, NOT WHAT WE TRIED (CORRECTIONS 71).
    if [ "$SUBMIT" = "1" ]; then
      if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
    else
      N=$((N+1)); printf '%s\n' "${CMD[*]}"
    fi
  done
done
# --- EMITTER END ---
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
echo "---- REGISTERED n = 4 on the BN control arms and n = 8 on the GN arms, 4 arms,"
echo "---- ONE batch.  D_BN and D_GN are both WITHIN this batch, so STANDING RULE 14 is"
echo "---- satisfied by construction and the cross-batch offset cancels.  NO contrast in"
echo "---- this batch crosses a batch or a box boundary. ----"
[ "$N" = "$NJOBS" ] || echo "!!! expected $NJOBS jobs, built $N -- DO NOT TREAT AS SUBMITTED"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
if [ "$GUARD_FAILS" -gt 0 ]; then
  echo "=============================================================="
  echo "!!! $GUARD_FAILS GUARD(S) FAILED.  The emitter above is VALIDATED; the batch is NOT."
  echo "!!! Nothing was submitted and nothing may be submitted until every guard"
  echo "!!! passes ON cluster_a, on the LIVE tree, with the patch applied."
  echo "!!! A dry run with failed guards is not a green light."
  echo "=============================================================="
  exit 2
fi
exit 0
