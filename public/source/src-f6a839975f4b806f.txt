def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names):
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def window(eps, end, k):
    tail = [eps.get(e) for e in range(end - 5, end)]
    if all(x is not None for x in tail):
        return mean([x[k] for x in tail])
    return None


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")]}
    rec["p100"], rec["t100"] = window(eps, CTRL_EPOCHS, 1), window(eps, CTRL_EPOCHS, 0)
    rec["pE"], rec["tE"] = window(eps, EPOCHS, 1), window(eps, EPOCHS, 0)
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["pE"] is not None and rec["p100"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest INTEGER job id among COMPLETE runs wins."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):

        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:

def tail_slope(eps, lo, hi):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, ep):
    return (r.get("epochs_requested") == str(ep)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, net, ep):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, ep) and r.get("network") == net and r.get("dataset") == DSET:
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
    return (math.sqrt(ss / df) if df else None), df, nc

def bite_check(arm, recs):
    """G-BITE for ONE run.  -> (ok, detail dict).  Pure function of the records."""
    names = [n for n, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    W = WEIGHTS[arm]
    d = {"n": 0, "bad_rec": 0, "bad_z": 0, "bad_m": 0, "worst_z": 0.0, "worst_m": 0.0,
         "informative": 0, "unweighted_match_on_informative": 0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for r in recs:
        ok = (r is not None and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS
              and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
              and isinstance(r.get("z_agg"), list) and len(r["z_agg"]) == 1 and isinstance(r["z_agg"][0], list)
              and len(r["z_agg"][0]) == len(groups)
              and isinstance(r.get("mom_pre"), list) and len(r["mom_pre"]) == 1 and isinstance(r["mom_pre"][0], list)
              and len(r["mom_pre"][0]) == len(groups)
              and isinstance(r.get("beta"), list) and len(r["beta"]) == len(groups)
              and isinstance(r.get("step"), int))
        if not ok:
            d["bad_rec"] += 1
            continue
        try:
            zt = [float(x) for x in r["z_tensor"]]
            mt = [float(x) for x in r["m_tensor"]]
            za = [float(x) for x in r["z_agg"][0]]
            ma = [float(x) for x in r["mom_pre"][0]]
            bb = [float(x) for x in r["beta"]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in zt + mt + za + ma + bb):
            d["bad_rec"] += 1
            continue
        d["n"] += 1
        ez = em = 0.0
        informative = False
        um = True
        for g, zg, mg in zip(groups, za, ma):
            sz = sum(abs(W.get(i, 1.0) * zt[i - 1]) for i in g) + 1e-30
            sm = sum(abs(W.get(i, 1.0) * mt[i - 1]) for i in g) + 1e-30
            wz = sum(W.get(i, 1.0) * zt[i - 1] for i in g)
            wm = sum(W.get(i, 1.0) * mt[i - 1] for i in g)
            uz = sum(zt[i - 1] for i in g)
            ez = max(ez, abs(wz - zg) / sz)
            em = max(em, abs(wm - mg) / sm)
            if W and abs(wz - uz) / sz > INFORM:
                informative = True
                um = um and abs(uz - zg) / sz <= BITE_TOL
        d["worst_z"], d["worst_m"] = max(d["worst_z"], ez), max(d["worst_m"], em)
        d["bad_z"] += ez > BITE_TOL
        d["bad_m"] += em > BITE_TOL
        if informative:
            d["informative"] += 1
            d["unweighted_match_on_informative"] += um
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] == N_RECORDS
          and d["bad_z"] == 0 and d["bad_m"] == 0)
    if W:
        ok = ok and d["informative"] >= MIN_INFORMATIVE and d["unweighted_match_on_informative"] == 0
    return ok, d

def classify(med_r):
    if med_r is None or med_r != med_r:
        return "UNMEASURED"
    if med_r <= PIN_RATIO_MAX:
        return "PINNED"
    if med_r >= FREE_RATIO_MIN:
        return "FREE"
    return "NEITHER"


def arm_class(classes):
    for k in ("PINNED", "FREE"):
        if sum(1 for c in classes if c == k) >= 2:
            return k
    return "NEITHER"


def group_stats(recs, k=0):
    """step-size-magnitude statistics of ONE group's beta trajectory (records in step order)."""
    b = [(r["step"] / float(STEPS_PER_EPOCH), float(r["beta"][k])) for r in recs
         if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > k]
    if not b:
        return None
    n = len(b)
    tail = [v for _e, v in b[-max(1, int(n * TAIL_FRAC)):]]
    lr2 = BETA_LO + math.log(PIN_RATIO_MAX)
    f2 = next((e for e, v in b if v <= lr2), None)
    pf = None
    for i in range(n):
        if all(v <= BETA_LO + EXACT_EPS for _e, v in b[i:]):
            pf = b[i][0]
            break
    post100 = [v for e, v in b if e >= CTRL_EPOCHS]
    return {"n": n, "terminal": b[-1][1], "med_r": statistics.median([math.exp(v - BETA_LO) for v in tail]),
            "first_r2": f2, "pinned_from": pf,
            "max_r_post100": max(math.exp(v - BETA_LO) for v in post100) if post100 else None,
            "off_exact_post100": sum(1 for v in post100 if v > BETA_LO + EXACT_EPS),
            "above_r2_after_first": (sum(1 for e, v in b if f2 is not None and e >= f2 and v > lr2)
                                     if f2 is not None else None),
            "at": dict((int(round(e)), v) for e, v in b if abs(e - round(e)) < 1e-9)}


def three_way(delta, lo_tag, mid_tag, hi_tag, bar):
    if delta is None:
        return "UNMEASURED"
    if delta <= -bar:
        return lo_tag
    if delta >= bar:
        return hi_tag
    return mid_tag

def decide_global(M100, div100):
    if div100:
        return "UNRESOLVED-DIVERGED"
    if M100["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M100["HEAD"] - M100["k01"] < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED"
    return None


def decide_a(M100, ME, div_k13_E=False, div_k33_E=False):
    """(a) K13 at 300 against K33 at 300 and against its own 100-epoch level."""
    if (M100["HEAD"] - M100["K33"] < SHAPE_GAP or M100["HEAD"] - M100["K13"] <= MATCH_BAR
            or M100["K13"] - M100["K33"] <= MATCH_BAR):
        return "K13-PREMISE-NOT-REPLICATED"
    if div_k33_E:
        return "UNRESOLVED-K33-DIVERGED"
    if div_k13_E:
        return "UNRESOLVED-K13-SPLIT"
    g = ME["K13"] - ME["K33"]
    drop = ME["K13"] - M100["K13"]
    if g < -MATCH_BAR:
        return "K13-BELOW-K33"
    if g <= MATCH_BAR:
        return "GRADED-WAS-A-DELAY"
    if ME["K13"] - ME["HEAD"] > MATCH_BAR:
        return "K13-ABOVE-HEAD"
    if drop >= MATCH_BAR and ME["HEAD"] - ME["K13"] <= MATCH_BAR:
        return "K13-RECOVERS-TO-HEAD"
    if drop >= -MATCH_BAR:
        return "K-DEPENDENT-EQUILIBRIUM"
    return "GRADED-PARTLY-A-DELAY"


def decide_b(M100, ME, div_head_E=False):
    """(b) HEAD's rescue at 300 against its in-run 100-epoch size (cgn3's map). -> (branch, rho)."""
    d100 = M100["HEAD"] - M100["k01"]
    dE = ME["HEAD"] - ME["k01"]
    rho = dE / d100 if abs(d100) > 1e-12 else None
    if div_head_E:
        return "UNRESOLVED-HEAD-DIVERGED", rho
    if dE <= REFUTE_BAR:
        return "RESCUE-COLLAPSES", rho
    if rho is not None and rho >= SURVIVE_FRAC and dE >= SUPPORT_BAR:
        return "RESCUE-SURVIVES", rho
    return "RESCUE-ATTENUATES", rho


def decide_c(M100, ME, k33_classes, k33_maxr_post100, div_k33_E=False):
    """(c) K33 at long horizon.  k33_classes / k33_maxr_post100: per seed."""
    if div_k33_E:
        return "K33-SPLIT-AT-300"
    rise = ME["K33"] - M100["K33"]
    if any(x is not None and x >= FREE_RATIO_MIN for x in k33_maxr_post100) or rise >= MATCH_BAR:
        return "K33-REOPENS"
    if (arm_class(k33_classes) == "PINNED" and all(x is not None and x <= PIN_RATIO_MAX for x in k33_maxr_post100)
            and -MATCH_BAR < rise < MATCH_BAR):
        return "K33-HOLDS-PINNED"
    return "K33-UNRESOLVED"

def pin_stamps(tag, gs, runs_arm, E_level, se_readbar):
    """the cgn3 pin-stamp set for ONE arm's group 0.  gs: per-seed group_stats; runs_arm: per-seed runs."""
    out = []
    classes = [classify(g["med_r"]) if g else "UNMEASURED" for g in gs]
    cls = arm_class(classes)
    out.append("%s-COMPLEMENT-%s" % (tag, cls))
    if cls == "PINNED":
        pins = [g["first_r2"] for g, c in zip(gs, classes) if c == "PINNED" and g["first_r2"] is not None]
        latest = max(pins)
        win = EPOCHS - latest
        out.append("%s-PIN-EPOCH:%.1f" % (tag, mean(pins)))
        out.append("%s-POST-PIN-WINDOW:%.1f" % (tag, win))
        out.append("%s-WINDOW-MEETS-W100" % tag if win >= W_POST else "%s-WINDOW-SHORT" % tag)
        if tag == "K13":
            out.append("K13-PIN-INSIDE-BOUND" if latest <= PIN_BOUND_R2 else "K13-PIN-AFTER-BOUND")
        if win >= POST_MIN_WINDOW:
            e0 = int(math.ceil(latest)) + 5
            post = mean([mean([r["eps"][e][1] for e in range(e0, e0 + 5)]) for r in runs_arm])
            out.append(three_way(E_level - post, "%s-DECAYS-AFTER-PIN" % tag, "%s-FROZEN-AT-PIN" % tag,
                                 "%s-IMPROVES-AFTER-PIN" % tag, se_readbar))
        else:
            out.append("%s-POST-PIN-UNMEASURED" % tag)
    else:
        if tag == "K13":
            out.append("UNRESOLVED-NOT-PINNED")
        out.append("%s-POST-PIN-UNMEASURED" % tag)
    return out, classes, cls

def _beta_profile(kind, k):
    """group-0 beta at record k (5 records / epoch).  Invented shapes for the pin stamps."""
    e = k / float(REC_PER_EPOCH)
    if kind == "pin36":
        return max(-15.0, -5.0 - 10.0 * e / 36.0)
    if kind == "pin64":
        return max(-15.0, -5.0 - 10.0 * e / 64.0)
    if kind == "pin160":
        return max(-15.0, -5.0 - 10.0 * e / 160.0)
    if kind == "pin280":
        return max(-15.0, -5.0 - 10.0 * e / 280.0)
    if kind == "hover":                         # never pins: r ~ 20
        return max(-12.0, -5.0 - 7.0 * e / 60.0)
    if kind == "neither":                       # r ~ 5
        return max(-13.4, -5.0 - 8.4 * e / 60.0)
    if kind == "pin38exc":                      # pinned from 38, excursions inside r <= 2 every 13th record
        b = max(-15.0, -5.0 - 10.0 * e / 38.0)
        return -14.5 if (e > 40 and k % 13 == 0) else b
    if kind == "reopen":                        # pinned at 38, re-opens to r ~ 150 after epoch 150
        return -10.0 if e >= 150 else max(-15.0, -5.0 - 10.0 * e / 38.0)
    raise ValueError(kind)


_E_PROFILE = {"k01": "pin36", "HEAD": "pin64", "K13": "pin160", "K33": "pin38exc"}


def _e_probe_records(arm, weights=None, n=N_RECORDS, seed=0, profile=None):
    """synthetic PROBE_TENSOR records consistent with `weights` (default: the arm's own)."""
    import random
    rnd = random.Random(1000 + seed)
    W = WEIGHTS[arm] if weights is None else weights
    names = [nm for nm, _q, _o in r18_tensors(True)]
    groups = arm_groups(arm, names)
    kind = profile or _E_PROFILE[arm]
    out = []
    for k in range(n):
        zt = [round(rnd.uniform(-1e-4, 1e-4), 9) for _ in range(NTENS)]
        mt = [round(rnd.uniform(-1e-4, 1e-4), 9) for _ in range(NTENS)]
        zt[IDX_HEAD - 1] = 5e-2 * (1 + 0.1 * rnd.random())
        mt[IDX_HEAD - 1] = 4e-2
        zt[IDX_TWIN - 1] = 1e-4 * (1 + 0.1 * rnd.random())
        mt[IDX_TWIN - 1] = 8e-5
        za = [sum(W.get(i, 1.0) * zt[i - 1] for i in g) for g in groups]
        ma = [sum(W.get(i, 1.0) * mt[i - 1] for i in g) for g in groups]
        b0 = _beta_profile(kind, k)
        beta = [b0] if len(groups) == 1 else [b0, max(-15.0, -5.0 - 10.0 * (k / 5.0) / 22.0)]
        out.append({"step": 100 * k, "beta": beta, "pt_b2": 0.9, "pt_mp": 0.99,
                    "z_tensor": zt, "m_tensor": mt, "z_agg": [za], "mom_pre": [ma]})
    return out

def _e_score(d, csvpath):

    buf = io.StringIO()
    try:

            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()

def _dict_from_scorer(fname, key, var="SPEC"):
    p = os.path.join(_HERE, fname)
    if not os.path.exists(p):
        return None
    import ast
    m = re.search(r"^%s = (\{.*?^\})" % var, open(p).read(), re.S | re.M)
    if not m:

    return ast.literal_eval(m.group(1)).get(key)

def main():
    ap = argparse.ArgumentParser(description="cvt5 registered scorer.  Documented invocation: "
                                             "python3 analysis/cVT5_plainhorizon_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt5/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
