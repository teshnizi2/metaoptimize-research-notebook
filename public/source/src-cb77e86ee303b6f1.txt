def fmt(v, nd=4):
    if v is None:
        return "--"
    if isinstance(v, float) and not math.isfinite(v):
        return "inf"
    return f"{v:.{nd}f}"


def pct(v):
    return "--" if v is None else f"{100.0 * v:.4f}"


def rule(n=110):

# ----------------------------------------------------------------- beta span
def beta_span(pat):
    """max - min of beta on the LAST record, averaged over seeds.

    N2's registered mechanism variable.  The pre-registration calls it
    "ADAPTATION EXTENT" and predicts the argmin flips to `node` at the stepsize
    whose weightwise span first exceeds ~5 log units -- and that the flip tracks
    span BETTER than it tracks ms.  Span grows sublinearly in ms (measured 4.9x
    for 10x), so the two orderings are genuinely different orderings and the
    four-point ladder can tell them apart.
    """
    import glob
    ds = sorted(glob.glob(pat))
    if not ds:
        return None, None, 0
    spans, travels = [], []
    for d in ds:
        R = records(d)
        spans.append(R[-1]["beta_true_max"] - R[-1]["beta_true_min"])
        travels.append(R[-1]["beta_true_max"] - R[0]["beta_true_max"])
    return mean(spans), mean(travels), len(ds)


# ------------------------------------------------------------------- selftest
def selftest():
    import json
    import tempfile
    checks = []

    def ok(name, cond):
        checks.append((name, bool(cond)))

    with tempfile.TemporaryDirectory() as td:
        # two seeds, known analytic answers, so span/travel are checkable by hand
        for s, (b0, b1) in enumerate([(-7.0, -2.0), (-7.0, -3.0)]):
            d = os.path.join(td, f"probe_w_x_s{s}")
            os.makedirs(d)
            with open(os.path.join(d, "probe.jsonl"), "w") as f:
                for t in range(100):
                    f.write(json.dumps(dict(
                        step=t * 5,
                        beta_true_min=-15.0 if t else -6.909,
                        beta_true_max=b0 + (b1 - b0) * t / 99.0,
                        n_at_lo=0, n_at_hi=0, n_beta=10)) + "\n")
        sp, tr, n = beta_span(os.path.join(td, "probe_w_x_s*"))
        ok("beta_span sees both seeds", n == 2)
        # seed0 span = -2 - (-15) = 13; seed1 = -3 - (-15) = 12; mean 12.5
        ok("span is max-min on the LAST record", abs(sp - 12.5) < 1e-9)
        # travel seed0 = -2 - (-7) = 5; seed1 = -3 - (-7) = 4; mean 4.5
        ok("travel is max_final - max_first", abs(tr - 4.5) < 1e-9)
        ok("empty glob returns None", beta_span(os.path.join(td, "nope*"))[2] == 0)

        # occupancy is imported, not reimplemented -- check the import is live and
        # that the synthetic arm reads as PINNED at the low guard (t>=1) and never
        # at the high one.  If this ever flips, the guard convention changed.
        o = occupancy(os.path.join(td, "probe_w_x_s0"), -15.0, 0.0)
        ok("imported occupancy pins the low guard", abs(o["rec_lo"] - 0.99) < 1e-9)
        ok("imported occupancy sees no high guard", o["rec_hi"] == 0.0)
        ok("coord denominator is live (PATCH_CLIPCOUNT)", o["coord_lo"] == 0.0)
        ok("n_beta is read from the record", o["n_beta"] == 10)

        # headroom must use the LAST-QUARTER velocity, never the whole run.
        # The ramp is linear so both are equal by construction -- what is checked
        # here is the SIGN and the vmax normalisation (STANDING RULE 7).
        h = headroom(os.path.join(td, "probe_w_x_s0"), 0.0, meta_stepsize=1e-3)
        ok("headroom head = hi - final max", abs(h["head"] - 2.0) < 1e-9)
        ok("headroom velocity is positive on a rising arm", h["v_q4"] > 0)

    # argmin logic: an UNRESOLVED rung must make the argmin UNDECIDABLE, never
    # default to one of the resolved ones (N1's own words).
    ok("argmin over 3 finite values", argmin_rung({"lay": 0.7, "node": 0.4, "w": 0.5})[0] == "node")
    ok("argmin refuses a missing rung", argmin_rung({"lay": 0.7, "node": None})[0] is None)
    ok("argmin UNDECIDED inside 0.02",
       argmin_rung({"lay": 0.7, "node": 0.400, "w": 0.415})[1] is False)
    ok("argmin decided outside 0.02",
       argmin_rung({"lay": 0.7, "node": 0.400, "w": 0.500})[1] is True)

    n_ok = sum(1 for _, c in checks if c)
    for name, c in checks:
        if not c:
            print(f"  FAIL: {name}")
    print(f"selftest: {n_ok}/{len(checks)} PASS")
    return 0 if n_ok == len(checks) else 1


def argmin_rung(vals, tol=0.02):
    """-> (rung, decided).  `decided` is False when the two smallest are within
    `tol`, which the ns5 header registers as the UNDECIDED branch ("if two rungs
    come back within 0.02 of each other, say UNDECIDED and buy the third seed").
    A missing/unresolved rung returns (None, False): an argmin over three rungs
    with one absent is undecidable, not a default to one of the other two."""
    if any(v is None or not math.isfinite(v) for v in vals.values()):
        return None, False
    if len(vals) < 3:
        return None, False
    order = sorted(vals.items(), key=lambda kv: kv[1])
    return order[0][0], (order[1][1] - order[0][1]) > tol

def score_bl5():
    rule()
    print("BL5 -- THE BOX-FREE BUDGET LADDER.  20 -> 40 epochs, R18, ms=1e-3, BETA_CLIP=-30:0.0")
    print("  Registered in bin/c52_budget_ladder.sh.  SCORING ORDER: B0, B0.3, B1.5, B1, B2, B2.5, B3, B4.")
    rule()

    arms = {g: f"{P}/probes_bl5/probe_{g}_e40_s*" for g in RUNGS}
    ctrl = {g: f"{P}/probes_cl5/probe_{g}_cU_s*" for g in RUNGS}

    print("\n(B0) VALIDITY GATE")
    print(f"  {'rung':6}{'n_dirs':>8}{'T':>8}{'n_beta':>14}{'byte-match':>12}{'beta MOVED':>12}")
    b0 = True

        print(f"  {g:6}{n:>8}{a['T']:>8}{a['n_beta']:>14,}"
              f"{('yes' if match else 'NO'):>12}{('yes' if moved else 'NO'):>12}")

          f"probe stride changed\n      and the window slices no longer mean what they mean in cl5)")

    print("\n(B0.3) **THE BOX-FREE GATE, SCORED FIRST.**  bar: < 5% of records at either guard")
    print("  The batch rests on a headroom EXTRAPOLATION, and an extrapolation is not a")
    print("  measurement.  If an arm exceeds the bar its B1 is UNINTERPRETABLE, NOT REFUTED,")
    print("  and its first_hi record IS the result -- the first direct measurement of the")
    print("  budget at which a box becomes unavoidable.")

def neff_table_window(root, window):
    """neff_table, but on an explicit window.  Shells out to the same selftested
    instrument through its NEW --window option, so B1.5 and B1 differ in exactly
    one argument and nothing else."""
    import re
    import subprocess
    out = subprocess.run(
        [sys.executable, os.path.join(HERE, "neff_instrument.py"),
         root, f"--window={window[0]:g}-{window[1]:g}"],
        capture_output=True, text=True).stdout
    row = re.compile(r"^\s+(lay|node|w|blk6)\s+([\d,]+)\s+(\d+)\s+\S+\s+\S+\s+"
                     r"([\d,\.]+)\s+([\d,\.]+)\s+([\d,\.]+)\s")
    res, fam = {}, None
    for ln in out.splitlines():
        m = re.match(r"=== family (\S+) ===", ln)
        if m:
            fam = m.group(1)
            res[fam] = {}
            continue
        m = row.match(ln)
        if m and fam:
            mm = int(m.group(2).replace(",", ""))
            nv = float(m.group(6).replace(",", ""))
            res[fam][m.group(1)] = (mm, nv, nv / mm)
    return res

def score_ns5():
    rule()
    print("NS5 -- THE NODEWISE-MINIMUM ONSET LADDER.  ms in {2e-4, 5e-4}, R18, box-free")
    print("  Registered in bin/c52_nodemin_onset.sh.  SCORING ORDER: N0, N0.3, N1, N2, N3.")
    rule()

    LAD = [("1e-4", f"{P}/probes_ml5_m4", "m4", "probe_{g}_m4_s*", 3),
           ("2e-4", f"{P}/probes_ns5", "m2p4", "probe_{g}_m2p4_s*", 2),
           ("5e-4", f"{P}/probes_ns5", "m5p4", "probe_{g}_m5p4_s*", 2),
           ("1e-3", f"{P}/probes_cl5", "cU", "probe_{g}_cU_s*", 3)]

    # ---- N0 VALIDITY (new rungs only; the controls are already in the record)
    print("\n(N0) VALIDITY GATE  (the two NEW stepsizes)")
    print(f"  {'ms':7}{'rung':6}{'n_dirs':>8}{'T':>7}{'n_beta':>14}{'byte-match':>12}{'beta MOVED':>12}")
    # The seed count is NOT a validity condition and must not be folded into one
    # (cycle 54).  This check read `n == 2` hard-coded; when `ns6-*` delivered the
    # THIRD seed at ms=2e-4 -- a seed this campaign deliberately bought -- N0 went
    # FAIL and printed "a n_beta mismatch VOIDS the ladder" while every byte-match
    # cell in its own table read `yes`.  The verdict was right about nothing and
    # the message named a cause that had not occurred.  What VOIDS the ladder is a
    # n_beta mismatch, a wrong T, or beta not moving; a changed seed count is data.
    n0, seedcounts = True, {}
    for ms, root, fam, pat, _ in LAD[1:3]:
        for g in RUNGS:
            a, n = agg(os.path.join(root, pat.format(g=g)), -30.0, 0.0)
            moved = a["min_final"] != a["max_final"]
            match = a["n_beta"] == NBETA[g]
            n0 &= (a["T"] == 2000) and moved and match
            seedcounts[(ms, g)] = n
            print(f"  {ms:7}{g:6}{n:>8}{a['T']:>7}{a['n_beta']:>14,}"
                  f"{('yes' if match else 'NO'):>12}{('yes' if moved else 'NO'):>12}")
    print(f"  --> N0 {'PASS' if n0 else 'FAIL -- n_beta mismatch / wrong T / beta did not move. VOIDS the ladder.'}")
    _uneven = sorted({(ms, n) for (ms, _g), n in seedcounts.items()})
    print(f"  seed counts (DATA, not a gate): " +
          "   ".join(f"ms={ms} n={n}" for ms, n in _uneven))

    # ---- N0.3 THE BOX-FREE GATE, SCORED FIRST
    print("\n(N0.3) **THE BOX-FREE GATE, SCORED FIRST**, for the fifth time in this campaign.")
    print("  Expected to pass trivially -- ms=1e-3 already passed at 0.0% and a SMALLER")
    print("  stepsize travels LESS -- so a FAILURE means something other than the")
    print("  meta-stepsize changed and the batch is VOID rather than interesting.")
    print("  It is scored anyway because 'expected to pass trivially' is the exact")
    print("  reasoning that made cycle 50 move the wrong wall.")
    print(f"\n  {'ms':7}{'rung':6}{'%rec LO':>10}{'%rec HI':>10}{'%coordLO':>11}"
          f"{'%coordHI':>11}{'final min':>12}{'final max':>12}")
    n03 = True
    for ms, root, fam, pat, _ in LAD[1:3]:
        for g in RUNGS:
            a, _n = agg(os.path.join(root, pat.format(g=g)), -30.0, 0.0)
            n03 &= max(a["rec_lo"], a["rec_hi"]) < BAR_BOXFREE
            print(f"  {ms:7}{g:6}{pct(a['rec_lo']):>10}{pct(a['rec_hi']):>10}"
                  f"{pct(a['coord_lo']):>11}{pct(a['coord_hi']):>11}"
                  f"{a['min_final']:>12.3f}{a['max_final']:>12.3f}")
    print(f"  --> N0.3 {'PASS -- all six new arms are box-free; the four-point ladder is box-free end to end' if n03 else 'FAIL -- the batch is VOID'}")

    # ---- N1 THE PRIMARY TEST
    print("\n(N1) THE PRIMARY TEST -- WHERE DOES argmin FLIP?")
    print("  argmin of N_eff/m over {lay, node, w}; VARIANCE instrument, STEADY half.")
    print("  Controls: ms=1e-4 -> **w**;  ms=1e-3 -> **node**.")
    print("  PREDICTION: the flip is MONOTONE in ms -- exactly one crossing in [1e-4, 1e-3],")
    print("  so the admissible outcomes at (2e-4, 5e-4) are (w,w), (w,node) or (node,node).")
    print("  REFUTATION: **(node, w)** -- the shape is not a monotone function of ms and")
    print("  CORRECTIONS 70 must be reported as holding at a POINT, not on an interval.")
    print("\n  Report the FULL three-number profile at every rung, never just the argmin.")

    tabs = {}
    for ms, root, fam, pat, _n in LAD:
        tabs[ms] = neff_table(root).get(fam, {})
    print(f"\n  {'ms':7}{'n':>3}{'lay':>10}{'node':>10}{'w':>10}{'argmin':>9}"
          f"{'decided':>9}   ordering")
    profiles = {}
    for ms, root, fam, pat, nseed in LAD:
        pr = {g: (tabs[ms].get(g)[2] if tabs[ms].get(g) else None) for g in RUNGS}
        profiles[ms] = pr
        am, dec = argmin_rung(pr)
        order = " < ".join(k for k, _ in sorted(
            ((k, v) for k, v in pr.items() if v is not None), key=lambda kv: kv[1]))
        # COUNT THE SEEDS THAT ARE ON DISK; DO NOT PRINT THE TABLE LITERAL (cycle 54).
        # This column printed LAD's hard-coded `nseed`, so after `ns6-*` landed the
        # third seed at ms=2e-4 the row still read "n 2" while the N_eff/m values
        # beside it HAD moved (w 0.2054 -> 0.2128, node 0.2244 -> 0.2271) because
        # neff_table globs the directories.  A provenance label that disagrees with
        # the number it labels is worse than no label.
        nreal = len(glob.glob(os.path.join(root, pat.format(g="w"))))
        flag = "" if nreal == nseed else f"  <-- n differs from the registration's {nseed}"
        print(f"  {ms:7}{nreal:>3}{fmt(pr['lay']):>10}{fmt(pr['node']):>10}{fmt(pr['w']):>10}"
              f"{str(am):>9}{('yes' if dec else 'NO'):>9}   {order}{flag}")

    n1_undecided = False
    seq = [argmin_rung(profiles[ms])[0] for ms, *_ in LAD]
    dec = [argmin_rung(profiles[ms])[1] for ms, *_ in LAD]
    mid = (seq[1], seq[2])
    print(f"\n  argmin sequence across the four-point ladder: "
          f"{' -> '.join(str(s) for s in seq)}")

    # AN UNDECIDED RUNG IS NOT ITS NOMINAL argmin.  The ns5 header registered the
    # tolerance and the consequence in the same sentence: "If two rungs come back
    # within 0.02 of each other, say UNDECIDED and buy the third seed then."
    # Reading the nominal minimum anyway is how a coin-flip becomes a result.
    und = [ms for (ms, *_), d in zip(LAD, dec) if not d]
    if und:
        print(f"\n  *** UNDECIDED RUNGS: {', '.join(und)} -- the two smallest N_eff/m are")
        print("      within the registered 0.02 tolerance, so the argmin there is a")
        print("      COIN FLIP and not a measurement.")
        for ms in und:
            pr = profiles[ms]
            two = sorted(((k, v) for k, v in pr.items() if v is not None), key=lambda kv: kv[1])[:2]
            alt = tuple(k for k, _ in two)
            print(f"      ms={ms}: {alt[0]} {pr[alt[0]]:.4f} vs {alt[1]} {pr[alt[1]]:.4f}"
                  f"  (gap {abs(pr[alt[0]] - pr[alt[1]]):.4f} < 0.02)")
        # is the REGISTERED REFUTATION one of the admissible readings?
        if not dec[1] and seq[2] == "w":
            alt2 = sorted(((k, v) for k, v in profiles["2e-4"].items() if v is not None),
                          key=lambda kv: kv[1])[:2]
            if "node" in [k for k, _ in alt2]:
                print("\n  --> N1 IS UNDECIDED, NOT PASSED.  With ms=2e-4 a coin flip between w")
                print("      and node, the two admissible readings of (2e-4, 5e-4) are")
                print("      (w, w) -- ADMISSIBLE, boundary between 5e-4 and 1e-3 -- and")
                print("      (node, w) -- **THE REGISTERED REFUTATION**.  A test whose")
                print("      refutation branch is live cannot be reported as a pass.")
                print("      ACTION, and it is the one the header registered in advance:")
                print("      BUY THE THIRD SEED AT ms=2e-4 (3 jobs) and score N1 then.")
                n1_undecided = True
    if n1_undecided:
        print("\n  (the verdict branches below are printed for the record only; N1's")
        print("   verdict is UNDECIDED above and that is the one that counts)")
    if mid == ("node", "w"):
        print("  --> N1 REFUTED.  (node, w) is the registered refutation: the shape is NOT a")
        print("      monotone function of the meta-stepsize.  CORRECTIONS 70 must be rewritten")
        print("      as holding at a POINT rather than on an interval.")
    elif mid in (("w", "w"), ("w", "node"), ("node", "node")):
        print("  --> N1 PASSES.  The flip is monotone in ms with exactly one crossing.")
        cross = {("w", "w"): "between 5e-4 and 1e-3",
                 ("w", "node"): "between 2e-4 and 5e-4",
                 ("node", "node"): "between 1e-4 and 2e-4"}[mid]
        print(f"      THE BOUNDARY IS {cross}.  CORRECTIONS 70's scope becomes an INTERVAL.")
    else:
        print(f"  --> N1 UNDECIDABLE: argmin sequence contains a None/undecided rung {mid}.")
        print("      An unresolved rung makes the argmin UNDECIDABLE, never a default.")

    # ---- N2 THE MECHANISM
    print("\n(N2) THE MECHANISM CANDIDATE, registered BEFORE the data landed.")
    print("  PREDICTION: the argmin flips to `node` at the stepsize whose WEIGHTWISE beta")
    print("  SPAN first exceeds ~5 log units, and the flip tracks SPAN better than ms.")
    print("  Span grows SUBLINEARLY in ms (measured 4.9x for 10x), so the two orderings")
    print("  are not the same ordering and a four-point ladder can separate them.")
    print(f"\n  {'ms':7}{'rung':6}{'beta span':>12}{'travel(max)':>14}{'N_eff/m':>10}")
    spans = {}
    for ms, root, fam, pat, _n in LAD:
        for g in RUNGS:
            sp, tr, n = beta_span(os.path.join(root, pat.format(g=g)))
            if g == "w":
                spans[ms] = sp
            pr = profiles[ms][g]
            print(f"  {ms:7}{g:6}{fmt(sp, 3):>12}{fmt(tr, 3):>14}{fmt(pr):>10}")
    print(f"\n  weightwise SPAN vs the argmin:")
    print(f"  {'ms':9}{'span (w)':>12}{'> 5 log units?':>17}{'argmin':>9}")
    for ms, *_ in LAD:
        sp = spans.get(ms)
        am = argmin_rung(profiles[ms])[0]
        print(f"  {ms:9}{fmt(sp, 3):>12}{('YES' if (sp or 0) > 5.0 else 'no'):>17}{str(am):>9}")
    hits = [(ms, (spans.get(ms) or 0) > 5.0, argmin_rung(profiles[ms])[0] == "node")
            for ms, *_ in LAD]
    agree = all(a == b for _, a, b in hits)
    print(f"\n  --> N2 {'CONFIRMED' if agree else 'REFUTED'}: the '>5 log units of span' predicate "
          f"{'matches' if agree else 'does NOT match'} the argmin==node predicate on "
          f"{sum(1 for _, a, b in hits if a == b)} of {len(hits)} rungs.")
    if agree:
        print("      CORRECTIONS 70 can be rewritten in terms of ADAPTATION EXTENT rather than")
        print("      meta-stepsize -- a better statement: it transfers across optimizers and")
        print("      budgets, and it PREDICTS the nodewise minimum should also appear at")
        print("      ms=1e-4 given a LONGER run.  That is directly testable against bl5.")
    else:
        print("      The flip tracks ms and not span.  70 stays scoped to the meta-stepsize")
        print("      and the mechanism stays OPEN.  Say so; do not retro-fit a second variable.")

    # ---- N3 ACCURACY
    print("\n(N3) ACCURACY, reported regardless and NOT a test.")
    print(f"\n  {'ms':9}{'lay':>10}{'node':>10}{'w':>10}")
    for ms, pre in (("1e-4", "ml5-{g}-m4"), ("2e-4", "ns5-{g}-m2p4"),
                    ("5e-4", "ns5-{g}-m5p4"), ("1e-3", "cl5-{g}-cU")):
        vs = []
        for g in RUNGS:
            p, n = plateau(pre.format(g=g))
            vs.append(f"{p:.3f}" if n else "--")
        print(f"  {ms:9}{vs[0]:>10}{vs[1]:>10}{vs[2]:>10}")
    print("  (a smaller meta-stepsize should train more slowly; recorded, not scored)")
