# ---------------------------------------------------------------------------
# Distribution tails.  The cluster env has no scipy; these are implemented here
# and validated against published critical values in the selftest.
# ---------------------------------------------------------------------------
def _betacf(a, b, x):
    MAXIT, EPS, FPMIN = 300, 3e-16, 1e-300
    qab, qap, qam = a + b, a + 1.0, a - 1.0
    c = 1.0
    d = 1.0 - qab * x / qap
    if abs(d) < FPMIN:
        d = FPMIN
    d = 1.0 / d
    h = d
    for m in range(1, MAXIT + 1):
        m2 = 2 * m
        aa = m * (b - m) * x / ((qam + m2) * (a + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        h *= d * c
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
        d = 1.0 + aa * d
        if abs(d) < FPMIN:
            d = FPMIN
        c = 1.0 + aa / c
        if abs(c) < FPMIN:
            c = FPMIN
        d = 1.0 / d
        de = d * c
        h *= de
        if abs(de - 1.0) < EPS:
            break
    return h


def betainc(a, b, x):
    if x <= 0.0:
        return 0.0
    if x >= 1.0:
        return 1.0
    lbeta = (math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b)
             + a * math.log(x) + b * math.log1p(-x))
    if x < (a + 1.0) / (a + b + 2.0):
        return math.exp(lbeta) * _betacf(a, b, x) / a
    return 1.0 - math.exp(lbeta) * _betacf(b, a, 1.0 - x) / b


def t_sf(t, df):
    """P(T > t) for Student-t on df degrees of freedom."""
    if df <= 0:
        return float("nan")
    x = df / (df + t * t)
    p = 0.5 * betainc(0.5 * df, 0.5, x)
    return p if t > 0 else 1.0 - p


def t_two(t, df):
    return 2.0 * t_sf(abs(t), df)


def binom_sf(k, n, p=0.5):
    """P(X >= k) for Binomial(n, p) -- the SIGN TEST, exact."""
    tot = 0.0
    for i in range(k, n + 1):
        tot += math.comb(n, i) * (p ** i) * ((1 - p) ** (n - i))
    return tot


def emax_normal(k):
    """E[max of k iid standard normals], exact for k <= 3."""
    if k == 1:
        return 0.0
    if k == 2:
        return 1.0 / math.sqrt(math.pi)
    if k == 3:
        return 3.0 / (2.0 * math.sqrt(math.pi))
    raise ValueError("only k <= 3 is registered")


# ---------------------------------------------------------------------------
# THE POOLED-VARIANCE CONTRAST ENGINE.
# At n=2 per cell a per-cell Welch has 1 df and is useless.  Every statistic
# below is instead a LINEAR COMBINATION of cell means with the within-cell
# variance POOLED ACROSS ALL SURVIVING CELLS.  This is registered as the primary
# estimator; per-cell Welch is printed beside it as a sanity line only.
# ---------------------------------------------------------------------------
class Pooled(object):
    def __init__(self, cells):
        """cells: {key: [values]} -- only box-free, complete runs."""
        self.cells = {k: list(v) for k, v in cells.items() if len(v) >= 1}
        num = den = 0.0
        for v in self.cells.values():
            if len(v) >= 2:
                num += (len(v) - 1) * st.variance(v)
                den += len(v) - 1
        self.df = int(den)
        self.sd = math.sqrt(num / den) if den > 0 else float("nan")

    def mean(self, k):
        v = self.cells.get(k)
        return st.mean(v) if v else None

    def n(self, k):
        return len(self.cells.get(k, []))

    def contrast(self, coeffs):
        """coeffs: {key: c}.  Returns (value, se, df) for sum c_k * mean_k."""
        val = 0.0
        var = 0.0
        for k, c in coeffs.items():
            m = self.mean(k)
            if m is None or self.n(k) < 1:
                return (float("nan"),) * 3
            val += c * m
            var += c * c / self.n(k)
        if not (self.df > 0) or not (self.sd == self.sd):
            return val, float("nan"), float("nan")
        return val, self.sd * math.sqrt(var), self.df


def welch(a, b):
    """(diff, se, df) for mean(a) - mean(b), Welch-unpaired.  SANITY LINE ONLY."""
    if len(a) < 2 or len(b) < 2:
        return (st.mean(a) - st.mean(b) if a and b else float("nan"),
                float("nan"), float("nan"))
    va, vb = st.variance(a), st.variance(b)
    na, nb = len(a), len(b)
    se = math.sqrt(va / na + vb / nb)
    if se == 0:
        return st.mean(a) - st.mean(b), 0.0, float("nan")
    df = (va / na + vb / nb) ** 2 / ((va / na) ** 2 / (na - 1) + (vb / nb) ** 2 / (nb - 1))
    return st.mean(a) - st.mean(b), se, df


# ---------------------------------------------------------------------------
# plateau5, re-derived from the .out epoch series.  NEVER read from the CSV
# `plateau` column, which is mean-of-last-20 and is not comparable across budgets.
# ---------------------------------------------------------------------------
EP_RE = re.compile(r"Epoch\s+(\d+).*?Test Accuracy:\s*([0-9.]+)", re.I)


def plateau5_from_out(path):
    tests = []
    with open(path, "r", errors="replace") as fh:
        for line in fh:
            m = EP_RE.search(line)
            if m:
                tests.append((int(m.group(1)), float(m.group(2))))
    if len(tests) < 5:
        return None, len(tests)
    tests.sort()
    return sum(v for _, v in tests[-5:]) / 5.0, len(tests)

# THE RAIL READER.  CORRECTED 2026-08-30, CORRECTIONS 117.1.
#
# The per-coordinate rail counts live in  n_at_lo / n_at_hi  over  n_beta.
# r["beta"] is a PER-TENSOR summary -- 62 entries on ResNet18 against
# n_beta = 14,420 -- and reading rails off it is BLIND to real clipping and,
# where it does fire, gets the arm ranking BACKWARDS.  Both failures were
# measured on data already on disk, in the plateau5 window:
#
#   fa1 (ms=3e-4, 100 ep, box -25:-2.3026).  The 62-list reads 0.000000 on
#   ALL 24 runs.  The coordinate counts read nodewise s1/s4/s5 rec_hi = 1.0000
#   -- a clipped coordinate in EVERY record -- while all 18 merged-tail runs
#   are exactly 0.  The ceiling binds, and it binds only on the arm the paper
#   is about.
#
#   ar1 (ms=3e-4, 100 ep, box -15:-2.3026).  The 62-list says chunk777 is
#   7x MORE floor-bound than nodewise (0.2305 vs 0.0325).  The coordinate
#   counts say nodewise is 75x more bound than chunk777 (0.1103 vs 0.00145).
#   The instrument inverted the sign of the arm asymmetry.
#
# The 62-list is retained as a THIRD, LABELLED column so the historical
# statistic stays visible, and is never used for a gate.
# ---------------------------------------------------------------------------
def rail_fracs(rec, lo, hi):
    """(coord_lo, coord_hi, rec_lo, rec_hi, tensor_lo, tensor_hi) or None.

    coord_*  : fraction of the run's beta COORDINATES at a rail (n_at_*/n_beta)
    rec_*    : 1 if ANY coordinate is at that rail in this record, else 0
    tensor_* : the historical 62-entry per-tensor statistic; DIAGNOSTIC ONLY
    """
    nb = rec.get("n_beta")
    nlo = rec.get("n_at_lo")
    nhi = rec.get("n_at_hi")
    if not isinstance(nb, (int, float)) or not nb or nlo is None or nhi is None:
        return None
    nb = float(nb)
    clo, chi = float(nlo) / nb, float(nhi) / nb
    rlo = 1.0 if float(nlo) > 0 else 0.0
    rhi = 1.0 if float(nhi) > 0 else 0.0
    b = rec.get("beta")
    if isinstance(b, list) and b:
        n = float(len(b))
        tlo = sum(1 for x in b if x <= lo + 1e-9) / n
        thi = sum(1 for x in b if x >= hi - 1e-9) / n
    else:
        tlo = thi = float("nan")
    return clo, chi, rlo, rhi, tlo, thi


def read_probe_occupancy(probe_dir, lo, hi, last_frac=0.25):
    """(coord_lo, coord_hi, rec_lo, rec_hi, n_records) over the LAST quarter of
    the run, or (None, None, None, None, 0) if no probe is on disk.

    coord_* is THE GATE (n_at_lo / n_at_hi over n_beta).  rec_* is printed
    beside it and never gates.  CORRECTIONS 117.1."""
    p = os.path.join(probe_dir, "probe.jsonl")
    if not os.path.exists(p):
        return None, None, None, None, 0
    recs = []
    with open(p, "r", errors="replace") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            try:
                recs.append(json.loads(line))
            except Exception:
                continue
    if not recs:
        return None, None, None, None, 0
    tail = recs[int(len(recs) * (1.0 - last_frac)):]
    cl, ch, rl, rh = [], [], [], []
    for r in tail:
        f = rail_fracs(r, lo, hi)
        if f is None:
            continue
        cl.append(f[0]); ch.append(f[1]); rl.append(f[2]); rh.append(f[3])
    if not cl:
        return None, None, None, None, len(recs)
    n = float(len(cl))
    return (sum(cl) / n, sum(ch) / n, sum(rl) / n, sum(rh) / n, len(recs))


def run_name(short, ms, seed):
    return "%s-%s-%s-s%d" % (TAG, short, MS_TAG[ms], seed)


def probe_name(short, ms, seed):
    return "probe_%s_%s_%s_s%d" % (short, MS_TAG[ms], TAG, seed)


def load_cell(runs_dir, probes_dir, short, ms, lo, hi):
    """Return {seed: dict(plateau5, rec_lo, rec_hi, boxfree, epochs, truncated)}."""
    out = {}
    for s in SEEDS:
        rn = run_name(short, ms, s)
        cands = []
        if os.path.isdir(runs_dir):
            for fn in os.listdir(runs_dir):
                if fn.startswith(rn) and fn.endswith(".out"):
                    cands.append(os.path.join(runs_dir, fn))
        if not cands:
            continue
        p5, nep = plateau5_from_out(sorted(cands)[-1])
        if p5 is None or nep < EPOCHS:
            out[s] = dict(plateau5=p5, epochs=nep, coord_lo=None, coord_hi=None,
                          rec_lo=None, rec_hi=None, boxfree=False, truncated=True)
            continue
        pd = os.path.join(probes_dir or runs_dir, probe_name(short, ms, s))
        clo, chi, rlo, rhi, nrec = read_probe_occupancy(pd, lo, hi)
        boxfree = (clo is not None and chi is not None
                   and clo < BOX_GATE and chi < BOX_GATE)
        out[s] = dict(plateau5=p5, epochs=nep, coord_lo=clo, coord_hi=chi,
                      rec_lo=rlo, rec_hi=rhi,
                      n_records=nrec, boxfree=boxfree, truncated=False)
    return out

def band_down(d):
    """The band for D_own, the matched-optima partition gap."""
    if d >= DECIDE:
        return "SURVIVES TUNING"
    if d > NULL_HALF:
        return "ATTENUATED"
    if d >= -NULL_HALF:
        return "NULL AT MATCHED OPTIMA"
    if d > -DECIDE:
        return "ATTENUATED-NEGATIVE"
    return "REVERSES"


def band_profile(means, ses_gap):
    """means: [(ms, value)] in MS_LIST order, value None if the cell is void.
    Returns (verdict, argmax_ms, bracketed) where `bracketed` says whether the
    argmax is enclosed by measured rungs on BOTH sides.

    CORRECTIONS 117.7.  The build returned an argmax on every branch and nothing
    downstream gated on EDGE-LOW / EDGE-HIGH, so an unbracketed argmax still
    printed "RULE 11 is closed ... the objection cannot be raised again".  An
    unbracketed argmax does not locate an optimum; it locates the edge of the
    grid.  `bracketed` is now returned and every RULE-11 sentence is gated on it.
    FLAT-PROFILE counts as bracketed ONLY in the sense that the stepsize cannot
    have decided the comparison over the measured range -- the caller says so in
    those words and does not say "closed"."""
    if any(v is None for _, v in means):
        return "UNRESOLVED (a cell is box-void or missing)", None, False
    vals = [v for _, v in means]
    span = max(vals) - min(vals)
    if span < ses_gap:
        return ("FLAT-PROFILE (span %.3f < %.3f pp, the resolution -- over the "
                "measured range the stepsize cannot have decided the comparison)"
                % (span, ses_gap)), means[vals.index(max(vals))][0], True
    i = vals.index(max(vals))
    if i == 0:
        return ("EDGE-LOW (argmax at the bottom rung, NOT bracketed below -- the "
                "optimum is at or below the grid floor and is NOT located)"), \
            means[0][0], False
    if i == len(vals) - 1:
        return ("EDGE-HIGH (argmax at the top rung, NOT bracketed above.  Above "
                "%.3g the box cannot be free at this budget, so the upper bracket "
                "is an ARITHMETIC bound, not a measurement)" % MS_MAX_FREE), \
            means[-1][0], False
    return "INTERIOR (argmax bracketed on both sides)", means[i][0], True

        # POST-SELECTION se.  CORRECTIONS 117.7 note: the se above is the
        # UNCONDITIONAL two-cell se and treats the selected cells as fixed.  When
        # two arms each independently select a rung from a profile whose true gaps
        # are below resolution, D_own's sampling distribution is a MIXTURE over
        # which pair of rungs got selected, and its spread exceeds that se.  A
        # small parametric bootstrap over the selection is reported beside it;
        # the fixed-cell figure remains PRIMARY (it is what the published D
        # readings are), and a reader who recomputes gets both.
        try:
            import random as _rnd
            _rnd.seed(20260830)
            mu = {k: pool.mean((k, m)) for k in (D_PAIR[0], D_PAIR[1])
                  for m in MS_LIST if (k, m) in cells}
            have = all((k, m) in cells for k in D_PAIR for m in MS_LIST)
            if have:
                sdc = pool.sd if pool.sd == pool.sd else SD_PLAN
                sec = sdc / math.sqrt(N_PER_CELL)
                draws = []
                for _ in range(20000):
                    pick = {}
                    for k in D_PAIR:
                        vals = [(pool.mean((k, m)) + _rnd.gauss(0.0, sec), m)
                                for m in MS_LIST]
                        pick[k] = max(vals)[0]
                    draws.append(pick[D_PAIR[0]] - pick[D_PAIR[1]])
                mb = sum(draws) / len(draws)
                sb = math.sqrt(sum((x - mb) ** 2 for x in draws) / (len(draws) - 1))
                P("    POST-SELECTION se (parametric bootstrap over the argmax, 20,000")
                P("    draws at the MEASURED pooled sd): %.3f against the fixed-cell"
                  % sb)
                P("    %.3f -- inflation %.2fx.  The fixed-cell figure stays PRIMARY;"
                  % (se, sb / se if se else float("nan")))
                P("    this one is what a reader recomputing the selection would get.")
            else:
                P("    POST-SELECTION se: NOT COMPUTED -- a cell is void, so the")
                P("    selection distribution is not defined.  Disclosed, not skipped.")
        except Exception as _e:
            P("    POST-SELECTION se: NOT COMPUTED (%s).  Disclosed, not skipped."
              % type(_e).__name__)
        b = band_down(d)
        # ANTI-OVERCLAIM: a refutation-shaped band may be DECLARED only if D_own is
        # separated from the WITHIN-BATCH shared-ms D at |t| >= 2.
        ks = (D_PAIR[0], MS_SHARED), (D_PAIR[1], MS_SHARED)
        dsh = dsh_se = float("nan")
        if ks[0] in cells and ks[1] in cells:
            dsh, dsh_se, _ = pool.contrast({ks[0]: 1.0, ks[1]: -1.0})
        dcoef = dict(coef)
        for k, c in ({ks[0]: -1.0, ks[1]: 1.0}).items():
            dcoef[k] = dcoef.get(k, 0.0) + c
        dcoef = {k: c for k, c in dcoef.items() if abs(c) > 1e-12}
        if dcoef:
            dd, dd_se, dd_df = pool.contrast(dcoef)
        else:
            dd, dd_se, dd_df = 0.0, 0.0, pool.df
        # THE ANTI-OVERCLAIM CLAUSE.  A refutation-shaped band may be DECLARED only
        # if D_own is separated from THIS BATCH's own shared-ms D at |t| >= 2.
        # THE VACUOUS CASE IS NOT AN UNDERPOWERED CASE and is not routed through the
        # UNDERPOWERED verdict: when both arms' argmax IS the shared rung, D_own and
        # D(shared) are the SAME NUMBER, there is no tuning move to be underpowered
        # against, and whatever band D_own takes is simply the band -- as a statement
        # about D IN THIS BOX, which is not the box any published D was measured in.
        vacuous = not dcoef
        declarable = (b == "SURVIVES TUNING") or vacuous \
            or (dd_se > 0 and abs(dd / dd_se) >= ANTI_OVERCLAIM_T)
        if declarable:
            P("    VERDICT: %s" % b)
            if vacuous and b != "SURVIVES TUNING":
                P("    **AND IT IS A STATEMENT ABOUT D IN THE BOX %s, NOT A REFUTATION" % CLIP)
                P("    OF THE PUBLISHED D.**  Both arms' argmax is the shared rung, so no")
                P("    tuning move was available and RULE 11 cannot be what moved D here.")
                P("    A low D_own in this box against a high D at -15/-25 is a BOX")
                P("    finding, and fa1 already established that this class of move")
                P("    changes the optimiser.  It would need its own batch to pursue.")
            res["H2"] = (b, d)
        else:
            P("    VERDICT: CONSISTENT-WITH-SHARED-ms (UNDERPOWERED).  The point")
            P("    estimate bands %s, but it is NOT separated from this batch's own" % b)
            P("    shared-ms D at |t| >= %.1f, so THIS MAY NOT BE WRITTEN UP AS A"
              % ANTI_OVERCLAIM_T)
            P("    REFUTATION.  At n=%d the smallest declarable move in D is %.3f pp."
              % (N_PER_CELL, DD_DECLARABLE))
            res["H2"] = ("CONSISTENT-WITH-SHARED-ms (UNDERPOWERED)", d)
        # the registered refutation, stated as its own line
        P("    REGISTERED REFUTATION: D_own <= %+0.2f (declarable), OR nodewise at its"
          % NULL_HALF)
        P("    own argmax >= chunk777 at its own argmax.  %s"
          % ("REFUTATION TRIGGERED." if (d <= NULL_HALF and declarable) or d <= 0
             else "NOT triggered."))
        # --- H3: the RULE-11 delta ----------------------------------------
        P("")
        P("H3  PRIMARY.  dD = D_own - D(ms=%s), WITHIN BATCH.  D(%s) is the rung this"
          % (MS_SHARED, MS_SHARED))
        P("    batch shares with the published readings, so dD is the RULE-11 delta")
        P("    with the box, the batch and the seeds all held fixed.")

            P("    D(ms=%s) = %+0.3f  se %.3f" % (MS_SHARED, dsh, dsh_se))

            pass

                P("    'D is unchanged at matched optima to within %.3f pp'." % (ANTI_OVERCLAIM_T * dd_se))

            elif dd <= -DECIDE:

                P("    resolvably when each arm is tuned.  THE HEADLINE MUST BE REWRITTEN")
                P("    as 'at a fixed shared meta-stepsize', which is a much weaker claim.")
                res["H3"] = ("SHARED-ms ARTEFACT", dd)
            else:
                P("    VERDICT: D MOVES RESOLVABLY WITH TUNING (%+0.3f) but not downward"
                  % dd)
                P("    past the -%.2f bar.  Report both D_own and D(%s) side by side."
                  % (DECIDE, MS_SHARED))
                res["H3"] = ("D MOVES, NOT REFUTED", dd)

    # --- H4: SECONDARY -- D and G per rung ---------------------------------
    P("")
    P("H4  SECONDARY.  D and G AT EACH RUNG SEPARATELY, all within batch.")
    P("    D = chunk777 - nodewise      count-matched +1 group  (14,421 vs 14,420)")
    P("    G = chunk2325 - nodewise1d   count-matched EXACTLY   ( 4,851 vs  4,851)")
    dvals = []
    for ms in MS_LIST:
        for leg, (a, b) in (("D", D_PAIR), ("G", G_PAIR)):
            ka, kb = (a, ms), (b, ms)
            if ka not in cells or kb not in cells:
                P("    %s @ ms=%-5s VOID (a cell is box-void or missing)" % (leg, ms))
                continue
            v, se, df = pool.contrast({ka: 1.0, kb: -1.0})
            wv, wse, wdf = welch(cells[ka], cells[kb])
            P("    %s @ ms=%-5s = %+0.3f  se %.3f  t %.2f (%d df)   [Welch sanity"
              " %+0.3f se %.3f]" % (leg, ms, v, se, v / se if se else float("nan"),
                                    df, wv, wse))
            if leg == "D":
                dvals.append((ms, v))
    if len(dvals) >= 2:
        dec = math.log10(float(dvals[-1][0]) / float(dvals[0][0]))
        P("    ms-SENSITIVITY OF D across %.3f decades of meta-stepsize: %+0.3f -> %+0.3f"

        P("    = %+0.3f pp per decade of ms, measured WITHIN ONE BATCH.  (The corpus"

            P("    cannot adjudicate two cells that share a box with each other.")

        P("    rl3's own ms=1e-3 nodewise cell is VOID -- the split stays unadjudicated.")

    P("H6  DESCRIPTIVE ONLY.")

    P("    T = nodewise1d - nodewise spans %d -> %d groups = %.4f decades and is"
      % (ARMS["node"][1], ARMS["n1d"][1], math.log10(ARMS["node"][1] / float(ARMS["n1d"][1]))))
    P("    COUNT-CONFOUNDED BY CONSTRUCTION.  T IS NOT TAIL REMOVAL and this scorer")
    P("    refuses to label it so; the tail-removal statistic is G, which is")
    P("    count-matched exactly.  Report T only as T = (D - G) + U.")
    for ms in MS_LIST:
        ka, kb = ("n1d", ms), ("node", ms)
        if ka in cells and kb in cells:
            v, se, df = pool.contrast({ka: 1.0, kb: -1.0})
            P("      T @ ms=%-5s = %+0.3f  se %.3f  (DESCRIPTIVE)" % (ms, v, se))
    P("    SIGN TESTS are omitted: at n=%d a 2-of-2 agreement has exact p=0.25 and"
      % N_PER_CELL)
    P("    carries no evidence.  Seed is statistically NULL on this cluster anyway.")

    P("")

    P("moments).  Direction C was dropped on its own pre-registered adoption gate.")

    P("RULE-11 result on ResNet34 or CIFAR-100, where no ladder exists; any")
    P("resolution of the gc-/flat split, which rl3 can only add a third reading to.")
    return res

def selftest():

        else:

        "the minimum declarable move in D must be 2*sd*sqrt(4/n) = %.5f"

        "resolving +-0.15 must be shown UNAFFORDABLE at this design (n=%.1f/cell)"

    chk(band_down(-0.15) == "NULL AT MATCHED OPTIMA"

    chk(band_down(-0.30) == "REVERSES" and band_down(-0.2999) == "ATTENUATED-NEGATIVE",

    flat = [("1e-4", 92.10), ("3e-4", 92.20)]
    v, am, brk = band_profile(flat, MIN_GAP)
    chk(v.startswith("FLAT-PROFILE"),
        "a 0.10 pp span must band FLAT-PROFILE at a %.3f pp resolution" % MIN_GAP)
    chk(brk is True,
        "a FLAT profile counts as bracketed -- the stepsize cannot have decided "
        "the comparison over the measured range")
    high = [("1e-4", 92.00), ("3e-4", 93.00)]
    v, am, brk = band_profile(high, MIN_GAP)
    chk(v.startswith("EDGE-HIGH") and am == "3e-4" and brk is False,

    low = [("1e-4", 93.00), ("3e-4", 92.00)]

    chk(float(MS_LIST[-1]) < MS_MAX_FREE < 1e-3,
        "the top rung must sit under the free limit and the limit under 1e-3")
    chk(math.exp(math.log(float(ALPHA0)) + 1e-3 * EPOCHS * STEPS_PER_EPOCH) > 1e15,
        "a ceiling that made ms=1e-3 free would imply a step size above 1e15 -- "
        "which is why the upper bracket is arithmetic and not purchasable")
    # the box must be strictly WIDER on BOTH rails than the anchors' box
    chk(lo < -15.0 and hi > -2.3026,
        "CLIP must be strictly wider than the anchors' -15:-2.3026 on BOTH rails")

    # --- COUNT MATCHING.  The G leg is EXACT; the D leg is +1 group. ---------
    chk(ARMS["c23"][1] == ARMS["n1d"][1],
        "the G leg must be an EXACT count match (4,851 vs 4,851)")
    chk(ARMS["ch7"][1] - ARMS["node"][1] == 1, "the D leg must be +1 group")
    # +1 group on 14,420 is negligible under ANY slope the record carries.
    dec_d = math.log10(ARMS["ch7"][1] / float(ARMS["node"][1]))
    for slope in (-0.407, -0.4906, -0.51, -0.5361, -0.213, -0.5549):
        chk(abs(slope * dec_d) < 0.005,
            "the D-leg count bias must be under 0.005 pp at slope %.4f" % slope)
    chk(abs(math.log10(ARMS["node"][1] / float(ARMS["n1d"][1])) - 0.4732) < 5e-4,
        "T must span 0.4732 decades -- the count confound it is refused for")

    # --- RUN AND PROBE NAMING must be unique across the 12 cells -------------
    names = {run_name(a, m, s) for a in ARMS for m in MS_LIST for s in SEEDS}
    probes = {probe_name(a, m, s) for a in ARMS for m in MS_LIST for s in SEEDS}
    chk(len(names) == 24 and len(probes) == 24,
        "all 24 run names and 24 probe dirs must be distinct")
    chk(all(run_name(a, m, s) in bs or MS_TAG[m] in bs
            for a in ARMS for m in MS_LIST for s in SEEDS),
        "the batch script must build the same ms tags this scorer reads")

    # --- HONESTY CLAUSES, asserted against this file's own text --------------
    for phrase, why in (
            ("CONSISTENT-WITH-SHARED-ms (UNDERPOWERED)", "the anti-overclaim verdict"),
            ("FLAT-PROFILE", "a flat profile must be a PRIMARY-ADMISSIBLE answer"),
            ("MAX-SELECTION BIAS", "the argmax selection bias must be disclosed"),
            ("CROSS-BOX POOLING IS FORBIDDEN", "the box restriction must be printed"),
            ("NO BOX IS PROVABLY FREE", "the ms=1e-3/3e-3 box limit must be printed"),
            ("T IS NOT TAIL REMOVAL", "T must be refused the tail-removal label"),
            ("FIRST-CLASS", "rec_hi must be first-class, not a drop criterion"),
            ("RULE 11 IS CLOSED BY THIS BATCH ON R18/CIFAR-10 ONLY",
             "the scope of the RULE-11 certificate must be printed"),
            ("40 of 100 epochs", "the brief's mis-derived 1.19 pp must be corrected"),
            ("stays OPEN on", "R34 and C100 must be declared still open")):
        chk(phrase in me, "%s (%r)" % (why, phrase))

    # --- the frozen gate constants -------------------------------------------
    chk(MIN_BOXFREE_CELL == 2 and MIN_BOXFREE_CELL <= len(SEEDS),
        "a cell needs at least %d box-free seeds to be usable" % MIN_BOXFREE_CELL)
    chk(BOX_ASYM == 0.10, "the arm-asymmetry bar is 0.10 absolute")
    chk(BOX_GATE == 0.05, "the per-seed occupancy gate is 5% of records")
    chk(len(SEEDS) == 3 and len(ARMS) == 4 and len(MS_LIST) == 2,
        "%d arms x %d stepsizes x %d seeds = %d jobs"
        % (len(ARMS), len(MS_LIST), len(SEEDS),
           len(ARMS) * len(MS_LIST) * len(SEEDS)))
    chk(MS_SHARED in MS_LIST, "the shared-ms rung must be one of the rungs")
    chk(all(m in ("1e-4", "3e-4") for m in MS_LIST),
        "BOTH rungs must be ones a published D was measured at: 1e-4 is "
        "mm1/pp1/cc1's and 3e-4 is ar1/fa1's")
    chk(sorted(MS_TAG) == sorted(MS_LIST), "every stepsize must have a filename tag")

    print("selftest: %d/%d PASS" % (ok, ok + fail) if fail == 0
          else "selftest: %d PASS, %d FAIL" % (ok, fail))
    return 0 if fail == 0 else 1


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runs", default="runs", help="directory holding rl3-*.out "
                    "(the .out files land in the ROOT of runs/)")
    ap.add_argument("--probes", default=None,
                    help="directory holding the probe_* dirs; defaults to "
                         "<runs>/%s, which is where the runner puts them" % TAG)
    a = ap.parse_args()
    if a.selftest:
        return selftest()
    if not os.path.isdir(a.runs):
        print("no rl3 runs at %s -- this batch has NOT been submitted.  Nothing to "
              "score, and that is the correct state before the operator's review."
              % a.runs)
        return 0
    probes = a.probes
    if probes is None:
        cand = os.path.join(a.runs, TAG)
        probes = cand if os.path.isdir(cand) else a.runs
    score(a.runs, probes)
    return 0


if __name__ == "__main__":
    sys.exit(main())
