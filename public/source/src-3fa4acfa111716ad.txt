c98_figures.py -- the paper's four figures, on ONE palette, from the CSV of record.

    python3 analysis/c98_figures.py --all                 # writes paper/figures/*.pdf + *.png
    python3 analysis/c98_figures.py --numbers             # prints every number it plots
    python3 analysis/c98_figures.py --fig f1 f3           # a subset

DESIGN RULES THIS SCRIPT ENFORCES, so a figure cannot silently break a house rule:

  R-A  plateau5 (mean of the last 5 test epochs) is the ONLY accuracy metric read.
       The CSV `plateau` column (mean of the last 20) is never touched.
  R-B  Admissibility is `window_ok == 1 AND complete == 1 AND plateau5 present`,
       applied before any arm mean is formed.
  R-C  Every contrast is WITHIN one batch. No cross-batch difference is ever drawn.
  R-D  CIFAR-10 and CIFAR-100 D's are NEVER placed on one percentage-point axis
       (they sit on ~7-8 pp and ~29-30 pp error budgets respectively).  Panel (a)
       of F1 therefore carries a hard axis break and panel (b) gives the
       commensurable quantity, D / headroom.

       number of distinct groups.  This is what makes ml2 a 3 v 3 cell, not 6 v 6.
  R-F  Nothing is hard-coded.  Every plotted value is re-derived here from
       results/all_runs.csv or, for the within-run budget series, from the raw
       per-epoch `.out` files.  --numbers prints them all for audit.

Palette: Okabe-Ito, colour-blind safe, identical across all four figures.
Colour encodes the BASE OPTIMISER; marker shape encodes the DATASET.

def _num(r, k):
    try:    return float(r[k])
    except (TypeError, ValueError, KeyError): return None

def admissible(r):
    return (r.get("window_ok") == "1" and r.get("complete") == "1"
            and _num(r, "plateau5") is not None)

def load(path=CSV):
    rows = list(csv.DictReader(open(path)))
    return rows, [r for r in rows if admissible(r)]

# The 18 collapsed pairs registered by the ARGS audit (docs/ARGS-AUDIT.md §5).
# analysis/args_repair.py writes them into the CSV's `dup_group` column; until it
# has been run with --apply the column is empty, so we fall back to the SAME
# registered table rather than silently over-counting.  One source of truth, and
# the result is identical before and after the patch is applied.
def _pending_dupes():
    try:
        sys.path.insert(0, HERE)
        from args_repair import GROUPS
        return {run: name for name, runs in GROUPS for run in runs}
    except Exception:
        return {}
PENDING_DUP = _pending_dupes()

def arm(rows, prefix, gran):
    """Admissible plateau5 values of one arm, with R-E dup_group collapsing.

    Returns (values, n_groups).  Rows carrying a dup_group -- from the CSV column
    if the repair has been applied, otherwise from the registered table above --
    are averaged WITHIN the group first, so a batch that ran one configuration
    twice under two names contributes ONE number, not two.  This is what makes
    ml2 a 3 v 3 cell (se 0.195) and not a 6 v 6 one (se 0.142).
    """
    sel = [r for r in rows if r["run"].startswith(prefix) and r["granularity"] == gran]
    groups, singles = {}, []
    for r in sel:
        dg = (r.get("dup_group") or "").strip() or PENDING_DUP.get(r["run"], "")
        if dg: groups.setdefault(dg, []).append(float(r["plateau5"]))
        else:  singles.append(float(r["plateau5"]))
    vals = singles + [st.mean(v) for v in groups.values()]
    return vals, len(vals)

def welch(a, b):
    d  = st.mean(a) - st.mean(b)
    se = math.sqrt(st.variance(a) / len(a) + st.variance(b) / len(b))
    return d, se, (d / se if se else float("nan"))

def cells(rows):
    out = []
    for (lab, net, ds, base, eta, ep, ch, nd, c23, n1d) in CELLS:
        if lab == GN_CELL and not WITH_GN: continue
        cv, cn = arm(rows, *ch); nv, nn = arm(rows, *nd)
        if not cv or not nv:
            sys.stderr.write("MISSING arm for cell %s -- skipped\n" % lab); continue
        d, se, t = welch(cv, nv)
        rec = dict(label=lab, network=net, dataset=ds, base=base, eta=eta, epochs=ep,
                   n=(cn, nn), D=d, seD=se, tD=t,
                   aligned=st.mean(nv), uniform=st.mean(cv),
                   headroom=100.0 - st.mean(nv), G=None, seG=None, tG=None,
                   DG=None, seDG=None, tDG=None, in12=(lab in POOL12))
        rec["rel"] = d / rec["headroom"]
        if c23 and n1d:
            gv, gn_ = arm(rows, *c23); hv, hn = arm(rows, *n1d)
            if gv and hv:
                g, gse, gt = welch(gv, hv)
                rec.update(G=g, seG=gse, tG=gt, DG=d - g,
                           seDG=math.sqrt(se**2 + gse**2))
                rec["tDG"] = rec["DG"] / rec["seDG"]
        out.append(rec)
    return out

# ---- DerSimonian-Laird, used by F2 -----------------------------------------
def meta(items):
    """items :: [(y, se)] -> (fixed-effect pool, se, Q, df, tau)."""
    w = [1.0 / se**2 for _, se in items]
    y = [v for v, _ in items]
    W = sum(w)
    m = sum(wi * yi for wi, yi in zip(w, y)) / W
    Q = sum(wi * (yi - m)**2 for wi, yi in zip(w, y))
    k = len(items)
    den = W - sum(wi**2 for wi in w) / W
    tau2 = max(0.0, (Q - (k - 1)) / den) if den > 0 else 0.0
    return m, 1.0 / math.sqrt(W), Q, k - 1, math.sqrt(tau2)

# ---- raw per-epoch series, used by F3 --------------------------------------
def series(pattern):
    """{seed: [test acc per epoch]} from the raw .out files, searched recursively."""
    out = {}
    for d in RUNDIRS:
        for f in glob(os.path.join(d, pattern)) + glob(os.path.join(d, "**", pattern),
                                                       recursive=True):
            m = re.search(r"-s(\d+)-\d+\.out$", f)
            if not m: continue
            txt = open(f, errors="replace").read()
            out[int(m.group(1))] = [float(x) for x in
                                    re.findall(r"Test Accuracy: ([0-9.]+)", txt)]
    return out

def pl5(v, B):
    """plateau5 evaluated at budget B: the mean of test epochs B-5 .. B-1."""
    return sum(v[B-5:B]) / 5.0

def budget_table():
    ch, nd = series("hz3-ch-s*.out"), series("hz3-node-s*.out")
    seeds  = sorted(set(ch) & set(nd))
    return ch, nd, seeds

def chi2_sf(x, k):
    """Upper tail of chi-square with k df, k >= 1, no scipy."""
    if k <= 0: return float("nan")
    if k % 2 == 0:                       # even df: closed form
        m = k // 2
        s, term = 0.0, 1.0
        for i in range(m):
            if i: term *= (x / 2.0) / i
            s += term
        return math.exp(-x / 2.0) * s
    # odd df: series for the regularised lower incomplete gamma
    a, xx = k / 2.0, x / 2.0
    if xx < a + 1:
        term, s, n = 1.0 / a, 1.0 / a, 0
        while abs(term) > 1e-14 * abs(s) and n < 10000:
            n += 1; term *= xx / (a + n); s += term
        return 1.0 - s * math.exp(-xx + a * math.log(xx) - math.lgamma(a))
    b, c, d, h = xx + 1 - a, 1e300, 1.0 / (xx + 1 - a), 1.0 / (xx + 1 - a)
    for i in range(1, 10000):
        an = -i * (i - a); b += 2
        d = an * d + b;  d = d if abs(d) > 1e-300 else 1e-300
        c = b + an / c;  c = c if abs(c) > 1e-300 else 1e-300
        d = 1.0 / d; de = d * c; h *= de
        if abs(de - 1.0) < 1e-14: break
    return math.exp(-xx + a * math.log(xx) - math.lgamma(a)) * h

def fig3(rows, cs, numbers=False):
    ch, nd, seeds = budget_table()
    if not seeds:
        sys.stderr.write("F3: no hz3 .out found under %s -- skipped\n" % RUNDIRS); return None
    Bs = [100, 200, 300]
    # which seeds are box-matched?  read the CSV's own beta_clip per run.
    box = {}
    for r in rows:
        m = re.match(r"hz3-(ch|node)-s(\d+)$", r["run"])
        if m: box.setdefault(int(m.group(2)), {})[m.group(1)] = r["beta_clip"]
    mismatch = sorted(s for s, v in box.items() if len(v) == 2 and v["ch"] != v["node"])
    keep = [s for s in seeds if s not in mismatch]
    # hz3q: the box- and class-matched re-run of the mismatched seed.  It REPLACES
    # that seed and adds none, so the repaired pool is the same six seeds.  With the
    # hz3q .out files absent the panel degrades to the two archived readings, which
    # is exactly what it drew before the repair existed.
    qch, qnd = series("hz3q-ch-s*.out"), series("hz3q-node-s*.out")
    rep = sorted(set(qch) & set(qnd) & set(mismatch))
    RCH, RND = (dict(ch), dict(nd)) if rep else (None, None)
    for s5 in rep:
        RCH[s5], RND[s5] = qch[s5], qnd[s5]

    fig, (axL, axR) = plt.subplots(1, 2, figsize=(7.4, 3.5),
                                   gridspec_kw=dict(width_ratios=[1.30, 1.0], wspace=0.42))
    col = BASE_C["SGDm"]

    # ---- left: per-seed within-run traces + the arm-set means
    for s in seeds:
        ys = [pl5(ch[s], B) - pl5(nd[s], B) for B in Bs]
        bad = s in mismatch
        axL.plot(Bs, ys, color=("#B0392B" if bad else col), lw=(1.1 if bad else 1.0),
                 alpha=(0.95 if bad else 0.28),
                 ls=("--" if bad else "-"), marker="o", ms=3.0, zorder=(4 if bad else 2),
                 label=("seed %d, archived: cross-box, cross-class" % s) if bad else None)
    for s5 in rep:
        ys = [pl5(RCH[s5], B) - pl5(RND[s5], B) for B in Bs]
        axL.plot(Bs, ys, color=REPC, lw=1.1, alpha=0.95, ls=(0, (1.2, 1.2)),
                 marker="o", ms=3.0, zorder=5,
                 label="seed %d, repaired: same box, same class" % s5)
    lines = [("archived, all %d seeds" % len(seeds), (ch, nd), seeds, col, "-"),
             ("archived, box-matched %d" % len(keep), (ch, nd), keep, "#00507F", (0, (5, 2)))]
    if rep:
        lines.append(("repaired, %d seeds" % len(seeds), (RCH, RND), seeds, REPC, (0, (1.2, 1.2))))
    for lab, (C_, N_), ss, c_, ls_ in lines:
        mu = [st.mean([pl5(C_[s], B) - pl5(N_[s], B) for s in ss]) for B in Bs]
        se = [st.stdev([pl5(C_[s], B) - pl5(N_[s], B) for s in ss]) / math.sqrt(len(ss)) for B in Bs]
        axL.errorbar(Bs, mu, yerr=[1.96 * x for x in se], color=c_, lw=2.0, ls=ls_,
                     marker="s", ms=5.0, capsize=3, capthick=1.0, zorder=6, label=lab)
    axL.axhline(0, color=INK, lw=0.8)
    axL.set_xticks(Bs); axL.set_xlabel("training budget (epochs)")
    axL.set_ylabel("D, paired WITHIN run  (pp)")
    axL.set_title("(a)  D across the budget, hz3, within run", loc="left", pad=7)
    axL.yaxis.grid(True, color=GRID, lw=0.6); axL.set_axisbelow(True)
    axL.set_ylim(-0.02, 1.62)
    axL.set_xlim(88, 312)
    h, l = axL.get_legend_handles_labels()
    order = [l.index(x) for x in sorted(l, key=lambda z: (not z.startswith(("archived,", "repaired,")), l.index(z)))]
    axL.legend([h[i] for i in order], [l[i] for i in order],

               handlelength=2.1, borderpad=0.5, labelspacing=0.35, fontsize=6.6,
               bbox_to_anchor=(0.50, 1.005))

    # ---- right: the slope, both ways
    rowsD = []
    slope_src = [("repaired, %d seeds" % len(seeds), (RCH, RND), seeds, REPC)] if rep else []
    slope_src += [("archived, box-matched %d" % len(keep), (ch, nd), keep, "#00507F"),
                  ("archived, all %d seeds" % len(seeds), (ch, nd), seeds, col)]
    for lab, (C_, N_), ss, c_ in slope_src:
        dd = [(pl5(C_[s], 300) - pl5(N_[s], 300)) - (pl5(C_[s], 100) - pl5(N_[s], 100))
              for s in ss]
        m_, se_ = st.mean(dd), st.stdev(dd) / math.sqrt(len(dd))
        rowsD.append((lab, m_, se_, m_ / se_, len(dd), c_))

                     capthick=1.0, marker="o", ms=6.0, mec="white", mew=0.8, zorder=4)
        axR.text(m_, i + 0.26, "%+.3f ± %.3f   t %.2f   (n=%d)" % (m_, se_, t_, n_),

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--all", action="store_true")
    ap.add_argument("--fig", nargs="*", default=[])
    ap.add_argument("--numbers", action="store_true")
    ap.add_argument("--csv", default=CSV)
    ap.add_argument("--outdir", default=FIGDIR)
    ap.add_argument("--with-gn", action="store_true",
                    help="re-include the GroupNorm cell that R0 item 1 removes")
    a = ap.parse_args()
    global WITH_GN
    WITH_GN = a.with_gn
    want = list(FIGS) if (a.all or not a.fig) else a.fig

    rows, adm = load(a.csv)
    cs = cells(adm)
    print("csv %s -- %d rows, %d admissible, %d count-matched cells"
          % (os.path.relpath(a.csv, ROOT), len(rows), len(adm), len(cs)))
    os.makedirs(a.outdir, exist_ok=True)
    for k in want:
        stem, fn = FIGS[k]
        fig = fn(rows, cs, numbers=a.numbers)
        if fig is None: continue
        for ext in ("pdf", "png"):
            p = os.path.join(a.outdir, "%s.%s" % (stem, ext))
            # deterministic output: no embedded build timestamp, so the artefact's
            # MANIFEST.md5 still verifies after `make figures` regenerates them.
            meta = {"CreationDate": None} if ext == "pdf" else {"Software": None}
            fig.savefig(p, metadata=meta)
            print("wrote %s" % os.path.relpath(p, ROOT))
        plt.close(fig)

if __name__ == "__main__":
    main()
