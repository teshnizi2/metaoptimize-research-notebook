  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"

  || guard_fail "guard 1c4: fresh-seed check did not pass"

   && grep -q "PASS D3(b) masked tensors" "$ILOG"; then

          "$BN_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$RUNNER_SHA" "$STAGEW" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"
