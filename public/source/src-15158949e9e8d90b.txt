# the `sets:` grammar, re-implemented here for --selftest ONLY.  It never runs
# at score time: G1 reads what the LIVE model actually composed, out of the
# manifest the launcher wrote from `HF.polish_the_stepsize_groups` itself.

_NS_INT = re.compile(r"^\d+$")
_NS_RANGE = re.compile(r"^(\d+)-(\d+)$")


def spec_to_groups(spec, names=MODEL_NAMES):
    """`sets:...` -> [[1-based index, ...], ...].  Mirrors PATCH_NAMESETS."""
    if not spec.startswith("sets:"):
        raise ValueError("not a sets: spec: %r" % spec)
    by_name = dict((n, i + 1) for i, n in enumerate(names))
    T = len(names)
    out, owner = [], {}
    for gi, g in enumerate(spec[len("sets:"):].split("/")):
        if not g:
            raise ValueError("empty group %d in %r" % (gi, spec))
        idx = []
        for item in g.split(","):
            m = _NS_RANGE.match(item)
            if m:
                lo, hi = int(m.group(1)), int(m.group(2))
                if not (1 <= lo <= hi <= T):
                    raise ValueError("range %r outside 1..%d" % (item, T))
                got = list(range(lo, hi + 1))
            elif _NS_INT.match(item):
                v = int(item)
                if not (1 <= v <= T):
                    raise ValueError("index %r outside 1..%d" % (item, T))
                got = [v]
            elif item in by_name:
                got = [by_name[item]]
            else:
                raise ValueError("%r is not an index, a range or a name" % item)
            for i in got:
                if i in owner:
                    raise ValueError("tensor %d claimed twice" % i)
                owner[i] = gi
            idx.extend(got)
        out.append(sorted(idx))
    if len(owner) != T:
        raise ValueError("%d tensors in no group" % (T - len(owner)))
    return out


def holes_of(coarse):
    s = set(coarse)
    return sorted(set(range(1, max(s) + 1)) - s)


def prefix_depth_of(coarse):
    s, k = set(coarse), 1
    while k in s:
        k += 1
    return k - 1


def tensor_class(name):
    """1 convolution | 2 normalisation scale | 0 normalisation shift."""
    if name.endswith("conv1.weight") or name.endswith("conv2.weight") \
            or ".shortcut.0." in name:
        return 1
    if name.endswith(".weight"):
        return 2
    return 0

def parse_argsline(line):
    try:
        toks = shlex.split(line)
    except ValueError:
        toks = line.split()
    occ, i, n = [], 0, len(toks)
    while i < n:
        t = toks[i]
        if t.startswith("--") and len(t) > 2:
            if "=" in t:
                f, v = t.split("=", 1)
                occ.append((f.lstrip("-"), v))
                i += 1
                continue
            f, vals, j = t.lstrip("-"), [], i + 1
            while j < n and not (toks[j].startswith("--") and len(toks[j]) > 2):
                vals.append(toks[j])
                j += 1
            occ.append((f, " ".join(vals)))
            i = j
            continue
        i += 1
    return occ


def tail5(d, budget):
    v = [d[e] for e in range(budget - 5, budget) if e in d]
    return (sum(v) / 5.0) if len(v) == 5 else None


def load_runs(runsdir):
    recs = []
    pats = [os.path.join(runsdir, PREFIX + "*.out"),
            os.path.join(runsdir, BATCH, PREFIX + "*.out")]
    files = sorted(set(sum([glob.glob(p) for p in pats], [])))
    for f in files:
        m = NAME_RE.match(os.path.basename(f))
        if not m:
            recs.append({"file": f, "bad_name": True})
            continue
        arm, seed, job = m.group(1), int(m.group(2)), m.group(3)
        te, tr, args, env, done = {}, {}, None, None, False
        with open(f, errors="ignore") as fh:
            for line in fh:
                a = ARGS_RE.match(line)
                if a and args is None:
                    args = a.group(1)
                e = ENV_RE.match(line)
                if e and env is None:
                    env = e.group(1).strip()
                if line.strip() == "RUN_DONE":
                    done = True
                x = EP_RE.search(line)
                if x:
                    te[int(x.group(1))] = float(x.group(2))
                y = EPTR_RE.search(line)
                if y:
                    tr[int(y.group(1))] = float(y.group(2))
        occ = parse_argsline(args) if args else []
        cnt = collections.Counter(f2 for f2, _ in occ)
        recs.append({
            "file": f, "bad_name": False, "arm": arm, "seed": seed, "job": job,
            "args": args, "env": env, "occ": occ, "eff": dict(occ),
            "repeated": sorted([k for k, v in cnt.items() if v > 1]),
            "run_done": done, "test": te, "train": tr, "n_ep": len(te),
            "p5": tail5(te, EPOCHS), "t5": tail5(tr, EPOCHS),
        })
    return recs

def _csv_rows(path=CSV, exclude_own=True):
    out = []
    with open(path) as fh:
        for r in csv.DictReader(fh):
            if (r.get("superseded") or "0") == "1":
                continue
            if exclude_own and str(r.get("run") or "").startswith(PREFIX):
                continue
            out.append(r)
    return out


def _batch_of(run):
    return re.split(r"[-_]", str(run))[0]


def _in_cell(r, epochs):
    return (r["network"] == NET and r["dataset"] == DSET
            and r["base"] == BASE_ALG and r["meta"] == META_ALG
            and r["meta_stepsize"] == MS and r["alpha0"] == ALPHA0
            and r["gamma"] == GAMMA and r["augment"] == AUG
            and r["beta_clip"] == CLIP_C and not (r.get("hier") or "").strip()
            and r["batch_size"] == str(BATCH_SIZE)
            and r["epochs_done"] == str(epochs)
            and r["collapsed"] == "0")


M2_NARROW = re.compile(r"^\[\d+,\d+\]$")


def _is_m2(g, wide):
    g = str(g or "")
    if M2_NARROW.match(g):
        return True
    if wide and g.startswith("sets:"):
        return g.count("/") == 1
    return False


def noise_floor_from_csv(path=CSV, epochs=EPOCHS, wide=False):
    cells = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not _in_cell(r, epochs) or not _is_m2(r["granularity"], wide):
            continue
        try:
            cells[(_batch_of(r["run"]), r["granularity"])].append(
                float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        nm += len(v)
        if len(v) >= 2:
            m = statistics.mean(v)
            ss += sum((x - m) ** 2 for x in v)
            df += len(v) - 1
            nc += 1
    return (math.sqrt(ss / df) if df else float("nan")), df, nc, nm


def batch_levels_from_csv(prefix, path=CSV):
    g = collections.defaultdict(list)
    for r in _csv_rows(path):
        if not str(r.get("run") or "").startswith(prefix):
            continue
        try:
            g[str(r["granularity"] or "")].append(float(r["plateau5"]))
        except (TypeError, ValueError):
            pass
    return dict((k, (statistics.mean(v), len(v))) for k, v in g.items())


def cell_granularity_levels(gran, path=CSV, epochs=EPOCHS):
    """Every same-cell row at one granularity, cpg1 EXCLUDED."""
    v, b = [], set()
    for r in _csv_rows(path):
        if _in_cell(r, epochs) and str(r["granularity"]) == gran:
            try:
                v.append(float(r["plateau5"]))
                b.add(_batch_of(r["run"]))
            except (TypeError, ValueError):
                pass
    return v, len(b)


def outs_100(prefix, armpat, runsdirs):
    """100-epoch arm means re-derived from RAW .out files.  {} when unreachable
    (a SKIP, never a FAIL)."""
    g, gt = collections.defaultdict(list), collections.defaultdict(list)
    seen = set()
    dirs = []
    for d in runsdirs:                 # de-duplicate: two spellings of the same
        rp = os.path.realpath(d)       # directory would double every file and
        if rp in seen:                 # silently turn a PASS into a SKIP
            continue
        seen.add(rp)
        dirs.append(d)
    for d in dirs:
        for f in glob.glob(os.path.join(d, prefix + "*.out")):
            m = re.match(armpat, os.path.basename(f))
            if not m:
                continue
            te, tr = {}, {}
            with open(f, errors="ignore") as fh:
                for line in fh:
                    x = EP_RE.search(line)
                    if x:
                        te[int(x.group(1))] = float(x.group(2))
                    y = EPTR_RE.search(line)
                    if y:
                        tr[int(y.group(1))] = float(y.group(2))
            v, w = tail5(te, 100), tail5(tr, 100)
            if v is not None:
                g[m.group(1)].append(v)
            if w is not None:
                gt[m.group(1)].append(w)
    return (dict((k, statistics.mean(v)) for k, v in g.items() if len(v) == 3),
            dict((k, statistics.mean(v)) for k, v in gt.items() if len(v) == 3))

def resolve_manifest(runsdir, explicit=None):
    if explicit:
        return (explicit, "given explicitly with --manifest")
    if not runsdir:
        return (None, "no runsdir")
    up = os.path.dirname(os.path.abspath(runsdir.rstrip("/")))
    cands = [(os.path.join(runsdir, BATCH, MANIFEST_NAME),
              "DEFAULT <runsdir>/%s/%s" % (BATCH, MANIFEST_NAME)),
             (os.path.join(runsdir, BATCH + "-" + MANIFEST_NAME),
              "DEFAULT <runsdir>/%s-%s" % (BATCH, MANIFEST_NAME)),
             (os.path.join(runsdir, MANIFEST_NAME),
              "DEFAULT <runsdir>/%s" % MANIFEST_NAME),
             (os.path.join(up, BATCH, MANIFEST_NAME),
              "DEFAULT <runsdir>/../%s/%s" % (BATCH, MANIFEST_NAME))]
    for p, how in cands:
        if os.path.exists(p):
            return (p, how)
    for pat in (os.path.join(runsdir, "*", MANIFEST_NAME),
                os.path.join(up, "*", BATCH, MANIFEST_NAME)):
        hits = sorted(g for g in glob.glob(pat) if BATCH in g)
        if len(hits) == 1:
            return (hits[0], "DEFAULT, found by glob %s" % pat)
    return (None, "NOT FOUND -- searched %s" % "; ".join(c[0] for c in cands))


def read_manifest(path):
    man = {"tensors": {}, "specs": {}, "raw": []}
    with open(path) as fh:
        for line in fh:
            line = line.rstrip("\n")
            man["raw"].append(line)
            p = line.split()
            if not p:
                continue
            if p[0] == "NUM_PARAM_TENSORS":
                man["T"] = int(p[1])
            elif p[0] == "TOTAL_PARAMS":
                man["total"] = int(p[1])
            elif p[0] == "TENSOR" and len(p) >= 4:
                man["tensors"][int(p[1])] = (p[2], int(p[3]))
            elif p[0] == "ARMSPEC":
                d, arm, key, i = {}, p[1], None, 2
                while i < len(p):
                    if p[i] in ("SPEC", "M", "SIZES", "PARAMS", "FIRSTGROUP",
                                "LASTGROUP"):
                        key = p[i]
                        d[key] = []
                    elif key:
                        d[key].append(p[i])
                    i += 1
                man["specs"][arm] = {
                    "spec": d.get("SPEC", [""])[0],
                    "m": int(d.get("M", ["0"])[0]),
                    "sizes": [int(x) for x in d.get("SIZES", ["0"])[0].split(",")],
                    "params": [int(x) for x in d.get("PARAMS", ["0"])[0].split(",")],
                    "first": d.get("FIRSTGROUP", []),
                    "last": d.get("LASTGROUP", []),
                }
    return man

def selftest():
    ok = True

    def chk(cond, label, got="", want=""):
        nonlocal ok
        print("  %-4s %-60s %s%s" % ("PASS" if cond else "FAIL", label, got,
                                     (" (want %s)" % want) if want else ""))
        if not cond:
            ok = False

    print("cW1_cpg1_score.py --selftest")
    print("THE DOCUMENTED INVOCATION: python3 analysis/cW1_cpg1_score.py <runsdir>")
    print("  (--manifest is OPTIONAL and DEFAULTED; see 164.2 and the header)")
    print("-" * 78)

    print("A. the arms, the specs, the seeds and the run count")
    chk(ARMS == ("k01", "kP", "kE", "kC", "kG"), "arms", " ".join(ARMS))
    chk(N_RUNS == 15, "run count", str(N_RUNS), "15")
    chk(SEEDS == (12, 13, 14), "seeds", str(SEEDS))
    chk(EPOCHS == 100, "the ONLY horizon", str(EPOCHS))
    chk(len(set(SPEC.values())) == 5, "all five specs are distinct")
    chk(SPEC["kP"] == CPR1_GRAN["kP"],
        "kP's spec string is BYTE-IDENTICAL to cpr1's kP", SPEC["kP"])
    chk(SPEC["kC"] == CPR1_GRAN["kC"],
        "kC's spec string is BYTE-IDENTICAL to cpr1's kC (a true replicate)")
    for a in SWEEP_ARMS:
        chk(SPEC[a].startswith("sets:"), "%s uses the sets: grammar" % a)

    print("B. RULE 20 SAFETY of every registered spec string")
    tok = re.compile(r"^[A-Za-z0-9_.,:/-]+$")
    for a in ARMS:
        s = SPEC[a]
        good = (bool(tok.match(s)) and not s.startswith("-") and "=" not in s
                and not any(c.isspace() for c in s)
                and "'" not in s and '"' not in s
                and not any(c in s for c in "[]*?{}~$`&|<>()"))
        chk(good, "%s is ONE shell token, no space/quote/=/glob/leading-dash" % a,
            "len %d" % len(s))
        rt = dict(parse_argsline(
            "--dataset CIFAR100 --stepsize-groups %s --seed 12" % s))
        chk(rt.get("stepsize-groups") == s and len(rt) == 3,
            "%s round-trips through an ARGS line as ONE value" % a)

    print("C. the model, the partitions and the BN-SHIFT BALLAST census")
    chk(len(MODEL) == N_TENSORS, "the frozen model has 62 tensors", str(len(MODEL)))
    chk(sum(MODEL_NUMEL) == TOTAL_PARAMS, "frozen model total params",
        str(sum(MODEL_NUMEL)), str(TOTAL_PARAMS))
    for nm, i in IDX.items():
        chk(MODEL_NAMES[i - 1] == nm, "tensor %d is %s (1-BASED)" % (i, nm))
    alias = all(tensor_class(MODEL_NAMES[i]) == (i + 1) % 3 for i in range(60))
    chk(alias, "class == (1-based index) mod 3 over tensors 1..60 -- the alias "
               "no prefix cut can break")
    for a in SWEEP_ARMS:
        g = spec_to_groups(SPEC[a])
        chk(len(g) == 2 and [len(x) for x in g] == [49, 13],
            "%s composes to m=2 sizes [49,13]" % a,
            "m=%d %s" % (len(g), [len(x) for x in g]))
        chk(tuple(g[0]) == COARSE_IDX[a],
            "%s coarse index set is EXACTLY as registered" % a)
        cp = sum(MODEL_NUMEL[i - 1] for i in g[0])
        chk(cp == COARSE_PARAMS[a], "%s coarse params" % a, str(cp),
            str(COARSE_PARAMS[a]))
        chk(TOTAL_PARAMS - cp == FINE_PARAMS[a], "%s fine params" % a,
            str(TOTAL_PARAMS - cp), str(FINE_PARAMS[a]))
        h, d = holes_of(g[0]), prefix_depth_of(g[0])
        print("       %-3s coarse max ordinal %2d  holes %-14s prefix depth %2d"
              % (a, max(g[0]), str(h), d))
        chk(IDX[T_BN2W] not in g[0],
            "%s does NOT put layer4.0.bn2.weight (tensor 50) in the coarse group"
            % a)
    ce, cc = spec_to_groups(SPEC["kE"])[0], spec_to_groups(SPEC["kC"])[0]
    chk(len(holes_of(ce)) == len(holes_of(cc)) == 3 and max(ce) == max(cc) == 52,
        "kE and kC are MATCHED on hole count (3) and max coarse ordinal (52)")
    chk(set(ce) ^ set(cc) == {IDX[T_BN1B], IDX[T_CONV2]},
        "kE ^ kC is EXACTLY {48 bn1.bias, 49 conv2.weight}")
    chk(COARSE_PARAMS["kG"] == CPR1_KS_COARSE
        and FINE_PARAMS["kG"] == CPR1_KS_FINE,

def primary_of(de):
    """The registered PRIMARY branch map.  Pure function of DE = kE - kP."""
    hits = [k for k in ACCOUNTS if abs(de - DE_PRED[k]) <= READ_BAR]
    if len(hits) == 1:
        return BRANCH_OF_ACCOUNT[hits[0]]
    if len(hits) > 1:
        return "UNRESOLVED-UNDERPOWERED"
    return "UNRESOLVED-PATTERN-UNREGISTERED"


def interaction_of(i):
    """The registered SECONDARY token.  Pure function of the 2x2 interaction."""
    if abs(i) <= INT_BAR:
        return "INTERACTION-ABSENT"
    return ("INTERACTION-PRESENT-POSITIVE" if i > 0
            else "INTERACTION-PRESENT-NEGATIVE")

            print("  !!! manifest says T=%s TOTAL=%s, registered %d / %d"
                  % (man.get("T"), man.get("total"), N_TENSORS, TOTAL_PARAMS))
        else:
            print("  live model: %d tensors, %d parameters -- ok"
                  % (man["T"], man["total"]))
        nmis = [i for i in range(1, N_TENSORS + 1)
                if i in man["tensors"]
                and (man["tensors"][i][0] != MODEL_NAMES[i - 1]
                     or man["tensors"][i][1] != MODEL_NUMEL[i - 1])]
        print("  manifest carries %d TENSOR lines; %d disagree with the frozen "
              "model" % (len(man["tensors"]), len(nmis)))
        for i in nmis[:6]:
            g1 = False
            print("  !!! tensor %d live %s/%d, registered %s/%d"
                  % (i, man["tensors"][i][0], man["tensors"][i][1],
                     MODEL_NAMES[i - 1], MODEL_NUMEL[i - 1]))
        if len(man["tensors"]) < N_TENSORS:
            print("  NOTE: fewer than %d TENSOR lines; the name-order check is"
                  % N_TENSORS)
            print("        partial, but the FIRSTGROUP/LASTGROUP check below is not.")
        gm = {}
        for a in SWEEP_ARMS:
            d = man["specs"].get(a)
            if not d:
                g1 = False
                print("  !!! no ARMSPEC line for %s" % a)
                continue
            if d["spec"] != SPEC[a]:
                g1 = False
                print("  !!! %s manifest spec %r != registered %r"
                      % (a, d["spec"], SPEC[a]))
            if d["m"] != 2 or d["sizes"] != [49, 13]:
                g1 = False
                print("  !!! %s composed to m=%d sizes %s, registered m=2 [49,13]"
                      % (a, d["m"], d["sizes"]))
            if len(d["first"]) + len(d["last"]) != N_TENSORS:
                g1 = False
                print("  !!! %s name lists cover %d tensors, expected %d"
                      % (a, len(d["first"]) + len(d["last"]), N_TENSORS))
            live_idx = set()
            unknown = [n for n in d["first"] if n not in MODEL_NAMES]
            for n in unknown[:3]:
                g1 = False
                print("  !!! %s coarse names %r, not a tensor of the frozen model"
                      % (a, n))
            for n in d["first"]:
                if n in MODEL_NAMES:
                    live_idx.add(MODEL_NAMES.index(n) + 1)
            if live_idx != set(COARSE_IDX[a]):
                g1 = False
                print("  !!! %s coarse index set differs from the registered one:"
                      % a)
                print("      extra %s  missing %s"
                      % (sorted(live_idx - set(COARSE_IDX[a])),
                         sorted(set(COARSE_IDX[a]) - live_idx)))
            if d["params"][0] != COARSE_PARAMS[a] \
                    or d["params"][1] != FINE_PARAMS[a]:
                g1 = False
                print("  !!! %s params %s, registered %d / %d"
                      % (a, d["params"], COARSE_PARAMS[a], FINE_PARAMS[a]))
            gm[a] = live_idx
            hh = holes_of(sorted(live_idx)) if live_idx else []
            print("  %-3s %s" % (a, d["spec"]))
            print("      sizes %-9s params %9d / %-9d   max ordinal %2d  holes %s"
                  % (str(d["sizes"]), d["params"][0], d["params"][1],
                     max(live_idx) if live_idx else -1, hh))
            print("      COARSE: %s" % " ".join(d["first"]))
            print("      FINE  : %s" % " ".join(d["last"]))
            if IDX[T_BN2W] in live_idx:
                g1 = False
                print("  !!! %s puts %s (tensor 50) in the COARSE group; no cpg1"
                      % (a, T_BN2W))
                print("      arm may, and every floor-saturated m=2 arm in the")
                print("      corpus has it there")
        if all(a in gm for a in SWEEP_ARMS):
            for a, b, want in (("kP", "kE", (T_BN1B, T_SC0W)),
                               ("kP", "kC", (T_CONV2, T_SC0W)),
                               ("kP", "kG", (T_CONV2, T_BN2B)),
                               ("kE", "kC", (T_BN1B, T_CONV2)),
                               ("kC", "kG", (T_BN2B, T_SC0W))):
                d = gm[a] ^ gm[b]
                w = set(IDX[x] for x in want)
                if d != w:
                    g1 = False
                    print("  !!! %s ^ %s = %s, registered %s"
                          % (a, b, sorted(d), sorted(w)))
                else:
                    print("  %s ^ %s is EXACTLY {%s} -- ok"
                          % (a, b, ", ".join(want)))
            szs = set(len(gm[a]) for a in SWEEP_ARMS)
            if szs != {49}:
                g1 = False
                print("  !!! coarse group sizes differ across arms: %s"
                      % sorted(szs))
            else:
                print("  all four coarse groups have the SAME SIZE (49) -- ok")
            hE, hC = holes_of(sorted(gm["kE"])), holes_of(sorted(gm["kC"]))
            if len(hE) == len(hC) == 3 and max(gm["kE"]) == max(gm["kC"]) == 52:
                print("  kE and kC MATCHED on hole count (3) and max ordinal (52)"
                      " -- ok")
            else:
                g1 = False
                print("  !!! kE/kC contiguity match broken: holes %s vs %s, max "
                      "%d vs %d" % (hE, hC, max(gm["kE"]), max(gm["kC"])))
            if COARSE_PARAMS["kG"] == CPR1_KS_COARSE:
                print("  kG's coarse mass %d is EXACTLY cpr1 kS's -- the mass "
                      "control -- ok" % COARSE_PARAMS["kG"])
    print("  G1 %s" % ("PASS" if g1 else "FAIL"))
    if not g1:
        verdicts.append("UNRESOLVED-NOT-COMPARABLE")

    # ---------------- the levels ----------------
    print("\nTHE ARMS.  plateau5 at %d epochs.  TEST and TRAIN." % EPOCHS)
    M, SD, MT = {}, {}, {}
    for a in ARMS:
        v = sorted([r["p5"] for r in cells.get(a, []) if r["p5"] is not None])
        w = sorted([r["t5"] for r in cells.get(a, []) if r["t5"] is not None])
        if v:
            M[a] = statistics.mean(v)
            SD[a] = statistics.stdev(v) if len(v) > 1 else 0.0
        if w:
            MT[a] = statistics.mean(w)
        print("  %-4s n=%d  TEST %s  sd %s  TRAIN %s   %s"
              % (a, len(v), fmt(M.get(a)), fmt(SD.get(a), 7, 4), fmt(MT.get(a)),
                 SPEC[a][:44]))
        if v:
            print("       per-seed TEST %s" % "  ".join("%.4f" % x for x in v))

    # ---------------- R2 ----------------
    print("\nR2 DIVERGENCE (bar %.2f pp)" % DEAD_BAR)
    dead = [r for r in good
            if r["p5"] is None or not math.isfinite(r["p5"])
            or r["p5"] <= DEAD_BAR or r["n_ep"] != EPOCHS]
    for r in dead:
        print("  !!! DEAD %s p5=%s n_ep=%d"
              % (os.path.basename(r["file"]), r["p5"], r["n_ep"]))
    print("  %d of %d alive -- %s" % (len(good) - len(dead), len(good),
                                      "PASS" if not dead else "FAIL"))
    if dead:
        verdicts.append("UNRESOLVED-DIVERGED")

    # ---------------- R3 ----------------
    print("\nR3 NOISE (bar %.4f pp = 3 * SIGMA_W)" % NOISY_BAR)
    noisy = [a for a in ARMS if SD.get(a, 0.0) > NOISY_BAR]
    for a in ARMS:
        print("  %-4s sd %8.4f  %s" % (a, SD.get(a, float("nan")),
                                       "NOISY" if a in noisy else "ok"))
    if noisy:
        verdicts.append("UNRESOLVED-NOISY")

    # ---------------- R-CTRL ----------------
    print("\nR-CTRL -- DOES THIS SEED TRIPLE REPRODUCE THE CELL?  IN BATCH.")
    if "kP" in M and ANCHOR in M:
        d = M["kP"] - M[ANCHOR]
        print("  kP %.4f - k01 %.4f = %+.4f pp  (bar %.4f = half of cpk3's own"
              % (M["kP"], M[ANCHOR], d, CTRL_BAR))
        print("  k49 - k01 at %d epochs).  %s"
              % (EPOCHS, "PASS" if d >= CTRL_BAR else "FAIL"))
        if d < CTRL_BAR:
            verdicts.append("UNRESOLVED-CONTROL")
        print("  DESCRIPTIVE, CROSS-BATCH, GATES NOTHING: cpr1's kP at the "
              "byte-identical")
        print("  spec read %.4f on seeds {9,10,11}; cpg1's kP reads %.4f, "
              "%+.4f pp = %s."
              % (CPR1_100["kP"], M["kP"], M["kP"] - CPR1_100["kP"],
                 se_str(M["kP"] - CPR1_100["kP"])))
    else:
        verdicts.append("UNRESOLVED-CONTROL")

    # ---------------- R-FLOOR ----------------
    print("\nR-FLOOR -- SATURATION AGAINST THIS BATCH'S OWN m=1 ANCHOR")
    print("  k01 = %s pp.  bar %.4f pp (= 2 * SE_FLOOR)."
          % (fmt(M.get(ANCHOR), 8), FLOOR_BAR))
    informative = {}
    for a in SWEEP_ARMS:
        if a in M and ANCHOR in M:
            above = M[a] - M[ANCHOR]
            informative[a] = above > FLOOR_BAR
            print("  %-4s %+8.4f above the floor = %7s -> %s"
                  % (a, above, se_str(above).strip(),
                     "INFORMATIVE" if informative[a]
                     else "NOT-INFORMATIVE (floor-saturated)"))
    for a in ("kP", "kE"):
        if not informative.get(a, False):
            verdicts.append("UNRESOLVED-SATURATED")
            print("  !!! %s is at the floor.  NO live account predicted that "
                  "(worst" % a)
            print("      live prediction was +23.2554 pp = 30.77 SE above it), so "
                  "the")
            print("      primary read is destroyed and the batch says so rather "
                  "than")
            print("      picking a branch.")
            break

    # ---------------- R-CEIL, DESCRIPTIVE ----------------
    print("\nR-CEIL -- DESCRIPTIVE ONLY, CROSS-BATCH, GATES NOTHING")
    print("  reference: the corpus's %d same-cell `layerwise` (m=62) rows over %d"
          % (L_CEIL_N, L_CEIL_BATCHES))
    print("  batches at %d epochs, mean %.4f.  There is NO in-batch ceiling "
          "anchor." % (EPOCHS, L_CEIL))
    for a in ARMS:
        if a in M:
            print("  %-4s %+8.4f pp below the reference" % (a, L_CEIL - M[a]))

    # ---------------- THE CONTRASTS ----------------
    print("\nTHE CONTRASTS.  DE = kE - kP (PRIMARY), DC = kC - kP, DG = kG - kP.")
    print("  read bar %.6f pp = 2 SE_ARM_DIFF (SE %.6f)" % (READ_BAR, SE_ARM_DIFF))
    de = (M["kE"] - M["kP"]) if ("kE" in M and "kP" in M) else None
    dc = (M["kC"] - M["kP"]) if ("kC" in M and "kP" in M) else None
    dg = (M["kG"] - M["kP"]) if ("kG" in M and "kP" in M) else None
    det = (MT["kE"] - MT["kP"]) if ("kE" in MT and "kP" in MT) else None
    dct = (MT["kC"] - MT["kP"]) if ("kC" in MT and "kP" in MT) else None
    dgt = (MT["kG"] - MT["kP"]) if ("kG" in MT and "kP" in MT) else None
    print("  contrast  what crosses vs kP                          "
          "TEST        in SE   TRAIN")
    for nm, v, vt, crossing in (
            ("DE", de, det, "OUT %s / IN %s" % (T_BN1B, T_SC0W)),
            ("DC", dc, dct, "OUT %s / IN %s" % (T_CONV2, T_SC0W)),
            ("DG", dg, dgt, "OUT %s / IN %s" % (T_CONV2, T_BN2B))):
        print("  %-9s %-45s %s %s %s"
              % (nm, crossing, fmt(v), se_str(v) if v is not None else "",
                 fmt(vt)))
    print("\n  PREDICTED, REGISTERED BEFORE ANY RUN EXISTED:")
    print("      account          DE         DC         DG         2x2 I")
    for k in ACCOUNTS:
        print("      %-14s %+9.4f  %+9.4f  %+9.4f  %+9.4f"
              % (ACCOUNT_NAME[k], DE_PRED[k], DC_PRED[k], DG_PRED[k],
                 INT_PRED[k]))
    print("      %-14s %9s  %9s  %+9.4f  %9s (kG only; DEAD before this batch)"
          % ("H-MASS", "-", "-", DG_D, "-"))
    if de is not None:
        print("\n  RESIDUALS ON THE PRIMARY CONTRAST DE:")
        for k in ACCOUNTS:
            r = de - DE_PRED[k]
            print("      vs %-14s %+9.4f = %s   %s"
                  % (ACCOUNT_NAME[k], r, se_str(r),
                     "WITHIN BAR" if abs(r) <= READ_BAR else "excluded"))
    inter = None
    if None not in (de, dc, dg):
        inter = de - (dc - dg)
        print("\n  THE 2x2 INTERACTION, FULLY IN-BATCH.")
        print("      I = (kE - kP) - (kC - kG) = %+.6f pp = %s"
              % (inter, se_str(inter, SE_INT)))
        print("      bar %.6f pp = 2 * SE_INT (%.6f)" % (INT_BAR, SE_INT))
        print("      I is, up to a ballast term bounded by |S48| + |S51| = "
              "%.6f pp," % (abs(S48) + abs(S51)))
        print("      EXACTLY the contextuality of %s." % T_SC0W)
        for k in ACCOUNTS:
            print("      vs %-14s %+9.4f = %s"
                  % (ACCOUNT_NAME[k], inter - INT_PRED[k],
                     se_str(inter - INT_PRED[k], SE_INT)))
    for a, nm in (("kC", "DC"), ("kG", "DG")):
        if a in informative and not informative[a] and ANCHOR in M and "kP" in M:
            bound = -(M["kP"] - M[ANCHOR] - FLOOR_BAR)
            print("\n  !!! %s IS FLOOR-SATURATED.  |%s| IS TRUNCATED and may be"
                  % (a, nm))
            print("      quoted ONLY as the BOUND %s <= %+.4f pp." % (nm, bound))

    # ---------------- THE VERDICT ----------------
    print("\n" + "=" * 78)
    if verdicts:
        final = verdicts[0]
        print("GATES FAILED, in order: %s" % " | ".join(verdicts))
    elif de is None:
        final = "UNRESOLVED-PATTERN-UNREGISTERED"
    else:
        primary = primary_of(de)
        toks = [primary]
        if inter is not None:
            itok = interaction_of(inter)
            toks.append(itok)
            want = {"CONTIGUITY-REFUTED-COGROUPING-OPERATIVE":
                    "INTERACTION-PRESENT-POSITIVE",
                    "CONTIGUITY-OPERATIVE": "INTERACTION-ABSENT",
                    "INERT-BEYOND-FIRST-HOLE":
                    "INTERACTION-PRESENT-NEGATIVE"}.get(primary)
            if want and itok != want:
                toks.append("INCONSISTENT-PRIMARY-vs-INTERACTION")
        else:
            toks.append("INTERACTION-UNAVAILABLE")
        if "kG" in informative:
            toks.append("COARSE-MASS-REFUTED-AT-EXACT-MATCH"
                        if informative["kG"] else "COARSE-MASS-NOT-REFUTED")
        if dc is not None:
            rr = dc - (CPR1_100["kC"] - CPR1_100["kP"])
            toks.append("KC-REPLICATES" if abs(rr) <= REPL_BAR
                        else "KC-DOES-NOT-REPLICATE")
            print("REPLICATION (cross-batch, DESCRIPTIVE, GATES NOTHING): cpg1's DC")
            print("  %+.6f vs cpr1's %+.6f, residual %+.6f = %s against a bar of"
                  % (dc, CPR1_100["kC"] - CPR1_100["kP"], rr,
                     se_str(rr, SE_INT)))
            print("  %.6f.  Different batch AND different seeds." % REPL_BAR)
        final = " | ".join(toks)
    print("PREDICTED BRANCHES (registered, all three live): %s"
          % " / ".join(BRANCH_OF_ACCOUNT[k] for k in ACCOUNTS))
    print("MEASURED: %s" % final)
    print("FINAL: %s" % final)
    print("=" * 78)

    print("\nSCOPE.  %s / %s / %s+%s / ms %s / alpha0 %s / gamma %s / batch %d /"
          % (DSET, NET, BASE_ALG, META_ALG, MS, ALPHA0, GAMMA, BATCH_SIZE))
    print("  %d epochs / BETA_CLIP=%s / AUGMENT=1 / HIER unset / PROBE=0 /"
          % (EPOCHS, CLIP_C))
    print("  seeds %s / this one named_parameters() order."
          % ",".join(str(s) for s in SEEDS))
    print("  NO claim at any other horizon and NO claim about attenuation: every")
    print("  anchor the predictions are built from is a 100-epoch readout.")
    print("  NO m claim: every sweep arm is m = 2 with sizes [49,13] and the")
    print("  scalar anchor is an anchor.")
    print("  NO CAPTURE and NO ceiling claim: there is no layerwise arm in this")
    print("  batch, R-CEIL is cross-batch and DESCRIPTIVE, and no in-batch")
    print("  ceiling anchor may be spliced in later.")
    print("  NO separation of tensor CLASS from a MASS THRESHOLD on the crossing")
    print("  tensor: structurally impossible on ResNet-18 (161.7c i).  This batch")
    print("  does not attempt it; it separates CONTEXT VARIABLES, not classes.")
    print("  NO test of 152.12 rival (c): PROBE=0, and at m=2 block_product has")
    print("  already reduced <h,g> to one scalar per GROUP.  It stays UNTESTED.")
    print("  NO generalisation outside layer4.0 and its boundary: the four")
    print("  tensors that move are 48, 49, 51 and 52 of 62.")
    print("  NO empirical claim about PATCH_NAMESETS' inertness from this batch:")
    print("  there is no kL arm and 164.10 forbids reading cpr1's R-EQUIV as")
    print("  proof.  Inertness rests on tests/test_namesets.py and on G1.")
    print("  The `scalar` anchor still takes a different code path in init_meta;")
    print("  it is a FLOOR REFERENCE ONLY and that confound is NOT lifted.")
    print("  Nothing here touches CIFAR-10, Tiny-ImageNet, ImageNet-489,")
    print("  ResNet-50, resnet18_blocks, any other meta-stepsize or optimiser")
    print("  pair, or any hierarchical or shrinkage operator.")
    return 0

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir", nargs="?", default=None)
    ap.add_argument("--manifest", default=None,
                    help="OPTIONAL.  Defaults to <runsdir>/cpg1/%s, then "
                         "<runsdir>/cpg1-%s, then <runsdir>/%s."
                         % (MANIFEST_NAME, MANIFEST_NAME, MANIFEST_NAME))
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    if a.selftest:
        return selftest()
    if not a.runsdir:
        ap.error("runsdir is required unless --selftest is given")
    return score(a.runsdir, a.manifest)


if __name__ == "__main__":
    sys.exit(main())
