#!/bin/bash
# =============================================================================
# cWD3_carrierwd.sh -- `cwd3`: ResNet18_c100 -- IS THE SCALAR COLLAPSE's WEIGHT-DECAY PRECONDITION CARRIED BY THE THREE
#   CARRIERS' OWN DECAY?  ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS {140,141,142}, ResNet18_c100 /
#   CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM, EVERY ARM SCALAR, --constraint=L4 ON EVERY JOB.  CORRECTIONS 275.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Every guard and scancel line is filtered on `cwd3-`.
#
# REGISTERED SCORER: analysis/cWD3_carrierwd_score.py (+ analysis/cwd3_design.py, and cwd1's analysis/cwd_design.py and
# analysis/cwd_common.py REUSED UNEDITED), committed and pushed BEFORE this batch is submitted (RULE 21).
#   DOCUMENTED INVOCATION:  python3 analysis/cWD3_carrierwd_score.py <runsdir>
#
# NO NEW HARNESS CODE.  The batch runs from cwd1's EXISTING isolated tree $WS/harness_cwd1/cifar10 (cvt8's tree +
# PATCH_DECAYMASK, HF.py 94aedc33..., UNCHANGED, re-verified by bin/cWD_stage_harness.sh cwd1) with cwd1's EXISTING runner
# $WS/jobs/run_cifar_cwd1.sh (34a8c90e..., archived as jobs/run_cifar_cwd1.sh, UNCHANGED).  PATCH_DECAYMASK's registered
# grammar `DECAY_MASK=normscale | <name>(+<name>)*` already admits a name list; guard 4f requires BOTH cwd1's inertness
# log for this tree (cwdinert1 5026046) AND cwd3's own bite log (bin/cWD3_realrun_bite.sbatch, the name-list strings on
# the real GPU path).
#
# THE ARMS (no hold patch on any arm; every arm scalar)
#   k01      DECAY_MASK unset
#   CARWD0   DECAY_MASK=layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight   (50,53,59: the ctd1 carriers)
#   CTLWD0   DECAY_MASK=layer4.0.bn1.weight+layer4.0.bn1.bias+layer4.1.bn1.weight            (47,48,56: cdep1 DEPTH)
#   CTL2WD0  DECAY_MASK=layer4.0.bn1.weight+layer4.1.bn1.weight                              (47,56: cdep1 DEPTH2)
#   NWD      DECAY_MASK=normscale                                                            (the 20 BN scales: cwd1's k01NWD)
#
# RULE 20.  Every ARGS value is ONE shell token.  DECAY_MASK rides sbatch's --export list and CANNOT ride the runner's ENV
# line: the patch prints a witness line at construction on EVERY run, which the ENV audit and the scorer's G-WITNESS read.
# analysis/argsline_guard.py IS NOT EDITED.  WEIGHT DECAY ON A NAMED SET IS NOT A CSV COLUMN: the 12 masked runs go to
# results/CORPUS-EXCLUSIONS.tsv at landing as DECAY_MASK rows (the kind exists since 269; corpus_exclusions.py NOT edited).
#
# HARDWARE (271.4(5), 273 bound (1)): every job carries --constraint=L4 and the partitions are gpu-short,gpu-l4-24g, so
# the whole batch runs on ONE accelerator type.  The scorer only DISCLOSES the device (G-HW); it never gates on it.
#
# COST.  cwd1 by sacct: 7.5333 GPU-h / 9 runs = 0.837 per ResNet18_c100 run.  15 jobs ~= 12.6 GPU-h expected;
#   HARD BOUND = WALL x 15 = 45 GPU-h.
#
# USAGE   bash bin/cWD3_carrierwd.sh [--dry-run]   |   bash bin/cWD3_carrierwd.sh --submit   (the ONLY submitting form)
# ENV     CWD3_REGISTERED_COMMIT=<40-hex sha>   on a host with no git repo, --submit requires it (271.6 F1: full sha only)
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
unset VOTE_W BETA_HOLD COMP_HOLD WINDOW_HOLD GROUP_HOLD REST_HOLD DECAY_MASK
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
B=cwd3
HARN=$WS/harness_cwd1/cifar10
HPAR=$WS/harness_cvt8/cifar10
RUNNER=$WS/jobs/run_cifar_cwd1.sh
SAVE=$WS/runs/cwd3

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true
export PYTHONDONTWRITEBYTECODE=1

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=ResNet18_c100
SEEDS="140 141 142"
MST=1e-3
A0=1e-6
NTENS=62
TOTPAR=11220132
G_ALL='scalar'
M_CAR='layer4.0.bn2.weight+layer4.0.shortcut.1.weight+layer4.1.bn2.weight'
M_CTL='layer4.0.bn1.weight+layer4.0.bn1.bias+layer4.1.bn1.weight'
M_CTL2='layer4.0.bn1.weight+layer4.1.bn1.weight'
M_NS='normscale'
ARMS="k01 CARWD0 CTLWD0 CTL2WD0 NWD"
NJOBS=15
WALL=03:00:00
GPU_CONSTRAINT=L4
BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_PRE_SHA=5197dc2eecbbcb6feb815f3798b3455171e6ad614deac913be5df48dd9ede1a9
HF_POST_SHA=94aedc33046afb12782ffb7f2eb729bafc7fcba5f42227fec373fa67b6f58ebf
RUNNER_SHA=34a8c90ee3a0bee00aa2d0b4a3ad5d16d461fbf5a1f0cda5797e44e44c158371
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL" "gpu-l4-24g")

SC="$REPO/analysis/cWD3_carrierwd_score.py"
DES="$REPO/analysis/cwd3_design.py"
WDES="$REPO/analysis/cwd_design.py"
COM="$REPO/analysis/cwd_common.py"
SELF="$REPO/bin/cWD3_carrierwd.sh"
SC9="$REPO/analysis/cVT9_dosewindow_score.py"
EXCL="$REPO/analysis/corpus_exclusions.py"
EXCLT="$REPO/results/CORPUS-EXCLUSIONS.tsv"
RRSB="$REPO/bin/cWD3_realrun_bite.sbatch"
TESTD="$REPO/tests/test_decaymask.py"
TESTDR="$REPO/tests/test_decaymask_realrun.py"
TESTD3="$REPO/tests/test_decaymask_realrun_cwd3.py"
TESTR="$REPO/tests/test_grouphold_realrun.py"
STAGEW="$REPO/bin/cWD_stage_harness.sh"
STAGE8="$REPO/bin/cVT8_stage_harness.sh"
PATCHD="$REPO/patches/patch_decaymask.py"
ARCHR="$REPO/jobs/run_cifar_cwd1.sh"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFDM="$HARN/Optimizers/HF.py.pre_decaymask"
BNH="$HARN/build_network.py"
ILOG1="$WS/runs/cwd1/inertness_decaymask.log"
ILOG="$SAVE/bite_decaymask_cwd3.log"

echo "=============================================================="
echo " cwd3 -- ResNet18_c100: is the collapse's WD precondition carried by the three carriers' own decay?  (PATCH_DECAYMASK)"
echo " $DSET / $NET   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C   ms=$MST alpha0=$A0  $EPOCHS epochs; seeds $SEEDS"
echo "   k01      $G_ALL"
echo "   CARWD0   $G_ALL   DECAY_MASK=$M_CAR"
echo "   CTLWD0   $G_ALL   DECAY_MASK=$M_CTL"
echo "   CTL2WD0  $G_ALL   DECAY_MASK=$M_CTL2"
echo "   NWD      $G_ALL   DECAY_MASK=$M_NS"
echo " 5 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cwd1's, UNCHANGED)   runner $RUNNER (cwd1's, UNCHANGED)   partitions $PARTS   constraint $GPU_CONSTRAINT   walltime $WALL"
echo " DOCUMENTED INVOCATION: python3 analysis/cWD3_carrierwd_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && [ -z "${BETA_HOLD+x}" ] && [ -z "${GROUP_HOLD+x}" ] && [ -z "${REST_HOLD+x}" ] && [ -z "${DECAY_MASK+x}" ] \
  && echo "guard 0b: VOTE_W, BETA_HOLD, GROUP_HOLD, REST_HOLD and DECAY_MASK are UNSET in the launcher's environment" \
  || guard_fail "guard 0b: a hold / mask variable is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" || guard_fail "guard 0: no CSV at $CSVP"

for f in "$SC" "$DES" "$WDES" "$COM" "$SELF" "$SC9" "$EXCL" "$EXCLT" "$RRSB" "$TESTD" "$TESTDR" "$TESTD3" "$TESTR" "$STAGEW" "$STAGE8" \
         "$PATCHD" "$ARCHR" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$HFDM" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

NEWF="analysis/cWD3_carrierwd_score.py analysis/cwd3_design.py bin/cWD3_carrierwd.sh bin/cWD3_realrun_bite.sbatch tests/test_decaymask_realrun_cwd3.py"
REUSEF="analysis/cwd_design.py analysis/cwd_common.py analysis/cVT9_dosewindow_score.py analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv tests/test_decaymask.py tests/test_decaymask_realrun.py tests/test_grouphold_realrun.py patches/patch_decaymask.py bin/cWD_stage_harness.sh bin/cVT8_stage_harness.sh bin/_lib_guards.sh analysis/argsline_guard.py jobs/run_cifar_cwd1.sh"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the registration files and reused files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CWD3_REGISTERED_COMMIT:-}" ]; then
    if printf '%s' "$CWD3_REGISTERED_COMMIT" | grep -qE '^[0-9a-f]{40}$'; then
      echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CWD3_REGISTERED_COMMIT (40 hex, 271.6 F1)"
    else
      guard_fail "guard 1b: CWD3_REGISTERED_COMMIT is not a full 40-hex commit name (271.6 F1): $CWD3_REGISTERED_COMMIT"
    fi
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CWD3_REGISTERED_COMMIT is unset"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CWD3_REGISTERED_COMMIT=<40-hex sha>"
  fi
fi
[ "$(sha "$ARCHR")" = "$RUNNER_SHA" ] && [ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] \
  && echo "guard 1b2: the archived jobs/run_cifar_cwd1.sh == \$WS/jobs/run_cifar_cwd1.sh == $RUNNER_SHA (cwd1's runner, UNCHANGED)" \
  || guard_fail "guard 1b2: the archived runner or the live runner is not $RUNNER_SHA"
[ "$(sha "$WDES")" = 03d2d7614915757798699ff75ca971a11fca2056d383649fecbbf5adf06c0195 ] && [ "$(sha "$COM")" = d7ddb49e71013e3a21eaf5129861ab43dbc16e181bbd92af7c70cdc95e82bf52 ] \
  && [ "$(sha "$PATCHD")" = cf88e7d25793b73c27e7a3bafcabd68e5499fe9007e803d871f25aaef75b2379 ] \
  && echo "guard 1b3: the REUSED cwd_design.py / cwd_common.py / patch_decaymask.py are cwd1's registered bytes (RULE 16)" \
  || guard_fail "guard 1b3: a reused cwd1 registration file moved"

SELFT=/tmp/cwd3_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
[ "$(grep -c '^  SKIP' "$SELFT")" = 0 ] && echo "guard 1c1: no SKIP (real records and the corpus were found on this host)" || guard_fail "guard 1c1: the selftest SKIPPED a section"
grep -qE "^  PASS O2 SUFFICES +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cWD3_carrierwd_score.py <runsdir>. exits 0" "$SELFT" && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows OUTSIDE cwd3 carry seed 140, 141 or 142" "$SELFT" && grep -q "^  PASS ZERO corpus rows carry the cwd3- prefix" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed / fresh-prefix checks ran and passed" \
  || guard_fail "guard 1c4: fresh-seed / prefix check did not pass"
grep -q "^  PASS B THE BROKEN NAME-LIST-MASK NULL ON REAL RECORDS" "$SELFT" && grep -q "^  PASS B THE WRONG-SET NULL ON REAL RECORDS" "$SELFT" \
  && grep -q "^  PASS B the 3 landed cwd1 k01NWD runs PASS G-BITE read as cwd3's NWD" "$SELFT" \
  && grep -q "^  PASS E the BROKEN NAME-LIST-MASK null's LEVELS block is BYTE-IDENTICAL" "$SELFT" \
  && echo "guard 1c5: G-BITE passes 3 landed cwd1 k01 and 3 k01NWD runs, FAILS both broken-mask nulls on landed records; the null's levels == NEEDS-MORE-THAN-THREE's" \
  || guard_fail "guard 1c5: the real-record G-BITE / broken-mask null checks did not pass"
grep -q "^  PASS SIGMA_R18ALL: the FROZEN literal 0.6481128684085689 IS the FILTERED value" "$SELFT" \
  && echo "guard 1c6: OPERATIONS 36 -- the frozen floor IS the corpus_exclusions.filter_rows value on the 3,253-row corpus, not the naive one" \
  || guard_fail "guard 1c6: the frozen floor was not shown to be the filtered value"
echo "guard 1c: scorer sha256 $(sha "$SC")"

grep -q '^cwd3-' "$CSVP" && guard_fail "guard 2: cwd3-* already in the CSV" || echo "guard 2: no cwd3-* row in the corpus"
NOUT=$(ls "$WS/runs"/cwd3-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cwd3-*.out on disk" || guard_fail "guard 2b: $NOUT cwd3-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (140|141|142)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 140/141/142" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 140/141/142"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cwd3-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cwd3-* jobs" || guard_fail "guard 2e: sacct shows $NS cwd3-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cwd3-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cwd3-* in squeue" || guard_fail "guard 2f: $NQ cwd3-* already queued"
fi

# ---- guard 4a: cvt8's tree unchanged; cwd1's tree consistent (cwd3 runs on it) -----------
bash "$STAGE8" >/tmp/cwd3_stage8.$$ 2>&1 \
  && { echo "guard 4a: cVT8_stage_harness.sh (verify mode) VERIFIES cvt8's tree:"; grep -E 'V-h|V-d|VERIFIED' /tmp/cwd3_stage8.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt8's tree is NOT consistent"; tail -20 /tmp/cwd3_stage8.$$; }
bash "$STAGEW" cwd1 >/tmp/cwd3_stagew.$$ 2>&1 \
  && { echo "guard 4a1: cWD_stage_harness.sh cwd1 (verify mode) VERIFIES cwd1's tree, which cwd3 uses UNCHANGED:"; grep -E 'V-a|V-h|V-d|V-f|VERIFIED' /tmp/cwd3_stagew.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a1: cwd1's tree is NOT consistent"; tail -20 /tmp/cwd3_stagew.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" || guard_fail "guard 3: CIFAR-100 not visible"
[ "$(sha "$BNH")" = "$BN_SHA" ] && echo "guard 3b: build_network.py == cvt8's ($BN_SHA)" || guard_fail "guard 3b: build_network.py"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFDM")" = "$HF_PRE_SHA" ] && [ "$(sha "$HPAR/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && echo "guard 3c: HF.py == $HF_POST_SHA (cwd1's); .pre_decaymask == cvt8's HF.py == $HF_PRE_SHA" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or cvt8's HF.py moved)"
[ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && echo "guard 3e: runner == $RUNNER_SHA (cwd1's run_cifar_cwd1.sh)" || guard_fail "guard 3e: runner sha"

# ---- guard 4b/4f: structural proof; cwd1's inertness log (the tree) AND cwd3's bite log (the name-list strings) ----
python3 "$TESTD" --pre "$HFDM" --post "$HFH" --structural-only >/tmp/cwd3_d0.$$ 2>&1 \
   && echo "guard 4b: PATCH_DECAYMASK structural D0 PASSES (deleting the three regions == cvt8's HF.py, byte for byte)" \
   || { guard_fail "guard 4b: D0 FAILED"; tail -8 /tmp/cwd3_d0.$$; }
if [ -f "$ILOG1" ] && grep -q "^TEST_DECAYMASK_RC 0$" "$ILOG1" && grep -q "^REALRUN_RC 0$" "$ILOG1" \
   && [ "$(grep -c '^ALL PASS$' "$ILOG1")" = 2 ] && [ "$(grep -c '^  FAIL' "$ILOG1")" = 0 ] && grep -q "^BATCH cwd1$" "$ILOG1" \
   && grep -q "^TREE_POST $HARN HF $(sha "$HFH") build_network $(sha "$BNH")$" "$ILOG1" \
   && grep -q "^PATCH_SHA256 $(sha "$PATCHD")$" "$ILOG1" && grep -q "^DESIGN_SHA256 $(sha "$WDES")$" "$ILOG1" \
   && grep -q "PASS RR1 k01          DECAY_MASK unset: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG1"; then
  echo "guard 4f1: cwd1's inertness log (cwdinert1) ALL PASS on THIS tree's HF, THIS patch: $(grep -c '^  PASS' "$ILOG1") PASS / $(grep -c '^  FAIL' "$ILOG1") FAIL (sha $(sha "$ILOG1" | cut -c1-8))"
else
  guard_fail "guard 4f1: cwd1's inertness log at $ILOG1 does not vouch for HF $(sha "$HFH")"
fi
if [ -f "$ILOG" ] && grep -q "^TEST_DECAYMASK_D0_RC 0$" "$ILOG" && grep -q "^REALRUN_RC 0$" "$ILOG" \
   && [ "$(grep -c '^ALL PASS$' "$ILOG")" = 2 ] && [ "$(grep -c '^  FAIL' "$ILOG")" = 0 ] && grep -q "^BATCH cwd3$" "$ILOG" \
   && grep -q "^NVIDIA L4, " "$ILOG" && grep -qx "EXPORT_PROBE $M_CAR" "$ILOG" \
   && grep -q "^TREE_POST $HARN HF $(sha "$HFH") build_network $(sha "$BNH")$" "$ILOG" \
   && grep -q "^DESIGN_SHA256 $(sha "$DES")$" "$ILOG" && grep -q "^CWD_DESIGN_SHA256 $(sha "$WDES")$" "$ILOG" \
   && grep -q "^DRIVER_SHA256 $(sha "$TESTD3")$" "$ILOG" && grep -q "^REUSED_DRIVER_SHA256 $(sha "$TESTDR")$" "$ILOG" \
   && grep -q "^PATCH_SHA256 $(sha "$PATCHD")$" "$ILOG" && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTR")$" "$ILOG" \
   && grep -q "PASS RR1 k01 DECAY_MASK unset: every parameter and buffer after the last step BITWISE-IDENTICAL" "$ILOG" \
   && [ "$(grep -c "PASS RR2 .* printed exactly the design's DECAY_MASK witness" "$ILOG")" = 4 ] \
   && [ "$(grep -c "PASS RR2 .* at every probe record dm_n == step+2" "$ILOG")" = 4 ] \
   && [ "$(grep -c "PASS RR3 .* w', h', m' BITWISE the formula" "$ILOG")" = 4 ] \
   && [ "$(grep -c "PASS RR4 .* matches ITS OWN set" "$ILOG")" = 4 ] \
   && grep -q "PASS RR5 a misspelt name" "$ILOG"; then
  echo "guard 4f: cwd3's bite log ALL PASS on THIS tree, THIS design, THIS driver: $(grep -c '^  PASS' "$ILOG") PASS / $(grep -c '^  FAIL' "$ILOG") FAIL (sha $(sha "$ILOG" | cut -c1-8))"
else
  guard_fail "guard 4f: no ALL PASS cwd3 bite log for HF $(sha "$HFH") / design $(sha "$DES") at $ILOG -- sbatch $RRSB, then copy its .out there"
fi

# ---- guard 4c': the launcher's literals == the design module's ------
python3 - "$DES" "$G_ALL" "$M_CAR" "$M_CTL" "$M_CTL2" "$M_NS" "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$ARMS" \
          "$BN_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$RUNNER_SHA" "$STAGEW" "$GPU_CONSTRAINT" <<'PYEOF' || guard_fail "guard 4c': launcher / design / stage-script literals disagree"
import sys, re, importlib.util
spm = importlib.util.spec_from_file_location("cwd3_design", sys.argv[1]); Dm = importlib.util.module_from_spec(spm); spm.loader.exec_module(Dm)
Bm = Dm.CWD3
(g, car, ctl, ctl2, ns, net, ntens, totpar, seeds, arms, bn, pre, post, runner, stage, gpu) = sys.argv[2:18]
bad = []
if tuple(arms.split()) != Bm.ARMS: bad.append("ARMS")
if dict((a, g) for a in Bm.ARMS) != Bm.SPEC: bad.append("SPEC")
if {"k01": "", "CARWD0": car, "CTLWD0": ctl, "CTL2WD0": ctl2, "NWD": ns} != Bm.DMASK: bad.append("DMASK")
