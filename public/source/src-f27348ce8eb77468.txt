"""RULE 13 -- rl3's guards, tested in BOTH DIRECTIONS.  The real configuration
must PASS and an injected violation must FAIL.  A green guard is evidence of
nothing until it has been shown to go red.

THE TWO GUARDS THIS BATCH ACTUALLY RESTS ON, and therefore the two whose red
direction matters most:

  GUARD 4  THE BETA-ALLOCATION GUARD.  Everything rl3 claims about RULE 10 --
           "m is held fixed across the three stepsizes" -- is exactly what
           guard 4 measures, and a wrong K silently produces a different
           partition that still runs to completion and still writes a number.
           It needs torch and the LIVE build_network, so its red direction can
           only be exercised ON THE CLUSTER; off-cluster it is SKIPPED loudly
           rather than reported as passing.  That distinction is the whole
           lesson of the previous build's five aborts.

  GUARD 8  THE BOX GUARD.  At ms=1e-3 and 3e-3 nothing is provably box-free, so
           the box is the single largest threat to every number rl3 produces.
           It is pure CSV arithmetic and runs anywhere.

Nothing here invokes --submit or sbatch: the guard bodies are extracted from the
script and driven directly as pure python.
"""
import os
import subprocess

import tempfile

HERE = os.path.dirname(os.path.abspath(__file__))
REPO = os.path.dirname(HERE)
SCRIPT = os.path.join(REPO, "bin", "c87_rule11_ladder.sh")
CSV = os.path.join(REPO, "results", "all_runs.csv")
src = open(SCRIPT).read()
PASS = FAIL = SKIP = 0


def report(name, ok, detail=""):
    global PASS, FAIL
    if ok:
        PASS += 1
        print("  PASS  %s" % name)
    else:
        FAIL += 1
        print("  FAIL  %s   %s" % (name, detail))


def skip(name, why):
    global SKIP
    SKIP += 1
    print("  SKIP  %s   (%s)" % (name, why))


def block(marker, delim="PYEOF"):
    """Extract the heredoc body that follows `marker`."""
    i = src.index(marker)
    j = src.index("<<'%s'" % delim, i)
    k = src.index("\n", j) + 1
    e = src.index("\n%s\n" % delim, k)
    return src[k:e]


def run_py(body, args, expect_fail, env=None):
    with tempfile.NamedTemporaryFile("w", suffix=".py", delete=False) as fh:
        fh.write(body)
        path = fh.name
    e = dict(os.environ)
    if env:
        e.update(env)
    try:
        r = subprocess.run([sys.executable, path] + args, capture_output=True,
                           text=True, env=e)
        out = (r.stdout + r.stderr).strip()
        last = out.splitlines()[-1] if out else ""
        ok = (r.returncode != 0) if expect_fail else (r.returncode == 0)
        return ok, last
    finally:
        os.unlink(path)

    report("RED: m(nodewise) off by ONE -> REFUSES", ok and "GUARD FAIL" in msg, msg)
    ok, msg = run_py(g4, g4args(m_chg="4852"), expect_fail=True)
    report("RED: the G leg's EXACT count match broken by one -> REFUSES",
           ok and "GUARD FAIL" in msg, msg)
    ok, msg = run_py(g4, g4args(kg="256"), expect_fail=True)
    report("RED: a chunk K BELOW the largest 1-D tensor (256 < 512) -> REFUSES, "
