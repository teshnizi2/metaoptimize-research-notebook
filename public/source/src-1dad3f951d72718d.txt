#!/usr/bin/env python3
# =============================================================================
# cLV1_live_term_census.py -- WHICH REPORTED VERDICTS READ THE LIVE CORPUS AT
# SCORE TIME?  A CENSUS OF THE REGISTERED SCORERS.  ZERO GPU.  READ-ONLY.
#
# WHY.  CORRECTIONS 207 found that cVK1's bars are max(frozen floor, in-batch,
# two terms re-derived from results/all_runs.csv AT SCORE TIME), so an ingest
# of OTHER batches' rows can move cVK1's verdict.  This file asks the same
# question of every scorer behind a claim in 199.10 / 204 and of every batch
# registered since cycle 140:
#
#   (1) CLASSIFY how each scorer's bars, thresholds, SE and gates are built:
#         FROZEN       literals fixed at registration, plus this batch's OWN
#                      in-batch data (which no other ingest can move);
#         LIVE-IN-MAX  a max() over a frozen floor and >= 1 term read from the
#                      corpus at score time;
#         LIVE-DIRECT  a bar or gate computed from the corpus with NO frozen
#                      floor.
#       Excluding the batch's OWN prefix does NOT make a term frozen: other
#       batches' ingests still move it.
#   (2) RE-DERIVE every live term on the corpus the batch was scored against
#       and on EVERY later committed corpus up to HEAD.
#   (3) FLIP MARGINS.  How far the live quantity (or sigma) must move from the
#       value actually used before the registered branch changes -- found by
#       running the scorer's OWN, UNEDITED score() in-process with one module
#       constant or one corpus-reader function replaced IN MEMORY, and reading
#       the FINAL line it prints.
#
# WHAT THIS FILE IS NOT.  It is not a scorer, it is not registered, it issues
# no verdict of record and it re-scores no batch into the record.  It writes
# nothing under analysis/, tests/, bin/, results/, docs/ or any runs dir:
# corpus snapshots (git show) and combined run dirs (symlinks) go to --work.
# Registered scorers are IMPORTED from their committed paths; their sha256 is
# printed at start and re-checked at exit.
#
# RULE 21: NO CLAIM.  The only claim is that this file was committed before
# its first execution.
#
# RUN, from the repo root (the default run dirs are the Mac mirrors beside the
# repo, ../runs_cluster_b and ../runs):
#   python3 analysis/cLV1_live_term_census.py --extra <dir> [--work <dir>]
#           [--section static|live|sigma|all] [--jobs N]
#   --extra  holds READ-ONLY copies fetched from cluster_b of
#            cpg1/PARTITION-MANIFEST.txt and ciso2/ (manifest + probes).
#            Without it cW1 and cIS2 cannot reach a FINAL line locally and are
#            reported UNSURE.
# =============================================================================

def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as fh:
        for b in iter(lambda: fh.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def git(*a):
    return subprocess.check_output(["git", "-C", REPO] + list(a)).decode().strip()


def snap(work, commit):
    """-> (path, nrows, sha8, short) of results/all_runs.csv at `commit`."""
    h = git("rev-parse", commit)
    p = os.path.join(work, "corpus-%s.csv" % h[:12])
    if not os.path.exists(p):
        data = subprocess.check_output(["git", "-C", REPO, "show", "%s:%s" % (h, CSV_REL)])
        with open(p, "wb") as fh:
            fh.write(data)
    with open(p, newline="") as fh:
        n = sum(1 for _ in csv.reader(fh)) - 1
    return p, n, sha256(p)[:8], h[:7]


def chain(work, start):
    """The corpus at `start`, then every later committed corpus up to HEAD."""
    base = git("rev-parse", start)
    later = git("log", "--reverse", "--format=%H", "%s..HEAD" % base, "--", CSV_REL)
    cs = [base] + [c for c in later.splitlines() if c]
    out, seen = [], set()
    for c in cs:
        s = snap(work, c)
        if s[2] in seen:
            continue
        seen.add(s[2])
        out.append(s)
    return out


_N = itertools.count()


def load(fname):
    path = os.path.join(AN, fname)
    name = "_cLV1_%s_%d" % (fname[:-3], next(_N))
    spec = importlib.util.spec_from_file_location(name, path)
    m = importlib.util.module_from_spec(spec)
    sys.modules[name] = m
    spec.loader.exec_module(m)
    return m


IO_FUNCS = ("load_outs", "load_probes", "read_outs", "read_probe_dirs", "read_runs", "load_runs",
            "collect", "read_out", "probe_summary", "outs_100", "cpk3_100_from_outs")


def memoize_io(m, cache):
    """Run-file readers only (never a corpus reader).  The cached object is
    returned as-is; each scan re-checks that a cached lambda=1 run prints the
    same FINAL as a fresh, uncached, unpatched run."""
    for nm in IO_FUNCS:
        f = m.__dict__.get(nm)
        if not callable(f):
            continue

        def w(*a, __f=f, __nm=nm, **k):
            key = (__nm, repr(a), repr(sorted(k.items())))
            if key not in cache:
                cache[key] = __f(*a, **k)
            return cache[key]
        m.__dict__[nm] = w


def pin_corpus(m, path):
    """Point every DEFAULT corpus path of the module at `path` (module globals
    AND function defaults that captured the path at definition time)."""
    old = set()
    for g in ("CSV", "DEFAULT_CSV", "CSV_DEFAULT"):
        if isinstance(m.__dict__.get(g), str):
            old.add(m.__dict__[g])
            m.__dict__[g] = path
    for v in list(m.__dict__.values()):
        if callable(v) and getattr(v, "__defaults__", None):
            d = tuple(path if (isinstance(x, str) and x in old) else x for x in v.__defaults__)
            if d != v.__defaults__:
                v.__defaults__ = d


def run(fname, args, patches=(), cache=None):
    m = load(fname)
    if cache is not None:
        memoize_io(m, cache)
    infos = [p(m) for p in patches]
    buf = io.StringIO()
    t0 = time.time()
    with contextlib.redirect_stdout(buf), contextlib.redirect_stderr(io.StringIO()):
        try:
            rc = m.score(*args)
        except SystemExit as e:
            rc = "exit %r" % (e.code,)
        except Exception as e:  # reported, never swallowed silently
            rc = "EXC %s: %s" % (type(e).__name__, e)
    sys.modules.pop(m.__name__, None)
    return buf.getvalue(), rc, infos, time.time() - t0


def final_of(key, txt):
    if key == "cdn2":
        b = re.findall(r"BRANCH:\s*(\S+)", txt)
        v = re.findall(r"VERDICT:\s*(\S+)", txt)
        return "%s || %s" % (b[-1] if b else "?", v[-1] if v else "?")
    if key == "cI2":
        v = re.findall(r"^VERDICT:\s*(.+)$", txt, re.M)
        return v[-1].strip() if v else "?"
    f = re.findall(r"FINAL:\s*(.+)$", txt, re.M)
    return f[-1].strip() if f else "?"


def primary_of(key, final):
    parts = [p.strip() for p in final.split("|")]

    if key == "cdn2":
        return final.split("||")[0].strip()
    if key == "cTD2":
        return parts[0]
    return parts[0]

def grade(ratio):
    if ratio is None:
        return "n/a"
    if ratio == float("inf"):
        return "ROBUST (sigma-free)"
    if ratio < 1.2:
        return "FRAGILE"
    if ratio < 2.0:
        return "MODERATE"
    return "ROBUST"


def fnum(x, p=4):
    if x is None:
        return "--"
    if isinstance(x, float) and math.isinf(x):
        return "inf"
    return "%.*f" % (p, x)

BAR_RE = re.compile(r"^(SIGMA\w*|SE_\w+|SE|\w*_BAR|BAND_K|FLOOR_K|FLOOR_BAND|TREE_BAND|V3_OFFSET|"
                    r"SUPPORT_BAR|REFUTE_BAR)$")


def static_one(key):
    fname = SC[key][0]
    path = os.path.join(AN, fname)
    src = open(path).read()
    lines = src.splitlines()
    t = ast.parse(src)
    funcs = {n.name: n for n in t.body if isinstance(n, ast.FunctionDef)}

    def callees(fn):
        return {c.func.id for c in ast.walk(fn) if isinstance(c, ast.Call)
                and isinstance(c.func, ast.Name) and c.func.id in funcs}
    reach, todo = set(), ["score"] if "score" in funcs else []
    while todo:
        f = todo.pop()
        if f in reach:
            continue
        reach.add(f)
        todo.extend(callees(funcs[f]) - reach)
    readers = sorted(f for f in reach
                     if "DictReader" in (ast.get_source_segment(src, funcs[f]) or "")
                     or "csv.reader" in (ast.get_source_segment(src, funcs[f]) or ""))
    bars = sorted({n.id for f in reach for n in ast.walk(funcs[f])
                   if isinstance(n, ast.Name) and BAR_RE.match(n.id)})
    cites = []
    for nd in CLASS[key][2]:
        ln = next((i + 1 for i, l in enumerate(lines) if nd in l), None)
        cites.append((ln, nd))
    return {"readers": readers, "bars": bars, "cites": cites, "nfun": len(reach)}


def section_static(ctx, R):
    print("=" * 100)
    print("SECTION S -- STATIC CLASSIFICATION (AST of the committed file; line numbers measured)")
    print("=" * 100)
    for key in ORDER:
        fname, batch, corr = SC[key][0], SC[key][1], SC[key][2]
        st = static_one(key)
        cls, live, _ = CLASS[key]
        R.setdefault(key, {}).update({"class": cls, "live": live, "readers": st["readers"]})
        print("\n[%s] analysis/%s  sha256 %s  batch %s  scored %s"
              % (key, fname, ctx["SHA0"][fname][:12], batch, corr))
        print("   CLASS: %s" % cls)
        print("   live terms: %s" % live)
        print("   score() call graph: %d functions; corpus readers reachable: %s"
              % (st["nfun"], ", ".join(st["readers"]) or "NONE -- score() never reads the corpus"))
        print("   bar/sigma names referenced in that call graph: %s" % (", ".join(st["bars"]) or "none"))
        for ln, nd in st["cites"]:
            print("     line %-5s %s" % (ln if ln else "NOT-FOUND", nd))

def indep_seed_levels(runsdir, prefix, arm, seeds, horizon):
    """INDEPENDENT parser: my own regex over the raw .out, highest job id wins,
    plateau5 = mean TEST over epochs horizon-5..horizon-1.  No scorer code."""
    vals = []
    for s in seeds:
        fs = glob.glob(os.path.join(runsdir, "%s-%s-s%d-*.out" % (prefix, arm, s)))
        fs = [f for f in fs if re.search(r"-s%d-(\d+)\.out$" % s, f)]
        if not fs:
            return None
        f = max(fs, key=lambda x: int(re.search(r"-(\d+)\.out$", x).group(1)))
        te = {}
        with open(f, errors="replace") as fh:
            for line in fh:
                mm = EP.match(line)
                if mm:
                    te[int(mm.group(1))] = float(mm.group(3))
        w = [te[e] for e in range(horizon - 5, horizon) if e in te]
        if len(w) != 5:
            return None
        vals.append(sum(w) / 5.0)
    return vals


def indep_cell(path, gran, excl, style):
    """INDEPENDENT corpus filter written from each scorer's documented cell
    (ResNet18_c100/CIFAR100, SGDm+Lion, ms 1e-3, a0 1e-6, gamma 1, AUGMENT=1,
    clip -15:-2.3026, HIER unset, batch 100, 100 epochs), own prefix excluded."""
    out = []
    with open(path, newline="") as fh:
        for r in csv.DictReader(fh):
            if r["run"].startswith(excl) or r["granularity"] != gran:
                continue
            common = (r["network"] == "ResNet18_c100" and r["dataset"] == "CIFAR100" and r["base"] == "SGDm"
                      and r["meta"] == "Lion" and r["beta_clip"] == "-15:-2.3026" and r["augment"] == "1"
                      and r["gamma"] == "1" and r["batch_size"] == "100" and not (r["hier"] or "").strip()
                      and r["epochs_done"] == "100")
            if not common:
                continue
            if style == "td":
                ok = (r["superseded"] == "0" and r["epochs_requested"] == "100"
                      and float(r["meta_stepsize"]) == 1e-3 and float(r["alpha0"]) == 1e-6)
            else:
                ok = ((r["superseded"] or "0") != "1" and r["collapsed"] == "0"
                      and r["meta_stepsize"] == "1e-3" and r["alpha0"] == "1e-6")
            if ok:
                try:
                    out.append(float(r["plateau5"]))
                except ValueError:
                    pass
    return out


def msd(v):
    n = len(v)
    mu = sum(v) / n
    return mu, math.sqrt(sum((x - mu) ** 2 for x in v) / (n - 1))


def min_k_band(vals, L, K):
    """WORST-CASE ingest: the fewest rows, all at ONE value v, that push L out of
    mean +/- K*sd.  Identical rows add no variance, so this is the fastest way an
    ingest can narrow a band; any spread in the new rows needs MORE of them."""
    n, S1, S2 = len(vals), sum(vals), sum(x * x for x in vals)
    best = None
    for i in range(0, 1601):
        v = L - 40.0 + 0.05 * i
        for k in range(1, 401):
            N = n + k
            mu = (S1 + k * v) / N
            var = max((S2 + k * v * v - N * mu * mu) / (N - 1), 0.0)
            if abs(L - mu) > K * math.sqrt(var):
                if best is None or k < best[0]:
                    best = (k, v)
                break
            if best is not None and k >= best[0]:
                break
    return best


def min_k_mean(vals, got, band, lo=0.0, hi=100.0):
    """WORST-CASE ingest for a mean-only gate |got - mean| <= band: fewest rows
    at one value v in [lo, hi]."""
    n, S1 = len(vals), sum(vals)
    best = None
    for v in (lo, hi):
        for k in range(1, 5001):
            mu = (S1 + k * v) / (n + k)
            if abs(got - mu) > band:
                if best is None or k < best[0]:
                    best = (k, v)
                break
    return best

def band_levels(key, m, ctx, csvp):
    """-> {arm: [(label, level), ...]} exactly as score() checks them."""
    out = {}
    if key == "cTD2":
        outs = m.read_outs(ctx["RUNS2"])
        batch = m.read_csv_batch(csvp)
        for arm in BAND[key][1]:
            for s in m.SEEDS:
                oo = outs.get((arm, s), [])
                row = batch.get("ctd1-%s-s%d" % (arm, s))
                if row is not None:
                    out.setdefault(arm, []).append(("s%d(csv)" % s, float(row["plateau5"])))
                elif oo and oo[-1]["epochs"] and len(oo[-1]["epochs"]) >= m.EPOCHS:
                    ep = oo[-1]["epochs"]
                    out.setdefault(arm, []).append(("s%d(.out)" % s, statistics.mean(e[2] for e in ep[-5:])))
    elif key in ("cIS1", "cDP1"):
        outs = m.load_outs(ctx["RUNS2"])
        for arm in BAND[key][1]:
            vs = [outs[(arm, s)][-1]["p5"] for s in m.SEEDS if (arm, s) in outs]
            if len(vs) == len(m.SEEDS) and None not in vs:
                out[arm] = [("mean3", statistics.mean(vs))]
    elif key == "cIS2":
        runs = m.load_outs(ctx["RD_CISO2"])
        for arm in BAND[key][1]:
            v = m.arm_mean(runs, arm, m.CTRL_EPOCHS, "test")[0]
            if v is not None:
                out[arm] = [("P100", v)]
    return out


def band_of(key, m, csvp, gran):
    if key == "cTD2":
        rows = m.read_csv_cell(csvp, gran)
        return m.floor_band(rows), [float(r["plateau5"]) for r in rows]
    b = m.band(csvp, gran)
    ep = m.CTRL_EPOCHS if key == "cIS2" else m.EPOCHS
    vals = []
    for r in m._csv_rows(csvp):
        ok = m._in_cell(r, ep) if key == "cIS2" else m._in_cell(r)
        if ok and str(r["granularity"]) == gran:
            try:
                vals.append(float(r["plateau5"]))
            except (TypeError, ValueError):
                pass
    return b, vals


def band_gates(ctx, R):
    print("\n" + "=" * 100)
    print("L2 -- LIVE-DIRECT BAND GATES (cTD2 [H], cIS1 [R], cDP1 [R], cIS2 [C]).  inside iff |L - mean| <= K*sd")
    print("      ratio = K*sd / |L - mean| = the factor the band sd must SHRINK by to exclude L")
    print("=" * 100)
    for key in ("cTD2", "cIS1", "cDP1", "cIS2"):
        if not runnable(key, ctx):
            print("\n[%s] SKIPPED -- run data not reachable locally (see --extra).  UNSURE." % key)
            R[key].update({"margin": "UNSURE (data not local)", "ratio": None})
            continue
        m = load(SC[key][0])
        K = getattr(m, BAND[key][0])
        prefix = SC[key][1] + "-"
        hist = chain(ctx["WORK"], SC[key][3])
        print("\n[%s] batch %s, K = %.1f, corpora %s (scoring) .. HEAD: %d committed corpora"
              % (key, SC[key][1], K, hist[0][3], len(hist)))
        worst = None
        rows_out = []
        for p, n, sh, c in hist:
            lev = band_levels(key, m, ctx, p)
            for arm, gran in BAND[key][1].items():
                b, vals = band_of(key, m, p, gran)
                if b is None:
                    continue
                iv = indep_cell(p, gran, prefix, "td" if key == "cTD2" else "is")
                imu, isd = msd(iv) if len(iv) > 1 else (float("nan"), float("nan"))
                agree = len(iv) == b["n"] and abs(imu - b["mean"]) < 1e-9 and abs(isd - b["sd"]) < 1e-9
                for lab, L in lev.get(arm, []):
                    dist = abs(L - b["mean"])
                    ratio = (K * b["sd"] / dist) if dist > 0 else float("inf")
                    slack = K * b["sd"] - dist
                    rows_out.append((c, n, arm, lab, L, b["n"], b["mean"], b["sd"], slack, slack / b["sd"], ratio, agree))
                    if worst is None or ratio < worst[0]:
                        worst = (ratio, c, arm, lab, L, b, vals)
        print("  %-8s %5s %-4s %-9s %-9s %4s %-9s %-7s %-8s %-7s %-8s %s"
              % ("commit", "rows", "arm", "level", "L", "n", "mean", "sd", "slack pp", "slack/sd", "ratio", "indep"))
        for (c, n, arm, lab, L, bn, mu, sd, sl, sls, ra, ag) in rows_out:
            print("  %-8s %5d %-4s %-9s %-9.4f %4d %-9.4f %-7.4f %-8.4f %-7.3f %-8.3f %s"
                  % (c, n, arm, lab, L, bn, mu, sd, sl, sls, ra, "AGREE" if ag else "DISAGREE"))
        first = [r for r in rows_out if r[0] == hist[0][3]]
        last = [r for r in rows_out if r[0] == hist[-1][3]]
        moved = any(abs(a[6] - b[6]) > 1e-12 or abs(a[7] - b[7]) > 1e-12 or a[5] != b[5]
                    for a, b in zip(first, last))
        ever_out = any(r[8] < 0 for r in rows_out)
        # decisiveness: shrink the tightest arm's sd to just past / just short of the flip
        ratio, c, arm, lab, L, b, vals = worst
        gran = BAND[key][1][arm]
        csv_sc = hist[0][0]

        def shrink(mm, f, gran=gran, K=K):
            if key == "cTD2":
                orig = mm.floor_band

                def fb(rows):
                    x = orig(rows)
                    if x and rows and rows[0].get("granularity") == gran:
                        x = dict(x)
                        x["sd"] *= f
                        x["lo"], x["hi"] = x["mean"] - K * x["sd"], x["mean"] + K * x["sd"]
                    return x
                mm.floor_band = fb
            else:
                orig = mm.band

                def bb(path, g):
                    x = orig(path, g)
                    if x and g == gran:
                        x = dict(x)
                        x["sd"] *= f
                        x["lo"], x["hi"] = x["mean"] - K * x["sd"], x["mean"] + K * x["sd"]
                    return x
                mm.band = bb
        args = score_args(key, ctx, csv_sc)
        f_real = final_of(key, run(SC[key][0], args)[0])
        f_in = final_of(key, run(SC[key][0], args, patches=(lambda mm: shrink(mm, (1.0 / ratio) * 1.001),))[0])
        f_out = final_of(key, run(SC[key][0], args, patches=(lambda mm: shrink(mm, (1.0 / ratio) * 0.999),))[0])
        f_head = final_of(key, run(SC[key][0], score_args(key, ctx, hist[-1][0]))[0])
        mk = min_k_band(vals, L, K)
        print("  tightest: %s %s %s L=%.4f, ratio %.3f (band sd must shrink x%.3f to flip)" % (c, arm, lab, L, ratio, 1 / ratio))
        print("  UNEDITED score(), scoring corpus:         %s" % f_real)
        print("    band sd x%.4f (just INSIDE):            %s" % ((1 / ratio) * 1.001, f_in))
        print("    band sd x%.4f (just OUTSIDE):           %s" % ((1 / ratio) * 0.999, f_out))
        print("  UNEDITED score(), HEAD corpus:            %s" % f_head)
        print("  worst-case ingest to flip (identical rows, the most narrowing): %s"
              % ("%d rows at %.2f pp into the %s cell (n %d)" % (mk[0], mk[1], gran, len(vals)) if mk else "none within 400 rows"))
        rec = REC.get(key)
        recok = rec and f_real.startswith(rec[1])
        R[key].update({"moved": moved, "ever_out": ever_out, "ratio": ratio,
                       "margin": "band sd must shrink x%.3f (ratio %.3f); worst-case ingest %s"
                                 % (1 / ratio, ratio, ("%d identical rows" % mk[0]) if mk else ">400 rows"),
                       "final_scoring": f_real, "final_head": f_head, "rec_match": recok,
                       "decisive": f_out.split("|")[0].strip() != f_real.split("|")[0].strip()})
        print("  -> moved since scoring: %s; any level ever outside its band: %s; gate decisive (flips branch): %s;"
              " scoring FINAL matches CORRECTIONS %s: %s"
              % (moved, ever_out, R[key]["decisive"], rec[0] if rec else "?", recok))
        if key == "cDP1":
            print("  [cDP1] saturation band (stamp DEPTH-/DEPTH2-FLOOR-SATURATED, the BOUND discipline):")
            outs = m.load_outs(ctx["RUNS2"])
            lv = {a: statistics.mean(outs[(a, s)][-1]["p5"] for s in m.SEEDS) for a in ("DEPTH", "DEPTH2")}
            smin = None
            for p, n, sh, cc in hist:
                sb = m.saturation_band(p)
                for a, L2 in lv.items():
                    s2 = min(L2 - sb["lo"], sb["hi"] - L2)
                    smin = s2 if smin is None else min(smin, s2)
                    print("    %-8s %5d %-6s %.4f in [%.4f, %.4f] (cap %.4f, %d+%d rows) slack %.4f pp"
                          % (cc, n, a, L2, sb["lo"], sb["hi"], sb["cap"], sb["n_scalar"], sb["n_extra"], s2))
            R[key]["sat_slack"] = smin


def archive_gates(ctx, R):
    print("\n" + "=" * 100)
    print("L3 -- LIVE-DIRECT ARCHIVE GATES (cK1 G3, cJ1 G3).  pass iff |in-batch anchor - corpus cell mean| <= 2.00 pp")
    print("=" * 100)
    pat = re.compile(r"^\s+(\S+)\s+in-batch ([\d.]+)\s+archived ([\d.]+) \(n=(\d+), (\d+) batches\)\s+([+-][\d.]+) pp", re.M)
    for key in ("cK1", "cJ1"):
        fname = SC[key][0]
        hist = chain(ctx["WORK"], SC[key][3])
        m = load(fname)
        band = m.TREE_BAND
        args = score_args(key, ctx, hist[0][0])

        def pinA(mm, path):
            orig = mm.archived_arms
            mm.archived_arms = lambda path_=None, _o=orig, _p=path: _o(_p)
        txt0 = run(fname, args, patches=(lambda mm: pinA(mm, hist[0][0]),))[0]
        txtH = run(fname, args, patches=(lambda mm: pinA(mm, hist[-1][0]),))[0]
        got = {a: float(g) for a, g, *_ in pat.findall(txt0)}
        f0, fH = final_of(key, txt0), final_of(key, txtH)
        print("\n[%s] batch %s  TREE_BAND %.2f  corpora %s .. HEAD (%d)" % (key, SC[key][1], band, hist[0][3], len(hist)))
        worst = None
        for p, n, sh, c in hist:
            g = m.archived_arms(p)
            for arm, gran in ((m.TAG_LO if key == "cK1" else m.ANCHOR_LO, "scalar"),
                              (m.TAG_HI if key == "cK1" else m.ANCHOR_HI, "layerwise")):
                ref, nn, nb = g[gran][0], g[gran][1], g[gran][2]
                d = got[arm] - ref
                ratio = band / abs(d) if d else float("inf")
                own = sum(1 for r in m._csv_rows(p) if m._cell_p5(r) is not None and r["granularity"] == gran
                          and str(r["run"]).startswith(SC[key][1] + "-"))
                print("  %-8s %5d %-4s in-batch %.4f archived %.4f (n %d, %d batches, OWN rows %d) d %+.4f slack %.4f ratio %.3f"
                      % (c, n, arm, got[arm], ref, nn, nb, own, d, band - abs(d), ratio))
                if worst is None or ratio < worst[0]:
                    vals = [m._cell_p5(r) for r in m._csv_rows(p) if m._cell_p5(r) is not None and r["granularity"] == gran]
                    worst = (ratio, c, arm, d, vals, got[arm])
        mk = min_k_mean(worst[4], worst[5], band)
        print("  UNEDITED score(), scoring corpus: %s" % f0)
        print("  UNEDITED score(), HEAD corpus:    %s" % fH)
        print("  tightest %s %s: ratio %.3f; worst-case ingest to flip: %s"
              % (worst[1], worst[2], worst[0], ("%d rows at %.0f pp" % mk) if mk else "none within 5000 rows"))
        R[key].update({"moved": True, "ratio": worst[0], "final_scoring": f0, "final_head": fH,
                       "margin": "|d| may grow x%.2f (slack %.3f pp); worst-case ingest %s"
                                 % (worst[0], band - abs(worst[3]), ("%d rows at %.0f pp" % mk) if mk else ">5000 rows")})

    print("=" * 100)

              % (outs[0][2] == outs[1][2], body[0] == body[1], rec[0] if rec else "?", ok))
        R[key].update({"final_scoring": outs[0][2], "final_head": outs[1][2], "rec_match": ok,
                       "final_moved": outs[0][2] != outs[1][2]})
        if key in ("cdn2", "cVG1", "cI2"):
            R[key].setdefault("moved", False)

SIGMA_FREE = {"cK1": "in-batch t-contrasts and unitless CAPTURE bars; no corpus sigma in score()",
              "cJ1": "as cK1", "cI2": "unitless ladder bars (DROP/SHAPE/CLIFF); no sigma",

def make_patch(key, lam):
    base, derived, pre, post = SIG[key]

    def patch(m):
        s0 = getattr(m, base)
        s = lam * s0
        lit = {n: getattr(m, n) for n, _ in derived}
        setattr(m, base, s)
        for fn, dfn in pre:
            getattr(m, fn).__defaults__ = dfn(m, s)
        rec = {}
        for n, f in derived:
            v = f(m, s)
            setattr(m, n, v)
            rec[n] = v
        for fn, dfn in post:
            getattr(m, fn).__defaults__ = dfn(m, s)
        if key == "cVI1":   # the two live terms scale with the frozen floor
            orig = m.pooled_sigma

            def ps(rows, extra=lambda r: True, _o=orig):
                t = _o(rows, extra)
                return ((t[0] * lam) if t[0] else t[0],) + tuple(t[1:])
            m.pooled_sigma = ps
        return {"s0": s0, "s": s, "checks": [(n, lit[n], rec[n]) for n, _ in derived]}
    return patch


def sigma_used(key, txt, info):
    if key == "cY1":
        mm = re.findall(r"USED ([\d.]+)", txt)
        return float(mm[-1]) if mm else info["s"]
    if key in ("cVG1", "cVI1"):
        mm = re.findall(r"SIGMA_USED\s+([\d.]+)", txt)
        return float(mm[-1]) if mm else info["s"]
    return info["s"]


def sigma_worker(job):
    key, ctx = job
    t0 = time.time()
    fname = SC[key][0]
    csvp = snap(ctx["WORK"], SC[key][3])[0]
    args = score_args(key, ctx, csvp)
    pin = (lambda mm: pin_corpus(mm, csvp),)
    fresh = final_of(key, run(fname, args, patches=pin)[0])
    cache = {}
    memo = {}

    def at(lam):
        if lam not in memo:
            txt, rc, infos, dt = run(fname, args, patches=pin + (make_patch(key, lam),), cache=cache)
            memo[lam] = (final_of(key, txt), sigma_used(key, txt, infos[-1]), infos[-1], rc)
        return memo[lam]
    f1, su1, info1, rc1 = at(1.0)
    cons = f1 == fresh and all(abs(a - b) <= 6e-6 for _, a, b in info1["checks"])
    grid = sorted(set([2 ** (k / 4.0) for k in range(-20, 21)] + [1.0]))
    for lam in grid:
        at(lam)
    res = {"key": key, "fresh": fresh, "f1": f1, "su1": su1, "consistent": cons, "checks": info1["checks"],
           "rc": rc1}
    for mode in ("primary", "full"):
        kf = (lambda f: primary_of(key, f)) if mode == "primary" else (lambda f: f)
        ref = kf(f1)
        out = {}
        for direction in ("down", "up"):
            seq = [l for l in grid if l < 1.0][::-1] if direction == "down" else [l for l in grid if l > 1.0]
            prev = 1.0
            hit = None
            for l in seq:
                if kf(at(l)[0]) != ref:
                    hit = (prev, l)
                    break
                prev = l
            if hit is None:
                out[direction] = None
                continue
            good, bad = hit
            for _ in range(14):
                mid = math.sqrt(good * bad)
                if kf(at(mid)[0]) == ref:
                    good = mid
                else:
                    bad = mid
            fb, sub, _, _ = at(bad)
            out[direction] = {"lam": bad, "sigma_used": sub, "final": fb,
                              "saturated": abs(sub - su1) < 1e-12}
        res[mode] = out
    res["su_grid"] = sorted((l, at(l)[1]) for l in grid)
    res["secs"] = time.time() - t0
    res["nruns"] = len(memo)
    return res


def section_sigma(ctx, R, jobs):
    print("\n" + "=" * 100)
    print("SECTION G -- SIGMA FLIP MARGINS.  lambda scales the registered sigma inside the UNEDITED score();")
    print("  every constant the file derives from it is re-derived by the file's own formula; grid 2^(k/4),")
    print("  k = -20..20 (x0.03 .. x32), then 14 bisections.  ratio = sigma_at_flip / sigma_used.")
    print("  A floor (in-batch sigma) the scorer's own max() imposes is honoured: below it sigma_used stops moving.")
    print("=" * 100)
    keys = [k for k in SIG if runnable(k, ctx)]
    skipped = [k for k in SIG if k not in keys]
    with ProcessPoolExecutor(max_workers=jobs) as ex:
        results = list(ex.map(sigma_worker, [(k, ctx) for k in keys]))
    for res in results:
        key = res["key"]
        print("\n[%s] %s  (%d runs, %.0f s)" % (key, SC[key][0], res["nruns"], res["secs"]))
        print("   fresh unpatched FINAL : %s" % res["fresh"])
        print("   lambda=1 patched FINAL: %s" % res["f1"])
        print("   patch consistent with the file's literals and FINAL: %s" % res["consistent"])
        for n, a, b in res["checks"]:
            print("       %-12s literal %.6f  re-derived %.6f  |d| %.1e" % (n, a, b, abs(a - b)))
        print("   sigma_used at scoring = %.6f" % res["su1"])
        rat = {}
        for mode in ("primary", "full"):
            for d in ("down", "up"):
                x = res[mode][d]
                if x is None:
                    print("   %-7s %-4s no change anywhere on x0.03..x32" % (mode, d))
                    rat[(mode, d)] = float("inf")
                    continue
                r = x["sigma_used"] / res["su1"]
                sat = " (sigma_used SATURATED at the in-batch floor -- unreachable)" if x["saturated"] else ""
                rat[(mode, d)] = float("inf") if x["saturated"] else (r if d == "up" else 1.0 / r)
                print("   %-7s %-4s flips at lambda %.4f, sigma_used %.6f (x%.3f)%s -> %s"
                      % (mode, d, x["lam"], x["sigma_used"], r, sat, x["final"]))
        pr = min(rat[("primary", "down")], rat[("primary", "up")])
        R[key].update({"sig_used": res["su1"], "sig_ratio_primary": pr,
                       "sig_ratio_full": min(rat[("full", "down")], rat[("full", "up")]),
                       "sig_consistent": res["consistent"], "sig_fresh": res["fresh"]})
        print("   -> PRIMARY-branch margin: sigma must move x%s; any-token margin x%s"
              % (fnum(pr, 3), fnum(R[key]["sig_ratio_full"], 3)))
    for k in skipped:
        print("\n[%s] SKIPPED -- cannot reach FINAL locally.  UNSURE." % k)
        R[k].update({"sig_ratio_primary": None})
    for k, why in SIGMA_FREE.items():
        print("\n[%s] sigma-free: %s -> margin inf" % (k, why))
        R[k].update({"sig_ratio_primary": float("inf")})

def section_summary(R):
    print("\n" + "=" * 100)
    print("SUMMARY -- scorer | batch | class | live terms | moved since scoring? | ever decisive/won? | flip margin | fragile?")
    print("=" * 100)
    for key in ORDER:
        r = R.get(key, {})
        cls = r.get("class", CLASS[key][0])
        live = r.get("ratio")
        print("\n%-5s %-8s %s" % (key, SC[key][1], cls))
        print("      live   : %s" % CLASS[key][1])
        if "moved" in r:
            print("      moved  : %s" % r["moved"])
        if "won" in r:
            print("      max() winners: %s" % ", ".join(r["won"]))
        if "ever_out" in r:
            print("      any level ever outside its band since scoring: %s" % r["ever_out"])
        if "margin" in r:
            print("      live-term margin : %s  -> %s" % (r["margin"], grade(live)))
        if "sig_ratio_primary" in r:
            print("      sigma margin (primary branch): x%s -> %s" % (fnum(r["sig_ratio_primary"], 3), grade(r["sig_ratio_primary"])))
        if "final_scoring" in r:
            print("      FINAL scoring corpus: %s" % r["final_scoring"])
            print("      FINAL HEAD corpus   : %s" % r["final_head"])


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs2", default=DEF_RUNS2)
    ap.add_argument("--runs1", default=DEF_RUNS1)
    ap.add_argument("--extra", default="")
    ap.add_argument("--work", default="")
    ap.add_argument("--section", default="all", choices=("static", "live", "sigma", "all"))
    ap.add_argument("--jobs", type=int, default=6)
    ap.add_argument("--json", default="")
    a = ap.parse_args()
    work = a.work or tempfile.mkdtemp(prefix="cLV1-")
    os.makedirs(work, exist_ok=True)
    sys.path.insert(0, AN)
    ctx = {"RUNS1": a.runs1, "RUNS2": a.runs2, "EXTRA": a.extra, "WORK": work, "RD_CISO2": ""}
    if a.extra and os.path.isdir(os.path.join(a.extra, "ciso2")):
        rd = os.path.join(work, "rd_ciso2")
        os.makedirs(rd, exist_ok=True)
        for f in glob.glob(os.path.join(a.runs2, "ciso2-*.out")):
            dst = os.path.join(rd, os.path.basename(f))
            if not os.path.lexists(dst):
                os.symlink(f, dst)
        if not os.path.lexists(os.path.join(rd, "ciso2")):
            os.symlink(os.path.join(a.extra, "ciso2"), os.path.join(rd, "ciso2"))
        ctx["RD_CISO2"] = rd
    files = sorted(set(v[0] for v in SC.values()))
    ctx["SHA0"] = {f: sha256(os.path.join(AN, f)) for f in files}
    head = git("rev-parse", "HEAD")
    hp = snap(work, "HEAD")
    print("cLV1_live_term_census.py  sha256 %s" % sha256(os.path.abspath(__file__)))
    print("repo HEAD %s   corpus at HEAD: %d rows, sha8 %s   work %s" % (head[:12], hp[1], hp[2], work))
    print("runs2 %s   runs1 %s   extra %s" % (a.runs2, a.runs1, a.extra or "(none)"))
    R = {k: {} for k in ORDER}
    if a.section in ("static", "all"):
        section_static(ctx, R)
    if a.section in ("live", "all"):
        section_live(ctx, R)
    if a.section in ("sigma", "all"):
        section_sigma(ctx, R, a.jobs)
    if a.section == "all":
        section_summary(R)
    after = {f: sha256(os.path.join(AN, f)) for f in files}
    same = after == ctx["SHA0"]
    print("\nregistered scorer files byte-identical before and after this run: %s (%d files)" % (same, len(files)))
    if a.json:
        with open(a.json, "w") as fh:
            json.dump(R, fh, indent=1, default=str)
    return 0 if same else 1


if __name__ == "__main__":
    sys.exit(main())
