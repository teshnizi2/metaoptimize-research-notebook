#!/bin/bash
# =============================================================================
# cVT8_stage_harness.sh -- BUILD OR VERIFY THE ISOLATED `cvt8` HARNESS TREE.  (CORRECTIONS 248.)
#   Touches NO existing file.  Submits nothing.
#
# THE BASE IS cvt7's TREE ($WS/harness_cvt7/cifar10: the LIVE ResNet harness the ISO batches ran from + PATCH_VOTEWEIGHT
# + PATCH_BETAHOLD + PATCH_GROUPHOLD; HF.py 2396f2be...; proved bitwise the LIVE tree when all switches are off, 243.4).
#
#   harness_cvt8/cifar10/<every file of cvt7's tree>   byte-identical (5 *.py + the four HF.py.pre_* backups), except
#   harness_cvt8/cifar10/Optimizers/HF.py              cvt7's HF.py + PATCH_RESTHOLD (HF_POST_SHA)
#   harness_cvt8/cifar10/Optimizers/HF.py.pre_resthold cvt7's HF.py bytes (2396f2be...)
#   harness_cvt8/cifar10/Optimizers/cvt8_isopath.json  the RULE I replay (ISOPATH_SHA)
#   harness_cvt8/cifar10/Optimizers/cvt8_testdown.json TEST-ONLY replay (tests only; no launcher line carries it)
#   harness_cvt8/cifar10/data -> the LIVE data (symlink, the same target as cvt7's)
#   jobs/run_cifar_cvt8.sh                              jobs/run_cifar.sh with EXACTLY the cd line changed
#   harness_cvt8/STAGE-MANIFEST.txt                     sha256 of every staged file
#
# USAGE   bash bin/cVT8_stage_harness.sh            # VERIFY ONLY (exit 0 iff consistent)
#         bash bin/cVT8_stage_harness.sh --stage    # build it (refuses if it exists)
# =============================================================================
set -u
MODE=verify; [ "${1:-}" = "--stage" ] && MODE=stage
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
BASE=$WS/harness_cvt7/cifar10
HROOT=$WS/harness_cvt8
DEST=$HROOT/cifar10
RUNNER_SRC=$WS/jobs/run_cifar.sh
RUNNER=$WS/jobs/run_cifar_cvt8.sh
PATCH_RH=$REPO/patches/patch_resthold.py
ISOJ=$REPO/patches/cvt8_isopath.json
TESTJ=$REPO/patches/cvt8_testdown.json
LIVE_BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_VW_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_CVT4_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
HF_PRE_SHA=2396f2be8102724c120ea5271d2d84acf3835ffb9a5c7947fc282121cf2a588f
HF_POST_SHA=5197dc2eecbbcb6feb815f3798b3455171e6ad614deac913be5df48dd9ede1a9
ISOPATH_SHA=08ab25f3a329cb260bb39fb72f3299c021e7169bf612fa27d539166296e28e70
TESTDOWN_SHA=733a429e74435b90e39183b4a8c7939e02ebd5b46bf154bba8d3e8ba533dc3b4
LIVE_TRAIN_SHA=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
LIVE_RUNNER_SHA=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
CVT7_RUNNER_SHA=805267e4cbcc64172d79824d553a2577ef7cdd55fb8f5e7731c0ea3251a04c56
RUNNER_SHA=1dbe4ff52eccbbb1c9936f4a687ed88172dd41c48b14ae60eb353b45f766f5e0
FAIL=0; bad() { echo "!!! STAGE FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }
module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true
base_files() { (cd "$BASE" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort); }

echo "cVT8_stage_harness: mode=$MODE  WS=$WS"
echo "  base (cvt7's tree, read only) $BASE"
echo "  cvt8 tree                     $DEST"
echo "  runner                        $RUNNER  (from $RUNNER_SRC)"

if [ "$MODE" = stage ]; then
  [ -e "$HROOT" ] && { echo "!!! $HROOT already exists -- refusing to overwrite. Verify it instead."; exit 2; }
  [ -e "$RUNNER" ] && { echo "!!! $RUNNER already exists -- refusing to overwrite."; exit 2; }
  for p in "$PATCH_RH" "$ISOJ" "$TESTJ"; do [ -f "$p" ] || { echo "!!! $p is missing"; exit 2; }; done
  [ "$(sha "$BASE/Optimizers/HF.py")" = "$HF_PRE_SHA" ] || { echo "!!! cvt7's HF.py sha != pinned $HF_PRE_SHA"; exit 2; }
  [ "$(sha "$ISOJ")" = "$ISOPATH_SHA" ] && [ "$(sha "$TESTJ")" = "$TESTDOWN_SHA" ] || { echo "!!! replay file shas != pinned"; exit 2; }
  [ "$(sha "$RUNNER_SRC")" = "$LIVE_RUNNER_SHA" ] || { echo "!!! run_cifar.sh sha != pinned"; exit 2; }
  mkdir -p "$DEST" || exit 2
  for f in $(base_files); do mkdir -p "$DEST/$(dirname "$f")"; cp -p "$BASE/$f" "$DEST/$f" || exit 2; done
  ln -s "$(readlink "$BASE/data")" "$DEST/data" || exit 2
  HF_PATH="$DEST/Optimizers/HF.py" python3 "$PATCH_RH" || { echo "!!! patch_resthold failed"; exit 2; }
  cp -p "$ISOJ" "$DEST/Optimizers/cvt8_isopath.json" && cp -p "$TESTJ" "$DEST/Optimizers/cvt8_testdown.json" || exit 2
  sed "s#^cd $LIVE\$#cd $DEST#" "$RUNNER_SRC" > "$RUNNER" || exit 2
  chmod --reference="$RUNNER_SRC" "$RUNNER" 2>/dev/null || chmod 755 "$RUNNER"
  { echo "STAGED_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"; echo "BASE $BASE"; echo "DEST $DEST"
    echo "PATCH_RESTHOLD_SHA256 $(sha "$PATCH_RH")"
    for f in $(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort); do echo "FILE $(sha "$DEST/$f") $f"; done
    echo "RUNNER $(sha "$RUNNER") $RUNNER"; echo "RUNNER_SRC $(sha "$RUNNER_SRC") $RUNNER_SRC"; } > "$HROOT/STAGE-MANIFEST.txt"
  echo "staged; now verifying"
fi

[ -d "$DEST" ] || { echo "!!! no cvt8 tree at $DEST -- run with --stage"; exit 2; }
N=0
for f in $(base_files); do
  [ "$f" = ./Optimizers/HF.py ] && continue
  if [ ! -f "$DEST/$f" ]; then bad "V-a $f missing from the cvt8 tree"; continue; fi
  [ "$(sha "$BASE/$f")" = "$(sha "$DEST/$f")" ] && N=$((N+1)) || bad "V-a $f differs from cvt7's tree"
done
echo "V-a: $N files byte-identical to cvt7's tree (Optimizers/HF.py excepted)"
EXTRA=$(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort | while read -r f; do
          case "$f" in
            (./Optimizers/HF.py.pre_resthold|./Optimizers/cvt8_isopath.json|./Optimizers/cvt8_testdown.json) ;;
            (*) [ -f "$BASE/$f" ] || echo "$f" ;;
          esac; done)
[ -z "$EXTRA" ] && echo "V-a2: no file in the cvt8 tree beyond cvt7's files, HF.py.pre_resthold and the two replay files" || bad "V-a2 extra files: $EXTRA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_resthold")" = "$HF_PRE_SHA" ] && echo "V-h: HF.py.pre_resthold == cvt7's HF.py $HF_PRE_SHA" \
  || bad "V-h HF.py.pre_resthold != $HF_PRE_SHA"
[ "$(sha "$DEST/Optimizers/HF.py")" = "$HF_POST_SHA" ] && echo "V-h2: cvt8 HF.py sha == pinned post $HF_POST_SHA" \
  || bad "V-h2 cvt8 HF.py sha $(sha "$DEST/Optimizers/HF.py") != pinned post $HF_POST_SHA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_grouphold")" = "$HF_CVT4_SHA" ] && [ "$(sha "$DEST/Optimizers/HF.py.pre_betahold")" = "$HF_VW_SHA" ] \
  && [ "$(sha "$DEST/Optimizers/HF.py.pre_voteweight")" = "$HF_LIVE_SHA" ] \
  && echo "V-h1: the chain .pre_voteweight (LIVE $HF_LIVE_SHA) -> .pre_betahold (cvt1's) -> .pre_grouphold (cvt4's) -> .pre_resthold (cvt7's) -> HF.py" \
  || bad "V-h1 the HF.py.pre_* chain is not the pinned one"
[ "$(sha "$BASE/Optimizers/HF.py")" = "$HF_PRE_SHA" ] && echo "V-h3: cvt7's HF.py is still $HF_PRE_SHA (untouched)" || bad "V-h3 cvt7's HF.py has moved"
grep -q PATCH_RESTHOLD "$LIVE/Optimizers/HF.py" "$BASE/Optimizers/HF.py" && bad "V-h4 the LIVE or cvt7's HF.py carries PATCH_RESTHOLD" \
  || echo "V-h4: neither the LIVE nor cvt7's HF.py carries a PATCH_RESTHOLD marker"
[ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] && echo "V-h5: the LIVE HF.py is still $HF_LIVE_SHA" || bad "V-h5 the LIVE HF.py has moved"
[ "$(sha "$DEST/Optimizers/cvt8_isopath.json")" = "$ISOPATH_SHA" ] && cmp -s "$DEST/Optimizers/cvt8_isopath.json" "$ISOJ" \
  && echo "V-j: Optimizers/cvt8_isopath.json == the repo's, sha $ISOPATH_SHA" || bad "V-j the replay file is not the pinned one"
[ "$(sha "$DEST/Optimizers/cvt8_testdown.json")" = "$TESTDOWN_SHA" ] && echo "V-j2: Optimizers/cvt8_testdown.json sha $TESTDOWN_SHA (TEST ONLY)" \
  || bad "V-j2 the test file is not the pinned one"
[ "$(sha "$DEST/build_network.py")" = "$LIVE_BN_SHA" ] && [ "$(sha "$DEST/train.py")" = "$LIVE_TRAIN_SHA" ] \
  && echo "V-b: build_network.py == the LIVE one $LIVE_BN_SHA; train.py == $LIVE_TRAIN_SHA" || bad "V-b build_network / train sha"
[ -L "$DEST/data" ] && [ "$(readlink "$DEST/data")" = "$(readlink "$BASE/data")" ] && [ -d "$DEST/data/cifar-100-python" ] \
  && echo "V-c: data -> $(readlink "$DEST/data") (symlink, cvt7's target); CIFAR-100 visible" || bad "V-c data symlink"
if [ -f "$RUNNER" ]; then
  DL=$(diff "$RUNNER_SRC" "$RUNNER" | grep '^[<>]')
  D=$(printf '%s\n' "$DL" | grep -c '^[<>]')
  if [ "$D" = 2 ] && echo "$DL" | grep -qx "< cd $LIVE" && echo "$DL" | grep -qx "> cd $DEST"; then
    echo "V-d: $RUNNER differs from run_cifar.sh in EXACTLY one line: cd $LIVE -> cd $DEST"
  else
    bad "V-d the runner diff is not exactly the cd line:"; diff "$RUNNER_SRC" "$RUNNER"
  fi
  [ "$RUNNER_SHA" = RUNNER_SHA_TBD ] && echo "V-d2: NOTE runner sha $(sha "$RUNNER") (not yet pinned)" \
    || { [ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && [ "$(sha "$RUNNER_SRC")" = "$LIVE_RUNNER_SHA" ] \
         && [ "$(sha "$WS/jobs/run_cifar_cvt7.sh")" = "$CVT7_RUNNER_SHA" ] \
         && echo "V-d2: runner sha == pinned $RUNNER_SHA; run_cifar.sh and cvt7's runner unchanged" || bad "V-d2 runner shas moved"; }
else
  bad "V-d no runner at $RUNNER"
fi
if python3 "$REPO/tests/test_resthold.py" --pre "$DEST/Optimizers/HF.py.pre_resthold" --post "$DEST/Optimizers/HF.py" \
     --structural-only >/tmp/cvt8_stage_rh.$$ 2>&1; then
  echo "V-f: tests/test_resthold.py R0 (deleting the four regions == cvt7's HF.py, byte for byte) passes ($(grep -c '^  PASS' /tmp/cvt8_stage_rh.$$) PASS)"
else
  bad "V-f tests/test_resthold.py R0 FAILS"; grep '^  FAIL' /tmp/cvt8_stage_rh.$$ | head
fi
rm -f /tmp/cvt8_stage_rh.$$
if python3 "$REPO/tests/test_grouphold.py" --pre "$DEST/Optimizers/HF.py.pre_grouphold" --post "$DEST/Optimizers/HF.py.pre_resthold" \
     --structural-only >/tmp/cvt8_stage_gh.$$ 2>&1; then
  echo "V-g: cvt7's tests/test_grouphold.py G0 (registered, unedited) passes on the intermediate pair ($(grep -c '^  PASS' /tmp/cvt8_stage_gh.$$) PASS)"
else
  bad "V-g tests/test_grouphold.py G0 FAILS on the intermediate pair"; grep '^  FAIL' /tmp/cvt8_stage_gh.$$ | head
fi
rm -f /tmp/cvt8_stage_gh.$$
[ "$FAIL" = 0 ] && { echo "cVT8_stage_harness: VERIFIED"; exit 0; } || { echo "cVT8_stage_harness: NOT CONSISTENT"; exit 2; }
