#!/bin/bash
# =============================================================================
# Cycle 53 -- N1's THIRD SEED.  cluster_a.  3 jobs.  Tag `ns6-*`.
#
# OPEN QUESTION IT ANSWERS.  It is the one `ns5`'s own header registered in
# advance, and the batch is being bought because that registration says to.
#
#   `bin/c52_nodemin_onset.sh`, under "NOT INCLUDED, deliberately":
#       "A third seed.  n=2 at the new rungs against n=3 at both controls; the
#        statistic is an ARGMIN over gaps of 0.1-0.3, not a mean difference of
#        0.006, so n=2 resolves it.  **If two rungs come back within 0.02 of each
#        other, say UNDECIDED and buy the third seed then.**"
#
#   They came back within 0.02.  Measured (FINDINGS 53.6), N_eff/m on the steady
#   half, VARIANCE instrument, all six new arms box-free at 0.0000%:
#
#       ms       n      lay     node        w    argmin   gap(2 smallest)
#       1e-4     3   0.7000   0.1697   0.1287       w           0.0410
#       2e-4     2   0.6516   0.2244   0.2054       ?         **0.0190**   <-- UNDECIDED
#       5e-4     2   0.6581   0.4483   0.3915       w           0.0568
#       1e-3     3   0.7258   0.4016   0.5045    node           0.1029
#
#   0.0190 < 0.020.  The argmin at ms=2e-4 is a COIN FLIP, and the two admissible
#   readings of the (2e-4, 5e-4) pair are:
#       (w, w)     -- ADMISSIBLE.  One crossing, boundary between 5e-4 and 1e-3.
#       (node, w)  -- **THE REGISTERED REFUTATION** of N1: the shape is not a
#                     monotone function of the meta-stepsize, and CORRECTIONS 70
#                     holds at a POINT rather than on an interval.
#
#   A TEST WHOSE REFUTATION BRANCH IS LIVE CANNOT BE REPORTED AS A PASS.  N1 is
#   recorded UNDECIDED this tick, not PASSED, and these 3 jobs settle it.
#
# WHY THIS IS THE CHEAPEST INFORMATIVE JOB IN THE QUEUE.  Three 20-epoch R18 runs,
# ~10-16 min each, against a claim (CORRECTIONS 70) that is currently the most
# paper-relevant thing the campaign has produced after the sign-agreement
# measurement itself.  Nothing else on offer moves a headline for 3 jobs.
#
# THE DESIGN.  Byte-identical to `ns5-*-m2p4-*` except --seed:
#   R18/CIFAR-10, SGDm/Lion, ms=2e-4, alpha0=1e-3, AUGMENT=1, BETA_CLIP=-30:0.0,
#   20 epochs, rungs w/node/lay, **seed 2 only**.
#   Pooled with the two ns5 seeds this gives n=3 at ms=2e-4, matching both
#   controls.
#
# PRE-REGISTERED PREDICTIONS -- score against these, not against what lands.
#
#   (M0) VALIDITY GATE.  n_records == 2000; n_beta byte-matches ns5:
#        w 11,173,962 / node 14,420 / lay 62; beta MOVED on all 3.
#
#   (M0.3) THE BOX-FREE GATE, SCORED FIRST.  < 5% of records at either guard,
#        per rung, from beta_true_min/max AND PATCH_CLIPCOUNT's per-coordinate
#        counts.  ns5's ms=2e-4 arms read 0.0000% on every denominator and this
#        one differs only in the seed, so a failure means the seed changed
#        something other than the seed, and the batch is VOID not interesting.
#        Scored anyway: "expected to pass trivially" is what moved the wrong wall
#        in cycle 50, and cycle 53 has just watched a pooled box-free row hide a
#        seed that bound (CORRECTIONS 72).
#
#   (M1) THE TEST.  Recompute the ms=2e-4 profile at n=3 and re-read N1.
#        PREDICTION: **argmin stays `w`, and the gap opens past 0.02** -- so the
#        sequence is w -> w -> w -> node and N1 PASSES with the boundary located
#        between 5e-4 and 1e-3.
#        WHY THAT DIRECTION: the neighbouring rungs on both sides have argmin w
#        with decided gaps (1e-4 gap 0.041, 5e-4 gap 0.057), and w is below node
#        at 2e-4 in BOTH ns5 seeds individually, not just in the mean.
#        REFUTATION: argmin flips to `node` with a gap > 0.02 --> the sequence is
#        w -> node -> w -> node, which is the registered N1 refutation.  Then
#        CORRECTIONS 70 is scoped to a POINT and must say so.
#        STILL-UNDECIDED BRANCH: if the gap remains <= 0.02 at n=3, DO NOT buy a
#        fourth seed.  Report the rung as a genuine near-tie and say that the
#        argmin is not the right summary near the crossing -- report the whole
#        profile, which is what N1's own text already demands.
#
#   (M2) REPORTED, NOT TESTED.  N2 was REFUTED this tick (the flip tracks the
#        meta-stepsize, NOT beta span: at ms=5e-4 the weightwise span is 8.27 log
#        units, well past N2's ~5 threshold, and the argmin is still `w`).  The
#        third seed's span is recorded for completeness; it cannot revive N2,
#        which failed on a rung this batch does not touch.
#
#   RESOLUTION.  rho_min is set by T and m and does not depend on the seed, so
#   ns5's floors carry over unchanged: w ~8.5e-09 RESOLVES, node ~7.0e-06
#   RESOLVES, lay ~1.5e-03 MARGINAL at m=62.  The layerwise rung is not the
#   argmin at any point on this ladder and is not load-bearing for M1.
#
# NOT INCLUDED, deliberately:
#   * A third seed at ms=5e-4.  Its gap is 0.057, decided, and nothing hinges on it.
#   * ms between 5e-4 and 1e-3.  Locating the crossing more precisely is only
#     worth buying once N1 is settled, and it is not settled until M1 scores.
#
# WHY cluster_a.  `ns5-*` is on cluster_a and this pools directly with it (CORRECTIONS
# 46(6), account assignment by comparator).
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. M0, M0.3 above, run not assumed, M0.3 FIRST
#   2. python3 analysis/c53_score.py --selftest        (14/14)
#   3. python3 analysis/neff_instrument.py --selftest  (23/23)
#   4. python3 analysis/probe5_window.py --selftest    (41/41)
#   5. Window scan on every arm.
#
# USAGE
#   bash c53_ns_thirdseed.sh            # dry run
#   bash c53_ns_thirdseed.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

WS=/cluster/account_a/metaopt                   # cluster_a
RUNNER=$WS/jobs/run_cifar.sh
HF=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10/Optimizers/HF.py
SAVE=$WS/runs/ns5                                  # SAME dir as ns5: it pools
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=account_a
PEND_CAP=40
NJOBS=3
BOX=-30:0.0
MST=2e-4
SEED=2
EPOCHS=20
SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

echo "=== c53 N1 THIRD SEED (cluster_a) ==="
echo "    ms=$MST  seed=$SEED  BETA_CLIP=$BOX  epochs=$EPOCHS  pools with ns5-*-m2p4-s0/s1"

for P in PATCH_PROBE5 PATCH_PROBE5_FIX PATCH_CLIPCOUNT; do
  grep -q "$P" "$HF" 2>/dev/null || { echo "GUARD FAIL: $P not present in $HF"; exit 1; }
done
echo "guard 1: PATCH_PROBE5 + PATCH_PROBE5_FIX + PATCH_CLIPCOUNT present  OK"

python3 -c "import py_compile; py_compile.compile('$HF', doraise=True)" 2>/dev/null || {
  echo "GUARD FAIL: $HF does not compile"; exit 1; }
echo "guard 2: HF.py compiles  OK"

# GUARD 3 -- "byte-identical except the seed" is a CLAIM and must be DIFFED
# (the uc5 lesson, CORRECTIONS 71).  Diff against ns5's OWN script.
NS5=$(dirname "$0")/c52_nodemin_onset.sh
if [ -r "$NS5" ]; then
  for f in "ResNet18" "CIFAR10" "SGDm" "Lion" "--alpha0 1e-3"; do
    grep -q -- "$f" "$NS5" || { echo "GUARD FAIL: '$f' absent from the ns5 control"; exit 1; }
  done
  grep -q -- "$MST" "$NS5" || {
    echo "GUARD FAIL: ns5 does not contain the stepsize $MST -- this does not pool"; exit 1; }
  grep -q -- 'BOX=-30:0.0' "$NS5" || {
    echo "GUARD FAIL: ns5's box is not -30:0.0 -- this is not a matched seed"; exit 1; }
  echo "guard 3: ns5 is R18/CIFAR10/SGDm/Lion, ms=$MST present, BOX=-30:0.0  OK"
else
  echo "guard 3: c52_nodemin_onset.sh not readable here -- SKIPPED (say so in FINDINGS)"
fi

# GUARD 4 -- the seed must not already exist, or this silently doubles an arm.
CLASH=$(ls -d "$SAVE"/probe_*_m2p4_s${SEED} 2>/dev/null | wc -l | tr -d ' ')
[ "$CLASH" = "0" ] || { echo "GUARD FAIL: $CLASH probe dirs for seed $SEED already exist in $SAVE"; exit 1; }
echo "guard 4: no existing seed-$SEED probe dirs in $SAVE  OK"

FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 5: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$SUBMIT" = "1" ]; then
  [ "$((PEND + NJOBS))" -le "$PEND_CAP" ] || {
    echo "ABORT: $PEND pending + $NJOBS would exceed the $PEND_CAP cap"; exit 1; }
fi

mkdir -p "$SAVE"
N=0
FAIL=0
for G in layerwise nodewise weightwise; do
  case $G in
    layerwise)  TAG=lay  ;;
    nodewise)   TAG=node ;;
    weightwise) TAG=w    ;;
  esac
  RN="ns5-${TAG}-m2p4-s${SEED}"        # ns5-* name so aggregate.py pools it
  PD="$SAVE/probe_${TAG}_m2p4_s${SEED}"
  CMD=(sbatch --job-name="$RN" --partition="$PARTS" --time=03:50:00
    --export="ALL,AUGMENT=1,BETA_CLIP=$BOX,PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=500,PROBE_DIR=$PD"
    "$RUNNER"
    --optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
    --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0
    --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
    --max-time 999:00:00 --gamma 1 --meta-stepsize "$MST" --alpha0 1e-3
    --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$SEED"
    --save-directory "$SAVE" --run-name "$RN")
  if [ "$SUBMIT" = "1" ]; then
    if "${CMD[@]}" >/dev/null; then N=$((N+1)); else FAIL=$((FAIL+1)); fi
  else
    N=$((N+1)); printf '%s\n' "${CMD[*]}"
  fi
done
echo "---- $N jobs ($( [ "$SUBMIT" = 1 ] && echo ACCEPTED BY SLURM || echo 'dry run, nothing submitted' )); $FAIL rejected ----"
[ "$FAIL" -gt 0 ] && echo "!!! $FAIL sbatch calls FAILED -- do not treat this batch as submitted"
echo "reduce with, IN THIS ORDER:"
echo "  1. M0.3 box-free gate, PER SEED, on ../probes_ns5"
echo "  2. python3 analysis/c53_score.py    # re-reads N1 at n=3 on the 2e-4 rung"
