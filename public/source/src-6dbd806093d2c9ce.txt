#!/bin/bash
# =============================================================================
# cVT1_voteweight.sh -- `cvt1`: SEPARATE IDENTITY FROM MAGNITUDE BY INTERVENING ON
#   THE VOTE.  ONE SUBMISSION, 15 JOBS (5 arms x 3 seeds), 100 EPOCHS, SEEDS
#   {78,79,80}, PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.
#   CORRECTIONS 227.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account; every guard and scancel line is filtered on `cvt1-`.
#
# REGISTERED SCORER: analysis/cVT1_voteweight_score.py, committed BEFORE this batch
# is submitted (RULE 21).  Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT1_voteweight_score.py <runsdir>
#
# THE NETWORK RUNS FROM cvt1's OWN ISOLATED TREE, NOT THE LIVE ONE AND NOT cpl1's.
#   $WS/harness_cvt1/cifar10 (bin/cVT1_stage_harness.sh: PATCH_PLAINNET, cpl1's
#   POST_SHA bytes, + PATCH_VOTEWEIGHT in HF.py) and $WS/jobs/run_cifar_cvt1.sh, which
#   differs from jobs/run_cifar.sh in EXACTLY its `cd` line.
#
# THE ARMS.  PlainNet indices (53 tensors); guard 4 re-derives every one on the LIVE
# model by NAME, constructs each arm's HF with its VOTE_W, and writes the manifest.
#   k01     scalar                                VOTE_W unset
#   HEAD    sets:1-49,51-53/layer4.1.bn2.weight   VOTE_W unset       (cpl2's HEAD)
#   MUTE    scalar                                VOTE_W=layer4.1.bn2.weight:0
#   DOSE    scalar                                VOTE_W=layer4.1.bn2.weight:0.1
#   INJECT  sets:1-49,51-53/layer4.1.bn2.weight   VOTE_W=layer4.1.bn1.weight:691
#
# RULE 20.  Every ARGS value is ONE shell token of [A-Za-z0-9_.,:/-]; MUTE/DOSE's
# ARGS lines are k01's and INJECT's are HEAD's except the run name.  VOTE_W rides
# sbatch's --export list (value class [A-Za-z0-9_.:/], no comma) and CANNOT ride the
# runner's ENV line (the runner may differ from run_cifar.sh in its cd line only):
# the patch prints `VOTE_W: off` / `VOTE_W: on ...` at construction on EVERY run,
# and that line is what the separate ENV audit and the scorer's G-VOTEW gate read.
# analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  cpl2: 9.7356 GPU-h / 15 PlainNet runs.  15 jobs ~= 9.7 GPU-h expected;
# HARD BOUND = WALL x 15 = 45 GPU-h.
#
# USAGE   bash bin/cVT1_voteweight.sh               # dry run: every guard, every
#         bash bin/cVT1_voteweight.sh --dry-run     #   composed line printed; submits nothing
#         bash bin/cVT1_voteweight.sh --submit      # the ONLY submitting form
# ENV     CVT1_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u

case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;

[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# the control arms must NOT inherit a VOTE_W through --export=ALL
unset VOTE_W

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"

# shellcheck disable=SC1091

echo " cvt1 -- separate identity from magnitude by intervening on the vote"

echo " harness tree $HARN   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY (co-primary): P_VOTE = HEAD - MUTE; P_INJECT = HEAD - INJECT"

NEWF="analysis/cVT1_voteweight_score.py bin/cVT1_voteweight.sh bin/cVT1_stage_harness.sh bin/cVT1_realrun_inertness.sbatch patches/patch_voteweight.py tests/test_voteweight.py tests/test_voteweight_realrun.py"

  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the seven registration files are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"

  elif [ "$SUBMIT" = 1 ]; then

  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"

  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"

  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"

  && echo "guard 1c5: HEAD (and INJECT's grouping) is byte-identical to cpl2's HEAD" || guard_fail "guard 1c5: HEAD not proved byte-identical to cpl2's"

  && echo "guard 1c6: $(grep "K re-derived" "$SELFT" | sed 's/^  PASS //')" || guard_fail "guard 1c6: K = 691 was not re-derived from cpl2's records on this host"
grep -q "^  PASS E BROKEN PATCH: MUTE's z is the UNWEIGHTED sum" "$SELFT" \
  && echo "guard 1c7: the broken-patch null at OWN-STEP-SIZE's levels reaches PATCH-NOT-VERIFIED, not a branch" || guard_fail "guard 1c7: broken-patch case did not pass"

python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys

if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt1-')

[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \

HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH")

  echo "guard 4e: tests/test_voteweight.py (V0-V6, CPU) ALL PASS on THIS HF and THIS build_network: $(grep -c '^  PASS' "$VLOG") PASS / $(grep -c '^  FAIL' "$VLOG") FAIL"

  guard_fail "guard 4e: no ALL PASS test_voteweight log for HF $HFSHA / BN $BNSHA at $VLOG -- run:

# ---- guard 4c': the literals here, in the scorer and in the stage script agree ------

          "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$POST_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$TESTV" "$TESTR" <<'PYEOF' \
  || guard_fail "guard 4c': launcher / scorer / stage-script / test literals disagree"
import sys, re, ast
sc, st = open(sys.argv[1]).read(), open(sys.argv[2]).read()
bad = []
def blk(txt, name):
    m = re.search(r"^%s = (\{[^\n]*\})[ \t]*(#.*)?$" % name, txt, re.M) or \
        re.search(r"^%s = (\{\n.*?^\})" % name, txt, re.S | re.M)
    return ast.literal_eval(m.group(1)) if m else None
spec, votew = blk(sc, "SPEC"), blk(sc, "VOTEW")
mine = dict(zip(("k01", "HEAD", "MUTE", "DOSE", "INJECT"), sys.argv[3:8]))
bad += ["SPEC:" + k for k in set(mine) | set(spec or {}) if mine.get(k) != (spec or {}).get(k)]
mw = {"k01": "", "HEAD": "", "MUTE": sys.argv[8], "DOSE": sys.argv[9], "INJECT": sys.argv[10]}
bad += ["VOTEW:" + k for k in set(mw) | set(votew or {}) if mw.get(k) != (votew or {}).get(k)]
def lit(name):
    mm = re.search(r"^%s = (.+?)(\s*#.*)?$" % name, sc, re.M)
    return ast.literal_eval(mm.group(1)) if mm else None
if lit("NET") != sys.argv[11]: bad.append("NET")
if lit("NTENS") != int(sys.argv[12]): bad.append("NTENS")
if lit("TOTPAR") != int(sys.argv[13]): bad.append("TOTPAR")
if tuple(lit("SEEDS") or ()) != tuple(int(x) for x in sys.argv[14].split()): bad.append("SEEDS")
for name, v in (("POST_SHA", sys.argv[15]), ("HF_PRE_SHA", sys.argv[16]), ("HF_POST_SHA", sys.argv[17])):
    b = re.search(r"^%s=([0-9a-f]{64})$" % name, st, re.M)
    if lit(name) != v or not b or b.group(1) != v: bad.append(name)
for tf in sys.argv[18:20]:
    tt = open(tf).read()
    tv = blk(tt, "VOTEW")
    tw = blk(tt, "WITNESS")
    sw = blk(sc, "WITNESS")
    if any(tv.get(k) != votew[k] for k in ("MUTE", "DOSE", "INJECT")): bad.append("VOTEW in " + tf)
    if any(tw.get(k) != sw[k] for k in ("MUTE", "DOSE", "INJECT")): bad.append("WITNESS in " + tf)
if bad:
    print("!!! guard 4c': disagreement on %s" % sorted(bad)); raise SystemExit(1)
print("guard 4c': 5 spec strings, 3 VOTE_W values, WITNESS lines, NET, NTENS, TOTPAR, SEEDS and the 3 pinned shas agree in launcher / scorer / stage script / both tests")
PYEOF

# ---- guard 4: THE LIVE-MODEL MANIFEST, built from the cvt1 tree, per arm with NAMES and its VOTE_W ----

python3 - "$HARN" "$MTMP" "$NTENS" "$TOTPAR" "$NET" "$SC" <<'PYEOF' || guard_fail "guard 4: live partition / VOTE_W / source check FAILED"
import sys, os, io, inspect, re, operator, contextlib, importlib.util
harn, out, ntens, totpar, net_name, scp = sys.argv[1], sys.argv[2], int(sys.argv[3]), int(sys.argv[4]), sys.argv[5], sys.argv[6]
spm = importlib.util.spec_from_file_location("cvt1sc", scp); S = importlib.util.module_from_spec(spm); spm.loader.exec_module(S)
sys.path.insert(0, harn); os.chdir(harn)
import torch, torch.nn as nn, torch.fx
from build_network import build_network
from Optimizers.HF import HF
net = build_network(net_name, "cpu"); par = build_network(S.PARENT, "cpu")
nps = [(n, p.data.size()) for n, p in net.named_parameters()]
names = [n for n, _ in nps]; numel = [int(p.numel()) for _n, p in net.named_parameters()]
pnames = [n for n, _ in par.named_parameters()]
own = {}
for mn, m in net.named_modules():
    for pn, _p in m.named_parameters(recurse=False):
        own[(mn + "." if mn else "") + pn] = type(m).__name__
T, tot = len(nps), sum(numel); bad = 0
print("guard 4: live %s from the cvt1 tree -- %d tensors, %d parameters (parent %s: %d)" % (net_name, T, tot, S.PARENT, len(pnames)))
if T != ntens or tot != totpar:
    print("!!! guard 4: live model is %d / %d, the batch assumes %d / %d" % (T, tot, ntens, totpar)); raise SystemExit(1)
removed = [n for n in pnames if n not in names]
if removed != list(S.SHORTCUT_9):
    print("!!! guard 4: PlainNet is not the parent minus exactly the 9 shortcut tensors"); bad += 1
def adds(m):
    g = torch.fx.symbolic_trace(m).graph
    return sum(1 for nd in g.nodes if (nd.op == "call_function" and nd.target in (operator.add, operator.iadd, torch.add))
               or (nd.op == "call_method" and nd.target in ("add", "add_", "__add__", "__iadd__")))
na, nr = adds(net), adds(par)
print("guard 4: torch.fx add ops -- %s %d, %s %d" % (net_name, na, S.PARENT, nr))
if na != 0 or nr != 8: bad += 1
