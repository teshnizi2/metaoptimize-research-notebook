#!/bin/bash
#SBATCH --partition=gpu-short
#SBATCH --gres=gpu:1
#SBATCH --cpus-per-task=8
#SBATCH --mem=16G
#SBATCH --time=02:30:00
#SBATCH --output=/cluster/account_b/metaopt/runs/%x-%j.out

# Generic CIFAR-10 MetaOptimize runner (account account_b).
# NOTE: always pass --max-time 999:00:00 -- train.py has its OWN time-based
# break in the epoch loop which silently truncates runs and makes them
# non-comparable. Slurm's --time above is the real bound.
set -e
module load Python/3.10.4-GCCcore-11.3.0
source /cluster/account_b/metaopt/envs/mo/bin/activate
cd /cluster/account_b/metaopt/harness_cvt1/cifar10

echo "NODE=$(hostname) | JOB=${SLURM_JOB_NAME} ${SLURM_JOB_ID} | AUGMENT=${AUGMENT:-0}"
echo "ARGS: $@"
echo "ENV: AUGMENT=${AUGMENT:-0} BETA_CLIP=${BETA_CLIP:-none} HIER=${HIER:-none} LAM=${LAM:-na} ETA_RATIO=${ETA_RATIO:-na} COS_TOTAL=${COS_TOTAL:-default} COS_WARMUP=${COS_WARMUP:-default} SCHED=${SCHED:-none} SCHED_TOTAL=${SCHED_TOTAL:-none} SCHED_WARMUP=${SCHED_WARMUP:-none} SCHED_MIN=${SCHED_MIN:-none} PROBE=${PROBE:-0} PROBE_DIR=${PROBE_DIR:-none} EB_RHO=${EB_RHO:-na} EB_LOG=${EB_LOG:-0}"
nvidia-smi --query-gpu=name,memory.total --format=csv,noheader
export PYTHONUNBUFFERED=1
python -u train.py "$@"
echo "RUN_DONE"
