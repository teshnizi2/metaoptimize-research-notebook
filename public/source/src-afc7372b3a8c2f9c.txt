#!/bin/bash
# =============================================================================
# cVT2_injectladder.sh -- `cvt2`: AN INJECT DOSE LADDER ON PlainNet18_c100 --
#   THRESHOLD OR GRADED IN K, AND WHAT WAS cvt1's 0.34 RESIDUE?
#   ONE SUBMISSION, 21 JOBS (7 arms x 3 seeds), 100 EPOCHS, SEEDS {87,88,89},
#   PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1 ON EVERY ARM.  CORRECTIONS 233.
#
# ACCOUNT: account_b (cluster_b).  WORKSPACE: /cluster/account_b/metaopt.  Nothing here
# touches the other account; every guard and scancel line is filtered on `cvt2-`.
#
# REGISTERED SCORER: analysis/cVT2_injectladder_score.py, committed BEFORE this batch
# is submitted (RULE 21).  Every bar is a frozen literal (O2).
#     DOCUMENTED INVOCATION:  python3 analysis/cVT2_injectladder_score.py <runsdir>
#
# NO NEW HARNESS CODE.  The network runs from cvt1's OWN isolated tree, UNCHANGED:
#   $WS/harness_cvt1/cifar10 (PATCH_PLAINNET + PATCH_VOTEWEIGHT, verified by cvt1's
#   bin/cVT1_stage_harness.sh in VERIFY mode) and $WS/jobs/run_cifar_cvt1.sh.
#   PATCH_VOTEWEIGHT already takes `<name>:<w>`; the only new code is the scorer,
#   this launcher and tests/test_voteweight_ladder_realrun.py (which re-proves
#   inertness and bite for EXACTLY these VOTE_W strings on a real GPU run).
#
# THE ARMS.  PlainNet indices (53 tensors); guard 4 re-derives every one on the LIVE
# model by NAME, constructs each arm's HF with its VOTE_W, and writes the manifest.
#   k01     scalar                                VOTE_W unset
#   HEAD    sets:1-49,51-53/layer4.1.bn2.weight   VOTE_W unset                     (K = 1)
#   K13     HEAD's string                         VOTE_W=layer4.1.bn1.weight:13
#   K33     HEAD's string                         VOTE_W=layer4.1.bn1.weight:33
#   K152    HEAD's string                         VOTE_W=layer4.1.bn1.weight:152
#   K691    HEAD's string                         VOTE_W=layer4.1.bn1.weight:691   (cvt1's INJECT)
#   K2000   HEAD's string                         VOTE_W=layer4.1.bn1.weight:2000
#   The K values are DERIVED from cvt1's probe records (analysis/cvt2_registration_derivations.py).
#
# RULE 20.  Every ARGS value is ONE shell token; the ladder arms' ARGS lines are HEAD's
# except the run name.  VOTE_W rides sbatch's --export list and CANNOT ride the runner's
# ENV line: the patch prints `VOTE_W: off` / `VOTE_W: on ...` at construction on EVERY
# run, and that line is what the separate ENV audit and the scorer's G-VOTEW gate read.
# analysis/argsline_guard.py IS NOT EDITED.
#
# COST.  cvt1: 10.3783 GPU-h / 15 PlainNet runs = 0.6919 GPU-h / run.  21 jobs ~= 14.5
# GPU-h expected; HARD BOUND = WALL x 21 = 63 GPU-h.
#
# USAGE   bash bin/cVT2_injectladder.sh               # dry run: every guard, every
#         bash bin/cVT2_injectladder.sh --dry-run     #   composed line printed; submits nothing
#         bash bin/cVT2_injectladder.sh --submit      # the ONLY submitting form
# ENV     CVT2_REGISTERED_COMMIT=<sha>   on a host with no git repo, --submit requires it
# =============================================================================
set -u
SUBMIT=0
case "${1:-}" in
  --submit)      SUBMIT=1 ;;
  --dry-run|"")  SUBMIT=0 ;;
  *) echo "usage: bash $0 [--dry-run|--submit]"; exit 2 ;;
esac
[ $# -le 1 ] || { echo "usage: bash $0 [--dry-run|--submit]"; exit 2; }
# the control arms must NOT inherit a VOTE_W through --export=ALL
unset VOTE_W
FAIL=0; guard_fail() { echo "!!! GUARD FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

USER_NAME="${USER:-$(whoami)}"
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HARN=$WS/harness_cvt1/cifar10
RUNNER=$WS/jobs/run_cifar_cvt1.sh
SAVE=$WS/runs/cvt2
CVT1SAVE=$WS/runs/cvt1

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

CLIP_C="-15:-2.3026"
EPOCHS=100; BATCH=100; GAMMA=1; AUG=1; PROBE=100; PT=1
DSET=CIFAR100
NET=PlainNet18_c100
SEEDS="87 88 89"
MST=1e-3
A0=1e-6
NTENS=53
TOTPAR=11046308
G_K01='scalar'
G_HEAD='sets:1-49,51-53/layer4.1.bn2.weight'
W_K13='layer4.1.bn1.weight:13'
W_K33='layer4.1.bn1.weight:33'
W_K152='layer4.1.bn1.weight:152'
W_K691='layer4.1.bn1.weight:691'
W_K2000='layer4.1.bn1.weight:2000'
ARMS="k01 HEAD K13 K33 K152 K691 K2000"
NJOBS=21
WALL=03:00:00
POST_SHA=e65e67738e45418045ccc7f737fb0ea5b2523a9294a4c3a288e543e95508a18b
HF_PRE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_POST_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
# shellcheck disable=SC1091
. "$REPO/bin/_lib_guards.sh"
PARTS=$(slurm_parts_for_wall "$WALL")

SC="$REPO/analysis/cVT2_injectladder_score.py"
CE="$REPO/analysis/corpus_exclusions.py"
CET="$REPO/results/CORPUS-EXCLUSIONS.tsv"
DERIV="$REPO/analysis/cvt2_registration_derivations.py"
SELF="$REPO/bin/cVT2_injectladder.sh"
LSB="$REPO/bin/cVT2_ladder_inertness.sbatch"
TESTL="$REPO/tests/test_voteweight_ladder_realrun.py"
SC1="$REPO/analysis/cVT1_voteweight_score.py"
STAGEF="$REPO/bin/cVT1_stage_harness.sh"
PATCHV="$REPO/patches/patch_voteweight.py"
PATCHF="$REPO/patches/patch_plainnet.py"
TESTV="$REPO/tests/test_voteweight.py"
TESTR="$REPO/tests/test_voteweight_realrun.py"
TESTT="$REPO/tests/test_probe_tensor.py"
TESTF="$REPO/tests/test_plainnet.py"
LIBF="$REPO/bin/_lib_guards.sh"
AGF="$REPO/analysis/argsline_guard.py"
CSVP="$REPO/results/all_runs.csv"
HFH="$HARN/Optimizers/HF.py"
HFVW="$HARN/Optimizers/HF.py.pre_voteweight"
HFPRE="$HARN/Optimizers/HF.py.pre_probe_tensor"
BNH="$HARN/build_network.py"

echo "=============================================================="
echo " cvt2 -- an INJECT dose ladder: threshold or graded in K, and what was the 0.34?"
echo " $DSET / $NET (PlainNet indices)   AUGMENT=$AUG  PROBE=$PROBE  PROBE_TENSOR=$PT  clamp $CLIP_C"
echo " ms=$MST alpha0=$A0  horizon $EPOCHS epochs; seeds $SEEDS"
echo "   k01    $G_K01      VOTE_W unset"
echo "   HEAD   $G_HEAD      VOTE_W unset"
for a in K13 K33 K152 K691 K2000; do v="W_$a"; echo "   $a  $G_HEAD      VOTE_W=${!v}"; done
echo " 7 arms x 3 seeds = $NJOBS jobs, ONE submission; mode: $([ "$SUBMIT" = 1 ] && echo SUBMIT || echo DRY-RUN)"
echo " harness tree $HARN (cvt1's, unchanged)   runner $RUNNER"
echo " partitions $PARTS   walltime $WALL"
echo " PRIMARY: SHAPE of K13/K33/K152 vs HEAD and K691; TOP = K691 - K2000"
echo " DOCUMENTED INVOCATION: python3 analysis/cVT2_injectladder_score.py <runsdir>"
echo "=============================================================="

[ -z "${VOTE_W+x}" ] && echo "guard 0b: VOTE_W is UNSET in the launcher's environment (control arms cannot inherit it)" \
                     || guard_fail "guard 0b: VOTE_W is set in the launcher's environment"
[ -f "$CSVP" ] && echo "guard 0: CSV present ($(wc -l < "$CSVP") lines, sha $(sha "$CSVP"))" \
               || guard_fail "guard 0: no CSV at $CSVP"

# ---- guard 1a: EVERY file this launcher needs --------------------------------
for f in "$SC" "$CE" "$CET" "$DERIV" "$SELF" "$LSB" "$TESTL" "$SC1" "$STAGEF" "$PATCHV" "$PATCHF" "$TESTV" "$TESTR" "$TESTT" \
         "$TESTF" "$LIBF" "$AGF" "$CSVP" "$RUNNER" "$HFH" "$HFVW" "$HFPRE" "$BNH"; do
  [ -f "$f" ] && echo "guard 1a: present  $(sha "$f" | cut -c1-16)  $f" || guard_fail "guard 1a: missing $f"
done

# ---- guard 1b: the registration files are COMMITTED (RULE 21) ------------------
NEWF="analysis/cVT2_injectladder_score.py analysis/cvt2_registration_derivations.py bin/cVT2_injectladder.sh bin/cVT2_ladder_inertness.sbatch tests/test_voteweight_ladder_realrun.py"
REUSEDF="analysis/corpus_exclusions.py results/CORPUS-EXCLUSIONS.tsv analysis/cVT1_voteweight_score.py bin/cVT1_stage_harness.sh patches/patch_voteweight.py patches/patch_plainnet.py tests/test_voteweight.py tests/test_voteweight_realrun.py tests/test_probe_tensor.py tests/test_plainnet.py bin/_lib_guards.sh analysis/argsline_guard.py"
if git -C "$REPO" rev-parse --git-dir >/dev/null 2>&1; then
  UNTR=""; for f in $NEWF $REUSEDF; do git -C "$REPO" ls-files --error-unmatch "$f" >/dev/null 2>&1 || UNTR="$UNTR $f"; done
  DIRTY=$(git -C "$REPO" status --porcelain -- $NEWF $REUSEDF 2>/dev/null)
  if [ -z "$UNTR" ] && [ -z "$DIRTY" ]; then
    echo "guard 1b: the five new registration files and the twelve reused ones are TRACKED and COMMITTED at $(git -C "$REPO" rev-parse --short HEAD) (RULE 21)"
  else
    guard_fail "guard 1b: not committed -- untracked:${UNTR:- none}; dirty: ${DIRTY:-none}"
  fi
else
  if [ -n "${CVT2_REGISTERED_COMMIT:-}" ]; then
    echo "guard 1b: no git repo on this host; the operator declares the staged tree is commit $CVT2_REGISTERED_COMMIT"
    echo "          (RULE 21 is re-verified by wall clock: that commit's %ct vs the earliest sacct Submit)"
  elif [ "$SUBMIT" = 1 ]; then
    guard_fail "guard 1b: no git repo at $REPO and CVT2_REGISTERED_COMMIT is unset -- declare the commit whose tree was staged"
  else
    echo "guard 1b: NOTE (dry run) -- no git repo at $REPO; --submit will require CVT2_REGISTERED_COMMIT=<sha>"
  fi
fi

# ---- guard 1c: the scorer selftest, INCLUDING its O2 proof, the filtered floor and the invocation ----
SELFT=/tmp/cvt2_selftest.$$
python3 "$SC" --selftest --runsdir "$WS/runs" --csv "$CSVP" >"$SELFT" 2>&1 \
  && echo "guard 1c: scorer --selftest PASSES ($(grep -c '^  PASS' "$SELFT") PASS, $(grep -c '^  FAIL' "$SELFT") FAIL, $(grep -c '^  SKIP' "$SELFT") SKIP)" \
  || { guard_fail "guard 1c: scorer --selftest FAILED"; grep '^  FAIL\|EXCEPTION' "$SELFT" | head -20; tail -5 "$SELFT"; }
[ "$(grep -c '^  SKIP' "$SELFT")" = 0 ] && echo "guard 1c0: 0 SKIP (every host-only check ran on this host)" \
  || guard_fail "guard 1c0: $(grep -c '^  SKIP' "$SELFT") selftest check(s) SKIPPED on this host"
grep -qE "^  PASS O2 SIGN-SAT-THRESHOLD +real\+invented " "$SELFT" && grep -q "^  PASS O2 non-vacuity" "$SELFT" \
  && echo "guard 1c2: the O2 invariance proof RAN against THIS host's real corpus plus invented rows at cvt2's cells, and passed" \
  || guard_fail "guard 1c2: the O2 invariance proof did not run against the real corpus on this host"
grep -q "^  PASS .python3 cVT2_injectladder_score.py <runsdir>. exits 0" "$SELFT" \
  && grep -q "^  PASS ... and on a dropped run exits 2" "$SELFT" \
  && echo "guard 1c3: the DOCUMENTED one-argument invocation reached FINAL on synthetic runs; a dropped run -> INCOMPLETE" \
  || guard_fail "guard 1c3: the documented invocation / dropped-run check did not pass"
grep -q "^  PASS ZERO corpus rows carry seed 87, 88 or 89" "$SELFT" \
  && echo "guard 1c4: the scorer's own fresh-seed check ran and passed" || guard_fail "guard 1c4: fresh-seed check did not pass"
grep -q "^  PASS SIGMA_PLAIN re-derives THROUGH filter_rows == frozen" "$SELFT" \
  && grep -q "^  PASS SIGMA_R18ALL re-derives THROUGH filter_rows == frozen" "$SELFT" \
  && grep -q "^  PASS the UNFILTERED PlainNet sigma is" "$SELFT" \
  && echo "guard 1c5: both frozen sigmas re-derive THROUGH corpus_exclusions.filter_rows; the unfiltered PlainNet sigma is NOT the literal (OPERATIONS s36)" \
  || guard_fail "guard 1c5: the filtered noise floor was not proved on this host"
grep -q "^  PASS the ladder re-derived from cvt1's probe records on this host" "$SELFT" \
  && echo "guard 1c6: $(grep "the ladder re-derived" "$SELFT" | sed 's/^  PASS //')" || guard_fail "guard 1c6: the ladder was not re-derived from cvt1's records on this host"
grep -q "^  PASS K691's VOTE_W and witness are BYTE-IDENTICAL to cvt1's INJECT" "$SELFT" \
  && echo "guard 1c7: K691 is byte-identical to cvt1's INJECT" || guard_fail "guard 1c7: K691 not proved identical to cvt1's INJECT"
[ "$(grep -c 'FLOOR GATE: no non-anchor arm is predicted at the floor' "$SELFT")" = 6 ] \
  && ! grep -q '^  FAIL .*FLOOR GATE' "$SELFT" \
  && echo "guard 1c8: FLOOR GATE passes under all 6 accounts (no non-anchor arm predicted at the floor)" \
  || guard_fail "guard 1c8: the floor gate did not pass under every account"
grep -q "^  PASS E WRONG K: K2000's records carry x691" "$SELFT" \
  && echo "guard 1c9: a run carrying the WRONG K reaches PATCH-NOT-VERIFIED, not a branch" || guard_fail "guard 1c9: wrong-K case did not pass"
echo "guard 1c: scorer sha256 $(sha "$SC")"
python3 "$CE" --check --runs "$WS/runs" >/tmp/cvt2_ce.$$ 2>&1 \
  && echo "guard 1d: corpus_exclusions.py --check PASS on this host's corpus ($(grep -c . "$CET") TSV lines)" \
  || { guard_fail "guard 1d: corpus_exclusions.py --check FAILED"; tail -6 /tmp/cvt2_ce.$$; }

# ---- guard 2: RULE 21 premise -- nothing of this batch exists yet ----------------
grep -q '^cvt2-' "$CSVP" && guard_fail "guard 2: cvt2-* already in the CSV" || echo "guard 2: no cvt2-* row in the corpus"
python3 - "$CSVP" "$SEEDS" "$NET" <<'PYEOF' || guard_fail "guard 2a/2c: corpus premise failed"
import csv, sys
rows = list(csv.DictReader(open(sys.argv[1])))
seeds = set(sys.argv[2].split())
hit = [r["run"] for r in rows if r.get("seed") in seeds]
gn = sorted({r["run"].split("-")[0] for r in rows if r.get("network") == sys.argv[3]})
print("guard 2c: rows carrying seed %s: %d" % ("/".join(sorted(seeds)), len(hit)))
print("guard 2a: %s appears only under prefixes %s" % (sys.argv[3], gn))
raise SystemExit(1 if hit or not set(gn) <= {"cpl1", "cpl2", "cvt1"} else 0)
PYEOF
NOUT=$(ls "$WS/runs"/cvt2-*.out 2>/dev/null | wc -l | tr -d ' ')
[ "${NOUT:-0}" = 0 ] && echo "guard 2b: no cvt2-*.out on disk" || guard_fail "guard 2b: $NOUT cvt2-*.out already exist"
NSEED=$(grep -h -m1 '^ARGS:' "$WS/runs"/*.out 2>/dev/null | grep -c -E -- '--seed (87|88|89)( |$)')
[ "${NSEED:-0}" = 0 ] && echo "guard 2d: ZERO .out ARGS lines under $WS/runs carry --seed 87/88/89" \
                      || guard_fail "guard 2d: $NSEED .out files already ran seed 87/88/89"
if command -v sacct >/dev/null 2>&1; then
  NS=$(sacct -u "$USER_NAME" -S 2025-01-01 -X -n -o JobName%60 2>/dev/null | awk '{print $1}' | grep -c '^cvt2-')
  [ "${NS:-0}" = 0 ] && echo "guard 2e: sacct all-time: 0 cvt2-* jobs" || guard_fail "guard 2e: sacct shows $NS cvt2-* jobs already"
  NQ=$(squeue -h -u "$USER_NAME" -o '%j' 2>/dev/null | grep -c '^cvt2-')
  [ "${NQ:-0}" = 0 ] && echo "guard 2f: 0 cvt2-* in squeue" || guard_fail "guard 2f: $NQ cvt2-* already queued"
fi
[ ! -e "$SAVE/PROVENANCE.txt" ] && echo "guard 2g: no submit-mode PROVENANCE.txt under $SAVE" || guard_fail "guard 2g: $SAVE/PROVENANCE.txt already exists"

# ---- guard 3/4a: cvt1's isolated tree is consistent (VERIFY mode) and sees CIFAR-100 -----------
bash "$STAGEF" >/tmp/cvt2_stage.$$ 2>&1 \
  && { echo "guard 4a: bin/cVT1_stage_harness.sh (VERIFY mode, unedited) VERIFIES cvt1's tree:"; grep -E 'V-|VERIFIED' /tmp/cvt2_stage.$$ | sed 's/^/          /'; } \
  || { guard_fail "guard 4a: cvt1's tree is NOT consistent"; tail -20 /tmp/cvt2_stage.$$; }
[ -d "$HARN/data/cifar-100-python" ] && echo "guard 3: cifar-100-python visible under $HARN/data" \
  || guard_fail "guard 3: CIFAR-100 not visible at $HARN/data/cifar-100-python"
[ "$(sha "$BNH")" = "$POST_SHA" ] && echo "guard 3b: build_network.py == cpl1's POST_SHA e65e6773...a18b" \
  || guard_fail "guard 3b: build_network.py is not cpl1's POST_SHA"
[ "$(sha "$HFH")" = "$HF_POST_SHA" ] && [ "$(sha "$HFVW")" = "$HF_PRE_SHA" ] && [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_PRE_SHA" ] \
  && [ "$(sha "$HFPRE")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "guard 3c: HF.py == HF_POST_SHA 3f2b98e1...; HF.py.pre_voteweight == the LIVE HF.py == HF_PRE_SHA 4732b74a...; .pre_probe_tensor == live" \
  || guard_fail "guard 3c: the tree's HF files are not the pinned ones (or the live HF.py moved)"
grep -q PATCH_VOTEWEIGHT "$LIVE/Optimizers/HF.py" && guard_fail "guard 3d: the LIVE HF.py carries PATCH_VOTEWEIGHT" \
  || echo "guard 3d: the LIVE HF.py carries no PATCH_VOTEWEIGHT"

# ---- guard 4b/4e/4f: the patch, proved inert and biting ------------------------------
python3 "$TESTT" --pre "$HFPRE" --post "$HFVW" --structural-only >/tmp/cvt2_struct.$$ 2>&1 \
   && echo "guard 4b: PATCH_PROBE_TENSOR structural inertness R0 PASSES on HF.py.pre_voteweight (= live)" \
   || { guard_fail "guard 4b: PATCH_PROBE_TENSOR structural R0 FAILED on HF.py.pre_voteweight"; tail -8 /tmp/cvt2_struct.$$; }
HFSHA=$(sha "$HFH"); BNSHA=$(sha "$BNH")
VLOG="$CVT1SAVE/inertness_voteweight.log"
if [ -f "$VLOG" ] && grep -q '^ALL PASS$' "$VLOG" && grep -q "^POST_SHA256 $HFSHA" "$VLOG" && grep -q "^BN_SHA256 $BNSHA" "$VLOG"; then
  echo "guard 4e: cvt1's tests/test_voteweight.py log (V0-V6, CPU) ALL PASS on THIS HF and THIS build_network: $(grep -c '^  PASS' "$VLOG") PASS / $(grep -c '^  FAIL' "$VLOG") FAIL"
else
  guard_fail "guard 4e: no ALL PASS test_voteweight log for HF $HFSHA / BN $BNSHA at $VLOG"
fi
RLOG="$CVT1SAVE/inertness_voteweight_realrun.log"
if [ -f "$RLOG" ] && grep -q '^ALL PASS$' "$RLOG" && grep -q "^TREE_POST $HARN HF $HFSHA build_network $BNSHA" "$RLOG" \
   && grep -q "^REALRUN_RC 0$" "$RLOG"; then
  echo "guard 4f: cvt1's real-run inertness log (RR0-RR3) ALL PASS on THIS tree: $(grep -c '^  PASS' "$RLOG") PASS / $(grep -c '^  FAIL' "$RLOG") FAIL"
else
  guard_fail "guard 4f: cvt1's real-run inertness log is not ALL PASS for this tree at $RLOG"
fi
LLOG="$SAVE/inertness_ladder_realrun.log"
if [ -f "$LLOG" ] && grep -q '^ALL PASS$' "$LLOG" && grep -q "^TREE_POST $HARN HF $HFSHA build_network $BNSHA" "$LLOG" \
   && grep -q "^REALRUN_RC 0$" "$LLOG" && grep -q "^REGISTERED_TEST_SHA256 $(sha "$TESTR")$" "$LLOG" \
   && grep -q "^THIS_TEST_SHA256 $(sha "$TESTL")$" "$LLOG" \
   && grep -q "PASS RR0 unpatched k01 vs itself: every parameter and buffer after the last step BITWISE-IDENTICAL" "$LLOG"; then
  echo "guard 4g0: tests/test_voteweight_ladder_realrun.py (RR0-RR3 for cvt2's EXACT strings, GPU, real CIFAR-100 steps) ALL PASS on THIS tree: $(grep -c '^  PASS' "$LLOG") PASS / $(grep -c '^  FAIL' "$LLOG") FAIL"
  python3 - "$SC" "$LLOG" <<'PYEOF' || guard_fail "guard 4g1: the ladder log's LADDER lines differ from the scorer's VOTEW / WITNESS"
import sys, importlib.util
sp = importlib.util.spec_from_file_location("s", sys.argv[1]); S = importlib.util.module_from_spec(sp); sp.loader.exec_module(S)
got = [ln.rstrip("\n") for ln in open(sys.argv[2]) if ln.startswith("LADDER ")]
want = ["LADDER %s %s %s" % (a, S.VOTEW[a] or "unset", S.WITNESS[a]) for a in S.ARMS]
bite = [a for a in S.LADDER if not any(ln.startswith("  PASS RR3 %-6s" % a) and "BITWISE" in ln for ln in open(sys.argv[2]))]
print("guard 4g1: the ladder log proved inertness/bite for exactly the scorer's %d arms' VOTE_W strings and witnesses: %s; RR3 isolated-group bitwise PASS on all ladder arms: %s" % (len(want), got == want, not bite))
raise SystemExit(0 if got == want and not bite else 1)
PYEOF
else
  guard_fail "guard 4g0: no ALL PASS ladder real-run log for HF $HFSHA and THIS test at $LLOG -- sbatch $LSB, then copy its .out there"
fi

# ---- guard 4c': the literals here, in the scorer, in the stage script and in cvt1's scorer agree ------
python3 - "$SC" "$STAGEF" "$SC1" "$G_K01" "$G_HEAD" "$W_K13" "$W_K33" "$W_K152" "$W_K691" "$W_K2000" \
          "$NET" "$NTENS" "$TOTPAR" "$SEEDS" "$POST_SHA" "$HF_PRE_SHA" "$HF_POST_SHA" "$ARMS" <<'PYEOF' \
  || guard_fail "guard 4c': launcher / scorer / stage-script literals disagree"
import sys, re, ast, importlib.util
sp = importlib.util.spec_from_file_location("s", sys.argv[1]); S = importlib.util.module_from_spec(sp); sp.loader.exec_module(S)
st = open(sys.argv[2]).read()
c1 = open(sys.argv[3]).read()
bad = []
arms = sys.argv[18].split()
if tuple(arms) != tuple(S.ARMS): bad.append("ARMS")
mine_spec = {"k01": sys.argv[4]}
mine_spec.update(dict((a, sys.argv[5]) for a in arms if a != "k01"))
if mine_spec != S.SPEC: bad.append("SPEC")
mw = {"k01": "", "HEAD": ""}
mw.update(dict(zip(("K13", "K33", "K152", "K691", "K2000"), sys.argv[6:11])))
if mw != S.VOTEW: bad.append("VOTEW")
if S.NET != sys.argv[11] or S.NTENS != int(sys.argv[12]) or S.TOTPAR != int(sys.argv[13]): bad.append("NET/NTENS/TOTPAR")
if tuple(S.SEEDS) != tuple(int(x) for x in sys.argv[14].split()): bad.append("SEEDS")
