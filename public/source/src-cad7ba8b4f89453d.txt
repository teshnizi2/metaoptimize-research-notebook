#!/usr/bin/env bash
# =============================================================================
# c74_tuned_weightwise_probe.sh -- `tw0`, cycle 74.
# cluster_a.  9 jobs.  100 epochs.  ms=1e-4 FIXED.  BETA_CLIP=-15:-2.3026 FIXED.
# Rungs weightwise/nodewise/layerwise x seeds 0-2, PROBE=5 AND PROBE5=1.
#
# WHY THIS EXISTS -- IT TESTS THIS TICK'S OWN HEADLINE AGAINST ITSELF.
#
# cycle 74 scored `wm9` and REFUTED its registered W2: tuned weightwise reaches
# plateau5 = 91.015 against a registered [92.2, 92.9], i.e. the granularity
# ladder is NOT flat once every rung is ms-tuned -- weightwise pays 1.25 pp
# against the nearest tuned rung (scalar 92.262) and 1.85 pp against the best
# (layerwise 92.867), where those four coarse rungs span only 0.604 pp among
# themselves.  Every one of wm9's 10 runs is >1.1 pp below the band, so the
# refutation does not lean on W1 (UNDECIDED, gap/SE 0.18).
#
# THERE IS EXACTLY ONE WAY THAT RESULT IS AN ARTIFACT, AND THIS BATCH TESTS IT.
# `wm9` ran with NO PROBE, deliberately, so its 13 axes would match the rs-/ms-
# cells it extends (CORRECTIONS 102.7 stated that cost in advance).  So NOTHING
# IS KNOWN ABOUT WHETHER wm9's WEIGHTWISE ARM WAS PINNED AGAINST ITS OWN CLIP
# BOX.  If it was, the 1.25 pp deficit is a clamp artifact and not a property of
# granularity at all -- and this tick's headline would be the same class of
# error as CORRECTIONS 99/100.
#
# THE PRIOR, MEASURED THIS TICK AT ZERO GPU COST, AND WHY IT IS NOT SUFFICIENT.
# `probes_ml5` sits in THIS EXACT BOX (-15, -2.3026; c55_neff_noise.BOXES) and
# its m4 rung is THIS EXACT ms (1e-4).  Its weightwise arms are box-free at
# BOTH guards on 3/3 seeds -- rec_lo = rec_hi = 0.0000, q4 both 0.0000.  So at
# ms=1e-4 in this box the clamp is slack at both ends.
#   **BUT ml5 IS A 20-EPOCH LADDER AND wm9 IS 100 EPOCHS**, and `bf9` measured
#   THIS TICK that the bind GROWS WITH BUDGET: at 80 ep, rec_hi goes 0.0000 on
#   layerwise 4/4 but BINDS on nodewise 2/4 and weightwise 3/4 (0.2101-0.5299).
#   A 20-epoch box-free reading therefore does NOT license a 100-epoch claim.
# That gap is exactly what this batch closes, and it can only be closed by
# running the probe at the budget the claim is made at.
#
# IT ALSO PAYS FOR A SECOND THING THE CAMPAIGN HAS NEVER HAD.
# Every per-coordinate sign-agreement / N_eff number in this project was taken
# at an UNTUNED weightwise operating point -- ms=1e-3 or coarser, which FINDINGS
# 72.4 shows is past the optimum of every swept rung.  ms=1e-4 is the argmax for
# weightwise (wm9), for layerwise AND for scalar (13-axis census), so this is
# the FIRST cell in which the field is measured with every rung at or beside its
# own best meta-stepsize.  Direction C's measurement has never been read there.
#
# THE PRE-REGISTERED GATES.  Nothing below may be edited after the data lands.
#
#   T0    VALIDITY.  n_records == 10000 (100 ep x 500 steps / stride 5) on every
#         dir; n_beta == {w 11173962, node 14420, lay 62} on EVERY record, not
#         just record 0; beta moved; epochs_done == 100 == requested.
#
#   T0.2  THE INSTRUMENT FIRED.  neg_counts.json present, n_tot == n_beta for
#         the rung, neg_counts.npy shape == (n_tot,) READ FROM ITS HEADER.
#         This is bf8's failure (PROBE=5 exported, PROBE5=1 never) and it is
#         also c74_bf9_score's own first-draft failure (shape inferred from file
#         size assuming int64; the array is int32).  STANDING RULE (19) twice.
#
#   T1    **THE PRIMARY GATE.  THE BOX CONFOUND ON THIS TICK'S HEADLINE.**
#         Is the WEIGHTWISE arm box-free at BOTH guards at 100 epochs, in wm9's
#         box, at wm9's argmax ms?  Scored per seed with c52_boxfree.occupancy,
#         the PUBLISHED rec_-based 5% gate, PRIMARY and UNCHANGED.  coord_lo /
#         coord_hi are reported ALONGSIDE and are NEVER the gate -- re-reading a
#         gate at whichever resolution flatters the data is the failure this
#         campaign has repeated most (CORRECTIONS 102.4).
#         **REGISTERED PREDICTION: weightwise is BOX-FREE on >= 2 of 3 seeds.**
#         CONFIRMS -> wm9's W2 refutation is NOT a box artifact.  The ladder is
#              genuinely not flat, the finest partition carries a real cost that
#              ms-tuning does not buy back, and -- because the clamp is then
#              provably slack -- the ceiling-runaway mechanism bf9 suggests is
#              RULED OUT as the explanation, leaving the deficit unexplained.
#              That is a sharper result than the deficit alone.
#         REFUTES (weightwise bound on >= 2 of 3) -> **wm9's W2 REFUTATION IS
#              CONFOUNDED WITH THE BOX AND MUST BE RESTATED.**  The 1.25 pp
#              deficit would then be a clamp artifact, the "ladder is not flat"
#              sentence is WITHDRAWN, and the next batch re-runs wm9 in a box
#              wide enough to be slack at 100 ep.
#         Either way it is a result, and it decides the status of the headline.
#         **NOTE THE DIRECTION OF THE CONFLICT OF INTEREST: the outcome that
#         CONFIRMS is the one that keeps this tick's own headline alive.  That
#         is precisely why the threshold is written down here, before the run.**
#
#   T2    ACCURACY REPRODUCTION.  **CAN ONLY VOID.**  tw0's weightwise plateau5
#         vs wm9's ms=1e-4 cell = 91.015 (n=2).  Bar |diff| <= 0.50 pp (STANDING
#         RULE 9's level bar).  CORRECTIONS 18 records that the probe leaves
#         training untouched; this gate MEASURES that rather than trusting it.
#         FAILS -> the probe perturbed training, and T1 is not evidence about
#         wm9 at all because it is no longer the same run.
#
#   T3    SIGN AGREEMENT AT A TUNED OPERATING POINT.  **DESCRIPTIVE.  NO
#         DIRECTION IS REGISTERED, and registering one would be theatre** -- no
#         measurement of this field at a tuned point exists to predict from.
#         Report, per rung and per seed: pbar, bias, a_raw, a_deb, bias_share,
#         from neff_instrument.agreement_stats, the same reducer every published
#         number in this campaign came through.
#         **THE "53.1 % OF 11.17M META-GRADIENTS AGREE IN SIGN" SENTENCE IS NOT
#         BEING REPRODUCED HERE AND MUST NOT BE WRITTEN.**  It has been refuted
#         as stated since cycle 42 (CORRECTIONS 26; KILLTEST-idea2 section 1):
#         two arms merged, an n-dependent null the quoted arm sits 1.9 pp BELOW,
#         and an alpha0=1e-6 scope whose excess collapses 16x at alpha0=1e-3.
#         What survives is the drift-vs-N slope and the block-size curve.
#
#   T4    N_eff/m AT A TUNED POINT, all three rungs, BOX-FREE CELLS ONLY, priced
#         in its own SE.  A bound rung ANYWHERE in the ranked set makes the
#         argmin UNINTERPRETABLE (CORRECTIONS 79) -- that rule is not relaxed
#         here just because this cell is better tuned than the ones before it.
#         Register the box `tw0` -> (-15.0, -2.3026) in c55_neff_noise.BOXES,
#         with THIS script's path:line, BEFORE scoring, or the root is SKIPPED.
#
# WHAT THIS BATCH DELIBERATELY DOES NOT DO
#   * It does NOT extend the wm9 ms grid.  W1 is UNDECIDED because the curve is
#     FLAT (3e-5..1e-3 spans 0.074 pp), not because the grid is too coarse.
#     Refining a flat curve buys nothing.  W1b already certified the argmax as
#     interior, which is what licensed W2 being scored at all.
#   * It does NOT chase the 80-epoch N_eff point.  See CORRECTIONS 103.4: three
#     batches (bd7, bf8, bf9 = 36 GPU-jobs) have now failed to produce it, and
#     the two available ceilings fail in OPPOSITE directions (HI=+2.0 pins the
#     fine rungs, HI=+6.0 collapsed weightwise 2/2 to 10.000).  That axis is
#     closed by decision, not by another batch.
#
# STRUCTURAL CHECKS BEFORE USING ANY NUMBER (Rule 4)
#   1. python3 analysis/c74_bf9_score.py  --selftest     (47/47)
#   2. python3 analysis/c74_wm9_score.py  --selftest     (49/49)
#   3. python3 analysis/c55_neff_noise.py --selftest     (51/51)
#   4. python3 analysis/c52_boxfree.py    --selftest     (27/27)
#   5. Register tw0 in c55_neff_noise.BOXES BEFORE scoring (see T4).
#   6. plateau5 is PRIMARY.  best_test inflates ~0.37 pp, the size of this effect.
#
# USAGE
#   bash c74_tuned_weightwise_probe.sh            # dry run: prints every sbatch line
#   bash c74_tuned_weightwise_probe.sh --submit   # submits, ONLY if the guards pass
# =============================================================================
set -u

SUBMIT=0
[ "${1:-}" = "--submit" ] && SUBMIT=1

WS=/cluster/account_a/metaopt
RUNNER=$WS/jobs/run_cifar.sh
SAVE=$WS/runs/tw0
PARTS=gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g
USER_NAME=$(whoami)

NJOBS=9
CLIP=-15:-2.3026               # wm9's box EXACTLY -- not a new box
MST=1e-4                       # wm9's argmax, AND layerwise's, AND scalar's
EPOCHS=100
WALL=03:50:00
WRITE_EVERY=500                # as bd7 and bf9 used
EXP_RECORDS=10000              # 100 ep x 500 steps / stride 5

# The repo path is passed IN, never derived from __file__ inside a heredoc:
# heredoc-fed python has __file__ == '<stdin>' and resolves against CWD, which
# silently read a stale Aug-20 CSV once already (CORRECTIONS 101.11).
REPO="$(cd "$(dirname "$0")/.." && pwd)"
CSVP="$REPO/results/all_runs.csv"

echo "=============================================================="
echo "  tw0 -- the tuned weightwise operating point, WITH the probe"
echo "    ms=$MST fixed, BETA_CLIP=$CLIP fixed, ${EPOCHS}ep"
echo "    rungs weightwise/nodewise/layerwise x seeds 0,1,2 = $NJOBS jobs"
echo "    T1 is PRIMARY: is weightwise box-free at BOTH guards at 100 ep?"
echo "    T1 REGISTERS: box-free on >= 2 of 3 seeds."
echo "    A REFUTE WITHDRAWS THIS TICK'S OWN 'ladder is not flat' HEADLINE."
echo "=============================================================="

# ---------------------------------------------------------------------------
# GUARD 0 -- the things that must simply exist.
# ---------------------------------------------------------------------------
[ -f "$CSVP" ] || { echo "GUARD FAIL: no CSV at $CSVP"; exit 1; }
[ -x "$RUNNER" ] || [ -f "$RUNNER" ] || { echo "GUARD FAIL: no runner at $RUNNER"; exit 1; }
echo "guard 0: runner and CSV present ($(wc -l < "$CSVP") CSV lines)"

# ---------------------------------------------------------------------------
# GUARD 1 -- THE PATCH IS PRESENT **AND ENABLED**.  STANDING RULE (19).
# bf8 burned 12 GPU-jobs because its guard checked the patch was PRESENT in
# HF.py.  It was present.  It was OFF: the script exported PROBE=5 (the STRIDE)
# and never PROBE5=1 (the INSTRUMENT).  This guard reads the gate out of the
# patched source, then asserts THIS SCRIPT'S OWN --export line sets all three
# variables, then diffs against bf9's export line -- the batch that is CONFIRMED
# to have fired (12/12 neg_counts.json, n_tot correct, npy shape correct).
# A self-reading guard MUST exclude its own source lines or it matches itself.
# ---------------------------------------------------------------------------
python3 - "$WS" "$REPO" "$0" <<'PYEOF' || exit 1
import os, re, sys
ws, repo, me = sys.argv[1], sys.argv[2], sys.argv[3]

# The optimizer the runner actually imports.  `jobs/run_cifar.sh` cds into
# .../cifar10 and runs train.py, which does `from Optimizers.build_optimizer
# import build_optimizer`, so THIS is the file that carries the gate.  The first
# draft of this guard looked for $WS/HF.py, which does not exist on either
# account -- it FAILED CLOSED and refused to submit, which is the behaviour a
# guard that names a path must have.  Do not relax it to a glob.
hf = os.path.join(ws, "MetaOptimize", "codes", "Supervised_tasks", "MetaOptimize",
                  "cifar10", "Optimizers", "HF.py")
if not os.path.exists(hf):
    sys.exit("GUARD FAIL: %s absent -- the patched optimizer is not where the "
             "runner imports it from" % hf)
src = open(hf).read()
if "PROBE5" not in src:
    sys.exit("GUARD FAIL: HF.py carries no PROBE5 gate -- patch_probe5.py is not applied")
m = re.search(r"os\.environ\.get\(\s*['\"]PROBE5['\"].*?\)\s*==\s*['\"]1['\"]", src)
if not m:
    sys.exit("GUARD FAIL: HF.py's PROBE5 gate is not the ==\"1\" form this script "
             "exports for.  Read the patch before assuming the variable name.")
print("guard 1a: HF.py gates the sign-count writer on PROBE5 == '1'")

# --- this script's own export line (EXCLUDING the guard's own text)
mine = [l for l in open(me).read().splitlines()
        if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
if len(mine) != 1:
    sys.exit("GUARD FAIL: expected exactly 1 --export line with PROBE_DIR, found %d" % len(mine))
line = mine[0]
for need in ("PROBE=5", "PROBE5=1", "PROBE5_WRITE_EVERY="):
    if need not in line:
        sys.exit("GUARD FAIL: this script's --export line is missing %r.  That is "
                 "EXACTLY bf8's bug: the stride without the instrument." % need)
print("guard 1b: this script exports PROBE=5 AND PROBE5=1 AND PROBE5_WRITE_EVERY")

# --- diff against bf9, the batch CONFIRMED to have fired
bf9 = os.path.join(repo, "bin", "c73_bf9_probe5.sh")
if not os.path.exists(bf9):
    sys.exit("GUARD FAIL: bf9's script is absent, so there is nothing to diff against")
ref = [l for l in open(bf9).read().splitlines()
       if l.strip().startswith('--export="ALL,') and "PROBE_DIR" in l]
if len(ref) != 1:
    sys.exit("GUARD FAIL: could not isolate bf9's export line")
def keys(l):
    return sorted(k.split("=")[0] for k in re.search(r'"ALL,([^"]+)"', l).group(1).split(","))
if keys(line) != keys(ref[0]):
    sys.exit("GUARD FAIL: export KEYS differ from bf9's confirmed-firing line:\n"
             "  tw0: %s\n  bf9: %s" % (keys(line), keys(ref[0])))
print("guard 1c: export keys identical to bf9's CONFIRMED-FIRING line  %s" % keys(line))
PYEOF

# ---------------------------------------------------------------------------
# GUARD 2 -- THE 13-AXIS SIGNATURE.  This batch must be comparable to `wm9`,
# because T2 reproduces wm9's accuracy and T1 makes a claim ABOUT wm9.  The
# signature is re-derived FROM THE CSV, never from a comment (STANDING RULE 1, 15).
# ---------------------------------------------------------------------------
python3 - "$CSVP" "$CLIP" "$EPOCHS" "$MST" <<'PYEOF' || exit 1
import csv, sys, statistics
csvp, clip, epochs, mst = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
rd = list(csv.DictReader(open(csvp)))
AX = ['network','dataset','batch_size','base','meta','alpha0','gamma',
      'augment','beta_clip','hier','lam','eta_ratio','epochs_done']
ref = [r for r in rd if r['run'].startswith('rs-lay-1e4')]
if not ref:
    sys.exit("GUARD FAIL: no rs-lay-1e4 reference row -- cannot derive the signature")
sig = {a: ref[0][a] for a in AX}
mine = {'network':'ResNet18','dataset':'CIFAR10','batch_size':'100','base':'SGDm',
        'meta':'Lion','alpha0':'1e-3','gamma':'1','augment':'1',
        'beta_clip':clip,'epochs_done':epochs}
bad = [(a, mine[a], sig[a]) for a in mine if mine[a] != sig[a]]
if bad:
    for a,m,s in bad: print("  axis %s: this batch %r vs reference %r" % (a,m,s))
    sys.exit("GUARD FAIL: tw0 does not match the ladder's axis signature")
for a in ('hier','lam','eta_ratio'):
    if sig[a] not in ('','na'):
        sys.exit("GUARD FAIL: reference has %s=%r; this batch sets no hierarchical "
                 "operator, so it is not the same arm" % (a, sig[a]))
print("guard 2: all 13 axes match the rs-lay-1e4 signature read FROM THE CSV")

# --- T2's reference must exist, and ms must really be wm9's argmax
w = [r for r in rd if r['run'].startswith('wm9-w-') and r.get('plateau5')]
if len(w) != 10:
    sys.exit("GUARD FAIL: expected 10 wm9 rows for the reference, found %d" % len(w))
cells = {}
for r in w:
    cells.setdefault(r['meta_stepsize'], []).append(float(r['plateau5']))
best = max(cells, key=lambda k: statistics.mean(cells[k]))
if float(best) != float(mst):
    sys.exit("GUARD FAIL: wm9's argmax is ms=%s, but this batch runs ms=%s -- it "
             "would not be the tuned operating point" % (best, mst))
ctrl = statistics.mean(cells[best])
print("guard 2b: wm9 argmax ms=%s, T2 reference plateau5 = %.3f (n=%d)"
      % (best, ctrl, len(cells[best])))
if not (90.0 < ctrl < 92.2):
    sys.exit("GUARD FAIL: the T2 reference %.3f is not where cycle 74 recorded it" % ctrl)

# --- and ms=1e-4 must be the argmax for layerwise and scalar too, or this is
#     not a cell where every rung is at its own best.
sigt = tuple(ref[0][a] for a in AX)
for g in ('layerwise','scalar'):
    cel = {}
    for r in rd:
        if (tuple(r[a] for a in AX) == sigt and r['granularity'] == g
                and r['collapsed'] in ('False','false','0','') and r.get('plateau5')):
            cel.setdefault(r['meta_stepsize'], []).append(float(r['plateau5']))
    if not cel:
        sys.exit("GUARD FAIL: no matched %s cells" % g)
    b = max(cel, key=lambda k: statistics.mean(cel[k]))
    if float(b) != float(mst):
        sys.exit("GUARD FAIL: %s's matched argmax is ms=%s, not %s -- this is not a "
                 "commonly-tuned operating point and the batch's premise is wrong"
                 % (g, b, mst))
    print("          %-10s matched argmax ms=%s  plateau5 %.3f (n=%d)"
          % (g, b, statistics.mean(cel[b]), len(cel[b])))
PYEOF

# ---------------------------------------------------------------------------
# GUARD 3 -- THE QUESTION IS ACTUALLY OPEN.  Every job must name an open
# question, and a batch already answered by the CSV must not run.  There must be
# NO existing matched weightwise run at this ms and budget carrying a probe.
# ---------------------------------------------------------------------------
python3 - "$CSVP" "$MST" "$EPOCHS" "$REPO" <<'PYEOF' || exit 1
import csv, os, sys, glob
csvp, mst, epochs, repo = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
rd = list(csv.DictReader(open(csvp)))
hit = [r['run'] for r in rd if r['granularity']=='weightwise'
       and r['epochs_done']==epochs and r['beta_clip']=='-15:-2.3026'
       and r['meta_stepsize'] and float(r['meta_stepsize'])==float(mst)]
print("guard 3: matched 100-ep weightwise runs at ms=%s in this box: %d  %s"
      % (mst, len(hit), hit))
if len(hit) > 2:
    sys.exit("GUARD FAIL: %d such runs already exist -- re-read the census before "
             "spending 9 more jobs" % len(hit))
# and none of them may already carry a probe dir
roots = glob.glob(os.path.join(repo, "..", "probes_*"))
have = [os.path.basename(r) for r in roots if os.path.basename(r) in ("probes_tw0",)]
if have:
    sys.exit("GUARD FAIL: %s already exists -- this batch has already been run" % have)
print("         no probes_tw0 root exists, so no probe data answers this yet")

# --- the ml5 prior this batch is built on must actually be what cycle 74 read.
ml5 = os.path.join(repo, "..", "probes_ml5")
if os.path.isdir(ml5):
    sys.path.insert(0, os.path.join(repo, "analysis"))
    from c52_boxfree import occupancy
    ds = sorted(glob.glob(os.path.join(ml5, "probe_w_m4_s*")))
    if len(ds) != 3:
        sys.exit("GUARD FAIL: expected 3 ml5 w_m4 dirs (the prior), found %d" % len(ds))
    bad = []
    for d in ds:
        o = occupancy(d, -15.0, -2.3026)
        if not (o['rec_lo'] < 0.05 and o['rec_hi'] < 0.05):
            bad.append((os.path.basename(d), o['rec_lo'], o['rec_hi']))
    if bad:
        sys.exit("GUARD FAIL: the ml5 prior does NOT hold -- %s.  This batch's whole "
                 "premise (slack box at ms=1e-4, 20 ep) is wrong." % bad)
    print("         ml5 prior VERIFIED: weightwise m4 box-free 3/3 at BOTH guards, 20 ep")
else:
    print("         (probes_ml5 not local -- the prior is cited, not re-verified here)")
PYEOF

# ---------------------------------------------------------------------------
# GUARD 4 -- wallclock, sized from MEASURED data, including the probe overhead.
# ---------------------------------------------------------------------------
python3 - "$WALL" "$CSVP" "$EPOCHS" <<'PYEOF' || exit 1
import csv, sys
def secs(t):
    p=[int(x) for x in t.split(":")]
    while len(p)<3: p.insert(0,0)
    return p[0]*3600+p[1]*60+p[2]
wall_min = secs(sys.argv[1])/60.0
rd = list(csv.DictReader(open(sys.argv[2])))
# the probed 80-ep weightwise runs are the honest reference: they carry the probe
probed = [float(r['wallclock_min']) for r in rd
          if r['run'].startswith('bf9-w-') and r.get('wallclock_min')]
plain = [float(r['wallclock_min']) for r in rd
         if r['granularity']=='weightwise' and r['epochs_done']=='100'
         and r.get('wallclock_min')]
if not probed or not plain:
    sys.exit("GUARD FAIL: no wallclock reference to size against")
proj = max(probed) * (int(sys.argv[3])/80.0)     # 80 ep -> 100 ep
need = max(proj, max(plain))
if need > wall_min:
    sys.exit("GUARD FAIL: projected %.0f min but --time is %.0f min -- would truncate"
             % (need, wall_min))
print("guard 4: probed 80-ep weightwise slowest %.0f min -> %.0f min at 100 ep; "
      "plain 100-ep slowest %.0f min; wall %.0f min (%.1fx headroom)"
      % (max(probed), proj, max(plain), wall_min, wall_min/need))
PYEOF

# ---------------------------------------------------------------------------
# GUARD 5 -- disk.  A weightwise probe dir is ~90 MB at 8000 records; this batch
# writes 10000 records on 3 weightwise dirs plus 6 small ones.  Want 20x room.
# ---------------------------------------------------------------------------
python3 - "$WS" <<'PYEOF' || exit 1
import os, shutil, sys
ws = sys.argv[1]
free_gb = shutil.disk_usage(ws).free / 1e9
need_gb = 3 * 0.115 + 6 * 0.01     # 3 weightwise @115MB + 6 small
if free_gb < 20 * need_gb:
    sys.exit("GUARD FAIL: %.1f GB free vs ~%.2f GB needed and this guard wants 20x"
             % (free_gb, need_gb))
print("guard 5: %.0f GB free vs ~%.2f GB needed  OK" % (free_gb, need_gb))
PYEOF

# ---------------------------------------------------------------------------
# GUARD 6 -- queue depth.  9 jobs is inside the ~20-job / 0.5%-RawUsage rule.
# ---------------------------------------------------------------------------
FS=$(sshare -U -u "$USER_NAME" -n -o FairShare 2>/dev/null | tr -d ' ' | head -1)
PEND=$(squeue -h -u "$USER_NAME" -t PENDING 2>/dev/null | wc -l | tr -d ' ')
echo "guard 6: FairShare=${FS:-unknown} (informational), pending=$PEND, adding $NJOBS"
if [ "$PEND" -gt 40 ]; then
  echo "GUARD FAIL: $PEND already pending, over the 40 cap"; exit 1
fi

# ---------------------------------------------------------------------------
mkdir -p "$SAVE"
N=0
FAIL=0
SGDM=(--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1
      --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0)
for G in weightwise nodewise layerwise; do
  if   [ "$G" = weightwise ]; then SHORT=w
  elif [ "$G" = nodewise ];   then SHORT=node
  else                             SHORT=lay
  fi
  for S in 0 1 2; do
    RN="tw0-${SHORT}-s${S}"
    CMD=(sbatch --job-name="$RN" --partition="$PARTS" --gres=gpu:1 --cpus-per-task=6
      --mem=14G --time="$WALL" --nice=0
      --export="ALL,AUGMENT=1,BETA_CLIP=${CLIP},PROBE=5,PROBE5=1,PROBE5_WRITE_EVERY=${WRITE_EVERY},PROBE_DIR=$SAVE/probe_${SHORT}_tw0_s${S}"
      "$RUNNER" "${SGDM[@]}"
      --dataset CIFAR10 --NN-name ResNet18 --batch-size 100
      --max-time 999:00:00 --gamma 1 --meta-stepsize "$MST" --alpha0 1e-3
      --num-epochs "$EPOCHS" --stepsize-groups "$G" --seed "$S"
      --save-directory "$SAVE" --run-name "$RN")
    # COUNT WHAT SLURM ACCEPTED, NOT WHAT WE TRIED (CORRECTIONS 71).
    if [ "$SUBMIT" = "1" ]; then
