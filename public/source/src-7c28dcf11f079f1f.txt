set -u
cd /cluster/account_a/metaopt
R=/cluster/account_a/metaopt/runs
P="gpu-short,gpu-l4-24g,gpu-2080ti-11g,gpu-mig-40g,gpu-a100-80g"
C="BETA_CLIP=-15:-2.3026"
SGDM="--optimizer HF --alg-base SGDm --momentum-param-base 0.99 --weight-decay-base 0.1 --alg-meta Lion --momentum-param-meta 0.99 --Lion-beta2-meta 0.9 --weight-decay-meta 0"
mkdir -p $R/p7free

# `p9-*` (written with the p7- prefix so bin/agree2.py picks them up unchanged).
#
# OPEN QUESTION.  Cycle 31 resolved 29.2 at n=10 for the m=6 rung only.  The rest of the
# R18 agreement ladder is n=3 (CIFAR-10) and n=1-2 (CIFAR-100), and CORRECTIONS Rule 3 says
# report the curve, not one cell.  Two claims currently rest on n<=2 cells:
#   (a) CIFAR-100's ladder is NON-monotone (peaks at m=62, not m=6)  -- C100 lay is n=2
#   (b) the dataset contrast INVERTS sign between m=6 and m=62       -- C100 lay/node n=2
# and CIFAR-100 has no drift-vs-log(m) slope of its own at usable n, so the campaign's
# headline sqrt(N) refutation is a ONE-DATASET result.
#
# This batch takes the ENTIRE ResNet18 ladder to n=10 on BOTH datasets, 4 granularities.
# 20-epoch probes, ~15 min each: the cheapest decisive jobs on the board.
prb(){ sbatch --parsable --job-name=$1 --partition=$P --gres=gpu:1 --cpus-per-task=6 --mem=14G --time=01:30:00 \
  --nice=0 \
  --export="ALL,AUGMENT=1,$C,PROBE=100,PROBE_DIR=$R/p7free/$1" jobs/run_cifar.sh $SGDM \
  --dataset $4 --NN-name $3 --batch-size 100 --max-time 999:00:00 --gamma 1 \
  --meta-stepsize 1e-3 --alpha0 1e-3 --num-epochs 20 --stepsize-groups $2 --seed $5 \
  --save-directory $R/p7free --run-name $1 >/dev/null; }

# CIFAR-10 / ResNet18 : s0-s2 exist for lay/node/w -> add s3..s9
for s in 3 4 5 6 7 8 9; do
  prb p7-r18-lay-s$s  layerwise  ResNet18 CIFAR10 $s
  prb p7-r18-node-s$s nodewise   ResNet18 CIFAR10 $s
  prb p7-r18-w-s$s    weightwise ResNet18 CIFAR10 $s
done

# CIFAR-100 / ResNet18_c100 : lay/node have s1,s2 -> add s0,s3..s9 ; w has s1 -> add s0,s2..s9
for s in 0 3 4 5 6 7 8 9; do
  prb p7-c100-lay-s$s  layerwise  ResNet18_c100 CIFAR100 $s
  prb p7-c100-node-s$s nodewise   ResNet18_c100 CIFAR100 $s
done
for s in 0 2 3 4 5 6 7 8 9; do
  prb p7-c100-w-s$s    weightwise ResNet18_c100 CIFAR100 $s
done

echo "P7_QUEUED=$(squeue -h -u account_a -o '%j' | grep -c '^p7-')"
echo "TOTAL_cluster_a=$(squeue -h -u account_a | wc -l)"
