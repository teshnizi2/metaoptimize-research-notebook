# analysis/cST2_carriervote_score.py (sha 8300f757...) prints on the landed `cst1` batch.  It imports

def build_model(n_classes):
    out = [("conv1.weight", 64 * 3 * 3 * 3), ("bn1.weight", 64), ("bn1.bias", 64)]
    inp = 64
    for stage, planes in enumerate((64, 128, 256, 512), start=1):
        for blk in range(2):
            pre = "layer%d.%d." % (stage, blk)
            out.append((pre + "conv1.weight", planes * inp * 3 * 3))
            out.append((pre + "bn1.weight", planes))
            out.append((pre + "bn1.bias", planes))
            out.append((pre + "conv2.weight", planes * planes * 3 * 3))
            out.append((pre + "bn2.weight", planes))
            out.append((pre + "bn2.bias", planes))
            if inp != planes:
                out.append((pre + "shortcut.0.weight", planes * inp))
                out.append((pre + "shortcut.1.weight", planes))
                out.append((pre + "shortcut.1.bias", planes))
            inp = planes
    out.append(("linear.weight", 512 * n_classes))
    out.append(("linear.bias", n_classes))
    return out

def tclass(name):
    if name.startswith("linear."):
        return "linear"
    if name.endswith(".bias"):
        return "bn_bias"
    if ".bn" in name or ".shortcut.1." in name or name.startswith("bn1."):
        return "bn_scale"
    return "conv"

def parse_sets(spec):
    if spec == "scalar":
        return [list(range(1, len(NAMES) + 1))]
    if spec == "layerwise":
        return [[i] for i in range(1, len(NAMES) + 1)]
    assert spec.startswith("sets:")
    groups = []
    for g in spec[5:].split("/"):
        idx = []
        for tok in g.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(NAMES.index(tok) + 1)
        groups.append(sorted(idx))
    return groups

def sgn(x):
    return 1 if x > 0 else (-1 if x < 0 else 0)


def fmt(x, nd=4):
    return "n/a" if x is None else ("%.*f" % (nd, x))


def med(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return None
    return v[n // 2] if n % 2 else (v[n // 2 - 1] + v[n // 2]) / 2.0


def avg(v):
    v = list(v)
    return sum(v) / len(v) if v else None


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def f32(x):
    return struct.unpack("f", struct.pack("f", x))[0]


def ulp32_bits(x):
    """The float32 spacing at |x|, derived by INCREMENTING THE BIT PATTERN -- a different route from the
    scorer's math.frexp implementation, so agreement between them is evidence, not a shared assumption."""
    v = abs(float(x))
    b = struct.unpack("<I", struct.pack("<f", v))[0]
    lo = struct.unpack("<f", struct.pack("<I", b))[0]
    hi = struct.unpack("<f", struct.pack("<I", b + 1))[0]
    return hi - lo


def ulp32_frexp(x):
    v = abs(float(x))
    if v == 0.0 or not math.isfinite(v):
        return 2.0 ** -149
    _m, e = math.frexp(v)
    return math.ldexp(1.0, e - 24)


def sign_tol(bpre, bpost, ms):
    return SIGN_ABS_BAR + SIGN_ULP_K * ulp32_bits(max(abs(bpre), abs(bpost))) / ms

def find_out(runsdir, arm, seed):
    pat = re.compile(r"^%s-%s-s%d-(\d+)\.out$" % (BATCH, arm, seed))
    hits = []
    for d in (runsdir, os.path.join(runsdir, BATCH)):
        if os.path.isdir(d):
            for fn in sorted(os.listdir(d)):
                m = pat.match(fn)
                if m:
                    hits.append((int(m.group(1)), os.path.join(d, fn)))
    return sorted(hits)


def read_out(path):
    txt = open(path, errors="replace").read()
    lines = txt.splitlines()
    te, tr = {}, {}
    for e, a, b in EP_RE.findall(txt):
        tr[int(e)] = float(a)
        te[int(e)] = float(b)
    r = {"path": path, "test": te, "train": tr,
         "args": [l for l in lines if l.startswith("ARGS:")],
         "env": [l for l in lines if l.startswith("ENV:")],
         "pt": [l for l in lines if l.startswith("PROBE_TENSOR:")],
         "foreign": [l for l in lines if l.startswith(FOREIGN_PREFIXES)],
         "done": any(l.strip() == "RUN_DONE" for l in lines),
         "tb": "Traceback (most recent call last)" in txt}
    tail = [te.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    trl = [tr.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    r["p5"] = sum(tail) / 5.0 if all(isinstance(x, float) and math.isfinite(x) for x in tail) else None
    r["t5"] = sum(trl) / 5.0 if all(isinstance(x, float) and math.isfinite(x) for x in trl) else None
    r["complete"] = sorted(te) == list(range(EPOCHS)) and r["done"] and not r["tb"] and r["p5"] is not None
    return r


def read_probe(runsdir, arm, seed):
    d = os.path.join(runsdir, BATCH, "probe_%s-%s-s%d" % (BATCH, arm, seed))
    recs, bad = [], 0
    for ln in open(os.path.join(d, "probe.jsonl"), errors="replace"):
        ln = ln.strip()
        if not ln:
            continue
        try:
            r = json.loads(ln)
        except ValueError:
            bad += 1
            continue
        if isinstance(r, dict):
            recs.append(r)
    return {"dir": d, "recs": recs, "bad": bad,
            "hdr": json.load(open(os.path.join(d, "probe_tensor.json")))}

def rec_L(r):
    b2 = float(r.get("pt_b2", B2))
    return [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])], b2


def shared_s(r, L, b2):
    za, ma = r["z_agg"][0], r["mom_pre"][0]
    if len(za) == 1 and len(ma) == 1:
        return sgn(b2 * ma[0] + (1.0 - b2) * za[0])
    return sgn(sum(L))


def fixes(L, idx, s):
    tot = sum(L[i] for i in idx)
    rest = sum(abs(L[j]) for j in range(len(L)) if j not in idx)
    return sgn(tot) == s and abs(tot) > rest


def carrying(L, s):
    order = sorted(range(len(L)), key=lambda i: -abs(L[i]))
    rem = sum(abs(x) for x in L)
    part = 0.0
    for k, i in enumerate(order, 1):
        part += L[i]
        rem -= abs(L[i])
        if sgn(part) == s and abs(part) > rem:
            return order[:k]
    return order


def vote_stats(recs):
    cs = set(C_IDX0)
    n = indet = dom = t3 = dt3 = det = dis = careq = down = pinned = dompin = 0
    shares, cls = [], {c: [] for c in CLASSES}
    modal, argmax = Counter(), Counter()
    dec = [[0, 0] for _ in range(10)]
    for r in recs:
        L, b2 = rec_L(r)
        s = shared_s(r, L, b2)
        if s == 0:
            indet += 1
            continue
        n += 1
        order = sorted(range(len(L)), key=lambda i: -abs(L[i]))
        top3 = order[:3]
        d = fixes(L, C_IDX0, s)
        dom += d
        t3 += set(top3) == cs
        dt3 += fixes(L, top3, s)
        modal[tuple(sorted(i + 1 for i in top3))] += 1
        argmax[order[0] + 1] += 1
        tot = sum(abs(x) for x in L)
        shares.append(sum(abs(L[i]) for i in C_IDX0) / tot if tot > 0 else 0.0)
        if tot > 0:
            for c in CLASSES:
                cls[c].append(sum(abs(L[i]) for i in range(len(L)) if CLASS_OF[i] == c) / tot)
        st = r.get("step")
        if isinstance(st, int) and 0 <= st < META_STEPS:
            dd = dec[st * 10 // META_STEPS]
            dd[0] += 1
            dd[1] += d
        if len(r.get("beta", [])) == 1 and (r.get("n_at_lo") or 0) >= 1:
            pinned += 1
            dompin += d
        maj = sgn(sum(1 for x in L if x > 0) - sum(1 for x in L if x < 0))
        if maj:
            det += 1
            if maj != s:
                dis += 1
                careq += set(carrying(L, s)) == cs
                down += s > 0
    return {"n": n, "indet": indet, "DOM_C": dom / n if n else None, "TOP3_C": t3 / n if n else None,
            "DOM_TOP3": dt3 / n if n else None, "SHARE_C": med(shares) if shares else None,
            "R_T": dis / det if det else None, "n_dis": dis, "CAR_EQ": careq / dis if dis else None,
            "DOWN": down / dis if dis else None, "PINNED_FRAC": pinned / n if n else None,
            "DOM_C_PINNED": dompin / pinned if pinned else None,

            "DECILE": [(d[1] / d[0]) if d[0] else None for d in dec]}

def decomposition(recs, groups1, ms):
    """Both bars at once: `bad_sign` is the SUCCESSOR's ms-aware count, `old_bad` the predecessor's absolute one."""
    g0 = [[i - 1 for i in g] for g in groups1]
    n = bshape = bz = bm = nsign = bsign = obsign = clamped = bshared = exact32 = 0
    wz = wm = worst_q = tol_max = 0.0
    msf = f32(ms)
    for r in recs:
        n += 1
        z, m = r["z_tensor"], r["m_tensor"]
        za, ma = r["z_agg"][0], r["mom_pre"][0]
        bpre, bpost = r["beta_pre"][0], r["beta"]
        b2 = float(r.get("pt_b2", B2))
        if not (len(z) == len(m) == 62 and len(za) == len(ma) == len(bpre) == len(bpost) == len(g0)):
            bshape += 1
            continue
        sz = sum(abs(v) for v in z) + 1e-30
        sm = sum(abs(v) for v in m) + 1e-30
        dz = max(abs(sum(z[i] for i in g) - za[k]) for k, g in enumerate(g0)) / sz
        dm = max(abs(sum(m[i] for i in g) - ma[k]) for k, g in enumerate(g0)) / sm
        wz, wm = max(wz, dz), max(wm, dm)
        bz += dz > TOL_Z
        bm += dm > TOL_M
        if len(g0) == 1:
            L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
            Lh = b2 * ma[0] + (1 - b2) * za[0]
            if abs(sum(L) - Lh) > 1e-4 * (sum(abs(x) for x in L) + 1e-30) and sgn(sum(L)) != sgn(Lh):
                bshared += 1
        for k in range(len(g0)):
            if not (BETA_LO + 1e-9 < bpost[k] < BETA_HI - 1e-9):
                clamped += 1
                continue
            nsign += 1
            s = sgn(b2 * ma[k] + (1 - b2) * za[k])
            q = -(bpost[k] - bpre[k]) / ms
            resid = abs(q - s)
            tol = sign_tol(bpre[k], bpost[k], ms)
            worst_q = max(worst_q, resid)
            tol_max = max(tol_max, tol)
            bsign += resid > tol
            obsign += resid > PRED_SIGN_BAR
            exact32 += (f32(f32(bpre[k]) - f32(msf * s)) == bpost[k])
    return {"n": n, "bad_shape": bshape, "bad_z": bz, "bad_m": bm, "worst_z": wz, "worst_m": wm,
            "n_sign": nsign, "bad_sign": bsign, "old_bad": obsign, "clamped": clamped, "bad_shared": bshared,
            "exact32": exact32, "worst_q": worst_q, "tol_max": tol_max}


def decomp_ok(d):
    n = max(d["n"], 1)
    return (d["n"] > 0 and d["bad_shape"] == 0 and d["bad_z"] / n <= DECOMP_FAIL_MAX
            and d["bad_m"] / n <= DECOMP_FAIL_MAX and d["n_sign"] > 0
            and d["bad_sign"] / max(d["n_sign"], 1) <= SIGN_FAIL_MAX and d["bad_shared"] / n <= DECOMP_FAIL_MAX)

CELLKEYS = ["network", "dataset", "granularity", "base", "meta", "meta_stepsize", "alpha0", "gamma", "augment",
            "beta_clip", "batch_size", "epochs_requested", "hier", "lam", "eta_ratio"]


def load_corpus(csv_path, tsv_path):
    rows = list(csv.DictReader(open(csv_path)))
    lines = [l.rstrip("\n") for l in open(tsv_path) if l.strip() and not l.startswith("#")]
    head = lines[0].split("\t")
    excl = set()
    for ln in lines[1:]:
        f = dict(zip(head, ln.split("\t")))
        excl.add((f["run"], f["job_id"]))
    return rows, [r for r in rows if (r.get("run"), r.get("job_id")) not in excl], excl


def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
            and (r.get("plateau5") or "").strip() != "")


def std_cell(r, net, dset, ms, a0):
    return (usable(r) and r.get("network") == net and r.get("dataset") == dset
            and r.get("epochs_requested") == "100" and r.get("augment") == "1"
            and r.get("beta_clip") == "-15:-2.3026" and r.get("meta_stepsize") == ms
            and r.get("alpha0") == a0 and r.get("batch_size") == "100")


def cell_rows(rows, net, dset, ms, a0, gran):
    return [r for r in rows if std_cell(r, net, dset, ms, a0) and r.get("granularity") == gran
            and r.get("hier", "") == "" and r.get("base") == "SGDm" and r.get("meta") == "Lion"
            and r.get("gamma") == "1"]


def pooled_sigma(rows, net, dset, ms="1e-3", a0="1e-6"):
    cells = {}
    for r in rows:
        if not std_cell(r, net, dset, ms, a0):
            continue
        cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = sum(v) / len(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
    return (math.sqrt(ss / df) if df else None), df, nc

      % ("DOM_C", nst["DOM_C"], int(round(nst["DOM_C"] * n_rec)), int(round(DOM_LO * n_rec)),
         int(round(nst["DOM_C"] * n_rec)) - int(round(DOM_LO * n_rec))))
    P("  D_ISO %+.4f pp clears the 20.0 bar by %.4f pp (%.2f SE); D_CTL %+.4f pp is %.4f pp inside the 5.0 bar "
      "(%.2f SE); k01 %.4f is %.4f pp under K01_MAX 34.0"
      % (d_iso, d_iso - RESCUE_BAR, (d_iso - RESCUE_BAR) / SE_PRIOR, d_ctl, NULL_BAR - d_ctl,
         (NULL_BAR - d_ctl) / SE_PRIOR, lev["k01"], K01_MAX - lev["k01"]))
    chk(nst["TOP3_C"] < TOP3_BAR and nst["DOM_C"] < DOM_HI and nst["DOM_C"] > DOM_LO,
        "both nomination bars are missed and the NOT-NOMINATED bar is cleared -> NOMINATION-PARTIAL")

    P("\n[CORPUS] sigma and the anchors, re-derived through the exclusions TSV at commit %s" % CORPUS_COMMIT)
    raw, kept, excl = load_corpus(a.csv, a.tsv)
    P("  rows %d -> %d after %d exclusion keys" % (len(raw), len(kept), len(excl)))
    chk(len(raw) == REG_CORPUS_ROWS, "the registration corpus has %d rows" % REG_CORPUS_ROWS)
    sig, df, ncell = pooled_sigma(kept, NET, DSET)
    P("  SIGMA_R18ALL filtered %r (df %d, %d cells)" % (sig, df, ncell))
    chk(abs(sig - SIGMA_PRIOR) < 1e-12 and df == 221 and ncell == 46,
        "SIGMA_R18ALL == SIGMA_PRIOR (df 221, 46 cells)")
    nsig, ndf, nnc = pooled_sigma(raw, NET, DSET)
    P("  the same WITHOUT the exclusions (the OPERATIONS 36 trap): %.6f (df %d, %d cells)" % (nsig, ndf, nnc))
    chk(nsig > sig, "the naive sigma is larger than the filtered one")
    anch = {"k01@3e-4 (cru1 n3)": cell_rows(kept, NET, DSET, "3e-4", A0, "scalar"),
            "layerwise@3e-4 (cru1 n3)": cell_rows(kept, NET, DSET, "3e-4", A0, "layerwise"),
            "k01@1e-3 (n44)": cell_rows(kept, NET, DSET, "1e-3", A0, "scalar")}
    for k, rows in anch.items():
        m = avg([float(r["plateau5"]) for r in rows])
        want, wn = EXPECT_ANCHOR[k]
        P("  anchor %-27s n=%d mean %.4f" % (k, len(rows), m))
        chk(len(rows) == wn and abs(round(m, 4) - want) < 5e-5, "anchor %s == the registered literal" % k)
    chk(not any(r["run"].startswith(BATCH + "-") for r in raw),
        "no cst1- row in the REGISTRATION corpus (the pre-launch freshness 256.2 recorded)")

    # ---- the REPLICA: the successor's own output, rebuilt from scratch ----------------------------
    R("[G-DECOMP] per-group sums of the capture vs the harness; applied step vs sign(L)")
    R("  tolerance (THE ONE CHANGE from cST1): |-(beta - beta_pre)/ms - sign(L)| <= %.1e + %.1f*ulp32(beta)/ms "
      "= %.4e at ms %g, |beta| < 16" % (SIGN_ABS_BAR, SIGN_ULP_K, SIGN_ABS_BAR + SIGN_ULP_K * ULP32_MAX_IN_CLIP / MS,
                                        MS))
    R("  cST1 applied an absolute %.1e; the float32 bound 0.5*ulp32/ms is %.4e here and exceeds it for "
      "ms < %.6e" % (PRED_SIGN_BAR, 0.5 * ULP32_MAX_IN_CLIP / MS, MS_CROSSOVER))
    for arm in ARMS:
        for s in SEEDS:
            d = dec[(arm, s)]
            R("  %-4s %s   %s" % ("PASS" if decomp_ok(d) else "FAIL", "G-DECOMP %s-s%d" % (arm, s),
                                  "shape %d z %d m %d (worst %.1e / %.1e) sign %d/%d clamped %d shared %d | "
                                  "worst|q-s| %.3e bar %.3e" % (d["bad_shape"], d["bad_z"], d["bad_m"], d["worst_z"],
                                                                d["worst_m"], d["bad_sign"], d["n_sign"],
                                                                d["clamped"], d["bad_shared"], d["worst_q"],
                                                                d["tol_max"])))
    R("")
    R("[NOMINATION] k01's records only, pooled over seeds (the registered check)")
    for s in SEEDS:
        describe(R, "k01-s%d" % s, per_seed[s])
    describe(R, "k01 POOLED", nst)
    R("  calibration at ms 1e-3 (registration): scalar DOM_C %.3f-%.3f pooled %.4f; layerwise 0.0000"
      % (CALIB_SCALAR_DOM + (CALIB_SCALAR_POOLED,)))
    R("")
    R("[LEVELS] plateau5 (train5) per run; arm means; contrasts in pp and SE (SE_PRIOR %.6f)" % SE_PRIOR)
    for arm in ARMS:
        v = [outs[(arm, s)]["p5"] for s in SEEDS]
        t = [outs[(arm, s)]["t5"] for s in SEEDS]
        R("  %-4s %s  mean %.4f  range %.3f  | train %s mean %.4f"
          % (arm, " ".join("%.3f" % x for x in v), lev[arm], rngs[arm], " ".join("%.3f" % x for x in t), trn[arm]))
    for nm, d in (("D_ISO = ISO - k01", d_iso), ("D_CTL = CTL - k01", d_ctl), ("P = ISO - CTL", pp)):
        R("  %-20s %+8.4f pp = %+8.2f SE" % (nm, d, d / SE_PRIOR))
    R("  in-batch sigma %.4f (df %d); SIGMA_PRIOR %.4f -> %s"
      % (sig_in, len(ins) - len(ARMS), SIGMA_PRIOR, "SIGMA-PRIOR-FROZEN"))
    R("")
    R("[DESCRIPTIVE, gating nothing] the counterfactual shared vote inside ISO and CTL; beta per group")
    describe(R, "ISO POOLED", iso_st)
    describe(R, "CTL POOLED", ctl_st)
    for arm in ARMS:
        for s in SEEDS:
            recs = probes[(arm, s)]["recs"]
            for c in range(N_COORDS[arm]):
                b = [x["beta"][c] for x in recs]
                pk = max(range(len(b)), key=lambda i: b[i])
                R("  beta %s-s%d g%d: peak %.3f at step %d; terminal %.3f; share at floor %.3f"
                  % (arm, s, c, b[pk], recs[pk]["step"], b[-1], sum(1 for x in b if x <= BETA_LO + 1e-6) / len(b)))
    R("  corpus anchors (non-gating, filtered): %s" % "; ".join("%s %.4f" % kv for kv in ANCHOR.items()))
    R("")
    R("BRANCH   %s" % branch)
    R("LICENCE  %s" % lic)
    R("NOT LICENSED  %s" % NOT_LICENSED)
    R("")
    R(final)

    P("\n" + "=" * 100)
    P("REPLICA -- the successor scorer's output from its [G-DECOMP] header to FINAL, rebuilt independently")
    P("=" * 100)
    for ln in REPLICA:
        P(ln)
    P("=" * 100)
    chk(final == EXPECT_FINAL, "FINAL re-derived == the successor scorer's FINAL")
    chk(branch == "NOMINATION-PARTIAL+ISO-RESCUES+CTL-NULL", "BRANCH re-derived", branch)
    chk(head is None, "this branch has NO headline sentence: the licence is composed word by word")

    P("\nVIOLATIONS %d" % len(VIOL))
    for v in VIOL:
        P("  ! %s" % v)
    sys.stdout.write("\n".join(OUT) + "\n")
    print("HOSTPATH runsdir  %s" % os.path.abspath(rd))
    print("HOSTPATH corpus   %s" % os.path.abspath(a.csv))
    return 0 if not VIOL else 1
