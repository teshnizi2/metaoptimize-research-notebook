def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))

# =============================================================================
# O2: the ONLY composition of sigma.  ONE argument.  NO corpus.
# =============================================================================
def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]

    """VOTE_W value: `<name>:0` for every silenced tensor, ascending index, joined by `/` ('' = unset)."""
    return "/".join("%s:0" % NAMES[i - 1] for i in SILENCED[arm])


def witness_of(arm):
    if not SILENCED[arm]:
        return "VOTE_W: off"
    return "VOTE_W: on type=scalar items=" + ",".join(
        "%d:%s:w=0.0:group=0:groupsize=53" % (i, NAMES[i - 1]) for i in SILENCED[arm])


VOTEW = dict((a, votew_of(a)) for a in ARMS)
WITNESS = dict((a, witness_of(a)) for a in ARMS)
WEIGHTS = dict((a, dict((i, 0.0) for i in SILENCED[a])) for a in ARMS)      # 1-based


def parse_sets(spec, names):
    """`sets:<g>/<g>` -> list of 1-based index lists (each sorted); no HF.py import."""
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names=None):
    names = names or NAMES
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    """(arm, seed) -> the run.  The highest INTEGER job id among COMPLETE runs wins; a truncated
    resubmission cannot shadow a complete run."""
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s"
                         % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes


def resolve(runsdir, explicit, base):
    if explicit:
        return explicit, "explicit"
    for c, how in ((os.path.join(runsdir, PREFIX, base), "<runsdir>/%s/" % PREFIX),
                   (os.path.join(runsdir, "%s-%s" % (PREFIX, base)), "<runsdir>/%s-" % PREFIX)):
        if os.path.exists(c):
            return c, how
    return None, "not found"


def resolve_manifest(runsdir, explicit):
    return resolve(runsdir, explicit, "PARTITION-MANIFEST.txt")


def read_kv(path):
    if not path or not os.path.exists(path):
        return None
    d = {}
    for ln in open(path):
        p = ln.split()
        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))
    return d


def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")


def std_cell(r, any_gran=False):
    return ((any_gran or r.get("granularity") in ("scalar", "layerwise"))
            and r.get("epochs_requested") == str(EPOCHS)
            and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0
            and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra, any_gran=False):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r, any_gran) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm

def load_probe(runsdir, arm, seed, prefix=PREFIX):
    """-> list of records, or None if the file is absent.  Unparseable lines are None entries so a
    garbled file cannot pass the record-count check.  json accepts NaN/Infinity tokens."""
    for base in (os.path.join(runsdir, prefix), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (prefix, arm, seed), "probe.jsonl")
        if os.path.exists(p):
            recs = []
            for ln in open(p, errors="replace"):
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    r = json.loads(ln)
                except ValueError:
                    recs.append(None)
                    continue
                recs.append(r if isinstance(r, dict) else None)
            return recs
    return None


def _wsum(W, vals, g):
    return sum(W.get(i, 1.0) * vals[i - 1] for i in g)


def bite_check(arm, recs):
    """G-BITE (+ G-BITE-SET) for ONE run.  -> (ok, detail).  Pure function of the records."""
    groups = arm_groups(arm)
    W = WEIGHTS[arm]
    only50 = WEIGHTS["MUTE50"]
    other = {"MUTEDOWN": WEIGHTS["MUTECTL"], "MUTECTL": WEIGHTS["MUTEDOWN"]}.get(arm)
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_z": 0, "bad_m": 0, "worst_z": 0.0, "worst_m": 0.0,
         "informative": 0, "unweighted_match_on_informative": 0,
         "set_informative": 0, "only50_match_on_set_informative": 0,
         "swap_informative": 0, "other_set_match": 0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for r in recs:
        ok = (r is not None and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS
              and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
              and isinstance(r.get("z_agg"), list) and len(r["z_agg"]) == 1 and isinstance(r["z_agg"][0], list)
              and len(r["z_agg"][0]) == len(groups)
              and isinstance(r.get("mom_pre"), list) and len(r["mom_pre"]) == 1 and isinstance(r["mom_pre"][0], list)
              and len(r["mom_pre"][0]) == len(groups))
        if not ok:
            d["bad_rec"] += 1
            continue
        try:
            zt = [float(x) for x in r["z_tensor"]]
            mt = [float(x) for x in r["m_tensor"]]
            za = [float(x) for x in r["z_agg"][0]]
            ma = [float(x) for x in r["mom_pre"][0]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in zt + mt + za + ma):
            d["nonfinite"] += 1
            continue
        d["n"] += 1
        sz = sum(abs(x) for x in zt) + 1e-30
        sm = sum(abs(x) for x in mt) + 1e-30
        wz = [_wsum(W, zt, g) for g in groups]
        wm = [_wsum(W, mt, g) for g in groups]
        uz = [sum(zt[i - 1] for i in g) for g in groups]
        ez = max(abs(p - q) for p, q in zip(wz, za)) / sz
        em = max(abs(p - q) for p, q in zip(wm, ma)) / sm
        d["worst_z"], d["worst_m"] = max(d["worst_z"], ez), max(d["worst_m"], em)
        d["bad_z"] += ez > BITE_TOL
        d["bad_m"] += em > BITE_TOL
        if W and max(abs(p - q) for p, q in zip(wz, uz)) / sz > INFORM:
            d["informative"] += 1
            d["unweighted_match_on_informative"] += max(abs(p - q) for p, q in zip(uz, za)) / sz <= BITE_TOL
        if other is not None:
            oz = [_wsum(only50, zt, g) for g in groups]
            if max(abs(p - q) for p, q in zip(wz, oz)) / sz > INFORM:
                d["set_informative"] += 1
                d["only50_match_on_set_informative"] += max(abs(p - q) for p, q in zip(oz, za)) / sz <= BITE_TOL
            xz = [_wsum(other, zt, g) for g in groups]
            if max(abs(p - q) for p, q in zip(wz, xz)) / sz > INFORM:
                d["swap_informative"] += 1
                d["other_set_match"] += max(abs(p - q) for p, q in zip(xz, za)) / sz <= BITE_TOL
    ok = (len(recs) == N_RECORDS and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == N_RECORDS
          and d["bad_z"] == 0 and d["bad_m"] == 0)
    if W:
        ok = ok and d["informative"] >= MIN_INFORMATIVE and d["unweighted_match_on_informative"] == 0
    if other is not None:
        ok = (ok and d["set_informative"] >= MIN_INFORMATIVE and d["only50_match_on_set_informative"] == 0
              and d["other_set_match"] == 0)
    return ok, d


def runaway_check(recs):
    """G-RUNAWAY for ONE run, from its own probe beta[0].  Pure function of the records."""
    d = {"n_post": 0, "at_ceil": 0, "drift": 0, "nonfinite": 0, "max_finite": None, "frac_ceil": 0.0,
         "frac_drift": 0.0}
    for r in recs or []:
        if not isinstance(r, dict) or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) \
                or not r["beta"]:
            continue
        try:
            b = float(r["beta"][0])
        except (TypeError, ValueError):
            continue
        if not math.isfinite(b):
            d["nonfinite"] += 1
            continue
        d["max_finite"] = b if d["max_finite"] is None else max(d["max_finite"], b)
        if r["step"] >= RUNAWAY_FROM_EPOCH * STEPS_PER_EPOCH:
            d["n_post"] += 1
            d["at_ceil"] += b >= BETA_CEIL - AT_CEIL_TOL
            d["drift"] += b >= DRIFT_BETA
    if d["n_post"]:
        d["frac_ceil"] = d["at_ceil"] / float(d["n_post"])
        d["frac_drift"] = d["drift"] / float(d["n_post"])
    d["runaway"] = d["nonfinite"] > 0 or d["frac_ceil"] >= RUNAWAY_FRAC
    d["drifts"] = d["frac_drift"] >= RUNAWAY_FRAC
    return d

# DESCRIPTIVE, NON-GATING probe readouts.  Nothing here reaches decide(), a gate or the FINAL line.
# =============================================================================
def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord \
                and isinstance(r.get("step"), int):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    at = [v <= CLAMP_LO + 1e-4 for _s, v in vals]
    pin = None
    for k in range(len(vals)):
        if all(at[k:]):
            pin = vals[k][0] / float(STEPS_PER_EPOCH)
            break
    pk = max(vals, key=lambda sv: sv[1])
    return {"last": vals[-1][1], "peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH),
            "frac_clamp": sum(at) / float(len(at)), "pin_epoch": pin, "n": len(vals)}


def descriptive_probe_readout(runsdir):
    print("\nDESCRIPTIVE, NON-GATING -- beta trajectories from the runs' own probe.jsonl")
    print("  (cvt1: k01 pinned at 36.2-36.6, MUTE at 40.2-40.6; HEAD's complement at 99.8, its isolated beta at 21-24.)")
    for a in ARMS:
        for s in SEEDS:
            recs = load_probe(runsdir, a, s)
            if not recs:
                continue
            for coord in ((0,) if PT_TYPE[a] == "scalar" else (0, 1)):
                bs = beta_summary(recs, coord)
                lab = "beta[0]" if PT_TYPE[a] == "scalar" else ("complement beta[0]" if coord == 0 else "isolated beta[1]")
                if bs:
                    print("  %-8s s%d  %-19s last %8.3f  peak %8.3f @ep %5.1f  at-floor-clamp frac %.3f  pinned-from-epoch %s"
                          % (a, s, lab, bs["last"], bs["peak"], bs["peak_epoch"], bs["frac_clamp"],
                             "never" if bs["pin_epoch"] is None else "%.1f" % bs["pin_epoch"]))
    print("\nDESCRIPTIVE, NON-GATING -- who votes in the shared sum from epoch 36 (L_i = w_i (b2 m_i + (1-b2) z_i));")
    print("  top 6 UNSILENCED voters by mean share of |L|, with the fraction of records each votes DOWN")
    for a in ("MUTE50", "MUTEDOWN", "MUTECTL"):
        W = WEIGHTS[a]
        for s in SEEDS:
            recs = [r for r in (load_probe(runsdir, a, s) or []) if isinstance(r, dict)
                    and isinstance(r.get("step"), int) and r["step"] >= 36 * STEPS_PER_EPOCH
                    and isinstance(r.get("m_tensor"), list) and len(r["m_tensor"]) == NTENS
                    and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS]
            recs = [r for r in recs if all(math.isfinite(float(x)) for x in r["m_tensor"] + r["z_tensor"])]
            if not recs:
                continue
            acc, down = {}, 0
            for r in recs:
                b2 = r.get("pt_b2", 0.9)
                L = [W.get(i, 1.0) * (b2 * r["m_tensor"][i - 1] + (1 - b2) * r["z_tensor"][i - 1]) for i in range(1, NTENS + 1)]
                tot = sum(abs(x) for x in L) + 1e-30
                down += sum(L) > 0
                for i in range(1, NTENS + 1):
                    if i in W:
                        continue
                    t_ = acc.setdefault(i, [0.0, 0])
                    t_[0] += abs(L[i - 1]) / tot
                    t_[1] += L[i - 1] > 0
            top = sorted(acc.items(), key=lambda kv: -kv[1][0])[:6]
            print("  %-8s s%d  %3d records, sum DOWN on %.3f | %s" % (
                a, s, len(recs), down / float(len(recs)),
                "  ".join("%d %s %.3f D%.2f" % (i, NAMES[i - 1], v[0] / len(recs), v[1] / float(len(recs))) for i, v in top)))


# =============================================================================
# THE BRANCH MAP -- a pure function of the in-batch means and two in-batch flags
# =============================================================================
def decide(M, diverged=False, runaway=None):
    """M: arm -> mean plateau5.  runaway: arm -> number of seeds that ran away (G-RUNAWAY)."""
    rw = dict((a, 0) for a in ARMS)
    rw.update(runaway or {})
    if rw["MUTEDOWN"] >= RUNAWAY_MIN_SEEDS:
        return "UP-RUNAWAY"
    if rw["MUTEDOWN"] >= 1:
        return "UNRESOLVED-RUNAWAY-SPLIT"
    if any(rw[a] >= 1 for a in ARMS if a != "MUTEDOWN"):
        return "CONTROL-RUNAWAY"
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    D_HEAD = M["HEAD"] - M["k01"]
    D_MUTE50 = M["MUTE50"] - M["k01"]
    D_DOWN = M["MUTEDOWN"] - M["k01"]
    D_CTL = M["MUTECTL"] - M["k01"]
    P_COAL = M["MUTEDOWN"] - M["MUTECTL"]
    G_DOWN = M["HEAD"] - M["MUTEDOWN"]
    if D_HEAD < GAP_MIN:
        return "POSITIVE-CONTROL-FAILED"
    if D_MUTE50 >= RESCUE_BAR:
        return "MUTE50-RESCUES"
    if D_DOWN < -NULL_BAR:
        return "MUTEDOWN-BELOW-K01"
    if D_CTL >= RESCUE_BAR:
        return "COALITION-OVER-NONSPECIFIC" if P_COAL >= RESCUE_BAR else "NONSPECIFIC"
    if D_CTL > NULL_BAR:
        return "CONTROL-WEAK"
    if D_DOWN >= RESCUE_BAR:
        return "COALITION" if G_DOWN <= MATCH_BAR else "COALITION-PARTIAL"
    if D_DOWN <= NULL_BAR:
        return "OWN-STEP-NECESSARY"
    return "DOWN-WEAK"

# =============================================================================
def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound / patch-not-verified, 2 incomplete."""
    print("=" * 78)
    print(" cvt3 -- DOES SILENCING THE WHOLE DOWN COALITION RESCUE WITHOUT AN OWN STEP SIZE?  (PlainNet18_c100)")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (NET, DSET, EPOCHS, ",".join(str(s) for s in SEEDS), AUG, CLIP, PROBE))
    for a in ARMS:
        print("   %-8s %-38s silenced %s" % (a, SPEC[a], " ".join(str(i) for i in SILENCED[a]) or "(VOTE_W unset)"))
    print(" COALITION (RULE C, rank order): %s" % " ".join(str(i) for i in COALITION))
    print(" CONTROL   (RULE K, pair order): %s" % " ".join(str(i) for i in CONTROL))
    print(" PRIMARY: P_COAL = MUTEDOWN - MUTECTL.  plateau5 = mean TEST over epochs %d..%d of each run's own .out."
          % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2); the corpus cannot move them.")
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:                      # a disclosure must never crash the scorer
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = read_runs(runsdir)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        note(n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing),
                                                            len(ARMS) * len(SEEDS), ",".join(missing))
        print("\n" + final)
        print("No branch is taken and NO NUMBER HERE IS QUOTABLE.")
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    every run's OWN ARGS line says what its file name registers")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = {"optimizer": "HF", "alg-base": "SGDm", "alg-meta": "Lion", "dataset": DSET,
                    "NN-name": NET, "batch-size": str(BATCH), "num-epochs": str(EPOCHS),
                    "meta-stepsize": MST, "alpha0": A0, "stepsize-groups": SPEC[a],
                    "seed": str(s), "run-name": "%s-%s-s%d" % (PREFIX, a, s),
                    "momentum-param-base": "0.99", "weight-decay-base": "0.1", "momentum-param-meta": "0.99",
                    "Lion-beta2-meta": "0.9", "weight-decay-meta": "0", "gamma": "1"}
            bad = [k for k, v in want.items() if f.get(k, [None])[-1] != v]
            hchk(not dup and not bad, "G-ARGS %s-s%d" % (a, s),
                 "repeated %s; wrong %s" % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad])
                 if (dup or bad) else "")

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(re.sub(r" PROBE_DIR=\S*", "", ln) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),
         "G-ENV one distinct ENV line, one per run", "%d distinct" % len(envs))
    for ln in envs:
        print("        %s" % ln)
    hchk(envs == [ENV_EXPECTED], "G-ENV the ENV line is cvt1's / cpl2's, byte for byte (PROBE_DIR stripped)")
    badpt = [(a, s) for (a, s), r in sorted(runs.items())
             if len(r["pt"]) != 1 or not PT_RE.match(r["pt"][0])
             or PT_RE.match(r["pt"][0]).group(1) != PT_TYPE[a]]
    hchk(not badpt, "G-ENV every run printed ONE `PROBE_TENSOR: on every=%d type=<its arm's> tensors=%d`"
         % (PROBE, NTENS), "bad: %s" % badpt[:5] if badpt else "%d runs" % len(runs))

    print("\nG-VOTEW   every run printed exactly ONE VOTE_W line, == its arm's registered witness")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            hchk(r["vw"] == [WITNESS[a]], "G-VOTEW %s-s%d (%d items)" % (a, s, len(SILENCED[a])),
                 "" if r["vw"] == [WITNESS[a]] else "got %r" % ([x[:120] for x in r["vw"][:2]],))

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == the frozen manifest, byte for byte")
    mp, how = resolve_manifest(runsdir, manifest_path)
    if not mp:
        hchk(False, "G-STRUCT manifest present", how)
    else:
        txt = open(mp).read()
        print("  manifest: %s (%s)" % (mp, how))
        hchk(txt == synthetic_manifest_text(),
             "G-STRUCT byte-identical to synthetic_manifest_text() (53 tensors, HEAD [52,1], the two sets, the VOTE_W table)",
             "%d vs %d bytes" % (len(txt), len(synthetic_manifest_text())))

    print("\nG-PROV    PROVENANCE.txt")
    pp, phow = resolve(runsdir, "", "PROVENANCE.txt")
    prov = read_kv(pp)
    if prov is None:
        hchk(False, "G-PROV PROVENANCE.txt present", phow)
    else:
        print("  provenance: %s (%s)" % (pp, phow))
        hchk(prov.get("MODE") == "submit", "G-PROV MODE submit", str(prov.get("MODE")))
        hchk(prov.get("BUILD_NETWORK_SHA256") == POST_SHA,
             "G-PROV build_network.py sha256 == POST_SHA %s..." % POST_SHA[:12], str(prov.get("BUILD_NETWORK_SHA256"))[:16])
        hchk(prov.get("HF_SHA256") == HF_POST_SHA,
             "G-PROV HF.py sha256 == HF_POST_SHA %s... (cvt1's PATCH_VOTEWEIGHT tree, unchanged)" % HF_POST_SHA[:12],
             str(prov.get("HF_SHA256"))[:16])
        hchk(prov.get("RUNNER_SHA256") == RUNNER_SHA, "G-PROV RUNNER_SHA256 == cvt1's runner %s..." % RUNNER_SHA[:12],
             str(prov.get("RUNNER_SHA256"))[:16])
        me = file_sha(os.path.abspath(__file__))
        hchk(prov.get("SCORER_SHA256") == me, "G-PROV SCORER_SHA256 == this scorer's sha256 %s..." % me[:12],
             str(prov.get("SCORER_SHA256"))[:16])

    print("\nLEVELS    plateau5 (TEST) and TRAIN, IN BATCH; tail slope = TEST OLS over epochs 80-99 (descriptive)")
    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)
    arm_t = dict((a, [runs[(a, s)]["train5"] for s in SEEDS]) for a in ARMS)
    for a in ARMS:
        v, t = arm_v[a], arm_t[a]
        sl = [tail_slope(runs[(a, s)]["eps"]) for s in SEEDS]
        print("  %-8s TEST %.4f  sd %s  range %.4f   TRAIN %.4f   tail slope %s pp/ep"
              % (a, mean(v), fmt(sd(v), 4), max(v) - min(v), mean(t), "/".join(fmt(x, 3) for x in sl)))
        print("           seeds: %s" % ", ".join("s%d %.4f (train %.4f)" % (s, x, y) for s, x, y in zip(SEEDS, v, t)))
    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    hi = max(M.values())
    print("\nG-FLOOR / G-CEIL")
    hchk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp (chance %.2f)" % (FLOOR_MIN, CHANCE), "max %.4f" % hi)
    hchk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)

    if harness:
        final = "FINAL: HARNESS-UNSOUND | %d harness gate(s) failed: %s" % (
            len(harness), "; ".join(h.split()[0] for h in harness))
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)
        for h in harness:
            print("   FAILED  %s" % h)
        for ln in LICENSE["HARNESS-UNSOUND"]:
            print("  " + ln)
        return 1, final

    print("\nG-BITE    the registered weight SETS reached the shared sum -- read from the runs' OWN probe records")
    print("  per finite record: z_agg == sum_i w_i z_tensor_i and mom_pre == sum_i w_i m_tensor_i (rel %.0e of sum|term|);" % BITE_TOL)
    print("  intervened runs: >= %d records where weighted and UNWEIGHTED sums differ by > %.0e, on none of which the" % (MIN_INFORMATIVE, INFORM))
    print("  unweighted sum matches.  G-BITE-SET (MUTEDOWN, MUTECTL): >= %d records where the registered set and the" % MIN_INFORMATIVE)
    print("  50-ONLY set differ by > %.0e, on none of which the 50-only sum matches; the OTHER arm's set matches NO record." % INFORM)
    bite_fail = []
    nonfin = {}
    for s in SEEDS:
        for a in ARMS:
            recs = load_probe(runsdir, a, s)
            ok, d = bite_check(a, recs)
            nonfin[(a, s)] = d["nonfinite"]
            extra = ""
            if a in ("MUTEDOWN", "MUTECTL"):
                extra = "  set-informative %d 50-only-match %d  swap-informative %d other-set-match %d" % (
                    d["set_informative"], d["only50_match_on_set_informative"], d["swap_informative"], d["other_set_match"])
            print("  %-4s G-BITE %-8s s%d  records %s  bad %d  nonfinite %d  worst z %.1e m %.1e  informative %d  unweighted-match %d%s%s"
                  % ("PASS" if ok else "FAIL", a, s, "absent" if recs is None else len(recs), d["bad_rec"], d["nonfinite"],
                     d["worst_z"], d["worst_m"], d["informative"], d["unweighted_match_on_informative"], extra,
                     ("  (%s)" % d["why"]) if "why" in d else ""))
            if not ok:
                bite_fail.append("%s-s%d" % (a, s))
    if bite_fail:
        final = "FINAL: PATCH-NOT-VERIFIED | G-BITE failed on %s" % ",".join(bite_fail)
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)
        for ln in LICENSE["PATCH-NOT-VERIFIED"]:
            print("  " + ln)
        return 1, final

    print("\nG-RUNAWAY the shared beta[0] trajectory, stamped BEFORE any branch is read (from the probe records)")
    print("  at ceiling iff beta[0] >= %.4f - %.2f; a seed runs away iff >= %.2f of its epoch >= %.0f records are at the"
          % (BETA_CEIL, AT_CEIL_TOL, RUNAWAY_FRAC, RUNAWAY_FROM_EPOCH))
    print("  ceiling or any record is non-finite; an arm runs away iff >= %d seeds do.  DRIFT: beta[0] >= %.1f (stamp only)."
          % (RUNAWAY_MIN_SEEDS, DRIFT_BETA))
    rw_seeds = dict((a, 0) for a in ARMS)
    drift_seeds = dict((a, 0) for a in ARMS)
    for a in ARMS:
        for s in SEEDS:
            d = runaway_check(load_probe(runsdir, a, s))
            rw_seeds[a] += d["runaway"]
            drift_seeds[a] += d["drifts"]
            print("  %-8s s%d  epoch>=17 records %3d  at-ceiling frac %.3f  drift frac %.3f  nonfinite %d  max finite beta[0] %s  -> %s"
                  % (a, s, d["n_post"], d["frac_ceil"], d["frac_drift"], d["nonfinite"], fmt(d["max_finite"], 3),
                     "RUNS AWAY" if d["runaway"] else "no runaway"))
    stamps_rw = ["RUNAWAY-%s-%dof3" % (a, rw_seeds[a]) for a in ARMS if rw_seeds[a]]
    stamps_rw += ["NONFINITE-%s" % a for a in ARMS if any(nonfin[(a, s)] for s in SEEDS)]
    if drift_seeds["MUTEDOWN"] >= RUNAWAY_MIN_SEEDS:
        stamps_rw.append("DRIFT-MUTEDOWN")
    print("  runaway stamps: %s" % (" | ".join(stamps_rw) or "none"))

    print("\nSIGMA     O2: max(frozen floor, in-batch) -- nothing from the corpus")
    ss = sum(sum((x - M[a]) ** 2 for x in arm_v[a]) for a in ARMS)
    df = sum(len(arm_v[a]) - 1 for a in ARMS)
    sigma_in = math.sqrt(ss / df)
    which, sigma_used, cands = compose_sigma(sigma_in)
    for nm, v in cands:
        print("  %-20s %.6f" % (nm, v))
    se = sigma_used * math.sqrt(2.0 / 3.0)
    print("  SIGMA_USED           %.6f  (= %s)" % (sigma_used, which))
    print("  SE_ARM_DIFF          %.6f  = SIGMA_USED * sqrt(2/3)  (df_inbatch %d)" % (se, df))

    D_HEAD = M["HEAD"] - M["k01"]
    D_MUTE50 = M["MUTE50"] - M["k01"]
    D_DOWN = M["MUTEDOWN"] - M["k01"]
    D_CTL = M["MUTECTL"] - M["k01"]
    G_DOWN = M["HEAD"] - M["MUTEDOWN"]
    P_COAL = M["MUTEDOWN"] - M["MUTECTL"]
    P_COAL_TR = T["MUTEDOWN"] - T["MUTECTL"]
    print("\nCONTRASTS (every one WITHIN batch)")
    print("  PRIMARY  P_COAL   = MUTEDOWN - MUTECTL = %.4f - %.4f = %+.4f pp = %+.2f SE   (TRAIN %+.4f)"
          % (M["MUTEDOWN"], M["MUTECTL"], P_COAL, P_COAL / se, P_COAL_TR))
    print("  KEY      D_DOWN   = MUTEDOWN - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)" % (D_DOWN, D_DOWN / se, T["MUTEDOWN"] - T["k01"]))
    print("  KEY      D_CTL    = MUTECTL  - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)" % (D_CTL, D_CTL / se, T["MUTECTL"] - T["k01"]))
    print("  KEY      D_HEAD   = HEAD     - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)  [positive control]"
          % (D_HEAD, D_HEAD / se, T["HEAD"] - T["k01"]))
    print("  KEY      D_MUTE50 = MUTE50   - k01 = %+.4f pp = %+.2f SE   (TRAIN %+.4f)  [cvt1's MUTE replicated]"
          % (D_MUTE50, D_MUTE50 / se, T["MUTE50"] - T["k01"]))
    print("           G_DOWN   = HEAD - MUTEDOWN = %+.4f pp = %+.2f SE   (TRAIN %+.4f)" % (G_DOWN, G_DOWN / se, T["HEAD"] - T["MUTEDOWN"]))
    print("  bars: GAP_MIN %.1f (%.2f SE) | RESCUE %.1f pp (%.2f SE) | NULL %.1f pp (%.2f SE) | MATCH %.1f pp (%.2f SE)"
          % (GAP_MIN, GAP_MIN / se, RESCUE_BAR, RESCUE_BAR / se, NULL_BAR, NULL_BAR / se, MATCH_BAR, MATCH_BAR / se))

    print("\nG-DIVERGE (a branch condition, not a harness gate)")
    div = [a for a in ARMS if max(arm_v[a]) - min(arm_v[a]) > DIVERGED_BAR]
    print("  %s every arm's seed range <= %.1f pp%s" % ("PASS" if not div else "FAIL", DIVERGED_BAR,
                                                     "" if not div else "  diverged: %s" % div))

    branch = decide(M, bool(div), rw_seeds)
    stamps = ["HARNESS-CLEAN", "PATCH-BITES", "SETS-FROM-CVT1-MUTE", "CTL-COUNT-CLASS-NOT-MASS", "NO-TIME-GATE",

              "SIGMA-PRIOR-FROZEN" if which == "SIGMA_PRIOR_frozen" else "SIGMA-INBATCH"]
    stamps += stamps_rw
    if div:
        stamps.append("DIVERGED-%s" % "-".join(div))
    stamps.append("POSITIVE-CONTROL-REPRODUCES" if D_HEAD >= GAP_MIN else "POSITIVE-CONTROL-FAILED")
    stamps.append("MUTE50-AT-K01" if D_MUTE50 <= NULL_BAR else ("MUTE50-WEAK" if D_MUTE50 < RESCUE_BAR else "MUTE50-RESCUES"))
    if M["MUTEDOWN"] - M["HEAD"] > MATCH_BAR:
        stamps.append("DOWN-ABOVE-HEAD")
    if D_CTL < -NULL_BAR:
        stamps.append("CTL-BELOW-K01")
    if any(abs(M[a] - M["k01"]) <= NULL_BAR for a in INTERVENED):
        stamps.append("FLOOR-READINGS-ARE-BOUNDS")
    stamps.append("TRAIN-AGREES" if ((P_COAL_TR > 0) == (P_COAL > 0)) else "TRAIN-DISAGREES")
    final = "FINAL: %s | %s" % (branch, " | ".join(stamps))
    print("\n" + "=" * 78)
    print(final)
    print("=" * 78)

    print("\nBETWEEN-BATCH, NON-GATING (frozen literals; NO other batch enters any cvt3 bar or contrast):")
    print("  cvt1 {78,79,80}: k01 %.4f  HEAD %.4f  MUTE %.4f  DOSE %.4f  INJECT %.4f"
          % (R_CVT1["k01"], R_CVT1["HEAD"], R_CVT1["MUTE"], R_CVT1["DOSE"], R_CVT1["INJECT"]))

          % (M["k01"], M["HEAD"], M["MUTE50"], M["MUTEDOWN"], M["MUTECTL"]))

    try:
        descriptive_probe_readout(runsdir)
    except Exception as ex:                      # a disclosure must never crash the scorer
        print("  (descriptive probe readout skipped: %s)" % type(ex).__name__)

    print("\nWHAT THIS DECIDES -- as registered, before any run existed:")
    for ln in LICENSE[branch]:
        print("  " + ln)
    print("\nWHAT IT DOES NOT LICENSE, UNCONDITIONALLY:")
    print("  * Anything about ResNet18_c100 (BN or GN) or VGG11_bn_c100: one network, one cell.")
    print("  * Anything about WHEN the silence acts: VOTE_W has no time gate; every weight is 0 from step 0, while the")
    print("    sets were chosen on post-epoch-36 records.  Dose and onset are confounded here as in cvt1 (230.4(1)).")
    print("  * A mass-matched control: MUTECTL holds ~1/30 of MUTEDOWN's mass on the rule's records (232.3).")
    print("  * Which members of the coalition are necessary: one set, one cut (1.00), one eligibility bar (0.90).")
    print("  * Any point agreement for an arm at the k01 level: such readings are BOUNDS (164.6).")
    print("  * Anything beyond 100 epochs.")
    return 0, final

            "--weight-decay-meta 0 --dataset CIFAR100 --NN-name %s --batch-size 100 "
            "--max-time 999:00:00 --gamma 1 --meta-stepsize 1e-3 --alpha0 1e-6 "
            "--num-epochs 100 --stepsize-groups %s --seed %d "
            "--save-directory /cluster/account_b/metaopt/runs/cvt3 --run-name %s"

             "PROBE_TENSOR: on every=100 type=%s tensors=%d meta_alg=Lion momentum_param=0.99 "
             "Lion_beta2=0.9 dir=/x" % (PT_TYPE[arm], NTENS)]

def _e_probe_records(arm, weights=None, n=N_RECORDS, seed=0, beta_mode="collapse", nan_from=None):
    """synthetic PROBE_TENSOR records consistent with `weights` (default: the arm's own).
    beta_mode: collapse (falls to the floor clamp), ceil (rises to the ceiling from epoch 20), drift
    (holds at -3.5).  nan_from: record index from which every value is NaN (a blow-up)."""
    import random
    rnd = random.Random(1000 + seed)
    W = WEIGHTS[arm] if weights is None else weights
    groups = arm_groups(arm)
    out = []
    for k in range(n):
        zt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        mt = [rnd.uniform(-1e-4, 1e-4) for _ in range(NTENS)]
        zt[IDX_HEAD - 1] = 5e-2 * (1 + 0.1 * rnd.random())
        mt[IDX_HEAD - 1] = 4e-2
        for i in COALITION:
            zt[i - 1] = 1e-2 * (1 + 0.1 * rnd.random())
            mt[i - 1] = 8e-3
        for i in CONTROL:
            zt[i - 1] = -2e-3 * (1 + 0.1 * rnd.random())
            mt[i - 1] = -1.5e-3
        za = [sum(W.get(i, 1.0) * zt[i - 1] for i in g) for g in groups]
        ma = [sum(W.get(i, 1.0) * mt[i - 1] for i in g) for g in groups]
        e = 100.0 * k / STEPS_PER_EPOCH
        if beta_mode == "ceil":
            b = -13.8 + 0.6 * e if e < 20 else BETA_CEIL
            b = min(b, BETA_CEIL)
        elif beta_mode == "drift":
            b = min(-13.8 + 0.6 * e, -3.5)
        else:
            b = max(-15.0, -6.0 - 0.02 * k)
        rec = {"step": 100 * k, "beta": [b] * len(groups), "pt_b2": 0.9, "pt_mp": 0.99,
               "z_tensor": zt, "m_tensor": mt, "z_agg": [za], "mom_pre": [ma], "beta_pre": [[b] * len(groups)]}
        if nan_from is not None and k >= nan_from:
            nan = float("nan")
            rec.update({"beta": [nan] * len(groups), "z_tensor": [nan] * NTENS, "m_tensor": [nan] * NTENS,
                        "z_agg": [[nan] * len(groups)], "mom_pre": [[nan] * len(groups)]})
        out.append(rec)
    return out


def _e_probe_template(root):
    """one directory of synthetic probe dirs for all 15 runs; cases symlink it."""

def _e_build(d, template, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, shadow=None,
             env_break=None, spec_break=None, vw_break=None, manifest=None, prov=None,
             probe_break=None, probe_drop=None):
    """probe_break: {(arm, seed): records}."""
    lv = dict(_E_BASE, **(levels or {}))
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(
        manifest if manifest is not None else synthetic_manifest_text())
    _e_prov(d, **(prov or {}))
    jid = 9800000
    pb = probe_break or {}
    for i, s in enumerate(SEEDS):
        for a in ARMS:
            jid += 1
            pname = "probe_%s-%s-s%d" % (PREFIX, a, s)
            if (a, s) in pb:
                p = os.path.join(d, PREFIX, pname)
                os.makedirs(p)
                with open(os.path.join(p, "probe.jsonl"), "w") as f:
                    for r in pb[(a, s)]:
                        f.write((json.dumps(r) if isinstance(r, dict) else r) + "\n")
            elif probe_drop != (a, s):
                os.symlink(os.path.join(template, pname), os.path.join(d, PREFIX, pname))
            if drop == (a, s):
                continue
            L = (over or {}).get((a, s), lv[a] + jit[i])
            env = _E_ENV.replace("AUGMENT=1", "AUGMENT=0") if env_break == (a, s) else _E_ENV
            spec = HEADSPEC if spec_break == (a, s) else None
            vw = vw_break[1] if vw_break and vw_break[0] == (a, s) else None
            _e_write(d, a, s, jid, L, env=env, spec=spec, vw=vw)
            if shadow == (a, s):
                _e_write(d, a, s, jid + 2000, L, n_ep=40, run_done=False)

def _e_score(d, csvpath):
    saved = GATE_FAIL[:]
    del GATE_FAIL[:]
    buf = io.StringIO()
    try:
        with contextlib.redirect_stdout(buf):
            rc, fin = score(d, csvpath, "")
    except Exception as ex:
        return None, "EXCEPTION %s: %s" % (type(ex).__name__, ex), buf.getvalue()
    finally:
        GATE_FAIL[:] = saved
    return rc, fin, buf.getvalue()

def _spec_from_scorer(fname, key):
    p = os.path.join(HERE, fname)
    if not os.path.exists(p):
        return None
    import ast
    m = re.search(r"^SPEC = (\{.*?^\})", open(p).read(), re.S | re.M)
    if not m:
        return None
    return ast.literal_eval(m.group(1)).get(key)

def main():
    ap = argparse.ArgumentParser(description="cvt3 registered scorer.  Documented invocation: "
                                             "python3 analysis/cVT3_downcoalition_score.py <runsdir>")
    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--runsdir", dest="runsdir_kw", default="")
    ap.add_argument("--csv", default=DEFAULT_CSV)
    ap.add_argument("--manifest", default="", help="defaults to <runsdir>/cvt3/PARTITION-MANIFEST.txt")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    runsdir = a.runsdir or a.runsdir_kw
    if a.selftest:
        return selftest(a.csv, runsdir or ".")
    if not runsdir:
        ap.error("a runsdir is required: python3 %s <runsdir>" % os.path.basename(__file__))
    rc, _fin = score(runsdir, a.csv, a.manifest)
    return rc


if __name__ == "__main__":
