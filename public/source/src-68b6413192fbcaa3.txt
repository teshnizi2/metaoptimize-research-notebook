#
# WHAT IT IS.  A second, independent implementation of every line the REGISTERED scorer
# analysis/cST1_carrier_contrast_score.py prints on the landed batch.  It imports NO repo module
# (no cCV0_carrier_vote_core, no corpus_exclusions, no scorer): stdlib only.  It rebuilds the model
# from the ResNet18 architecture, rebuilds the partition manifest byte for byte, re-derives every
# gate including the decomposition counts, re-derives the levels from the RAW `.out` files and the
# corpus sigma / anchors through the exclusions TSV, and reproduces the scorer's FINAL line.
#
# THE SCORER'S FINAL ON THIS BATCH IS A GATE, NOT A BRANCH:
#     FINAL: UNRESOLVED-DECOMPOSITION | G-DECOMP k01-s112
# so the scorer printed NO nomination, level or branch line.  This parser therefore has two parts:
#   (1) a REPLICA of what the scorer did print (the gate ledger and that FINAL line), and
#   (2) a DESCRIPTIVE part -- the nomination statistics and the levels -- which is NOT licensed by
#       any registered branch, is labelled DESCRIPTIVE / UNSURE, and is printed because the gate
#       that fired is a TOLERANCE defect, not a harness failure.  The evidence for that claim is
#       re-derived here and is the parser's own (the scorer makes no such claim):
#       on EVERY unclamped coordinate of all 9 runs the posted beta equals, bit for bit,
#       float32(beta_pre - float32(ms) * sign(b2*mom_pre + (1-b2)*z_agg)).  The gate's test is
#       |-(beta - beta_pre)/ms - sign(L)| <= 1e-3, an ABSOLUTE bar on a quotient whose float32
#       quantisation noise is ulp(beta)/ms; at |beta| ~ 13.8 that is ~9.5e-7 / 3e-4 ~ 3.2e-3,
#       so at ms 3e-4 the bar cannot be met however correct the harness is.  RULE 16: the scorer is
#       NOT edited here; the defect is reported and a frozen successor must be registered before any
#       cst1 branch word is read.
#
#     python3 analysis/cst1_attack_indep.py <runsdir> [--csv <all_runs.csv>] [--tsv <CORPUS-EXCLUSIONS.tsv>]
#
# Exit 0 iff VIOLATIONS 0.  Host paths appear only on the HOSTPATH lines at the end, so the output is
# byte-identical on cluster_b and on the Mac apart from those lines.
#
# THE CORPUS IT USES: results/all_runs.csv and results/CORPUS-EXCLUSIONS.tsv at repo commit 1cb52f7
# ("ingest cvt8 + cvt9"), sha256 1801245410857d8a9e66a68e6ec633ccf6cf9e92d0efcec8cf3d7e2077b94b0a
# (3,127 rows) and ff9533348f9f2f0db04c66e7f6cbe8f191cd05bc59cb1c7dc359976db7cbe191.

def build_model(n_classes):
    out = [("conv1.weight", 64 * 3 * 3 * 3), ("bn1.weight", 64), ("bn1.bias", 64)]
    inp = 64
    for stage, planes in enumerate((64, 128, 256, 512), start=1):
        for blk in range(2):
            pre = "layer%d.%d." % (stage, blk)
            out.append((pre + "conv1.weight", planes * inp * 3 * 3))
            out.append((pre + "bn1.weight", planes))
            out.append((pre + "bn1.bias", planes))
            out.append((pre + "conv2.weight", planes * planes * 3 * 3))
            out.append((pre + "bn2.weight", planes))
            out.append((pre + "bn2.bias", planes))
            if inp != planes:
                out.append((pre + "shortcut.0.weight", planes * inp))
                out.append((pre + "shortcut.1.weight", planes))
                out.append((pre + "shortcut.1.bias", planes))
            inp = planes
    out.append(("linear.weight", 512 * n_classes))
    out.append(("linear.bias", n_classes))
    return out

def tclass(name):
    if name.startswith("linear."):
        return "linear"
    if name.endswith(".bias"):
        return "bn_bias"
    if ".bn" in name or ".shortcut.1." in name or name.startswith("bn1."):
        return "bn_scale"
    return "conv"

def parse_sets(spec):
    """PATCH_NAMESETS grammar: `scalar`, `layerwise`, or `sets:<g>/<g>/...` where a group is a comma list of
    1-based indices, `a-b` ranges, or tensor names.  -> list of sorted 1-based index lists."""
    if spec == "scalar":
        return [list(range(1, len(NAMES) + 1))]
    if spec == "layerwise":
        return [[i] for i in range(1, len(NAMES) + 1)]
    assert spec.startswith("sets:")
    groups = []
    for g in spec[5:].split("/"):
        idx = []
        for tok in g.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(NAMES.index(tok) + 1)
        groups.append(sorted(idx))
    return groups

    return "\n".join(L) + "\n"

def sgn(x):
    return 1 if x > 0 else (-1 if x < 0 else 0)


def fmt(x, nd=4):
    return "n/a" if x is None else ("%.*f" % (nd, x))


def med(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return None
    return v[n // 2] if n % 2 else (v[n // 2 - 1] + v[n // 2]) / 2.0


def avg(v):
    v = list(v)
    return sum(v) / len(v) if v else None


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def f32(x):
    return struct.unpack("f", struct.pack("f", x))[0]

def find_out(runsdir, arm, seed):
    pat = re.compile(r"^%s-%s-s%d-(\d+)\.out$" % (BATCH, arm, seed))
    hits = []
    for d in (runsdir, os.path.join(runsdir, BATCH)):
        if os.path.isdir(d):
            for fn in sorted(os.listdir(d)):
                m = pat.match(fn)
                if m:
                    hits.append((int(m.group(1)), os.path.join(d, fn)))
    return sorted(hits)


def read_out(path):
    txt = open(path, errors="replace").read()
    lines = txt.splitlines()
    te, tr = {}, {}
    for e, a, b in EP_RE.findall(txt):
        tr[int(e)] = float(a)
        te[int(e)] = float(b)
    r = {"path": path, "test": te, "train": tr,
         "args": [l for l in lines if l.startswith("ARGS:")],
         "env": [l for l in lines if l.startswith("ENV:")],
         "pt": [l for l in lines if l.startswith("PROBE_TENSOR:")],
         "foreign": [l for l in lines if l.startswith(FOREIGN_PREFIXES)],
         "done": any(l.strip() == "RUN_DONE" for l in lines),
         "tb": "Traceback (most recent call last)" in txt}
    tail = [te.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    trl = [tr.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    r["p5"] = sum(tail) / 5.0 if all(isinstance(x, float) and math.isfinite(x) for x in tail) else None
    r["t5"] = sum(trl) / 5.0 if all(isinstance(x, float) and math.isfinite(x) for x in trl) else None
    r["complete"] = sorted(te) == list(range(EPOCHS)) and r["done"] and not r["tb"] and r["p5"] is not None
    return r


def read_probe(runsdir, arm, seed):
    d = os.path.join(runsdir, BATCH, "probe_%s-%s-s%d" % (BATCH, arm, seed))
    recs, bad = [], 0
    for ln in open(os.path.join(d, "probe.jsonl"), errors="replace"):
        ln = ln.strip()

    return {"dir": d, "recs": recs, "bad": bad,
            "hdr": json.load(open(os.path.join(d, "probe_tensor.json")))}

def rec_L(r):
    b2 = float(r.get("pt_b2", B2))
    return [b2 * m + (1.0 - b2) * z for m, z in zip(r["m_tensor"], r["z_tensor"])], b2


def shared_s(r, L, b2):
    za, ma = r["z_agg"][0], r["mom_pre"][0]
    if len(za) == 1 and len(ma) == 1:
        return sgn(b2 * ma[0] + (1.0 - b2) * za[0])
    return sgn(sum(L))


def fixes(L, idx, s):
    tot = sum(L[i] for i in idx)
    rest = sum(abs(L[j]) for j in range(len(L)) if j not in idx)
    return sgn(tot) == s and abs(tot) > rest


def carrying(L, s):
    order = sorted(range(len(L)), key=lambda i: -abs(L[i]))
    rem = sum(abs(x) for x in L)
    part = 0.0
    for k, i in enumerate(order, 1):
        part += L[i]
        rem -= abs(L[i])
        if sgn(part) == s and abs(part) > rem:
            return order[:k]
    return order


def vote_stats(recs):
    cs = set(C_IDX0)
    n = indet = dom = t3 = dt3 = det = dis = careq = down = pinned = dompin = 0
    shares, cls = [], {c: [] for c in CLASSES}
    modal, argmax = Counter(), Counter()
    dec = [[0, 0] for _ in range(10)]
    for r in recs:
        L, b2 = rec_L(r)
        s = shared_s(r, L, b2)
        if s == 0:
            indet += 1
            continue
        n += 1
        order = sorted(range(len(L)), key=lambda i: -abs(L[i]))
        top3 = order[:3]
        d = fixes(L, C_IDX0, s)
        dom += d
        t3 += set(top3) == cs
        dt3 += fixes(L, top3, s)
        modal[tuple(sorted(i + 1 for i in top3))] += 1
        argmax[order[0] + 1] += 1
        tot = sum(abs(x) for x in L)
        shares.append(sum(abs(L[i]) for i in C_IDX0) / tot if tot > 0 else 0.0)
        if tot > 0:
            for c in CLASSES:
                cls[c].append(sum(abs(L[i]) for i in range(len(L)) if CLASS_OF[i] == c) / tot)
        st = r.get("step")
        if isinstance(st, int) and 0 <= st < META_STEPS:
            dd = dec[st * 10 // META_STEPS]
            dd[0] += 1
            dd[1] += d
        if len(r.get("beta", [])) == 1 and (r.get("n_at_lo") or 0) >= 1:
            pinned += 1
            dompin += d
        maj = sgn(sum(1 for x in L if x > 0) - sum(1 for x in L if x < 0))
        if maj:
            det += 1
            if maj != s:
                dis += 1
                careq += set(carrying(L, s)) == cs
                down += s > 0
    return {"n": n, "indet": indet, "DOM_C": dom / n if n else None, "TOP3_C": t3 / n if n else None,
            "DOM_TOP3": dt3 / n if n else None, "SHARE_C": med(shares) if shares else None,
            "R_T": dis / det if det else None, "n_dis": dis, "CAR_EQ": careq / dis if dis else None,
            "DOWN": down / dis if dis else None, "PINNED_FRAC": pinned / n if n else None,
            "DOM_C_PINNED": dompin / pinned if pinned else None,
            "MODAL_TOP3": modal.most_common(3), "ARGMAX": argmax.most_common(5),
            "CLASS_SHARE": dict((c, med(v) if v else None) for c, v in cls.items()),
            "DECILE": [(d[1] / d[0]) if d[0] else None for d in dec]}

def decomposition(recs, groups1, ms):
    g0 = [[i - 1 for i in g] for g in groups1]
    n = bshape = bz = bm = nsign = bsign = clamped = bshared = exact32 = 0
    wz = wm = worst_q = 0.0
    for r in recs:
        n += 1
        z, m = r["z_tensor"], r["m_tensor"]
        za, ma = r["z_agg"][0], r["mom_pre"][0]
        bpre, bpost = r["beta_pre"][0], r["beta"]
        b2 = float(r.get("pt_b2", B2))
        if not (len(z) == len(m) == 62 and len(za) == len(ma) == len(bpre) == len(bpost) == len(g0)):
            bshape += 1
            continue
        sz = sum(abs(v) for v in z) + 1e-30
        sm = sum(abs(v) for v in m) + 1e-30
        dz = max(abs(sum(z[i] for i in g) - za[k]) for k, g in enumerate(g0)) / sz
        dm = max(abs(sum(m[i] for i in g) - ma[k]) for k, g in enumerate(g0)) / sm
        wz, wm = max(wz, dz), max(wm, dm)
        bz += dz > TOL_Z
        bm += dm > TOL_M
        if len(g0) == 1:
            L = [b2 * mi + (1 - b2) * zi for mi, zi in zip(m, z)]
            Lh = b2 * ma[0] + (1 - b2) * za[0]
            if abs(sum(L) - Lh) > 1e-4 * (sum(abs(x) for x in L) + 1e-30) and sgn(sum(L)) != sgn(Lh):
                bshared += 1
        for k in range(len(g0)):
            if not (BETA_LO + 1e-9 < bpost[k] < BETA_HI - 1e-9):
                clamped += 1
                continue
            nsign += 1
            Lh = b2 * ma[k] + (1 - b2) * za[k]
            q = -(bpost[k] - bpre[k]) / ms
            worst_q = max(worst_q, abs(q - sgn(Lh)))
            bsign += abs(q - sgn(Lh)) > 1e-3
            exact32 += (f32(f32(bpre[k]) - f32(f32(ms) * sgn(Lh))) == bpost[k])
    return {"n": n, "bad_shape": bshape, "bad_z": bz, "bad_m": bm, "worst_z": wz, "worst_m": wm,
            "n_sign": nsign, "bad_sign": bsign, "clamped": clamped, "bad_shared": bshared,
            "exact32": exact32, "worst_q": worst_q}


def decomp_ok(d):
    n = max(d["n"], 1)
    return (d["n"] > 0 and d["bad_shape"] == 0 and d["bad_z"] / n <= DECOMP_FAIL_MAX
            and d["bad_m"] / n <= DECOMP_FAIL_MAX and d["n_sign"] > 0
            and d["bad_sign"] / max(d["n_sign"], 1) <= SIGN_FAIL_MAX and d["bad_shared"] / n <= DECOMP_FAIL_MAX)

def load_corpus(csv_path, tsv_path):
    rows = list(csv.DictReader(open(csv_path)))
    lines = [l.rstrip("\n") for l in open(tsv_path) if l.strip() and not l.startswith("#")]
    head = lines[0].split("\t")
    excl = set()
    for ln in lines[1:]:
        f = dict(zip(head, ln.split("\t")))
        excl.add((f["run"], f["job_id"]))
    return rows, [r for r in rows if (r.get("run"), r.get("job_id")) not in excl], excl


def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0" and r.get("complete") == "1"
            and (r.get("plateau5") or "").strip() != "")


def std_cell(r, net, dset, ms, a0):
    return (usable(r) and r.get("network") == net and r.get("dataset") == dset
            and r.get("epochs_requested") == "100" and r.get("augment") == "1"
            and r.get("beta_clip") == "-15:-2.3026" and r.get("meta_stepsize") == ms
            and r.get("alpha0") == a0 and r.get("batch_size") == "100")


def cell_rows(rows, net, dset, ms, a0, gran):
    return [r for r in rows if std_cell(r, net, dset, ms, a0) and r.get("granularity") == gran
            and r.get("hier", "") == "" and r.get("base") == "SGDm" and r.get("meta") == "Lion"
            and r.get("gamma") == "1"]


def pooled_sigma(rows, net, dset, ms="1e-3", a0="1e-6"):
    cells = {}
    for r in rows:
        if not std_cell(r, net, dset, ms, a0):
            continue
        cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc = 0.0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = sum(v) / len(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
    return (math.sqrt(ss / df) if df else None), df, nc

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("runsdir")
    here = os.path.dirname(os.path.abspath(__file__))
    ap.add_argument("--csv", default=os.path.join(os.path.dirname(here), "results", "all_runs.csv"))
    ap.add_argument("--tsv", default=os.path.join(os.path.dirname(here), "results", "CORPUS-EXCLUSIONS.tsv"))
    a = ap.parse_args()
    rd = a.runsdir

    P("=" * 100)
    P("cst1_attack_indep -- INDEPENDENT re-derivation of CORRECTIONS 256's scorer output (no repo import)")
    P("=" * 100)

    P("\n[FILES] sha256 of every file read (host-independent)")
    outs, probes = {}, {}
    for arm in ARMS:
        for s in SEEDS:
            hits = find_out(rd, arm, s)
            chk(len(hits) == 1, "one .out for %s-s%d" % (arm, s), "%d" % len(hits))
            job, path = hits[-1]
            outs[(arm, s)] = read_out(path)
            outs[(arm, s)]["job"] = job
            probes[(arm, s)] = read_probe(rd, arm, s)
            P("  %-14s job %d  out %s" % ("%s-s%d" % (arm, s), job, sha256(path)))
            P("  %-14s %11s probe.jsonl %s" % ("", "", sha256(os.path.join(probes[(arm, s)]["dir"], "probe.jsonl"))))
    for f in ("PROVENANCE.txt", "PARTITION-MANIFEST.txt"):
        P("  %-14s %s %s" % ("batch", f, sha256(os.path.join(rd, BATCH, f))))
    csv_sha, tsv_sha = sha256(a.csv), sha256(a.tsv)
    P("  %-14s all_runs.csv %s" % ("corpus", csv_sha))
    P("  %-14s CORPUS-EXCLUSIONS.tsv %s" % ("corpus", tsv_sha))
    P("  corpus commit %s (the commit that last wrote both files; the corpus cst1 was registered against)"
      % CORPUS_COMMIT)
    chk(csv_sha == REG_CSV_SHA, "all_runs.csv sha == the scorer's REG_CORPUS_SHA")
    chk(tsv_sha == REG_TSV_SHA, "CORPUS-EXCLUSIONS.tsv sha == the registered one")

    P("\n[MODEL] names and numels built from the ResNet18_c100 architecture, then checked against each header")
    chk(len(MODEL) == 62 and sum(NUMELS) == 11220132, "62 tensors, %d parameters" % sum(NUMELS))
    chk(tuple(NAMES[i - 1] for i in C_IDX1) == CARRIER_NAMES, "C = {50, 53, 59} are the three carriers by name")
    chk(tuple(NAMES[i - 1] for i in ISOLATED["CTL"]) == ("layer4.0.bn1.weight", "layer4.0.bn1.bias",
                                                         "layer4.1.bn1.weight"),
        "CTL = {47, 48, 56} is the matched non-carrier layer4 triple (one member is a bias)")
    giso, gctl = parse_sets(SPEC["ISO"]), parse_sets(SPEC["CTL"])
    chk([len(g) for g in giso] == [59, 3] and sorted(giso[1]) == list(C_IDX1),
        "ISO parses to [59, 3] with the carriers isolated")
    chk([len(g) for g in gctl] == [59, 3] and sorted(gctl[1]) == [47, 48, 56],
        "CTL parses to [59, 3] with the control triple isolated")
    chk(sum(NUMELS[i - 1] for i in C_IDX1) == 1536 and sum(NUMELS[i - 1] for i in (47, 48, 56)) == 1536,
        "ISO and CTL isolate 1,536 parameters each (3 x 512) -- count-matched")
    man = open(os.path.join(rd, BATCH, "PARTITION-MANIFEST.txt")).read()
    mt = manifest_text()
    chk(man == mt, "PARTITION-MANIFEST.txt rebuilt byte for byte (%d bytes)" % len(mt), "%d on disk" % len(man))
    chk(len(mt) == EXPECT_MANIFEST_BYTES, "manifest length == the registered %d" % EXPECT_MANIFEST_BYTES)
    for arm in ARMS:
        for s in SEEDS:
            chk(tuple(probes[(arm, s)]["hdr"].get("param_numels", ())) == NUMELS,
                "header numels %s-s%d == the built model" % (arm, s))

    P("\n[PROVENANCE] runs/%s/PROVENANCE.txt against the literals CORRECTIONS 256.10 records" % BATCH)
    prov = {}
    for ln in open(os.path.join(rd, BATCH, "PROVENANCE.txt")):
        p = ln.split()
        if len(p) >= 2:
            prov.setdefault(p[0], " ".join(p[1:]))
    for k, want in (("MODE", "submit"), ("HF_SHA256", HF_SHA), ("BUILD_NETWORK_SHA256", BUILD_NETWORK_SHA),
                    ("TRAIN_PY_SHA256", TRAIN_PY_SHA), ("RUNNER_SHA256", RUNNER_SHA), ("CORE_SHA256", CORE_SHA),
                    ("SCORER_SHA256", SCORER_SHA)):
        chk(prov.get(k) == want, "PROVENANCE %s" % k, "got %s" % prov.get(k))
    sc = os.path.join(here, "cST1_carrier_contrast_score.py")
    if os.path.isfile(sc):
        chk(sha256(sc) == SCORER_SHA, "the scorer file on disk is the registered one (sha)")

    P("\n[RUN LINES] ARGS / ENV / PROBE_TENSOR / foreign / records / coordinate count, from each run's own .out")
    for arm in ARMS:
        for s in SEEDS:
            r, pr, tag = outs[(arm, s)], probes[(arm, s)], "%s-s%d" % (arm, s)
            chk(r["complete"], "%s: RUN_DONE, epochs 0..99, no traceback" % tag)
            m = re.match(r"^(.*--save-directory )(\S+)( --run-name \S+)$", r["args"][0]) if len(r["args"]) == 1 else None
            ws = m.group(2)[:-len("/runs/" + BATCH)] if m and m.group(2).endswith("/runs/" + BATCH) else None
            chk(ws is not None and r["args"][0] == args_line(arm, s, ws), "%s: ARGS == the registered line" % tag)
            pdir = None
            if len(r["env"]) == 1:
                mm = re.search(r" PROBE_DIR=(\S+)", r["env"][0])
                pdir = mm.group(1) if mm else None
            chk(len(r["env"]) == 1 and re.sub(r" PROBE_DIR=\S+", "", r["env"][0]) == "ENV: " + ENV_FIXED
                and pdir is not None and pdir.endswith("/runs/%s/probe_%s-%s" % (BATCH, BATCH, tag)),
                "%s: ENV == the registered line, PROBE_DIR is the run's own" % tag)
            mp = PT_RE.match(r["pt"][0]) if len(r["pt"]) == 1 else None
            chk(mp is not None and mp.group(1) == PT_TYPE[arm] and mp.group(2) == pdir,
                "%s: one PROBE_TENSOR line, type=%s, dir == PROBE_DIR" % (tag, PT_TYPE[arm]))
            chk(not r["foreign"], "%s: no intervention line (the live tree ran)" % tag)
            chk(pr["bad"] == 0 and [x.get("step") for x in pr["recs"]] == list(range(0, META_STEPS, PROBE_EVERY)),
                "%s: 500 records at steps 0..49,900" % tag)
            nc = set(len(x.get("beta", [])) for x in pr["recs"])
            nz = set(len((x.get("z_agg") or [[]])[0]) for x in pr["recs"])
            chk(nc == {N_COORDS[arm]} and nz == {N_COORDS[arm]},
                "%s: every record carries %d coordinate(s) -- the BROKEN-SPEC null is excluded" % (tag, N_COORDS[arm]))

    # ---- the gate that fired, re-derived, and what it actually means ------------------------------
    P("\n[G-DECOMP] the gate the registered scorer failed on, re-derived count for count")
    dec = {}
    for arm in ARMS:
        for s in SEEDS:
            d = decomposition(probes[(arm, s)]["recs"], parse_sets(SPEC[arm]), MS)
            dec[(arm, s)] = d
            P("  %-10s shape %d z %d m %d (worst %.1e / %.1e) sign %d/%d clamped %d shared %d"
              % ("%s-s%d" % (arm, s), d["bad_shape"], d["bad_z"], d["bad_m"], d["worst_z"], d["worst_m"],
                 d["bad_sign"], d["n_sign"], d["clamped"], d["bad_shared"]))
            got = (d["bad_shape"], d["bad_z"], d["bad_m"], d["bad_sign"], d["n_sign"], d["clamped"], d["bad_shared"])
            chk(got == EXPECT_DECOMP[(arm, s)], "%s-s%d: counts == the registered scorer's" % (arm, s), str(got))
    P("\n[WHY IT FIRED -- this parser's own evidence, which the scorer does not compute]")
    tot_sign = sum(dec[k]["n_sign"] for k in dec)
    tot_exact = sum(dec[k]["exact32"] for k in dec)
    worst_q = max(dec[k]["worst_q"] for k in dec)
    P("  unclamped coordinates over the 9 runs: %d;  posted beta == float32(beta_pre - float32(ms)*sign(L)): %d"
      % (tot_sign, tot_exact))
    P("  worst |-(beta - beta_pre)/ms - sign(L)| over all of them: %.3e   (the gate's bar is 1.000e-03)" % worst_q)
    ulp = 2.0 ** -23 * 2.0 ** int(math.floor(math.log2(13.8)))
    P("  one float32 ulp at |beta| ~ 13.8 is %.3e; ulp / ms = %.3e at ms %g, and %.3e at ms 1e-3"
      % (ulp, ulp / MS, MS, ulp / 1e-3))
    chk(tot_exact == tot_sign,
        "every applied step is EXACTLY float32(beta_pre - ms*sign(L)): the harness applied the registered update")
    chk(worst_q < 2.0 * ulp / MS, "the whole residual is float32 quantisation of beta, not a wrong step")
    P("  READING: the failure is the gate's ABSOLUTE 1e-3 bar on a quotient whose quantisation noise scales as")
    P("  ulp(beta)/ms.  It is met at ms 1e-3 (cct1: 0 failures on the same code) and cannot be met at ms 3e-4.")
    P("  RULE 16: the registered scorer is NOT edited here.  Its FINAL stands; a frozen successor with an")
    P("  ms-aware tolerance must be registered before any cst1 BRANCH word is read.")

    # ---- REPLICA of what the scorer printed ------------------------------------------------------
    final = "FINAL: UNRESOLVED-DECOMPOSITION | G-DECOMP %s" % "k01-s112"
    P("\n" + "=" * 100)
    P("REPLICA of the registered scorer's FINAL line")
    P("=" * 100)
    P(final)
    chk(final == EXPECT_FINAL, "FINAL re-derived == the registered scorer's FINAL")
    chk(not decomp_ok(dec[("k01", 112)]), "the first failing run is k01-s112 (the ledger's first FAIL)")

    # ---- DESCRIPTIVE: nothing below is licensed by a registered branch ----------------------------
    P("\n" + "=" * 100)
    P("DESCRIPTIVE / UNSURE -- NOT a registered reading: the scorer never reached its NOMINATION, LEVELS or")
    P("BRANCH section, so no cst1 licence sentence exists.  The numbers below are this parser's own.")
    P("=" * 100)
    P("\n[DESCRIPTIVE NOMINATION] k01's 1,500 records, the 256.3 definitions (bars DOM_C 0.50 / 0.10, TOP3 0.50)")
    for s in SEEDS:
        describe("k01-s%d" % s, vote_stats(probes[("k01", s)]["recs"]))
    nst = vote_stats([x for s in SEEDS for x in probes[("k01", s)]["recs"]])
    describe("k01 POOLED", nst)
    P("  calibration at ms 1e-3 (registration): scalar DOM_C 0.784-0.820 pooled %.4f; layerwise 0.0000"
      % CALIB_SCALAR_POOLED)
    for a_ in ("ISO", "CTL"):
        describe("%s POOLED" % a_, vote_stats([x for s in SEEDS for x in probes[(a_, s)]["recs"]]))
    P("\n[DESCRIPTIVE LEVELS] plateau5 = mean TEST of epochs 95..99 from the RAW .out; TRAIN beside it")
    lev, rngs, trn = {}, {}, {}
    for arm in ARMS:
        v = [outs[(arm, s)]["p5"] for s in SEEDS]
        t = [outs[(arm, s)]["t5"] for s in SEEDS]
        lev[arm], rngs[arm], trn[arm] = avg(v), max(v) - min(v), avg(t)
        P("  %-4s test %s  mean %.4f  range %.3f  | train %s mean %.4f"
          % (arm, " ".join("%.3f" % x for x in v), lev[arm], rngs[arm], " ".join("%.3f" % x for x in t), trn[arm]))
    d_iso, d_ctl, pp = lev["ISO"] - lev["k01"], lev["CTL"] - lev["k01"], lev["ISO"] - lev["CTL"]
    for nm, d in (("D_ISO = ISO - k01", d_iso), ("D_CTL = CTL - k01", d_ctl), ("P = ISO - CTL", pp)):
        P("  %-20s %+8.4f pp = %+8.2f SE" % (nm, d, d / SE_PRIOR))
    ins = [outs[(arm, s)]["p5"] - lev[arm] for arm in ARMS for s in SEEDS]
    sig_in = math.sqrt(sum(x * x for x in ins) / (len(ins) - len(ARMS)))
    P("  in-batch sigma %.4f (df %d); SIGMA_PRIOR %.6f frozen; k01 vs K01_MAX %.1f: %s"
      % (sig_in, len(ins) - len(ARMS), SIGMA_PRIOR, K01_MAX, "below" if lev["k01"] <= K01_MAX else "ABOVE"))
    P("\n[DESCRIPTIVE BETA] per arm, per seed, per group: peak, terminal, share of records at the -15 floor")
    for arm in ARMS:
        for s in SEEDS:
            recs = probes[(arm, s)]["recs"]
            for c in range(N_COORDS[arm]):
                b = [x["beta"][c] for x in recs]
                pk = max(range(len(b)), key=lambda i: b[i])
                P("  beta %s-s%d g%d: peak %.3f at step %d; terminal %.3f; share at floor %.3f"
                  % (arm, s, c, b[pk], recs[pk]["step"], b[-1], sum(1 for x in b if x <= BETA_LO + 1e-6) / len(b)))

    P("\n[DESCRIPTIVE, WHAT THE FROZEN BARS WOULD SAY -- UNSURE, NOT A LICENCE]")
    P("  This is arithmetic on frozen literals, NOT a registered reading: the registered scorer stopped at a gate")
    P("  and never ran decide(), so `cst1` HAS no branch word and no licence sentence until a successor scorer is")
    P("  registered.  Reported only so the near-bar risk is visible before that successor is written.")
    nom = ("NOMINATED" if (nst["DOM_C"] >= DOM_HI and nst["TOP3_C"] >= TOP3_BAR)
           else "NOT-NOMINATED" if nst["DOM_C"] <= DOM_LO else "NOMINATION-PARTIAL")
    isow = ("ISO-RESCUES" if d_iso >= RESCUE_BAR else "ISO-NO-RESCUE" if d_iso <= NULL_BAR else "ISO-PARTIAL")
    ctlw = ("CTL-RESCUES" if d_ctl >= RESCUE_BAR else "CTL-NULL" if d_ctl <= NULL_BAR else "CTL-PARTIAL")
    P("  DOM_C %.4f vs the 0.50 bar: %+.4f      TOP3_C %.4f vs 0.50: %+.4f      -> %s"
      % (nst["DOM_C"], nst["DOM_C"] - DOM_HI, nst["TOP3_C"], nst["TOP3_C"] - TOP3_BAR, nom))
    P("  D_ISO %+.4f vs 20.0 -> %s;  D_CTL %+.4f vs 5.0 -> %s;  k01 %.4f vs K01_MAX 34.0 -> %s"
      % (d_iso, isow, d_ctl, ctlw, lev["k01"], "collapse gate not tripped" if lev["k01"] <= K01_MAX
         else "SCALAR-NOT-COLLAPSED"))
    P("  the three words would read %s+%s+%s -- BOTH nomination bars are missed by less than 0.03, so the word is"
      % (nom, isow, ctlw))
    P("  bar-sensitive and the DOM_C decile profile (0.00 x5 then ~1.00) says it is a TIME-COURSE, not a level.")

    P("\n[CORPUS] sigma and the anchors, re-derived through the exclusions TSV")
    raw, kept, excl = load_corpus(a.csv, a.tsv)
    P("  rows %d -> %d after %d exclusion keys" % (len(raw), len(kept), len(excl)))
    chk(len(raw) == REG_CORPUS_ROWS, "the corpus has the registered %d rows" % REG_CORPUS_ROWS)
    sig, df, ncell = pooled_sigma(kept, NET, DSET)
    P("  SIGMA_R18ALL filtered %r (df %d, %d cells)" % (sig, df, ncell))
    chk(abs(sig - SIGMA_PRIOR) < 1e-12 and df == 221 and ncell == 46,
        "SIGMA_R18ALL == SIGMA_PRIOR (df 221, 46 cells)")
    nsig, ndf, nnc = pooled_sigma(raw, NET, DSET)
    P("  the same WITHOUT the exclusions (the OPERATIONS 36 trap): %.6f (df %d, %d cells)" % (nsig, ndf, nnc))
    chk(nsig > sig, "the naive sigma is larger than the filtered one")
    anch = {"k01@3e-4 (cru1 n3)": cell_rows(kept, NET, DSET, "3e-4", A0, "scalar"),
            "layerwise@3e-4 (cru1 n3)": cell_rows(kept, NET, DSET, "3e-4", A0, "layerwise"),
            "k01@1e-3 (n44)": cell_rows(kept, NET, DSET, "1e-3", A0, "scalar")}
    for k, rows in anch.items():
        m = avg([float(r["plateau5"]) for r in rows])
        want, wn = EXPECT_ANCHOR[k]
        P("  anchor %-27s n=%d mean %.4f" % (k, len(rows), m))
        chk(len(rows) == wn and abs(round(m, 4) - want) < 5e-5, "anchor %s == the registered literal" % k)
    chk(not any(r["run"].startswith(BATCH + "-") for r in raw), "no cst1- row in the corpus (freshness)")

    P("\n" + "=" * 100)
    P("VIOLATIONS %d" % len(VIOL))
    for v in VIOL:
        P("  ! %s" % v)
    sys.stdout.write("\n".join(OUT) + "\n")
    print("HOSTPATH runsdir  %s" % os.path.abspath(rd))
    print("HOSTPATH corpus   %s" % os.path.abspath(a.csv))
    return 0 if not VIOL else 1


if __name__ == "__main__":
    sys.exit(main())
