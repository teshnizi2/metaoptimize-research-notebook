"""187's OPERATIVE statistic on VGG: on the scalar arm's PINNED records, does
removing a candidate set FLIP the sign of the remainder sum of L_i?
L_i = 0.9*m_tensor_i + 0.1*z_tensor_i.  Independent of every registered scorer."""
import json, os, re, sys, itertools, collections
R = sys.argv[1] if len(sys.argv) > 1 else "/cluster/account_b/metaopt/runs"
man = open(R + "/cvg1-PARTITION-MANIFEST.txt").read()
names = {int(i): n for i, n, s in re.findall(r"^TENSOR (\d+) (\S+) (\d+)$", man, re.M)}

def load(arm, seed):
    p = "%s/cvg1/probe_cvg1-%s-%s/probe.jsonl" % (R, arm, seed)
    recs = [json.loads(l) for l in open(p)]
    return recs

CANDS = {
    "{23} bn8.weight ONLY":        [23],
    "{20,23} bn7+bn8.weight":      [20, 23],
    "{14,17,20,23} four 512 BN":   [14, 17, 20, 23],
    "{14,17,20} the other three":  [14, 17, 20],
    "{22} conv8.weight":           [22],
    "{25} linear.weight":          [25],
}
print("=== SCALAR ARM (k01), PINNED records only ===")
print("  For each candidate set S:  TOTAL = sign(sum_i L_i) [what the single beta follows];")
print("  REM = sign(sum_{i not in S} L_i).  'FLIPS' = fraction of pinned records where REM != TOTAL.")
for seed in ("s31", "s32", "s33"):
    recs = load("k01", seed)
    pin = [r for r in recs if r["beta"][0] <= -14.999]
    tot_down = 0
    for r in pin:
        L = [0.9 * r["m_tensor"][i] + 0.1 * r["z_tensor"][i] for i in range(26)]
        if sum(L) > 0: tot_down += 1
    print("\n  k01-%s: %d pinned records | sum_i L_i > 0 (beta DOWN) on %d/%d = %.4f"
          % (seed, len(pin), tot_down, len(pin), tot_down / len(pin)))
    for label, S in CANDS.items():
        flips = 0; domin = 0
        for r in pin:
            L = [0.9 * r["m_tensor"][i] + 0.1 * r["z_tensor"][i] for i in range(26)]
            tot = sum(L)
            rem = sum(L[i - 1] for i in range(1, 27) if i not in S)
            sset = sum(L[i - 1] for i in S)
            if (rem > 0) != (tot > 0): flips += 1
            if abs(sset) > abs(rem) and (sset > 0) != (rem > 0): domin += 1
        print("     %-28s remainder-sign FLIPS on %.4f | set DOMINATES remainder on %.4f"
              % (label, flips / len(pin), domin / len(pin)))
