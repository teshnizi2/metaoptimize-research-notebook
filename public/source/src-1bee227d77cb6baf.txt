#!/bin/bash
# =============================================================================
# cVT7_stage_harness.sh -- BUILD OR VERIFY THE ISOLATED `cvt7` HARNESS TREE.
#   (CORRECTIONS 243.)  Touches NO existing file.  Submits nothing.
#
# THE BASE IS THE ResNet HARNESS THE ISO BATCHES RAN FROM.  ciso1 (185), cdep1 (188) and ciso2 (190) ran from the
# LIVE shared tree $WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10 with the runner jobs/run_cifar.sh;
# their PROVENANCE.txt files record HF_SHA256 4732b74a... (= HF_LIVE_SHA below), and ciso2's also TRAIN_PY_SHA256
# 3fea309e...  cvt7 needs PATCH_GROUPHOLD in HF.py; the live tree must stay byte-untouched, and cvt1's / cvt4's
# trees carry PlainNet's build_network.py.  So cvt7 gets its own tree built FROM THE LIVE TREE:
#
#   harness_cvt7/cifar10/*.py                every live *.py file, byte-identical, except Optimizers/HF.py
#                                            (build_network.py stays the LIVE one, c7998883...: NO PlainNet patch)
#   harness_cvt7/cifar10/Optimizers/HF.py.pre_probe_tensor   the live backup, byte-identical (as cvt1's tree)
#   harness_cvt7/cifar10/data  ->            SYMLINK to the live tree's data/ (no copy)
#   harness_cvt7/cifar10/Optimizers/HF.py.pre_voteweight     the LIVE HF.py bytes          (4732b74a...)
#   harness_cvt7/cifar10/Optimizers/HF.py.pre_betahold       + PATCH_VOTEWEIGHT  == cvt1's HF.py (3f2b98e1...)
#   harness_cvt7/cifar10/Optimizers/HF.py.pre_grouphold      + PATCH_BETAHOLD    == cvt4's HF.py (84b345ad...)
#   harness_cvt7/cifar10/Optimizers/HF.py                    + PATCH_GROUPHOLD   (HF_POST_SHA)
#   jobs/run_cifar_cvt7.sh                   jobs/run_cifar.sh with EXACTLY the cd line changed
#   harness_cvt7/STAGE-MANIFEST.txt          sha256 of every staged file
#
# The two intermediate HF files are the SAME BYTES as cvt1's and cvt4's registered HF.py (pinned below and re-checked
# in verify mode), so every inertness proof of PATCH_VOTEWEIGHT (227) and PATCH_BETAHOLD (237) is about these bytes;
# tests/test_grouphold_realrun.py then proves the whole OFF tree BITWISE against the LIVE tree on ResNet18_c100.
#
# USAGE
#   bash bin/cVT7_stage_harness.sh            # VERIFY ONLY (exit 0 iff consistent)
#   bash bin/cVT7_stage_harness.sh --stage    # build it (refuses if it exists)
# =============================================================================
set -u
MODE=verify; [ "${1:-}" = "--stage" ] && MODE=stage
export METAOPT_WS=${METAOPT_WS:-/cluster/account_b/metaopt}
WS=$METAOPT_WS
REPO="$(cd "$(dirname "$0")/.." && pwd)"
LIVE=$WS/MetaOptimize/codes/Supervised_tasks/MetaOptimize/cifar10
HROOT=$WS/harness_cvt7
DEST=$HROOT/cifar10
RUNNER_SRC=$WS/jobs/run_cifar.sh
RUNNER=$WS/jobs/run_cifar_cvt7.sh
PATCH_VW=$REPO/patches/patch_voteweight.py
PATCH_BH=$REPO/patches/patch_betahold.py
PATCH_GH=$REPO/patches/patch_grouphold.py
LIVE_BN_SHA=c79988833c4399a06cc84d15c0830f54d86f59feeec345f47a2287c1cf2ba77d
HF_LIVE_SHA=4732b74aa3a10508896e92eccced5c89051c353fa8a01a3af21aab9aeda0cecd
HF_VW_SHA=3f2b98e1a279ceccd0de81019f63d999be1718782f7ce7f479496b7ff3fa4e52
HF_PRE_SHA=84b345adf6df8024bbe4c694fcbbaffc6211d1c7b7ae4b9ae933e353298a3d96
HF_POST_SHA=2396f2be8102724c120ea5271d2d84acf3835ffb9a5c7947fc282121cf2a588f
LIVE_TRAIN_SHA=3fea309e172609c1ee1be143d5f1847404426df31869a63c4cefbb1ebddfcab7
LIVE_RUNNER_SHA=a0d0a1b904e64ea5a3de8c0649b58454beff0322419c97ea9e6d77481cfd9f3a
RUNNER_SHA=805267e4cbcc64172d79824d553a2577ef7cdd55fb8f5e7731c0ea3251a04c56
CVT1_HF=$WS/harness_cvt1/cifar10/Optimizers/HF.py
CVT4_HF=$WS/harness_cvt4/cifar10/Optimizers/HF.py
FAIL=0; bad() { echo "!!! STAGE FAIL: $*"; FAIL=1; }
sha() { (sha256sum "$1" 2>/dev/null || shasum -a 256 "$1") | awk '{print $1}'; }

module load Python/3.10.4-GCCcore-11.3.0 >/dev/null 2>&1 || true
# shellcheck disable=SC1091
[ -f "$WS/envs/mo/bin/activate" ] && . "$WS/envs/mo/bin/activate" 2>/dev/null || true

live_py() { (cd "$LIVE" && find . -name '*.py' -not -path '*/__pycache__/*' -not -path './data/*' | sort); }

echo "cVT7_stage_harness: mode=$MODE  WS=$WS"
echo "  live tree   $LIVE  (the tree ciso1 / cdep1 / ciso2 ran from; read only)"
echo "  cvt7 tree   $DEST"
echo "  runner      $RUNNER  (from $RUNNER_SRC)"

if [ "$MODE" = stage ]; then
  [ -e "$HROOT" ] && { echo "!!! $HROOT already exists -- refusing to overwrite. Verify it instead."; exit 2; }
  [ -e "$RUNNER" ] && { echo "!!! $RUNNER already exists -- refusing to overwrite."; exit 2; }
  for p in "$PATCH_VW" "$PATCH_BH" "$PATCH_GH"; do [ -f "$p" ] || { echo "!!! $p is missing"; exit 2; }; done
  [ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] || { echo "!!! the LIVE HF.py sha != pinned $HF_LIVE_SHA"; exit 2; }
  [ "$(sha "$LIVE/build_network.py")" = "$LIVE_BN_SHA" ] || { echo "!!! the LIVE build_network.py sha != pinned $LIVE_BN_SHA"; exit 2; }
  [ "$(sha "$LIVE/train.py")" = "$LIVE_TRAIN_SHA" ] || { echo "!!! the LIVE train.py sha != pinned $LIVE_TRAIN_SHA"; exit 2; }
  [ "$(sha "$RUNNER_SRC")" = "$LIVE_RUNNER_SHA" ] || { echo "!!! run_cifar.sh sha != pinned $LIVE_RUNNER_SHA"; exit 2; }
  mkdir -p "$DEST" || exit 2
  for f in $(live_py); do
    mkdir -p "$DEST/$(dirname "$f")"; cp -p "$LIVE/$f" "$DEST/$f" || exit 2
  done
  cp -p "$LIVE/Optimizers/HF.py.pre_probe_tensor" "$DEST/Optimizers/HF.py.pre_probe_tensor" || exit 2
  ln -s "$LIVE/data" "$DEST/data" || exit 2
  HF_PATH="$DEST/Optimizers/HF.py" python3 "$PATCH_VW" || { echo "!!! patch_voteweight failed"; exit 2; }
  [ "$(sha "$DEST/Optimizers/HF.py")" = "$HF_VW_SHA" ] || { echo "!!! live + PATCH_VOTEWEIGHT != cvt1's HF.py $HF_VW_SHA"; exit 2; }
  HF_PATH="$DEST/Optimizers/HF.py" python3 "$PATCH_BH" || { echo "!!! patch_betahold failed"; exit 2; }
  [ "$(sha "$DEST/Optimizers/HF.py")" = "$HF_PRE_SHA" ] || { echo "!!! + PATCH_BETAHOLD != cvt4's HF.py $HF_PRE_SHA"; exit 2; }
  HF_PATH="$DEST/Optimizers/HF.py" python3 "$PATCH_GH" || { echo "!!! patch_grouphold failed"; exit 2; }
  sed "s#^cd $LIVE\$#cd $DEST#" "$RUNNER_SRC" > "$RUNNER" || exit 2
  chmod --reference="$RUNNER_SRC" "$RUNNER" 2>/dev/null || chmod 755 "$RUNNER"
  { echo "STAGED_UTC $(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo "LIVE $LIVE"; echo "DEST $DEST"
    echo "PATCH_VOTEWEIGHT_SHA256 $(sha "$PATCH_VW")"
    echo "PATCH_BETAHOLD_SHA256 $(sha "$PATCH_BH")"
    echo "PATCH_GROUPHOLD_SHA256 $(sha "$PATCH_GH")"
    for f in $(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort); do
      echo "FILE $(sha "$DEST/$f") $f"; done
    echo "RUNNER $(sha "$RUNNER") $RUNNER"
    echo "RUNNER_SRC $(sha "$RUNNER_SRC") $RUNNER_SRC"; } > "$HROOT/STAGE-MANIFEST.txt"
  echo "staged; now verifying"
fi

# ---------------------------------------------------------------- VERIFY ------
[ -d "$DEST" ] || { echo "!!! no cvt7 tree at $DEST -- run with --stage"; exit 2; }
N=0
for f in $(live_py); do
  [ "$f" = ./Optimizers/HF.py ] && continue
  if [ ! -f "$DEST/$f" ]; then bad "V-a $f missing from the cvt7 tree"; continue; fi
  [ "$(sha "$LIVE/$f")" = "$(sha "$DEST/$f")" ] && N=$((N+1)) || bad "V-a $f differs from the LIVE tree"
done
echo "V-a: $N live *.py files byte-identical to the LIVE tree (Optimizers/HF.py excepted)"
EXTRA=$(cd "$DEST" && find . -type f -not -path '*/__pycache__/*' -not -path './data/*' | sort | while read -r f; do
          case "$f" in
            (./Optimizers/HF.py.pre_probe_tensor|./Optimizers/HF.py.pre_voteweight|./Optimizers/HF.py.pre_betahold|./Optimizers/HF.py.pre_grouphold) ;;
            (*.py) [ -f "$LIVE/$f" ] || echo "$f" ;;
            (*) echo "$f" ;;
          esac; done)
[ -z "$EXTRA" ] && echo "V-a2: no file in the cvt7 tree beyond the live *.py files and the four HF.py.pre_* backups" || bad "V-a2 extra files: $EXTRA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_probe_tensor")" = "$(sha "$LIVE/Optimizers/HF.py.pre_probe_tensor")" ] \
  && echo "V-a3: HF.py.pre_probe_tensor == the live backup" || bad "V-a3 HF.py.pre_probe_tensor != the live backup"
[ "$(sha "$DEST/Optimizers/HF.py.pre_voteweight")" = "$HF_LIVE_SHA" ] && echo "V-h0: HF.py.pre_voteweight == the LIVE HF.py == $HF_LIVE_SHA (ISO's)" \
  || bad "V-h0 HF.py.pre_voteweight != $HF_LIVE_SHA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_betahold")" = "$HF_VW_SHA" ] && echo "V-h1: HF.py.pre_betahold == live + PATCH_VOTEWEIGHT == cvt1's HF.py $HF_VW_SHA" \
  || bad "V-h1 HF.py.pre_betahold != $HF_VW_SHA"
[ "$(sha "$DEST/Optimizers/HF.py.pre_grouphold")" = "$HF_PRE_SHA" ] && echo "V-h: HF.py.pre_grouphold == + PATCH_BETAHOLD == cvt4's HF.py $HF_PRE_SHA" \
  || bad "V-h HF.py.pre_grouphold sha != pinned $HF_PRE_SHA"
[ "$(sha "$DEST/Optimizers/HF.py")" = "$HF_POST_SHA" ] && echo "V-h2: cvt7 HF.py sha == pinned post $HF_POST_SHA" \
  || bad "V-h2 cvt7 HF.py sha $(sha "$DEST/Optimizers/HF.py") != pinned post $HF_POST_SHA"
[ -f "$CVT1_HF" ] && [ "$(sha "$CVT1_HF")" = "$HF_VW_SHA" ] && [ -f "$CVT4_HF" ] && [ "$(sha "$CVT4_HF")" = "$HF_PRE_SHA" ] \
  && echo "V-h3: cvt1's HF.py is still $HF_VW_SHA and cvt4's still $HF_PRE_SHA (untouched; the intermediates equal them)" \
  || bad "V-h3 cvt1's or cvt4's HF.py has moved"
grep -q PATCH_GROUPHOLD "$LIVE/Optimizers/HF.py" "$CVT1_HF" "$CVT4_HF" && bad "V-h4 the LIVE, cvt1's or cvt4's HF.py carries PATCH_GROUPHOLD" \
  || echo "V-h4: neither the LIVE, cvt1's nor cvt4's HF.py carries a PATCH_GROUPHOLD marker"
[ "$(sha "$LIVE/Optimizers/HF.py")" = "$HF_LIVE_SHA" ] && echo "V-h5: the LIVE HF.py is still $HF_LIVE_SHA (untouched)" \
  || bad "V-h5 the LIVE HF.py has moved ($(sha "$LIVE/Optimizers/HF.py"))"
[ "$(sha "$DEST/build_network.py")" = "$LIVE_BN_SHA" ] && echo "V-b: cvt7 build_network.py == the LIVE one $LIVE_BN_SHA (no PlainNet patch)" \
  || bad "V-b cvt7 build_network.py != $LIVE_BN_SHA"
[ "$(sha "$DEST/train.py")" = "$LIVE_TRAIN_SHA" ] && echo "V-b2: cvt7 train.py == $LIVE_TRAIN_SHA (ciso2's recorded TRAIN_PY_SHA256)" \
  || bad "V-b2 cvt7 train.py != $LIVE_TRAIN_SHA"
[ -L "$DEST/data" ] && [ "$(readlink "$DEST/data")" = "$LIVE/data" ] && echo "V-c: data -> $LIVE/data (symlink)" \
  || bad "V-c data is not a symlink to $LIVE/data"
[ -d "$DEST/data/cifar-100-python" ] && echo "V-c2: CIFAR-100 visible through the symlink" || bad "V-c2 cifar-100-python not visible"
if [ -f "$RUNNER" ]; then
  DL=$(diff "$RUNNER_SRC" "$RUNNER" | grep '^[<>]')
  D=$(printf '%s\n' "$DL" | grep -c '^[<>]')
  if [ "$D" = 2 ] && echo "$DL" | grep -qx "< cd $LIVE" && echo "$DL" | grep -qx "> cd $DEST"; then
    echo "V-d: $RUNNER differs from run_cifar.sh in EXACTLY one line: cd $LIVE -> cd $DEST"
  else
    bad "V-d the runner diff is not exactly the cd line:"; diff "$RUNNER_SRC" "$RUNNER"
  fi
  [ "$(sha "$RUNNER")" = "$RUNNER_SHA" ] && [ "$(sha "$RUNNER_SRC")" = "$LIVE_RUNNER_SHA" ] \
    && echo "V-d2: runner sha == pinned $RUNNER_SHA; run_cifar.sh still $LIVE_RUNNER_SHA" || bad "V-d2 runner / run_cifar.sh sha moved"
else
  bad "V-d no runner at $RUNNER"
fi
if python3 "$REPO/tests/test_grouphold.py" --pre "$DEST/Optimizers/HF.py.pre_grouphold" --post "$DEST/Optimizers/HF.py" \
     --structural-only >/tmp/cvt7_stage_gh.$$ 2>&1; then
  echo "V-f: tests/test_grouphold.py G0 (structure: deleting the four regions == cvt4's HF.py, byte for byte) passes ($(grep -c '^  PASS' /tmp/cvt7_stage_gh.$$) PASS)"
else
  bad "V-f tests/test_grouphold.py G0 FAILS on the cvt7 tree"; grep '^  FAIL' /tmp/cvt7_stage_gh.$$ | head
fi
rm -f /tmp/cvt7_stage_gh.$$
if python3 "$REPO/tests/test_betahold.py" --pre "$DEST/Optimizers/HF.py.pre_betahold" --post "$DEST/Optimizers/HF.py.pre_grouphold" \
     --structural-only >/tmp/cvt7_stage_bh.$$ 2>&1; then
  echo "V-f2: tests/test_betahold.py B0 (registered, unedited) passes on the intermediate pair ($(grep -c '^  PASS' /tmp/cvt7_stage_bh.$$) PASS)"
else
  bad "V-f2 tests/test_betahold.py B0 FAILS on the intermediate pair"; grep '^  FAIL' /tmp/cvt7_stage_bh.$$ | head
fi
rm -f /tmp/cvt7_stage_bh.$$
if python3 "$REPO/tests/test_voteweight.py" --pre "$DEST/Optimizers/HF.py.pre_voteweight" --post "$DEST/Optimizers/HF.py.pre_betahold" \
     --cifar-dir "$DEST" --structural-only >/tmp/cvt7_stage_vw.$$ 2>&1; then
  echo "V-f3: tests/test_voteweight.py V0 (registered, unedited) passes on the LIVE HF.py -> + PATCH_VOTEWEIGHT pair ($(grep -c '^  PASS' /tmp/cvt7_stage_vw.$$) PASS)"
else
  bad "V-f3 tests/test_voteweight.py V0 FAILS on the live pair"; grep '^  FAIL' /tmp/cvt7_stage_vw.$$ | head
fi
rm -f /tmp/cvt7_stage_vw.$$
[ "$FAIL" = 0 ] && { echo "cVT7_stage_harness: VERIFIED"; exit 0; } || { echo "cVT7_stage_harness: NOT CONSISTENT"; exit 2; }
