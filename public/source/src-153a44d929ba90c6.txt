#!/usr/bin/env python3
# =============================================================================
# cSV1_shadowvote_score.py -- THE REGISTERED SCORER FOR `csv1`:
#   N1 (LIMITS-PREP 5.1): IS A LARGE APPLIED STEP SIZE ON THE CARRIER NECESSARY FOR THE STALL, WITH ITS VOTE KEPT IN THE
#   SHARED SUM?  (CORRECTIONS 262)
#   ONE SUBMISSION, 18 JOBS, 100 EPOCHS, SEEDS {136,137,138,139}, PlainNet18_c100 / CIFAR-100, PROBE=100, PROBE_TENSOR=1
#   ON EVERY ARM, run from the ISOLATED tree $WS/harness_csv1 = cvt9's tree + PATCH_SHADOWVOTE (patches/patch_shadowvote.py).
#
# Committed BEFORE any csv1 run exists (STANDING RULE 21).  RUN IT UNEDITED (RULE 16).
#     python3 analysis/cSV1_shadowvote_score.py <runsdir>
# --manifest and --csv are OPTIONAL and DEFAULTED.
#
# FROZEN BARS (O2): every bar, sigma and tolerance score() uses is a LITERAL here; the only data-dependent terms are the
# in-batch arm means (the anchors k01 and HEAD), SIGMA_INBATCH through compose_sigma() (a max() with the frozen floor),
# the INERT arm's paired gap, and the registered vote-share statistic read from the batch's own probe records.
#
# THE ARMS (every arm but HEAD in the SCALAR grouping; idx 50 = layer4.1.bn2.weight)
#   k01        scalar                                                     floor anchor                     seeds 136-138
#   INERT      scalar  SHADOW_VOTE=shadow:shared:layer4.1.bn2.weight       the on-path at identity (== k01)  seed 136
#   SHADOWLOW  scalar  SHADOW_VOTE=shadow:floor:layer4.1.bn2.weight        50 APPLIED at the floor, its vote the SHADOW
#                                                                          term (built at the shared step)  seeds 136-139
#   NAIVELOW   scalar  SHADOW_VOTE=natural:floor:layer4.1.bn2.weight       50 APPLIED at the floor, its vote the NATURAL
#                                                                          term (built at the floor)        seeds 136-139
#   MUTE       scalar  VOTE_W=layer4.1.bn2.weight:0                        50 applied at the shared step, vote x 0 (cvt1)
#                                                                                                           seeds 136-138
#   HEAD       sets:1-49,51-53/layer4.1.bn2.weight                         rescue anchor (cvt1 / cvt4)      seeds 136-138
#   A 2x2 in (applied: shared | floor) x (vote: shared-step-sized | removed/shrunk): k01, MUTE, SHADOWLOW, NAIVELOW.
#
# STATES (SHADOWLOW, NAIVELOW, MUTE): AT-HEAD iff L >= HEAD - 5; AT-K01 iff L <= k01 + 2; else BETWEEN.
# VOTE-SHARE (registered, from probe records): per run, the median over records with epoch in [16, 36) of
#   |z_tensor[50]| / sum_i |z_tensor[i]|; per arm, the median over its runs.  DOMINANT iff share(k01) >= 0.30 and
#   share(SHADOWLOW) >= 0.50 x share(k01).  (Landed k01, 12 runs: per-run medians 0.51-0.56; csv1_registration_derivations.)
#
# BRANCH MAP (decide(); first match wins):  UNRESOLVED-DIVERGED (a seed range > 5 on an arm with >= 2 seeds)
#   -> SCALAR-NOT-COLLAPSED (k01 > 20) -> CONTROL-FAILED (HEAD - k01 < 20) -> INERT-DEPARTED (|INERT - k01(s136)| > 2)
#   -> NAIVELOW AT-K01: APPLIED-STEP-NOT-NECESSARY | NAIVELOW BETWEEN: NAIVE-PARTIAL
#   -> NAIVELOW AT-HEAD: SHADOWLOW AT-K01: VOTE-ROUTE-SUFFICES
#                        SHADOWLOW AT-HEAD: APPLIED-STEP-NECESSARY (if DOMINANT) else SHADOW-VOTE-NOT-DOMINANT
#                        SHADOWLOW BETWEEN: BOTH-ROUTES (if DOMINANT) else SHADOW-VOTE-NOT-DOMINANT
# CO-PRIMARY (in batch; bounded in neither direction): P_VOTE = NAIVELOW - SHADOWLOW; P_APPLIED = SHADOWLOW - k01.
# KEY: D_HEAD = HEAD - k01; P_NAIVE = NAIVELOW - k01; P_MUTE = MUTE - k01; P_INERT = INERT(s136) - k01(s136).
# DESCRIPTIVE: the 2x2 interaction I = (NAIVELOW - SHADOWLOW) - (MUTE - k01).
# G-BITE (-> PATCH-NOT-VERIFIED, before any level): 500 records; Lion recomputed on EVERY arm; z_agg == the arm's
#   (weighted / grouped) sum of z_tensor; no sv_ / bh_ / ch_ / wh_ key on k01, MUTE, HEAD; on INERT / SHADOWLOW / NAIVELOW
#   the sv_* records audit (sv_n == step + 2, the registered mode and index, a_shared == exp(beta_pre), a_applied == the floor or the
#   shared value, sv_z_vote == z_tensor[50], dapp/dref == a_applied/a_shared), the PHYSICAL applied-step check on every
#   record whose shared step would move 50 by >= 1e-3 (LOW: ||dw|| <= 0.01 ||dref||; INERT: within 5 %), and the TRACE
#   check (SHADOWLOW: ||vote trace|| >= ||natural|| on >= 90 % of records and never reversed by 2x; NAIVELOW mirrored;
#   INERT: the two traces and terms identical).
# plateau5 PRIMARY (epochs 95..99 of each run's own .out); TRAIN beside TEST; the probe readout after FINAL is
# DESCRIPTIVE and cannot move it.
# =============================================================================

def mean(v):
    return sum(v) / len(v) if v else None


def sd(v):
    if len(v) < 2:
        return None
    m = mean(v)
    return math.sqrt(sum((x - m) ** 2 for x in v) / (len(v) - 1))


def median(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return None
    return v[n // 2] if n % 2 else 0.5 * (v[n // 2 - 1] + v[n // 2])


def file_sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()


def lion_natural(beta_pre, mom_pre, z):
    a, b = B2 * mom_pre, (1 - B2) * z
    L = a + b
    tie = abs(L) <= TIE_REL * (abs(a) + abs(b))
    s = (L > 0) - (L < 0)
    return max(LO, min(HI, beta_pre - MS * s)), tie


def compose_sigma(sigma_in):
    cands = [("SIGMA_PRIOR_frozen", SIGMA_PRIOR), ("SIGMA_INBATCH", sigma_in)]
    which, v = max(cands, key=lambda kv: kv[1])
    return which, v, cands

def r18_tensors(plain):
    t = [("conv1.weight", 64 * 3 * 9, "Conv2d"), ("bn1.weight", 64, "BatchNorm2d"), ("bn1.bias", 64, "BatchNorm2d")]
    inp = 64
    for li, w in enumerate((64, 128, 256, 512), 1):
        for b in (0, 1):
            stride = 2 if (li > 1 and b == 0) else 1
            cin = inp if b == 0 else w
            pre = "layer%d.%d." % (li, b)
            t += [(pre + "conv1.weight", w * cin * 9, "Conv2d"), (pre + "bn1.weight", w, "BatchNorm2d"),
                  (pre + "bn1.bias", w, "BatchNorm2d"), (pre + "conv2.weight", w * w * 9, "Conv2d"),
                  (pre + "bn2.weight", w, "BatchNorm2d"), (pre + "bn2.bias", w, "BatchNorm2d")]
            if not plain and (stride != 1 or cin != w):
                t += [(pre + "shortcut.0.weight", w * cin, "Conv2d"), (pre + "shortcut.1.weight", w, "BatchNorm2d"),
                      (pre + "shortcut.1.bias", w, "BatchNorm2d")]
        inp = w
    return t + [("linear.weight", 512 * 100, "Linear"), ("linear.bias", 100, "Linear")]


NAMES = [n for n, _q, _o in r18_tensors(True)]


def parse_sets(spec, names):
    assert spec.startswith("sets:"), spec
    out = []
    for grp in spec[5:].split("/"):
        idx = []
        for tok in grp.split(","):
            if re.match(r"^\d+-\d+$", tok):
                a, b = (int(x) for x in tok.split("-"))
                idx += list(range(a, b + 1))
            elif re.match(r"^\d+$", tok):
                idx.append(int(tok))
            else:
                idx.append(names.index(tok) + 1)
        out.append(sorted(idx))
    return out


def arm_groups(arm, names=None):
    names = names or NAMES
    if SPEC[arm] == "scalar":
        return [list(range(1, len(names) + 1))]
    return parse_sets(SPEC[arm], names)

            lines.append("ARMSPEC %s SPEC %s TYPE blockwise M %d SIZES %s PARAMS %s GROUP1 %s"
                         % (a, SPEC[a], len(g), ",".join(str(len(x)) for x in g),
                            ",".join(str(sum(numel[i - 1] for i in x)) for x in g),
                            " ".join(names[i - 1] for i in g[1])))
        lines.append("SEEDS %s %s" % (a, ",".join(str(s) for s in SEEDS_OF[a])))

        lines.append("SVWITNESS %s %s" % (a, WITNESS_SV[a]))

def parse_args_line(line):
    toks = line[len("ARGS:"):].split()
    flags, cur = {}, None
    for t in toks:
        if t.startswith("--"):
            cur = t[2:]
            flags.setdefault(cur, []).append(None)
        elif cur is not None:
            v = flags[cur][-1]
            flags[cur][-1] = t if v is None else v + " " + t
    return flags


def parse_run(path):
    txt = open(path, errors="replace").read()
    eps = {}
    for a, b, c in EPTR_RE.findall(txt):
        eps[int(a)] = (float(b), float(c))
    lines = txt.splitlines()
    rec = {"eps": eps, "n_epochs": len(eps),
           "traceback": "Traceback (most recent call last)" in txt,
           "run_done": any(ln.strip() == "RUN_DONE" for ln in lines),
           "args": [ln for ln in lines if ln.startswith("ARGS:")],
           "env": [ln for ln in lines if ln.startswith("ENV:")],
           "pt": [ln for ln in lines if ln.startswith("PROBE_TENSOR:")],
           "vw": [ln for ln in lines if ln.startswith("VOTE_W")],
           "sv": [ln for ln in lines if ln.startswith("SHADOW_VOTE")],
           "bh": [ln for ln in lines if ln.startswith("BETA_HOLD")],
           "ch": [ln for ln in lines if ln.startswith("COMP_HOLD")],
           "wh": [ln for ln in lines if ln.startswith("WINDOW_HOLD")],
           "plateau5": None, "train5": None}
    tail = [eps.get(e) for e in range(EPOCHS - 5, EPOCHS)]
    if all(x is not None for x in tail) and all(math.isfinite(x[0]) and math.isfinite(x[1]) for x in tail):
        rec["plateau5"] = mean([x[1] for x in tail])
        rec["train5"] = mean([x[0] for x in tail])
    rec["complete"] = (sorted(eps) == list(range(EPOCHS)) and rec["run_done"]
                       and not rec["traceback"] and rec["plateau5"] is not None)
    return rec


def read_runs(runsdir):
    cand, notes = {}, []
    if not runsdir or not os.path.isdir(runsdir):
        return {}, notes
    for fn in sorted(os.listdir(runsdir)):
        m = OUT_RE.match(fn)
        if not m:
            continue
        arm, seed, jid = m.group(1), int(m.group(2)), int(m.group(3))
        if (arm, seed) not in RUNS:
            notes.append("%s is not a registered (arm, seed) -- ignored" % fn)
            continue
        rec = parse_run(os.path.join(runsdir, fn))
        rec.update({"file": fn, "job": jid, "arm": arm, "seed": seed})
        cand.setdefault((arm, seed), []).append(rec)
    got = {}
    for k, lst in cand.items():
        comp = [r for r in lst if r["complete"]]
        pick = max(comp or lst, key=lambda r: r["job"])
        if len(lst) > 1:
            notes.append("%s-s%d has %d files (%d complete); using %s" % (k[0], k[1], len(lst), len(comp), pick["file"]))
        got[k] = pick
    return got, notes

        return None
    d = {}
    for ln in open(path):

        if len(p) >= 2:
            d.setdefault(p[0], " ".join(p[1:]))

def tail_slope(eps, lo=80, hi=99):
    xs = [e for e in range(lo, hi + 1) if e in eps]
    if len(xs) < 3:
        return None
    ys = [eps[e][1] for e in xs]
    mx, my = mean(xs), mean(ys)
    den = sum((x - mx) ** 2 for x in xs)
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den if den else None

def usable(r):
    return (r.get("superseded") == "0" and r.get("collapsed") == "0"
            and r.get("complete") == "1" and (r.get("plateau5") or "").strip() != "")

    return (r.get("epochs_requested") == str(EPOCHS) and r.get("augment") == AUG and r.get("beta_clip") == CLIP
            and r.get("meta_stepsize") == MST and r.get("alpha0") == A0 and r.get("batch_size") == str(BATCH))


def pooled_sigma(rows, extra):
    cells = {}
    for r in rows:
        if usable(r) and std_cell(r) and extra(r):
            cells.setdefault(tuple(r.get(k, "") for k in CELLKEYS), []).append(float(r["plateau5"]))
    ss, df, nc, nm = 0.0, 0, 0, 0
    for v in cells.values():
        if len(v) < 2:
            continue
        m = mean(v)
        ss += sum((x - m) ** 2 for x in v)
        df += len(v) - 1
        nc += 1
        nm += len(v)
    return (math.sqrt(ss / df) if df else None), df, nc, nm

def load_probe(runsdir, arm, seed, prefix=PREFIX):
    for base in (os.path.join(runsdir, prefix), runsdir):
        p = os.path.join(base, "probe_%s-%s-s%d" % (prefix, arm, seed), "probe.jsonl")

SV_LIST_KEYS = ("sv_a_shared", "sv_a_applied", "sv_z_vote", "sv_z_other", "sv_dw", "sv_dref", "sv_dapp",
                "sv_hv_norm", "sv_ho_norm")
SV_KEYS = ("sv_n", "sv_vote", "sv_applied", "sv_idx") + SV_LIST_KEYS
FOREIGN_PREFIXES = ("bh_", "ch_", "wh_")


def _groupvec(r, key, ng):
    v = r.get(key)
    if not (isinstance(v, list) and len(v) == 1 and isinstance(v[0], list) and len(v[0]) == ng):
        return None
    return [float(x) for x in v[0]]


def _close(a, b, rel, absol=1e-30):
    return abs(a - b) <= rel * max(abs(a), abs(b)) + absol


def weights_of(arm):
    return {IDX_HEAD - 1: 0.0} if arm == "MUTE" else {}


def zagg_expected(arm, zt):
    W = weights_of(arm)
    groups = arm_groups(arm)
    return [sum(W.get(i - 1, 1.0) * zt[i - 1] for i in g) for g in groups]


def sv_audit(arm, recs, every=PROBE):
    """the SHADOW_VOTE half of G-BITE for one run, a pure function of its records.  -> (ok, detail)."""
    vote, applied = MODE[arm]
    d = {"sv_bad_keys": 0, "sv_bad_n": 0, "sv_bad_mode": 0, "sv_bad_ashared": 0, "sv_bad_aapplied": 0,
         "sv_bad_zvote": 0, "sv_bad_ratio": 0, "sv_bad_inert": 0, "sv_phys_records": 0, "sv_bad_phys": 0,
         "sv_trace_records": 0, "sv_trace_ok": 0, "sv_trace_reversed": 0, "sv_worst_zvote": 0.0}
    for k, r in enumerate(recs):
        if not isinstance(r, dict):
            d["sv_bad_keys"] += 1
            continue
        try:
            s = int(r["step"])
            if any(key not in r for key in SV_KEYS) or any(not (isinstance(r[key], list) and len(r[key]) == 1)
                                                            for key in SV_LIST_KEYS + ("sv_idx",)):
                d["sv_bad_keys"] += 1
                continue
            a_sh, a_ap = float(r["sv_a_shared"][0]), float(r["sv_a_applied"][0])
            zv, zo = float(r["sv_z_vote"][0]), float(r["sv_z_other"][0])
            dw, dref, dapp = float(r["sv_dw"][0]), float(r["sv_dref"][0]), float(r["sv_dapp"][0])
            hv, ho = float(r["sv_hv_norm"][0]), float(r["sv_ho_norm"][0])
            bpre = float(r["beta_pre"][0][0])
            zt50 = float(r["z_tensor"][IDX_HEAD - 1])
        except (KeyError, TypeError, ValueError, IndexError):
            d["sv_bad_keys"] += 1
            continue
        vals = [a_sh, a_ap, zv, zo, dw, dref, dapp, hv, ho, bpre, zt50]
        if not all(math.isfinite(x) for x in vals):
            d["sv_bad_keys"] += 1
            continue
        d["sv_bad_n"] += int(r["sv_n"]) != s + 2
        d["sv_bad_mode"] += not (r["sv_vote"] == vote and r["sv_applied"] == applied and r["sv_idx"] == [IDX_HEAD])
        d["sv_bad_ashared"] += not _close(a_sh, math.exp(bpre), SV_REL)
        if applied == "floor":
            d["sv_bad_aapplied"] += not _close(a_ap, FLOOR_ALPHA, INERT_REL)
        else:
            d["sv_bad_aapplied"] += a_ap != a_sh
        ez = abs(zv - zt50) / (max(abs(zv), abs(zt50)) + 1e-30)
        d["sv_worst_zvote"] = max(d["sv_worst_zvote"], ez if (zv or zt50) else 0.0)
        d["sv_bad_zvote"] += not _close(zv, zt50, SV_REL)
        if dref > 0:
            d["sv_bad_ratio"] += not _close(dapp / dref, a_ap / a_sh, RATIO_REL)
        else:
            d["sv_bad_ratio"] += 1
        if applied == "shared":
            d["sv_bad_inert"] += not (_close(zo, zv, INERT_REL) and _close(ho, hv, INERT_REL) and _close(dapp, dref, INERT_REL))
        if dref >= DREF_MIN:
            d["sv_phys_records"] += 1
            if applied == "floor":
                d["sv_bad_phys"] += dw > DW_LOW_MAX * dref
            else:
                d["sv_bad_phys"] += abs(dw - dref) > DW_INERT_REL * dref
        if k >= 1 and applied == "floor":
            d["sv_trace_records"] += 1
            big, small = (hv, ho) if vote == "shadow" else (ho, hv)
            d["sv_trace_ok"] += big >= small
            d["sv_trace_reversed"] += small >= TRACE_FACTOR * big
    ok = (d["sv_bad_keys"] == 0 and d["sv_bad_n"] == 0 and d["sv_bad_mode"] == 0 and d["sv_bad_ashared"] == 0
          and d["sv_bad_aapplied"] == 0 and d["sv_bad_zvote"] == 0 and d["sv_bad_ratio"] == 0 and d["sv_bad_inert"] == 0
          and d["sv_bad_phys"] == 0)
    if applied == "floor":
        ok = (ok and d["sv_trace_records"] > 0 and d["sv_trace_ok"] >= TRACE_FRAC * d["sv_trace_records"]
              and d["sv_trace_reversed"] == 0)
    return ok, d


def bite_check(arm, recs, every=PROBE, n_records=N_RECORDS):
    """G-BITE for ONE run.  -> (ok, detail).  Pure function of the records."""
    ng = len(arm_groups(arm))
    d = {"n": 0, "nonfinite": 0, "bad_rec": 0, "bad_step": 0, "bad_lion": 0, "ties": 0, "bad_zagg": 0,
         "keys_on_control": 0, "foreign_keys": 0, "worst_lion": 0.0, "worst_zagg": 0.0}
    if recs is None:
        d["why"] = "no probe.jsonl"
        return False, d
    for k, r in enumerate(recs):
        if r is None or not isinstance(r.get("step"), int) or not isinstance(r.get("beta"), list) \
                or len(r["beta"]) != ng or not isinstance(r.get("z_tensor"), list) or len(r["z_tensor"]) != NTENS:
            d["bad_rec"] += 1
            continue
        try:
            beta = [float(x) for x in r["beta"]]
            bpre = _groupvec(r, "beta_pre", ng)
            mom = _groupvec(r, "mom_pre", ng)
            z = _groupvec(r, "z_agg", ng)
            zt = [float(x) for x in r["z_tensor"]]
        except (TypeError, ValueError):
            d["bad_rec"] += 1
            continue
        if bpre is None or mom is None or z is None:
            d["bad_rec"] += 1
            continue
        if not all(math.isfinite(x) for x in beta + bpre + mom + z + zt):
            d["nonfinite"] += 1
            continue
        s = r["step"]
        if s != every * k:
            d["bad_step"] += 1
        d["n"] += 1
        for g in range(ng):
            nat, tie = lion_natural(bpre[g], mom[g], z[g])
            if tie:
                d["ties"] += 1
                continue
            e = abs(nat - beta[g])
            d["worst_lion"] = max(d["worst_lion"], e)
            d["bad_lion"] += e > BH_TOL
        tot = sum(abs(x) for x in zt) + 1e-30
        ez = max(abs(p - q) for p, q in zip(zagg_expected(arm, zt), z)) / tot
        d["worst_zagg"] = max(d["worst_zagg"], ez)
        d["bad_zagg"] += ez > ZAGG_REL
        has_sv = any(key.startswith("sv_") for key in r)
        d["foreign_keys"] += any(key.startswith(FOREIGN_PREFIXES) for key in r)
        if arm not in SV_ARMS:
            d["keys_on_control"] += has_sv
    ok = (len(recs) == n_records and d["bad_rec"] == 0 and d["n"] + d["nonfinite"] == n_records
          and d["bad_step"] == 0 and d["bad_lion"] == 0 and d["bad_zagg"] == 0 and d["keys_on_control"] == 0
          and d["foreign_keys"] == 0)
    if arm in SV_ARMS:
        ok2, d2 = sv_audit(arm, recs, every)
        d.update(d2)
        ok = ok and ok2
    return ok, d


def vote_share(recs, win=DOM_WIN):
    """the registered statistic: median over records with epoch in [win) of |z_tensor[50]| / sum |z_tensor|."""
    v = []
    for r in recs or []:
        if not isinstance(r, dict) or not isinstance(r.get("z_tensor"), list) or len(r["z_tensor"]) != NTENS:
            continue
        ep = r.get("step", -1) / float(STEPS_PER_EPOCH)
        if not (win[0] <= ep < win[1]):
            continue
        try:
            zt = [abs(float(x)) for x in r["z_tensor"]]
        except (TypeError, ValueError):
            continue
        tot = sum(zt)
        if tot > 0 and math.isfinite(tot):
            v.append(zt[IDX_HEAD - 1] / tot)
    return median(v)

def beta_summary(recs, coord):
    vals = []
    for r in recs:
        if isinstance(r, dict) and isinstance(r.get("beta"), list) and len(r["beta"]) > coord \
                and isinstance(r.get("step"), int):
            try:
                v = float(r["beta"][coord])
            except (TypeError, ValueError):
                continue
            if math.isfinite(v):
                vals.append((r["step"], v))
    if not vals:
        return None
    pk = max(vals, key=lambda sv: sv[1])
    pin = None
    for i in range(len(vals)):
        if all(v <= LO + 0.01 for _s, v in vals[i:]):
            pin = vals[i][0] / float(STEPS_PER_EPOCH)
            break
    return {"peak": pk[1], "peak_epoch": pk[0] / float(STEPS_PER_EPOCH), "pin": pin}

def state(L, M):
    if L >= M["HEAD"] - MATCH_BAR:
        return "AT-HEAD"
    if L <= M["k01"] + NULL_BAR:
        return "AT-K01"
    return "BETWEEN"


def decide(M, diverged=False, inert_gap=0.0, dominant=True):
    """M: arm -> mean plateau5; inert_gap: INERT(s136) - k01(s136); dominant: the registered vote-share flag."""
    if diverged:
        return "UNRESOLVED-DIVERGED"
    if M["k01"] > K01_MAX:
        return "SCALAR-NOT-COLLAPSED"
    if M["HEAD"] - M["k01"] < GAP_MIN:
        return "CONTROL-FAILED"
    if abs(inert_gap) > INERT_BAR:
        return "INERT-DEPARTED"
    sn, ss = state(M["NAIVELOW"], M), state(M["SHADOWLOW"], M)
    if sn == "AT-K01":
        return "APPLIED-STEP-NOT-NECESSARY"
    if sn == "BETWEEN":
        return "NAIVE-PARTIAL"
    if ss == "AT-K01":
        return "VOTE-ROUTE-SUFFICES"
    if not dominant:
        return "SHADOW-VOTE-NOT-DOMINANT"
    if ss == "AT-HEAD":
        return "APPLIED-STEP-NECESSARY"
    return "BOTH-ROUTES"

        s = every * k
        ep = s / float(STEPS_PER_EPOCH)
        sh = share if DOM_WIN[0] <= ep < DOM_WIN[1] else 0.02
        others = [rnd.uniform(-1e-3, 1e-3) for _ in range(NTENS - 1)]
        so = sum(abs(x) for x in others)
        z50 = (1 if rnd.random() < 0.7 else -1) * so * sh / (1 - sh)
        zt = [f32(x) for x in others[:IDX_HEAD - 1] + [z50] + others[IDX_HEAD - 1:]]
        zsum = zagg_expected(arm, zt)
        if mute_broken:
            zsum = [sum(zt)] if ng == 1 else zsum
        mom = [rnd.uniform(-1e-3, 1e-3) for _ in range(ng)]
        bpre = [f32(max(LO, min(HI, -9.0 + 5.0 * math.sin(k / 40.0)))) for _ in range(ng)]
        beta = [f32(lion_natural(bpre[g], mom[g], zsum[g])[0]) for g in range(ng)]
        if touch:
            beta[0] = f32(beta[0] + 0.002)
        rec = {"step": s, "pt_b2": 0.9, "pt_mp": 0.99}
        if (arm in SV_ARMS and not strip_sv) or add_sv:
            vote, applied = MODE.get(arm, ("shadow", "floor"))
            a_sh = f32(math.exp(bpre[0]))
            if ashared_broken:
                a_sh = f32(a_sh * 1.5)
            a_ap = a_sh if (applied == "shared" or applied_broken) else FLOOR_ALPHA
            C = 2.2
            dref, dapp = a_sh * C, a_ap * C
            dw = dapp if applied == "shared" or applied_broken or phys_broken else min(dapp, 2.5e-6)
            if phys_broken:
                dw = dref
            hs, hn = 50.0 * a_sh * (k + 1), 50.0 * FLOOR_ALPHA * (k + 1)
            if applied == "shared":
                hv = ho = hs
            elif vote == "shadow":
                hv, ho = hs, hn
            else:
                hv, ho = hn, hs
            if trace_swapped:
                hv, ho = ho, hv
            zv = zt[IDX_HEAD - 1] * (1.01 if zvote_broken else 1.0)
            zo = zv if applied == "shared" else zv * (0.001 if vote == "shadow" else 1000.0)
            if inert_other:
                zo = zv * 1.5
            rec.update({"sv_n": s + 2 + (1 if n_off else 0), "sv_vote": vote, "sv_applied": applied,
                        "sv_idx": [IDX_HEAD + (1 if idx_broken else 0)], "sv_a_shared": [a_sh], "sv_a_applied": [a_ap],
                        "sv_z_vote": [zv], "sv_z_other": [zo], "sv_dw": [dw], "sv_dref": [dref], "sv_dapp": [dapp],
                        "sv_hv_norm": [hv], "sv_ho_norm": [ho]})
        if add_bh:
            rec.update({"bh_n": s + 2, "bh_active": 0, "bh_nat": beta[-1], "bh_held": beta[-1]})
        rec.update({"beta": beta, "beta_pre": [bpre], "mom_pre": [mom], "z_agg": [zsum], "z_tensor": zt,
                    "m_tensor": [0.0] * NTENS})
        out.append(rec)
    return out

def _e_score(d, csvpath):

    ap.add_argument("--runsdir", dest="runsdir_kw", default="")

        return selftest(a.csv, runsdir or ".")
