def _fin(x):
    return isinstance(x, (int, float)) and not isinstance(x, bool) and math.isfinite(x)


def _med(v):
    v = sorted(v)
    n = len(v)
    if not n:
        return float("nan")
    return v[n // 2] if n % 2 else 0.5 * (v[n // 2 - 1] + v[n // 2])


# =============================================================================
# THE VOTE (caw1's code, unchanged in substance; the meta alg now comes from the arm, not the pairing)
# =============================================================================
def vote_terms(r, meta_alg):
    """-> (L list, applied scalar) for ONE scalar-arm record, or None if the record lacks the keys or is non-finite."""
    try:
        mp, b2 = float(r["pt_mp"]), float(r["pt_b2"])
        z, m = r["z_tensor"], r["m_tensor"]
        mom, za = float(r["mom_pre"][0][0]), float(r["z_agg"][0][0])
    except (KeyError, TypeError, IndexError, ValueError):
        return None
    if len(z) != NTENS or len(m) != NTENS or not all(_fin(x) for x in z) or not all(_fin(x) for x in m):
        return None
    if meta_alg == "Adam":
        L = [mp / (1.0 - mp) * mi + zi for mi, zi in zip(m, z)]
        app = mp * mom + za
    else:
        L = [b2 * mi + (1.0 - b2) * zi for mi, zi in zip(m, z)]
        app = b2 * mom + (1.0 - b2) * za
    if not (_fin(app) and all(_fin(x) for x in L)):
        return None
    return L, app


def _sg(x):
    return (x > 0) - (x < 0)


def decomposition(recs, meta_alg):
    """-> (n_usable, n_bad_sum, n_bad_sign, n_nonfinite).  sum_i L_i must equal the harness's applied term."""
    n = bad = bads = nonf = 0
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            nonf += 1
            continue
        L, app = v
        s = sum(L)
        scale = sum(abs(x) for x in L)
        n += 1
        bad += not (abs(s - app) <= DECOMP_TOL * scale or abs(s - app) <= 1e-30)
        bads += _sg(s) != _sg(app) and abs(app) > DECOMP_TOL * scale
    return n, bad, bads, nonf


def fit_carriers(recs, meta_alg, k=3):
    """top-k tensors by signed contribution sum_r s_r * L_i,r (s_r = the applied sign), on the FIT records only."""
    acc = [0.0] * NTENS
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            continue
        L, app = v
        s = _sg(app)
        for i, x in enumerate(L):
            acc[i] += s * x
    return tuple(sorted(sorted(range(NTENS), key=lambda i: (-acc[i], i))[:k]))


def dom_c(recs, meta_alg, Cset):
    """-> (share, n): share of usable records on which C alone fixes the applied sign."""
    n = d = 0
    for r in recs or []:
        v = vote_terms(r, meta_alg) if isinstance(r, dict) else None
        if v is None:
            continue
        L, app = v
        s = _sg(app)
        sc = sum(L[i] for i in Cset)
        rest = sum(abs(L[i]) for i in range(NTENS) if i not in Cset)
        n += 1
        d += (s != 0 and _sg(sc) == s and abs(sc) > rest)
    return (d / n if n else float("nan")), n


def dom_reading(recs_by_seed, meta_alg, fit_seed=FIT_SEED, score_seeds=SCORE_SEEDS, seeds=SEEDS):
    Cx = fit_carriers(recs_by_seed.get(fit_seed), meta_alg)
    held = [r for s in score_seeds for r in (recs_by_seed.get(s) or [])]
    share, n = dom_c(held, meta_alg, Cx)
    alls = [r for s in seeds for r in (recs_by_seed.get(s) or [])]
    dshare, dn = dom_c(alls, meta_alg, CARRIER_IDX0)
    return {"C": Cx, "dom": share, "n": n, "decl_dom": dshare, "decl_n": dn}


# =============================================================================
# THE DOSE: the realised per-step shrink a*wd from the arm's own probe records
# =============================================================================
def rec_shrink(r, wd):
    b = r.get("beta") if isinstance(r, dict) else None
    if not isinstance(b, list) or not b or not all(_fin(x) for x in b):
        return None
    return _med([math.exp(x) for x in b]) * wd


def dose_of_seed(recs, wd):
    """-> (peak, pre_peak, late_median) of the per-record shrink; None entries for an empty set."""
    allv, pre, late = [], [], []
    for r in recs or []:
        s = rec_shrink(r, wd)
        if s is None or not isinstance(r.get("step"), int):
            continue
        allv.append(s)
        if r["step"] <= DOSE_PRE_END:
            pre.append(s)
        if r["step"] >= 4 * N_RECORDS * D.PROBE // 5:
            late.append(s)
    return (max(allv) if allv else float("nan"), max(pre) if pre else float("nan"),
            _med(late) if late else float("nan"))


def dose_word(rho):
    if not _fin(rho):
        return "UNREAD"
    if rho < DOSE_BAR_LO:
        return "BELOW"
    if rho > DOSE_BAR_HI:
        return "ABOVE"
    return "MATCHED"

def cell_state(Lseeds, kseeds):
    L, k = mean(Lseeds), mean(kseeds)
    if L < REF_MIN or (max(Lseeds) - min(Lseeds)) > DIVERGED_BAR:
        return "UNREADABLE"
    ncol = sum(1 for x in kseeds if x <= R50 * L)
    if 1 <= ncol <= len(kseeds) - 1:
        return "SPLIT"
    if k <= R50 * L:
        return "COLLAPSE"
    if L - k < GAP_BAR:
        return "NOGAP"
    return "PARTIAL"


def decide(states, k01_mean, rho_x):
    """states: {"M","L","A","X": state}.  -> branch token.  Pure function."""
    if k01_mean > K01_MAX:
        return "CONTROL-NOT-COLLAPSED"
    a = states["A"]
    if a == "UNREADABLE":
        return "UNRESOLVED-REFERENCE-A"
    if a == "COLLAPSE":
        return "HAZARD-AT-STANDARD-RECIPE"
    if a in ("SPLIT", "PARTIAL"):
        return "PARTIAL-AT-STANDARD-RECIPE"
    x = states["X"]
    if x in ("COLLAPSE", "SPLIT"):
        return "NO-COLLAPSE-AT-STANDARD-DOSE-ONLY"
    if x == "NOGAP" and _fin(rho_x):
        return ("NO-COLLAPSE-PAIRING-IMMUNE-AT-CONTROL-DOSE" if rho_x >= DOSE_BAR_LO
                else "NO-COLLAPSE-CONTROL-DOSE-NOT-REACHED")
    return "NO-COLLAPSE-DOSE-ARM-UNRESOLVED"


def _pcu(st):
    return "P" if st == "NOGAP" else ("C" if st in ("COLLAPSE", "SPLIT") else "U")


def attribution(states):
    m, l = _pcu(states["M"]), _pcu(states["L"])
    return {("C", "P"): "ATTR-BASE-PROTECTS", ("P", "C"): "ATTR-META-PROTECTS", ("P", "P"): "ATTR-EITHER-PROTECTS",
            ("C", "C"): "ATTR-NEITHER-ALONE"}.get((m, l), "ATTR-UNRESOLVED")


def gate_word(branch, states, doms):
    """doms: {arm: dom_reading} for the scalar arms incl. K01 (None if unreadable)."""
    if branch == "PARTIAL-AT-STANDARD-RECIPE" or branch.startswith("NO-COLLAPSE-"):
        return "GATE-DOES-NOT-FIRE"
    if branch != "HAZARD-AT-STANDARD-RECIPE":
        return "GATE-UNREADABLE"
    kc = doms.get(D.CONTROL)
    if kc is None or not (kc["dom"] >= DOM_BAR):
        return "GATE-UNREADABLE-DOMC-CONTROL"
    ac = doms.get(D.SCALAR_OF["A"])
    if states["A"] == "COLLAPSE" and ac is not None and ac["dom"] >= DOM_BAR:
        return "GATE-FIRES-A"
    return "GATE-COLLAPSE-WITHOUT-DOMINANCE"

def score(runsdir, csvpath, manifest_path=""):
    """-> (rc, FINAL line).  rc 0 branch taken, 1 harness-unsound, 2 incomplete."""
    print("=" * 78)
    print(" caw2 -- THE STANDARD-RECIPE GATE, WITH EACH CONFOUND IN ITS OWN CELL.  Does the shared-step-size collapse")
    print("         survive the META swap (M), the BASE swap (L), both (A: AdamW+Adam, the standard recipe) and a 10x")
    print("         dose (X), against the in-batch mechanism cell K01?")
    print(" %s / %s   %d epochs   seeds %s   AUGMENT=%s  BETA_CLIP=%s  PROBE=%d  PROBE_TENSOR=1"
          % (D.NET, D.DSET, EPOCHS, ",".join(str(s) for s in SEEDS), D.AUG, D.CLIP, D.PROBE))
    for a in ARMS:
        print("   %-4s %-5s+%-4s %-9s wd %-4s  %s" % (a, D.BASE_OF[a], D.META_OF[a], D.SPEC[a], D.WD[a], D.ROLE[a]))
    print(" CO-PRIMARY: G_A = AL - AS.  Every state is WITHIN its cell.  plateau5 = mean TEST over epochs %d..%d."
          % (EPOCHS - 5, EPOCHS - 1))
    print(" BARS ARE FROZEN LITERALS (O2).  ALPHA-INDEPENDENT DECAY IS NOT TESTED.")
    print("=" * 78)
    try:
        rows = read_corpus(csvpath)
        disc = "not readable" if rows is None else "%d rows after corpus_exclusions.filter_rows, %s- excluded" % (len(rows), PREFIX)
    except Exception as ex:
        disc = "not readable (%s)" % type(ex).__name__
    print("corpus: %s  (DISCLOSURE ONLY -- no bar, sigma, branch or stamp reads it)" % disc)

    runs, rnotes = C.read_runs(runsdir, PREFIX, ARMS, EPOCHS)
    print("\nCOMPLETE  %d runs, epochs 0..%d, RUN_DONE, no traceback, finite plateau5" % (len(ARMS) * len(SEEDS), EPOCHS - 1))
    for n_ in rnotes:
        print("  NOTE %s" % n_)
    missing = []
    for s in SEEDS:
        for a in ARMS:
            r = runs.get((a, s))
            ok = r is not None and r["complete"]
            print("  %-4s %s-%s-s%d  %s" % ("OK" if ok else "MISS", PREFIX, a, s,
                  "absent" if r is None else "%d epoch lines, RUN_DONE=%s, traceback=%s, plateau5 %s (train %s)"
                  % (r["n_epochs"], r["run_done"], r["traceback"], fmt(r["plateau5"]), fmt(r["train5"]))))
            if not ok:
                missing.append("%s-s%d" % (a, s))
    if missing:
        final = "FINAL: INCOMPLETE | %d/%d | missing %s" % (len(ARMS) * len(SEEDS) - len(missing), len(ARMS) * len(SEEDS),
                                                            ",".join(missing))
        print("\n" + final)
        for ln in LICENSE["INCOMPLETE"]:
            print("  " + ln)
        return 2, final

    harness = []

    def hchk(cond, label, extra=""):
        print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))
        if not cond:
            harness.append(label)
        return bool(cond)

    print("\nG-ARGS    [HARNESS-NULL] every run's OWN ARGS line == its arm's registered flags (base, meta, decay, grain)")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            if len(r["args"]) != 1:
                hchk(False, "G-ARGS %s-s%d exactly one ARGS line" % (a, s), str(len(r["args"])))
                continue
            f = C.parse_args_line(r["args"][0])
            dup = [k for k, v in f.items() if len(v) > 1]
            want = dict(D.args_pairs(a, s, "<save>"))
            bad = [k for k, v in want.items() if k != "save-directory" and f.get(k, [None])[-1] != v]
            extra = sorted(set(f) - set(want))
            sv = f.get("save-directory", [""])[-1] or ""
            ok = not dup and not bad and not extra and sv.rstrip("/").endswith("/" + PREFIX)
            hchk(ok, "G-ARGS %s-s%d (%s+%s %s wd %s)" % (a, s, D.BASE_OF[a], D.META_OF[a], D.SPEC[a], D.WD[a]),
                 "" if ok else "repeated %s; wrong %s; unregistered %s; save %s"
                 % (dup, ["%s=%s" % (k, f.get(k, [None])[-1]) for k in bad], extra, sv))

    print("\nG-ENV     the environment rode the ENV line on every run")
    envs = sorted(set(" ".join(t for t in ln.split() if not t.startswith("PROBE_DIR=")) for r in runs.values() for ln in r["env"]))
    hchk(len(envs) == 1 and all(len(r["env"]) == 1 for r in runs.values()),
         "G-ENV one distinct ENV line (PROBE_DIR stripped), one per run", "%d distinct" % len(envs))
    hchk(envs == [D.ENV_EXPECTED], "G-ENV the ENV line is cvt1 ... cwd5's, byte for byte (PROBE_DIR stripped)")

    print("\nG-PT      [HARNESS-NULL] ONE PROBE_TENSOR line per run == its arm's witness (grain, 62 tensors, META alg)")

        for a in ARMS:

            hchk(ok, "G-PT %s-s%d" % (a, s), "" if ok else "got %r" % ([x[:150] for x in pt],))

    print("\nG-WITNESS the five `off` lines once each; no COMP_HOLD / WINDOW_HOLD line")
    for s in SEEDS:
        for a in ARMS:
            r = runs[(a, s)]
            ok = all(r[k.split(":")[0]] == [k] for k in D.OFF_LINES) and not r["COMP_HOLD"] and not r["WINDOW_HOLD"]
            hchk(ok, "G-WITNESS %s-s%d" % (a, s), "" if ok else str(dict((k, r[k]) for k in C.KINDS if r[k])))

    print("\nG-STRUCT  the batch's own PARTITION-MANIFEST.txt == caw2_design.manifest_text(), byte for byte")
    mp, how = resolve_manifest(runsdir, manifest_path)
    if not mp:
        hchk(False, "G-STRUCT manifest present", how)
    else:
        print("  manifest: %s (%s)" % (mp, how))
        hchk(open(mp).read() == D.manifest_text(), "G-STRUCT byte-identical")

    print("\nG-PROV    PROVENANCE.txt")
    pp, phow = resolve(runsdir, "", "PROVENANCE.txt")
    prov = read_kv(pp)
    if prov is None:
        hchk(False, "G-PROV PROVENANCE.txt present", phow)
    else:
        print("  provenance: %s (%s)" % (pp, phow))
        me = file_sha(os.path.abspath(__file__))
        for key, want, lab in (("MODE", "submit", "MODE submit"),
                               ("BUILD_NETWORK_SHA256", D.BN_SHA, "build_network.py == cvt8's"),
                               ("HF_SHA256", D.HF_SHA, "HF.py == cwd1's tree, UNCHANGED"),
                               ("RUNNER_SHA256", D.RUNNER_SHA, "runner == run_cifar_cwd1.sh, UNCHANGED"),
                               ("SCORER_SHA256", me, "SCORER == this file"),
                               ("DESIGN_SHA256", file_sha(D.__file__), "DESIGN == the caw2_design.py this scorer loaded"),
                               ("CWD_DESIGN_SHA256", file_sha(D.W.__file__), "the reused cwd_design.py this scorer loaded"),
                               ("COMMON_SHA256", file_sha(C.__file__), "COMMON == the cwd_common.py this scorer loaded")):

    print("\nG-PROBE   [HARNESS-NULL] %d records per run; the step-size count is the GRAIN's (1 or %d); on the scalar"
          " arms the per-tensor vote sums to the harness's applied term (within %g of the vote's mass sum|L_i|) with its"
          " sign, under the arm's OWN meta alg" % (N_RECORDS, NTENS, DECOMP_TOL))
    recs_all, nonfin = {}, {}
    for s in SEEDS:
        for a in ARMS:
            recs = C.load_probe(runsdir, PREFIX, a, s)
            recs_all[(a, s)] = recs
            if recs is None:
                hchk(False, "G-PROBE %s-s%d probe.jsonl present" % (a, s))
                continue
            want_nb = 1 if D.SPEC[a] == "scalar" else NTENS
            badrec = sum(1 for r in recs if not (isinstance(r, dict) and isinstance(r.get("step"), int)
                                                   and isinstance(r.get("beta"), list) and len(r["beta"]) == want_nb
                                                   and isinstance(r.get("z_tensor"), list) and len(r["z_tensor"]) == NTENS))
            line = "%d records, %d malformed / wrong step-size count" % (len(recs), badrec)
            ok = len(recs) == N_RECORDS and badrec == 0
            if D.SPEC[a] == "scalar":
                n, bsum, bsign, nf = decomposition(recs, D.META_OF[a])
                nonfin[(a, s)] = nf
                line += "; decomposition (%s): %d usable, %d off, %d wrong sign, %d non-finite (disclosed)" % (
                    D.META_OF[a], n, bsum, bsign, nf)
                ok = ok and bsum == 0 and bsign == 0 and n > 0
            hchk(ok, "G-PROBE %s-s%d (%s, %d step size(s))" % (a, s, D.SPEC[a], want_nb), line)

    arm_v = dict((a, [runs[(a, s)]["plateau5"] for s in SEEDS]) for a in ARMS)
    arm_t = dict((a, [runs[(a, s)]["train5"] for s in SEEDS]) for a in ARMS)
    M = dict((a, mean(arm_v[a])) for a in ARMS)
    T = dict((a, mean(arm_t[a])) for a in ARMS)
    print("\nG-FLOOR / G-CEIL")
    hi = max(M.values())
    hchk(hi >= FLOOR_MIN, "G-FLOOR max arm mean >= %.2f pp" % FLOOR_MIN, "max %.4f" % hi)
    hchk(hi <= CEIL_MAX, "G-CEIL max arm mean <= %.2f pp" % CEIL_MAX, "max %.4f" % hi)

    if harness:
        final = "FINAL: HARNESS-UNSOUND | %d harness gate(s) failed: %s" % (
            len(harness), "; ".join(sorted(set(h.split()[0] for h in harness))))
        print("\n" + "=" * 78)
        print(final)
        print("=" * 78)

            print("   FAILED  %s" % h)

            print("  " + ln)

    print("\nTHE VOTE    DOM_C per scalar arm: C fitted on seed %d (top-3 by signed contribution), scored on seeds %s;"

        if any(nonfin.get((a, s)) for s in SEEDS):
            stamps.append("NONFINITE-RECORDS-%s" % a)
    stamps.append(("HW-UNIFORM-%s" % hwset[0].replace(" ", "_")) if len(hwset) == 1 else "HW-MIXED")

    print("=" * 78)

    print("REGISTERED PREDICTION (UNSURE): %s; this batch: A-%s, %s" % (" / ".join(PREDICTION), states["A"], attr))
    print("\nWHAT THIS DECIDES -- as registered, before any run existed:")
    for ln in LICENSE[branch]:
        print("  " + ln)
    if branch not in ("CONTROL-NOT-COLLAPSED",):

def _e_curve(level):
    return [level if e >= EPOCHS - 5 else level * (1 - math.exp(-(e + 1) / 12.0)) for e in range(EPOCHS)]


def _e_write(d, arm, s, jid, level, n_ep=EPOCHS, run_done=True, env=_E_ENV, args=None, pt=None, offs=None):
    lines = ["NODE=[HOST_2f2f29d9e1] | JOB=%s-%s-s%d %d | AUGMENT=1" % (PREFIX, arm, s, jid),
             "ARGS: %s" % (args if args is not None else D.args_string(arm, s, _E_SAVE)), env, "NVIDIA L4, 23034 MiB"]
    lines += list(offs if offs is not None else D.OFF_LINES)
    lines.append(pt if pt is not None else "%s dir=/x/probe_%s-%s-s%d" % (D.pt_witness_prefix(arm), PREFIX, arm, s))
    for e, v in enumerate(_e_curve(level)[:n_ep]):
        lines.append("Epoch %d, Train Accuracy: %.2f %%, Test Accuracy: %.2f %%" % (e, min(v + 20, 99.9), v))

    open(os.path.join(d, "%s-%s-s%d-%d.out" % (PREFIX, arm, s, jid)), "w").write("\n".join(lines) + "\n")


def _e_records(arm, seed, dominate=False, nb=None, decomp_break=False, beta=None, rng=None):
    """500 synthetic PROBE_TENSOR records.  dominate: the three declared carriers carry the applied sign and outweigh
    the rest.  The target per-tensor terms L_i are drawn, m_i drawn non-zero, and z_i solved from the ARM's OWN meta
    algebra, with mom_pre and z_agg the harness's aggregates -> the decomposition is exact under that meta alg only."""
    rng = rng or random.Random(seed * 7 + len(arm))
    meta = D.META_OF[arm]
    mp, b2 = (0.9, 0.9) if meta == "Adam" else (0.99, 0.9)
    nb = nb if nb is not None else (1 if D.SPEC[arm] == "scalar" else NTENS)
    bv = _E_BETA[arm] if beta is None else beta
    out = []
    for k in range(N_RECORDS):
        L = [rng.gauss(0, 1e-6) for _ in range(NTENS)]
        if dominate:
            for i in CARRIER_IDX0:
                L[i] = 1e-4
        m = [rng.gauss(0, 1e-6) for _ in range(NTENS)]     # NON-ZERO probe EMA, so the two metas' algebra differ
        if meta == "Adam":
            z = [Li - mp / (1 - mp) * mi for Li, mi in zip(L, m)]
            mom = sum(m) / (1 - mp)                         # the harness's unnormalised sum, from the probe's EMA
        else:
            z = [(Li - b2 * mi) / (1 - b2) for Li, mi in zip(L, m)]
            mom = sum(m)
        za = sum(z)
        if decomp_break:
            za = -za - 1e-3
        out.append({"step": k * D.PROBE, "beta": [bv] * nb, "z_tensor": z, "m_tensor": m,
                    "mom_pre": [[mom]], "z_agg": [[za]], "pt_mp": mp, "pt_b2": b2, "beta_pre": [[bv] * nb]})
    return out


def _e_probe_templates(root):
    t = {}
    os.makedirs(root, exist_ok=True)
    for a in ARMS:
        for s in SEEDS:
            for dom in (False, True):
                p = os.path.join(root, "tpl_%s_%d_%d.jsonl" % (a, s, dom))
                with open(p, "w") as fh:
                    for r in _e_records(a, s, dominate=dom):
                        fh.write(json.dumps(r) + "\n")
                t[(a, s, dom)] = p
    return t


def _e_build(d, tpl, levels=None, jit=(0.0, 0.2, -0.2), over=None, drop=None, dom=(), prov_over=None, man_break=False,
             rec_over=None):
    os.makedirs(os.path.join(d, PREFIX), exist_ok=True)
    lv = dict(_E_BASE)
    lv.update(levels or {})
    jid = 7000000
    for s, j in zip(SEEDS, jit):
        for a in ARMS:
            jid += 1
            if drop == (a, s):
                continue
            L = lv[a] if not isinstance(lv[a], (list, tuple)) else lv[a][SEEDS.index(s)]
            kw = dict((over or {}).get((a, s), {}))
            _e_write(d, a, s, jid, L + (0 if isinstance(lv[a], (list, tuple)) else j), **kw)
            pd = os.path.join(d, PREFIX, "probe_%s-%s-s%d" % (PREFIX, a, s))

                    for r in rec_over[(a, s)]:
                        fh.write(json.dumps(r) + "\n")
            else:
                os.symlink(tpl[(a, s, a in dom)], os.path.join(pd, "probe.jsonl"))

    open(os.path.join(d, PREFIX, "PARTITION-MANIFEST.txt"), "w").write(man)
    prov = {"MODE": "submit", "BUILD_NETWORK_SHA256": D.BN_SHA, "HF_SHA256": D.HF_SHA, "RUNNER_SHA256": D.RUNNER_SHA,
            "SCORER_SHA256": file_sha(os.path.abspath(__file__)), "DESIGN_SHA256": file_sha(D.__file__),
            "CWD_DESIGN_SHA256": file_sha(D.W.__file__), "COMMON_SHA256": file_sha(C.__file__)}
    prov.update(prov_over or {})
    open(os.path.join(d, PREFIX, "PROVENANCE.txt"), "w").write("".join("%s %s\n" % kv for kv in prov.items()))

def chk(cond, label, extra=""):
    print("  %-4s %s%s" % ("PASS" if cond else "FAIL", label, ("   " + extra) if extra else ""))

def _find_out(runsdir, name_prefix):
    if not runsdir or not os.path.isdir(runsdir):
        return None
    for fn in sorted(os.listdir(runsdir)):
        if fn.startswith(name_prefix) and fn.endswith(".out"):
            return os.path.join(runsdir, fn)
    return None

def main():

    ap.add_argument("runsdir", nargs="?", default="")
    ap.add_argument("--csv", default=os.path.join(REPO, "results", "all_runs.csv"))

    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--runsdir", dest="runsdir_opt", default="")
    a = ap.parse_args()
    if a.selftest:
        raise SystemExit(selftest(a.csv, a.runsdir_opt or a.runsdir))
    if not a.runsdir:
        ap.error("runsdir required")
    rc, _final = score(a.runsdir, a.csv, a.manifest)
    raise SystemExit(rc)
