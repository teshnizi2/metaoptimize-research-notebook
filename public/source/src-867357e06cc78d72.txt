# =============================================================================
# cY2_cru1_probe.py -- THE SUCCESSOR PROBE READER FOR BATCH `cru1`.
#
# WHY THIS FILE EXISTS, STATED BEFORE ANY NUMBER IT PRODUCES IS READ.
# -----------------------------------------------------------------------------
# `analysis/cY1_cru1_score.py` is the REGISTERED SCORER for `cru1` (committed
# 180755c, 2026-09-08T08:33:38+02:00, 193 s before the batch's earliest sacct
# Submit).  Its `[PROBE]` section reads each record looking for a key named
# `beta_min` or `min_beta`.  **THE LIVE PROBE WRITES NEITHER.**  The actual
# record schema, read off the first `cru1` probe file on disk, is
#
#   step, beta[], z_mean[], z_std[], snr[], frac_neg, frac_zero, z_skew,
#   mom_norm, h_absmax, beta_true_min, beta_true_max, n_beta, n_at_lo,
#   n_at_hi, t_neg[], t_zero[], t_n[], z_sub, z_sub_k, probe_idx[]
#
# so the minimum is `beta_true_min`, not `beta_min`.  `cY1`'s reader therefore
# aggregates nothing and its probe section degrades to its registered SKIP.
#
# STANDING RULE 16 IS ABSOLUTE: `cY1` IS **NOT** EDITED.  Its data now exist, and
# a registered scorer whose data exist is FROZEN even when it is wrong -- it is
# wrong in public and a SUCCESSOR file says so.  Precedent: `cN1`/`cN2`
# (CORRECTIONS 149).  This is that successor.
#
# THIS FILE MAKES NO RULE 21 CLAIM.  It has no runs of its own -- it is a pure
# re-analysis of `.jsonl` files already on disk, at ZERO GPU cost -- so the
# normal wall-clock proof (commit time vs earliest sacct Submit) DOES NOT EXIST
# for it.  It claims ONLY commit-before-first-execution.  Precedent: `cQ1`
# (CORRECTIONS 149), `cS1` (CORRECTIONS 159).
#
# WHAT IT DOES **NOT** DO, AND MAY NEVER BE MADE TO DO.  It produces NO verdict,
# NO branch, and NO bar.  `cru1`'s verdict is whatever `cY1` returns, and
# `cY1`'s registration already states that the probe is "a SKIP, never a FAIL:
# no verdict depends on it."  This file exists ONLY to instrument the confound
# `cY1`'s registration named in advance -- that lowering `ms` tunes the
# meta-optimiser AND mechanically limits how far beta can descend -- by
# measuring the descent directly.  Numbers here are MECHANISM EVIDENCE reported
# alongside the verdict; they can never overturn it.
#
# RUN
#   python3 analysis/cY2_cru1_probe.py                 # defaults, no arguments
#   python3 analysis/cY2_cru1_probe.py --probe-root <dir>       (override only)
#   python3 analysis/cY2_cru1_probe.py --selftest
# The default probe root is /cluster/account_b/metaopt/runs/cru1, the directory the
# launcher's PROBE_DIR points every job at.  There is no argument to forget.
# =============================================================================

from __future__ import annotations

import argparse
import collections
import glob
import json
import math
import os
import re
import statistics
import sys

DEFAULT_ROOTS = ("/cluster/account_b/metaopt/runs/cru1",
                 os.path.expanduser("~/metaopt/runs/cru1"),
                 "runs/cru1")

# The design, copied from cY1 and CHECKED against it by --selftest so the two
# cannot drift.
BETA_LO, BETA_HI = -15.0, -2.3026
EPOCHS, BATCH_SIZE = 100, 100
META_STEPS = EPOCHS * (50000 // BATCH_SIZE)
LADDER = {"1e-3": ("1e-5", "3e-5", "1e-4", "3e-4", "1e-3", "3e-3"),
          "1e-6": ("1e-4", "3e-4", "1e-3", "3e-3")}
RUN_RE = re.compile(r"^probe_cru1-(sc|lay)-m(\S+?)-a(\S+?)-s(\d+)$")


def reach(ms, a0):
    b0 = math.log(float(a0))
    T = META_STEPS * float(ms)
    du = BETA_HI - b0
    r = T / du
    st = ("FROZEN" if r < 0.5 else "STARVED" if r < 1.0
          else "ADEQUATE" if r < 3.0 else "FREE")
    return T, du, r, st


def resolve_root(explicit=None):
    if explicit:
        return explicit, "given explicitly with --probe-root"
    for d in DEFAULT_ROOTS:
        if os.path.isdir(d) and glob.glob(os.path.join(d, "probe_*")):
            return d, "DEFAULT %s" % d
    return None, "no default probe root contains a probe_* directory"


def read_one(path):
    """-> (n_records, last_step, beta_min_over_run, beta_max_over_run,
            frac_records_with_a_coord_at_lo, frac_at_hi, terminal_beta_min,
            first_step_at_lo)"""
    n = 0
    bmin = bmax = None
    lo_hits = hi_hits = 0
    last = None
    first_lo = None
    with open(path, errors="ignore") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            try:
                j = json.loads(line)
            except ValueError:
                continue
            n += 1
            last = j
            bn, bx = j.get("beta_true_min"), j.get("beta_true_max")
            if bn is not None:
                bn = float(bn)
                bmin = bn if bmin is None else min(bmin, bn)
            if bx is not None:
                bx = float(bx)
                bmax = bx if bmax is None else max(bmax, bx)
            if float(j.get("n_at_lo", 0) or 0) >= 1:
                lo_hits += 1
                if first_lo is None:
                    first_lo = j.get("step")
            if float(j.get("n_at_hi", 0) or 0) >= 1:
                hi_hits += 1
    if not n:
        return None
    term = last.get("beta_true_min")
    return (n, last.get("step"), bmin, bmax, lo_hits / float(n),
            hi_hits / float(n), None if term is None else float(term), first_lo)


def report(root):
    print("=" * 78)
    print("cY2_cru1_probe.py -- MECHANISM EVIDENCE ONLY.  NO VERDICT, NO BRANCH,")
    print("NO BAR.  cru1's verdict is whatever analysis/cY1_cru1_score.py returns.")
    print("probe root: %s" % root)
    print("=" * 78)
    files = sorted(glob.glob(os.path.join(root, "probe_*", "probe.jsonl")))
    if not files:
        print("no probe.jsonl under %s -- nothing to report." % root)
        return 0
    rows = {}
    for f in files:
        m = RUN_RE.match(os.path.basename(os.path.dirname(f)))
        if not m:
            continue
        r = read_one(f)
        if r:
            rows[(m.group(1), m.group(2), m.group(3), int(m.group(4)))] = r
    print("\n%d probe files read, %d with a parseable run name and >=1 record."
          % (len(files), len(rows)))
    print("\nPER-RUN (a run still in flight reports its progress so far --")
    print("`last step` below %d means the run has NOT finished)" % META_STEPS)
    print("%-4s %-6s %-6s %-4s %8s %9s %10s %10s %9s %9s"
          % ("arm", "ms", "a0", "seed", "records", "last step", "beta min",
             "beta max", "frac@lo", "frac@hi"))
    for k in sorted(rows):
        n, last, bmin, bmax, flo, fhi, term, first_lo = rows[k]
        print("%-4s %-6s %-6s %-4d %8d %9s %10.4f %10.4f %9.4f %9.4f"
              % (k[0], k[1], k[2], k[3], n, last,
                 bmin if bmin is not None else float("nan"),
                 bmax if bmax is not None else float("nan"), flo, fhi))

    print("\nPER-CELL MEANS, with the registered reachability alongside.")
    print("TRAVEL = %d * ms is the TOTAL distance beta can move, because" % META_STEPS)
    print("|dbeta| = ms EXACTLY (HF.Lion_meta_update with --weight-decay-meta 0).")
    print("%-4s %-6s %-6s %3s %9s %8s %-9s %10s %9s %11s"
          % ("arm", "ms", "a0", "n", "TRAVEL", "T/D_up", "reach", "beta min",
             "frac@lo", "|beta-beta0|"))
    agg = collections.defaultdict(list)
    for k, v in rows.items():
        agg[(k[0], k[1], k[2])].append(v)
    for k in sorted(agg, key=lambda k: (k[0], k[2], float(k[1]))):
        vs = agg[k]
        T, du, r, st = reach(k[1], k[2])
        bmins = [v[2] for v in vs if v[2] is not None]
        flos = [v[4] for v in vs]
        b0 = math.log(float(k[2]))
        span = [max(abs((v[2] or b0) - b0), abs((v[3] or b0) - b0)) for v in vs]
        print("%-4s %-6s %-6s %3d %9.2f %8.3f %-9s %10.4f %9.4f %11.4f"
              % (k[0], k[1], k[2], len(vs), T, r, st,
                 statistics.mean(bmins) if bmins else float("nan"),
                 statistics.mean(flos), statistics.mean(span)))

    print("\nHOW TO READ THIS, REGISTERED IN ADVANCE.")
    print("  * `frac@lo` is the fraction of probe records in which AT LEAST ONE")
    print("    beta coordinate sits on the clamp floor %g.  CORRECTIONS 155" % BETA_LO)
    print("    measured the CIFAR-10 analogue at ms=3e-4: 0.318 for scalar (1/1")
    print("    coordinates) against 0.055 coordinate-occupancy for layerwise.")
    print("  * `|beta-beta0|` is how far beta actually moved.  At a FROZEN rung it")
    print("    is bounded above by TRAVEL by arithmetic, so a value at or near")
    print("    TRAVEL means the ladder, not the loss surface, set the level --")
    print("    which is exactly the confound cY1's registration named in advance.")
    print("  * A cell whose `last step` is below %d is STILL RUNNING and its"
          % META_STEPS)
    print("    numbers are partial.  Nothing here may be quoted until cY1's G1")
    print("    reports 60/60.")
    return 0


def selftest():
    P = F = 0

    def chk(ok, label, detail=""):
        nonlocal P, F
        if ok:
            P += 1
            print("  PASS  %s %s" % (label, detail))
        else:
            F += 1
            print("  FAIL  %s %s" % (label, detail))

    print("cY2_cru1_probe.py --selftest")
    chk(META_STEPS == 50000, "S1 meta-step count", "-> %d" % META_STEPS)
    here = os.path.dirname(os.path.abspath(__file__))
    cy1 = os.path.join(here, "cY1_cru1_score.py")
    if os.path.isfile(cy1):
        import importlib.util
        spec = importlib.util.spec_from_file_location("cy1", cy1)
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        chk(dict(m.LADDER) == LADDER and m.META_STEPS == META_STEPS
            and (m.BETA_LO, m.BETA_HI) == (BETA_LO, BETA_HI),
            "S2 this file's design constants match the REGISTERED scorer's",
            "(ladders, meta-steps, clamp)")
        chk("beta_min" in open(cy1).read(),
            "S3 the defect this successor exists for is real: cY1's probe reader",
            "looks for `beta_min`, a key the live probe does not write")
    else:
        chk(False, "S2 cY1_cru1_score.py not found next to this file", cy1)
    fr = [(ms, a0) for a0 in LADDER for ms in LADDER[a0]
          if reach(ms, a0)[3] == "FROZEN"]
    chk(set(fr) == {("1e-5", "1e-3"), ("3e-5", "1e-3"), ("1e-4", "1e-6")},
        "S4 the FROZEN rungs reproduce", "-> %s" % fr)
    print("SELFTEST: %d PASS, %d FAIL" % (P, F))
    return F


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--probe-root", default=None,
                    help="OPTIONAL -- defaulted to the launcher's PROBE_DIR root")
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    if a.selftest:
        return 1 if selftest() else 0
    root, how = resolve_root(a.probe_root)
    if not root:
        print("no probe root found (%s)" % how)
        return 1
    return report(root)


if __name__ == "__main__":
    sys.exit(main())
